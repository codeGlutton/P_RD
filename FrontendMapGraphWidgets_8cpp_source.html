46void UFrontendMapLineWidget::SetLineStyle(bool bIsOpenPath)
47{
48 // 틴트도 시안(WBP 클래스 디폴트)이 정본 — C++ 하드코딩 금지.
49 const FLinearColor Tint = bIsOpenPath ? mOpenTint : mLockedTint;
50 UTexture2D* PathTexture = bIsOpenPath ? mSolidTexture.Get() : mDashedTexture.Get();
51 if (LineImage != nullptr && PathTexture != nullptr)
52 {
53 // 경로 텍스처를 길이 방향으로 타일링해 이어 붙인다(늘리면 문양이 뭉개진다).
54 FSlateBrush LineBrush;
55 LineBrush.DrawAs = ESlateBrushDrawType::Image;
56 LineBrush.Tiling = ESlateBrushTileType::Horizontal;
57 LineBrush.ImageSize = FVector2D(PathTexture->GetSizeX(), PathTexture->GetSizeY());
58 LineBrush.SetResourceObject(PathTexture);
59 LineImage->SetBrush(LineBrush);
60 LineImage->SetColorAndOpacity(Tint);
61 LineImage->SetVisibility(ESlateVisibility::HitTestInvisible);
62 if (LinePanel != nullptr)
63 {
64 LinePanel->SetBrushColor(FLinearColor(0.f, 0.f, 0.f, 0.f));
65 }
66 return;
67 }
68
69 // 텍스처 미주입(빌더 미실행) 시 기존 단색 Border 폴백.
70 if (LineImage != nullptr)
71 {
72 LineImage->SetVisibility(ESlateVisibility::Collapsed);
73 }
74 SetLineColor(Tint);
75}

101UTexture2D* UFrontendMapNodeWidget::GetTypeIconTexture(ERoomType RoomType) const
102{
103 // 시안 빌더가 주입한 클래스 디폴트가 정본, 하드코딩 경로는 빌더 미실행 시 폴백(생성자에서 로드한 텍스처로 대체).
104 UTexture2D* Icon = nullptr;
105 switch (RoomType)
106 {
107 case ERoomType::Treasure: Icon = mIconTreasureTexture ? mIconTreasureTexture.Get() : mFallbackIconTreasure.Get(); break;
108 case ERoomType::Shop: Icon = mIconShopTexture ? mIconShopTexture.Get() : mFallbackIconShop.Get(); break;
109 case ERoomType::Monster: Icon = mIconMonsterTexture ? mIconMonsterTexture.Get() : mFallbackIconMonster.Get(); break;
110 case ERoomType::EliteMonster: Icon = mIconEliteTexture ? mIconEliteTexture.Get() : mFallbackIconElite.Get(); break;
111 case ERoomType::BossMonster: Icon = mIconBossTexture ? mIconBossTexture.Get() : mFallbackIconBoss.Get(); break;
112 default: break;
113 }
114 return Icon;
115}

148 // 아이콘 위주 노드: 타입 아이콘이 주인공. 상태는 링 텍스처(시안)가 1순위, 없으면 상태색 프레임 폴백.
149 bool bHasIcon = false;
150 if (NodePanel != nullptr)
151 {
152 UTexture2D* Icon = GetTypeIconTexture(RoomType);
153 if (Icon != nullptr)
154 {
155 NodePanel->SetBrushFromTexture(Icon);
156 // 잠김은 자물쇠 링이 이미 표현하므로 아이콘은 살짝만 누른다. 값은 시안(nodeStyle.lockedIconTint)이 정본.
157 NodePanel->SetBrushColor(RoomState == EMapRoomState::Locked && !bIsCurrentRoom

165 }
166 }
167
168 UTexture2D* RingTexture = GetStateRingTexture(RoomState, bIsCurrentRoom);
169 const bool bHasRing = NodeRingImage != nullptr && RingTexture != nullptr;
170 if (NodeRingImage != nullptr)
171 {
172 if (bHasRing)
173 {
174 NodeRingImage->SetBrushFromTexture(RingTexture, false);
175 NodeRingImage->SetColorAndOpacity(FLinearColor::White);
176 NodeRingImage->SetVisibility(ESlateVisibility::HitTestInvisible);

181 }
182 }
183
184 if (NodeTypeStripe != nullptr)
185 {
186 // 링이 상태를 말하면 색 프레임은 완전히 숨긴다. 링이 없을 때만 기존 상태색 프레임 폴백.
