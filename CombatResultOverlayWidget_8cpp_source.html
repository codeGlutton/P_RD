143void UCombatResultOverlayWidget::RefreshSlate()
144{
145 if (mRewardPanel.IsValid())
146 {
147 mRewardPanel->SetVisibility(mMode == ECombatResultOverlayMode::VictoryReward ? EVisibility::Visible : EVisibility::Collapsed);

149 if (mContinuePanel.IsValid())

153 if (mTitleText.IsValid())
