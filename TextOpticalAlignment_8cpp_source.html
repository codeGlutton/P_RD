13 TOptional<float> MeasureOffsetY(const UTextBlock& TextBlock)
14 {
15 if (!FSlateApplication::IsInitialized())

18 }
19
20 FSlateRenderer* Renderer = FSlateApplication::Get().GetRenderer();
21 if (Renderer == nullptr)
22 {
23 return {};
24 }

26 FString DisplayString = TextBlock.GetText().ToString();
27 switch (TextBlock.GetTextTransformPolicy())
28 {
29 case ETextTransformPolicy::ToLower:
30 DisplayString = FTextTransformer::ToLower(DisplayString);
31 break;
32 case ETextTransformPolicy::ToUpper:
33 DisplayString = FTextTransformer::ToUpper(DisplayString);
34 break;
35 default:
36 break;
37 }
38
39 if (DisplayString.IsEmpty())

41 return 0.0f;
42 }
43
44 const FSlateFontInfo& FontInfo = TextBlock.GetFont();
45 const TSharedRef<FSlateFontCache> FontCache = Renderer->GetFontCache();
46 const TextBiDi::ETextDirection BaseDirection =
47 TextBiDi::ComputeBaseDirection(DisplayString);
48 const FShapedGlyphSequenceRef GlyphSequence =
49 FontCache->ShapeBidirectionalText(DisplayString, FontInfo, 1.0f,
50 BaseDirection, GetDefaultTextShapingMethod());
51
52 const float MaxTextHeight = static_cast<float>(GlyphSequence->GetMaxTextHeight());
53 const float TextBaseline = static_cast<float>(GlyphSequence->GetTextBaseline());
54 float InkTop = TNumericLimits<float>::Max();
55 float InkBottom = -TNumericLimits<float>::Max();
56 bool bFoundVisibleGlyph = false;
57
58 for (const FShapedGlyphEntry& Glyph : GlyphSequence->GetGlyphsToRender())
59 {
60 if (!Glyph.bIsVisible)
61 {
62 continue;
63 }
64
65 // 외곽선은 중심을 바꾸지 않고 위아래로 같은 양만 늘어나므로 본체만 잰다.
66 const FShapedGlyphFontAtlasData AtlasData =
67 FontCache->GetShapedGlyphFontAtlasData(Glyph, FFontOutlineSettings());
68 if (!AtlasData.Valid)
69 {
70 continue;
71 }
72
73 const float BitmapRenderScale = Glyph.GetBitmapRenderScale();
74 if (FMath::IsNearlyZero(BitmapRenderScale))
75 {
76 continue;
77 }
78
79 const float GlyphTop = -static_cast<float>(AtlasData.VerticalOffset)
80 + static_cast<float>(Glyph.YOffset)
81 + (MaxTextHeight + TextBaseline) / BitmapRenderScale;
82 const float GlyphBottom = GlyphTop
83 + static_cast<float>(AtlasData.VSize) * BitmapRenderScale;
84 InkTop = FMath::Min(InkTop, GlyphTop);
85 InkBottom = FMath::Max(InkBottom, GlyphBottom);
86 bFoundVisibleGlyph = true;
87 }
88
89 if (!bFoundVisibleGlyph)

91 return {};
92 }

95 const float LineBoxCenter = static_cast<float>(FontInfo.OutlineSettings.OutlineSize)
96 + MaxTextHeight * 0.5f;
97 const float InkCenter = (InkTop + InkBottom) * 0.5f;
98 return LineBoxCenter - InkCenter;
