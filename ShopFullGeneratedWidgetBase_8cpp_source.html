44void UShopFullGeneratedWidgetBase::UpdateResponsiveZoneLayout(
45 const FVector2D& ViewportSize)
46{
47 constexpr float DesignWidth = 1600.f;
48 constexpr float DesignHeight = 1000.f;
49
50 if (ViewportSize.X <= 0.f || ViewportSize.Y <= 0.f)
51 {
52 return;
53 }
54
55 // ShopMasterScale uses ScaleToFit. On tall screens this leaves equal vertical
56 // gutters around the 1600x1000 design board. Counter-translate only the edge
57 // zones in design-local units: header reaches the viewport top, actions reach
58 // the viewport bottom, while CenterZone remains on the viewport center line.
59 const float MasterScale = FMath::Min(
60 ViewportSize.X / DesignWidth, ViewportSize.Y / DesignHeight);
61 if (MasterScale <= UE_SMALL_NUMBER)
62 {
63 return;
64 }
65
66 const float ScaledDesignHeight = DesignHeight * MasterScale;
67 const float VerticalGutter = FMath::Max(0.f,
68 (ViewportSize.Y - ScaledDesignHeight) * .5f);
69 const float LocalEdgeOffset = VerticalGutter / MasterScale;
70
71 if (mResponsiveTopZone != nullptr)
72 {
73 mResponsiveTopZone->SetRenderTranslation(FVector2D(0.f, -LocalEdgeOffset));
74 }
75 if (mResponsiveCenterZone != nullptr)
76 {
77 mResponsiveCenterZone->SetRenderTranslation(FVector2D::ZeroVector);
78 }
79 if (mResponsiveBottomZone != nullptr)
80 {
81 mResponsiveBottomZone->SetRenderTranslation(FVector2D(0.f, LocalEdgeOffset));
82 }
83}
