48void USettingsPanelWidget::RefreshValueModelFromCurrentOptions()
49{
50 FSettingsPanelValueModel CurrentValueModel = mValueModel;
51 if (const UGameInstance* GameInstance = GetGameInstance())
52 {
53 if (const UPersistentDataSubsystem* PersistentDataSubsystem = GameInstance->GetSubsystem<UPersistentDataSubsystem>())
54 {
55 if (const UOptionPersistData* OptionData = PersistentDataSubsystem->GetOptionPersistData())
56 {
57 CurrentValueModel.mMasterVolume = OptionData->GetVolume(EGameVolumeType::Master);
58 CurrentValueModel.mBgmVolume = OptionData->GetVolume(EGameVolumeType::BGM);
59 CurrentValueModel.mSfxVolume = OptionData->GetVolume(EGameVolumeType::SFX);
60 CurrentValueModel.mUiVolume = OptionData->GetVolume(EGameVolumeType::UI);
61 CurrentValueModel.mUseKoreanLanguage = OptionData->GetLanguage() == ELanguageType::KOREAN;
62 CurrentValueModel.mFpsLimit = OptionData->GetFpsLimit();
63 }
64 }
65 }
66 ApplyValueModel(CurrentValueModel);
67}

81void USettingsPanelWidget::HandleResetButtonClicked()
82{
83 OnResetRequested.Broadcast();
84 // 전용 수신자가 생기기 전까지의 기본 적용: 옵션을 CDO 기본값으로 되돌리고
85 // (ResetOptions -> ClearOption -> ApplyCurrentOptions로 볼륨/언어/해상도 즉시 재적용),
86 // 패널 UI도 기본 값 모델로 갱신한다(ApplyValueModel은 콜백 가드로 이벤트 재발신 없음).
87 // 초기화도 커밋 행위다 — 되돌린 옵션을 즉시 디스크에 저장한다.
88 if (ARDGameModeBase* GameModeBase = GetWorld()->GetAuthGameMode<ARDGameModeBase>())
89 {
90 GameModeBase->ResetFromOptionPanel();
91 }
92 RefreshValueModelFromCurrentOptions();
93}
