11void USettingsPanelWidget::ApplyValueModel(const FSettingsPanelValueModel& ValueModel)
12{
13 mValueModel = RDSettingsPanel::SanitizeValueModel(ValueModel);
14
15 mIsApplyingValueModel = true;
16 if (BgmVolumeSlider != nullptr)
17 {
18 BgmVolumeSlider->SetValue(mValueModel.mBgmVolume);
19 }
20 if (SfxVolumeSlider != nullptr)
21 {
22 SfxVolumeSlider->SetValue(mValueModel.mSfxVolume);
23 }
24 if (UiVolumeSlider != nullptr)
25 {
26 UiVolumeSlider->SetValue(mValueModel.mUiVolume);
27 }
28 if (ScreenShakeCheckBox != nullptr)
29 {
30 ScreenShakeCheckBox->SetIsChecked(mValueModel.mScreenShakeEnabled);
31 }
32 if (VibrationCheckBox != nullptr)
33 {
34 VibrationCheckBox->SetIsChecked(mValueModel.mVibrationEnabled);
35 }
36 if (MasterVolumeSlider != nullptr)
37 {
38 MasterVolumeSlider->SetValue(mValueModel.mMasterVolume);
39 }
40 if (EffectsCheckBox != nullptr)

42 EffectsCheckBox->SetIsChecked(mValueModel.mEffectsEnabled);
43 }
44 mIsApplyingValueModel = false;
45 SyncText();
46 UpdateGraphicsSelectionIndicators();
47}

49void USettingsPanelWidget::RefreshValueModelFromCurrentOptions()
50{
51 FSettingsPanelValueModel CurrentValueModel = mValueModel;
52 if (ARDGameModeBase* GameModeBase = GetWorld()->GetAuthGameMode<ARDGameModeBase>())
53 {
54 if (const UOptionPersistData* OptionData = GameModeBase->GetOptionPersistData())
55 {
56 CurrentValueModel.mMasterVolume = OptionData->GetVolume(EGameVolumeType::Master);
57 CurrentValueModel.mBgmVolume = OptionData->GetVolume(EGameVolumeType::BGM);
58 CurrentValueModel.mSfxVolume = OptionData->GetVolume(EGameVolumeType::SFX);
59 CurrentValueModel.mUiVolume = OptionData->GetVolume(EGameVolumeType::UI);
60 CurrentValueModel.mUseKoreanLanguage = OptionData->GetLanguage() == ELanguageType::KOREAN;
61 CurrentValueModel.mFpsLimit = OptionData->GetFpsLimit();
62 CurrentValueModel.mQualityLevel = RDSettingsPanel::FromOverallQuality(StaticCast<int32>(OptionData->GetOverallQuality()));
63 CurrentValueModel.mScreenShakeEnabled = OptionData->IsCameraShakeEnabled();
64 CurrentValueModel.mEffectsEnabled = OptionData->IsEffectVFXEnabled();
65 CurrentValueModel.mVibrationEnabled = OptionData->IsVibrationEnabled();
66 }
67 }
68 ApplyValueModel(CurrentValueModel);
69}

83void USettingsPanelWidget::HandleResetButtonClicked()
84{
85 OnResetRequested.Broadcast();
86 // 전용 수신자가 생기기 전까지의 기본 적용: 옵션을 CDO 기본값으로 되돌리고
87 // (ResetOptions -> ClearOption -> ApplyCurrentOptions로 볼륨/언어/해상도 즉시 재적용),
88 // 패널 UI도 기본 값 모델로 갱신한다(ApplyValueModel은 콜백 가드로 이벤트 재발신 없음).
89 // 초기화도 커밋 행위다 — 되돌린 옵션을 즉시 디스크에 저장한다.
90 if (ARDGameModeBase* GameModeBase = GetWorld()->GetAuthGameMode<ARDGameModeBase>())
91 {
92 GameModeBase->ResetFromOptionPanel();
93 }
94 RefreshValueModelFromCurrentOptions();
95}

104void USettingsPanelWidget::HandleLowQualityButtonClicked()
105{

112 if (ARDGameModeBase* GameModeBase = GetWorld()->GetAuthGameMode<ARDGameModeBase>())
113 {
114 GameModeBase->SetOverallQuality(StaticCast<EOverallQualityType>(RDSettingsPanel::ToQualityRequestValue(mValueModel.mQualityLevel)));
115 }
