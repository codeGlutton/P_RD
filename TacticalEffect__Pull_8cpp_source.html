13void UTacticalEffectExecutionCalculation_Pull::Execute(const FTacticalEffectCustomExecutionParameters& ExecutionParams, FTacticalEffectCustomExecutionOutput& OutExecutionOutput) const
14{
15 Super::Execute(ExecutionParams, OutExecutionOutput);
16
17 UAttributeSetComponentModel* SourceAttributeSetCompModel = ExecutionParams.GetSourceAttributeSetComponentModel();
18 checkf(SourceAttributeSetCompModel != nullptr, TEXT("소스 컴포넌트 모델 nullptr"));
19
20 UAttributeSetComponentModel* TargetAttributeSetCompModel = ExecutionParams.GetTargetAttributeSetComponentModel();
21 checkf(TargetAttributeSetCompModel != nullptr, TEXT("타겟 컴포넌트 모델 nullptr"));
22
23 UBoardActorModel* SourceModel = SourceAttributeSetCompModel->GetOwnerModel<UBoardActorModel>();
24 checkf(SourceModel != nullptr, TEXT("당기기 시전자가 보드 액터가 아님"));
25
26 UBoardActorModel* TargetModel = TargetAttributeSetCompModel->GetOwnerModel<UBoardActorModel>();
27 checkf(TargetModel != nullptr, TEXT("당기기 대상자가 보드 액터가 아님"));
28
29 IBoardCombatTarget* TargetCombatTarget = Cast<IBoardCombatTarget>(TargetModel);
30 checkf(TargetCombatTarget != nullptr, TEXT("당기기 대상자가 전투 대상이 아님"));
31
32 UBoardMovementComponentModel* TargetMoveCompModel = TargetCombatTarget->GetBoardMovementComponentModel();
33 checkf(TargetMoveCompModel != nullptr, TEXT("타겟 움직임 컴포넌트 모델 nullptr"));
34
35 // 타일맵은 대상의 이동 컴포넌트에서 획득 (같은 보드 위이므로 동일한 맵)
36 UTileMapModel* TileMap = TargetMoveCompModel->GetTileMap();
37 checkf(TileMap != nullptr, TEXT("타일 맵 nullptr"));
38
39 // 당겨지는 경로 계산 (거리 소진, 시전자에 붙음, 중간 막힘 중 먼저 오는 곳까지로 짧아짐)
40 const TArray<FTileIndex> PullPath = TileMap->GetPullPath(SourceModel->GetTileTransform().mIndex, TargetModel->GetTileTransform().mIndex, ExecutionParams.GetOwningSpec().GetStackCount());
41
42 // 한 칸도 당기지 못하거나 대상이 이동 중이면 아무것도 안 함 (이동 중 당기기는 지원하지 않음)
43 const int32 PathNum = PullPath.Num();
44 if (PathNum >= 2 && TargetMoveCompModel->IsMoving() == false)
45 {
46 // 정지 상태 대상: 즉시 당기기 시작
47 TargetMoveCompModel->PullAlongPath(PullPath);
48
49 /* 로그 작성 */
50
51 for (int32 PathIndex = 1; PathIndex < PathNum; ++PathIndex)
52 {
53 FSRPGTileEffectEventLog Log;
54 Log.mOccupancyState = ESRPGTileOccupancyState::Move;
55 Log.mPreTileIndex = PullPath[PathIndex - 1];
56 Log.mNextTileIndex = PullPath[PathIndex];
57
58 GetWorldEventLogger(TargetModel)->LogTileEffect(TargetModel->GetModelId(), TargetModel->GetClass(), Log);
59 }
60 }
61
62 OutExecutionOutput.MarkDynamicMagnitudeHandledManually();
63 OutExecutionOutput.MarkStackCountHandledManually();
64}
