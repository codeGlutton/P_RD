100 /*

106 {
107 return Option.mJobType == JobType;
108 });

112 void AppendLockedCharacterOption(TArray<FFrontendCharacterOption>& Options, EPlayerJobType JobType)
113 {
114 /*
115 * 아직 DataAsset이 준비되지 않은 캐릭터도 화면 슬롯은 유지한다.
116 * WBP 레이아웃 확인과 기획상 "추후 열릴 캐릭터" 표현을 위해 더미가 아니라 비활성 View 데이터로 추가한다.
117 * bSelectable=false와 DisabledReason을 같이 내려, CharacterSelectWidget이 Confirm을 막고 이유 문구를 표시할 수 있게 한다.
118 */
119 FFrontendCharacterOption NewOption;
120 NewOption.mIndex = Options.Num();
121 NewOption.mDisplayName = GetPlayerJobName(JobType);
122 NewOption.mRoleText = GetPlayerJobRole(JobType);
123 // 잠긴 카드도 직업별 일러스트는 보여줄 수 있게 화면 아트 키를 채운다.
124 NewOption.mJobType = JobType;
125 // "데이터 없음" 문구 대신 직업 설명을 표시(기획: not ready 삭제). 잠금 사유는 내부적으로만 유지.
126 NewOption.mDescription = GetPlayerJobDescription(JobType);
127 NewOption.mDisabledReason = NSLOCTEXT("FrontendGameMode", "LockedCharacterReason", "준비 중인 캐릭터입니다");
128 NewOption.mSelectable = false;
129 Options.Add(MoveTemp(NewOption));
130 }

147 void SortCharacterOptionsForClassSelect(TArray<FFrontendCharacterOption>& Options)
148 {
149 Options.StableSort([](const FFrontendCharacterOption& Left, const FFrontendCharacterOption& Right)
150 {
151 const int32 LeftOrder = GetClassSelectSortOrder(Left.mJobType);
152 const int32 RightOrder = GetClassSelectSortOrder(Right.mJobType);
153 if (LeftOrder != RightOrder)
154 {
155 return LeftOrder < RightOrder;
156 }
157
158 return Left.mIndex < Right.mIndex;
159 });
160
161 for (int32 OptionIndex = 0; OptionIndex < Options.Num(); ++OptionIndex)
162 {
163 Options[OptionIndex].mIndex = OptionIndex;
164 }
165 }

176// InGameSettings를 프론트엔드에도 준비하는 이유:

273bool AFrontendGameMode::StartNewRun(const FPrimaryAssetId& PlayerUnitId, int32 Difficulty)
274{
275 /*
276 * 캐릭터 선택 화면에서 Confirm이 눌린 뒤의 실제 시작 지점이다.
277 * 먼저 선택된 캐릭터로 RunPersistData를 만들고, 그 다음 Stage1 첫 방을 프리로드/전환한다.
278 * UI는 "시작 요청"만 하고, 저장 데이터 생성과 방 전환 순서는 GameMode가 고정한다.

283 return false;
284 }
285
286 checkf(CreateRunData(PlayerUnitId, Difficulty) == true, TEXT("런 데이터 생성 오류"));
287 checkf(PreloadAndTransitionRoomAsync(EStageLevelType::Stage1) == true, TEXT("스테이지 1 처음 방으로 전환 실패"));
288 return true;
289}

292// Continue는 세이브 파일을 여기서 새로 불러오는 기능이 아니라,
293// 이미 활성화되어 있는 RunPersistData의 현재 row/column으로 방 전환을 요청하는 기능이다.
294// 타이틀에서는 월드맵이나 다음 방 선택을 열지 않고, 방 입장 이후의 UI/선택/저장 흐름은 RoomGameModeBase가 맡는다.
295bool AFrontendGameMode::ContinueRunFromTitle()
296{
297 /*
298 * Continue는 "세이브 파일을 불러온다"가 아니라 "이미 활성화된 Run의 현재 방으로 돌아간다"는 의미다.
299 * 세이브 슬롯 선택/로드는 프론트엔드에 진입하기 전 또는 별도 UI 흐름에서 끝나 있어야 하고,
300 * 이 함수는 RunPersistData가 이미 준비되어 있다는 전제에서 방 전환만 시작한다.
301 *
302 * 타이틀에서는 더 이상 지도 미리보기나 다음 방 선택을 열지 않는다.
303 * 타이틀은 시작/이어하기 같은 프론트엔드 선택만 담당하고,
304 * 현재 방 UI, 월드맵 표시, 다음 방 선택, 저장 후 전환은 실제 방에 들어간 뒤 RoomGameMode가 맡는다.
305 */
306 if (mWasNextRoomPreloadRequested == true)
307 {
308 UE_LOG(LogFrontendGameMode, Log, TEXT("방 전환 시 추가 로직 요청 불가"));
309 return false;
310 }
311
312 /*
313 * Continue 버튼은 TitleMenuWidget에서 활성 Run이 있을 때만 보이지만,
314 * UI 상태가 갱신되기 전 클릭되거나 외부 흐름에서 직접 호출될 수 있으므로 GameMode에서도 다시 확인한다.
315 */
316 const URunPersistData* RunPersistData = GetRunPersistData();
317 if (RunPersistData == nullptr || RunPersistData->IsActive() == false)
318 {
319 UE_LOG(LogFrontendGameMode, Log, TEXT("이어갈 런 데이터가 없음"));
320 return false;
321 }
322
323 int32 CurrentRowIndex = INDEX_NONE;
324 int32 CurrentColumnIndex = INDEX_NONE;
325 RunPersistData->GetCurrentRoomIndex(OUT CurrentRowIndex, OUT CurrentColumnIndex);
326
327 /*
328 * RunPersistData에는 "현재 방 좌표"만 저장되어 있으므로,
329 * 전환 요청 전에 해당 좌표가 현재 Stage에 실제로 존재하는 방인지 확인한다.
330 * 좌표가 깨진 상태로 PreloadAndTransitionRoomAsync()에 넘기면 잘못된 방 전환 요청이 되기 때문이다.
331 */
332 const FStage& Stage = RunPersistData->GetStage();
333 if (Stage.HasRoom(CurrentRowIndex, CurrentColumnIndex) == false)
334 {
335 UE_LOG(LogFrontendGameMode, Log, TEXT("현재 런의 방 위치가 Stage에 없음"));
336 return false;
337 }
338
339 /*
340 * 여기서부터는 기존 방 전환 공통 흐름을 사용한다.
341 * 플레이어/스테이지/방 에셋 프리로드, 페이드/로딩 UI, 실제 레벨 전환 순서는 RDGameModeBase와 RoomTransitionSubsystem이 처리한다.
342 */
343 checkf(PreloadAndTransitionRoomAsync(CurrentRowIndex, CurrentColumnIndex) == true, TEXT("이어하기 방 전환 실패"));
344 return true;
345}

348// 프론트엔드는 아직 실제 전투 방 안이 아니므로 전투 정리나 방 퇴장 처리를 하지 않는다.
349// RunPersistData만 비우고, 이후 TitleMenuWidget이 메뉴 상태를 다시 읽어 Continue 버튼 표시 여부를 갱신한다.
350bool AFrontendGameMode::AbandonRunFromTitle()
351{
352 /*
353 * 타이틀 설정 화면에서 기존 런을 포기할 때 호출된다.
354 * 여기서는 전투 방이 아니므로 방 전환을 일으키지 않고 RunPersistData만 비운다.
