65 bool HasCharacterOptionForJob(const TArray<FFrontendCharacterOption>& Options, EPlayerJobType JobType)
66 {
67 /*
68 * 실제 PlayerUnit DataAsset에서 특정 직업 카드가 내려왔는지 확인한다.
69 * 현재 Archer/Mage는 아직 실제 데이터가 준비되지 않은 경우가 있어, 없는 직업만 잠김 카드로 보강한다.
70 * 이렇게 해야 "데이터가 없어서 빈 칸"과 "잠긴 캐릭터로 보여주려는 의도"가 UI에서 구분된다.
71 */
72 const FText JobText = GetPlayerJobName(JobType);
73 return Options.ContainsByPredicate([&JobText](const FFrontendCharacterOption& Option)
74 {
75 return Option.mRoleText.EqualTo(JobText);
76 });
77 }

79 void AppendLockedCharacterOption(TArray<FFrontendCharacterOption>& Options, EPlayerJobType JobType)
80 {
81 /*
82 * 아직 DataAsset이 준비되지 않은 캐릭터도 화면 슬롯은 유지한다.
83 * WBP 레이아웃 확인과 기획상 "추후 열릴 캐릭터" 표현을 위해 더미가 아니라 비활성 View 데이터로 추가한다.
84 * bSelectable=false와 DisabledReason을 같이 내려, CharacterSelectWidget이 Confirm을 막고 이유 문구를 표시할 수 있게 한다.
85 */
86 FFrontendCharacterOption NewOption;
87 NewOption.mIndex = Options.Num();
88 NewOption.mDisplayName = GetPlayerJobName(JobType);
89 NewOption.mRoleText = GetPlayerJobName(JobType);
90 NewOption.mDescription = NSLOCTEXT("FrontendGameMode", "LockedCharacterDescription", "Character data is not ready");
91 NewOption.mDisabledReason = NewOption.mDescription;
92 NewOption.mSelectable = false;
93 Options.Add(MoveTemp(NewOption));
94 }

210 /*

234bool AFrontendGameMode::ContinueRunFromTitle()
235{
236 /*
237 * Continue는 "세이브 파일을 불러온다"가 아니라 "이미 활성화된 Run의 현재 방으로 돌아간다"는 의미다.
238 * 세이브 슬롯 선택/로드는 프론트엔드에 진입하기 전 또는 별도 UI 흐름에서 끝나 있어야 하고,
239 * 이 함수는 RunPersistData가 이미 준비되어 있다는 전제에서 방 전환만 시작한다.
240 *
241 * 타이틀에서는 더 이상 지도 미리보기나 다음 방 선택을 열지 않는다.
242 * 타이틀은 시작/이어하기 같은 프론트엔드 선택만 담당하고,
243 * 현재 방 UI, 월드맵 표시, 다음 방 선택, 저장 후 전환은 실제 방에 들어간 뒤 RoomGameMode가 맡는다.
244 */
245 if (mWasNextRoomPreloadRequested == true)
246 {
247 UE_LOG(LogFrontendGameMode, Log, TEXT("방 전환 시 추가 로직 요청 불가"));
248 return false;
249 }
250
251 /*
252 * Continue 버튼은 TitleMenuWidget에서 활성 Run이 있을 때만 보이지만,
253 * UI 상태가 갱신되기 전 클릭되거나 외부 흐름에서 직접 호출될 수 있으므로 GameMode에서도 다시 확인한다.
254 */
255 const URunPersistData* RunPersistData = GetRunPersistData();
256 if (RunPersistData == nullptr || RunPersistData->IsActive() == false)
257 {
258 UE_LOG(LogFrontendGameMode, Log, TEXT("이어갈 런 데이터가 없음"));
259 return false;
260 }
261
262 int32 CurrentRowIndex = INDEX_NONE;
263 int32 CurrentColumnIndex = INDEX_NONE;
264 RunPersistData->GetCurrentRoomIndex(OUT CurrentRowIndex, OUT CurrentColumnIndex);
265
266 /*
267 * RunPersistData에는 "현재 방 좌표"만 저장되어 있으므로,
268 * 전환 요청 전에 해당 좌표가 현재 Stage에 실제로 존재하는 방인지 확인한다.
269 * 좌표가 깨진 상태로 PreloadAndTransitionRoomAsync()에 넘기면 잘못된 방 전환 요청이 되기 때문이다.
270 */
271 const FStage& Stage = RunPersistData->GetStage();
272 if (Stage.HasRoom(CurrentRowIndex, CurrentColumnIndex) == false)
273 {
274 UE_LOG(LogFrontendGameMode, Log, TEXT("현재 런의 방 위치가 Stage에 없음"));
275 return false;
276 }
277
278 /*
279 * 여기서부터는 기존 방 전환 공통 흐름을 사용한다.
280 * 플레이어/스테이지/방 에셋 프리로드, 페이드/로딩 UI, 실제 레벨 전환 순서는 RDGameModeBase와 RoomTransitionSubsystem이 처리한다.
281 */
282 checkf(PreloadAndTransitionRoomAsync(CurrentRowIndex, CurrentColumnIndex) == true, TEXT("이어하기 방 전환 실패"));
283 return true;
284}

324bool AFrontendGameMode::GetCharacterOptions(TArray<FFrontendCharacterOption>& OutOptions) const
325{
326 /*
327 * CharacterSelectWidget은 PlayerUnit DataAsset 구조를 직접 알지 않는다.
328 * 이 함수가 DataAsset을 UI 전용 DTO로 바꿔주면, WBP 쪽은 이름/직업/스탯/초상화/선택 가능 여부만 보고 카드를 그릴 수 있다.
329 */
330 OutOptions.Reset();
331
332 const TArray<TSoftObjectPtr<UStaticPlayerUnitSpawnData>>& PlayerUnitDatas = GetPlayerUnitDatas();
333 for (const TSoftObjectPtr<UStaticPlayerUnitSpawnData>& PlayerUnitData : PlayerUnitDatas)
334 {
335 const UStaticPlayerUnitSpawnData* LoadedPlayerUnitData = PlayerUnitData.Get();
336 if (LoadedPlayerUnitData == nullptr)
337 {
338 UE_LOG(LogRD, Warning, TEXT("선택 가능한 Player Unit 로드 실패"));
339 continue;
340 }
341
342 FFrontendCharacterOption NewOption;
343 NewOption.mIndex = OutOptions.Num();
344 NewOption.mDisplayName = LoadedPlayerUnitData->mDisplayName.IsEmpty()
345 ? FText::FromName(LoadedPlayerUnitData->GetPrimaryAssetId().PrimaryAssetName)
346 : LoadedPlayerUnitData->mDisplayName;
347 NewOption.mRoleText = GetPlayerJobName(LoadedPlayerUnitData->mJobType);
348 NewOption.mDescription = LoadedPlayerUnitData->mDescription;
349 NewOption.mMaxHP = FMath::RoundToInt(LoadedPlayerUnitData->GetDefaultMaxHP(DefaultDifficulty));
350 NewOption.mDice = LoadedPlayerUnitData->mDiceDatas.Num();
351 NewOption.mGold = FMath::RoundToInt(LoadedPlayerUnitData->GetDefaultMoney(DefaultDifficulty));
352 NewOption.mStatSummary = FText::Format(
353 NSLOCTEXT("FrontendGameMode", "CharacterStatSummary", "HP {0} / Dice {1} / Gold {2}"),
354 FText::AsNumber(NewOption.mMaxHP),
355 FText::AsNumber(NewOption.mDice),
356 FText::AsNumber(NewOption.mGold));
357 NewOption.mPortrait = LoadedPlayerUnitData->mPortrait;
358 NewOption.mIcon = LoadedPlayerUnitData->mIcon;
359 NewOption.mPlayerUnitId = LoadedPlayerUnitData->GetPrimaryAssetId();
360 NewOption.mSelectable = PlayerUnitData.IsValid() && !LoadedPlayerUnitData->mClass.IsNull();
361 OutOptions.Add(MoveTemp(NewOption));
362 }
363
364 // TODO : 아쳐와 메이지가 추가된 뒤에 삭제할 부분
365
366 if (!HasCharacterOptionForJob(OutOptions, EPlayerJobType::Archer))
367 {
368 AppendLockedCharacterOption(OutOptions, EPlayerJobType::Archer);
369 }
370 if (!HasCharacterOptionForJob(OutOptions, EPlayerJobType::Mage))
371 {
372 AppendLockedCharacterOption(OutOptions, EPlayerJobType::Mage);
373 }
374
375 // TODO : 아쳐와 메이지가 추가된 뒤에 삭제할 부분
376
377 return OutOptions.IsEmpty() == false;
378}

388const TArray<TSoftObjectPtr<UStaticPlayerUnitSpawnData>>& AFrontendGameMode::GetPlayerUnitDatas() const
389{
390 /*
391 * 선택 가능한 캐릭터 목록은 FrontendRoom DataAsset(DA_TestFrontend)의 mPlayableUnits가 기준이다.
392 * Intro/FrontendRoom 프리로드 단계에서 FrontendRoom과 같은 번들의 PlayerUnit들이 메모리에 올라와 있어야 하고,
393 * 여기서는 이미 로드된 PrimaryAsset 객체에서 SoftObjectPtr 목록만 꺼내 캐시한다.
394 *
395 * 왜 CSV나 별도 임시 목록을 쓰지 않는가:
396 * 프론트엔드 방 정보 안에 "이 화면에서 선택 가능한 캐릭터"가 이미 들어 있으므로,
397 * UI 쪽에서 새 데이터 소스를 만들면 에셋 설정과 화면 표시가 갈라진다.
398 */
399 if (mPlayerUnitDataCache.IsEmpty() == true)
400 {
401 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
402 if (AssetManager != nullptr)
403 {
404 const UGamePlaySettings* GamePlaySettings = GetDefault<UGamePlaySettings>();
405 if (GamePlaySettings != nullptr)
406 {
407 TObjectPtr<UStaticFrontendRoomSpawnData> FrontendRoomSpawnData = AssetManager->GetPrimaryAssetObject<UStaticFrontendRoomSpawnData>(GamePlaySettings->mFrontendRoomId);
408 mPlayerUnitDataCache = FrontendRoomSpawnData->mPlayableUnits;
409 }
410 }
411 }
412 return mPlayerUnitDataCache;
413}

423bool AFrontendGameMode::IsPlayerUnitIdValid(const FPrimaryAssetId& PlayerUnitId) const
424{
425 /*
426 * Confirm 버튼이 보낸 PlayerUnitId가 실제 프론트엔드 방의 선택 후보 안에 있는지 마지막으로 확인한다.
427 * UI에서 비활성 카드를 막아도, 런 생성 직전 GameMode가 다시 검증해야 잘못된 ID로 StartRun()이 호출되지 않는다.
428 */
429 if (PlayerUnitId.IsValid() == false)
430 {
431 return false;
432 }
433
434 const TArray<TSoftObjectPtr<UStaticPlayerUnitSpawnData>>& PlayerUnitDatas = GetPlayerUnitDatas();
435 bool IsFound = PlayerUnitDatas.ContainsByPredicate([&PlayerUnitId](const TSoftObjectPtr<UStaticPlayerUnitSpawnData>& PlayerUnitData) {
436 const UStaticPlayerUnitSpawnData* LoadedPlayerUnitData = PlayerUnitData.Get();
437 if (LoadedPlayerUnitData == nullptr)
438 {
439 return false;
440 }
441 return LoadedPlayerUnitData->GetPrimaryAssetId() == PlayerUnitId;
442 });
443
444 return IsFound;
445}

493bool AFrontendGameMode::CreateRunData(const FPrimaryAssetId& PlayerUnitId, int32 Difficulty)
494{
495 /*
496 * RunPersistData 생성은 캐릭터/난이도 검증이 끝난 뒤 GameProfileSubsystem에 위임한다.
497 * 이 함수가 UI DTO나 WBP 상태를 읽지 않는 이유는, 런 생성 규칙을 화면 구조와 분리하기 위해서다.
498 */
499 if (IsPlayerUnitIdValid(PlayerUnitId) == false)
500 {
501 UE_LOG(LogFrontendGameMode, Log, TEXT("플레이어 유닛 데이터를 찾을 수 없음"));
502 return false;
503 }
504 if (IsDifficultyValid(Difficulty) == false)
505 {
506 UE_LOG(LogFrontendGameMode, Log, TEXT("불가능한 난이도 선택"));
507 return false;
508 }
509
510 UGameProfileSubsystem* GameProfileSubsystem = GetGameInstance()->GetSubsystem<UGameProfileSubsystem>();
511 checkf(GameProfileSubsystem != nullptr, TEXT("게임 프로파일 서브시스템 nullptr 오류"));
512
513 if (GetUserPersistData()->IsActive() == false)
514 {
515 /*
516 * 세이브가 없는 첫 실행은 Intro에서 유저 데이터 로드 실패 후 빈 UserData로 프론트엔드에 들어온다.
517 * 새 런 생성은 유저 프로필을 전제로 하므로, 첫 런 시작 시점에 기본 유저를 먼저 만든다.
518 */
519 GameProfileSubsystem->MakeUser(NSLOCTEXT("FrontendGameMode", "DefaultUserName", "Player"));
520 }
521
522 GameProfileSubsystem->StartRun(PlayerUnitId, Difficulty);
523 return true;
524}
