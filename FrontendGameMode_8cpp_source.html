160 {
161 Options.StableSort([](const FFrontendCharacterOption& Left, const FFrontendCharacterOption& Right)
162 {
163 const int32 LeftOrder = GetClassSelectSortOrder(Left.mJobType);
164 const int32 RightOrder = GetClassSelectSortOrder(Right.mJobType);
165 if (LeftOrder != RightOrder)
166 {
167 return LeftOrder < RightOrder;
168 }
169
170 return Left.mIndex < Right.mIndex;
171 });
172
173 for (int32 OptionIndex = 0; OptionIndex < Options.Num(); ++OptionIndex)
174 {
175 Options[OptionIndex].mIndex = OptionIndex;
176 }

317bool AFrontendGameMode::StartNewRun(const TArray<FPrimaryAssetId>& PlayerUnitIds, int32 Difficulty)
318{
319 /*
320 * 캐릭터 선택 화면에서 Confirm이 눌린 뒤의 실제 시작 지점이다.
321 * 먼저 선택된 캐릭터로 RunPersistData를 만들고, 그 다음 Stage1 첫 방을 프리로드/전환한다.
322 * UI는 "시작 요청"만 하고, 저장 데이터 생성과 방 전환 순서는 GameMode가 고정한다.
323 */
324 if (mWasNextRoomPreloadRequested == true)
325 {
326 UE_LOG(LogFrontendGameMode, Log, TEXT("방 전환 시 추가 로직 요청 불가"));
327 return false;
328 }
329
330 const bool IsRunDataCreated = CreateRunData(PlayerUnitIds, Difficulty);
331 checkf(IsRunDataCreated == true, TEXT("런 데이터 생성 오류"));
332 if (IsRunDataCreated == false)
333 {
334 return false;
335 }
336
337 // 새 Run의 시작 스테이지는 Stage1 고정
338 EStageLevelType FirstStageLevel = EStageLevelType::Stage1;
339
340#if !UE_BUILD_SHIPPING
341 // 디버깅용 시작 스테이지 고정이 켜져 있으면 지정 레벨로 대체 (꺼져 있으면 기존 흐름 그대로)
342 if (TryGetFixedFirstStageLevelForDebugging(OUT FirstStageLevel) == true)
343 {
344 UE_LOG(LogFrontendGameMode, Warning, TEXT("시작 스테이지를 %s로 고정 (Stage.FixedFirstStageLevel)"), *GFixedFirstStageLevelNameForDebugging);
345 }
346#endif
347
348 const bool IsTransitionStarted = PreloadAndTransitionRoomAsync(FirstStageLevel);
349 checkf(IsTransitionStarted == true, TEXT("시작 스테이지 처음 방으로 전환 실패"));
350 return IsTransitionStarted;
351}

353bool AFrontendGameMode::ContinueRunFromTitle()
354{
355 /*
356 * Continue는 "세이브 파일을 불러온다"가 아니라 "이미 활성화된 Run의 현재 방으로 돌아간다"는 의미다.
357 * 세이브 슬롯 선택/로드는 프론트엔드에 진입하기 전 또는 별도 UI 흐름에서 끝나 있어야 하고,
358 * 이 함수는 RunPersistData가 이미 준비되어 있다는 전제에서 방 전환만 시작한다.
359 *
360 * 타이틀에서는 더 이상 지도 미리보기나 다음 방 선택을 열지 않는다.
361 * 타이틀은 시작/이어하기 같은 프론트엔드 선택만 담당하고,
362 * 현재 방 UI, 월드맵 표시, 다음 방 선택, 저장 후 전환은 실제 방에 들어간 뒤 RoomGameMode가 맡는다.
363 */
364 if (mWasNextRoomPreloadRequested == true)
365 {
366 UE_LOG(LogFrontendGameMode, Log, TEXT("방 전환 시 추가 로직 요청 불가"));
367 return false;
368 }
369
370 /*
371 * Continue 버튼은 TitleMenuWidget에서 활성 Run이 있을 때만 보이지만,
372 * UI 상태가 갱신되기 전 클릭되거나 외부 흐름에서 직접 호출될 수 있으므로 GameMode에서도 다시 확인한다.
373 */
374 const URunPersistData* RunPersistData = GetRunPersistData();
375 if (RunPersistData == nullptr || RunPersistData->IsActive() == false)
376 {
377 UE_LOG(LogFrontendGameMode, Log, TEXT("이어갈 런 데이터가 없음"));
378 return false;
379 }
380
381 int32 CurrentRowIndex = INDEX_NONE;
382 int32 CurrentColumnIndex = INDEX_NONE;
383 RunPersistData->GetCurrentRoomIndex(OUT CurrentRowIndex, OUT CurrentColumnIndex);
384
385 /*
386 * RunPersistData에는 "현재 방 좌표"만 저장되어 있으므로,
387 * 전환 요청 전에 해당 좌표가 현재 Stage에 실제로 존재하는 방인지 확인한다.
388 * 좌표가 깨진 상태로 PreloadAndTransitionRoomAsync()에 넘기면 잘못된 방 전환 요청이 되기 때문이다.
389 */
390 const FStage& Stage = RunPersistData->GetStage();
391 if (Stage.HasRoom(CurrentRowIndex, CurrentColumnIndex) == false)
392 {
393 UE_LOG(LogFrontendGameMode, Log, TEXT("현재 런의 방 위치가 Stage에 없음"));
394 return false;
395 }
396
397 /*
398 * 여기서부터는 기존 방 전환 공통 흐름을 사용한다.
399 * 플레이어/스테이지/방 에셋 프리로드, 페이드/로딩 UI, 실제 레벨 전환 순서는 RDGameModeBase와 RoomTransitionSubsystem이 처리한다.
400 */
401 const bool IsTransitionStarted = PreloadAndTransitionRoomAsync(CurrentRowIndex, CurrentColumnIndex);
402 checkf(IsTransitionStarted == true, TEXT("이어하기 방 전환 실패"));
403 return IsTransitionStarted;
404}

406bool AFrontendGameMode::AbandonRunFromTitle()
407{
408 /*

428bool AFrontendGameMode::GetCharacterOptions(TArray<FFrontendCharacterOption>& OutOptions) const
429{
430 /*
431 * CharacterSelectWidget은 PlayerUnit DataAsset 구조를 직접 알지 않는다.
432 * 이 함수가 DataAsset을 UI 전용 DTO로 바꿔주면, WBP 쪽은 이름/직업/스탯/초상화/선택 가능 여부만 보고 카드를 그릴 수 있다.
433 */
434 OutOptions.Reset();
435
436 const TArray<TSoftObjectPtr<UStaticPlayerUnitSpawnData>>& PlayerUnitDatas = GetPlayerUnitDatas();
437 for (const TSoftObjectPtr<UStaticPlayerUnitSpawnData>& PlayerUnitData : PlayerUnitDatas)
438 {
439 const UStaticPlayerUnitSpawnData* LoadedPlayerUnitData = PlayerUnitData.Get();
440 if (LoadedPlayerUnitData == nullptr)
441 {
442 UE_LOG(LogRD, Warning, TEXT("선택 가능한 Player Unit 로드 실패"));
443 continue;
444 }
445
446 FFrontendCharacterOption NewOption;
447 // mIndex는 클릭 이벤트의 안정 키다. WBP 카드 배열 위치와 같게 시작하지만 UI는 이 값을 기준으로 다시 찾는다.
448 NewOption.mIndex = OutOptions.Num();

465 GetWorld(), UPlayerUnitAttributeSet::StaticClass(),
466 UUnitAttributeSet::GetRechargeSpeedPointAttribute(), DefaultDifficulty));

470 FText::AsNumber(NewOption.mGold));

472 NewOption.mIcon = LoadedPlayerUnitData->mIcon;
473 // 용병 선택과 전투 카드가 같은 스킬 DA 이름·아이콘을 보여 주도록
474 // 게임모드가 실제 후보 데이터에 함께 실어 보낸다.

486 // 로드한 DA의 정적 스펙을 UI 전용 값으로 함께 내려준다. 전투 중에만

488 FFrontendSkillOption& SkillOption = NewOption.mSkillDetails.AddDefaulted_GetRef();

491 SkillOption.mIcon = Skill->mIcon;
492 SkillOption.mCooldownTurns = FMath::Max(Skill->mCooldownDuration, 0);
493 SkillOption.mAimPattern = Skill->mAimPattern;
494 SkillOption.mAimRange = Skill->mAimRange;
495 SkillOption.mEffectPattern = Skill->mEffectPattern;
496 SkillOption.mEffectArea = Skill->mEffectArea;
497 SkillOption.mAimBlockerMask = Skill->mAimBlockerMask;
498 SkillOption.mEffectBlockerMask = Skill->mEffectBlockerMask;

502 SkillOption.mActionPointCost = FMath::Max(UnitSkill->mRequiredActionPoint, 0);
503 }

518 }

527 OutOptions.Add(MoveTemp(NewOption));
528 }

530 // [합의필요] Archer/Mage DataAsset이 준비되면 placeholder 보강을 제거하고 DA_TestFrontend.mPlayableUnits만 단일 출처로 둔다.

567 TObjectPtr<UStaticFrontendRoomSpawnData> FrontendRoomSpawnData = AssetManager->GetPrimaryAssetObject<UStaticFrontendRoomSpawnData>(GamePlaySettings->mFrontendRoomId);
568 mPlayerUnitDataCache = FrontendRoomSpawnData->mPlayableUnits;
569 }
570 }

575bool AFrontendGameMode::IsAnyPlayerUnitIdValid(const TArray<FPrimaryAssetId>& PlayerUnitIds) const
576{
577 /*
578 * Confirm 버튼이 보낸 PlayerUnitId가 실제 프론트엔드 방의 선택 후보 안에 있는지 마지막으로 확인한다.
579 * UI에서 비활성 카드를 막아도, 런 생성 직전 GameMode가 다시 검증해야 잘못된 ID로 StartRun()이 호출되지 않는다.
580 */
581
582 bool IsValid = false;
583 for (const FPrimaryAssetId& PlayerUnitId : PlayerUnitIds)
584 {
585 if (PlayerUnitId.IsValid() == false)
586 {
587 continue;
588 }
589
590 const TArray<TSoftObjectPtr<UStaticPlayerUnitSpawnData>>& PlayerUnitDatas = GetPlayerUnitDatas();
591 IsValid = PlayerUnitDatas.ContainsByPredicate([&PlayerUnitId](const TSoftObjectPtr<UStaticPlayerUnitSpawnData>& PlayerUnitData) {
592 const UStaticPlayerUnitSpawnData* LoadedPlayerUnitData = PlayerUnitData.Get();
593 if (LoadedPlayerUnitData == nullptr)
594 {
595 return false;
596 }
597 return LoadedPlayerUnitData->GetPrimaryAssetId() == PlayerUnitId;
598 });
599
600 if (IsValid == true)
601 {
602 break;
603 }
604 }
605
606 return IsValid;
607}

700bool AFrontendGameMode::CreateRunData(const TArray<FPrimaryAssetId>& PlayerUnitIds, int32 Difficulty)
701{

718 checkf(GameProfileSubsystem != nullptr, TEXT("게임 프로파일 서브시스템 nullptr 오류"));

720 if (GetUserPersistData()->IsActive() == false)

723 * 세이브가 없는 첫 실행은 Intro에서 유저 데이터 로드 실패 후 빈 UserData로 프론트엔드에 들어온다.
724 * 새 런 생성은 유저 프로필을 전제로 하므로, 첫 런 시작 시점에 기본 유저를 먼저 만든다.
725 */
726 GameProfileSubsystem->MakeUser(NSLOCTEXT("FrontendGameMode", "DefaultUserName", "Player"));
727 }
728
729 GameProfileSubsystem->StartRun(PlayerUnitIds, Difficulty);
730 return true;
731}
