285void UCombatTileMapHUDWidget::UpdateUnitHpBars()
286{
287 // 지도(풀스크린) 열림 중에는 유닛 머리 위 HP바를 숨긴다 — 탑바만 남기는 뷰.
288 // 이 함수가 매 틱 HP바를 강제 표시(라인 283)하므로, SetCombatPlayControlsVisible에서 한 번 숨기는 것으로는
289 // 다음 틱에 되살아난다. 여기서 게이트해야 지도 뷰 동안 계속 숨겨진다.
290 if (mCombatControlsHidden)
291 {
292 for (FUnitHpBarWidget& Bar : mUnitHpBars)
293 {
294 if (Bar.mRoot != nullptr) { Bar.mRoot->SetVisibility(ESlateVisibility::Collapsed); }
295 }
296 return;
297 }
298
299 if (mCombatUIModel == nullptr)
300 {
301 return;
302 }
303
304 APlayerController* PlayerController = GetOwningPlayer();
305 if (PlayerController == nullptr)
306 {
307 return;
308 }
309
310 // 카메라 줌(직교 OrthoWidth)에 비례해 HP바 크기를 키운다 — 줌인하면 유닛과 함께 바도 커진다.
311 // 기준 OrthoWidth(2000) 대비 배율, 과도한 확대/축소는 클램프.
312 float ZoomScale = 1.0f;
313 if (PlayerController->PlayerCameraManager != nullptr)
314 {
315 const float CurrentOrthoWidth = PlayerController->PlayerCameraManager->GetCameraCacheView().OrthoWidth;
316 if (CurrentOrthoWidth > KINDA_SMALL_NUMBER)
317 {
318 ZoomScale = FMath::Clamp(UnitHpBarBaseOrthoWidth / CurrentOrthoWidth, 0.6f, 3.0f);
319 }
320 }
321 const FVector2D BarRenderScale(UnitHpBarRenderScale * ZoomScale, UnitHpBarRenderScale * ZoomScale);
322
323 const TArray<FUnitUI>& Units = mCombatUIModel->GetUnitUIs();
324 for (int32 BarIndex = 0; BarIndex < mUnitHpBars.Num(); ++BarIndex)
325 {
326 FUnitHpBarWidget& Bar = mUnitHpBars[BarIndex];
327 if (Bar.mRoot == nullptr || Units.IsValidIndex(BarIndex) == false)
328 {
329 continue;
330 }
331 const FUnitUI& Unit = Units[BarIndex];
332
333 const float Percent = Unit.mMaxHP > 0.0f ? FMath::Clamp(Unit.mHP / Unit.mMaxHP, 0.0f, 1.0f) : 0.0f;
334
335 // 채움 색: 아군=초록, 적=빨강 텍스처. (없으면 기존 브러시 유지)
336 if (Bar.mFillImage != nullptr)
337 {
338 UTexture2D* FillTexture = Unit.mIsPlayer ? mUnitHpFillGreenTexture : mUnitHpFillRedTexture;
339 if (FillTexture != nullptr)
340 {
341 Bar.mFillImage->SetBrushResourceObject(FillTexture);
342 }
343 }
344
345 // 채움 드레인: 채움 이미지 슬롯 폭을 원본 폭 × 퍼센트로 줄인다(좌상단 고정이라 오른쪽부터 HP가 빠진다).
346 if (Bar.mFillClipSlot != nullptr && Bar.mFillFullWidth > 0.0f)
347 {
348 FVector2D FillSize = Bar.mFillClipSlot->GetSize();
349 FillSize.X = Bar.mFillFullWidth * Percent;
350 Bar.mFillClipSlot->SetSize(FillSize);
351 }
352
353 // HP 숫자: 현재/최대.
354 if (Bar.mValueText != nullptr)
355 {
356 Bar.mValueText->SetText(FText::AsNumber(FMath::RoundToInt(Unit.mHP)));
357 }
358
359 // 방어도: 0보다 클 때만 HP바 왼쪽에 아이콘+수치 표시.
360 const int32 DefensePoint = FMath::RoundToInt(Unit.mDefensePoint);
361 const ESlateVisibility DefenseVisibility = DefensePoint > 0
362 ? ESlateVisibility::HitTestInvisible
363 : ESlateVisibility::Collapsed;
364 if (Bar.mDefenseIcon != nullptr)
365 {
366 Bar.mDefenseIcon->SetVisibility(DefenseVisibility);
367 }
368 if (Bar.mDefenseText != nullptr)
369 {
370 Bar.mDefenseText->SetText(FText::AsNumber(DefensePoint));
371 Bar.mDefenseText->SetVisibility(DefenseVisibility);
372 }
373
374 // 유닛 월드 위치를 화면(위젯) 좌표로 투영. 화면 밖이면 숨긴다.
375 // 이동을 매 프레임 따라가도록 뷰 액터의 라이브 위치를 우선 투영(유효 시). 없으면 스냅샷(mWorldLocation) 폴백.
376 const FVector ProjectLocation = Unit.mViewActor.IsValid()
377 ? Unit.mViewActor->GetActorLocation()
378 : Unit.mWorldLocation;
379 FVector2D ScreenPosition;
380 const bool bOnScreen = UWidgetLayoutLibrary::ProjectWorldLocationToWidgetPosition(
381 PlayerController, ProjectLocation, ScreenPosition, false);
382 if (bOnScreen == false)
383 {
384 Bar.mRoot->SetVisibility(ESlateVisibility::Collapsed);
385 continue;
386 }
387
388 if (UCanvasPanelSlot* RootSlot = Cast<UCanvasPanelSlot>(Bar.mRoot->Slot))
389 {
390 RootSlot->SetPosition(ScreenPosition + FVector2D(0.0f, UnitHpBarHeadOffsetY)); // 유닛 머리 위로 띄운다.
391 }
392
393 // 줌 배율 반영(하단 중앙 피벗이라 커져도 밑변이 머리 위 투영점에 고정).
394 Bar.mRoot->SetRenderScale(BarRenderScale);
395
396 // HP바 밑 상태이상 아이콘/개수 갱신(온스크린 확정된 바에만).

417void UCombatTileMapHUDWidget::UpdateUnitHpBarStatus(FUnitHpBarWidget& Bar, const FUnitUI& Unit) const
418{
419 const int32 SlotCount = Bar.mStatusIcons.Num();
420
421 int32 Shown = 0;
422 int32 Iconable = 0;
423 for (const FStatusEffectUI& Status : Unit.mStatusEffects)
424 {
425 UTexture2D* Icon = ResolveStatusIcon(Status.mTag);
426 if (Icon == nullptr)
427 {
428 continue; // 전용 아이콘 없는 상태는 표시 생략.
429 }
430 ++Iconable;
431
432 if (Shown >= SlotCount)
433 {
434 continue; // 슬롯 초과분은 아래 오버플로로만 집계.
435 }
436
437 if (UImage* IconImage = Bar.mStatusIcons[Shown])
438 {
439 IconImage->SetBrushResourceObject(Icon);
440 IconImage->SetVisibility(ESlateVisibility::HitTestInvisible);
441 }
442 if (Bar.mStatusCountTexts.IsValidIndex(Shown))
443 {
444 if (UTextBlock* CountText = Bar.mStatusCountTexts[Shown])
445 {
446 if (Status.mStackCount > 1)
447 {
448 CountText->SetText(FText::AsNumber(Status.mStackCount));
449 CountText->SetVisibility(ESlateVisibility::HitTestInvisible);
450 }
451 else
452 {
453 CountText->SetVisibility(ESlateVisibility::Collapsed);
454 }
455 }
456 }
457 ++Shown;
458 }
459
460 // 채우지 못한 뒤쪽 슬롯 숨김.
461 for (int32 SlotIndex = Shown; SlotIndex < SlotCount; ++SlotIndex)
462 {
463 if (UImage* IconImage = Bar.mStatusIcons[SlotIndex]) { IconImage->SetVisibility(ESlateVisibility::Collapsed); }
464 }
465 for (int32 SlotIndex = Shown; SlotIndex < Bar.mStatusCountTexts.Num(); ++SlotIndex)
466 {
467 if (UTextBlock* CountText = Bar.mStatusCountTexts[SlotIndex]) { CountText->SetVisibility(ESlateVisibility::Collapsed); }
468 }
469
470 // 5칸 초과 상태 개수를 "+N"으로.
471 if (Bar.mStatusOverflowText != nullptr)
472 {
473 const int32 Overflow = Iconable - SlotCount;
474 if (Overflow > 0)
475 {
476 Bar.mStatusOverflowText->SetText(FText::FromString(FString::Printf(TEXT("+%d"), Overflow)));
477 Bar.mStatusOverflowText->SetVisibility(ESlateVisibility::HitTestInvisible);
478 }
479 else
480 {
481 Bar.mStatusOverflowText->SetVisibility(ESlateVisibility::Collapsed);
482 }
483 }
