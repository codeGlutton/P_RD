37 bool HasCanonicalModernLayout(UWidgetBlueprint* Blueprint)
38 {
39 if (Blueprint == nullptr || Blueprint->WidgetTree == nullptr
40 || Blueprint->WidgetTree->FindWidget(TEXT("MonsterNamePlate")) == nullptr
41 || Blueprint->WidgetTree->FindWidget(TEXT("MonsterSkillDivider")) == nullptr)
42 {
43 return false;
44 }
45
46 const UImage* Row = Cast<UImage>(
47 Blueprint->WidgetTree->FindWidget(TEXT("MonsterRowNormal_0")));
48 const UImage* PortraitFrame = Cast<UImage>(
49 Blueprint->WidgetTree->FindWidget(TEXT("MonsterPortraitFrame")));
50 const UObject* RowResource = Row != nullptr
51 ? Row->GetBrush().GetResourceObject() : nullptr;
52 const UObject* FrameResource = PortraitFrame != nullptr
53 ? PortraitFrame->GetBrush().GetResourceObject() : nullptr;
54 return RowResource != nullptr && FrameResource != nullptr
55 && RowResource->GetPathName() == CanonicalRowTexturePath
56 && (FrameResource->GetPathName() == LegacyPortraitFramePath
57 || FrameResource->GetPathName() == PortraitCellPath);
58 }

87 void RepairSkillInputOnly()
88 {
89 UWidgetBlueprint* Blueprint = LoadObject<UWidgetBlueprint>(nullptr, AssetPath);
90 if (HasCanonicalModernLayout(Blueprint) == false)
91 {
92 UE_LOG(LogTemp, Error,
93 TEXT("RD_MONSTER_TAB_REPAIR refused: canonical modern layout signature is missing"));
94 return;
95 }
96
97 UCanvasPanel* Canvas = Cast<UCanvasPanel>(
98 Blueprint->WidgetTree->FindWidget(TEXT("MonsterTabCanvas")));
99 if (Canvas == nullptr)
100 {
101 UE_LOG(LogTemp, Error,
102 TEXT("RD_MONSTER_TAB_REPAIR refused: MonsterTabCanvas is missing"));
103 return;
104 }
105
106 Blueprint->Modify();
107 Blueprint->WidgetTree->Modify();
108
109 UTexture2D* PortraitCell = LoadObject<UTexture2D>(nullptr, PortraitCellPath);
110 UImage* PortraitFrame = Cast<UImage>(
111 Blueprint->WidgetTree->FindWidget(TEXT("MonsterPortraitFrame")));
112 UImage* DetailPortrait = Cast<UImage>(
113 Blueprint->WidgetTree->FindWidget(TEXT("MonsterDetailPortrait")));
114 UWidget* DetailPortraitLayer = Blueprint->WidgetTree->FindWidget(
115 TEXT("MonsterDetailPortraitScale"));
116 if (PortraitCell == nullptr || PortraitFrame == nullptr
117 || DetailPortrait == nullptr || DetailPortraitLayer == nullptr)
118 {
119 UE_LOG(LogTemp, Error,
120 TEXT("RD_MONSTER_TAB_REPAIR missing portrait cell widgets or texture"));
121 return;
122 }
123 FSlateBrush PortraitCellBrush = PortraitFrame->GetBrush();
124 PortraitCellBrush.SetResourceObject(PortraitCell);
125 PortraitCellBrush.DrawAs = ESlateBrushDrawType::Image;
126 PortraitCellBrush.Margin = FMargin(0.f);
127 PortraitFrame->SetBrush(PortraitCellBrush);
128 PortraitFrame->SetColorAndOpacity(FLinearColor::White);
129 PortraitFrame->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
130 DetailPortrait->SetClipping(EWidgetClipping::ClipToBoundsAlways);
131 DetailPortrait->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
132 if (UCanvasPanelSlot* FrameSlot = Cast<UCanvasPanelSlot>(PortraitFrame->Slot))
133 {
134 if (const UCanvasPanelSlot* PortraitLayerSlot =
135 Cast<UCanvasPanelSlot>(DetailPortraitLayer->Slot))
136 {
137 // T_KitA_Cell_Normal은 중앙이 불투명한 셀이므로 초상화 위에
138 // 얹으면 그림을 가린다. 셀을 매트로 아래에 두고, 내부 초상화를
139 // 위에서 정사각 크롭한다.
140 FrameSlot->SetZOrder(PortraitLayerSlot->GetZOrder() - 1);
141 }
142 }
143 for (int32 Index = 0; Index < 4; ++Index)
144 {
145 const FName SlotName(*FString::Printf(TEXT("MonsterSkillSlot_%d"), Index));
146 const FName ButtonName(*FString::Printf(TEXT("MonsterSkillButton_%d"), Index));
147 UImage* SkillSlot = Cast<UImage>(Blueprint->WidgetTree->FindWidget(SlotName));
148 UCanvasPanelSlot* AuthoredSlot = SkillSlot != nullptr
149 ? Cast<UCanvasPanelSlot>(SkillSlot->Slot) : nullptr;
150 if (SkillSlot == nullptr || SkillSlot->GetParent() != Canvas
151 || AuthoredSlot == nullptr)
152 {
153 UE_LOG(LogTemp, Error,
154 TEXT("RD_MONSTER_TAB_REPAIR refused: authored socket %s is invalid"),
155 *SlotName.ToString());
156 return;
157 }
158
159 UButton* Button = Cast<UButton>(Blueprint->WidgetTree->FindWidget(ButtonName));
160 if (Button == nullptr)
161 {
162 Button = Blueprint->WidgetTree->ConstructWidget<UButton>(
163 UButton::StaticClass(), ButtonName);
164 }
165 if (Button->GetParent() != Canvas)
166 {
167 if (UPanelWidget* Parent = Button->GetParent())
168 {
169 Parent->RemoveChild(Button);
170 }
171 Canvas->AddChildToCanvas(Button);
172 }
173
174 UCanvasPanelSlot* ButtonSlot = CastChecked<UCanvasPanelSlot>(Button->Slot);
175 ButtonSlot->SetAnchors(AuthoredSlot->GetAnchors());
176 ButtonSlot->SetAlignment(AuthoredSlot->GetAlignment());
177 ButtonSlot->SetAutoSize(AuthoredSlot->GetAutoSize());
178 ButtonSlot->SetPosition(AuthoredSlot->GetPosition());
179 ButtonSlot->SetSize(AuthoredSlot->GetSize());
180 ButtonSlot->SetZOrder(24);
181 StyleTransparentButton(Button);
182 Expose(Blueprint, Button);
183 }
184
185 FBlueprintEditorUtils::MarkBlueprintAsStructurallyModified(Blueprint);
186 FKismetEditorUtilities::CompileBlueprint(Blueprint);
187 const FString Filename = FPackageName::LongPackageNameToFilename(
188 Blueprint->GetOutermost()->GetName(), FPackageName::GetAssetPackageExtension());
189 if (UPackage::SavePackage(Blueprint->GetPackage(), Blueprint, *Filename,
190 FSavePackageArgs()) == false)
191 {
192 UE_LOG(LogTemp, Error, TEXT("RD_MONSTER_TAB_REPAIR save failed"));
193 return;
194 }
195
196 UE_LOG(LogTemp, Display,
197 TEXT("RD_MONSTER_TAB_REPAIR success portrait=T_KitA_Cell_Normal added=MonsterSkillButton_0..3"));
198 }
