44ESRPGCommandResult USRPGSkillAction::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
45{
46 ESRPGCommandResult Result = Super::HandleCommand(Command);
47 if (Result == ESRPGCommandResult::Handled)
48 {
49 return Result;
50 }
51
52 switch (Command.Get().GetCommandType())
53 {
54 case ESRPGCommandType::SkillCast:
55 {
56 /* 스킬 실행 */
57
58 const FSRPGSkillCastCommand& SkillCastCommand = Command.Get<FSRPGSkillCastCommand>();
59
60 USkillComponentModel* SkillCompModel = mInstigator->GetSkillComponentModel();
61 checkf(SkillCompModel != nullptr, TEXT("스킬 컴포넌트 모델 nullptr"));
62
63 UTileMapModel* TileMap = GetTileMap();
64 checkf(TileMap != nullptr, TEXT("타일 맵 nullptr"));
65
66 if (SkillCompModel->IsAnySkillActivated() == true)
67 {
68 // 이미 스킬 실행 중 무시
69 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
70 }
71
72 // 이동 중 방해받아 계획한 위치에 도달하지 못할 수도 있으므로,
73 // 스킬 사용이 가능한지 한 번 더 확인
74 if (SkillCompModel->CanActiveSkill(SkillCastCommand.mSkillIndex) == false)
75 {
76 MarkActionCompleted(ESRPGActionResult::Cancelled);
77 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
78 }
79
80 // 밀치기 등으로 인해 명령을 만든 시점과 현재 시점의 위치가 다를 수 있음
81 // 현재 위치에서 조준할 수 없는 타일인 지 한 번 더 확인하고, 조준 불가능하면 액션 취소
82 if (SkillCompModel->GetAimableTiles(TileMap, SkillCastCommand.mSkillIndex).Contains(SkillCastCommand.mTargetIndex) == false)
83 {
84 MarkActionCompleted(ESRPGActionResult::Cancelled);
85 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
86 }
87
88 FOnEndSkillUI Callback;
89 Callback.AddWeakLambda(this, [this](const FActiveSkillContext& Context, const UStaticSkillData* PreSkillData) {
90 FSimulationOption Option;
91 Option.mDuration = ESimulationDurtaion::AllPlayerTurnEnd;
92 Option.mSkipAIActions = true;
93 Option.mReservedCommand.InitializeAs<FSRPGTurnEndCommand>();
94 GetCombatModel()->OnSimulateAllPlayerTurn.Broadcast(Option);
95
96 MarkActionCompleted(ESRPGActionResult::Succeeded);
97 });
98
99 SkillCompModel->ForcedActivateSkill(TileMap, SkillCastCommand.mSkillIndex, SkillCastCommand.mTargetIndex, MoveTemp(Callback));
100
101 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
102 }
103 }
104
105 return ESRPGCommandResult::Ignored;
106}
