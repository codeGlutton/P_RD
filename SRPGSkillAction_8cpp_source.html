41ESRPGCommandResult USRPGSkillAction::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
42{
43 ESRPGCommandResult Result = Super::HandleCommand(Command);
44 if (Result == ESRPGCommandResult::Handled)
45 {
46 return Result;
47 }
48
49 switch (Command.Get().GetCommandType())
50 {
51 case ESRPGCommandType::SkillCast:
52 {
53 /* 스킬 실행 */
54
55 const FSRPGSkillCastCommand& SkillCastCommand = Command.Get<FSRPGSkillCastCommand>();
56
57 USkillComponentModel* SkillCompModel = mInstigator->GetSkillComponentModel();
58 checkf(SkillCompModel != nullptr, TEXT("스킬 컴포넌트 모델 nullptr"));
59
60 UTileMapModel* TileMap = GetTileMap();
61 checkf(TileMap != nullptr, TEXT("타일 맵 nullptr"));
62
63 if (SkillCompModel->IsAnySkillActivated() == true)
64 {
65 // 이미 스킬 실행 중 무시
66 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
67 }
68
69 FOnEndSkillUI Callback;
70 Callback.AddWeakLambda(this, [this](const FActiveSkillContext& Context, const UStaticSkillData* PreSkillData) {
71 MarkActionCompleted(ESRPGActionResult::Succeeded);
72 });
73
74 SkillCompModel->ActivateSkill(TileMap, SkillCastCommand.mSkillIndex, SkillCastCommand.mTargetIndex, SkillCastCommand.mDiceSum, MoveTemp(Callback));
75
76 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
77 }
78 }
79
80 return ESRPGCommandResult::Ignored;
81}
