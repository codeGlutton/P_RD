52bool FTacticalSimulationTests::RunTest(const FString& Parameters)
53{
54 // 1. 월드 및 서브시스템 획득
55 UWorld* World = GetAnyGameWorld();
56 if (World == nullptr)
57 {
58 World = GWorld;
59 }
60 if (World == nullptr)
61 {
62 AddError(TEXT("유효한 UWorld를 찾는 데 실패했습니다."));
63 return false;
64 }
65
66 USimulationSubsystem* SimSubsystem = World->GetSubsystem<USimulationSubsystem>();
67 if (TestNotNull(TEXT("USimulationSubsystem이 초기화되어야 합니다."), SimSubsystem) == false)
68 {
69 return false;
70 }
71
72 URoomInstance* RoomInstance = FSimulationSubsystemTestAccessor::GetRoomInstance(SimSubsystem);
73 if (TestNotNull(TEXT("RoomInstance가 초기화되어야 합니다."), RoomInstance) == false)
74 {
75 return false;
76 }
77
78 // UTacticalFrameworkModel 모크 등록 (기존 등록 상태가 없을 때만 등록)
79 UTacticalFrameworkModel* FrameworkModel = Cast<UTacticalFrameworkModel>(RoomInstance->mAliveSubsystemModels.FindRef(UTacticalFrameworkModel::StaticClass()));
80 if (FrameworkModel == nullptr)
81 {
82 FrameworkModel = NewObject<UTacticalFrameworkModel>(RoomInstance);
83 RoomInstance->mAliveSubsystemModels.Add(UTacticalFrameworkModel::StaticClass(), FrameworkModel);
84 }
85
86 // 2. 모델 팩토리를 통해 액터 모델 생성 (RoomInstance에 소속되어 복제 대상이 됨)
87 UTASActorModelMock* MockActorModel = SimSubsystem->GetModelFactory().NewModel<UTASActorModelMock>();
88 if (TestNotNull(TEXT("MockActorModel이 팩토리를 통해 생성되어야 합니다."), MockActorModel) == false)
89 {
90 return false;
91 }
92
93 UAttributeSetComponentModel* CompModel = MockActorModel->GetAttributeComponent();
94 if (TestNotNull(TEXT("AttributeSetComponentModel이 유효해야 합니다."), CompModel) == false)
95 {
96 return false;
97 }
98
99 // 기본 방어력 세팅
100 CompModel->SetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute(), 10.f);
101 TestEqual(TEXT("초기 기본 방어력은 10.f이어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
102 TestEqual(TEXT("초기 현재 방어력은 10.f이어야 합니다."), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
103
104 // 3. 무한 이펙트 적용 (방어력 +30)
105 UTacticalEffectContext* EffectContext = CompModel->MakeEffectContext();
106 EffectContext->SetInstigator(MockActorModel);
107 EffectContext->SetAttributeSetComponentModel(CompModel);
108
109 TSharedPtr<FTacticalEffectSpec> InfiniteEffect = CompModel->MakeOutgoingSpec(UTestInfiniteTacticalEffect::StaticClass(), EffectContext);
110 InfiniteEffect->mDynamicMagnitude = 30.f;
111 FActiveTacticalEffectHandle ActiveHandle = CompModel->ApplyTacticalEffectSpecToSelf(*InfiniteEffect);
112
113 TestTrue(TEXT("ActiveHandle이 유효해야 합니다."), ActiveHandle.IsValid());
114 TestEqual(TEXT("기본 방어력은 10.f로 유지되어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
115 TestEqual(TEXT("현재 방어력은 40.f이어야 합니다. (기본 10 + 모디파이어 30)"), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 40.f);
116
117 // 4. 시뮬레이션 상태로 변경하여 복제 유도
118 AddInfo(TEXT("=== 시뮬레이션 상태로 전환 (RoomInstance 복제) ==="));
119 FSimulationSubsystemTestAccessor::SetSimState(SimSubsystem, ESRPGSimulationState::RunningSimulation);
120
121 // 복제된 룸 인스턴스 획득
122 URoomInstance* SimRoomInstance = FSimulationSubsystemTestAccessor::GetRoomInstance(SimSubsystem);
123 if (TestNotNull(TEXT("SimRoomInstance가 유효해야 합니다."), SimRoomInstance) == false)
124 {
125 return false;
126 }
127 TestTrue(TEXT("SimRoomInstance는 원본 RoomInstance와 다른 인스턴스여야 합니다."), SimRoomInstance != RoomInstance);
128
129 // 복제된 액터 모델 검색
130 int32 ModelId = MockActorModel->GetModelId();
131 UTASActorModelMock* ClonedActorModel = Cast<UTASActorModelMock>(SimRoomInstance->mAliveWorldModels.FindRef(ModelId));
132 if (TestNotNull(TEXT("SimRoomInstance 내에 ClonedActorModel이 존재해야 합니다."), ClonedActorModel) == false)
133 {
134 return false;
135 }
136 TestTrue(TEXT("ClonedActorModel은 원본과 다른 객체여야 합니다."), ClonedActorModel != MockActorModel);
137
138 UAttributeSetComponentModel* ClonedCompModel = ClonedActorModel->GetAttributeComponent();
139 if (TestNotNull(TEXT("ClonedCompModel이 유효해야 합니다."), ClonedCompModel) == false)
140 {
141 return false;
142 }
143
144 // 복제 직후 수치 일치 여부 확인
145 TestEqual(TEXT("[복제본] 기본 방어력은 10.f이어야 합니다."), ClonedCompModel->GetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
146 TestEqual(TEXT("[복제본] 현재 방어력은 40.f이어야 합니다."), ClonedCompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 40.f);
147
148 // 5. 복제본에서 이펙트 해제 테스트
149 AddInfo(TEXT("=== 복제된 모델에서 활성 이펙트 제거 ==="));
150 bool Removed = ClonedCompModel->RemoveActiveTacticalEffect(ActiveHandle);
151 TestTrue(TEXT("복제된 컴포넌트에서 이펙트가 성공적으로 제거되어야 합니다."), Removed);
152
153 // 6. 결과 검증 (격리 여부 확인)
154 // 복제본은 방어력이 원래 수치(10)로 복구되어야 함
155 TestEqual(TEXT("[복제본] 이펙트 제거 후, 방어력이 10.f로 복구되어야 합니다."), ClonedCompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
156
157 // 원본은 여전히 방어력 버프가 유지되어 40이어야 함
158 TestEqual(TEXT("[원본] 방어력이 40.f로 유지되어야 합니다."), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 40.f);
159
160 // 7. 환경 정리
161 AddInfo(TEXT("=== 게임 상태 복원 및 환경 정리 ==="));
162 FSimulationSubsystemTestAccessor::SetSimState(SimSubsystem, ESRPGSimulationState::RunningGame);
163
164 SimSubsystem->GetModelFactory().DestroyModel(MockActorModel);
165
166 return true;
167}
