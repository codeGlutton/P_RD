54{
55 FTestObjectScope Scope;
56
57 /* 월드와 프레임워크 모델 확보 (파티 연결 시 속성 초기화에 필요) */
58 UWorld* World = GetRestoreTestWorld();
59 if (TestNotNull(TEXT("유효한 월드"), World) == false)
60 {
61 return false;
62 }
63 if (TestNotNull(TEXT("전략 프레임워크 모델"), GetWorldSubsystemModel<UTacticalFrameworkModel>(World)) == false)
64 {
65 return false;
66 }
67
68 /* 실제 DA: 기사 스폰 데이터, 패시브만 있는 아티팩트 */
69 UStaticPlayerUnitSpawnData* KnightData = LoadObject<UStaticPlayerUnitSpawnData>(nullptr, TEXT("/Game/BP/DataAsset/Unit/PlayerUnit/DA_TestKnightPlayerUnit.DA_TestKnightPlayerUnit"));
70 if (TestNotNull(TEXT("기사 스폰 DA"), KnightData) == false)
71 {
72 return false;
73 }
74 const FPrimaryAssetId ArtifactId(TEXT("Artifact"), TEXT("DA_Artifact_A001_ChaliceOfLife"));
75
76 /* 파티와 구성원 2명 생성 (복원 경로처럼 월드 소속으로 초기화) */
77 UPartyModel* PartyModel = Scope.New<UPartyModel>(World);
78 UMockPartyMemberModel* MemberA = Scope.New<UMockPartyMemberModel>(World);
79 UMockPartyMemberModel* MemberB = Scope.New<UMockPartyMemberModel>(World);
80 for (UMockPartyMemberModel* Member : { MemberA, MemberB })
81 {
82 Member->SetStaticSpawnData(KnightData);
83 Member->Initialize();
84 }
85 PartyModel->Initialize();
86
87 /* 영구데이터에 아티팩트 시드 후 동기화 (셋째 슬롯은 빈 자리) */
88 UPartyArtifactRestoreTestData* PersistData = Scope.New<UPartyArtifactRestoreTestData>();
89 PersistData->SeedArtifact(ArtifactId);
90 TArray<TObjectPtr<UPlayerUnitModel>> Players = { MemberA, MemberB, nullptr };
91 PersistData->Sync(PartyModel, Players);
92
93 /* 파티 목록 복원 */
94 TestEqual(TEXT("파티 아티팩트 1개"), PartyModel->GetPartyArtifactComponentModel()->GetPartyArtifacts().Num(), 1);
95
96 /* 구성원 전원 장착과 패시브 설치 */
97 const FGameplayTag StartRoomTiming = FGameplayTag::RequestGameplayTag(FName(TEXT("GameplayAbility.Passive.OnStartRoom")));
98 for (UMockPartyMemberModel* Member : { MemberA, MemberB })
99 {
100 TestEqual(TEXT("구성원 장착 1개"), Member->GetArtifactComponentModel()->GetArtifacts().Num(), 1);
101 TestEqual(TEXT("구성원 패시브 1개"), Member->GetPassiveComponentModel()->GetPassives().Num(), 1);
102 TestEqual(TEXT("OnStartRoom 패시브 1개"), Member->GetPassiveComponentModel()->GetPassivesByTiming(StartRoomTiming).Num(), 1);
103 }
104
105 /* 빈 슬롯은 그대로 */
106 TestNull(TEXT("셋째 슬롯 비어 있음"), PartyModel->GetPlayerUnitModel(2));
107
108 return true;
109}
