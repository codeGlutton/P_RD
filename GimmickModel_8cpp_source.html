32bool UGimmickModel::TryTriggerGimmick(const FTileIndex& AimedTileIndex, TSharedPtr<FPresentationBarrier> PresentationBarrier)
33{
34 // 수명을 다 썼거나 죽은 기믹은 발동 불가
35 if (mRemainingTriggerCount == 0 || IsDead() == true)
36 {
37 return false;
38 }
39
40 USkillComponentModel* SkillComp = GetSkillComponentModel();
41 checkf(SkillComp != nullptr, TEXT("스킬 컴포넌트 nullptr"));
42
43 // 시전 중 재진입 방지 (지속형 기믹이 연출 중에 다시 트리거되는 경우)
44 if (SkillComp->IsAnySkillActivated() == true)
45 {
46 return false;
47 }
48
49 // 수명 차감 (무제한은 음수 유지)
50 // 뷰가 없으면 시전이 아래 호출 안에서 동기로 끝나므로, 종료 콜백이 소진을 볼 수 있게 먼저 차감
51 if (mRemainingTriggerCount > 0)
52 {
53 --mRemainingTriggerCount;
54 }
55
56 // 장착된 스킬 강제 시전 (기믹은 행동력 개념이 없으므로 소모 검사 없이 시전)
57 FOnEndSkillUI EndCallback;
58 EndCallback.AddUObject(this, &UGimmickModel::OnGimmickSkillEnd);
59
60 // 배리어를 받았으면, 배리어 포인터를 캡처한 빈 람다를 종료 콜백에 추가
61 // 스킬이 끝나면 콜백이 정리되고, 람다가 사라지면서 캡처된 포인터도 함께 해제됨
62 if (PresentationBarrier.IsValid() == true)
63 {
64 EndCallback.AddLambda([PresentationBarrier](const FActiveSkillContext&, const UStaticSkillData*) {});
65 }
66
67 SkillComp->ForcedActivateSkill(GetTileMap(), mTriggerSkillIndex, AimedTileIndex, EndCallback);
68 return true;
69}

82void UGimmickModel::OnGimmickSkillEnd(const FActiveSkillContext& Context, const UStaticSkillData* SkillData)
83{
84 // 수명을 다 썼으면 죽음 상태로 표시 (유닛 사망과 같은 태그, IsDead()가 이 태그를 봄)
85 // 실제 제거는 이번 행동이 끝날 때 ClearDeadActorModels가 죽은 액터들을 치우면서 함께 처리됨
86 if (mRemainingTriggerCount == 0)
87 {
88 GetAttributeComponentModel()->AddLooseGameplayTag(EffectTags::GameplayEffect_ActorState_Dead);
89
90 USRPGCombatModel* CombatModel = GetWorldSubsystemModel<USRPGCombatModel>(this);
91 checkf(CombatModel != nullptr, TEXT("전투 모델 nullptr"));
92
93 CombatModel->EvaluateCombatStates();
94 }
