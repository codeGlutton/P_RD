139void UCarouselPanelWidget::SelectCarouselItem(int32 ItemIndex)
140{
141 /*
142 * 첫 선택 전에는 패널이 가로 목록처럼 보인다.
143 * 사용자가 카드를 선택하면 mCarouselActivated를 켜고, 선택된 카드를 기준으로 원형 캐러셀 배치를 계산한다.
144 */
145 if (ItemIndex < 0 || ItemIndex >= GetClampedCarouselItemDisplayCount())
146 {
147 return;
148 }
149
150 const int32 PreviousIndex = mSelectedCarouselIndex;
151 const bool bWasCarouselActivated = mCarouselActivated;
152 mCarouselActivated = true;
153 mSelectedCarouselIndex = ItemIndex;
154 if (PreviousIndex != mSelectedCarouselIndex || !bWasCarouselActivated)
155 {
156 HandleCarouselSelectionChanged(PreviousIndex, mSelectedCarouselIndex);
157 }
158 ApplyCarouselLayout(bWasCarouselActivated);
159}

161void UCarouselPanelWidget::ApplyCarouselLayout(bool bAnimate)
162{
163 /*
164 * 실제 카드 디자인은 WBP가 담당하고, C++은 계산된 카드 묶음의 배치와 깊이감만 적용한다.
165 */
166 const int32 ItemCount = GetClampedCarouselItemDisplayCount();
167 if (ItemCount > 0 && (mSelectedCarouselIndex == INDEX_NONE || mSelectedCarouselIndex >= ItemCount))
168 {
169 mSelectedCarouselIndex = 0;
170 }
171
172 TArray<FCarouselItemLayout> TargetLayouts;
173 BuildCarouselLayouts(TargetLayouts);
174 if (bAnimate == true && mCarouselActivated == true && mCarouselTransitionDuration > 0.0f)
175 {
176 StartCarouselLayoutTransition(TargetLayouts);
177 return;
178 }
179
180 mCarouselTransitionActive = false;
181 ApplyCarouselLayouts(TargetLayouts);
182}
