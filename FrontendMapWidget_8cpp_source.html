670bool UFrontendMapWidget::RefreshMap()
671{
672 RefreshLocalizedTextCache();
673 HideUnusedMapTextSurfaces();
674 mEnterRequested = false;
675
676 if (MapGraphCanvas == nullptr)
677 {
678 return false;
679 }
680
681 TArray<FMapRoomView> Rooms;
682 bool bHasRooms = false;
683 bool bShouldScrollToStart = false;
684 if (ARoomGameModeBase* RoomGameMode = GetWorld() != nullptr ? GetWorld()->GetAuthGameMode<ARoomGameModeBase>() : nullptr)
685 {
686 bHasRooms = RoomGameMode->GetMapRoomViews(Rooms);
687
688 FRunControlView RunControlView;
689 bShouldScrollToStart = RoomGameMode->GetRunControlView(OUT RunControlView) && RunControlView.mIsAtStageStart;
690 }
691 else
692 {
693 UE_LOG(LogRD, Warning, TEXT("FrontendMapWidget: RoomGameMode is not available. WorldMap data must be provided by RoomGameMode."));
694 }
695
696 // 그래프 높이는 행 수 x 행 간격이므로, 데이터를 읽은 뒤에 레이아웃을 확정한다.
697 int32 MaxRowIndex = 0;
698 for (const FMapRoomView& Room : Rooms)
699 {
700 MaxRowIndex = FMath::Max(MaxRowIndex, Room.mRow);
701 }
702 mCachedRowCount = bHasRooms ? MaxRowIndex + 1 : 0;
703 ConfigureMapGraphLayout();
704
705 if (!bHasRooms)
706 {
707 HideUnusedMapGraphWidgets(0, 0);
708 SetMapStatusText(mMapUnavailableStatusText);
709 SetMapPreviewText(FrontendMapText(TEXT("NoRoomSelected")), mMapUnavailableStatusText, FText::GetEmpty(), MutedColor);
710 SetEnterButtonVisible(false);
711 if (EnterRoomButton != nullptr)
712 {
713 EnterRoomButton->SetIsEnabled(false);
714 }
715 return false;
716 }
717
718 bool bHasSelectedRoom = false;
719 FText SelectedRoomTitle;
720 FText SelectedRoomDescription;
721 FText SelectedRoomState;
722 FSlateColor SelectedRoomStateColor = MutedColor;
723 UWidget* FocusMapNodeWidget = nullptr;
724 UWidget* CurrentRoomNodeWidget = nullptr; // 진입 가능 방이 없을 때(전투 승리 직후 등) 스크롤 폴백 대상.
725 const FVector2D GraphSize = GetMapGraphContentSize();
726
727 TMap<FIntPoint, FVector2D> NodeCenters;
728 for (const FMapRoomView& Room : Rooms)
729 {
730 NodeCenters.Add(FIntPoint(Room.mRow, Room.mColumn), GetMapRoomNodeCenter(Rooms, Room, GraphSize));
731 }
732
733 // 현재 위치 = 방문(Cleared) 경로의 마지막 행. 선택된 경로에서는 행마다 방문 방이 하나뿐이다.
734 FIntPoint CurrentCoord(INDEX_NONE, INDEX_NONE);
735 FIntPoint SelectedCoord(INDEX_NONE, INDEX_NONE);
736 for (const FMapRoomView& Room : Rooms)
737 {
738 if (Room.mState == EMapRoomState::Cleared && (CurrentCoord.X == INDEX_NONE || Room.mRow > CurrentCoord.X))
739 {
740 CurrentCoord = FIntPoint(Room.mRow, Room.mColumn);
741 }
742 if (Room.mSelected)
743 {
744 SelectedCoord = FIntPoint(Room.mRow, Room.mColumn);
745 }
746 }
747
748 int32 UsedLineCount = 0;
749 for (const FMapRoomView& Room : Rooms)
750 {
751 const FVector2D* FromCenter = NodeCenters.Find(FIntPoint(Room.mRow, Room.mColumn));
752 if (FromCenter == nullptr)
753 {
754 continue;
755 }
756
757 for (int32 NextColumn : Room.mNextRoomColumns)
758 {
759 const FMapRoomView* NextRoom = FindMapRoom(Rooms, Room.mRow + 1, NextColumn);
760 const FVector2D* ToCenter = NodeCenters.Find(FIntPoint(Room.mRow + 1, NextColumn));
761 if (NextRoom == nullptr || ToCenter == nullptr)
762 {
763 continue;
764 }
765
766 const FVector2D Delta = *ToCenter - *FromCenter;
767 const float Length = Delta.Size();
768 if (Length <= KINDA_SMALL_NUMBER)
769 {
770 continue;
771 }
772
773 FFrontendMapLinePoolEntry* LineEntry = AcquireMapLineWidget(UsedLineCount++);
774 if (LineEntry == nullptr || LineEntry->mLineWidget == nullptr)
775 {
776 continue;
777 }
778
779 UFrontendMapLineWidget* ConnectionLine = LineEntry->mLineWidget;
780 // 실선=열린 길 / 점선=잠긴 길 — 텍스처/틴트 모두 시안(WBP 클래스 디폴트) 소유.
781 ConnectionLine->SetLineStyle(Room.mState != EMapRoomState::Locked);
782
783 FWidgetTransform LineTransform;
784 LineTransform.Angle = FMath::RadiansToDegrees(FMath::Atan2(Delta.Y, Delta.X));
785 ConnectionLine->SetRenderTransform(LineTransform);
786
787 const float LineThickness = ConnectionLine->GetLineThickness();
788 if (UCanvasPanelSlot* LineSlot = Cast<UCanvasPanelSlot>(ConnectionLine->Slot))
789 {
790 LineSlot->SetSize(FVector2D(Length, LineThickness));
791 LineSlot->SetPosition(*FromCenter - FVector2D(0.f, LineThickness * 0.5f));
792 LineSlot->SetZOrder(0);
793 }
794 }
795 }
796
797 int32 UsedNodeCount = 0;
798 for (const FMapRoomView& Room : Rooms)
799 {
800 const FText StateText = GetMapRoomStateText(Room);
801 const FSlateColor StateColor = Room.mState == EMapRoomState::Locked
802 ? LockedColor
803 : (Room.mState == EMapRoomState::Selected ? TitleColor : ReadyColor);
804
805 const bool bShouldFocusRoom = Room.mCanEnter || Room.mSelected;
806 if (bShouldFocusRoom)
807 {
808 bHasSelectedRoom = true;
809 SelectedRoomTitle = Room.mTitle;
810 SelectedRoomDescription = Room.mDescription;
811 SelectedRoomState = StateText;
812 SelectedRoomStateColor = StateColor;
813 }
814
815 FFrontendMapNodePoolEntry* NodeEntry = AcquireMapNodeWidget(UsedNodeCount++);
816 if (NodeEntry == nullptr || NodeEntry->mNodeWidget == nullptr)
817 {
818 continue;
819 }
820
821 UFrontendMapNodeWidget* NodeWidget = NodeEntry->mNodeWidget;
822 NodeWidget->SetNodeEnabled(Room.mSelectable && IsFrontendMapNavigationEnabled() && !mEnterRequested);
823 NodeWidget->SetNodeVisual(
824 Room.mRow,
825 Room.mColumn,
826 GetMapRoomDebugNodeLabel(Room),
827 FText::GetEmpty(),
828 GetMapRoomPanelColor(Room),
829 GetMapRoomTypeColor(Room.mType),
830 GetMapRoomTextColor(Room),
831 FSlateColor(FLinearColor(0.f, 0.f, 0.f, 0.f)),
832 Room.mType,
833 Room.mState,
834 FIntPoint(Room.mRow, Room.mColumn) == CurrentCoord);
835 if (bShouldFocusRoom)
836 {
837 FocusMapNodeWidget = NodeWidget;
838 }
839 if (FIntPoint(Room.mRow, Room.mColumn) == CurrentCoord)
840 {
841 CurrentRoomNodeWidget = NodeWidget;
842 }
843
844 const FVector2D NodeSize = GetMapNodeSize();
845 const FVector2D Center = NodeCenters.FindChecked(FIntPoint(Room.mRow, Room.mColumn));
846 if (UCanvasPanelSlot* NodeSlot = Cast<UCanvasPanelSlot>(NodeWidget->Slot))
847 {
848 NodeSlot->SetSize(NodeSize);
849 NodeSlot->SetPosition(Center - NodeSize * 0.5f);
850 NodeSlot->SetZOrder(1);
851 }
852 }
853
854 HideUnusedMapGraphWidgets(UsedLineCount, UsedNodeCount);
855 UpdateOverlayMarkers(NodeCenters, CurrentCoord, SelectedCoord);
856
857 if (bHasSelectedRoom)
858 {
859 SetMapPreviewText(SelectedRoomTitle, SelectedRoomDescription, SelectedRoomState, SelectedRoomStateColor);
860 }
861 else
862 {
863 SetMapPreviewText(FrontendMapText(TEXT("NoRoomSelected")), mMapReadyStatusText, FText::GetEmpty(), MutedColor);
864 }
865
866 if (EnterRoomButton != nullptr)
867 {
868 const bool bNavigationEnabled = IsFrontendMapNavigationEnabled();
869 SetEnterButtonVisible(bNavigationEnabled);
870 EnterRoomButton->SetIsEnabled(bNavigationEnabled && bHasSelectedRoom && !mEnterRequested);
871 }
872
873 SetEnterButtonText(mEnterText);
874 SetMapStatusText(mStatusOverrideText.IsEmpty() ? mMapReadyStatusText : mStatusOverrideText);
875 if (bShouldScrollToStart && MapScrollBox != nullptr)
876 {
877 MapScrollBox->ScrollToEnd();
878 }
879 else if (MapScrollBox != nullptr)
880 {
881 // 진입 가능/선택 노드가 없으면(전투 승리 직후 잠금 지도 등) 현재 위치 노드로 스크롤한다 — 최상단에 머무르지 않게.
882 UWidget* ScrollTargetWidget = FocusMapNodeWidget != nullptr ? FocusMapNodeWidget : CurrentRoomNodeWidget;
883 if (ScrollTargetWidget != nullptr)
884 {
885 MapScrollBox->ScrollWidgetIntoView(ScrollTargetWidget, false, EDescendantScrollDestination::Center, 48.f);
886 }
887 }
888 return true;
889}

1181void UFrontendMapWidget::GetNodeAreaLayout(float& OutLeftFrac, float& OutRightFrac, float& OutTopPx, float& OutBottomPx, float InGraphWidth) const
1182{
1183 // 시안 마커: 앵커 X 분수 = 좌우 경계, 오프셋 Top/Bottom = 상하 여백(px). 에디터에서 박스를 옮기면 여기로 반영된다.
1184 OutLeftFrac = MapNodeAreaLeftFracFallback;
1185 OutRightFrac = MapNodeAreaRightFracFallback;
1186 OutTopPx = MapNodeAreaTopPxFallback;
1187 OutBottomPx = MapNodeAreaBottomPxFallback;
1188
1189 if (Map_NodeArea == nullptr)
1190 {
1191 return;
1192 }
1193 const UCanvasPanelSlot* AreaSlot = Cast<UCanvasPanelSlot>(Map_NodeArea->Slot);
1194 if (AreaSlot == nullptr)
1195 {
1196 return;
1197 }
1198 const FAnchors Anchors = AreaSlot->GetAnchors();
1199 if (Anchors.Maximum.X > Anchors.Minimum.X)
1200 {
1201 OutLeftFrac = Anchors.Minimum.X;
1202 OutRightFrac = Anchors.Maximum.X;
1203 }
1204 const FMargin Offsets = AreaSlot->GetOffsets();
1205 OutTopPx = FMath::Max(0.f, Offsets.Top);
1206 OutBottomPx = FMath::Max(0.f, Offsets.Bottom);
1207
1208 if (InGraphWidth <= 64.f)
1209 {
1210 return;
1211 }
1212
1213 float LeftPx = OutLeftFrac * InGraphWidth;
1214 float RightPx = OutRightFrac * InGraphWidth;
1215
1216 const float DecorLeftSafePx = FMath::Clamp(InGraphWidth * MapNodeAreaNarrowLeftSafeFrac, MapNodeAreaMinLeftSafePx, MapNodeAreaMaxLeftSafePx);

1256{
1257 /*

1261 * mPositionOffsetRate는 기계적 배열을 깨는 지터, 마지막에 Map_NodeArea 안으로 Clamp한다.
1262 */
1263 int32 MaxColumn = 0;
1264 for (const FMapRoomView& Candidate : Rooms)
1265 {
1266 MaxColumn = FMath::Max(MaxColumn, Candidate.mColumn);
1267 }
1268
1269 float LeftFrac = 0.f;
1270 float RightFrac = 0.f;
1271 float TopPx = 0.f;
1272 float BottomPx = 0.f;
1273 GetNodeAreaLayout(OUT LeftFrac, OUT RightFrac, OUT TopPx, OUT BottomPx, GraphSize.X);

1275
1276 const float NodeHalfX = NodeSize.X * 0.5f;
1277 const float NodeHalfY = NodeSize.Y * 0.5f;
1278
1279 const float GraphMinX = NodeHalfX;
1280 const float GraphMaxX = FMath::Max(GraphMinX, GraphSize.X - NodeHalfX);

1282 const float AreaMaxX = FMath::Clamp(GraphSize.X * RightFrac - NodeHalfX, GraphMinX, GraphMaxX);
1283 const float ClampMinX = FMath::Min(AreaMinX, AreaMaxX);
1284 const float ClampMaxX = FMath::Max(AreaMinX, AreaMaxX);
1285 const float AreaCenterX = (ClampMinX + ClampMaxX) * 0.5f;

1287 const float ColPitch = MaxColumn <= 0
1288 ? 0.f
1289 : FMath::Min(GetMapColPitchMax(), AvailWidth / StaticCast<float>(MaxColumn));
1290 const float X = AreaCenterX
1291 + (StaticCast<float>(Room.mColumn) - StaticCast<float>(MaxColumn) * 0.5f) * ColPitch;
1292
1293 const float GraphMinY = NodeHalfY;
1294 const float GraphMaxY = FMath::Max(GraphMinY, GraphSize.Y - NodeHalfY);
1295 const float AreaTopY = FMath::Clamp(TopPx + NodeHalfY, GraphMinY, GraphMaxY);
1296 const float AreaBottomY = FMath::Clamp(GraphSize.Y - BottomPx - NodeHalfY, GraphMinY, GraphMaxY);
1297 const float ClampMinY = FMath::Min(AreaTopY, AreaBottomY);
1298 const float ClampMaxY = FMath::Max(AreaTopY, AreaBottomY);
1299
1300 const float Y = ClampMaxY - StaticCast<float>(Room.mRow) * GetMapRowPitch();
1301
1302 const FVector2D Offset(Room.mPositionOffsetRate.X * 18.f, Room.mPositionOffsetRate.Y * 18.f);
1303 return FVector2D(
1304 FMath::Clamp(X + Offset.X, ClampMinX, ClampMaxX),
1305 FMath::Clamp(Y + Offset.Y, ClampMinY, ClampMaxY));
