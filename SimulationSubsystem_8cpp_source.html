54{
55 checkf(mSimulationState == ESRPGSimulationState::RunningGame, TEXT("이미 시뮬레이션 중"));

57
58 USRPGCombatModel* SRPGCombatModel = GetWorldSubsystemModel<USRPGCombatModel>(this);

60 SRPGCombatModel->ForcedAdvanceUntilNextAction(NextCommand, NeedEndCurrentAction);

62 TArray<FSRPGTurnEventLog> ResultLogs = GetEventLogger().PopSRPGLogs();
63
64 SetSimulationState(ESRPGSimulationState::RunningGame);
