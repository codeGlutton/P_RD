55TArray<FSRPGTurnEventLog> USimulationSubsystem::PlaySimulation(FSimulationOption Option)
56{
57 TRACE_CPUPROFILER_EVENT_SCOPE(RDSimulation);
58
59 checkf(mSimulationState == ESRPGSimulationState::RunningGame, TEXT("이미 시뮬레이션 중"));
60 SetSimulationState(ESRPGSimulationState::RunningSimulation);
61
62 USRPGCombatModel* SRPGCombatModel = GetWorldSubsystemModel<USRPGCombatModel>(this);
63
64 /* 전처리 */
65
66 if (Option.mReservedCommand.IsValid() == true)
67 {
68 SRPGCombatModel->ForcedClearActions();
69 }
70
71 /* 옵션 세팅 */
72
73 if (Option.mSkipAIActions == true)
74 {
75 SRPGCombatModel->RequestSkipAIActions();
76 }
77 if (Option.mDuration == ESimulationDurtaion::NextAction)
78 {
79 SRPGCombatModel->RequestAdvanceUntilNextAction();
80 }
81 else
82 {
83 SRPGCombatModel->RequestAdvanceUntilAllPlayerTurn();
84 }
85
86 /* 시뮬 시작 */
87
88 if (Option.mReservedCommand.IsValid() == true)
89 {
90 USRPGCommandRouterModel* CommandRouterModel = GetWorldSubsystemModel<USRPGCommandRouterModel>(this);
91 checkf(CommandRouterModel != nullptr, TEXT("명령 라우터 서브시스템 모델 nullptr"));
92 CommandRouterModel->SummitCommand(Option.mReservedCommand);
93 }
94 else
95 {
96 SRPGCombatModel->ForcedBeginTurn();
97 }
98
99 TArray<FSRPGTurnEventLog> ResultLogs = GetEventLogger().PopSRPGLogs();
100 SetSimulationState(ESRPGSimulationState::RunningGame);
101 return MoveTemp(ResultLogs);
102}
