50 FSlateBrush TextureBrush(UTexture2D* Source, const FLinearColor& Tint = FLinearColor::White)
51 {

140 UWidgetBlueprint* EnsureBlueprint(const TCHAR* AssetPath, const TCHAR* AssetName,

142 {

154 AssetName, PackagePath, UWidgetBlueprint::StaticClass(), Factory));

157 void ResetTree(UWidgetBlueprint* Blueprint, UClass* FinalParent)
158 {
159 Blueprint->Modify();

161 if (UWidget* PreviousRoot = Blueprint->WidgetTree->RootWidget)
162 {
163 Blueprint->ParentClass = UUserWidget::StaticClass();
164 TSet<UWidget*> Widgets;
165 Widgets.Add(PreviousRoot);
166 FWidgetBlueprintEditorUtils::DeleteWidgets(Blueprint, MoveTemp(Widgets),
167 FWidgetBlueprintEditorUtils::EDeleteWidgetWarningType::DeleteSilently);
168 }
169 Blueprint->ParentClass = FinalParent;

528 void Verify()
529 {
530 UWidgetBlueprint* Blueprint = LoadObject<UWidgetBlueprint>(nullptr, MainAssetPath);
531 checkf(Blueprint != nullptr && Blueprint->GeneratedClass != nullptr,
532 TEXT("Landscape world-map WBP is missing"));
533 checkf(Blueprint->GeneratedClass->IsChildOf(UFrontendMapLandscapeWidget::StaticClass()),
534 TEXT("Landscape world-map WBP parent is wrong"));
535 UWidgetBlueprint* LineBlueprint = LoadObject<UWidgetBlueprint>(nullptr, LineAssetPath);
536 UWidgetBlueprint* NodeBlueprint = LoadObject<UWidgetBlueprint>(nullptr, NodeAssetPath);
537 checkf(LineBlueprint != nullptr && LineBlueprint->GeneratedClass != nullptr,
538 TEXT("Landscape world-map line WBP is missing"));
539 checkf(NodeBlueprint != nullptr && NodeBlueprint->GeneratedClass != nullptr,
540 TEXT("Landscape world-map node WBP is missing"));
541 const UFrontendMapLandscapeWidget* Defaults =
542 CastChecked<UFrontendMapLandscapeWidget>(Blueprint->GeneratedClass->GetDefaultObject());
543 checkf(Defaults->GetLandscapeLineWidgetClass().Get() == LineBlueprint->GeneratedClass,
544 TEXT("Main landscape WBP does not serialize the line WBP class"));
545 checkf(Defaults->GetLandscapeNodeWidgetClass().Get() == NodeBlueprint->GeneratedClass,
546 TEXT("Main landscape WBP does not serialize the node WBP class"));
547 const FName Required[] = {
548 TEXT("LandscapeMapBackdrop"), TEXT("MapGraphSize"), TEXT("MapGraphCanvas"),
549 TEXT("Map_ParchmentBody"), TEXT("MapTitleArea"),
550 TEXT("Map_NodeArea"), TEXT("Map_NodeMetrics"), TEXT("MapTitleText"),
551 TEXT("EnterRoomButton"), TEXT("EnterButtonText"), TEXT("CloseButton"),
552 TEXT("Map_CurrentMarker"),
553 TEXT("Map_SelectGlow"), TEXT("Map_LegendGroup")
554 };
555 for (const FName Name : Required)
556 {
557 checkf(Blueprint->WidgetTree->FindWidget(Name) != nullptr,
558 TEXT("Landscape world-map required widget missing: %s"), *Name.ToString());
559 }
560 UE_LOG(LogTemp, Display, TEXT("RD_WORLDMAP_LANDSCAPE_VERIFY success"));
561 }
