86void ACombatDicePreviewActor::ApplyDiceBodyMaterialPreset()
87{
88 if (mDiceMaterial == nullptr)
89 {
90 return;
91 }
92
93 const int32 VariantIndex = mDiceMaterialVariantIndex != INDEX_NONE
94 ? mDiceMaterialVariantIndex
95 : GetDefaultVariantForFaceCount(mCurrentFaceCount);
96 const FDiceBodyMaterialPreset& Preset = GetDiceBodyMaterialPreset(VariantIndex);
97 const float TintWeight = mCurrentFaceCount == 2 ? 0.22f : 0.26f;
98 const FLinearColor BodyColor = FMath::Lerp(Preset.mBaseColor, mDiceTintColor, TintWeight);
99
100 mDiceMaterial->SetVectorParameterValue(TEXT("Color"), BodyColor);
101 // 현재 M_DiceBody가 노출하지 않는 파라미터는 무시된다. 노출되면 각 variant의 질감 차이까지 동작한다.
102 mDiceMaterial->SetScalarParameterValue(TEXT("Metallic"), Preset.mMetallic);
103 mDiceMaterial->SetScalarParameterValue(TEXT("Specular"), Preset.mSpecular);
104 mDiceMaterial->SetScalarParameterValue(TEXT("Roughness"), Preset.mRoughness);
105 mDiceMaterial->SetScalarParameterValue(TEXT("EmissiveStrength"), Preset.mEmissiveStrength);
