14TArray<FActiveTacticalEffectHandle> FSkillEffectLayer_GetDefense::ApplyFactorEffect(IBoardCombatTarget* ActorModel, const UBoardCombatTargetSnapshotData* Snapshot) const
15{
16 const UGameBalanceSettings* GameBalanceSettings = GetDefault<UGameBalanceSettings>();
17 checkf(GameBalanceSettings != nullptr, TEXT("게임 밸런스 세팅 nullptr"));
18
19 UAttributeSetComponentModel* AttributeSetComponentModel = ActorModel->GetAttributeComponentModel();
20 checkf(AttributeSetComponentModel != nullptr, TEXT("속성 컴포넌트 nullptr"));
21
22 UTacticalEffectContext* EffectContext = AttributeSetComponentModel->MakeEffectContext();
23
24 TArray<FActiveTacticalEffectHandle> EffectHandles;
25
26 /* 포인트를 Factor에 임시 추가 */
27 {
28 const float StatusDefenseUpRatio = GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_Infinite_Buff_Dexterity];
29 const float StatusDefenseDownRatio = GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_Infinite_Debuff_Dexterity];
30
31 int32 StatusDefense = (
32 Snapshot->mEffectCounts.FindRef(EffectTags::GameplayEffect_StatusEffect_Infinite_Buff_Dexterity, 0) * StatusDefenseUpRatio -
33 Snapshot->mEffectCounts.FindRef(EffectTags::GameplayEffect_StatusEffect_Infinite_Debuff_Dexterity, 0) * StatusDefenseDownRatio
34 );
35
36 TSharedPtr<FTacticalEffectSpec> EffectSpec = AttributeSetComponentModel->MakeOutgoingSpec(UTacticalEffect_DefenseFactor_AddBase::StaticClass(), EffectContext);
37 EffectSpec->mDynamicMagnitude = mDefenseGain + StatusDefense;
38 EffectHandles.Add(AttributeSetComponentModel->ApplyTacticalEffectSpecToSelf(*EffectSpec));
39 }
40
41 return EffectHandles;
42}

56void FSkillEffectLayer_GetDefense::CommitEffect(const FSkillEffectCommitParams& Params) const
57{
58 UAttributeSetComponentModel* AttributeSetComponentModel = Params.mInstigator->GetAttributeComponentModel();
59 checkf(AttributeSetComponentModel != nullptr, TEXT("속성 컴포넌트 nullptr"));
60
61 UTacticalEffectContext* EffectContext = AttributeSetComponentModel->MakeEffectContext();
62
63 /* 방어력 증가 적용 */
64 const int32 TargetNum = Params.mTargets.Num();
65 for (int32 i = 0; i < TargetNum; ++i)
66 {
67 const TScriptInterface<IBoardCombatTarget>& OtherCombatTarget = Params.mTargets[i];
68 UAttributeSetComponentModel* OtherAttributeSetComponentModel = OtherCombatTarget->GetAttributeComponentModel();
69 checkf(OtherAttributeSetComponentModel != nullptr, TEXT("속성 컴포넌트 nullptr"));
70
71 TSharedPtr<FTacticalEffectSpec> EffectSpec = AttributeSetComponentModel->MakeOutgoingSpec(UTacticalEffect_GetDefense::StaticClass(), EffectContext);
72 EffectSpec->SetInstigatorSnapshotData(Params.mInstigatorSnapshot);
73 EffectSpec->SetTargetSnapshotData(Params.mTargetSnapshots[i]);
74 AttributeSetComponentModel->ApplyTacticalEffectSpecToTarget(*EffectSpec, OtherAttributeSetComponentModel);
75 }
76}
