47void UMockShopDriver::PushShop()

54 FShopUI Shop;

56 for (int32 i = 0; i < GMockEntryCount; ++i)

58 FShopItemUI Item;

60 Item.mKind = GMockEntries[i].Kind;
61 Item.mName = FText::FromString(GMockEntries[i].Name);
62 Item.mPrice = GMockEntries[i].Price;
63 Item.mIsSoldOut = mSoldOut.IsValidIndex(i) && mSoldOut[i];
64 Item.mIsAffordable = !Item.mIsSoldOut && mGold >= Item.mPrice;
65 Shop.mItems.Add(Item);

71void UMockShopDriver::HandleBuy(int32 SlotIndex)
72{
73 if (SlotIndex < 0 || SlotIndex >= GMockEntryCount)
74 {
75 return;
76 }

78 {
79 return;
80 }
81 const int32 Price = GMockEntries[SlotIndex].Price;
82 if (mGold < Price)
83 {
84 UE_LOG(LogRD, Display, TEXT("MockShopDriver: not enough gold for slot %d"), SlotIndex);
85 return;
86 }
87 mGold -= Price;

89 UE_LOG(LogRD, Display, TEXT("MockShopDriver: bought slot %d, gold=%d"), SlotIndex, mGold);
