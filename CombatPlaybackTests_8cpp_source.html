49IMPLEMENT_SIMPLE_AUTOMATION_TEST(FCombatPlaybackMixerTest, "P_RD.Combat.Playback.MixingAndLifecycle", EAutomationTestFlags::EditorContext | EAutomationTestFlags::EngineFilter)
50bool FCombatPlaybackMixerTest::RunTest(const FString&)
51{
52 auto* World = GEditor->GetEditorWorldContext().World();
53 auto* Actor = World->SpawnActor<AActor>();
54 auto* Mixer = NewObject<UTimeScaleComponent>(Actor);
55 Mixer->RegisterComponent();
56 const float Original = World->GetWorldSettings()->TimeDilation;
57 auto Playback = Mixer->RequestTimeScale(Actor, 8.f);
58 Mixer->SetTimeScaleImmediately(Playback, 8.f);
59 TestEqual(TEXT("8x playback"), World->GetWorldSettings()->TimeDilation, 8.f);
60 auto Slow = Mixer->RequestTimeScale(Actor, .2f);
61 Mixer->TickComponent(.01f, LEVELTICK_All, nullptr);
62 TestTrue(TEXT("Cinematic and playback multiply"), FMath::IsNearlyEqual(World->GetWorldSettings()->TimeDilation, 1.6f, 0.0001f));
63 Mixer->SetTimeScaleImmediately(Playback, 2.f);
64 TestTrue(TEXT("Changing playback preserves cinematic"), FMath::IsNearlyEqual(World->GetWorldSettings()->TimeDilation, .4f, 0.0001f));
65 Mixer->ReleaseTimeScale(Slow); Mixer->TickComponent(.01f, LEVELTICK_All, nullptr);
66 TestEqual(TEXT("Ending cinematic preserves selected playback"), World->GetWorldSettings()->TimeDilation, 2.f);
67 auto Other = Mixer->RequestTimeScale(Actor, .5f);
68 Mixer->TickComponent(.01f, LEVELTICK_All, nullptr);
69 Mixer->ReleaseTimeScaleImmediately(Playback);
70 TestEqual(TEXT("Leaving combat removes only playback request"), World->GetWorldSettings()->TimeDilation, .5f);
71 Mixer->ReleaseTimeScaleImmediately(Other);
72 TestEqual(TEXT("No leftover speed"), World->GetWorldSettings()->TimeDilation, 1.f);
73 Mixer->DestroyComponent();
74 World->GetWorldSettings()->TimeDilation = Original; Actor->Destroy();
75 return !HasAnyErrors();
76}

140IMPLEMENT_SIMPLE_AUTOMATION_TEST(FCombatPlaybackLifecycleTest, "P_RD.Combat.Playback.ControllerLifecycle", EAutomationTestFlags::EditorContext | EAutomationTestFlags::EngineFilter)
141bool FCombatPlaybackLifecycleTest::RunTest(const FString&)
142{
143 auto* World = GEditor->GetEditorWorldContext().World();
144 auto* Controller = World->SpawnActor<APlayerController>();
145 auto* Camera = World->SpawnActor<ACombatCameraPawn>();
146 World->AddController(Controller);
147 Controller->Possess(Camera);
148 auto* Owner = World->SpawnActor<AActor>();
149 auto* Playback = NewObject<UCombatPlaybackComponent>(Owner);
150 Playback->RegisterComponent();
151 Playback->BeginPlay();
152 auto* Model = NewObject<UCombatUIModel>(Owner);
153 Playback->StartPlayback(Model);
154 TestTrue(TEXT("Playback connects to current camera"), Model->IsPlaybackSpeedAvailable());
155 for (int32 Expected : {2,3,4,5,6,7,8,1,2})

157 Model->RequestCyclePlaybackSpeed();

159 TestEqual(TEXT("Model reflects applied selection"), Model->GetPlaybackSpeed(), Expected);
160 }
161 for(int32 I=0; I<20; ++I) Playback->TickComponent(.016f,LEVELTICK_All,nullptr);
162 TestEqual(TEXT("Ticks never stack the persistent multiplier"), World->GetWorldSettings()->TimeDilation,2.f);
163 Playback->StopPlayback();
164 TestEqual(TEXT("Combat completion resets playback"), World->GetWorldSettings()->TimeDilation,1.f);
165 TestFalse(TEXT("Result HUD cannot change speed"), Model->IsPlaybackSpeedAvailable());
166 Model->RequestCyclePlaybackSpeed();
167 TestEqual(TEXT("Stopped model has no active callback"), World->GetWorldSettings()->TimeDilation,1.f);
168 Playback->StartPlayback(Model);
169 TestEqual(TEXT("Next combat restores selected speed"), World->GetWorldSettings()->TimeDilation,2.f);
170 Playback->EndPlay(EEndPlayReason::LevelTransition);
171 TestEqual(TEXT("Abandon or level transition cleans up playback"), World->GetWorldSettings()->TimeDilation,1.f);
172 Playback->DestroyComponent(); Owner->Destroy(); Controller->UnPossess(); Camera->Destroy(); Controller->Destroy();
173 return !HasAnyErrors();
174}
