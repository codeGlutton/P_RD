15 bool ShouldShowPip(TCHAR PipName, int32 ResultValue)
16 {
17 switch (ResultValue)
18 {
19 case 1:
20 return PipName == TCHAR('C');
21 case 2:
22 return PipName == TCHAR('A') || PipName == TCHAR('E');
23 case 3:
24 return PipName == TCHAR('A') || PipName == TCHAR('C') || PipName == TCHAR('E');
25 case 4:
26 return PipName == TCHAR('A') || PipName == TCHAR('B') || PipName == TCHAR('D') || PipName == TCHAR('E');
27 case 5:
28 return PipName == TCHAR('A') || PipName == TCHAR('B') || PipName == TCHAR('C') || PipName == TCHAR('D') || PipName == TCHAR('E');
29 default:
30 return PipName == TCHAR('A') || PipName == TCHAR('B') || PipName == TCHAR('D') || PipName == TCHAR('E');

156UBorder* UDicePanelWidget::EnsureRuntimeDicePip(int32 DiceIndex, TCHAR PipName, const FVector2D& Position) const
157{
158 const FString PipWidgetName = FString::Printf(TEXT("DicePip%c_%d"), PipName, DiceIndex);
159 if (UBorder* ExistingPip = FindDiceBorder(PipWidgetName))

174
175 UBorder* NewPip = WidgetTree->ConstructWidget<UBorder>(UBorder::StaticClass(), FName(*PipWidgetName));
176 if (NewPip == nullptr)

178 return nullptr;
