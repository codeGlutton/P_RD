30bool UBossCollapseWidget::LoadStage(int32 InStage)
31{
32 if (!ShouldPlay(true, true, InStage)) return false;
33 const FString Base = TEXT("/Game/SVN/OutSideAsset/AICreation/UI/StageVictory/BossCollapse/");
34 Atlas = LoadObject<UTexture2D>(nullptr, *FString::Printf(TEXT("%sT_BossCollapse%d.T_BossCollapse%d"), *Base, InStage, InStage));
35 Sound = LoadObject<USoundBase>(nullptr, *FString::Printf(TEXT("%sS_BossCollapse%d.S_BossCollapse%d"), *Base, InStage, InStage));
36 if (!Atlas || !Sound) return false;
37 StageNumber = InStage;
38 Panels.Reset();
39 // Match the five-pixel separator inset in the approved preview, including odd atlas sizes.
40 // GetSizeX/Y can report a 32x32 compilation placeholder on a cold editor/game launch.
41 // ImportedSize is stable both before resource compilation and in cooked builds.
42 const FIntPoint ImportedSize = Atlas->GetImportedSize();
43 if (ImportedSize.X <= 20 || ImportedSize.Y <= 20) return false;
44 const float W = ImportedSize.X, H = ImportedSize.Y;
45 for (int32 I = 0; I < 4; ++I)
46 {
47 FSlateBrush Brush;
48 Brush.SetResourceObject(Atlas);
49 Brush.DrawAs = ESlateBrushDrawType::Image;
50 Brush.ImageSize = FVector2D(1280,720);
51 const FVector2f Min((I%2*FMath::FloorToInt(W/2)+5)/W, (I/2*FMath::FloorToInt(H/2)+5)/H);
52 Brush.SetUVRegion(FBox2f(Min, Min+FVector2f((FMath::FloorToInt(W/2)-10)/W,(FMath::FloorToInt(H/2)-10)/H)));
53 Panels.Add(Brush);
54 }
55 return true;
56}

138 const auto& T = Timings[StageNumber-1];
139 const FVector2D View = G.GetLocalSize();
140 const float Scale = FMath::Max(View.X/1280.f,View.Y/720.f);
141 if (Scale <= 0.f) return Layer;
142 FVector2D Shake = FVector2D::ZeroVector;
143 auto AddImpact = [&](float At, float Amplitude)
144 {
145 if (Elapsed < At) return;
146 const float Dt=Elapsed-At, Env=FMath::Exp(-15.f*Dt);
147 Shake += FVector2D(.65f*Amplitude*Env*FMath::Sin(99.f*Dt),Amplitude*Env*FMath::Cos(87.f*Dt));
148 };
149 if (StageNumber==1) { AddImpact(.42f,6); AddImpact(T.Land,17); }
150 if (StageNumber==2) { AddImpact(.10f,9); AddImpact(.62f,5); AddImpact(T.Land,22); }
151 if (StageNumber==3) { AddImpact(1.12f,3); AddImpact(T.Land,8); }
152 const float SceneScale = Scale*1.05f;
153 const FVector2D Origin = (View-FVector2D(1280,720)*SceneScale)*.5f-Shake*Scale;
154 const FSlateBrush* White = FCoreStyle::Get().GetBrush(TEXT("WhiteBrush"));
155 Out.PushClip(FSlateClippingZone(G));

161 const FVector2D Inset = Border ? FVector2D(9,9) : FVector2D::ZeroVector;
162 FSlateBrush Brush = Panels[Index];
163 // Crop to the panel's aspect ratio rather than stretching the boss.
164 const FVector2D Inner = Size-Inset*2;
165 FBox2f UV=Brush.GetUVRegion();
166 const float Ratio=Inner.X/Inner.Y;
167 if (Ratio>1280.f/720.f) { const float Remove=(UV.Max.Y-UV.Min.Y)*(1.f-(1280.f/720.f)/Ratio)*.5f; UV.Min.Y+=Remove;UV.Max.Y-=Remove; }
168 else { const float Remove=(UV.Max.X-UV.Min.X)*(1.f-Ratio/(1280.f/720.f))*.5f; UV.Min.X+=Remove;UV.Max.X-=Remove; }
169 Brush.SetUVRegion(UV);
170 FSlateDrawElement::MakeRotatedBox(Out,Layer++,G.ToPaintGeometry(Inner*SceneScale,FSlateLayoutTransform(Origin+(Pos+Inset)*SceneScale)),&Brush,ESlateDrawEffect::None,Angle,TOptional<FVector2D>(),FSlateDrawElement::RelativeToElement,FLinearColor::White);
171 };
172 Draw(0,{0,0},{1280,720},0,false);
173 if (Elapsed>=T.Detail && Elapsed<T.Land)
174 {
175 if (StageNumber==1) Draw(1,{195,195-850*Arrival(Elapsed,T.Detail,.08f)},{1110,570},-.055f,true);
176 if (StageNumber==2) Draw(1,{200+1450*Arrival(Elapsed,T.Detail,.10f),20},{1090,595},.045f,true);
177 if (StageNumber==3) Draw(1,{175+1450*Arrival(Elapsed,T.Detail,.10f),150},{1130,615},-.075f,true);
178 }
179 if (Elapsed>=T.LandStart && Elapsed<T.Rest)

182 Out.PopClip();
