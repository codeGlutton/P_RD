212bool ARDGameModeBase::PreloadAndTransitionRoomAsync(int32 RoomRowIndex, int32 RoomColumnIndex)
213{
214 if (mWasNextRoomPreloadRequested == true)
215 {
216 UE_LOG(LogRDGameMode, Log, TEXT("다음 방에 대한 Preload 중복 요청"));
217 return false;
218 }
219
220 URoomTransitionSubsystem* RoomTransitionSubsystem = GetGameInstance()->GetSubsystem<URoomTransitionSubsystem>();
221 checkf(RoomTransitionSubsystem != nullptr, TEXT("방 전환 서브시스템 nullptr 오류"));
222
223 mWasNextRoomPreloadRequested = true;
224 UE_LOG(LogRDGameMode, Log, TEXT("다음 방 Preload 중"));
225
226 const bool RequireExternalReady = mShowFadeOutUIOnTransition || mWaitExternalWorkOnTransition;
227 const bool AutoTransition = mShowLoadingNotifyUIOnTransition == false;
228 checkf(
229 RoomTransitionSubsystem->PreloadRoomAsync(
230 RoomRowIndex,
231 RoomColumnIndex,
232 FOnReadyToTransition::CreateUObject(this, &ARDGameModeBase::OnReadyToTransition),
233 FOnPreTransitNextRoom::CreateUObject(this, &ARDGameModeBase::OnPreTransition),
234 RequireExternalReady,
235 AutoTransition
236 ) == true,
237 TEXT("다음 방 Preload 오류")
238 );
239
240 if (RequireExternalReady == true)
241 {
242 if (mShowFadeOutUIOnTransition == true)
243 {
244 /* 페이드 아웃 애니메이션 실행 */
245
246 /*
247 * 호출부마다 "페이드아웃이 끝나면 MarkExternalReadyForTransition()" 콜백을 반복해서 만들지 않는다.
248 * 모든 방 전환 경로가 같은 후속 작업을 하므로, 그 콜백은 전환 전용 함수에 묶었다.
249 * 여기서 말하는 같은 후속 작업은 "페이드아웃 완료 -> 로딩 알림 표시 -> ExternalReady 전달 ->
250 * 준비 완료 콜백에서 로딩 알림 종료 -> TransitLoadedRoom()" 순서다.
251 * 일반 방, 새 스테이지 첫 방, 프론트엔드 방 전환 모두 이 순서를 공유한다.
252 * 여기서는 "방 전환용 페이드아웃을 시작한다"는 요청만 남긴다.
253 */
254 StartFadeOutUIForRoomTransition();
255 }
256 }
257 else if (mShowLoadingNotifyUIOnTransition == true)
258 {
259 OpenLoadingNotifyUI();
260 }
261 return true;
