203{
204 checkf(mWorldWidgets.Contains(EWorldWidgetType::FadeInOut) == true, TEXT("FadeInOut이 준비되지 않음"));
205
206 UWorldWidgetSubsystem* WorldWidgetSubsystem = GetWorld()->GetSubsystem<UWorldWidgetSubsystem>();
207 checkf(WorldWidgetSubsystem != nullptr, TEXT("월드 위젯 서브시스템 nullptr 오류"));
208
209 UFadeInOutWidget* FadeInOutWidget = WorldWidgetSubsystem->GetWorldWidget<UFadeInOutWidget>(EWorldWidgetType::FadeInOut);
210 checkf(FadeInOutWidget != nullptr, TEXT("페이드 인 앤 아웃 위젯 nullptr 오류"));
211
212 auto TryToExecuteFadeOutCallback = [Callback = MoveTemp(OnEndFadeOutAnimation), FadeInOutWidget](const ARDGameModeBase* GameMode) {
213 if (EnumHasAllFlags(GameMode->mFadeOutStateFlag, ERDFadeOutStateFlag::ReadyToCallback) == false)
214 {

216 }
217 GameMode->mFadeOutStateFlag = ERDFadeOutStateFlag::None;
218 Callback.ExecuteIfBound(FadeInOutWidget);
219 };
220
221 FadeInOutWidget->OpenUI();
222 FadeInOutWidget->StartFadeOut(FOnEndFadeOutAnimation::CreateWeakLambda(this, [this, TryToExecuteFadeOutCallback](UFadeInOutWidget* Widget) {
223 EnumAddFlags(mFadeOutStateFlag, ERDFadeOutStateFlag::FadeAnimationEnded);
224 TryToExecuteFadeOutCallback(this);
225 }));
226
227 if (mBgmComponent != nullptr && mBgmComponent->IsPlaying() == true)
228 {
229 mBgmComponent->OnAudioFinishedNative.AddWeakLambda(this, [this, TryToExecuteFadeOutCallback](UAudioComponent* AudioComponent) {
230 EnumAddFlags(mFadeOutStateFlag, ERDFadeOutStateFlag::FadeBGMEnded);
231 TryToExecuteFadeOutCallback(this);
232 });
233 mBgmComponent->FadeOut(mBGMFadeOutDuration, 0.f);
234 }
235 else

237 EnumAddFlags(mFadeOutStateFlag, ERDFadeOutStateFlag::FadeBGMEnded);

336{
337 if (mWasNextRoomPreloadRequested == true)
338 {
339 UE_LOG(LogRDGameMode, Log, TEXT("다음 방에 대한 Preload 중복 요청"));

343 URoomTransitionSubsystem* RoomTransitionSubsystem = GetGameInstance()->GetSubsystem<URoomTransitionSubsystem>();
344 checkf(RoomTransitionSubsystem != nullptr, TEXT("방 전환 서브시스템 nullptr 오류"));
345
346 mWasNextRoomPreloadRequested = true;
347 UE_LOG(LogRDGameMode, Log, TEXT("다음 방 Preload 중"));
348
349 const bool RequireExternalReady = mShowFadeOutUIOnTransition || mWaitExternalWorkOnTransition;
350 const bool AutoTransition = mShowLoadingNotifyUIOnTransition == false;
351 checkf(

355 FOnReadyToTransition::CreateUObject(this, &ARDGameModeBase::OnReadyToTransition),

357 RequireExternalReady,
358 AutoTransition
359 ) == true,
360 TEXT("다음 방 Preload 오류")
361 );
362
363 if (RequireExternalReady == true)
364 {
365 if (mShowFadeOutUIOnTransition == true)
366 {
367 /* 페이드 아웃 애니메이션 실행 */
368
369 /*
370 * 호출부마다 "페이드아웃이 끝나면 MarkExternalReadyForTransition()" 콜백을 반복해서 만들지 않는다.
371 * 모든 방 전환 경로가 같은 후속 작업을 하므로, 그 콜백은 전환 전용 함수에 묶었다.
372 * 여기서 말하는 같은 후속 작업은 "페이드아웃 완료 -> 로딩 알림 표시 -> ExternalReady 전달 ->
373 * 준비 완료 콜백에서 로딩 알림 종료 -> TransitLoadedRoom()" 순서다.
374 * 일반 방, 새 스테이지 첫 방, 프론트엔드 방 전환 모두 이 순서를 공유한다.
375 * 여기서는 "방 전환용 페이드아웃을 시작한다"는 요청만 남긴다.
376 */
377 StartFadeOutUIForRoomTransition();
378 }
379 }
380 else if (mShowLoadingNotifyUIOnTransition == true)
381 {
382 OpenLoadingNotifyUI();
383 }
384 return true;
