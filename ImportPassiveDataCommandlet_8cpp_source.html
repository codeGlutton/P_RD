55int32 UImportPassiveDataCommandlet::Main(const FString& Params)
56{
57 using namespace ImportPassiveDataPrivate;
58 // -csv 는 필수, -ids 는 선택
59 FString CsvPath;
60 if (FParse::Value(*Params, TEXT("csv="), CsvPath) == false)
61 {
62 UE_LOG(LogDataAssetImport, Error, TEXT("사용법: -run=ImportPassiveData -csv=<경로> [-ids=P001,P015]"));
63 return 1;
64 }
65 FString IdsText;
66 TArray<FString> OnlyIds;
67 if (FParse::Value(*Params, TEXT("ids="), IdsText))
68 {
69 IdsText.ParseIntoArray(OnlyIds, TEXT(","));
70 }
71
72 // CSV 읽기
73 FImportCsvTable Table;
74 if (DataAssetImportUtils::LoadCsv(CsvPath, Table) == false)
75 {
76 return 1;
77 }
78
79 // 필수 컬럼이 하나라도 없으면 이름을 찍고 중단. 생성 제외 체크박스는 없어도 됨
80 const TCHAR* RequiredColumns[] = {
81 Column::Title, Column::Id,
82 Column::ActivateTiming, Column::DeactivateTiming, Column::CounterResetTiming, Column::CaptureTiming,
83 Column::EffectTarget, Column::Quantifier, Column::EffectClass, Column::MagnitudeKind, Column::MagnitudeText,
84 Column::ConditionOp, Column::ConditionLhsKind, Column::ConditionLhsText, Column::ConditionRhsKind, Column::ConditionRhsText,
85 Column::CaptureKind, Column::CaptureKey, Column::CaptureText,
86 };
87 bool bColumnsOk = true;
88 for (const TCHAR* Name : RequiredColumns)
89 {
90 if (Table.HasColumn(Name) == false)
91 {
92 UE_LOG(LogDataAssetImport, Error, TEXT("CSV에 컬럼이 없음: %s"), Name);
93 bColumnsOk = false;
94 }
95 }
96 if (bColumnsOk == false)
97 {
98 return 1;
99 }
100
101 // 행마다 DA 생성 또는 덮어쓰기
102 int32 NumCreated = 0;
103 int32 NumUpdated = 0;
104 int32 NumSkipped = 0;
105 int32 NumFailed = 0;
106 TSet<FString> SeenIds;
107 for (int32 Row = 0; Row < Table.mRows.Num(); ++Row)
108 {
109 const FString Id = Table.Get(Row, Column::Id).TrimStartAndEnd();
110 if (Id.IsEmpty() || (OnlyIds.Num() > 0 && OnlyIds.Contains(Id) == false))
111 {
112 continue;
113 }
114 SeenIds.Add(Id);
115
116 // 생성 제외 체크박스가 켜져 있으면 건너뜀
117 if (DataAssetImportUtils::IsChecked(Table.Get(Row, Column::Exclude)))
118 {
119 UE_LOG(LogDataAssetImport, Display, TEXT("%s: 생성 제외"), *Id);
120 ++NumSkipped;
121 continue;
122 }
123
124 // DA를 만들거나 로드해서 채우고, 검증한 뒤 저장
125 bool bCreated = false;
126 UStaticPassiveData* Data = DataAssetImportUtils::FindOrCreateAsset<UStaticPassiveData>(PackagePath, AssetPrefix + Id, bCreated);
127 if (Data == nullptr || FillPassiveData(Table, Row, Data) == false)
128 {
129 UE_LOG(LogDataAssetImport, Error, TEXT("%s: 생성 실패"), *Id);
130 ++NumFailed;
131 continue;
132 }
133 DataAssetImportUtils::ValidateAsset(Data);
134 if (DataAssetImportUtils::SaveAsset(Data) == false)
135 {
136 ++NumFailed;
137 continue;
138 }
139 bCreated ? ++NumCreated : ++NumUpdated;
140 }
141
142 // -ids 에 있는데 CSV에 없는 ID 경고
143 for (const FString& Id : OnlyIds)
144 {
145 if (SeenIds.Contains(Id) == false)
146 {
147 UE_LOG(LogDataAssetImport, Warning, TEXT("CSV에 없는 ID: %s"), *Id);
148 }
149 }
150
151 UE_LOG(LogDataAssetImport, Display, TEXT("패시브 DA 임포트 끝: 생성 %d, 갱신 %d, 제외 %d, 실패 %d"), NumCreated, NumUpdated, NumSkipped, NumFailed);
152 return (NumFailed == 0) ? 0 : 1;
153}

155bool UImportPassiveDataCommandlet::FillPassiveData(const FImportCsvTable& Table, int32 RowIndex, UStaticPassiveData* Data) const
156{
157 using namespace ImportPassiveDataPrivate;
158 const FString Id = Table.Get(RowIndex, Column::Id).TrimStartAndEnd();
159
160 // 설명: 제목에서 앞의 ID를 뗌
161 // 예) "P001 전투 시작 시, ..." -> "전투 시작 시, ..."
162 FString Title = Table.Get(RowIndex, Column::Title).TrimStartAndEnd();
163 if (Title.StartsWith(Id))
164 {
165 Title = Title.RightChop(Id.Len()).TrimStart();
166 }
167 Data->mDescription = FText::FromString(Title);
168
169 // 시점 4종. 비어 있으면 빈 태그
170 bool bOk = true;
171 bOk &= DataAssetImportUtils::ParseTimingTag(Table.Get(RowIndex, Column::ActivateTiming), Data->mActivateTimingTag);
172 bOk &= DataAssetImportUtils::ParseTimingTag(Table.Get(RowIndex, Column::DeactivateTiming), Data->mDeactivateTimingTag);
173 bOk &= DataAssetImportUtils::ParseTimingTag(Table.Get(RowIndex, Column::CounterResetTiming), Data->mCounterResetTimingTag);
174 bOk &= DataAssetImportUtils::ParseTimingTag(Table.Get(RowIndex, Column::CaptureTiming), Data->mCaptureTimingTag);
175
176 // 효과 대상(자신/대상)과 수량 조건(Any/All)
177 Data->mEffectTarget = (Table.Get(RowIndex, Column::EffectTarget).TrimStartAndEnd() == TEXT("대상")) ? EPassiveEffectTarget::Targets : EPassiveEffectTarget::Self;
178 Data->mTargetQuantifier = (Table.Get(RowIndex, Column::Quantifier).TrimStartAndEnd() == TEXT("All")) ? EPassiveTargetQuantifier::All : EPassiveTargetQuantifier::Any;
179
180 // 효과: 이펙트 클래스는 여러 개 추가 가능. 단, 수치는 하나만 있으므로 동일한 수치가 들어감
181 Data->mEffects.Reset();
182 FPassiveOperand Magnitude;
183 bOk &= DataAssetImportUtils::ParseOperand(Table.Get(RowIndex, Column::MagnitudeKind), Table.Get(RowIndex, Column::MagnitudeText), Magnitude);
184 TArray<FString> EffectNames;
185 Table.Get(RowIndex, Column::EffectClass).ParseIntoArray(EffectNames, TEXT(","));
186 for (const FString& EffectName : EffectNames)
187 {
188 UClass* EffectClass = DataAssetImportUtils::FindEffectClass(EffectName);
189 if (EffectClass == nullptr)
190 {
191 bOk = false;
192 continue;
193 }
194 FPassiveEffectEntry Entry;
195 Entry.mEffectClass = EffectClass;
196 Entry.mMagnitude = Magnitude;
197 Data->mEffects.Add(Entry);
198 }
199
200 // 조건: 연산자 칸이 비어 있으면 조건 없음. 있으면 좌변/연산자/우변 하나
201 Data->mConditions.Reset();
202 const FString OpText = Table.Get(RowIndex, Column::ConditionOp).TrimStartAndEnd();
203 if (OpText.IsEmpty() == false)
204 {
205 FPassiveCondition Condition;
206 bOk &= DataAssetImportUtils::ParseOperand(Table.Get(RowIndex, Column::ConditionLhsKind), Table.Get(RowIndex, Column::ConditionLhsText), Condition.mLhs);
207 bOk &= DataAssetImportUtils::ParseCompareOp(OpText, Condition.mOp);
208 bOk &= DataAssetImportUtils::ParseOperand(Table.Get(RowIndex, Column::ConditionRhsKind), Table.Get(RowIndex, Column::ConditionRhsText), Condition.mRhs);
209 Data->mConditions.Add(Condition);
210 }
211
212 // 캡처: 종류 칸이 비어 있으면 캡처 없음. 있으면 키 + 피연산자 하나
213 Data->mCaptureOperands.Reset();
214 const FString CaptureKind = Table.Get(RowIndex, Column::CaptureKind).TrimStartAndEnd();
215 if (CaptureKind.IsEmpty() == false)
216 {
217 FPassiveCaptureEntry Capture;
218 Capture.mKey = FName(*Table.Get(RowIndex, Column::CaptureKey).TrimStartAndEnd());
219 bOk &= DataAssetImportUtils::ParseOperand(CaptureKind, Table.Get(RowIndex, Column::CaptureText), Capture.mOperand);
220 Data->mCaptureOperands.Add(Capture);
221 }
222
223 // 캡처한 타겟에게 발동: 문서에 이 옵션 컬럼이 없으므로 시점을 조합해서 판단
224 // 발동이 OnEndUsingSkill 이고 캡처가 OnStartApplyingEffect 이면 대상을 캡처했다가 스킬 끝에 판정하므로 켬
225 const FGameplayTag OnEndUsingSkill = FGameplayTag::RequestGameplayTag(FName(TEXT("GameplayAbility.Passive.OnEndUsingSkill")));
226 const FGameplayTag OnStartApplyingEffect = FGameplayTag::RequestGameplayTag(FName(TEXT("GameplayAbility.Passive.OnStartApplyingEffect")));
227 Data->mActivateOnCapturedTargets = (Data->mActivateTimingTag == OnEndUsingSkill && Data->mCaptureTimingTag == OnStartApplyingEffect);
228
229 return bOk;
230}
