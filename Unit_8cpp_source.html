185void AUnit::OnStartMoveStep(const FTileTransform& NextTileTransform, const FTransform& TargetWorldTransform, TSharedPtr<FPresentationBarrier> Barrier, float RemainingPathDistance)
186{
187 // 목표타일의 월드트랜스폼과 배리어 보관
188 mMoveTargetTransform = TargetWorldTransform;

190 // 목표 이후 남은 거리 저장 (제동거리 계산에 사용)
191 mRemainingPathDistance = RemainingPathDistance;
192 // @note 현재 속도(mCurrentMoveSpeed)는 초기화하지 않음 (중간에 멈칫하지 않게)

196 if (mCapsuleComp != nullptr)
197 {
198 FVector TargetLocation = mMoveTargetTransform.GetLocation();
199 TargetLocation.Z += mCapsuleComp->GetScaledCapsuleHalfHeight();
200 mMoveTargetTransform.SetLocation(TargetLocation);
201 }
202
203 // 루트가 Movable이 아니면 SetActorLocation이 실패하므로 보정 (BP 설정 실수 방어)
204 if (RootComponent != nullptr && RootComponent->Mobility != EComponentMobility::Movable)

207 }

229void AUnit::SpawnHitVFX(const FApplyNiagaraSpawnData& NiagaraSpawnData, ETileActorDirection LocalDirection) const
230{
231 if (NiagaraSpawnData.mNiagaraSystem == nullptr)
232 {
233 return;
234 }
235
236 // 소켓의 컴포넌트 기준 Transform
237 const FTransform SocketLocalTransform = GetMesh()->GetSocketTransform(NiagaraSpawnData.mSocketName, RTS_Component);
238
239 // 추가 방향 절대 회전 Transform

245 // 월드 트랜스폼으로 변환

251 NiagaraSpawnData.mNiagaraSystem,
252 GetMesh(),
253 NAME_None,
254 FinalLocalTransform.GetLocation(),
255 FinalLocalTransform.Rotator(),
256 FinalLocalTransform.GetScale3D(),
257 EAttachLocation::KeepRelativeOffset,
258 true,
259 ENCPoolMethod::None
260 );
261 }
262 else
263 {
264 UNiagaraFunctionLibrary::SpawnSystemAtLocation(
265 GetWorld(),
266 NiagaraSpawnData.mNiagaraSystem,
267 FinalWorldTransform.GetLocation(),
268 FinalWorldTransform.Rotator(),
269 FinalWorldTransform.GetScale3D()

274void AUnit::Tick(float DeltaSeconds)
275{
276 Super::Tick(DeltaSeconds);
277
278 // 이동스텝 연출 중이 아니면 틱 끔
279 if (mMoveBarrier.IsValid() == false)
280 {
281 SetActorTickEnabled(false);
282 return;
283 }
284
285 // 목표 속도 결정
286 // - 기본은 최대속도
287 // - 제동거리에 들어가면 감속하면서 느려짐
288 const FVector CurrentLocation = GetActorLocation();
289 const float RemainingDistance = FVector::Dist(CurrentLocation, mMoveTargetTransform.GetLocation());
290
291 // 남은 거리에서 멈출 수 있는 속도 계산
292 float TargetSpeed = mMaxMoveSpeed;
293 if (mDeceleration > 0.0f)
294 {
295 // 제동 기준 거리 = 이번 스텝 남은 거리 + 이후 경로 거리 (최종 목적지 기준으로 감속)
296 // 목표가 많이 남으면 아래 BrakeSpeed가 최고속도보다 커지므로 최고속도로 순항하고
297 // 목표가 가까워져서 제동거리 안에 들어가면 BrakeSpeed가 최고속도보다 낮아지므로 BrakeSpeed로 느려짐
298 const float BrakeDistance = RemainingDistance + mRemainingPathDistance;
299 // @details
300 // 등가속도 공식은 `v^2 = v0^2 + 2as` 인데,
301 // 최종속도 v=0 이 되면서 멈추는 거니까 이 공식은 `0 = v0^2 + 2as`
302 // 이때 감속하니까 가속도는 -가 되고 2as를 좌변으로 넘기면 `2as = v0^2`
303 // 따라서 `v0 = sqrt(2as)`
304 const float BrakeSpeed = FMath::Sqrt(2.0f * mDeceleration * BrakeDistance);
305 TargetSpeed = FMath::Min(TargetSpeed, BrakeSpeed);
306 }
307
308 // 현재 속도를 목표 속도로 가감속 (올릴 땐 가속도, 내릴 땐 감속도 사용)
309 // 해당 가감속도가 0이면 즉시 목표 속도 도달
310 const float InterpRate = (TargetSpeed < mCurrentMoveSpeed)
311 ? mDeceleration
312 : mAcceleration;
313 mCurrentMoveSpeed = (InterpRate > 0.0f)
314 ? FMath::FInterpConstantTo(mCurrentMoveSpeed, TargetSpeed, DeltaSeconds, InterpRate)
315 : TargetSpeed;
316
317 // 위치: 현재벡터를, 현재위치에서 다음위치로 현재 속도만큼 이동한 값 계산
318 const FVector NewLocation = FMath::VInterpConstantTo(
319 CurrentLocation,
320 mMoveTargetTransform.GetLocation(),
321 DeltaSeconds, mCurrentMoveSpeed);
322 // 회전: 현재회전을, 다음 타일을 바라보도록 지정한 각도만큼 변경한 값 계산
323 const FRotator NewRotation = FMath::RInterpConstantTo(
324 GetActorRotation(),
325 mMoveTargetTransform.Rotator(),
326 DeltaSeconds, mRotationSpeed);
327 SetActorLocationAndRotation(NewLocation, NewRotation);
328
329 // 현재 속도 저장
330 // 애니메이션이 GetVelocity()로 읽어서 활용 가능
331 mCurrentMoveVelocity
332 = (DeltaSeconds > 0.0f) // 혹시 0으로 나누는 경우를 방지
333 ? (NewLocation - CurrentLocation) / DeltaSeconds
334 : FVector::ZeroVector;
335
336 // 위치와 회전이 모두 목표와 일치하면 -> 해당 타일에 도착
337 if (NewLocation.Equals(mMoveTargetTransform.GetLocation()) &&
338 NewRotation.Equals(mMoveTargetTransform.Rotator()))
339 {
340 // 최종 목적지 도착이면 정지 상태로 초기화 (다음 이동은 0부터 다시 가속)
341 if (mRemainingPathDistance <= KINDA_SMALL_NUMBER)
342 {
343 mCurrentMoveSpeed = 0.0f;
344 mCurrentMoveVelocity = FVector::ZeroVector;
345 }
346 // 도착했으면 틱 해제
347 SetActorTickEnabled(false);
348 // 베리어 해제
349 // -> MoveAction의 OnStepPresentationFinished() 호출
350 // -> StartStep(NextTile) 해서 타일간 이동 반복
351 mMoveBarrier.Reset();
352 }
353}
