17UStaticCombatRoomSpawnData* UFirstPlayTutorialSubsystem::PrepareScenario(
18 UStaticCombatRoomSpawnData* Original, const TArray<TObjectPtr<UPlayerUnitModel>>& Party)
19{
20 ScenarioRoom = nullptr;
21 ScenarioUnit.Reset();
22 ScenarioSkillIndex = INDEX_NONE;
23 auto& Flow = GetUserMutableData()->GuidedTutorial;
24 // Stage Builder has already chosen the entire room, including its background and spawn setting.
25 auto* Authored = Cast<UStaticTutorialRoomSpawnData>(Original);
26 if (!Authored || !Flow.NeedsFirstRoom(GetUserMutableData()->TutorialProgress.Skipped) || Party.IsEmpty() || !Party[0]) return Original;
27 TArray<FText> Errors;
28 if (!Authored->ValidateLayout(Errors))
29 {
30 for (const auto& Error : Errors) UE_LOG(LogTemp, Error, TEXT("Tutorial DA: %s"), *Error.ToString());
31 return Original;
32 }
33 const auto& Skills = Party[0]->GetSkillComponentModel()->GetSkills();
34 TArray<const UStaticUnitSkillData*> SkillData;
35 for (const auto& Skill : Skills) SkillData.Add(Cast<UStaticUnitSkillData>(Skill.mData));
36 ScenarioSkillIndex = FFirstBattleScenario::SelectSkill(SkillData);
37 if (ScenarioSkillIndex != INDEX_NONE) ScenarioSkillName = SkillData[ScenarioSkillIndex]->mName;
38 if (ScenarioSkillIndex == INDEX_NONE)
39 {
40 UE_LOG(LogTemp, Error, TEXT("First battle scenario needs a basic attack in party slot 0."));
41 return Original;
42 }
43 ScenarioRoom = Authored;
44 ScenarioUnit = Party[0];
45 if (!Flow.CoreComplete)
46 {
47 // Combat positions are recreated on room load, so resume the authored opening from its start.
48 Flow.Stage = EGuidedStage::OpenSkills;
49 Flow.Lesson = EGuidedLesson::None;
50 }
51 UE_LOG(LogTemp, Display, TEXT("Tutorial authored room: %s skill=%d"), *Authored->GetPathName(), ScenarioSkillIndex);
52 return ScenarioRoom;
53}

75bool UFirstPlayTutorialSubsystem::IsScenarioPointerAllowed(const FVector2D& Position) const
76{
77 if (!IsScenarioGuiding()) return true;
78 const auto* Combat = GetWorldSubsystemModel<USRPGCombatModel>(GetWorld());
79 const auto* Map = Combat ? Combat->GetTileMap() : nullptr;
80 auto* PC = GetWorld()->GetFirstPlayerController();
81 if (!Map || !PC) return false;
82 FVector2D Pixel, Viewport;
83 USlateBlueprintLibrary::AbsoluteToViewport(this, Position, Pixel, Viewport);
84 FHitResult Hit;
85 if (!PC->GetHitResultAtScreenPosition(Pixel, RDTraceChannels::TileAnyTrace, false, Hit)) return false;
86 FTileIndex Tile = FTileIndex::Invalid;
87 if (Hit.GetActor() == Map->GetView<AActor>()) Tile = Map->WorldToTileIndex(Hit.ImpactPoint);
88 else if (auto* View = Cast<IObjectView>(Hit.GetActor()))
89 if (const auto* Model = View->GetModel<UBoardActorModel>()) Tile = Model->GetTileTransform().mIndex;
90 return IsScenarioTileAllowed(Tile);
91}

93bool UFirstPlayTutorialSubsystem::IsScenarioCommandAllowed(ECombatInputType Type, int32 Payload) const
94{
95 if (IsEncounterHintVisible()) return false;
96 if (!IsScenarioGuiding()) return true;
97 const auto Stage = GetUserMutableData()->GuidedTutorial.Stage;
98 switch (Type)
99 {
100 case ECombatInputType::Move: return Stage == EGuidedStage::SelectMove;
101 case ECombatInputType::SelectSkill: return Stage == EGuidedStage::SelectSkill && Payload == ScenarioSkillIndex;
102 case ECombatInputType::LongPressSkill: return Stage == EGuidedStage::HoldSkill && Payload == ScenarioSkillIndex;
103 case ECombatInputType::Confirm: return Stage == EGuidedStage::ConfirmMove || Stage == EGuidedStage::ConfirmSkill;
104 case ECombatInputType::EndTurn: return Stage == EGuidedStage::EndTurn;
105 case ECombatInputType::LongPressUnit: return Payload == INDEX_NONE; // Closing the actual skill details.
106 case ECombatInputType::FocusUnit: return true; // Camera anchoring requested by the instructed card.
107 default: return false;
108 }
109}

111bool UFirstPlayTutorialSubsystem::GetScenarioFocus(TArray<FVector2D>& Corners) const
112{
113 Corners.Reset();
114 if (!IsScenarioGuiding()) return false;
115 const auto Stage = GetUserMutableData()->GuidedTutorial.Stage;
116 FTileIndex Tile = FTileIndex::Invalid;
117 if (Stage == EGuidedStage::MoveTile || Stage == EGuidedStage::ConfirmMove) Tile = GetScenarioMoveTile();
118 if (Stage == EGuidedStage::SkillTarget || Stage == EGuidedStage::ConfirmSkill) Tile = GetScenarioEnemyTile();
119 if (Tile == FTileIndex::Invalid) return false;
120 const auto* Combat = GetWorldSubsystemModel<USRPGCombatModel>(GetWorld());
121 const auto* Map = Combat ? Combat->GetTileMap() : nullptr;
122 auto* PC = GetWorld()->GetFirstPlayerController();
123 if (!Map || !PC) return false;
124 const FVector Center = Map->TileToWorldLocation(Tile) + FVector(0, 0, 3);
125 const FVector X = (Map->TileToWorldLocation(Tile + FTileIndex(1, 0)) - Map->TileToWorldLocation(Tile)) * .46;
126 const FVector Y = (Map->TileToWorldLocation(Tile + FTileIndex(0, 1)) - Map->TileToWorldLocation(Tile)) * .46;
127 for (const FVector Point : {Center-X-Y, Center+X-Y, Center+X+Y, Center-X+Y})
128 {
129 FVector2D Pixel, Absolute;
130 if (!PC->ProjectWorldLocationToScreen(Point, Pixel)) { Corners.Reset(); return false; }
131 // Keep desktop absolute coordinates; the true option subtracts the window position.
132 USlateBlueprintLibrary::ScreenToWidgetAbsolute(this, Pixel, Absolute);
133 Corners.Add(Absolute);
134 }
135 return true;
136}
