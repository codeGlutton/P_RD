24UWidget *CloneRow(UWidgetTree *Tree, UWidget *Source, int32 Index)
25{
26 FString Name = Source->GetName();
27 Name.RemoveFromEnd(TEXT("_0"));
28 Name += FString::Printf(TEXT("_%d"), Index);
29 if (auto *SourcePanel = Cast<UPanelWidget>(Source))
30 {
31 auto *Copy = Tree->ConstructWidget<UPanelWidget>(Source->GetClass(), *Name);
32 Copy->SetVisibility(Source->GetVisibility());
33 Copy->SetRenderTransform(Source->GetRenderTransform());
34 Copy->SetRenderTransformPivot(Source->GetRenderTransformPivot());
35 if (auto* SourceScale=Cast<UScaleBox>(Source))
36 {
37 auto* Scale=CastChecked<UScaleBox>(Copy);
38 Scale->SetStretch(SourceScale->GetStretch());
39 Scale->SetStretchDirection(SourceScale->GetStretchDirection());
40 Scale->SetUserSpecifiedScale(SourceScale->GetUserSpecifiedScale());
41 }
42 if (auto *SourceButton = Cast<UButton>(Source))
43 {
44 auto *Button = CastChecked<UButton>(Copy);
45 Button->SetStyle(SourceButton->GetStyle());
46 Button->SetBackgroundColor(SourceButton->GetBackgroundColor());
47 Button->SetColorAndOpacity(SourceButton->GetColorAndOpacity());
48 }
49 for (auto *Child : SourcePanel->GetAllChildren())
50 {
51 auto *ChildCopy = CloneRow(Tree, Child, Index);
52 Copy->AddChild(ChildCopy, Child->Slot);
53 if (auto *Layout = Cast<UCanvasPanelSlot>(Child->Slot))
54 {
55 auto *NewLayout = CastChecked<UCanvasPanelSlot>(ChildCopy->Slot);
56 NewLayout->SetLayout(Layout->GetLayout());
57 NewLayout->SetAutoSize(Layout->GetAutoSize());
58 NewLayout->SetZOrder(Layout->GetZOrder());
59 }
60 }
61 return Copy;
62 }
63 auto *Copy = DuplicateObject<UWidget>(Source, Tree, *Name);
64 Copy->Slot = nullptr;
65 return Copy;
66}
