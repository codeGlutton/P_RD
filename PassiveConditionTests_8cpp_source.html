65IMPLEMENT_SIMPLE_AUTOMATION_TEST(
66 FPassiveConditionCompareTests,

68 EAutomationTestFlags::EditorContext | EAutomationTestFlags::ProductFilter

70
71bool FPassiveConditionCompareTests::RunTest(const FString& Parameters)
72{
73 // 대소 비교
74 TestTrue(TEXT("3 < 5"), FPassiveCondition::Compare(3.f, EPassiveCompareOp::Less, 5.f));
75 TestFalse(TEXT("5 < 5"), FPassiveCondition::Compare(5.f, EPassiveCompareOp::Less, 5.f));
76 TestTrue(TEXT("5 <= 5"), FPassiveCondition::Compare(5.f, EPassiveCompareOp::LessEqual, 5.f));
77 TestTrue(TEXT("5 >= 5"), FPassiveCondition::Compare(5.f, EPassiveCompareOp::GreaterEqual, 5.f));
78 TestTrue(TEXT("6 > 5"), FPassiveCondition::Compare(6.f, EPassiveCompareOp::Greater, 5.f));
79 TestFalse(TEXT("5 > 5"), FPassiveCondition::Compare(5.f, EPassiveCompareOp::Greater, 5.f));
80
81 // 근사 비교: float 계산 오차를 같은 값으로 취급
82 TestTrue(TEXT("50 == 100*0.5"), FPassiveCondition::Compare(50.f, EPassiveCompareOp::Equal, 100.f * 0.5f));
83 TestTrue(TEXT("50 != 51"), FPassiveCondition::Compare(50.f, EPassiveCompareOp::NotEqual, 51.f));
84
85 // 배수 판정: 6은 3의 배수, 7은 아님, 0으로는 나눌 수 없어 거짓
86 TestTrue(TEXT("6 % 3 == 0"), FPassiveCondition::Compare(6.f, EPassiveCompareOp::ModuloZero, 3.f));
87 TestFalse(TEXT("7 % 3 != 0"), FPassiveCondition::Compare(7.f, EPassiveCompareOp::ModuloZero, 3.f));
88 TestFalse(TEXT("x % 0 → 거짓"), FPassiveCondition::Compare(6.f, EPassiveCompareOp::ModuloZero, 0.f));
89
90 return true;
91}

93// 피연산자 계산: 속성(배수) / 태그 하위 합 / 카운터 / 스냅샷 없음
94IMPLEMENT_SIMPLE_AUTOMATION_TEST(
95 FPassiveConditionResolveTests,
96 "P_RD.TAS.Passive.Condition.Resolve",
97 EAutomationTestFlags::EditorContext | EAutomationTestFlags::ProductFilter
98)
99
100bool FPassiveConditionResolveTests::RunTest(const FString& Parameters)
101{
102 FTestObjectScope Scope;
103
104 // 소유자: HP 30 / MaxHP 100, 디버프 약화 1 + 취약 2
105 UBoardCombatTargetSnapshotData* Owner = MakeSnapshot(Scope, 30.f, 100.f, FTileIndex(0, 0));
106 Owner->mEffectCounts.Add(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Weakness, 1);
107 Owner->mEffectCounts.Add(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Vulnerability, 2);
108 Owner->mEffectCounts.Add(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Buff_Vigor, 1);
109
110 FPassiveActivateContext Ctx;
111 Ctx.mOwnerSnapshot = Owner;
112
113 FDynamicPassiveData_Generic State;
114 State.mCounter = 6;
115
116 float Value = 0.f;
117
118 // 속성값 + 배수: MaxHP 100 x 0.5 = 50
119 const FPassiveOperand HalfMaxHP = AttrOp(EPassiveOperandSource::Self, UCombatTargetAttributeSet::GetMaxHPAttribute(), 0.5f);
120 TestTrue(TEXT("MaxHP 계산 성공"), HalfMaxHP.Resolve(Ctx, INDEX_NONE, State, Value));
121 TestEqual(TEXT("MaxHP x 0.5 == 50"), Value, 50.f);
122
123 // 조건식: HP(30) <= MaxHP x 0.5(50)
124 FPassiveCondition HalfHP;
125 HalfHP.mLhs = AttrOp(EPassiveOperandSource::Self, UCombatTargetAttributeSet::GetHPAttribute());
126 HalfHP.mOp = EPassiveCompareOp::LessEqual;
127 HalfHP.mRhs = HalfMaxHP;
128 TestTrue(TEXT("HP 30 <= 50 통과"), HalfHP.Evaluate(Ctx, INDEX_NONE, State));
129
130 // 태그 개수: 부모 태그(Debuff)로 하위 합 3, Buff는 제외
131 FPassiveOperand DebuffCount = KindOp(EPassiveOperandKind::TagCount);
132 DebuffCount.mTag = EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff;
133 TestTrue(TEXT("TagCount 계산 성공"), DebuffCount.Resolve(Ctx, INDEX_NONE, State, Value));
134 TestEqual(TEXT("디버프 합 == 3"), Value, 3.f);
135
136 // 카운터 + 배수 판정: 6 % 3 == 0
137 FPassiveCondition Every3rd;
138 Every3rd.mLhs = KindOp(EPassiveOperandKind::Counter);
139 Every3rd.mOp = EPassiveCompareOp::ModuloZero;
140 Every3rd.mRhs = ConstOp(3.f);
141 TestTrue(TEXT("카운터 6, 3의 배수 통과"), Every3rd.Evaluate(Ctx, INDEX_NONE, State));
142
143 // 스냅샷 없음: 속성 조건은 탈락, 고정값끼리는 통과
144 FPassiveActivateContext EmptyCtx;
145 TestFalse(TEXT("스냅샷 없으면 속성 조건 탈락"), HalfHP.Evaluate(EmptyCtx, INDEX_NONE, State));
146 FPassiveCondition ConstOnly;
147 ConstOnly.mLhs = ConstOp(1.f);
148 ConstOnly.mOp = EPassiveCompareOp::Equal;
149 ConstOnly.mRhs = ConstOp(1.f);
150 TestTrue(TEXT("고정값 조건은 스냅샷 없어도 통과"), ConstOnly.Evaluate(EmptyCtx, INDEX_NONE, State));
151
152 return true;
153}

155// 위치/캡처 기반 피연산자: 거리 / 캡처값 / 이동 거리 / 미캡처
156IMPLEMENT_SIMPLE_AUTOMATION_TEST(
157 FPassiveConditionSpatialTests,
158 "P_RD.TAS.Passive.Condition.Spatial",
159 EAutomationTestFlags::EditorContext | EAutomationTestFlags::ProductFilter

161
162bool FPassiveConditionSpatialTests::RunTest(const FString& Parameters)
163{
164 FTestObjectScope Scope;
165
166 // 소유자 (0,0), 타겟 0 (2,3), 타겟 1 (1,0)
167 FPassiveActivateContext Ctx;
168 Ctx.mOwnerSnapshot = MakeSnapshot(Scope, 100.f, 100.f, FTileIndex(0, 0));
169 Ctx.mTargetSnapshots.Add(MakeSnapshot(Scope, 100.f, 100.f, FTileIndex(2, 3)));
170 Ctx.mTargetSnapshots.Add(MakeSnapshot(Scope, 100.f, 100.f, FTileIndex(1, 0)));
171
172 FDynamicPassiveData_Generic State;
173 float Value = 0.f;
174
175 // 거리: 타겟 0은 2+3=5, 타겟 1은 1 (맨해튼 방식)
176 const FPassiveOperand Distance = KindOp(EPassiveOperandKind::Distance);
177 TestTrue(TEXT("거리 계산 성공"), Distance.Resolve(Ctx, 0, State, Value));
178 TestEqual(TEXT("타겟 0 거리 == 5"), Value, 5.f);
179 TestTrue(TEXT("거리 계산 성공"), Distance.Resolve(Ctx, 1, State, Value));
180 TestEqual(TEXT("타겟 1 거리 == 1"), Value, 1.f);
181 TestFalse(TEXT("범위 밖 타겟 인덱스는 실패"), Distance.Resolve(Ctx, 2, State, Value));
182
183 // 이동 거리: 캡처 전엔 실패, 캡처 후 소유자가 (0,0)→현재 (0,0)이면 0, 타겟 0은 (0,3)→(2,3)이면 2
184 FPassiveOperand SelfMoved = KindOp(EPassiveOperandKind::MovedDistance, EPassiveOperandSource::Self);
185 TestFalse(TEXT("미캡처면 이동 거리 실패"), SelfMoved.Resolve(Ctx, INDEX_NONE, State, Value));
186 State.mCapturedSelfTile = FTileIndex(0, 0);
187 State.mCapturedTargetTiles = { FTileIndex(0, 3), FTileIndex(1, 0) };
188 TestTrue(TEXT("소유자 이동 거리 계산 성공"), SelfMoved.Resolve(Ctx, INDEX_NONE, State, Value));
189 TestEqual(TEXT("소유자 이동 거리 == 0"), Value, 0.f);
190 const FPassiveOperand TargetMoved = KindOp(EPassiveOperandKind::MovedDistance, EPassiveOperandSource::Target);
191 TestTrue(TEXT("타겟 이동 거리 계산 성공"), TargetMoved.Resolve(Ctx, 0, State, Value));
192 TestEqual(TEXT("타겟 0 이동 거리 == 2"), Value, 2.f);
193
194 // 캡처값: 키 "HP"에 소유자 80, 타겟별 [60, 40]. 없는 키는 실패
195 FPassiveCaptureSlot Slot;
196 Slot.mSelf = 80.f;
197 Slot.mTargets = { 60.f, 40.f };
198 State.mCaptures.Add(TEXT("HP"), Slot);
199 FPassiveOperand CapturedSelf = KindOp(EPassiveOperandKind::Captured, EPassiveOperandSource::Self);
200 CapturedSelf.mCaptureKey = TEXT("HP");
201 TestTrue(TEXT("소유자 캡처값 계산 성공"), CapturedSelf.Resolve(Ctx, INDEX_NONE, State, Value));
202 TestEqual(TEXT("소유자 캡처값 == 80"), Value, 80.f);
203 FPassiveOperand CapturedTarget = KindOp(EPassiveOperandKind::Captured, EPassiveOperandSource::Target);
204 CapturedTarget.mCaptureKey = TEXT("HP");
205 TestTrue(TEXT("타겟 캡처값 계산 성공"), CapturedTarget.Resolve(Ctx, 1, State, Value));
206 TestEqual(TEXT("타겟 1 캡처값 == 40"), Value, 40.f);
207 CapturedSelf.mCaptureKey = TEXT("None");
208 TestFalse(TEXT("없는 캡처 키는 실패"), CapturedSelf.Resolve(Ctx, INDEX_NONE, State, Value));
209
210 return true;
211}
