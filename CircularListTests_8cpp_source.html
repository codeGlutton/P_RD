21bool FCircularListInt32Tests::RunTest(const FString& Parameters)
22{
23 {
24 TCircularDoubleLinkedList<int32> HeadTestObj;
25
26 AddInfo(TEXT("객체 생성 검증"));
27 {
28 TestNotNull(TEXT("TCircularDoubleLinkedList<int32> 반드시 생성"), &HeadTestObj);
29 }
30
31 AddInfo(TEXT("초기화 검증"));
32 {
33 TestEqual(TEXT("TCircularDoubleLinkedList<int32> 초기 노드 0개"), HeadTestObj.IsEmpty(), true);
34 TestEqual(TEXT("TCircularDoubleLinkedList<int32> 초기 헤드 노드 nullptr"), HeadTestObj.GetHead() == nullptr, true);
35 TestEqual(TEXT("TCircularDoubleLinkedList<int32> 초기 테일 노드 nullptr"), HeadTestObj.GetTail() == nullptr, true);
36 }
37
38 const int32 HeadFirstNumber = 0;
39 HeadTestObj.AddHead(HeadFirstNumber);
40
41 AddInfo(TEXT("AddHead 및 순환 구조 검증"));
42 for (int32 i = 1; i <= 30; ++i)
43 {
44 HeadTestObj.AddHead(i);
45 TestEqual(TEXT("AddHead 헤드 값 검증"), HeadTestObj.GetHead()->GetValue(), i);
46 TestEqual(TEXT("AddHead 다음 값 검증"), HeadTestObj.GetHead()->GetNextNode()->GetValue(), i - 1);
47 TestEqual(TEXT("AddHead 이전 값 검증"), HeadTestObj.GetHead()->GetPrevNode()->GetValue(), HeadFirstNumber);
48 }
49 }
50
51 {
52 TCircularDoubleLinkedList<int32> TailTestObj;
53 TailTestObj.AddTail(10);
54 TailTestObj.AddTail(20);
55 TailTestObj.AddTail(30);
56
57 AddInfo(TEXT("AddTail 및 순환 구조 검증"));
58 {
59 TestEqual(TEXT("AddTail: 개수 검증"), TailTestObj.Num(), 3);
60 TestEqual(TEXT("AddTail: 헤드 값 검증"), TailTestObj.GetHead()->GetValue(), 10);
61 TestEqual(TEXT("AddTail: 테일 값 검증"), TailTestObj.GetTail()->GetValue(), 30);
62 }
63
64 AddInfo(TEXT("순환성 검증"));
65 {
66
67 TestEqual(TEXT("순환성: Tail->Next == Head"), TailTestObj.GetTail()->GetNextNode(), TailTestObj.GetHead());
68 TestEqual(TEXT("순환성: Head->Prev == Tail"), TailTestObj.GetHead()->GetPrevNode(), TailTestObj.GetTail());
69 }
70
71 AddInfo(TEXT("InsertNode 검증"));
72 {
73 auto Node20 = TailTestObj.FindNode(20);
74 TailTestObj.InsertNode(15, Node20); // 10 -> 15 -> 20 -> 30
75 TestEqual(TEXT("InsertNode: 중간 삽입 후 값 검증"), Node20->GetPrevNode()->GetValue(), 15);
76 TestEqual(TEXT("InsertNode: 삽입된 노드의 앞 노드 검증"), TailTestObj.FindNode(15)->GetPrevNode()->GetValue(), 10);
77 }
78
79 AddInfo(TEXT("RemoveNode 검증"));
80 {
81 TailTestObj.RemoveNode(15); // 10 -> 20 -> 30
82 TestFalse(TEXT("RemoveNode: 삭제된 노드 미포함 확인"), TailTestObj.Contains(15));
83 TestEqual(TEXT("RemoveNode: 삭제 후 개수 확인"), TailTestObj.Num(), 3);
84 TestEqual(TEXT("RemoveNode: 삭제 후 연결 확인"), TailTestObj.FindNode(10)->GetNextNode()->GetValue(), 20);
85 }
86
87 AddInfo(TEXT("Iterator 검증"));
88 {
89 auto Iter = TCircularDoubleLinkedList<int32>::TIterator(TailTestObj.GetHead());
90 TestEqual(TEXT("Iterator: 첫 번째"), *Iter, 10);
91 ++Iter;
92 TestEqual(TEXT("Iterator: 두 번째"), *Iter, 20);
93 ++Iter;
94 TestEqual(TEXT("Iterator: 세 번째"), *Iter, 30);
95 ++Iter;
96 TestEqual(TEXT("Iterator: 순환 후 첫 번째"), *Iter, 10); // 다시 10으로 돌아옴
97 --Iter;
98 TestEqual(TEXT("Iterator: 역순 순환 후 마지막"), *Iter, 30); // 다시 30으로 돌아감
99 }
100
101 AddInfo(TEXT("Empty 검증"));
102 {
103 TailTestObj.Empty();
104 TestTrue(TEXT("Empty: 비어있음 확인"), TailTestObj.IsEmpty());
105 TestNull(TEXT("Empty: 헤드 nullptr 확인"), TailTestObj.GetHead());
106 }
107 }
108
109 return true;
110}
