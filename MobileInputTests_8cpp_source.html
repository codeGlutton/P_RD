28{
29 UWorld* World = GEditor->GetEditorWorldContext().World();
30 auto* HUD = CreateWidget<UCombatLayoutHUDWidget>(World);
31 if (!TestNotNull(TEXT("Board HUD"), HUD)) return false;
32 auto* Model = NewObject<UCombatUIModel>();
33 auto* Listener = NewObject<URDMobileInputTestListener>();
34 Model->OnCombatWorldTouch.AddDynamic(Listener, &URDMobileInputTestListener::WorldTouch);
35 FTurnUI Turn;
36 Turn.mPhase = ECombatBuildPhaseUI::Preview;
37 Model->SetTurnUI(Turn);
38 HUD->mUIModel = Model;
39 const FGeometry Geometry;
40 auto Touch = [](uint32 Id, double X) { return FPointerEvent(uint32(0), Id, FVector2D(X, 100), FVector2D(X, 100), 1.f, true); };
41 auto Begin = [&](uint32 Id, double X) { HUD->NativeOnTouchStarted(Geometry, Touch(Id, X)); };
42 auto End = [&](uint32 Id, double X) { HUD->NativeOnTouchEnded(Geometry, Touch(Id, X)); };
43 Begin(0, 100); Begin(1, 300);
44 HUD->HandleBoardLongPress();
45 End(1, 300); End(0, 100);
46 TestEqual(TEXT("Two contacts never make a tile tap"), Listener->Taps, 0);
47 TestEqual(TEXT("Second contact cancels the pending hold"), Listener->Holds, 0);
48 Begin(3, 100); End(3, 100);
49 TestEqual(TEXT("A fresh single contact works after both fingers lift"), Listener->Taps, 1);
50 Begin(0, 100);
51 // This contact is consumed by a child button: only the preprocessor sees it.
52 HUD->ObserveBoardTouchDown(1);
53 HUD->ObserveBoardTouchUp(1);
54 HUD->HandleBoardLongPress();
55 End(0, 100);
56 TestEqual(TEXT("A second finger on UI also consumes the board gesture"), Listener->Taps, 1);
57 TestEqual(TEXT("Lifting the secondary finger cannot revive a hold"), Listener->Holds, 0);
58 Begin(0, 100);
59 HUD->NativeOnTouchMoved(Geometry, Touch(0, 110));
60 End(0, 110);
61 TestEqual(TEXT("Touch jitter below the shared pan threshold remains a tap"), Listener->Taps, 2);
62 Begin(0, 100);
63 HUD->NativeOnTouchMoved(Geometry, Touch(0, 130));
64 HUD->NativeOnTouchMoved(Geometry, Touch(0, 100));
65 End(0, 100);
66 TestEqual(TEXT("A drag returning to its origin cannot become a tap"), Listener->Taps, 2);
67 const TSet<FKey> Buttons{EKeys::LeftMouseButton};
68 auto Mouse = [&](double X) { return FPointerEvent(0, FVector2D(X, 100), FVector2D(X, 100), Buttons, EKeys::LeftMouseButton, 0, FModifierKeysState()); };
69 HUD->NativeOnMouseButtonDown(Geometry, Mouse(100));
70 HUD->NativeOnMouseMove(Geometry, Mouse(104));
71 HUD->NativeOnMouseButtonUp(Geometry, Mouse(104));
72 TestEqual(TEXT("Desktop pan and board tap use the same 3px threshold"), Listener->Taps, 2);
73 Begin(0, 100);
74 HUD->HandleBoardLongPress();
75 End(0, 100);
76 TestEqual(TEXT("Ordinary single-finger hold remains available"), Listener->Holds, 1);
77 TestEqual(TEXT("A hold release is not also a tap"), Listener->Taps, 2);
78 World->GetTimerManager().ClearAllTimersForObject(HUD);
79 return !HasAnyErrors();
80}

133 EAutomationTestFlags::EditorContext | EAutomationTestFlags::EngineFilter)
134bool FRDCenteredNotchTest::RunTest(const FString&)
135{
136 auto& App = FSlateApplication::Get();
137 const bool HadCustom = App.IsCustomSafeZoneSet();
138 const FMargin Previous = App.GetCustomSafeZone();
139 const auto GlobalScale = SSafeZone::GetGlobalSafeZoneScale();
140 SSafeZone::SetGlobalSafeZoneScale(TOptional<float>());
141 int32 Cases = 0;
142 for (const FVector2D Pixels : {FVector2D(1920,1080), FVector2D(2400,1080), FVector2D(3200,1440)})
143 for (const float Dpi : {1.f, 1.5f, 2.f})

150 FMargin(96,0,24,48), FMargin(0), FMargin(48,24,48,24)})

152 App.SetCustomSafeZone(FMargin(2*Inset.Left/Pixels.X, 2*Inset.Top/Pixels.Y,

155 Safe->SlatePrepass(Dpi);

157 Safe->ArrangeChildren(Viewport, Arranged);
158 if (!TestEqual(TEXT("One safe content root"), Arranged.Num(), 1)) continue;
159 const FGeometry& Content = Arranged[0].Geometry;
160 const FVector2D Origin = Viewport.AbsoluteToLocal(Content.LocalToAbsolute(FVector2D::ZeroVector)) * Dpi;
161 const FVector2D Extent = Content.GetLocalSize() * Dpi;
162 TestTrue(TEXT("HUD center stays at viewport center with either notch orientation"),
163 (Origin + Extent*.5).Equals(Pixels*.5, .1));
164 TestTrue(TEXT("All four physical unsafe edges are excluded"), Origin.X >= Inset.Left-1 && Origin.Y >= Inset.Top-1
165 && Origin.X+Extent.X <= Pixels.X-Inset.Right+1 && Origin.Y+Extent.Y <= Pixels.Y-Inset.Bottom+1);
166 if (Inset == FMargin(0))
167 TestTrue(TEXT("No cutout leaves the full viewport available"), Origin.IsNearlyZero() && Extent.Equals(Pixels,.1));
168 ++Cases;
169 }
170 }
171 if (HadCustom) App.SetCustomSafeZone(Previous); else App.ResetCustomSafeZone();
172 SSafeZone::SetGlobalSafeZoneScale(GlobalScale);
173 AddInfo(FString::Printf(TEXT("Checked %d live safe-area arrangements across resolutions, DPI and notch rotation."), Cases));
174 return !HasAnyErrors();
175}
