114 const TSoftObjectPtr<UStaticSkillData>& SkillSoft = PlayerData->mSkillDatas[i];
115 if (const UStaticSkillData* SkillData = SkillSoft.LoadSynchronous())
116 {

118 }
119 }
120 }

235void UPlayerUnitPersistData::BindPlayerUnitEvent(UPlayerUnitModel* PlayerUnit)
236{
237 checkf(PlayerUnit != nullptr, TEXT("플레이어 유닛 nullptr"));
238 UAttributeSetComponentModel* AttributeSetComponentModel = PlayerUnit->GetAttributeComponentModel();
239 checkf(AttributeSetComponentModel != nullptr, TEXT("어빌리티 시스템 컴포넌트 nullptr"));
240 USkillComponentModel* SkillComponentModel = PlayerUnit->GetSkillComponentModel();
241 checkf(SkillComponentModel != nullptr, TEXT("플레이어 스킬 컴포넌트 nullptr"));
242
243 /* 죽음으로 인한 파티 해제 처리 */
244
245 AttributeSetComponentModel->RegisterTacticalTagEvent(EffectTags::GameplayEffect_ActorState_Dead, ETacticalTagEventType::NewOrRemoved).AddWeakLambda(
246 PlayerUnit, [PlayerUnit](const FGameplayTag Tag, int32 Count) {
247 if (Count <= 0)
248 {
249 return;
250 }
251
252 UPartyModel* OwnerParty = PlayerUnit->GetOwnerParty();
253 if (OwnerParty != nullptr)
254 {
255 OwnerParty->RemovePlayerUnitModel(PlayerUnit);
256 }
257 }
258 );
259
260 // 레벨 추적
261 PlayerUnit->OnChangePlayerLevel.AddWeakLambda(this, [this](UPlayerUnitModel* Model, int32 PlayerLevel) {
262 mPlayerLevel = PlayerLevel;
263 });
264
265 // 스텟 추적
266 AttributeSetComponentModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetMaxHPAttribute()).AddWeakLambda(this, [this, AttributeSetComponentModel](const FTacticalAttributeChangeData& Data) {
267 mMaxHP = AttributeSetComponentModel->GetAttributeBaseValue(Data.mAttribute);
268 });
269 AttributeSetComponentModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetHPAttribute()).AddWeakLambda(this, [this, AttributeSetComponentModel](const FTacticalAttributeChangeData& Data) {
270 mHP = AttributeSetComponentModel->GetAttributeBaseValue(Data.mAttribute);
271 });
272 AttributeSetComponentModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetExpAttribute()).AddWeakLambda(this, [this, AttributeSetComponentModel](const FTacticalAttributeChangeData& Data) {
273 mExp = AttributeSetComponentModel->GetAttributeBaseValue(Data.mAttribute);
274 });
275
276 // 패시브 스택 비용 태그의 개수 변화를 추적
277 AttributeSetComponentModel->RegisterTacticalTagEvent(EffectTags::GameplayEffect_Cost_PassiveStack, ETacticalTagEventType::AnyCountChange).AddWeakLambda(this, [this](const FGameplayTag Tag, int32 Count) {
278 if (Count <= 0)
279 {
280 mTagCountMap.Remove(Tag);
281 }
282 else
283 {
284 mTagCountMap.Add(Tag, Count);
285 }
286 });
287
288 // 플레이어 스킬 추적
289 SkillComponentModel->OnChangeSkillUI.AddWeakLambda(this, [this](int32 SkillIndex, const UStaticSkillData* PreSkillData, const UStaticSkillData* NewSkillData)
290 {
291 // 음수 인덱스 방어
292 if (SkillIndex < 0)
293 {
294 return;
295 }
296
297 // 슬롯이 부족하면 확장 후 기록
298 if (SkillIndex >= mSkillIds.Num())
299 {
300 mSkillIds.SetNum(SkillIndex + 1);
301 }
302
303 FPrimaryAssetId NextSkillId;
304 if (NewSkillData != nullptr)
305 {
306 NextSkillId = NewSkillData->GetPrimaryAssetId();
307 }
308 mSkillIds[SkillIndex] = NextSkillId;
309 });
310}
