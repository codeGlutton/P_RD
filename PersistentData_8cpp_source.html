74

77 mIsNewData = false;
78
79 bool IsFound;
80
81 IsFound = false;
82 mMaxHP = ASC->GetGameplayAttributeValue(UPlayerUnitAttributeSet::GetMaxHPAttribute(), IsFound);
83 checkf(IsFound == true, TEXT("최대 체력 속성 미발견"));
84
85 IsFound = false;
86 mHP = ASC->GetGameplayAttributeValue(UPlayerUnitAttributeSet::GetHPAttribute(), IsFound);
87 checkf(IsFound == true, TEXT("체력 속성 미발견"));
88
89 IsFound = false;
90 mExp = ASC->GetGameplayAttributeValue(UPlayerUnitAttributeSet::GetExpAttribute(), IsFound);

92
93 IsFound = false;
94 mMoney = ASC->GetGameplayAttributeValue(UPlayerUnitAttributeSet::GetMoneyAttribute(), IsFound);
95 checkf(IsFound == true, TEXT("돈 속성 미발견"));
96
97 return;
98 }
99
100 ASC->ApplyModToAttribute(UPlayerUnitAttributeSet::GetMaxHPAttribute(), EGameplayModOp::Override, mMaxHP);
101 ASC->ApplyModToAttribute(UPlayerUnitAttributeSet::GetHPAttribute(), EGameplayModOp::Override, mHP);
102 ASC->ApplyModToAttribute(UPlayerUnitAttributeSet::GetExpAttribute(), EGameplayModOp::Override, mExp);
103 ASC->ApplyModToAttribute(UPlayerUnitAttributeSet::GetMoneyAttribute(), EGameplayModOp::Override, mMoney);
104 for (auto& Pair : mTagCountMap)

106 ASC->AddLooseGameplayTag(Pair.Key, Pair.Value);
107 }
