138void UPlayerUnitPersistData::SyncPlayerPersistData(UPlayerUnitModel* PlayerUnit)
139{
140 checkf(PlayerUnit != nullptr, TEXT("플레이어 유닛 nullptr"));
141 UAttributeSetComponentModel* ASCModel = PlayerUnit->GetAttributeComponentModel();
142 checkf(ASCModel != nullptr, TEXT("어빌리티 시스템 컴포넌트 nullptr"));
143
144 // 최초 등록: 유닛의 현재 속성값을 저장본으로 캡처한 뒤 더 진행하지 않는다(복원 단계 건너뜀).
145 if (mIsNewData == true)
146 {
147 mIsNewData = false;
148 }
149
150 // 저장본 -> 유닛 복원. ETacticalModOp::Override(정수 3, 구 EGameplayModOp::Override 대체)는
151 // 누적이 아닌 "덮어쓰기"라 저장된 절대값을 그대로 다시 세팅한다.
152 // ※ ETacticalModOp 의 정수값을 구 enum과 동일하게 유지하는 이유: 직렬화 호환 / Aggregator의 op-값 배열 인덱싱 / CoreRedirect 매핑.
153 ASCModel->ApplyModToAttribute(UPlayerUnitAttributeSet::GetMaxHPAttribute(), ETacticalModOp::Override, mMaxHP);
154 ASCModel->ApplyModToAttribute(UPlayerUnitAttributeSet::GetHPAttribute(), ETacticalModOp::Override, mHP);
155 ASCModel->ApplyModToAttribute(UPlayerUnitAttributeSet::GetExpAttribute(), ETacticalModOp::Override, mExp);
156 ASCModel->ApplyModToAttribute(UPlayerUnitAttributeSet::GetMoneyAttribute(), ETacticalModOp::Override, mMoney);
157
158 // (여러 방 내에서 유지되는 특수한 패시브 스택) Loose 게임플레이 태그를 개수(Pair.Value)만큼 다시 부여한다.
159 for (auto& Pair : mTagCountMap)
160 {
161 ASCModel->AddLooseGameplayTag(Pair.Key, Pair.Value);
162 }
163
164 // 플레이어 스킬 동기화
165 USkillComponentModel* SkillComponentModel = PlayerUnit->GetSkillComponentModel();
166 if (SkillComponentModel != nullptr)
167 {
168 SkillComponentModel->SetSkillFrom(GetSkillIds());
169 }
170
171 // 플레이어 장비 동기화
172 UEquipmentComponentModel* EquipmentComponentModel = PlayerUnit->GetEquipmentComponentModel();
173 if (EquipmentComponentModel != nullptr)
174 {
175 EquipmentComponentModel->EquipFrom(GetEquipmentIds());
176 }
177
178 // 플레이어 주사위 동기화
179 UDicePoolModel* DicePoolModel = PlayerUnit->GetDicePoolModel();
180 if (DicePoolModel != nullptr)
181 {
182 DicePoolModel->BuildFromDiceIds(GetDiceIds());
183 }

192void UPlayerUnitPersistData::BindPlayerUnitEvent(UPlayerUnitModel* PlayerUnit)
193{
194 checkf(PlayerUnit != nullptr, TEXT("플레이어 유닛 nullptr"));
195 UAttributeSetComponentModel* ASCModel = PlayerUnit->GetAttributeComponentModel();
196 checkf(ASCModel != nullptr, TEXT("어빌리티 시스템 컴포넌트 nullptr"));
197
198 ASCModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetMaxHPAttribute()).AddLambda([this](const FTacticalAttributeChangeData& Data) {
199 mMaxHP = Data.mNewValue;
200 });
201 ASCModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetHPAttribute()).AddLambda([this](const FTacticalAttributeChangeData& Data) {
202 mHP = Data.mNewValue;
203 });
204 ASCModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetExpAttribute()).AddLambda([this](const FTacticalAttributeChangeData& Data) {
205 mExp = Data.mNewValue;
206 });
207 ASCModel->GetTacticalAttributeValueChangeDelegate(UPlayerUnitAttributeSet::GetMoneyAttribute()).AddLambda([this](const FTacticalAttributeChangeData& Data) {
208 mMoney = Data.mNewValue;
209 });
210
211 // 패시브 스택 비용 태그의 개수 변화를 추적: 현재 개수를 맵에 저장하되, 0이 되면 항목을 제거해 저장본을 깔끔히 유지한다.
212 ASCModel->RegisterTacticalTagEvent(EffectTags::GameplayEffect_Cost_PassiveStack, EGameplayTagEventType::AnyCountChange).AddLambda([this](const FGameplayTag Tag, int32 Count) {
213 mTagCountMap[Tag] = Count;
214 if (Count == 0)
215 {
216 mTagCountMap.Remove(Tag);
217 }
218 });
219
220 // 플레이어 스킬 추적
221 USkillComponentModel* SkillComponentModel = PlayerUnit->GetSkillComponentModel();
222 checkf(SkillComponentModel != nullptr, TEXT("플레이어 스킬 컴포넌트 nullptr"));
223 SkillComponentModel->OnChangeSkillUI.AddLambda([this](int32 SkillIndex, const UStaticSkillData* PreSkillData, const UStaticSkillData* NewSkillData) {
224 if (mSkillIds.Num() < SkillIndex)
225 {
226 FPrimaryAssetId NextSkillId;
227 if (NewSkillData != nullptr)
228 {
229 NextSkillId = NewSkillData->GetPrimaryAssetId();
230 }
231 mSkillIds[SkillIndex] = NextSkillId;
232 }
233 });
234
235 // TODO: 장비 변경 시 대리자에 바인딩. 이걸로 영구 데이터도 동기화
236 // TODO: 주사위 변경 시 대리자에 바인딩. 이걸로 영구 데이터도 동기화

245void URunPersistData::StartRun(const FPrimaryAssetId& PlayerUnitId, int32 Difficulty)
246{
247 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
248 checkf(AssetManager != nullptr, TEXT("에셋 매니저 nullptr"));
249
250 const UStaticPlayerUnitSpawnData* PlayerData = AssetManager->GetPrimaryAssetObject<UStaticPlayerUnitSpawnData>(PlayerUnitId);
251 checkf(PlayerData != nullptr, TEXT("플레이어 데이터 nullptr"));
252
253 ClearRun();
254
255 // 시드 초기 세팅
256 {
257 mStageBuildStream.Initialize(FMath::Rand32());
258 mEventStream.Initialize(FMath::Rand32());
259 }
260
261 // 플레이어 기본 데이터 세팅
262 {
263 mPlayerUnitId = PlayerUnitId;
264 mDifficulty = Difficulty;
265 }
266
267 // 플레이어 기본 속성 세팅
268 {
269 mMaxHP = PlayerData->GetDefaultAttributeValue(GetWorld(), UPlayerUnitAttributeSet::StaticClass(), UPlayerUnitAttributeSet::GetMaxHPAttribute(), mPlayerLevel);
270 const float DefaultHP = PlayerData->GetDefaultAttributeValue(GetWorld(), UPlayerUnitAttributeSet::StaticClass(), UPlayerUnitAttributeSet::GetHPAttribute(), mPlayerLevel);
271 mHP = DefaultHP > 0.0f ? FMath::Min(DefaultHP, mMaxHP) : mMaxHP;
272 mExp = 0.f;
273 mMoney = PlayerData->GetDefaultAttributeValue(GetWorld(), UPlayerUnitAttributeSet::StaticClass(), UPlayerUnitAttributeSet::GetMoneyAttribute(), mPlayerLevel);
274 }
275
276 // 스킬 기본 값 세팅
277 {
278 constexpr int32 PlayerSkillSlot = 6;
279 mSkillIds.Init(FPrimaryAssetId(), PlayerSkillSlot);
280
281 const int32 StaticSkillCount = PlayerData->mSkillDatas.Num();
282 for (int32 i = 0; i < StaticSkillCount; ++i)
283 {
284 const TSoftObjectPtr<UStaticSkillData>& SkillSoft = PlayerData->mSkillDatas[i];
285 if (const UStaticSkillData* SkillData = SkillSoft.LoadSynchronous())
286 {
287 mSkillIds[i] = SkillData->GetPrimaryAssetId();
288 }
289 }
290 }
291
292 // 장비 기본 값 세팅
293 {
294 for (const TSoftObjectPtr<UStaticEquipmentData>& EquipmentSoft : PlayerData->mEquipmentDatas)
295 {
296 if (const UStaticEquipmentData* EquipmentData = EquipmentSoft.LoadSynchronous())
297 {
298 mEquipmentIds.Add(EquipmentData->GetPrimaryAssetId());
299 }
300 }
301 }
302
303 // 주사위 기본 값 세팅
304 {
305 for (const TSoftObjectPtr<UStaticDiceData>& DiceSoft : PlayerData->mDiceDatas)
306 {
307 if (const UStaticDiceData* DiceData = DiceSoft.LoadSynchronous())
308 {
309 mDiceIds.Add(DiceData->GetPrimaryAssetId());
310 }
311 }
312 }
