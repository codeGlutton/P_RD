122// 설정 기능은 타이틀과 인게임에서 같은 WBP_SettingsPanel 월드 위젯을 공유한다.
123// 타이틀에서는 저장 후 종료/포기하기 같은 런 액션이 보이면 안 되므로 PanelMode를 Title로 바꾸고 Run 액션 상태를 비활성화한다.
124// 왜 타이틀 내부 ScreenSwitcher로 보여주지 않는가:
125// 설정 패널은 HUD 하위 화면이 아니라 공용 팝업이다. InGameSettings 월드 위젯을 OpenUI()로 열어야
126// 타이틀/인게임 모두 AddToViewport, ZOrder, Back/Close 처리 규칙을 같은 경로로 검증할 수 있다.
127void UTitleMenuWidget::OpenSettingsPanel()
128{
129 USettingsPanelWidget* TitleSettingsPanel = GetTitleSettingsPanel();
130 if (TitleSettingsPanel == nullptr)
131 {
132 SetStatusText(mMainOnlyStatusText);
133 return;
134 }
135
136 TitleSettingsPanel->OnBackRequested.AddUniqueDynamic(this, &UTitleMenuWidget::HandleSettingsPanelBackRequested);
137 TitleSettingsPanel->SetPanelMode(ESettingsPanelMode::Title);
138 // 같은 SettingsPanel 인스턴스를 인게임에서도 쓰므로, 타이틀에서만 런 액션 영역을 숨긴 상태로 갱신한다.
139 TitleSettingsPanel->RefreshPanelState(false, false);
140 TitleSettingsPanel->HideAbandonConfirm();
141 TitleSettingsPanel->SetStatusText(FText::GetEmpty());
142
143 TitleSettingsPanel->OpenUI();
144 SetStatusText(FText::GetEmpty());
145}
