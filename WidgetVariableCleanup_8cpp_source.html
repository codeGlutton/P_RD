12/*
13 * 위젯을 지운 뒤 남은 변수 GUID 를 걷어낸다.
14 *
15 * 왜 필요한가
16 * -----------
17 * 배치를 파이썬으로 다시 짤 때 ``WidgetTree`` 의 자식을 통째로 갈아 끼운다.
18 * 그러면 옛 위젯 객체는 사라지는데 블루프린트의 **변수 목록**에는 그 이름의
19 * GUID 가 그대로 남는다. 컴파일할 때마다
20 *
21 * Variable [X] was deleted but still has a GUID referenced by WidgetBlueprint
22 *
23 * 가 이름 수만큼 뜬다(설정 패널 재배치 직후 실측 98개). 게임은 돌지만 매 부팅
24 * 로그가 이 줄로 덮여 진짜 문제를 못 본다.
25 *
26 * 파이썬에는 이 GUID 를 지우는 길이 없다. ``UWidgetBlueprint::OnVariableRemoved``
27 * 는 C++ 전용이라, 에디터 명령 하나로 열어 준다.
28 *
29 * 안전 장치
30 * ---------
31 * **트리에 같은 이름의 위젯이 실제로 없을 때만 지운다.** 살아 있는 위젯의
32 * 변수를 지우면 그 위젯이 블루프린트 그래프에서 사라져 배선이 끊긴다.
33 */

46 static void Clean(const TArray<FString>& Args)
47 {
48 if (Args.Num() == 0)
49 {
50 UE_LOG(LogTemp, Error,
51 TEXT("RD_WIDGET_VAR_CLEAN needs an asset path, e.g. /Game/UI/WBP_SettingsPanel"));
52 return;
53 }
54
55 for (const FString& AssetPath : Args)
56 {
57 UWidgetBlueprint* Blueprint = LoadObject<UWidgetBlueprint>(nullptr, *AssetPath);
58 if (Blueprint == nullptr || Blueprint->WidgetTree == nullptr)
59 {
60 UE_LOG(LogTemp, Error, TEXT("RD_WIDGET_VAR_CLEAN missing %s"), *AssetPath);
61 continue;
62 }
63
64 // 지금 트리에 실제로 있는 이름을 먼저 모은다. 이 목록에 든 변수는
65 // 절대 건드리지 않는다.
66 TSet<FName> LiveNames;
67 Blueprint->WidgetTree->ForEachWidget([&LiveNames](UWidget* Widget)
68 {
69 if (Widget != nullptr)
70 {
71 LiveNames.Add(Widget->GetFName());
72 }
73 });
74
75 // 위젯 변수는 NewVariables 가 아니라 WidgetVariableNameToGuidMap 에
76 // 산다. 컴파일러는 변수 여부와 무관하게 모든 SourceWidget 이름에 GUID가
77 // 있기를 요구한다(WidgetBlueprintCompiler.cpp:794). 파이썬으로 새 위젯을
78 // 만들면 이 맵에 자동 등록되지 않으므로, 빠진 쪽도 여기서 채운다.
79 int32 Added = 0;
80 Blueprint->WidgetTree->ForEachWidget([Blueprint, &Added](UWidget* Widget)
81 {
82 if (Widget != nullptr &&
83 Blueprint->WidgetVariableNameToGuidMap.Contains(Widget->GetFName()) == false)
84 {
85 Blueprint->WidgetVariableNameToGuidMap.Add(
86 Widget->GetFName(), FGuid::NewDeterministicGuid(Widget->GetPathName()));
87 ++Added;
88 }
89 });
90
91 TArray<FName> Stale;
92 for (const TPair<FName, FGuid>& Entry : Blueprint->WidgetVariableNameToGuidMap)
93 {
94 if (LiveNames.Contains(Entry.Key) == false)
95 {
96 Stale.Add(Entry.Key);
97 }
98 }
99 for (const FName& Name : Stale)
100 {
101 Blueprint->WidgetVariableNameToGuidMap.Remove(Name);
102 Blueprint->OnVariableRemoved(Name);
103 }
104
105 // 컴파일러도 이 짝을 스스로 걷어내지만, 걷어낸 결과를 저장하지 않으면
106 // 다음 부팅에 그대로 되살아난다. 지운 게 없어도 늘 저장한다.
107 FKismetEditorUtilities::CompileBlueprint(Blueprint);
108 const bool bSaved = SaveBlueprint(Blueprint);
109 UE_LOG(LogTemp, Display,
110 TEXT("RD_WIDGET_VAR_CLEAN %s live=%d added=%d removed=%d saved=%d"),
111 *AssetPath, LiveNames.Num(), Added, Stale.Num(), bSaved ? 1 : 0);
112 }
113 }
114}
