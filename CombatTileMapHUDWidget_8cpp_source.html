237 mSelectedSkillIndex = INDEX_NONE;
238 HideSkillDetail();
239 RefreshSkillRailWidgets();
240 RefreshDiceAssignmentText();
241
242 /*
243 * HUD 루트가 빈 영역(타일맵) 탭도 받아야 NativeOnTouchStarted에서 RequestWorldTouch로
244 * 좌표를 게임플레이에 넘길 수 있다. 스킬레일/주사위 같은 자식 버튼은 여전히 우선 처리되고,
245 * 버튼 밖(월드) 탭만 루트가 받아 좌표를 전달한다. (SelfHitTestInvisible이면 월드 탭이 통과돼버림)
