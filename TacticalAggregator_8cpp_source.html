60float FTacticalAggregator::StaticExecModOnBaseValue(float BaseValue, TEnumAsByte<ETacticalModOp::Type> ModifierOp, float EvaluatedMagnitude)
61{
62 switch (ModifierOp)
63 {
64 case ETacticalModOp::Override:
65 {
66 // 덮어쓰기: 기존 값을 무시하고 크기 값으로 치환한다.
67 BaseValue = EvaluatedMagnitude;
68 break;
69 }
70 case ETacticalModOp::AddBase:
71 case ETacticalModOp::AddFinal:
72 {
73 // 합산 계열(AddBase=베이스 합산, AddFinal=최종 합산): 즉시 적용 경로에서는 둘 다 단순 가산.
74 BaseValue += EvaluatedMagnitude;
75 break;
76 }
77 case ETacticalModOp::MultiplyAdditive:
78 case ETacticalModOp::MultiplyCompound:
79 {
80 // 곱 계열(MultiplyAdditive=배율 가산, MultiplyCompound=거듭제곱 곱): 즉시 적용에선 둘 다 단순 곱.
81 BaseValue *= EvaluatedMagnitude;
82 break;
83 }
84 case ETacticalModOp::DivideAdditive:
85 {
86 // 나눗셈 가산: 0으로 나누기는 무시(값 보존)한다.
87 if (FMath::IsNearlyZero(EvaluatedMagnitude) == false)
88 {
89 BaseValue /= EvaluatedMagnitude;
90 }
91 break;
92 }
93 }
94 return BaseValue;
95}

117void FTacticalAggregator::RemoveAggregatorMod(FActiveTacticalEffectHandle ActiveHandle)
118{
119 checkf(ActiveHandle.IsValid() == true, TEXT("비활성 핸들을 제거하려 시도"));
120
121 for (int32 ModOpIdx = 0; ModOpIdx < UE_ARRAY_COUNT(mMods); ++ModOpIdx)
122 {
123 mMods[ModOpIdx].RemoveAllSwap([&ActiveHandle](const FTacticalAggregatorMod& Element) {
124 return (Element.mActiveHandle == ActiveHandle);
125 }, EAllowShrinking::No);
126 }
127
128 BroadcastOnDirty();

152float FTacticalAggregator::EvaluateWithBase(float BaseValue) const
153{
154 // 덮어쓰기 모디파이어가 존재하면 첫 값으로 즉시 반환
155 for (const FTacticalAggregatorMod& Mod : mMods[ETacticalModOp::Override])
156 {
157 return Mod.mEvaluatedMagnitude;
158 }
159
160 /* 연산 종류별 누적값 계산 */
161
162 float Additive = SumMods(mMods[ETacticalModOp::AddBase], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::AddBase));
163 float Multiplicative = SumMods(mMods[ETacticalModOp::MultiplyAdditive], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::MultiplyAdditive));
164 float Division = SumMods(mMods[ETacticalModOp::DivideAdditive], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::DivideAdditive));
165 float FinalAdd = SumMods(mMods[ETacticalModOp::AddFinal], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::AddFinal));
166 float CompoundMultiply = MultiplyMods(mMods[ETacticalModOp::MultiplyCompound]);
167
168 // 0 나눗셈 방지. 분모가 0에 가까우면 1로 치환
169 if (FMath::IsNearlyZero(Division) == true)
170 {
171 UE_LOG(LogAttributeSetComp, Log, TEXT("속성 변경 시, 변경 값을 0으로 나눌 수 없음. 1로 치환"));
172 Division = 1.f;
173 }
174
175 // 최종 연산자에 따른 수식 계산
176 return ((BaseValue + Additive) * Multiplicative / Division * CompoundMultiply) + FinalAdd;
177}

199void FTacticalAggregator::BroadcastOnDirty()
200{
201 UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(mOwner->GetWorld());
202 checkf(TacticalFrameworkModel != nullptr, TEXT("전략 프레임워크 모델 nullptr"));
203
204 // 배치가 열려 있고 알릴 대상이 있으면 전파를 보류하고 대기열에만 등록
205 if (TacticalFrameworkModel->GetGlobalBatchCount() > 0 && (mDependentEffects.Num() > 0 || OnDirty.IsBound()))
206 {
207 TacticalFrameworkModel->AddAggregatorDirty(this);
208 return;
209 }
210
211 /* 재귀 검사 */
212
213 const int32 MAX_DIRTY = 10;
214 if (mDirtyCount > MAX_DIRTY)
215 {
216 checkf(false, TEXT("재귀적으로 해당 속성을 계속 변경해주고 있음"));
217 return;
218 }
219
220 mDirtyCount++;
221 OnDirty.Broadcast(this);
222
223 /* 값 갱신 요청 */
224
225 TArray<FActiveTacticalEffectHandle> DependentEffectsCopy = mDependentEffects;
226 mDependentEffects.Empty();
227
228 for (FActiveTacticalEffectHandle& Handle : DependentEffectsCopy)
229 {
230 UAttributeSetComponentModel* ASC = Handle.GetOwningAttributeSetComponentModel();
231 if (ASC != nullptr)
232 {
233 /* ASC가 아직 살아있다면 재 등록 */
234
235 ASC->OnMagnitudeDependencyChange(Handle, this);
236 mDependentEffects.Add(Handle);
237 }
238 }
239
240 mDirtyCount--;
241}
