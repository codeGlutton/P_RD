122float FTacticalAggregator::StaticExecModOnBaseValue(float BaseValue, TEnumAsByte<ETacticalModOp::Type> ModifierOp, float EvaluatedMagnitude)
123{
124 switch (ModifierOp)
125 {
126 case ETacticalModOp::Override:
127 {
128 // 덮어쓰기: 기존 값을 무시하고 크기 값으로 치환한다.
129 BaseValue = EvaluatedMagnitude;
130 break;
131 }
132 case ETacticalModOp::AddBase:
133 case ETacticalModOp::AddFinal:
134 {
135 // 합산 계열(AddBase=베이스 합산, AddFinal=최종 합산): 즉시 적용 경로에서는 둘 다 단순 가산.
136 BaseValue += EvaluatedMagnitude;
137 break;
138 }
139 case ETacticalModOp::MultiplyAdditive:
140 case ETacticalModOp::MultiplyCompound:
141 {
142 // 곱 계열(MultiplyAdditive=배율 가산, MultiplyCompound=거듭제곱 곱): 즉시 적용에선 둘 다 단순 곱.
143 BaseValue *= EvaluatedMagnitude;
144 break;
145 }
146 case ETacticalModOp::DivideAdditive:
147 {
148 // 나눗셈 가산: 0으로 나누기는 무시(값 보존)한다.
149 if (FMath::IsNearlyZero(EvaluatedMagnitude) == false)
150 {
151 BaseValue /= EvaluatedMagnitude;
152 }
153 break;
154 }
155 }
156 return BaseValue;
157}

201void FTacticalAggregator::RemoveAggregatorMod(FActiveTacticalEffectHandle ActiveHandle)
202{
203 checkf(ActiveHandle.IsValid() == true, TEXT("비활성 핸들을 제거하려 시도"));
204
205 // UE_ARRAY_COUNT(mMods) == ETacticalModOp::Max. 모든 연산 버킷을 빠짐없이 훑는다.
206 for (int32 ModOpIdx = 0; ModOpIdx < UE_ARRAY_COUNT(mMods); ++ModOpIdx)
207 {
208 mMods[ModOpIdx].RemoveAllSwap([&ActiveHandle](const FTacticalAggregatorMod& Element) {
209 return (Element.mActiveHandle == ActiveHandle);
210 }, EAllowShrinking::No);
211 }
212
213 BroadcastOnDirty();
214}

226void FTacticalAggregator::UpdateAggregatorMod(FActiveTacticalEffectHandle ActiveHandle, const FTacticalAttribute& Attribute, const FTacticalEffectSpec& Spec, FActiveTacticalEffectHandle InHandle)
227{
228 RemoveAggregatorMod(ActiveHandle);
229
230 for (int32 ModIdx = 0; ModIdx < Spec.mModifierValues.Num(); ++ModIdx)
231 {
232 const FTacticalModifierInfo& ModDef = Spec.mEffectClass->mModifiers[ModIdx];
233 // 이 Aggregator가 담당하는 속성을 대상으로 한 모디파이어만 반영한다.
234 if (ModDef.mAttribute == Attribute)
235 {
236 // mModifierOp는 ETacticalModOp(구 EGameplayModOp 치환) — 연산 종류를 그대로 버킷에 전달.
237 AddAggregatorMod(Spec.GetModifierMagnitude(ModIdx), ModDef.mModifierOp, InHandle);
238 }
239 }
240
241 BroadcastOnDirty();
242}

274float FTacticalAggregator::EvaluateWithBase(float BaseValue) const
275{
276 // 1) Override 단락: 덮어쓰기 모디파이어가 존재하면 첫 값으로 즉시 반환(다른 연산 전부 무시).
277 for (const FTacticalAggregatorMod& Mod : mMods[ETacticalModOp::Override])
278 {
279 return Mod.mEvaluatedMagnitude;
280 }
281
282 // 2) 연산 종류별 누적. Bias는 각 연산의 항등값(합산=0, 곱/나눗셈=1)으로, 모디파이어 부재 시 무효원소가 됨.
283 float Additive = SumMods(mMods[ETacticalModOp::Additive], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::Additive));
284 float Multiplicitive = SumMods(mMods[ETacticalModOp::Multiplicitive], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::Multiplicitive));
285 float Division = SumMods(mMods[ETacticalModOp::Division], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::Division));
286 float FinalAdd = SumMods(mMods[ETacticalModOp::AddFinal], TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::AddFinal));
287 // MultiplyCompound는 가산이 아니라 곱(복리)으로 누적 → 별도 MultiplyMods 사용.
288 float CompoundMultiply = MultiplyMods(mMods[ETacticalModOp::MultiplyCompound]);
289
290 // 3) 0 나눗셈 방지: 분모가 0에 가까우면 1로 치환(값 보존).
291 if (FMath::IsNearlyZero(Division) == true)
292 {
293 UE_LOG(LogAttributeSetComp, Log, TEXT("속성 변경 시, 변경 값을 0으로 나눌 수 없음. 1로 치환"));
294 Division = 1.f;
295 }
296 // 최종 합성 수식: 베이스 합산 → 배율/나눗셈/복리곱 → 최종 합산 순.
297 return ((BaseValue + Additive) * Multiplicitive / Division * CompoundMultiply) + FinalAdd;
298}

310float FTacticalAggregator::SumMods(const TArray<FTacticalAggregatorMod>& Mods, float Bias)
311{
312 float Sum = Bias;
313 for (const FTacticalAggregatorMod& Mod : Mods)
314 {
315 // (magnitude - Bias)를 더해 항등값 중복을 제거 → 모디파이어 부재 시 정확히 Bias가 남는다.
316 Sum += (Mod.mEvaluatedMagnitude - Bias);
317 }
318 return Sum;
319}

349void FTacticalAggregator::BroadcastOnDirty()
350{
351 UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(mOwner->GetWorld());
352 checkf(TacticalFrameworkModel != nullptr, TEXT("전략 프레임워크 모델 nullptr"));
353
354 // 배치가 열려 있고 알릴 대상이 있으면 즉시 전파를 보류하고 대기열에만 등록한다.
355 if (TacticalFrameworkModel->GetGlobalBatchCount() > 0 && (mDependentEffects.Num() > 0 || OnDirty.IsBound()))
356 {
357 TacticalFrameworkModel->AddAggregatorDirty(this);
358 return;
359 }
360
361 /* 재귀 검사 */
362
363 // 리스너 콜백이 다시 이 속성을 변경하는 무한 순환을 깊이 제한으로 차단한다.
364 const int32 MAX_DIRTY = 10;
365 if (mDirtyCount > MAX_DIRTY)
366 {
367 checkf(false, TEXT("재귀적으로 해당 속성을 계속 변경해주고 있음"));
368 return;
369 }
370
371 mDirtyCount++;
372 OnDirty.Broadcast(this);
373
374 /* 값 갱신 요청 */
375
376 // 순회 중 mDependentEffects가 변형될 수 있으므로 복사본을 떠서 돌고, 살아있는 항목만 재등록한다.
377 TArray<FActiveTacticalEffectHandle> DependentEffectsCopy = mDependentEffects;
378 mDependentEffects.Empty();
379
380 for (FActiveTacticalEffectHandle& Handle : DependentEffectsCopy)
381 {
382 UAttributeSetComponentModel* ASC = Handle.GetOwningAttributeSetComponentModel();
383 if (ASC != nullptr)
384 {
385 /* ASC가 아직 살아있다면 재 등록 */
386
387 // 의존 이펙트에 크기 재계산을 알리고, 유효한 핸들만 다시 의존 목록에 보존한다(소멸된 ASC는 누락 → 정리).
388 ASC->OnMagnitudeDependencyChange(Handle, this);
389 mDependentEffects.Add(Handle);
390 }
391 }
392
393 mDirtyCount--;
394}
