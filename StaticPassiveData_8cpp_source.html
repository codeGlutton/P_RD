50 void ValidateOperand(const FPassiveOperand& Operand, const FString& Label, const UStaticPassiveData& Data, FDataValidationContext& Context)
51 {
52 switch (Operand.mKind)
53 {
54 // 속성값: 읽을 속성 필수
55 case EPassiveOperandKind::Attribute:
56 if (Operand.mAttribute.IsValid() == false)
57 {
58 Context.AddError(FText::FromString(Label + TEXT(": 속성 미지정")));
59 }
60 break;
61
62 // 태그 개수: 집계할 태그 필수
63 case EPassiveOperandKind::TagCount:
64 if (Operand.mTag.IsValid() == false)
65 {
66 Context.AddError(FText::FromString(Label + TEXT(": 태그 미지정")));
67 }
68 break;
69
70 // 캡처값: 키 필수, 캡처 시점과 캡처 목록에 짝이 있어야 함
71 case EPassiveOperandKind::Captured:
72 if (Operand.mCaptureKey.IsNone())
73 {
74 Context.AddError(FText::FromString(Label + TEXT(": 캡처 키 미지정")));
75 }
76 else
77 {
78 if (Data.mCaptureTimingTag.IsValid() == false)
79 {
80 Context.AddError(FText::FromString(Label + TEXT(": 캡처값을 쓰는데 캡처 시점이 비어있음")));
81 }
82 const bool bKeyDefined = Data.mCaptureOperands.ContainsByPredicate(
83 [&Operand](const FPassiveCaptureEntry& Entry) { return Entry.mKey == Operand.mCaptureKey; });
84 if (bKeyDefined == false)
85 {
86 Context.AddError(FText::FromString(FString::Printf(TEXT("%s: 캡처 키 '%s'가 캡처 목록에 없음"), *Label, *Operand.mCaptureKey.ToString())));
87 }
88 }
89 break;
90
91 // 이동 거리: 캡처 시점의 타일을 읽으므로 캡처 시점 필수
92 case EPassiveOperandKind::MovedDistance:
93 if (Data.mCaptureTimingTag.IsValid() == false)
94 {
95 Context.AddError(FText::FromString(Label + TEXT(": 이동 거리를 쓰는데 캡처 시점이 비어있음")));
96 }
97 break;
98
99 // 카운터: 리셋 시점이 없으면 전투를 넘어 계속 누적되므로 경고
100 case EPassiveOperandKind::Counter:
101 if (Data.mCounterResetTimingTag.IsValid() == false)
102 {
103 Context.AddWarning(FText::FromString(Label + TEXT(": 카운터를 쓰는데 리셋 시점이 비어있음")));
104 }
105 break;
106
107 // 커스텀: 계산 클래스 필수
108 case EPassiveOperandKind::Custom:
109 if (Operand.mCustomClass == nullptr)
110 {
111 Context.AddError(FText::FromString(Label + TEXT(": 커스텀 클래스 미지정")));
112 }
113 break;
114
115 default:
116 break;
117 }
118 }

121EDataValidationResult UStaticPassiveData::IsDataValid(FDataValidationContext& Context) const
122{
123 const EDataValidationResult SuperResult = Super::IsDataValid(Context);
124
125 // 발동 시점은 필수
126 if (mActivateTimingTag.IsValid() == false)
127 {
128 Context.AddError(FText::FromString(TEXT("발동 시점이 비어있음")));
129 }
130
131 // 타이밍 태그 4종의 계열 검사
132 ValidateTimingTag(mActivateTimingTag, TEXT("발동 시점"), Context);
133 ValidateTimingTag(mDeactivateTimingTag, TEXT("해제 시점"), Context);
134 ValidateTimingTag(mCounterResetTimingTag, TEXT("리셋 시점"), Context);
135 ValidateTimingTag(mCaptureTimingTag, TEXT("캡처 시점"), Context);
136
137 // 캡처 목록이 있으면 캡처 시점도 필수
138 if (mCaptureOperands.Num() > 0 && mCaptureTimingTag.IsValid() == false)
139 {
140 Context.AddError(FText::FromString(TEXT("캡처 목록이 있는데 캡처 시점이 비어있음")));
141 }
142
143 // 캡처한 타겟에게 발동한다면, 타겟을 모을 캡처 타이밍 태그가 있는 지 확인
144 if (mActivateOnCapturedTargets && mCaptureTimingTag.IsValid() == false)
145 {
146 Context.AddError(FText::FromString(TEXT("캡처한 타겟에게 발동하는데 캡처 시점이 비어있음")));
147 }
148
149 // 캡처 엔트리 검사
150 for (int32 Index = 0; Index < mCaptureOperands.Num(); ++Index)
151 {
152 const FPassiveCaptureEntry& Entry = mCaptureOperands[Index];
153 const FString Prefix = FString::Printf(TEXT("캡처[%d]"), Index);
154
155 // 저장 키 필수
156 if (Entry.mKey.IsNone())
157 {
158 Context.AddError(FText::FromString(Prefix + TEXT(": 키 미지정")));
159 }
160
161 // 캡처는 속성값과 태그 개수만 허용
162 if (Entry.mOperand.mKind != EPassiveOperandKind::Attribute
163 && Entry.mOperand.mKind != EPassiveOperandKind::TagCount)
164 {
165 Context.AddError(FText::FromString(Prefix + TEXT(": 캡처는 속성값과 태그 개수만 허용")));
166 }
167
168 ValidateOperand(Entry.mOperand, Prefix + TEXT(" 피연산자"), *this, Context);
169 }
170
171 // 조건 검사
172 for (int32 Index = 0; Index < mConditions.Num(); ++Index)
173 {
174 const FPassiveCondition& Condition = mConditions[Index];
175 const FString Prefix = FString::Printf(TEXT("조건[%d]"), Index);
176
177 ValidateOperand(Condition.mLhs, Prefix + TEXT(" 좌변"), *this, Context);
178 ValidateOperand(Condition.mRhs, Prefix + TEXT(" 우변"), *this, Context);
179
180 // 배수 연산의 경우에만 우변이 0 이상인 지 체크
181 if (Condition.mOp == EPassiveCompareOp::ModuloZero
182 && Condition.mRhs.mKind == EPassiveOperandKind::Const
183 && Condition.mRhs.mConst <= 0.f)
184 {
185 Context.AddError(FText::FromString(Prefix + TEXT(": 배수 연산의 우변이 0 이하")));
186 }
187 }
188
189 // 효과가 없으면 아무 일도 하지 않는 패시브이므로 경고
190 if (mEffects.Num() == 0)
191 {
192 Context.AddWarning(FText::FromString(TEXT("효과 목록이 비어있음")));
193 }
194
195 // 해제 시점 있으면 이펙트를 직접 사용해야 함: 그래야 건 것만큼 뺄 수 있음
196 bool bHasRemovableEffect = false;
197
198 // 효과 검사
199 for (int32 Index = 0; Index < mEffects.Num(); ++Index)
200 {
201 const FPassiveEffectEntry& Entry = mEffects[Index];
202 const FString Prefix = FString::Printf(TEXT("효과[%d]"), Index);
203
204 // 이펙트 클래스 필수
205 if (Entry.mEffectClass.IsNull())
206 {
207 Context.AddError(FText::FromString(Prefix + TEXT(": 이펙트 클래스 미지정")));
208 }
209 else
210 {
211 // 실시간 계산형 이펙트는 시전자 스냅샷을 요구하는데 패시브에는 아직 없으므로 막아 둠
212 // TODO: NotifyPassive에 스냅샷 첨부하면 이 검사는 제거
213 const UClass* EffectClass = Entry.mEffectClass.LoadSynchronous();
214 const UTacticalEffect* EffectCDO = (EffectClass != nullptr) ? EffectClass->GetDefaultObject<UTacticalEffect>() : nullptr;
215 if (EffectCDO != nullptr && EffectCDO->mExecutions.Num() > 0)
216 {
217 Context.AddError(FText::FromString(Prefix + TEXT(": 실시간 계산형 이펙트는 패시브에서 사용 불가")));
218 }
219
220 // 즉시형이 아닌 효과는 활성 이펙트로 남아 해제 가능
221 if (EffectCDO != nullptr && EffectCDO->mDurationPolicy != ETacticalEffectDurationType::Instant)
222 {
223 bHasRemovableEffect = true;
224 }
225
226 // 해제 시점 없으면 Get* 형태의 이펙트를 사용해야 함
227 if (EffectCDO != nullptr && EffectCDO->IsA<UTacticalEffect_Status>())
228 {
229 if (mDeactivateTimingTag.IsValid() == false)
230 {
231 Context.AddWarning(FText::FromString(Prefix + TEXT(": 해제 시점 없이 상태이상 클래스 직접 지정. Get 계열 이펙트 사용")));
232 }
233
234 // 상태이상은 수치가 스택 수가 됨. 고정값이면 1 이상인지 확인
235 if (Entry.mMagnitude.mKind == EPassiveOperandKind::Const && FMath::FloorToInt(Entry.mMagnitude.mConst) <= 0)
236 {
237 Context.AddError(FText::FromString(Prefix + TEXT(": 상태이상 스택 수는 1 이상")));
238 }
239 }
240 }
241
242 ValidateOperand(Entry.mMagnitude, Prefix + TEXT(" 수치"), *this, Context);
243 }
244
245 // 해제 시점이 있는데 효과가 전부 즉시형이면 해제할 것이 없으므로 경고
246 if (mDeactivateTimingTag.IsValid() && mEffects.Num() > 0 && bHasRemovableEffect == false)
247 {
248 Context.AddWarning(FText::FromString(TEXT("해제 시점이 있는데 효과가 전부 즉시형이라 해제할 것이 없음")));
249 }
250
251 // 패시브 클래스 필수
252 if (mPassiveClass.IsNull())
253 {
254 Context.AddError(FText::FromString(TEXT("패시브 클래스 미지정")));
255 }
256
257 // 에러가 있으면 검증 실패
258 return (Context.GetNumErrors() > 0) ? EDataValidationResult::Invalid : SuperResult;
259}
