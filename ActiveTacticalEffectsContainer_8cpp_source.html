122void FActiveTacticalEffectsContainer::DecrementLock()
123{
124 if (--mScopedLockCount == 0)
125 {
126 /* 추가 작업 */
127
128 // 지연 추가 연결 리스트를 head부터 tail 직전까지 순회한다.
129 FActiveTacticalEffect* CurPendingEffect = mPendingTacticalEffectHead;
130 FActiveTacticalEffect* EndPendingEffect = *mPendingTacticalEffectTail;
131
132 while (CurPendingEffect != EndPendingEffect)
133 {
134 if (CurPendingEffect->mIsPendingRemove == false)
135 {
136 /* 추가가 미루어진 이펙트 추가 */
137
138 mTacticalEffects.Add(MoveTemp(*CurPendingEffect));
139 }
140 else
141 {
142 /* 추가가 미루어진 이펙트는 이미 제거됨 */
143
144 // 추가되기도 전에 제거 표시된 노드는 본 배열로 옮기지 않고, 예약 제거 카운트만 상쇄한다.
145 mPendingRemoveCount--;
146 }
147 CurPendingEffect = CurPendingEffect->mPendingNext;
148 }
149 // 지연 리스트를 다시 빈 상태로 되돌린다(노드 메모리는 재사용을 위해 유지).
150 mPendingTacticalEffectTail = &mPendingTacticalEffectHead;
151
152 /* 제거 작업 */
153
154 // RemoveAtSwap을 쓰므로 뒤에서 앞으로(역순) 순회해야 인덱스가 흔들리지 않는다.
155 for (int32 index = mTacticalEffects.Num() - 1; index >= 0 && mPendingRemoveCount > 0; --index)
156 {
157 FActiveTacticalEffect& Effect = mTacticalEffects[index];
158
159 if (Effect.mIsPendingRemove)
160 {
161 // 해당 이펙트 핸들이 사라짐을 알리기
162 Effect.mHandle.RemoveFromGlobalMap();
163
164 mTacticalEffects.RemoveAtSwap(index, EAllowShrinking::No);
165
166 mPendingRemoveCount--;
167 }
168 }
169
170 // 위 두 루프가 끝나면 예약된 제거가 정확히 모두 소진되어 있어야 한다.
171 // [방어 가드] 위 가드로 flush가 중단된 손상 상황에서 checkf 어설트가 2차 크래시를 내지 않도록,
172 // 불변식 위반을 기록 후 복구한다(정상 경로에선 원본 checkf와 동일하게 침묵).
173 if (mPendingRemoveCount != 0)
174 {
175 UE_LOG(LogTemp, Error, TEXT("[TASDBG] DecrementLock: pendingRemove 잔존 %d — 예약된 이펙트 증감 로직 에러(원본 checkf 지점)"), mPendingRemoveCount);
176 mPendingRemoveCount = 0;
177 }
178 }
179}

316FActiveTacticalEffect* FActiveTacticalEffectsContainer::ApplyTacticalEffectSpec(const FTacticalEffectSpec& Spec, bool& FoundExistingStackableGE)
317{
318 // 적용 과정 동안 배열 변동을 막기 위해 Scope Lock을 건다(추가는 필요 시 지연 처리됨).
319 TACTICAL_EFFECT_SCOPE_LOCK();
320
321 checkf(Spec.mEffectClass != nullptr, TEXT("적용하려는 Effect 클래스가 없음"));
322 FoundExistingStackableGE = false;
323
324 FActiveTacticalEffect* AppliedActiveEffect = nullptr;
325 FActiveTacticalEffect* ExistingStackableEffect = FindStackableActiveTacticalEffect(Spec);
326
327 int32 PreStackCount = 0;
328 int32 NewStackCount = 0;
329
330 if (ExistingStackableEffect != nullptr)
331 {
332 /* 스태킹 Effect의 경우 */
333
334 FoundExistingStackableGE = true;
335
336 FTacticalEffectSpec& ExistingSpec = ExistingStackableEffect->mSpec;
337 PreStackCount = ExistingSpec.GetStackCount();
338 NewStackCount = ExistingSpec.GetStackCount() + Spec.GetStackCount();
339
340 checkf(ExistingSpec.mDynamicMagnitude == Spec.mDynamicMagnitude, TEXT("스태킹되는 이펙트는 수치가 다를 수 없음"));
341
342 ExistingStackableEffect->mSpec = Spec;
343 ExistingStackableEffect->mSpec.SetStackCount(NewStackCount);
344
345 AppliedActiveEffect = ExistingStackableEffect;
346 }
347 else
348 {
349 /* 일반 Effect의 경우 */
350
351 FActiveTacticalEffectHandle NewHandle = FActiveTacticalEffectHandle::GenerateNewHandle(mOwner->GetWorld(), mOwner.Get());
352 if (mTacticalEffects.GetSlack() <= 0)
353 {
354 /* 배열 크기 변동으로 재할당이 필요한 경우 */
355
356 check(mPendingTacticalEffectTail != nullptr);
357
358 if (*mPendingTacticalEffectTail == nullptr)
359 {
360 /* 비어있는 경우 */
361
362 AppliedActiveEffect = new FActiveTacticalEffect(NewHandle, Spec);
363 *mPendingTacticalEffectTail = AppliedActiveEffect;
364 }
365 else
366 {
367 /* 할당 공간이 남아있는 경우 */
368
369 **mPendingTacticalEffectTail = FActiveTacticalEffect(NewHandle, Spec);
370 AppliedActiveEffect = *mPendingTacticalEffectTail;
371 }
372 mPendingTacticalEffectTail = &AppliedActiveEffect->mPendingNext;
373 }
374 else
375 {
376 /* 배열 크기 재할당이 필요없는 경우 */
377
378 AppliedActiveEffect = new(mTacticalEffects) FActiveTacticalEffect(NewHandle, Spec);
379 }
380 }
381
382 // 적용된 Spec의 각 모디파이어 최종 크기를 미리 계산해 둔다.
383 FTacticalEffectSpec& AppliedEffectSpec = AppliedActiveEffect->mSpec;
384 AppliedEffectSpec.CalculateModifierMagnitudes();
385
386 if (ExistingStackableEffect != nullptr)
387 {
388 // 기존 이펙트에 스택이 합쳐진 경우: 스택 변경 후처리만 수행.
389 OnStackCountChange(*ExistingStackableEffect, PreStackCount, NewStackCount);
390 }
391 else
392 {
393 // 새 이펙트가 추가된 경우: 태그/모디파이어 부여 등 추가 후처리 수행.
394 InternalOnActiveTacticalEffectAdded(*AppliedActiveEffect);
395 }
396
397 return AppliedActiveEffect;
398}

406void FActiveTacticalEffectsContainer::ExecuteActiveEffectsFrom(FTacticalEffectSpec& Spec)
407{
408 if (mOwner.IsValid() == false)
409 {
410 return;
411 }
412
413 FTacticalEffectSpec& SpecToUse = Spec;
414 SpecToUse.CalculateModifierMagnitudes();
415
416 /* 모디파이어 실행 */
417
418 bool ModifierSuccessfullyExecuted = false;
419
420 check(SpecToUse.mModifierValues.Num() == SpecToUse.mEffectClass->mModifiers.Num());
421 for (int32 ModIdx = 0; ModIdx < SpecToUse.mModifierValues.Num(); ++ModIdx)
422 {
423 const FTacticalModifierInfo& ModDef = SpecToUse.mEffectClass->mModifiers[ModIdx];
424
425 // [PR #191 enum 치환] ModDef.mModifierOp는 ETacticalModOp(구 EGameplayModOp 대체). 평가 데이터에 그대로 실어
426 // InternalExecuteMod -> ApplyModToAttribute로 전달되며, 거기서 op별 연산식이 base 값에 적용된다.
427 FTacticalModifierEvaluatedData EvalData(ModDef.mAttribute, ModDef.mModifierOp, SpecToUse.GetModifierMagnitude(ModIdx));
428 ModifierSuccessfullyExecuted |= InternalExecuteMod(SpecToUse, EvalData);
429 }
430
431 Spec.mEffectClass->OnExecuted(*this, Spec);
432}

550bool FActiveTacticalEffectsContainer::InternalRemoveActiveTacticalEffect(int32 Idx, int32 StacksToRemove, bool bPrematureRemoval)
551{
552 // Lock이 걸려 있으면 즉시 배열 제거 대신 지연 제거 경로를 탄다.
553 bool IsLocked = (mScopedLockCount > 0);
554 TACTICAL_EFFECT_SCOPE_LOCK();
555
556 if (Idx < GetNumTacticalEffects())
557 {
558 FActiveTacticalEffect& Effect = *GetActiveTacticalEffect(Idx);
559 if (!ensure(!Effect.mIsPendingRemove))
560 {
561 return true;
562 }
563
564 FTacticalEffectRemovalInfo GameplayEffectRemovalInfo;
565 GameplayEffectRemovalInfo.mActiveEffect = &Effect;
566 GameplayEffectRemovalInfo.mStackCount = Effect.mSpec.GetStackCount();
567 GameplayEffectRemovalInfo.mEffectContext = Effect.mSpec.GetContext();
568
569 if (StacksToRemove > 0 && Effect.mSpec.GetStackCount() > StacksToRemove)
570 {
571 // 보유 스택보다 적게 제거하는 경우: 이펙트는 유지하고 스택 수만 감소시킨다.
572 int32 StartingStackCount = Effect.mSpec.GetStackCount();
573 Effect.mSpec.SetStackCount(StartingStackCount - StacksToRemove);
574 OnStackCountChange(Effect, StartingStackCount, Effect.mSpec.GetStackCount());
575 return false;
576 }
577
578 InternalOnActiveTacticalEffectRemoved(Effect, GameplayEffectRemovalInfo);
579
580 bool ModifiedArray = false;
581 if (IsLocked == true)
582 {
583 // Lock 중: 즉시 제거하지 않고 카운트만 올려 Lock 해제 시 일괄 제거(DecrementLock)되게 한다.
584 mPendingRemoveCount++;
585 }
586 else
587 {
588 // Lock이 없으면 즉시 글로벌 맵에서 핸들을 지우고 배열에서 swap 제거.
589 Effect.mHandle.RemoveFromGlobalMap();
590 check(Idx < mTacticalEffects.Num());
591 mTacticalEffects.RemoveAtSwap(Idx);
592 ModifiedArray = true;
593 }
594 return ModifiedArray;
595 }
596
597 return false;
598}

610void FActiveTacticalEffectsContainer::AddActiveTacticalEffectGrantedTagsAndModifiers(FActiveTacticalEffect& Effect)
611{
612 check(Effect.mSpec.mEffectClass != nullptr);
613 check(mOwner != nullptr);
614
615 TACTICAL_EFFECT_SCOPE_LOCK();
616
617 for (int32 ModIdx = 0; ModIdx < Effect.mSpec.mModifierValues.Num(); ++ModIdx)
618 {
619 if (Effect.mSpec.mEffectClass->mModifiers.IsValidIndex(ModIdx) == false)
620 {
621 continue;
622 }
623
624 const FTacticalModifierInfo& ModInfo = Effect.mSpec.mEffectClass->mModifiers[ModIdx];
625 // 소유자가 해당 속성의 AttributeSet을 갖고 있지 않으면 등록을 건너뛴다.
626 if (mOwner->HasAttributeSetForAttribute(ModInfo.mAttribute) == false)
627 {
628 continue;
629 }
630
631 float EvaluatedMagnitude = Effect.mSpec.GetModifierMagnitude(ModIdx);
632 FTacticalAggregator* Aggregator = FindOrCreateAttributeAggregator(Effect.mSpec.mEffectClass->mModifiers[ModIdx].mAttribute).Get();
633 if (ensure(Aggregator))
634 {
635 // op(ETacticalModOp)별 버킷에 모디파이어를 추가. 이펙트 핸들로 추후 제거 식별이 가능하다.
636 Aggregator->AddAggregatorMod(EvaluatedMagnitude, ModInfo.mModifierOp, Effect.mHandle);
637 }
638 }
639
640 // 부여 태그를 +1 카운트로 태그 맵에 반영하고, 추가 사실을 소유자에게 알린다.
641 mOwner->UpdateTagMap(Effect.mSpec.mEffectClass->GetGrantedTags(), 1);
642 mOwner->OnActiveTacticalEffectAddedDelegateToSelf.Broadcast(mOwner.Get(), Effect.mSpec, Effect.mHandle);
643}

650void FActiveTacticalEffectsContainer::RemoveActiveTacticalEffectGrantedTagsAndModifiers(const FActiveTacticalEffect& Effect)
651{
652 for (const FTacticalModifierInfo& Mod : Effect.mSpec.mEffectClass->mModifiers)
653 {
654 if (Mod.mAttribute.IsValid() == true)
655 {
656 if (const TSharedPtr<FTacticalAggregator>* RefPtr = mAttributeAggregatorMap.Find(Mod.mAttribute))
657 {
658 // 이 이펙트 핸들로 등록됐던 모디파이어들을 op 버킷에서 제거(추가 시의 핸들과 매칭).
659 RefPtr->Get()->RemoveAggregatorMod(Effect.mHandle);
660 }
661 }
662 }
663 // 추가 때 +1 했던 부여 태그를 -1로 상쇄한다.
664 mOwner->UpdateTagMap(Effect.mSpec.mEffectClass->GetGrantedTags(), -1);

675void FActiveTacticalEffectsContainer::SetAttributeBaseValue(FTacticalAttribute Attribute, float BaseValue)
676{
677 checkf(mOwner != nullptr, TEXT("변경 ASC 대상이 존재하지 않음"));
678 const UTacticalAttributeSet* Set = mOwner->GetAttributeSet_Internal(Attribute.GetAttributeSetClass());
679 checkf(Set != nullptr, TEXT("변경 AttributeSet 대상이 존재하지 않음"));
680
681 float OldBaseValue = 0.0f;
682 Set->PreAttributeBaseChange(Attribute, BaseValue);
683
684 /* 속성에서 베이스 값 변경 */
685
686 const FStructProperty* StructProperty = CastField<FStructProperty>(Attribute.GetUProperty());
687 checkf(StructProperty != nullptr, TEXT("변경 속성 대상이 존재하지 않음"));
688 // 속성 데이터는 FTacticalAttributeData(virtual 보유로 vptr이 있어 FGameplayAttributeData와 메모리 레이아웃이 다름).
689 // 반드시 FTacticalAttributeData로 접근해야 base 값이 올바른 오프셋에 읽고 쓰인다.
690 FTacticalAttributeData* DataPtr = StructProperty->ContainerPtrToValuePtr<FTacticalAttributeData>(const_cast<UTacticalAttributeSet*>(Set));
691 checkf(DataPtr != nullptr, TEXT("변경 FTacticalAttributeData 대상이 존재하지 않음"));
692 OldBaseValue = DataPtr->GetBaseValue();
693 DataPtr->SetBaseValue(BaseValue);
694
695 /* 새로운 베이스 값으로 현재 값 계산 */
696
697 TSharedPtr<FTacticalAggregator>* RefPtr = mAttributeAggregatorMap.Find(Attribute);
698 if (RefPtr != nullptr)
699 {
700 FTacticalAggregator* Aggregator = RefPtr->Get();
701 checkf(Aggregator != nullptr, TEXT("변경 FTacticalAggregator 대상이 존재하지 않음"));
702
703 OldBaseValue = Aggregator->GetAttributeBaseValue();
704 Aggregator->SetAttributeBaseValue(BaseValue);
705 }
706 else
707 {
708 // 추가 값이 없다면, 베이스 값이 곧 현재 값
709 UpdateAttributeCurrentValue(Attribute, BaseValue);
710 }
711
712 Set->PostAttributeBaseChange(Attribute, OldBaseValue, GetAttributeBaseValue(Attribute));
713}
