45void UTacticalEffectExecutionCalculation_SystemHeal::Execute(const FTacticalEffectCustomExecutionParameters& ExecutionParams, FTacticalEffectCustomExecutionOutput& OutExecutionOutput) const
46{
47 Super::Execute(ExecutionParams, OutExecutionOutput);
48
49 UAttributeSetComponentModel* TargetAttributeSetCompModel = ExecutionParams.GetTargetAttributeSetComponentModel();
50 checkf(TargetAttributeSetCompModel != nullptr, TEXT("소스 컴포넌트 모델 nullptr"));
51
52 const float MaxHP = TargetAttributeSetCompModel->GetAttributeCurrentValue(UCombatTargetAttributeSet::GetMaxHPAttribute());
53 const float BaseHeal = MaxHP * FMath::Max(0.f, GetHealRatio());
54
55 // 최종 힐
56 const int32 TotalHeal =
57 FMath::Floor(
58 ExecutionParams.GetOwningSpec().GetStackCount() *
59 ExecutionParams.GetOwningSpec().mDynamicMagnitude *
60 BaseHeal
61 );
62
63 /* 체력 힐 */
64 {
65 const float HPDiff = TotalHeal;
66 if (HPDiff > 0)
67 {
68 OutExecutionOutput.AddOutputModifier(FTacticalModifierEvaluatedData(UCombatTargetAttributeSet::GetHPAttribute(), ETacticalModOp::AddBase, HPDiff));
69 }
70 }
71
72 OutExecutionOutput.MarkDynamicMagnitudeHandledManually();
73 OutExecutionOutput.MarkStackCountHandledManually();
74}

199void UTacticalEffectExecutionCalculation_Attack::Execute(const FTacticalEffectCustomExecutionParameters& ExecutionParams, FTacticalEffectCustomExecutionOutput& OutExecutionOutput) const
200{
201 Super::Execute(ExecutionParams, OutExecutionOutput);
202
203 const UGameBalanceSettings* GameBalanceSettings = GetDefault<UGameBalanceSettings>();
204 checkf(GameBalanceSettings != nullptr, TEXT("게임 밸런스 세팅 nullptr"));
205
206 UBoardCombatTargetSnapshotData* SourceSnapshotData = ExecutionParams.GetOwningSpec().GetInstigatorSnapshotData();
207 checkf(SourceSnapshotData != nullptr, TEXT("소스 스냅샷 nullptr"));
208
209 UBoardCombatTargetSnapshotData* TargetSnapshotData = ExecutionParams.GetOwningSpec().GetTargetSnapshotData();
210 checkf(TargetSnapshotData != nullptr, TEXT("타겟 스냅샷 nullptr"));
211
212 // 약화
213 const bool IsOwnerWeakness = SourceSnapshotData->mEffectCounts.Contains(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Weakness);
214 const float OwnerWeaknessRatio = IsOwnerWeakness == true ? GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Weakness] : 1.f;
215
216 // 취약
217 const bool IsTargetVulnerability = TargetSnapshotData->mEffectCounts.Contains(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Vulnerability);
218 const float TargetVulnerabilityRatio = IsTargetVulnerability == true ? GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Vulnerability] : 1.f;
219
220 // 최종 공격력과 방어력
221 const int32 TotalAttack =
222 FMath::Floor(
223 ExecutionParams.GetOwningSpec().GetStackCount() *
224 ExecutionParams.GetOwningSpec().mDynamicMagnitude *
225 TargetVulnerabilityRatio *
226 OwnerWeaknessRatio *
227 SourceSnapshotData->mAttributes[UCombatTargetAttributeSet::GetAttackFactorAttribute()]
228 );
229 const int32 TotalDefense =
230 FMath::Floor(
231 TargetSnapshotData->mAttributes[UCombatTargetAttributeSet::GetDefenseAttribute()]
232 );
233
234 /* 방어력 까기 */
235 {
236 const float DefenseDiff = -FMath::Min(TotalAttack, TotalDefense);
237 if (DefenseDiff < 0)
238 {
239 OutExecutionOutput.AddOutputModifier(FTacticalModifierEvaluatedData(UCombatTargetAttributeSet::GetDefenseAttribute(), ETacticalModOp::AddBase, DefenseDiff));
240
241 /* 로그 적용 */
242 FSRPGAttributeEffectEventLog Log;
243 Log.mEffectAttribute = UCombatTargetAttributeSet::GetDefenseAttribute();
244 Log.mMagnitude = DefenseDiff;
245
246 UAttributeSetComponentModel* TargetAttributeSetCompModel = ExecutionParams.GetTargetAttributeSetComponentModel();
247 const UActorModel* Target = TargetAttributeSetCompModel->GetOwnerModel();
248 GetWorldEventLogger(Target)->LogAttributeEffect(Target->GetModelId(), Target->GetClass(), Log);
249 }
250 }
251 /* 체력 까기 */
252 {
253 const float HPDiff = TotalDefense - TotalAttack;
254 if (HPDiff < 0)
255 {
256 OutExecutionOutput.AddOutputModifier(FTacticalModifierEvaluatedData(UCombatTargetAttributeSet::GetHPAttribute(), ETacticalModOp::AddBase, HPDiff));
257
258 /* 로그 적용 */
259 FSRPGAttributeEffectEventLog Log;
260 Log.mEffectAttribute = UCombatTargetAttributeSet::GetHPAttribute();
261 Log.mMagnitude = HPDiff;
262 Log.mIsCritical = ExecutionParams.GetOwningSpec().mIsCritical;
263
264 UAttributeSetComponentModel* TargetAttributeSetCompModel = ExecutionParams.GetTargetAttributeSetComponentModel();
265 const UActorModel* Target = TargetAttributeSetCompModel->GetOwnerModel();
266 GetWorldEventLogger(Target)->LogAttributeEffect(Target->GetModelId(), Target->GetClass(), Log);
267 }
268 }
269
270 OutExecutionOutput.MarkDynamicMagnitudeHandledManually();
271 OutExecutionOutput.MarkStackCountHandledManually();
272}
