109TArray<FPlayerLevelUpData> UPlayerUnitModel::CalculateLevelChange(int32 StartLevel, float StartExp, float ExpGain) const
110{
111 TArray<FPlayerLevelUpData> Result;
112
113 if (ExpGain <= 0.f)
114 {
115 return Result;
116 }
117
118 const int32 SafeMaxLevel = FMath::Max(1, ULevelAttributeSet::GetMaxLevel(this));
119
120 const int32 SafeStartLevel = FMath::Max(1, StartLevel);
121 const float SafeStartExp = FMath::Max(0.f, StartExp);
122 const float SafeExpGain = FMath::Max(0.f, ExpGain);
123
124 int32 CurLevel = SafeStartLevel;
125 float CurStartExp = SafeStartExp;
126 float CurCarryExp = SafeStartExp + SafeExpGain;
127
128 while (SafeMaxLevel > CurLevel)
129 {
130 float CurMaxExp = ULevelAttributeSet::GetMaxExp(this, CurLevel);
131 if (FMath::IsWithinInclusive(CurMaxExp, 1.f, CurCarryExp) == false)
132 {
133 break;
134 }
135
136 FPlayerLevelUpData LevelUpData;
137 LevelUpData.mMaxExp = CurMaxExp;
138
139 LevelUpData.mPreLevel = CurLevel;
140 LevelUpData.mPreExp = CurStartExp;
141
142 {
143 ++CurLevel;
144 CurCarryExp -= CurMaxExp;
145 CurStartExp = 0.f;
146 }
147
148 LevelUpData.mCurLevel = CurLevel;
149 LevelUpData.mCurExp = CurMaxExp;
150 LevelUpData.mCarryExp = CurCarryExp;
151
152 Result.Add(LevelUpData);
153 }
154
155 return Result;
156}
