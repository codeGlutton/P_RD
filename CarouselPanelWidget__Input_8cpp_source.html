70bool UCarouselPanelWidget::BeginCarouselPress(const FVector2D& ScreenPosition)
71{
72 /*
73 * 눌림 시작 시점에는 선택을 확정하지 않는다.
74 * 사용자가 손가락을 살짝 움직인 뒤 놓는 경우를 스크롤/드래그로 볼 수 있어,
75 * FinishCarouselPress()에서 같은 카드 위에서 짧게 끝났는지 다시 확인한다.
76 */
77 const int32 ItemIndex = FindCarouselItemIndexAtPosition(ScreenPosition);
78 if (ItemIndex == INDEX_NONE)
79 {
80 return false;
81 }
82
83 mPressedCarouselIndex = ItemIndex;
84 mCarouselPressPosition = ScreenPosition;
85 return true;
86}

88bool UCarouselPanelWidget::FinishCarouselPress(const FVector2D& ScreenPosition)
89{
90 /*
91 * Press 시작 카드와 Release 카드가 같고 이동 거리가 작을 때만 탭으로 인정한다.
92 * 터치 스크롤이나 주사위 드래그가 단순 선택으로 오인되지 않게 하기 위한 기준이다.
93 */
94 const int32 ReleasedItemIndex = FindCarouselItemIndexAtPosition(ScreenPosition);
95 const bool bIsTap = mPressedCarouselIndex != INDEX_NONE
96 && ReleasedItemIndex == mPressedCarouselIndex
97 && FVector2D::Distance(mCarouselPressPosition, ScreenPosition) <= FCarouselLayoutPolicy::TapDistanceThreshold;
98
99 if (bIsTap)
100 {
101 SelectCarouselItem(mPressedCarouselIndex);
102 }
103
104 mPressedCarouselIndex = INDEX_NONE;
105 mCarouselPressPosition = FVector2D::ZeroVector;
106 return bIsTap;
107}

109int32 UCarouselPanelWidget::FindCarouselItemIndexAtPosition(const FVector2D& ScreenPosition) const
110{
111 /*
112 * 캐러셀에서는 카드들이 겹쳐 보일 수 있다.
113 * 화면 좌표 아래에 여러 카드가 겹치면 ZOrder가 가장 높은, 즉 가장 앞에 있는 카드를 선택한다.
114 */
115 int32 BestItemIndex = INDEX_NONE;
116 int32 BestZOrder = TNumericLimits<int32>::Lowest();
117
118 const int32 ItemCount = GetClampedCarouselItemDisplayCount();
119 for (int32 Index = 0; Index < ItemCount; ++Index)
120 {
121 const UWidget* Item = mCarouselItems[Index];
122 if (Item == nullptr || !Item->GetCachedGeometry().IsUnderLocation(ScreenPosition))
123 {
124 continue;
125 }
126
127 const int32 ZOrder = GetCarouselItemZOrder(Index);
128 if (ZOrder >= BestZOrder)
129 {
130 BestZOrder = ZOrder;
131 BestItemIndex = Index;
132 }
133 }
134
135 return BestItemIndex;
136}

138int32 UCarouselPanelWidget::GetCarouselItemZOrder(int32 ItemIndex) const
139{
140 /*
141 * CanvasPanelSlot이 있으면 실제 배치 ZOrder를 사용한다.
142 * WBP 구조가 바뀌어 CanvasPanelSlot이 아니더라도 선택 판정이 완전히 깨지지 않도록 0을 반환한다.
143 */
144 const UWidget* Item = mCarouselItems.IsValidIndex(ItemIndex) ? mCarouselItems[ItemIndex].Get() : nullptr;
145 if (Item == nullptr)
146 {
147 return TNumericLimits<int32>::Lowest();
148 }
149
150 if (const UCanvasPanelSlot* CanvasSlot = Cast<UCanvasPanelSlot>(Item->Slot))
151 {
152 return CanvasSlot->GetZOrder();
153 }
154
155 return 0;
156}
