26void FStaticSkillDataPropertyCustomization::CustomizeDetails(IDetailLayoutBuilder& DetailBuilder)
27{
28 /* 대상 에셋 포인터 구하기 */
29
30 TArray<TWeakObjectPtr<UObject>> ObjectsBeingCustomized;
31 DetailBuilder.GetObjectsBeingCustomized(ObjectsBeingCustomized);
32
33 if (ObjectsBeingCustomized.Num() == 0)
34 {
35 return;
36 }
37
38 mSelectedSkillDatas.Reset();
39 for (const TWeakObjectPtr<UObject>& ObjectBeingCustomized : ObjectsBeingCustomized)
40 {
41 TWeakObjectPtr<UStaticSkillData> TargetSkillData = Cast<UStaticSkillData>(ObjectBeingCustomized.Get());
42 if (TargetSkillData.IsValid() == false)
43 {
44 continue;
45 }
46
47 mSelectedSkillDatas.Add(TargetSkillData);
48 }
49
50 /* mDescription 프로퍼티 행 우측에 버튼 배치 커스터마이징 */
51
52 IDetailCategoryBuilder& Category = DetailBuilder.EditCategory("UI");
53 TSharedRef<IPropertyHandle> DescriptionHandle = DetailBuilder.GetProperty(GET_MEMBER_NAME_CHECKED(UStaticSkillData, mDescription));
54
55 if (DescriptionHandle->IsValidHandle() == true)
56 {
57 IDetailPropertyRow& DescriptionPropertyRow = Category.AddProperty(DescriptionHandle);
58 DescriptionPropertyRow.CustomWidget()
59 .NameContent()
60 [
61 DescriptionHandle->CreatePropertyNameWidget()
62 ]
63 .ValueContent()
64 .MinDesiredWidth(350.0f)
65 .MaxDesiredWidth(600.0f)
66 [
67 SNew(SHorizontalBox)
68 + SHorizontalBox::Slot()
69 .FillWidth(1.0f)
70 [
71 DescriptionHandle->CreatePropertyValueWidget()
72 ]
73 + SHorizontalBox::Slot()
74 .AutoWidth()
75 .Padding(4.0f, 0.0f, 0.0f, 0.0f)
76 .VAlign(VAlign_Center)
77 [
78 SNew(SButton)
79 .Text(LOCTEXT("GenerateDescription", "자동 생성"))
80 .ToolTipText(LOCTEXT("GenerateDescriptionToolTip", "스킬 설정값에 맞춰 Description 텍스트를 자동 생성합니다."))
81 .OnClicked(this, &FStaticSkillDataPropertyCustomization::OnGenerateDescription)
82 ]
83 ];
84 }
85}

87FReply FStaticSkillDataPropertyCustomization::OnGenerateDescription() const
88{
89 for (const TWeakObjectPtr<UStaticSkillData>& SkillData : mSelectedSkillDatas)
90 {
91 if (SkillData.IsValid() == false)
92 {
93 continue;
94 }
95
96 SkillData->Modify();
97 SkillData->mDescription = SkillData->MakeDescription();
98 SkillData->MarkPackageDirty();
99 }
100 return FReply::Handled();
101}
