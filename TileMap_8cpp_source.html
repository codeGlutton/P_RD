293void ATileMap::RebuildTileInstances()
294{

305void ATileMap::RefreshTileVisuals()
306{
307 // 컴포넌트가 없으면 처리 불가
308 if (mTileMeshComponent == nullptr)
309 {
310 return;
311 }
312
313 // 에디터에서 교체된 메시 반영
314 mTileMeshComponent->SetStaticMesh(mTileMesh);
315
316 // 테두리 파라미터를 원본 에셋 수정 없이 푸시하기 위한 타일 전용 MID 생성 후 적용
317 if (mTileMaterial != nullptr)
318 {
319 mTileMID = UMaterialInstanceDynamic::Create(mTileMaterial, this);
320 mTileMeshComponent->SetMaterial(0, mTileMID);
321 }
322 ApplyBorderParameters();
323
324 // 타일별 하이라이트 RGBA를 담을 custom data 슬롯 4개 (0=R,1=G,2=B,3=A)
325 mTileMeshComponent->SetNumCustomDataFloats(4);
326
327 // 기존 인스턴스 모두 제거 후 재생성
328 mTileMeshComponent->ClearInstances();
329
330 // 메시가 없거나 크기가 비정상이면 빈 그리드로 둠
331 if (mTileMesh == nullptr || mModel->GetWidth() <= 0 || mModel->GetHeight() <= 0 || mTileSize <= 0.0f)
332 {
333 return;
334 }
335
336 // 엔진 Plane은 100cm 기준 → 타일 크기에 맞춰 스케일 환산, 시각 비율로 칸 사이 틈 생성
337 const float PlaneBaseSize = 100.0f;
338 const float InstanceScaleXY = (mTileSize / PlaneBaseSize) * mTileVisualScale;
339
340 // Width x Height 만큼 타일 인스턴스를 로컬 공간에 배치
341 for (int32 y = 0; y < mModel->GetHeight(); ++y)
342 {
343 for (int32 x = 0; x < mModel->GetWidth(); ++x)
344 {
345 // 타일 (x,y) 중심의 로컬 위치
346 const FVector LocalLocation(x * mTileSize, y * mTileSize, 0.0f);
347 const FTransform InstanceTransform(FRotator::ZeroRotator, LocalLocation, FVector(InstanceScaleXY, InstanceScaleXY, 1.0f));
348 mTileMeshComponent->AddInstance(InstanceTransform, /*bWorldSpace=*/false);
349 }
350 }
351
352 // 새 인스턴스에 기본 구분색과 하이라이트를 다시 칠함
353 for (int32 Index = 0; Index < mHighlights.Num(); ++Index)
354 {
355 RefreshTileCustomData(Index);
356 }
357}

359void ATileMap::ApplyBorderParameters()
360{
361 // 테두리 정보가 없으면 바로 리턴
362 if (mTileMID == nullptr)
363 {
364 return;
365 }
366
367 // 타일은 mTileVisualScale만큼 축소/확대 되므로
368 // cm 단위로 설정한 테두리 굵기도 실제 그려지는 타일판 크기에 맞춰 재계산
369 const float TileVisualSize = mTileSize * mTileVisualScale;
370 const float BorderWidthUV = TileVisualSize > 0.0f ? FMath::Clamp(mTileBorderWidth / TileVisualSize, 0.0f, 0.5f) : 0.0f;
371
372 mTileMID->SetVectorParameterValue(TEXT("BorderColor"), mTileBorderStyle.mColor);
373 mTileMID->SetScalarParameterValue(TEXT("BorderWidth"), BorderWidthUV);
374}

381FTransform ATileMap::TileToWorldTransform(const FTileTransform& TileTransform) const
382{
383 // 월드 위치 획득
384 const FVector WorldLocation = TileToWorldLocation(TileTransform.mIndex);
385 // 타일맵 메시 컴포넌트의 YAW 회전을 타일에 적용 (그래야 같은 방향을 바라보니까)
386 const FQuat WorldRotation = mTileMeshComponent->GetComponentQuat() * FRotator(0.0f, DirectionToYaw(TileTransform.mDirection), 0.0f).Quaternion();
387 // 타일맵 메시 컴포넌트의 스케일을 타일에 적용
388 return FTransform(WorldRotation, WorldLocation, mTileMeshComponent->GetComponentScale());
389}

391FVector ATileMap::TileToWorldLocation(const FTileIndex& TileIndex) const
392{
393 // 타일 (x,y) 중심의 로컬 위치 (RebuildTileInstances의 인스턴스 배치식과 동일)
394 // 메시 피벗이 중심인 엔진 Plane 기준이라 (X*TileSize, Y*TileSize)가 곧 타일 중심

438void ATileMap::RefreshTileCustomData(int32 LinearIndex)
439{
440 // 컴포넌트/인덱스 유효성 (인스턴스 인덱스 = 타일 1D 인덱스)
441 if (mTileMeshComponent == nullptr || !mHighlights.IsValidIndex(LinearIndex))
442 return;
443
444 const ETileHighlightFlag Flags = mHighlights[LinearIndex];
445
446 // 플래그 해석 (Select가 겹치면 최우선 → Effect 억제)
447 const bool bHasAim = EnumHasAnyFlags(Flags, ETileHighlightFlag::Aim);
448 const bool bHasSelect = EnumHasAnyFlags(Flags, ETileHighlightFlag::Select);
449 const bool bHasEffect = EnumHasAnyFlags(Flags, ETileHighlightFlag::Effect) && !bHasSelect;
450
451 // 스타일 색을 프리멀티플라이드(알파 곱한 RGB + 커버리지 알파)로 변환 — 타일 위 Mix용
452 auto Premultiply = [](const FTileHighlightStyle& Style)
453 {
454 const FLinearColor& C = Style.mColor;
455 return FLinearColor(C.R * C.A, C.G * C.A, C.B * C.A, C.A);
456 };
457
458 // 최종색 (기본=머티리얼 바탕 위에 기본 구분색만 얹은 상태)
459 FLinearColor Accum = Premultiply(mTileBaseStyle);
460
461 if (bHasSelect)
462 {
463 // 최우선: 선택 색만 칠함 (Aim/Effect 무시)
464 Accum = Premultiply(mSelectStyle);
465 }
466 else if (bHasEffect)
467 {
468 // 펄스: [아래 레이어 표시] ↔ [Effect 자기 색] 크로스페이드 (둘 다 타일 위)
469 // 저점 = Aim 있으면 Aim 색, 없으면 기본 구분색 / 고점 = Effect 색
470 const FLinearColor Low = bHasAim ? Premultiply(mAimStyle) : Premultiply(mTileBaseStyle);
471 const FLinearColor High = Premultiply(mEffectStyle);
472
473 // 펄스 파동(0~1)으로 저점↔고점 보간
474 const float Time = GetWorld() ? GetWorld()->GetTimeSeconds() : 0.0f;
475 const float PulseWave = 0.5f - 0.5f * FMath::Cos(2.0f * PI * Time / mPulsePeriod);
476 Accum = FMath::Lerp(Low, High, PulseWave);
477 }
478 else if (bHasAim)
479 {
480 // Aim만: 자기 색
481 Accum = Premultiply(mAimStyle);
482 }
483
484 // custom data 슬롯에 기록 (마지막 슬롯에서 렌더 상태 갱신)
485 mTileMeshComponent->SetCustomDataValue(LinearIndex, 0, Accum.R);
486 mTileMeshComponent->SetCustomDataValue(LinearIndex, 1, Accum.G);
487 mTileMeshComponent->SetCustomDataValue(LinearIndex, 2, Accum.B);
488 mTileMeshComponent->SetCustomDataValue(LinearIndex, 3, Accum.A, /*bMarkRenderStateDirty=*/true);
489}

491void ATileMap::SetTileHighlight(const TArray<FTileIndex>& Tiles, ETileHighlightFlag Flag)
492{
493 if (Flag == ETileHighlightFlag::None)
494 return;
495
496 // 이 레이어를 새로 칠하면 무효가 되는 하위(더 구체적) 의존 레이어
497 // 드릴다운: Aim → Select → Effect (상위를 새로 하면 하위는 무효, 상위는 맥락으로 보존)
498 ETileHighlightFlag Dependents = ETileHighlightFlag::None;
499 if (EnumHasAnyFlags(Flag, ETileHighlightFlag::Aim))
500 Dependents |= ETileHighlightFlag::Select | ETileHighlightFlag::Effect;
501 if (EnumHasAnyFlags(Flag, ETileHighlightFlag::Select))
502 Dependents |= ETileHighlightFlag::Effect;
503
504 // 모든 타일에서 끌 비트 = Flag 자신 + 하위 의존 레이어
505 const ETileHighlightFlag ClearBits = Flag | Dependents;
506
507 // 새로 켤 타일 집합 (맵 밖 좌표는 무시)
508 TSet<int32> NewOn;
509 for (const FTileIndex& Tile : Tiles)
510 {
511 const int32 Index = mModel->TileIndexToLinearIndex(Tile);
512 if (Index != INDEX_NONE)
513 NewOn.Add(Index);
514 }
515
516 // 전체 타일 순회: ClearBits 끄고, 지정 타일이면 Flag 켜기
517 for (int32 Index = 0; Index < mHighlights.Num(); ++Index)
518 {
519 const ETileHighlightFlag Before = mHighlights[Index];
520
521 // 이 Flag와 하위 의존 레이어 비트를 끔 (상위/무관 비트는 보존)
522 ETileHighlightFlag After = Before & ~ClearBits;
523
524 // 지정된 타일이면 Flag 비트를 켬
525 if (NewOn.Contains(Index))
526 After |= Flag;
527
528 // 바뀐 타일만 custom data 갱신
529 if (After != Before)
530 {
531 mHighlights[Index] = After;
532 RefreshTileCustomData(Index);
533 }
534 }
535}

537void ATileMap::ClearTileHighlight(ETileHighlightFlag Flag)
538{
539 if (Flag == ETileHighlightFlag::None)
540 return;
541
542 // 전체 타일에서 지정 비트만 끔 (캐스케이드 없음, 무관 비트는 보존)
543 for (int32 Index = 0; Index < mHighlights.Num(); ++Index)
544 {
545 const ETileHighlightFlag Before = mHighlights[Index];
546 const ETileHighlightFlag After = Before & ~Flag;
547
548 // 바뀐 타일만 custom data 갱신
549 if (After != Before)
550 {
551 mHighlights[Index] = After;
552 RefreshTileCustomData(Index);
553 }
554 }

571void ATileMap::SetMovePath(const TArray<FTileIndex>& PathTiles)
572{
573 // 기존 표시 제거 후 다시 그림
574 ClearMovePath();
575
576 // 경로가 비었으면 표시할 것 없음 (해제와 동일)
577 if (PathTiles.Num() == 0)
578 return;
579
580 // 에디터에서 교체된 메시/머티리얼 반영
581 if (mPathArrowComponent != nullptr)
582 {
583 mPathArrowComponent->SetStaticMesh(mPathArrowMesh);
584 if (mPathArrowMaterial != nullptr)
585 mPathArrowComponent->SetMaterial(0, mPathArrowMaterial);
586 }
587 if (mPathEndComponent != nullptr)
588 {
589 mPathEndComponent->SetStaticMesh(mPathEndMesh);
590 if (mPathEndMaterial != nullptr)
591 mPathEndComponent->SetMaterial(0, mPathEndMaterial);
592 }
593
594 // 화살표/마커 균일 스케일 (타일 크기에 맞춤)
595 const float ArrowScale = (mTileSize / 100.0f) * mPathArrowScale;
596
597 // 마지막을 제외한 각 타일에 '다음 타일을 향하는' 화살표 배치 (인스턴스 순서 = 경로 순서)
598 const int32 LastIndex = PathTiles.Num() - 1;
599 if (mPathArrowComponent != nullptr)
600 {
601 for (int32 Index = 0; Index < LastIndex; ++Index)
602 {
603 const FTileIndex& Tile = PathTiles[Index];
604 const FTileIndex& Next = PathTiles[Index + 1];
605
606 // 진행 방향 스텝 → yaw 회전
607 const FTileIndex Step(Next.mX - Tile.mX, Next.mY - Tile.mY);
608 const FRotator Rotation(0.0f, StepToYaw(Step), 0.0f);
609
610 // 타일 중심 로컬 위치 + Z 오프셋 (RebuildTileInstances의 배치식과 동일, bWorldSpace=false)
611 const FVector Location(Tile.mX * mTileSize, Tile.mY * mTileSize, mPathHeightOffset);
612 const FTransform InstanceTransform(Rotation, Location, FVector(ArrowScale));
613 mPathArrowComponent->AddInstance(InstanceTransform, /*bWorldSpace=*/false);
614 }
615 }
616
617 // 마지막(도착) 타일엔 도착 마커 배치 (인디케이터 메시가 방향성이 있어 진입 방향으로 회전)
618 if (mPathEndComponent != nullptr)
619 {
620 const FTileIndex& EndTile = PathTiles[LastIndex];
621
622 // 진입 방향 = 마지막 스텝(직전 타일 → 도착 타일), 마지막 화살표와 같은 방향을 가리킴
623 // 단일 타일 경로(직전 타일 없음)면 진입 방향이 없어 회전 없음(+X)
624 FRotator Rotation = FRotator::ZeroRotator;
625 if (LastIndex >= 1)
626 {
627 const FTileIndex& PrevTile = PathTiles[LastIndex - 1];
628 const FTileIndex Step(EndTile.mX - PrevTile.mX, EndTile.mY - PrevTile.mY);
629 Rotation = FRotator(0.0f, StepToYaw(Step), 0.0f);
630 }
631
632 const FVector Location(EndTile.mX * mTileSize, EndTile.mY * mTileSize, mPathHeightOffset);
633 const FTransform InstanceTransform(Rotation, Location, FVector(ArrowScale));
634 mPathEndComponent->AddInstance(InstanceTransform, /*bWorldSpace=*/false);
635 }
636
637 // 표시 중인 경로 길이 기록 (틱 펄스 대상 판단 + 도착 마커 위상 인덱스)
638 mPathLength = PathTiles.Num();
639
640 // 최초 1회 펄스 색 기록 (이후 틱마다 자동 갱신)
641 RefreshPathPulse();
642}

661void ATileMap::RefreshPathPulse()
662{
663 const float Time = GetWorld() ? GetWorld()->GetTimeSeconds() : 0.0f;
664
665 // 칸당 위상 밀림(라디안) — 경로 전체에 고점 mPathFlowCycles개가 흐르도록 길이로 정규화 (길이 무관 일정한 흐름)
666 // Span은 시작~도착 타일 간격(mPathLength-1) 기준이라 화살표와 도착 마커가 같은 파동을 공유함
667 const int32 Span = FMath::Max(1, mPathLength - 1);
668 const float PhasePerTile = 2.0f * PI * mPathFlowCycles / Span;
669
670 // 한 인스턴스에 펄스 색을 기록하는 헬퍼
671 auto WriteInstance = [this](UInstancedStaticMeshComponent* Component, int32 InstanceIndex, const FLinearColor& Color, float Wave)
672 {
673 // 펄스를 상한/하한 값으로 변환해서 색에 곱하면 최종 색이 나옴
674 const float Brightness = FMath::Lerp(mPathPulseMinBrightness, mPathPulseMaxBrightness, Wave);
675 Component->SetCustomDataValue(InstanceIndex, 0, Color.R * Brightness);
676 Component->SetCustomDataValue(InstanceIndex, 1, Color.G * Brightness);
677 Component->SetCustomDataValue(InstanceIndex, 2, Color.B * Brightness);
678 Component->SetCustomDataValue(InstanceIndex, 3, 1.0f, /*bMarkRenderStateDirty=*/true);
679 };
680
681 // 화살표: 인스턴스 순서(=경로 순서)마다 위상차를 줘 흐르게
682 if (mPathArrowComponent != nullptr)
683 {
684 const int32 Count = mPathArrowComponent->GetInstanceCount();
685 for (int32 Index = 0; Index < Count; ++Index)
686 {
687 const float Phase = 2.0f * PI * Time / mPathPulsePeriod - Index * PhasePerTile;
688 const float Wave = 0.5f - 0.5f * FMath::Cos(Phase);
689 WriteInstance(mPathArrowComponent, Index, mPathArrowStyle.mColor, Wave);
690 }
691 }
692
693 // 도착 마커: 경로 끝 인덱스의 위상으로 이어지게 (화살표 흐름의 다음 칸)
694 if (mPathEndComponent != nullptr && mPathEndComponent->GetInstanceCount() > 0)
695 {
696 const int32 EndPhaseIndex = FMath::Max(0, mPathLength - 1);
697 const float Phase = 2.0f * PI * Time / mPathPulsePeriod - EndPhaseIndex * PhasePerTile;
698 const float Wave = 0.5f - 0.5f * FMath::Cos(Phase);
699 WriteInstance(mPathEndComponent, 0, mPathEndStyle.mColor, Wave);
700 }
701}
