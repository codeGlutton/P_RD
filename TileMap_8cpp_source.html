25ATileMap::ATileMap()
26{
27 // 그리드는 OnConstruction에서 재생성되므로 틱 불필요
28 PrimaryActorTick.bCanEverTick = false;
29
30 // 루트 컴포넌트 생성 및 지정
31 RootComponent = CreateDefaultSubobject<USceneComponent>(TEXT("Root"));
32
33 // 타일 그리드용 인스턴스드 메시 컴포넌트 생성 및 루트에 부착
34 mTileMeshComponent = CreateDefaultSubobject<UInstancedStaticMeshComponent>(TEXT("TileMesh"));
35 mTileMeshComponent->SetupAttachment(RootComponent);
36 // 시각화 전용이므로 충돌 비활성화
37 mTileMeshComponent->SetCollisionEnabled(ECollisionEnabled::NoCollision);
38
39 // 기본 타일 메시로 엔진 기본 Plane(100x100cm, +Z 향) 지정
40 static ConstructorHelpers::FObjectFinder<UStaticMesh> PlaneMeshFinder(TEXT("/Engine/BasicShapes/Plane.Plane"));
41 if (PlaneMeshFinder.Succeeded())
42 {
43 mTileMesh = PlaneMeshFinder.Object;
44 mTileMeshComponent->SetStaticMesh(mTileMesh);
45 }
46}

56void ATileMap::RebuildTileInstances()
57{
58 // 컴포넌트가 없으면 처리 불가
59 if (mTileMeshComponent == nullptr)
60 {
61 return;
62 }
63
64 // 에디터에서 교체된 메시/머티리얼 반영
65 mTileMeshComponent->SetStaticMesh(mTileMesh);
66 if (mTileMaterial != nullptr)
67 {
68 mTileMeshComponent->SetMaterial(0, mTileMaterial);
69 }
70
71 // 기존 인스턴스 모두 제거 후 재생성
72 mTileMeshComponent->ClearInstances();
73
74 // 메시가 없거나 크기가 비정상이면 빈 그리드로 둠
75 if (mTileMesh == nullptr || mWidth <= 0 || mHeight <= 0 || mTileSize <= 0.0f)
76 {
77 return;
78 }
79
80 // 엔진 Plane은 100cm 기준 → 타일 크기에 맞춰 스케일 환산, 시각 비율로 칸 사이 틈 생성
81 const float PlaneBaseSize = 100.0f;
82 const float InstanceScaleXY = (mTileSize / PlaneBaseSize) * mTileVisualScale;
83
84 // Width x Height 만큼 타일 인스턴스를 로컬 공간에 배치
85 for (int32 y = 0; y < mHeight; ++y)
86 {
87 for (int32 x = 0; x < mWidth; ++x)
88 {
89 // 타일 (x,y) 중심의 로컬 위치
90 const FVector LocalLocation(x * mTileSize, y * mTileSize, 0.0f);
91 const FTransform InstanceTransform(FRotator::ZeroRotator, LocalLocation, FVector(InstanceScaleXY, InstanceScaleXY, 1.0f));
92 mTileMeshComponent->AddInstance(InstanceTransform, /*bWorldSpace=*/false);
93 }
94 }
95}
