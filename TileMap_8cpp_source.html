287 * 포인터가 null 로 덮일 수 있다. 컴포넌트 인스턴싱이 끝난 이 시점에 현재

290 auto Rebind = [this](
291 TObjectPtr<UInstancedStaticMeshComponent>& Target,
292 const FName CurrentName,
293 const FName LegacyName = NAME_None)
294 {
295 UInstancedStaticMeshComponent* Resolved =
296 Cast<UInstancedStaticMeshComponent>(GetDefaultSubobjectByName(CurrentName));
297
298 if (Resolved == nullptr && !LegacyName.IsNone())
299 {
300 Resolved = Cast<UInstancedStaticMeshComponent>(GetDefaultSubobjectByName(LegacyName));
301 }
302
303 if (Resolved != nullptr)
304 {
305 Target = Resolved;
306 }
307 };
308
309 Rebind(mMovePathSet.mStraightComponent, TEXT("MovePathStraight"), TEXT("PathArrow"));
310 Rebind(mMovePathSet.mTurnLeftComponent, TEXT("MovePathTurnLeft"), TEXT("PathTurnLeft"));
311 Rebind(mMovePathSet.mTurnRightComponent, TEXT("MovePathTurnRight"), TEXT("PathTurnRight"));

316 Rebind(mPushPathSet.mTurnRightComponent, TEXT("PushPathTurnRight"));
317 Rebind(mPushPathSet.mEndComponent, TEXT("PushPathEnd"));
318
319 Rebind(mWaypointComponent, TEXT("Waypoint"));

441void ATileMap::RefreshTileVisuals()
442{
443 mPulsingHighlights.Reset();
444 // 컴포넌트가 없으면 처리 불가
445 if (mTileMeshComponent == nullptr)
446 {
447 return;
448 }
449
450 // 에디터에서 교체된 메시 반영
451 mTileMeshComponent->SetStaticMesh(mTileMesh);
452
453 // 테두리 파라미터를 원본 에셋 수정 없이 푸시하기 위한 타일 전용 MID 생성 후 적용
454 if (mTileMaterial != nullptr)
455 {
456 mTileMID = UMaterialInstanceDynamic::Create(mTileMaterial, this);
457 mTileMeshComponent->SetMaterial(0, mTileMID);
458 }

460

463

465 mTileMeshComponent->ClearInstances();
466
467 // 메시가 없거나 크기가 비정상이면 빈 그리드로 둠 (박스도 크기 0)
468 if (mTileMesh == nullptr || mModel->GetWidth() <= 0 || mModel->GetHeight() <= 0 || mTileSize <= 0.0f)
469 {
470 mPickingBoxComponent->SetBoxExtent(FVector::ZeroVector);
471 return;
472 }
473
474 // 엔진 Plane은 100cm 기준 → 타일 크기에 맞춰 스케일 환산, 시각 비율로 칸 사이 틈 생성
475 const float PlaneBaseSize = 100.0f;
476 const float InstanceScaleXY = (mTileSize / PlaneBaseSize) * mTileVisualScale;
477
478 // Width x Height 만큼 타일 인스턴스를 로컬 공간에 배치
479 for (int32 y = 0; y < mModel->GetHeight(); ++y)
480 {
481 for (int32 x = 0; x < mModel->GetWidth(); ++x)
482 {
483 // 타일 (x,y) 중심의 로컬 위치
484 const FVector LocalLocation(x * mTileSize, y * mTileSize, 0.0f);
485 const FTransform InstanceTransform(FRotator::ZeroRotator, LocalLocation, FVector(InstanceScaleXY, InstanceScaleXY, 1.0f));
486 mTileMeshComponent->AddInstance(InstanceTransform, /*bWorldSpace=*/false);
487 }
488 }
489
490 // 픽킹 박스를 타일 그리드 전체 크기로 갱신
491 // 중심: 그리드 중앙 (타일 중심 0..W-1 의 가운데), Z는 두께만큼 아래로 내려 상면이 타일 상면과 일치
492 // 반크기: 그리드 절반에서 아주 조금 축소
493 // 그리드 가장자리 좌표는 타일 크기로 나누면 정확히 x.5 가 되어 반올림 시 존재하지 않는 인덱스로 올라감
494 // 예) 타일 100, 3칸: 가장자리 250 → 250/100 = 2.5 → 반올림 3 (인덱스는 0~2 뿐)
495 // 박스를 아주 조금 줄이면 가장자리 히트 자체가 안 생겨 이 경우를 피함
496 const float BoxHalfThickness = 0.5f;
497 const float BoxEpsilon = 0.1f;
498 const float HalfWidth = mModel->GetWidth() * mTileSize * 0.5f - BoxEpsilon;
499 const float HalfHeight = mModel->GetHeight() * mTileSize * 0.5f - BoxEpsilon;
500 const FVector BoxCenter(
501 (mModel->GetWidth() - 1) * mTileSize * 0.5f,
502 (mModel->GetHeight() - 1) * mTileSize * 0.5f,
503 -BoxHalfThickness);
504 mPickingBoxComponent->SetRelativeLocation(BoxCenter);
505 mPickingBoxComponent->SetBoxExtent(FVector(HalfWidth, HalfHeight, BoxHalfThickness));
506
507 // 크기가 다르면 레벨 로드 직후 등 초기화가 안 된 상태이므로 초기화 처리
508 if (mHighlights.Num() != mModel->GetWidth() * mModel->GetHeight())
509 {
510 mHighlights.Init(ETileHighlightFlag::None, mModel->GetWidth() * mModel->GetHeight());
511 }
512
513 // 새 인스턴스에 기본 구분색과 하이라이트를 다시 칠함
514 for (int32 Index = 0; Index < mHighlights.Num(); ++Index)
515 {
516 RefreshTileCustomData(Index);
517 }

599void ATileMap::RefreshTileCustomData(int32 LinearIndex, bool bUpdatePulseMembership)
600{
601 // 컴포넌트/인덱스 유효성 (인스턴스 인덱스 = 타일 1D 인덱스)
602 if (mTileMeshComponent == nullptr || !mHighlights.IsValidIndex(LinearIndex))
603 return;
604
605 const ETileHighlightFlag Flags = mHighlights[LinearIndex];
606
607 // 플래그 해석 (Select가 겹치면 최우선 → Effect 억제)
608 const bool bHasAim = EnumHasAnyFlags(Flags, ETileHighlightFlag::Aim);
609 const bool bHasSelect = EnumHasAnyFlags(Flags, ETileHighlightFlag::Select);
610 const bool bHasEffect = EnumHasAnyFlags(Flags, ETileHighlightFlag::Effect) && !bHasSelect;
611 if (bUpdatePulseMembership)
612 {
613 if (bHasEffect) mPulsingHighlights.Add(LinearIndex);
614 else mPulsingHighlights.Remove(LinearIndex);
615 }
616
617 // 스타일 색을 프리멀티플라이드(알파 곱한 RGB + 커버리지 알파)로 변환 — 타일 위 Mix용
618 auto Premultiply = [](const FTileHighlightStyle& Style)
619 {
620 const FLinearColor& C = Style.mColor;
621 return FLinearColor(C.R * C.A, C.G * C.A, C.B * C.A, C.A);
622 };
623
624 // 최종색 (기본=머티리얼 바탕 위에 기본 구분색만 얹은 상태)
625 FLinearColor Accum = Premultiply(mTileBaseStyle);
626
627 if (bHasSelect)
628 {
629 // 최우선: 선택 색만 칠함 (Aim/Effect 무시)
630 Accum = Premultiply(mSelectStyle);
631 }
632 else if (bHasEffect)
633 {
634 // 펄스: [아래 레이어 표시] ↔ [Effect 자기 색] 크로스페이드 (둘 다 타일 위)
635 // 저점 = Aim 있으면 Aim 색, 없으면 기본 구분색 / 고점 = Effect 색
636 const FLinearColor Low = bHasAim ? Premultiply(mAimStyle) : Premultiply(mTileBaseStyle);
637 const FLinearColor High = Premultiply(mEffectStyle);
638
639 // 펄스 파동(0~1)으로 저점↔고점 보간
640 const float Time = GetWorld() ? GetWorld()->GetTimeSeconds() : 0.0f;
641 const float PulseWave = 0.5f - 0.5f * FMath::Cos(2.0f * PI * Time / FMath::Max(mPulsePeriod, KINDA_SMALL_NUMBER));
642 Accum = FMath::Lerp(Low, High, PulseWave);
643 }
644 else if (bHasAim)
645 {
646 // Aim만: 자기 색
647 Accum = Premultiply(mAimStyle);
648 }
649
650 // custom data 슬롯에 기록 (마지막 슬롯에서 렌더 상태 갱신)
651 mTileMeshComponent->SetCustomDataValue(LinearIndex, 0, Accum.R);
652 mTileMeshComponent->SetCustomDataValue(LinearIndex, 1, Accum.G);
653 mTileMeshComponent->SetCustomDataValue(LinearIndex, 2, Accum.B);
654 mTileMeshComponent->SetCustomDataValue(LinearIndex, 3, Accum.A, /*bMarkRenderStateDirty=*/bUpdatePulseMembership);
655}

657void ATileMap::SetTileHighlight(const TArray<FTileIndex>& Tiles, ETileHighlightFlag Flag)
658{
659 if (Flag == ETileHighlightFlag::None)
660 return;
661
662 // 이 레이어를 새로 칠하면 무효가 되는 하위(더 구체적) 의존 레이어
663 // 드릴다운: Aim → Select → Effect (상위를 새로 하면 하위는 무효, 상위는 맥락으로 보존)
664 ETileHighlightFlag Dependents = ETileHighlightFlag::None;
665 if (EnumHasAnyFlags(Flag, ETileHighlightFlag::Aim))
666 Dependents |= ETileHighlightFlag::Select | ETileHighlightFlag::Effect;
667 if (EnumHasAnyFlags(Flag, ETileHighlightFlag::Select))
668 Dependents |= ETileHighlightFlag::Effect;
669
670 // 모든 타일에서 끌 비트 = Flag 자신 + 하위 의존 레이어
671 const ETileHighlightFlag ClearBits = Flag | Dependents;
672
673 // 새로 켤 타일 집합 (맵 밖 좌표는 무시)
674 TSet<int32> NewOn;
675 for (const FTileIndex& Tile : Tiles)
676 {
677 const int32 Index = mModel->TileIndexToLinearIndex(Tile);
678 if (Index != INDEX_NONE)
679 NewOn.Add(Index);
680 }
681
682 // 전체 타일 순회: ClearBits 끄고, 지정 타일이면 Flag 켜기
683 for (int32 Index = 0; Index < mHighlights.Num(); ++Index)
684 {
685 const ETileHighlightFlag Before = mHighlights[Index];
686
687 // 이 Flag와 하위 의존 레이어 비트를 끔 (상위/무관 비트는 보존)
688 ETileHighlightFlag After = Before & ~ClearBits;
689
690 // 지정된 타일이면 Flag 비트를 켬
691 if (NewOn.Contains(Index))
692 After |= Flag;
693
694 // 바뀐 타일만 custom data 갱신
695 if (After != Before)
696 {
697 mHighlights[Index] = After;
698 RefreshTileCustomData(Index);
699 }
700 }
701}

723void ATileMap::SetThreatRange(const TArray<FTileIndex>& MoveTiles, const TArray<FTileIndex>& AttackTiles)
724{
725 // 기존 위협범위 삭제
726 ClearThreatRange();
727
728 if (mThreatMoveComponent == nullptr || mThreatAttackComponent == nullptr || mModel == nullptr || mTileSize <= 0.0f)
729 return;
730
731 // 머티리얼 선택: 전용 머티리얼이 있으면 그걸 사용하고, 없으면 타일 머티리얼 재사용
732 // 왜 생성자에서 안 하고 여기서 하냐면: 그래야 에디터에서 지정한 값을 사용할 수 있음
733 UMaterialInterface* Material = mThreatMaterial != nullptr ? mThreatMaterial.Get() : mTileMaterial.Get();
734 if (Material != nullptr)
735 {
736 mThreatMoveComponent->SetMaterial(0, Material);
737 mThreatAttackComponent->SetMaterial(0, Material);
738 }
739
740 // 위협 범위 관련 크기 설정
741 const float PlaneBaseSize = 100.0f;
742 const float VisualSize = mTileSize * mTileVisualScale;
743 const float BandWidth = VisualSize * mThreatBandWidthRatio;
744 const float BandHalf = (VisualSize - BandWidth) * 0.5f;
745 const float InnerSize = VisualSize - 2.0f * BandWidth;
746
747 // 인스턴스 색을 기록하는 헬퍼 (알파를 미리 곱해서 계산해 둠)
748 auto SetInstanceColor = [](UInstancedStaticMeshComponent* Component, int32 InstanceIndex, const FLinearColor& Color)
749 {
750 Component->SetCustomDataValue(InstanceIndex, 0, Color.R * Color.A);
751 Component->SetCustomDataValue(InstanceIndex, 1, Color.G * Color.A);
752 Component->SetCustomDataValue(InstanceIndex, 2, Color.B * Color.A);
753 Component->SetCustomDataValue(InstanceIndex, 3, Color.A, /*bMarkRenderStateDirty=*/true);
754 };
755
756 // 밴드 막대 4개의 배치표 (타일 중심 기준 {중심 오프셋, 크기})
757 // 세로 막대 2개는 전체 길이, 가로 막대 2개는 모서리에서 겹쳐 진해지지 않도록 밴드 폭만큼 줄임
758 const struct { FVector2D Center; FVector2D Size; } Bars[] = {
759 { FVector2D(+BandHalf, 0.0f), FVector2D(BandWidth, VisualSize) },
760 { FVector2D(-BandHalf, 0.0f), FVector2D(BandWidth, VisualSize) },
761 { FVector2D(0.0f, +BandHalf), FVector2D(InnerSize, BandWidth) },
762 { FVector2D(0.0f, -BandHalf), FVector2D(InnerSize, BandWidth) },
763 };
764
765 // 최대이동범위: 타일마다 밴드 막대 4개 배치
766 for (const FTileIndex& Tile : MoveTiles)
767 {
768 // 맵 밖 좌표 무시
769 if (mModel->IsValidIndex(Tile) == false)
770 continue;
771
772 const FVector TileCenter(Tile.mX * mTileSize, Tile.mY * mTileSize, mThreatHeightOffset);
773 for (const auto& Bar : Bars)
774 {
775 const FVector Location = TileCenter + FVector(Bar.Center.X, Bar.Center.Y, 0.0f);

900void ATileMap::AppendPath(FPathArrowSet& Set, const TArray<FMovePathTile>& PathTiles)
901{
902 // 경로가 비었으면 추가할 것 없음
903 if (PathTiles.Num() == 0)
904 return;
905
906 // 에디터에서 교체된 메시/머티리얼 반영
907 auto ApplyMeshAndMaterial = [&Set](UInstancedStaticMeshComponent* Component, UStaticMesh* Mesh)
908 {
909 if (Component == nullptr)
910 return;
911 Component->SetStaticMesh(Mesh);
912 if (Set.mMaterial != nullptr)
913 Component->SetMaterial(0, Set.mMaterial);
914 };
915 ApplyMeshAndMaterial(Set.mStraightComponent, Set.mStraightMesh);
916 ApplyMeshAndMaterial(Set.mTurnLeftComponent, Set.mTurnLeftMesh);
917 ApplyMeshAndMaterial(Set.mTurnRightComponent, Set.mTurnRightMesh);
918 ApplyMeshAndMaterial(Set.mEndComponent, Set.mEndMesh);
919
920 // 경유지 타일 집합
921 // 경유지가 일반화살표와 겹치지 않게 미리 집합을 만들어서 비교할 때 활용
922 // 왔다갔다 경로로 같은 타일이 여러 번 경유지면 마지막 등장 위치를 기록 (마커 덮어쓰기 판정용)
923 TSet<FTileIndex> WaypointTiles;
924 TMap<FTileIndex, int32> LastWaypointIndexes;
925 for (int32 Index = 0; Index < PathTiles.Num(); ++Index)
926 {
927 if (PathTiles[Index].mIsWaypoint)
928 {
929 WaypointTiles.Add(PathTiles[Index].mIndex);
930 LastWaypointIndexes.Add(PathTiles[Index].mIndex, Index);
931 }
932 }
933
934 // 화살표와 도착지마커를 타일사이즈에 맞게 스케일 조정
935 const float ArrowScale = (mTileSize / 100.0f) * mPathArrowScale;
936
937 // 타일의 위상차 계산:
938 // 한 주기 2*PI를 간격의 개수(타일 수 - 1)로 나누면 한 주기에 각 타일들이 가져야할 위상차가 됨
939 // 여기에 한 주기에 돌아야 할 사이클 수(기본은 1)를 곱하면 최종 위상차가 됨
940 const int32 Span = FMath::Max(1, PathTiles.Num() - 1);
941 const float PhasePerTile = 2.0f * PI * Set.mFlowCycles / Span;
942
943 // 경유지 순번 (등장 순서대로 1부터 부여)
944 int32 WaypointNumber = 0;
945
946 // 도착지 마커인 마지막 타일을 제외하고 화살표 배치
947 const int32 LastIndex = PathTiles.Num() - 1;
948 for (int32 Index = 0; Index < LastIndex; ++Index)
949 {
950 const FTileIndex& Tile = PathTiles[Index].mIndex;
951 const FTileIndex& Next = PathTiles[Index + 1].mIndex;
952
953 // 경유지 타일이면 화살표 대신 순번 마커 조립
954 if (PathTiles[Index].mIsWaypoint)
955 {
956 ++WaypointNumber;
957
958 // 같은 타일의 앞선 경유지는 마커 생략 (마지막 순번이 덮어씀)
959 // 도착 타일과 겹치면 마커 생략 (도착 마커가 덮어씀)
960 if (LastWaypointIndexes[Tile] == Index && Tile != PathTiles[LastIndex].mIndex)
961 AppendWaypointMarker(Tile, WaypointNumber);
962 continue;
963 }
964
965 // 경유지 타일을 다시 지나가는 경우도 화살표 생략 (마커 우선)
966 if (WaypointTiles.Contains(Tile))
967 continue;
968
969 // 진출 방향 스텝
970 const FTileIndex OutStep = Next - Tile;
971
972 // 기본은 직진 화살표
973 EPathArrowKind Kind = EPathArrowKind::Straight;
974 float Yaw = StepToYaw(OutStep);
975
976 if (Index > 0)
977 {
978 // 진입/진출 방향의 벡터 외적 부호로 회전 판별
979 // 0: 직진
980 // 양수: 우회전
981 // 음수: 좌회전
982 const FTileIndex& Prev = PathTiles[Index - 1].mIndex;
983 const FTileIndex InStep = Tile - Prev;
984 const int32 Cross = InStep.mX * OutStep.mY - InStep.mY * OutStep.mX;
985
986 // 회전 타일이면 회전 화살표로 교체
987 // @note 회전 화살표 메시가 없으면 직진 화살표 사용 (회전 화살표 메시만 없는 경우는 없겠지만...)
988 if (Cross > 0 && Set.mTurnRightMesh != nullptr)
989 {
990 Kind = EPathArrowKind::TurnRight;
991 Yaw = StepToYaw(InStep);
992 }
993 else if (Cross < 0 && Set.mTurnLeftMesh != nullptr)
994 {
995 Kind = EPathArrowKind::TurnLeft;
996 Yaw = StepToYaw(InStep);
997 }
998 }
999
1000 // 타일 중심 로컬 위치 + Z 오프셋에 배치 (바닥에서 아주 살짝 뜬 위치, Z축은 별도 스케일로 납작하게)
1001 const FVector Location(Tile.mX * mTileSize, Tile.mY * mTileSize, mPathHeightOffset);
1002 const FTransform InstanceTransform(FRotator(0.0f, Yaw, 0.0f), Location, FVector(ArrowScale, ArrowScale, ArrowScale * mPathArrowZScale));
1003 AddTileInstance(Set, Kind, Tile, InstanceTransform, Index * PhasePerTile);
1004 }
1005
1006 // 마지막 타일엔 도착지 마커 배치 (진입 방향과 같은 방향)
1007 {
1008 const FTileIndex& EndTile = PathTiles[LastIndex].mIndex;
1009
1010 // 진입 방향 = 마지막 스텝(직전 타일 → 도착 타일), 마지막 화살표와 같은 방향을 가리킴
1011 // 단일 타일 경로(직전 타일 없음)면 진입 방향이 없어 회전 없음(+X)
1012 FRotator Rotation = FRotator::ZeroRotator;
1013 if (LastIndex >= 1)
1014 {
1015 const FTileIndex& PrevTile = PathTiles[LastIndex - 1].mIndex;
1016 const FTileIndex Step = EndTile - PrevTile;
1017 Rotation = FRotator(0.0f, StepToYaw(Step), 0.0f);
1018 }
1019
1020 // 도착지 마커의 위상 오프셋 설정
1021 const FVector Location(EndTile.mX * mTileSize, EndTile.mY * mTileSize, mPathHeightOffset);
1022 const FTransform InstanceTransform(Rotation, Location, FVector(ArrowScale));
1023 AddTileInstance(Set, EPathArrowKind::End, EndTile, InstanceTransform, LastIndex * PhasePerTile);
1024 }
1025
1026 // 경로 개수 증가
1027 ++Set.mPathCount;
1028
1029 // 최초 1회 펄스 색 기록 (이후 틱마다 자동 갱신)
1030 RefreshPathPulse(Set);
1031}

1098void ATileMap::RemoveTileInstance(FPathArrowSet& Set, const FTileIndex& Tile)
1099{
1100 // 해당 타일에 표시가 있는 지 확인해서 없으면 그냥 리턴
1101 const TPair<EPathArrowKind, int32>* Found = Set.mTileInstances.Find(Tile);
1102 if (Found == nullptr)
1103 return;
1104
1105 // 삭제할 인스턴스의 종류와 인덱스 저장 (인덱스는 뒤에서 오프셋 삭제할 때 사용)
1106 const EPathArrowKind Kind = Found->Key;
1107 const int32 RemovedIndex = Found->Value;
1108
1109 UInstancedStaticMeshComponent* Component = GetComponent(Set, Kind);
1110 if (Component == nullptr)
1111 return;
1112
1113 // 인스턴스 삭제
1114 Component->RemoveInstance(RemovedIndex);
1115
1116 // 오프셋 삭제
1117 TArray<float>& Offsets = GetPhaseOffsets(Set, Kind);
1118 if (Offsets.IsValidIndex(RemovedIndex))
1119 Offsets.RemoveAt(RemovedIndex);
1120
1121 // 맵에서 해당 타일표시 삭제
1122 Set.mTileInstances.Remove(Tile);
1123
1124 // 인스턴스 삭제로 뒤 번호들이 한 칸씩 당겨졌으니, 맵에 남은 같은 종류의 더 큰 번호들도 1씩 당김
1125 for (TPair<FTileIndex, TPair<EPathArrowKind, int32>>& TileInstancePair : Set.mTileInstances)
1126 {
1127 if (TileInstancePair.Value.Key == Kind && TileInstancePair.Value.Value > RemovedIndex)
1128 --TileInstancePair.Value.Value;
1129 }
1130}

1137void ATileMap::RefreshPathPulse(FPathArrowSet& Set)
1138{
1139 // 게임 시작부터의 누적 시간 (일시정지 시 멈추는 게임 시간)
1140 const float Time = GetWorld() ? GetWorld()->GetTimeSeconds() : 0.0f;
1141
1142 // 누적시간을 라디안으로 변환.
1143 // 뒤에서 cos() 함수에 넣어서 0~1 사이의 값으로 만들고,
1144 // 이 값에서 인스턴스별 오프셋을 빼면 인스턴스의 현재 위상이 나옴.
1145 // 인스턴스는 '현재 위상'만큼의 밝기로 표현됨.
1146 const float TimePhase = 2.0f * PI * Time / Set.mPulsePeriod;
1147
1148 // 한 인스턴스에 펄스 색을 기록하는 헬퍼
1149 auto WriteInstance = [&Set](UInstancedStaticMeshComponent* Component, int32 InstanceIndex, const FLinearColor& Color, float Wave)
1150 {
1151 // 펄스를 상한/하한 값으로 변환해서 색에 곱하면 최종 색이 나옴
1152 const float Brightness = FMath::Lerp(Set.mPulseMinBrightness, Set.mPulseMaxBrightness, Wave);
1153 Component->SetCustomDataValue(InstanceIndex, 0, Color.R * Brightness);
1154 Component->SetCustomDataValue(InstanceIndex, 1, Color.G * Brightness);
1155 Component->SetCustomDataValue(InstanceIndex, 2, Color.B * Brightness);
1156 Component->SetCustomDataValue(InstanceIndex, 3, 1.0f, /*bMarkRenderStateDirty=*/true);
1157 };
1158
1159 // 종류별로 인스턴스를 돌며 저장된 오프셋으로 위상 평가 (화살표는 화살표 색, 마커는 마커 색)
1160 const EPathArrowKind Kinds[] = { EPathArrowKind::Straight, EPathArrowKind::TurnLeft, EPathArrowKind::TurnRight, EPathArrowKind::End };
1161 for (const EPathArrowKind Kind : Kinds)
1162 {
1163 UInstancedStaticMeshComponent* Component = GetComponent(Set, Kind);
1164 if (Component == nullptr)
1165 continue;
1166
1167 const TArray<float>& Offsets = GetPhaseOffsets(Set, Kind);
1168 const FLinearColor& Color = (Kind == EPathArrowKind::End) ? Set.mEndStyle.mColor : Set.mArrowStyle.mColor;
1169
1170 const int32 Count = Component->GetInstanceCount();
1171 for (int32 Index = 0; Index < Count; ++Index)
1172 {
1173 // 오프셋 기록이 없으면 0으로 폴백 (크래시 대신 동시 펄스로 보이는 안전망)
1174 const float Offset = Offsets.IsValidIndex(Index) ? Offsets[Index] : 0.0f;
1175
1176 // 이 인스턴스의 현재 위상(시간항 - 자기 오프셋)을 cos에 넣어서 [-1, +1] 사이 값으로 변환 후,
1177 // [0, 1] 사이 값으로 최종 변환 -> 이 값이 밝기의 최종 값이 됨
1178 const float Wave = 0.5f - 0.5f * FMath::Cos(TimePhase - Offset);
1179 WriteInstance(Component, Index, Color, Wave);
1180 }
1181 }

1242void ATileMap::AppendMarkerBar(const FVector& TileCenter, EMarkerBar Bar)
1243{

1245 return;
1246
1247 // 마커 치수: 테두리 바깥 크기와 막대 굵기는 프로퍼티, 숫자 크기는 테두리 안쪽에 맞춤

1249 const float Thickness = mWaypointBarThickness;
1250 const float DigitHeight = (FrameSize - 2.0f * Thickness) * 0.55f;
1251 const float DigitWidth = DigitHeight * 0.55f;
1252
1253 // 세그먼트 길이: 이음새마다 살짝 띄워 실제 디지털 숫자처럼 보이게 함
1254 const float SegmentLengthH = DigitWidth - Thickness * 1.2f;
1255 const float SegmentLengthV = DigitHeight * 0.5f - Thickness * 1.2f;
1256
1257 // 막대 종류별 중심 위치(u=가로, v=세로)와 크기 결정
1258 const float FrameHalf = (FrameSize - Thickness) * 0.5f;
1259 float CenterU = 0.0f, CenterV = 0.0f, SizeU = 0.0f, SizeV = 0.0f;
1260 switch (Bar)
1261 {
1262 case EMarkerBar::FrameTop: CenterV = +FrameHalf; SizeU = FrameSize; SizeV = Thickness; break;
1263 case EMarkerBar::FrameBottom: CenterV = -FrameHalf; SizeU = FrameSize; SizeV = Thickness; break;
1264 case EMarkerBar::FrameLeft: CenterU = -FrameHalf; SizeU = Thickness; SizeV = FrameSize - 2.0f * Thickness; break;
1265 case EMarkerBar::FrameRight: CenterU = +FrameHalf; SizeU = Thickness; SizeV = FrameSize - 2.0f * Thickness; break;
1266 case EMarkerBar::DigitTop: CenterV = +DigitHeight * 0.5f; SizeU = SegmentLengthH; SizeV = Thickness; break;
1267 case EMarkerBar::DigitMiddle: SizeU = SegmentLengthH; SizeV = Thickness; break;
1268 case EMarkerBar::DigitBottom: CenterV = -DigitHeight * 0.5f; SizeU = SegmentLengthH; SizeV = Thickness; break;
1269 case EMarkerBar::DigitTopLeft: CenterU = -DigitWidth * 0.5f; CenterV = +DigitHeight * 0.25f; SizeU = Thickness; SizeV = SegmentLengthV; break;
1270 case EMarkerBar::DigitTopRight: CenterU = +DigitWidth * 0.5f; CenterV = +DigitHeight * 0.25f; SizeU = Thickness; SizeV = SegmentLengthV; break;
1271 case EMarkerBar::DigitBottomLeft: CenterU = -DigitWidth * 0.5f; CenterV = -DigitHeight * 0.25f; SizeU = Thickness; SizeV = SegmentLengthV; break;
1272 case EMarkerBar::DigitBottomRight: CenterU = +DigitWidth * 0.5f; CenterV = -DigitHeight * 0.25f; SizeU = Thickness; SizeV = SegmentLengthV; break;

1274 }

1276 // 마커 좌표(u,v)를 타일 로컬로 변환: v는 +X(숫자 위쪽), u는 +Y(숫자 오른쪽), yaw로 카메라 방향 보정
