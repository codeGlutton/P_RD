33 // 이름으로 아티팩트 ID 검색 (정확 일치 우선, 부분 일치는 하나일 때만)
34 bool FindArtifactId(const FString& Query, FPrimaryAssetId& OutId)
35 {
36 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
37 if (AssetManager == nullptr)
38 {
39 return false;
40 }
41
42 TArray<FPrimaryAssetId> Ids;
43 AssetManager->GetPrimaryAssetIdList(ArtifactPrimaryAssetTypes::GetArtifactType(), Ids);
44
45 TArray<FPrimaryAssetId> Matched;
46 for (const FPrimaryAssetId& Id : Ids)
47 {
48 const FString Name = Id.PrimaryAssetName.ToString();
49 if (Name.Equals(Query, ESearchCase::IgnoreCase))
50 {
51 OutId = Id;
52 return true;
53 }
54 if (Name.Contains(Query, ESearchCase::IgnoreCase))
55 {
56 Matched.Add(Id);
57 }
58 }
59
60 if (Matched.Num() == 1)
61 {
62 OutId = Matched[0];
63 return true;
64 }
65 for (const FPrimaryAssetId& Id : Matched)
66 {
67 UE_LOG(LogRD, Warning, TEXT("후보: %s"), *Id.PrimaryAssetName.ToString());
68 }
69 UE_LOG(LogRD, Warning, TEXT("아티팩트 검색 실패: '%s' (일치 %d개)"), *Query, Matched.Num());
70 return false;
71 }

73 // 파티에 아티팩트 추가
74 void AddArtifact(const TArray<FString>& Args, UWorld* World)
75 {
76 if (Args.Num() < 1)
77 {
78 UE_LOG(LogRD, Warning, TEXT("사용법: RD.AddArtifact <에셋이름>"));
79 return;
80 }
81
82 UPartyArtifactComponentModel* PartyArtifacts = FindPartyArtifacts(World);
83 if (PartyArtifacts == nullptr)
84 {
85 UE_LOG(LogRD, Warning, TEXT("파티 없음: 방 안에서만 사용 가능"));
86 return;
87 }
88
89 FPrimaryAssetId ArtifactId;
90 if (FindArtifactId(Args[0], ArtifactId) == false)
91 {
92 return;
93 }
94 if (PartyArtifacts->AddArtifact(ArtifactId) == false)
95 {
96 UE_LOG(LogRD, Warning, TEXT("아티팩트 추가 실패: %s"), *ArtifactId.ToString());
97 return;
98 }
99 UE_LOG(LogRD, Log, TEXT("아티팩트 추가: %s (보유 %d개)"), *ArtifactId.PrimaryAssetName.ToString(), PartyArtifacts->GetPartyArtifacts().Num());
100 }
