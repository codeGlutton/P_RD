1// 에디터에서 게임 텍스트의 미리보기 언어를 바꾸는 콘솔 커맨드.
2//
3// 에디터 자동화(-ExecCmds="Automation RunTests ...")는 -culture= 를 무시하고
4// 에디터의 게임 로컬라이제이션 미리보기 언어를 따른다. 언어별 WBP 캡처를
5// 헤드리스로 뽑으려면 테스트 실행 전에 이 커맨드로 언어를 바꿔야 한다.
6//
7// UnrealEditor-Cmd P_RD.uproject -ExecCmds="RD.Editor.SetPreviewLanguage en,
8// Automation RunTests P_RD.UI.CombatLayout.Capture; Quit" ...
9

16static FAutoConsoleCommand GSetPreviewLanguageCommand(
17 TEXT("RD.Editor.SetPreviewLanguage"),
18 TEXT("게임 텍스트 미리보기 언어를 바꾼다. 사용: RD.Editor.SetPreviewLanguage <en|ko>"),
19 FConsoleCommandWithArgsDelegate::CreateLambda([](const TArray<FString>& Args)
20 {
21 if (Args.Num() < 1)
22 {
23 UE_LOG(LogTemp, Warning,
24 TEXT("RD.Editor.SetPreviewLanguage: 문화 코드가 필요하다 (예: en, ko)"));
25 return;
26 }
27 const FString& Culture = Args[0];
28 FInternationalization::Get().SetCurrentCulture(Culture);
29 FTextLocalizationManager::Get().EnableGameLocalizationPreview(Culture);
30 // 에디터는 부팅 언어의 locres 만 라이브 테이블에 올려 둔다. 다른
31 // 언어로 바꿀 때는 리소스를 강제로 다시 읽어야 한다(런타임 경로와 동일).
32 FTextLocalizationManager::Get().RefreshResources();
33 FTextLocalizationManager::Get().WaitForAsyncTasks();
34 // 진단: 알려진 키가 이 시점에 어떤 언어로 리졸브되는지 찍는다.
35 FText Probe;
36 FText::FindTextInLiveTable_Advanced(
37 TEXT("CombatLayoutHUD"), TEXT("StatusWeakness"), Probe);
38 UE_LOG(LogTemp, Display,
39 TEXT("RD.Editor.SetPreviewLanguage: '%s' 적용됨. Probe(StatusWeakness)='%s' Culture=%s"),
40 *Culture, *Probe.ToString(),
41 *FInternationalization::Get().GetCurrentCulture()->GetName());
42 }));
