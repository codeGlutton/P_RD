54 bool IsStageStartPoint(const FStage& Stage, int32 RowIndex, int32 ColumnIndex)
55 {
56 /*
57 * Stage 시작점은 "첫 번째 행의 시작 열"로 정의된다.
58 * 월드맵 UI에서는 이 노드를 일반 방과 다르게 START 표시로 그려 현재 런의 출발점을 알 수 있게 한다.
59 */
60 return RowIndex == 0 && ColumnIndex == Stage.mStartColumn;
61 }

71 bool IsNextRoomFromCurrentPath(const FStage& Stage, int32 CurrentRowIndex, int32 CurrentColumnIndex, int32 RowIndex, int32 ColumnIndex)
72 {
73 if (Stage.HasRoom(CurrentRowIndex, CurrentColumnIndex) == false)
74 {
75 return false;
76 }
77 if (RowIndex != CurrentRowIndex + 1)
78 {
79 return false;
80 }
81 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
82 {
83 return false;
84 }
85
86 const FRoom& CurrentRoom = Stage.mRoomRows[CurrentRowIndex].mRooms[CurrentColumnIndex].Get<FRoom>();
87 return CurrentRoom.mNextRoomColumns.Contains(ColumnIndex);
88 }

97 EMapRoomState ResolveRoomState(const FStage& Stage, const FRoom& Room, int32 CurrentRowIndex, int32 CurrentColumnIndex, int32 SelectedRowIndex, int32 SelectedColumnIndex)
98 {
99 if (Room.mWasSelected == true)
100 {
101 return EMapRoomState::Cleared;
102 }
103
104 if (IsNextRoomFromCurrentPath(Stage, CurrentRowIndex, CurrentColumnIndex, Room.mRow, Room.mColumn) == true)
105 {
106 if (Room.mRow == SelectedRowIndex && Room.mColumn == SelectedColumnIndex)
107 {
108 return EMapRoomState::Selected;
109 }
110 return EMapRoomState::Ready;
111 }
112
113 return EMapRoomState::Locked;
114 }

348bool ARoomGameModeBase::GetMapRoomViews(TArray<FMapRoomView>& OutRooms) const
349{
350 OutRooms.Reset();
351
352 const URunPersistData* RunPersistData = GetRunPersistData();
353 if (RunPersistData == nullptr || RunPersistData->IsActive() == false)
354 {
355 UE_LOG(LogRD, Warning, TEXT("런 데이터 미존재로 Map Room View 표기 불가"));
356
357 return false;
358 }
359
360 int32 CurrentRowIndex = 0;
361 int32 CurrentColumnIndex = 0;
362 RunPersistData->GetCurrentRoomIndex(OUT CurrentRowIndex, OUT CurrentColumnIndex);
363
364 const FStage& Stage = RunPersistData->GetStage();
365 for (int32 RowIndex = 0; RowIndex < Stage.mRoomRows.Num(); ++RowIndex)
366 {
367 const FRoomRow& RoomRow = Stage.mRoomRows[RowIndex];
368 for (int32 ColumnIndex = 0; ColumnIndex < RoomRow.mRooms.Num(); ++ColumnIndex)
369 {
370 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
371 {
372 continue;
373 }
374
375 const FRoom& Room = RoomRow.mRooms[ColumnIndex].Get<FRoom>();
376
377 const bool bIsStartPoint = IsStageStartPoint(Stage, RowIndex, ColumnIndex);
378 const EMapRoomState RoomState = ResolveRoomState(
379 Stage,
380 Room,
381 CurrentRowIndex,
382 CurrentColumnIndex,
383 mSelectedRoomRow,
384 mSelectedRoomColumn
385 );
386
387 FMapRoomView NewView;
388 NewView.mRow = RowIndex;
389 NewView.mColumn = ColumnIndex;
390 NewView.mType = Room.mType;
391 NewView.mState = RoomState;
392 NewView.mTitle = bIsStartPoint ? NSLOCTEXT("RoomGameModeBase", "StartPointTitle", "Start") : GetRoomTitle(Room.mType);
393 NewView.mDescription = bIsStartPoint ? GetStartPointDescription(Room) : GetRoomDescription(Room);
394 NewView.mNextRoomColumns = Room.mNextRoomColumns;
395 NewView.mPositionOffsetRate = Room.mPositionOffsetRate;
396 NewView.mSelectable = RoomState == EMapRoomState::Ready;
397 NewView.mSelected = RoomState == EMapRoomState::Selected;
398 NewView.mVisited = RoomState == EMapRoomState::Cleared;
399 NewView.mCanEnter = RoomState == EMapRoomState::Selected;
400 NewView.mIsStartPoint = bIsStartPoint;
401 OutRooms.Add(MoveTemp(NewView));
402 }
403 }
404
405 return OutRooms.IsEmpty() == false;
406}

416bool ARoomGameModeBase::GetRunControlView(FRunControlView& OutView) const
417{
418 /*
419 * TopMenuBar와 FrontendMapWidget이 현재 런 요약을 표시할 때 쓰는 UI DTO다.
420 * 위젯이 URunPersistData를 직접 읽지 않게 하고, GameMode가 "화면에 보여줄 값"만 골라 내려준다.
421 */
422 OutView = FRunControlView();
423 OutView.mHasActiveRun = HasActiveRun();
424 OutView.mCanSaveRun = false;
425 OutView.mCanAbandonRun = CanAbandonRun();
426
427 if (OutView.mHasActiveRun == false)
428 {
429 return false;
430 }
431
432 const URunPersistData* RunPersistData = GetRunPersistData();
433 RunPersistData->GetCurrentRoomIndex(OUT OutView.mRow, OUT OutView.mColumn);
434 OutView.mIsAtStageStart = IsStageStartPoint(RunPersistData->GetStage(), OutView.mRow, OutView.mColumn);
435 OutView.mPlayerLevel = RunPersistData->GetPlayerLevel();
436 OutView.mDifficulty = RunPersistData->GetDifficulty();
437 return true;
438}

447bool ARoomGameModeBase::GetRunControlState(OUT int32& RowIndex, OUT int32& ColumnIndex, OUT int32& PlayerLevel, OUT int32& Difficulty) const
448{
449 /*
450 * BP/WBP나 기존 호출부가 구조체 대신 개별 값으로 현재 런 상태를 받아야 할 때 쓰는 호환 API다.
451 * 내부 기준은 GetRunControlView() 하나로 유지해, 표시 데이터 계산이 두 군데로 갈라지지 않게 한다.
452 */
453 FRunControlView RunView;

458 PlayerLevel = 0;
459 Difficulty = 0;
460 return false;
461 }
462
463 RowIndex = RunView.mRow;
464 ColumnIndex = RunView.mColumn;
465 PlayerLevel = RunView.mPlayerLevel;
466 Difficulty = RunView.mDifficulty;
467 return true;
