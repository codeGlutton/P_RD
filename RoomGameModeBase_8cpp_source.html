57 bool IsStageStartPoint(const FStage& Stage, int32 RowIndex, int32 ColumnIndex)

59 /*

74 bool IsNextRoomFromCurrentPath(const FStage& Stage, int32 CurrentRowIndex, int32 CurrentColumnIndex, int32 RowIndex, int32 ColumnIndex)
75 {
76 if (Stage.HasRoom(CurrentRowIndex, CurrentColumnIndex) == false)
77 {
78 return false;
79 }
80 if (RowIndex != CurrentRowIndex + 1)
81 {
82 return false;
83 }
84 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
85 {
86 return false;
87 }
88
89 const FRoom& CurrentRoom = Stage.mRoomRows[CurrentRowIndex].mRooms[CurrentColumnIndex].Get<FRoom>();
90 return CurrentRoom.mNextRoomColumns.Contains(ColumnIndex);
91 }

100 EMapRoomState ResolveRoomState(const FStage& Stage, const FRoom& Room, int32 CurrentRowIndex, int32 CurrentColumnIndex, int32 SelectedRowIndex, int32 SelectedColumnIndex)
101 {
102 if (Room.mWasSelected == true)
103 {
104 return EMapRoomState::Cleared;
105 }
106
107 if (IsNextRoomFromCurrentPath(Stage, CurrentRowIndex, CurrentColumnIndex, Room.mRow, Room.mColumn) == true)
108 {
109 if (Room.mRow == SelectedRowIndex && Room.mColumn == SelectedColumnIndex)
110 {
111 return EMapRoomState::Selected;
112 }
113 return EMapRoomState::Ready;
114 }
115
116 return EMapRoomState::Locked;
117 }

346bool ARoomGameModeBase::GetMapRoomViews(TArray<FMapRoomView>& OutRooms) const
347{
348 OutRooms.Reset();
349
350 const URunPersistData* RunPersistData = GetRunPersistData();
351 if (RunPersistData == nullptr || RunPersistData->IsActive() == false)
352 {
353 UE_LOG(LogRD, Warning, TEXT("런 데이터 미존재로 Map Room View 표기 불가"));
354
355 return false;
356 }
357
358 int32 CurrentRowIndex = 0;
359 int32 CurrentColumnIndex = 0;
360 RunPersistData->GetCurrentRoomIndex(OUT CurrentRowIndex, OUT CurrentColumnIndex);
361
362 const FStage& Stage = RunPersistData->GetStage();
363 for (int32 RowIndex = 0; RowIndex < Stage.mRoomRows.Num(); ++RowIndex)
364 {
365 const FRoomRow& RoomRow = Stage.mRoomRows[RowIndex];
366 for (int32 ColumnIndex = 0; ColumnIndex < RoomRow.mRooms.Num(); ++ColumnIndex)
367 {
368 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
369 {
370 continue;
371 }
372
373 const FRoom& Room = RoomRow.mRooms[ColumnIndex].Get<FRoom>();
374
375 const bool bIsStartPoint = IsStageStartPoint(Stage, RowIndex, ColumnIndex);
376 const EMapRoomState RoomState = ResolveRoomState(
377 Stage,
378 Room,
379 CurrentRowIndex,
380 CurrentColumnIndex,
381 mSelectedRoomRow,
382 mSelectedRoomColumn
383 );
384
385 FMapRoomView NewView;
386 NewView.mRow = RowIndex;
387 NewView.mColumn = ColumnIndex;
388 NewView.mType = Room.mType;
389 NewView.mState = RoomState;
390 NewView.mTitle = bIsStartPoint ? NSLOCTEXT("RoomGameModeBase", "StartPointTitle", "Start") : GetRoomTitle(Room.mType);
391 NewView.mDescription = bIsStartPoint ? GetStartPointDescription(Room) : GetRoomDescription(Room);
392 NewView.mNextRoomColumns = Room.mNextRoomColumns;
393 NewView.mPositionOffsetRate = Room.mPositionOffsetRate;
394 NewView.mSelectable = RoomState == EMapRoomState::Ready;
395 NewView.mSelected = RoomState == EMapRoomState::Selected;
396 NewView.mVisited = RoomState == EMapRoomState::Cleared;
397 NewView.mCanEnter = RoomState == EMapRoomState::Selected;
398 NewView.mIsStartPoint = bIsStartPoint;
399 OutRooms.Add(MoveTemp(NewView));
400 }
401 }
402
403 return OutRooms.IsEmpty() == false;
404}

414bool ARoomGameModeBase::GetRunControlView(FRunControlView& OutView) const
415{
416 /*
417 * FrontendMapWidget이 현재 런 요약을 표시할 때 쓰는 UI DTO다.
418 * 위젯이 URunPersistData를 직접 읽지 않게 하고, GameMode가 "화면에 보여줄 값"만 골라 내려준다.
419 */
420 OutView = FRunControlView();
421 OutView.mHasActiveRun = HasActiveRun();
422 OutView.mCanSaveRun = false;
423 OutView.mCanAbandonRun = CanAbandonRun();
424
425 if (OutView.mHasActiveRun == false)
426 {
427 return false;
428 }
429
430 const URunPersistData* RunPersistData = GetRunPersistData();
431 RunPersistData->GetCurrentRoomIndex(OUT OutView.mRow, OUT OutView.mColumn);
432 OutView.mIsAtStageStart = IsStageStartPoint(RunPersistData->GetStage(), OutView.mRow, OutView.mColumn);
433 OutView.mPlayerLevel = RunPersistData->GetPlayerLevel();
434 OutView.mDifficulty = RunPersistData->GetDifficulty();
435 return true;

445bool ARoomGameModeBase::GetRunControlState(OUT int32& RowIndex, OUT int32& ColumnIndex, OUT int32& PlayerLevel, OUT int32& Difficulty) const
446{
447 /*
448 * BP/WBP나 기존 호출부가 구조체 대신 개별 값으로 현재 런 상태를 받아야 할 때 쓰는 호환 API다.
449 * 내부 기준은 GetRunControlView() 하나로 유지해, 표시 데이터 계산이 두 군데로 갈라지지 않게 한다.
450 */
451 FRunControlView RunView;

453 {
454 RowIndex = 0;
455 ColumnIndex = 0;
456 PlayerLevel = 0;
457 Difficulty = 0;
458 return false;
459 }
460
461 RowIndex = RunView.mRow;
462 ColumnIndex = RunView.mColumn;
463 PlayerLevel = RunView.mPlayerLevel;
464 Difficulty = RunView.mDifficulty;
465 return true;
