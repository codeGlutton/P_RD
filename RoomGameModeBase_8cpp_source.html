73 bool IsStageStartPoint(const FStage& Stage, int32 RowIndex, int32 ColumnIndex)
74 {
75 /*
76 * Stage 시작점은 "첫 번째 행의 시작 열"로 정의된다.
77 * 월드맵 UI에서는 이 노드를 일반 방과 다르게 START 표시로 그려 현재 런의 출발점을 알 수 있게 한다.
78 */
79 return RowIndex == 0 && ColumnIndex == Stage.mStartColumn;
80 }

90 bool IsNextRoomFromCurrentPath(const FStage& Stage, int32 CurrentRowIndex, int32 CurrentColumnIndex, int32 RowIndex, int32 ColumnIndex)
91 {
92 if (Stage.HasRoom(CurrentRowIndex, CurrentColumnIndex) == false)
93 {
94 return false;
95 }
96 if (RowIndex != CurrentRowIndex + 1)
97 {
98 return false;
99 }
100 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
101 {
102 return false;
103 }
104
105 const FRoom& CurrentRoom = Stage.mRoomRows[CurrentRowIndex].mRooms[CurrentColumnIndex].Get<FRoom>();
106 return CurrentRoom.mNextRoomColumns.Contains(ColumnIndex);
107 }

116 EMapRoomState ResolveRoomState(const FStage& Stage, const FRoom& Room, int32 CurrentRowIndex, int32 CurrentColumnIndex, int32 SelectedRowIndex, int32 SelectedColumnIndex)
117 {
118 if (Room.mWasSelected == true)
119 {
120 return EMapRoomState::Cleared;
121 }
122
123 if (IsNextRoomFromCurrentPath(Stage, CurrentRowIndex, CurrentColumnIndex, Room.mRow, Room.mColumn) == true)
124 {
125 if (Room.mRow == SelectedRowIndex && Room.mColumn == SelectedColumnIndex)
126 {
127 return EMapRoomState::Selected;
128 }
129 return EMapRoomState::Ready;
130 }
131
132 return EMapRoomState::Locked;
133 }

520bool ARoomGameModeBase::CompleteRunFromRoom()
521{
522 if (mSaveAndExitPending || mWasNextRoomPreloadRequested) return false;
523 if (!mFinalRunClosed)
524 {
525 if (!HasActiveRun()) return false;
526 const FStage& Stage = GetRunPersistData()->GetStage();
527 if (Stage.mStageLevel != EStageLevelType::Stage3 || !Stage.mClearData.mIsCleared
528 || Stage.GetCurrentRoom().mType != ERoomType::BossMonster) return false;
529 }
530 auto* Saves = GetGameInstance()->GetSubsystem<USaveGameSubsystem>();
531 if (!Saves) return false;
532 if (!mFinalRunClosed)
533 {
534 ClearRunPersistData();
535 mFinalRunClosed = true;
536 }
537 // Retry saving without counting the run twice or leaving for the title on failure.
538 if (!Saves->SaveUser() || !Saves->SaveRun())
539 {
540 UE_LOG(LogRDGameMode, Error, TEXT("Final run save failed; waiting for retry"));
541 return false;
542 }
543 UE_LOG(LogRDGameMode, Display, TEXT("RD_STAGE_VICTORY final run saved and closed"));
544 return PreloadAndTransitionFrontendRoomAsync();
545}

547void ARoomGameModeBase::SaveAndExitRunFromRoomAsync(
548 FOnRoomSaveAndExitComplete Completion)
549{
550 if (mSaveAndExitPending || mWasNextRoomPreloadRequested || !HasActiveRun())
551 {
552 Completion.ExecuteIfBound(false);
553 return;
554 }
555
556 USaveGameSubsystem* SaveGameSubsystem = GetGameInstance() != nullptr
557 ? GetGameInstance()->GetSubsystem<USaveGameSubsystem>() : nullptr;
558 if (SaveGameSubsystem == nullptr)
559 {
560 Completion.ExecuteIfBound(false);
561 return;
562 }
563
564 mSaveAndExitPending = true;
565 SaveGameSubsystem->RestoreCheckpointOnNextFrontend();
566 SaveGameSubsystem->SaveRunAsync(FAsyncSaveGameToSlotDelegate::CreateWeakLambda(
567 this,
568 [this, MovedCompletion = MoveTemp(Completion)](
569 const FString& SlotName, int32 UserIndex, bool bSaveSucceeded) mutable
570 {
571 const bool bTransitionStarted = bSaveSucceeded
572 && PreloadAndTransitionFrontendRoomAsync();
573 if (!bTransitionStarted)
574 {
575 mSaveAndExitPending = false;
576 GetGameInstance()->GetSubsystem<USaveGameSubsystem>()->CancelCheckpointFrontendRestore();
577 }
578 MovedCompletion.ExecuteIfBound(bTransitionStarted);
579 }));
580}

582bool ARoomGameModeBase::GetMapRoomViews(TArray<FMapRoomView>& OutRooms) const
583{
584 OutRooms.Reset();
585
586 const URunPersistData* RunPersistData = GetRunPersistData();
587 if (RunPersistData == nullptr || RunPersistData->IsActive() == false)
588 {
589 UE_LOG(LogRD, Warning, TEXT("런 데이터 미존재로 Map Room View 표기 불가"));
590
591 return false;
592 }
593
594 int32 CurrentRowIndex = 0;
595 int32 CurrentColumnIndex = 0;
596 RunPersistData->GetCurrentRoomIndex(OUT CurrentRowIndex, OUT CurrentColumnIndex);
597
598 const FStage& Stage = RunPersistData->GetStage();
599 for (int32 RowIndex = 0; RowIndex < Stage.mRoomRows.Num(); ++RowIndex)
600 {
601 const FRoomRow& RoomRow = Stage.mRoomRows[RowIndex];
602 for (int32 ColumnIndex = 0; ColumnIndex < RoomRow.mRooms.Num(); ++ColumnIndex)
603 {
604 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
605 {
606 continue;
607 }
608
609 const FRoom& Room = RoomRow.mRooms[ColumnIndex].Get<FRoom>();
610
611 const bool bIsStartPoint = IsStageStartPoint(Stage, RowIndex, ColumnIndex);
612 const EMapRoomState RoomState = ResolveRoomState(
613 Stage,
614 Room,
615 CurrentRowIndex,
616 CurrentColumnIndex,
617 mSelectedRoomRow,
618 mSelectedRoomColumn
619 );
620
621 FMapRoomView NewView;
622 NewView.mRow = RowIndex;
623 NewView.mColumn = ColumnIndex;
624 NewView.mType = Room.mType;
625 NewView.mState = RoomState;

893bool ARoomGameModeBase::GetPartyRosterView(FPartyRosterView& OutView) const
894{
895 /*
896 * 인벤토리(GetInventoryView)와 같은 규칙 -- 밀지 않고 물어보게 둔다.
897 * 직업명/초상/스킬 아이콘 해석까지 여기서 끝내고, 위젯은 받은 값만 그린다.
898 */
899 OutView = FPartyRosterView();
900
901 UPartyModel* PartyModel = GetPartyModel();
902 if (PartyModel == nullptr)
903 {
904 return false;
905 }
906
907 const TArray<TObjectPtr<UPlayerUnitModel>>& Members = PartyModel->GetPlayerUnitModels();
908 for (int32 MemberIndex = 0; MemberIndex < Members.Num(); ++MemberIndex)
909 {
910 const UPlayerUnitModel* Member = Members[MemberIndex].Get();
911 if (Member == nullptr)
912 {
913 // 빈 파티 슬롯은 명단에 싣지 않는다. 상세 왕복은 mMemberIndex 로 한다.
914 continue;
915 }
916
917 FPartyRosterMemberView& Row = OutView.mMembers.AddDefaulted_GetRef();
918 Row.mMemberIndex = MemberIndex;
919 Row.mJobName = GetRosterJobName(Member);
920 Row.mLevel = Member->GetPlayerLevel();
921 Row.mPortrait = GetRosterHeadPortrait(Member);
922
923 if (const UAttributeSetComponentModel* Attributes = Member->GetAttributeComponentModel())
924 {
925 Row.mHP = FMath::RoundToInt(Attributes->GetAttributeCurrentValue(
926 UPlayerUnitAttributeSet::GetHPAttribute()));
927 Row.mMaxHP = FMath::RoundToInt(Attributes->GetAttributeCurrentValue(
928 UPlayerUnitAttributeSet::GetMaxHPAttribute()));
929 Row.mAP = FMath::RoundToInt(Attributes->GetAttributeCurrentValue(
930 UUnitAttributeSet::GetActionPointAttribute()));
931 Row.mMaxAP = FMath::RoundToInt(Attributes->GetAttributeCurrentValue(
932 UUnitAttributeSet::GetRechargeActionPointAttribute()));
933 Row.mSpeed = FMath::RoundToInt(Attributes->GetAttributeCurrentValue(
934 UUnitAttributeSet::GetRechargeSpeedPointAttribute()));
935 }
936
937 if (const USkillComponentModel* SkillComponent = Member->GetSkillComponentModel())
938 {
939 const TArray<FSkillEntry>& Entries = SkillComponent->GetSkills();
940 for (int32 EntryIndex = 0; EntryIndex < Entries.Num(); ++EntryIndex)
941 {
942 const UStaticSkillData* Skill = Entries[EntryIndex].mData;
943 if (Skill == nullptr)
944 {
945 continue;
946 }
947 FPartyRosterSkillView& SkillRow = Row.mSkills.AddDefaulted_GetRef();
948 SkillRow.mName = Skill->mName;
949 SkillRow.mIcon = Skill->mIcon.LoadSynchronous();
950 /* 상세 조회는 컴포넌트 원본 슬롯 index 로 해야 한다. */
951 SkillRow.mSlotIndex = EntryIndex;
952 if (const UStaticUnitSkillData* UnitSkill = Cast<UStaticUnitSkillData>(Skill))
953 {
954 SkillRow.mActionPointCost = FMath::Max(UnitSkill->mRequiredActionPoint, 0);
955 }
956 if (SkillRow.mIcon == nullptr)
957 {
958 // 아이콘 미설정 DA 는 화면이 이름 글자로 폴백한다. 데이터 소유자가
959 // 알아챌 수 있게 생산자에서 한 번씩 흔적을 남긴다.
960 UE_LOG(LogTemp, Verbose,
961 TEXT("파티 명단 스킬 아이콘 미설정: %s (Member=%d, Slot=%d)"),
962 *Skill->GetName(), MemberIndex, EntryIndex);
963 }
964 }
965 }
966 }
967
968 /*
969 * 파티 공용 골드/아티팩트도 같은 판에 싣는다. 레일 인벤토리 페이지가
970 * 이 값만 보고 그린다 -- 조립 규칙(희귀도 색·이름 폴백)은 인벤토리와 같다.
971 */
972 if (const UAttributeSetComponentModel* PartyAttributes =
973 PartyModel->GetAttributeComponentModel())
974 {
975 OutView.mGold = FMath::RoundToInt(PartyAttributes->GetAttributeCurrentValue(
976 UPartyAttributeSet::GetMoneyAttribute()));
977 }
978 if (const UPartyArtifactComponentModel* PartyArtifacts =
979 PartyModel->GetPartyArtifactComponentModel())
980 {
981 const TArray<TObjectPtr<UStaticArtifactData>>& Artifacts =
982 PartyArtifacts->GetPartyArtifacts();
983 OutView.mArtifacts.Reserve(Artifacts.Num());
984 for (int32 ArtifactIndex = 0; ArtifactIndex < Artifacts.Num(); ++ArtifactIndex)
985 {
986 const UStaticArtifactData* Artifact = Artifacts[ArtifactIndex];
987 if (Artifact == nullptr)
988 {
989 continue;
990 }
991
992 FPartyRosterArtifactView& ArtifactRow = OutView.mArtifacts.AddDefaulted_GetRef();
993 ArtifactRow.mArtifactIndex = ArtifactIndex;
994 ArtifactRow.mName = Artifact->mName.IsEmpty() == false
995 ? Artifact->mName
996 : FText::Format(
997 NSLOCTEXT("RoomGameModeBase", "ArtifactFallbackName", "Artifact {0}"),
998 FText::AsNumber(ArtifactIndex + 1));
999 ArtifactRow.mIcon = ResolveRoomArtifactInventoryIcon(Artifact);
1000 ArtifactRow.mRarityColor = GetInventoryRarityColor(Artifact->mRarityType);
1001 ArtifactRow.mDetail = NSLOCTEXT(
1002 "RoomGameModeBase", "PartyArtifact", "Party-wide effect");
1003 }
1004 }

1007 return true;
