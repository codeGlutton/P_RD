107void UTopMenuBarWidget::ApplyInputPassThrough()
108{
109 SetVisibility(ESlateVisibility::SelfHitTestInvisible);
110
111 if (WidgetTree != nullptr)
112 {
113 UWidget* RootWidget = WidgetTree->RootWidget;
114 WidgetTree->ForEachWidget([RootWidget](UWidget* Widget)
115 {
116 if (Widget == nullptr || !ShouldKeepWidgetVisibility(Widget->GetVisibility()))
117 {
118 return;
119 }
120
121 if (Widget == RootWidget)
122 {
123 Widget->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
124 return;
125 }
126
127 if (Cast<UButton>(Widget) != nullptr)
128 {
129 Widget->SetVisibility(ESlateVisibility::Visible);
130 return;
131 }
132
133 Widget->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
134 });
135 }
136
137 ConfigureDesignerButton(MapButton);
138 ConfigureDesignerButton(SettingsButton);
139 ConfigureDesignerButton(DiceButton);
140 ConfigureDesignerButton(SkillButton);

379void UTopMenuBarWidget::CloseFloatingPanels(EWorldWidgetType ExceptWorldWidgetType) const
380{
381 /*
382 * 탑바에서 여는 네 패널은 모두 같은 "현재 조작 중인 팝업" 자리를 공유한다.
383 * 예를 들어 MAP 위에 DICE가 겹치면 아래 지도 버튼이 눌리는지, 위 주사위가 눌리는지 애매해진다.
384 * 그래서 새 패널을 열기 전에 나머지 패널을 명시적으로 닫는다.
385 */
386 constexpr EWorldWidgetType FloatingPanels[] = {
387 EWorldWidgetType::WorldMap,
388 EWorldWidgetType::InGameSettings,
389 EWorldWidgetType::DicePanel,
390 EWorldWidgetType::SkillPanel
391 };
392
393 for (const EWorldWidgetType FloatingPanel : FloatingPanels)
394 {
395 if (FloatingPanel != ExceptWorldWidgetType)
396 {
397 CloseWorldWidget(FloatingPanel);
398 }
399 }
400}

630void UTopMenuBarWidget::CloseSettingsPanelAndRestoreVictoryWorldMap()
631{
632 /*
633 * 설정 패널 닫기 애니메이션이 있다면, 그 애니메이션이 끝난 뒤 월드맵을 복원한다.

637 SettingsPanelWidget != nullptr && SettingsPanelWidget->IsOpened())
