18void FBoardEventSectionEditorBase::PaintEventName(FSequencerSectionPainter& Painter, int32 LayerId, const FString& InEventString, float PixelPos, bool IsEventValid) const
19{
20 static const int32 FontSize = 10;
21 static const float BoxOffsetPx = 10.f;
22 static const TCHAR* WarningString = TEXT("\xf071");
23
24 const FSlateFontInfo FontAwesomeFont = FAppStyle::Get().GetFontStyle("FontAwesome.10");
25 const FSlateFontInfo SmallLayoutFont = FCoreStyle::GetDefaultFontStyle("Bold", 10);
26 const FLinearColor DrawColor = FAppStyle::GetSlateColor("SelectionColor").GetColor(FWidgetStyle());
27
28 TSharedRef<FSlateFontMeasure> FontMeasureService = FSlateApplication::Get().GetRenderer()->GetFontMeasureService();
29
30 static FVector2D WarningSize = FontMeasureService->Measure(WarningString, FontAwesomeFont);
31 const FMargin WarningPadding = (IsEventValid || InEventString.Len() == 0) ? FMargin(0.f) : FMargin(0.f, 0.f, 4.f, 0.f);
32 const FMargin BoxPadding = FMargin(4.0f, 2.0f);
33
34 const FVector2D TextSize = FontMeasureService->Measure(InEventString, SmallLayoutFont);
35 const FVector2D IconSize = IsEventValid ? FVector2D::ZeroVector : WarningSize;
36 const FVector2D PaddedIconSize = IconSize + WarningPadding.GetDesiredSize();
37 const FVector2D BoxSize = FVector2D(TextSize.X + PaddedIconSize.X, FMath::Max(TextSize.Y, PaddedIconSize.Y)) + BoxPadding.GetDesiredSize();
38
39 bool IsDrawLeft = (Painter.SectionGeometry.Size.X - PixelPos) < (BoxSize.X + 22.f) - BoxOffsetPx;
40 float BoxPositionX = IsDrawLeft ? PixelPos - BoxSize.X - BoxOffsetPx : PixelPos + BoxOffsetPx;
41 if (BoxPositionX < 0.f)
42 {
43 BoxPositionX = 0.f;
44 }
45
46 FVector2D BoxOffset = FVector2D(BoxPositionX, Painter.SectionGeometry.Size.Y * .5f - BoxSize.Y * .5f);
47 FVector2D IconOffset = FVector2D(BoxPadding.Left, BoxSize.Y * .5f - IconSize.Y * .5f);
48 FVector2D TextOffset = FVector2D(IconOffset.X + PaddedIconSize.X, BoxSize.Y * .5f - TextSize.Y * .5f);
49
50 /* 배경 */
51
52 FSlateDrawElement::MakeBox(
53 Painter.DrawElements,
54 LayerId + 1,
55 Painter.SectionGeometry.ToPaintGeometry(BoxSize, FSlateLayoutTransform(BoxOffset)),
56 FAppStyle::GetBrush("WhiteBrush"),
57 ESlateDrawEffect::None,
58 FLinearColor::Black.CopyWithNewOpacity(0.5f)
59 );
60
61 if (IsEventValid == false)
62 {
63 /* 경고 메세지 */
64
65 FSlateDrawElement::MakeText(
66 Painter.DrawElements,
67 LayerId + 2,
68 Painter.SectionGeometry.ToPaintGeometry(IconSize, FSlateLayoutTransform(BoxOffset + IconOffset)),
69 WarningString,
70 FontAwesomeFont,
71 Painter.bParentEnabled ? ESlateDrawEffect::None : ESlateDrawEffect::DisabledEffect,
72 FAppStyle::GetWidgetStyle<FTextBlockStyle>("Log.Warning").ColorAndOpacity.GetSpecifiedColor()
73 );
74 }
75
76 /* 이벤트 메세지 */
77
78 FSlateDrawElement::MakeText(
79 Painter.DrawElements,
80 LayerId + 2,
81 Painter.SectionGeometry.ToPaintGeometry(TextSize, FSlateLayoutTransform(BoxOffset + TextOffset)),
82 InEventString,
83 SmallLayoutFont,
84 Painter.bParentEnabled ? ESlateDrawEffect::None : ESlateDrawEffect::DisabledEffect,
85 DrawColor
86 );
87}

105int32 FBoardEventTriggerSectionEditor::OnPaintSection(FSequencerSectionPainter& Painter) const
106{
107 int32 LayerId = Painter.PaintSectionBackground();
108
109 UBoardEventTriggerSection* EventTriggerSection = Cast<UBoardEventTriggerSection>(WeakSection.Get());
110 if (EventTriggerSection == nullptr || IsSectionSelected() == false)
111 {
112 return LayerId;
113 }
114
115 const FTimeToPixel& TimeToPixelConverter = Painter.GetTimeConverter();
116 TArrayView<const FFrameNumber> Times = EventTriggerSection->mEventChannel.GetData().GetTimes();
117
118 TRange<FFrameNumber> EventSectionRange = EventTriggerSection->GetRange();
119 for (int32 KeyIndex = 0; KeyIndex < Times.Num(); ++KeyIndex)
120 {
121 FFrameNumber EventTime = Times[KeyIndex];
122 if (EventSectionRange.Contains(EventTime))
123 {
124 const float PixelPos = TimeToPixelConverter.FrameToPixel(EventTime);
125 PaintEventName(Painter, LayerId, TEXT("TriggerEvent"), PixelPos, true);
126 }
127 }
128
129 return LayerId + 3;
130}

139
140 UBoardEventDurationSection* EventDurationSection = Cast<UBoardEventDurationSection>(WeakSection.Get());
141 if (EventDurationSection == nullptr)

143 return LayerId;
144 }
145
