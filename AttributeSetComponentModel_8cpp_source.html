12void UAttributeSetComponentModel::Initialize()
13{
14 Super::Initialize();
15
16 // 뒤에서부터 순회하며 null(소멸/직렬화 누락) 엔트리를 제거 — 역순이라 RemoveAt 인덱스가 안정적
17 for (int32 Idx = mSpawnedAttributes.Num() - 1; Idx >= 0; --Idx)
18 {
19 if (mSpawnedAttributes[Idx] == nullptr)

21 mSpawnedAttributes.RemoveAt(Idx);

27 GetObjectsWithOuter(GetOwnerModel(), ChildObjects, false, RF_NoFlags, EInternalObjectFlags::Garbage);
28
29 for (UObject* Obj : ChildObjects)
30 {
31 UTacticalAttributeSet* Set = Cast<UTacticalAttributeSet>(Obj);
32 if (Set)
33 {
34 mSpawnedAttributes.AddUnique(Set); // 중복 방지하며 등록
35 }
36 }
37
38 mActiveAttributeEffects.RegisterWithOwnerModel(this);

280FActiveTacticalEffectHandle UAttributeSetComponentModel::ApplyTacticalEffectSpecToSelf(const FTacticalEffectSpec& Spec)
281{
282 // 이펙트 적용 중 활성 이펙트 컨테이너 변경을 방지하는 스코프 락
283 FScopedActiveTacticalEffectLock ScopeLock(mActiveAttributeEffects);
284 // "현재 적용 중인 이펙트" 컨텍스트를 스코프 동안 설정(중첩 적용 추적용)
285 FScopeCurrentTacticalEffectBeingApplied ScopedTEApplication(GetWorld(), &Spec, this);
286
287 /* 진입 검사 */
288
289 // 이펙트 자체의 적용 조건(태그 요구사항 등)을 통과하지 못하면 무효 핸들 반환
290 if (Spec.mEffectClass->CanApply(mActiveAttributeEffects, Spec) == false)
291 {
292 return FActiveTacticalEffectHandle();
293 }
294
295 // 모디파이어가 가리키는 속성이 하나라도 무효면 적용 중단
296 for (const FTacticalModifierInfo& Mod : Spec.mEffectClass->mModifiers)
297 {
298 if (Mod.mAttribute.IsValid() == false)
299 {
300 return FActiveTacticalEffectHandle();
301 }
302 }
303
304 FActiveTacticalEffectHandle MyHandle(INDEX_NONE, GetWorld());
305 bool FoundExistingStackableGE = false;
306
307 FActiveTacticalEffect* AppliedEffect = nullptr;
308 FTacticalEffectSpec* OurCopyOfSpec = nullptr;
309 TUniquePtr<FTacticalEffectSpec> StackSpec;
310 {
311 if (Spec.mEffectClass->mDurationPolicy != ETacticalEffectDurationType::Instant)
312 {
313 AppliedEffect = mActiveAttributeEffects.ApplyTacticalEffectSpec(Spec, FoundExistingStackableGE);
314 if (AppliedEffect == nullptr)
315 {
316 return FActiveTacticalEffectHandle();
317 }
318
319 MyHandle = AppliedEffect->mHandle;
320 OurCopyOfSpec = &(AppliedEffect->mSpec);
321 }
322
323 if (OurCopyOfSpec == nullptr)
324 {
325 /* 인스턴스 Effect */
326
327 StackSpec = MakeUnique<FTacticalEffectSpec>(Spec);
328 OurCopyOfSpec = StackSpec.Get();
329
330 UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(this);
331 check(TacticalFrameworkModel != nullptr);
332 TacticalFrameworkModel->GlobalPreTacticalEffectSpecApply(*OurCopyOfSpec, this);
333 }
334 }
335
336 UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(this);
337 check(TacticalFrameworkModel != nullptr);
338 TacticalFrameworkModel->SetCurrentAppliedTE(OurCopyOfSpec);
339
340 if (Spec.mEffectClass->mDurationPolicy == ETacticalEffectDurationType::Instant)
341 {
342 ExecuteTacticalEffect(*OurCopyOfSpec);
343 }
344
345 Spec.mEffectClass->OnApplied(mActiveAttributeEffects, *OurCopyOfSpec);
346
347 UAttributeSetComponentModel* InstigatorASCModel = Spec.GetContext()->GetAttributeSetComponentModel();
348 OnTacticalEffectAppliedToSelf(InstigatorASCModel, *OurCopyOfSpec, MyHandle);
349
350 if (InstigatorASCModel != nullptr)
351 {
352 InstigatorASCModel->OnTacticalEffectAppliedToTarget(this, *OurCopyOfSpec, MyHandle);
353 }
354
355 return MyHandle;
356}

590void UAttributeSetComponentModel::UpdateTagMapSingle_Internal(const FGameplayTag& Tag, int32 CountDelta)
591{
592 if (CountDelta > 0)
593 {
594 // 카운트 증가: 0→양수로 새로 존재하게 되면 true 반환 → 추가 통지
595 if (mTacticalTagCountContainer.UpdateTagCount(Tag, CountDelta))
596 {
597 OnTagUpdated(Tag, true);
598 }
599 }
600 else if (CountDelta < 0)
601 {
602 // 카운트 감소: 부모 태그 제거를 지연 처리해 순회 중 변경을 피한다
603 TArray<FDeferredTagChangeDelegate> DeferredTagChangeDelegates;
604 if (mTacticalTagCountContainer.UpdateTagCount_DeferredParentRemoval(Tag, CountDelta, DeferredTagChangeDelegates))
605 {

619void UAttributeSetComponentModel::UpdateTagMap_Internal(const FGameplayTagContainer& Container, int32 CountDelta)
620{
621 if (CountDelta > 0)
622 {
623 for (TArray<FGameplayTag>::TConstIterator TagIt = Container.CreateConstIterator(); TagIt; ++TagIt)
624 {
625 const FGameplayTag& Tag = *TagIt;
626 if (mTacticalTagCountContainer.UpdateTagCount(Tag, CountDelta))
627 {
628 OnTagUpdated(Tag, true);
629 }
630 }
631 }
632 else if (CountDelta < 0)
633 {
634 TArray<FGameplayTag> RemovedTags;
635 RemovedTags.Reserve(Container.Num());
636 TArray<FDeferredTagChangeDelegate> DeferredTagChangeDelegates;
637
638 for (TArray<FGameplayTag>::TConstIterator TagIt = Container.CreateConstIterator(); TagIt; ++TagIt)
639 {
640 const FGameplayTag& Tag = *TagIt;
641 if (mTacticalTagCountContainer.UpdateTagCount_DeferredParentRemoval(Tag, CountDelta, DeferredTagChangeDelegates))
642 {
643 // 0으로 떨어진 태그만 수집
644 RemovedTags.Add(Tag);
645 }
646 }
647
648 if (RemovedTags.Num() > 0)
649 {
650 // 부모 태그 계층을 한 번에 재구성
651 mTacticalTagCountContainer.FillParentTags();
652 }
653
654 // 지연된 부모 태그 변화 델리게이트 실행
655 for (FDeferredTagChangeDelegate& Delegate : DeferredTagChangeDelegates)
656 {
657 Delegate.Execute();
658 }
659
660 // 사라진 태그들에 대한 제거 통지(계층 재구성 이후라 일관된 상태에서 통지됨)
661 for (FGameplayTag& Tag : RemovedTags)
662 {
663 OnTagUpdated(Tag, false);
664 }
665 }
