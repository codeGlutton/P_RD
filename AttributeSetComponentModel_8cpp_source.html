410FActiveTacticalEffectHandle UAttributeSetComponentModel::ApplyTacticalEffectSpecToSelf(const FTacticalEffectSpec& Spec)
411{
412 // 이펙트 적용 중 활성 이펙트 컨테이너 변경을 방지하는 스코프 락
413 FScopedActiveTacticalEffectLock ScopeLock(mActiveAttributeEffects);
414 // "현재 적용 중인 이펙트" 컨텍스트를 스코프 동안 설정(중첩 적용 추적용)
415 FScopeCurrentTacticalEffectBeingApplied ScopedGEApplication(GetWorld(), &Spec, this);
416
417 /* 진입 검사 */
418
419 // 이펙트 자체의 적용 조건(태그 요구사항 등)을 통과하지 못하면 무효 핸들 반환
420 if (Spec.mEffectClass->CanApply(mActiveAttributeEffects, Spec) == false)
421 {
422 return FActiveTacticalEffectHandle();
423 }
424
425 // 모디파이어가 가리키는 속성이 하나라도 무효면 적용 중단
426 for (const FTacticalModifierInfo& Mod : Spec.mEffectClass->mModifiers)
427 {
428 if (Mod.mAttribute.IsValid() == false)
429 {
430 return FActiveTacticalEffectHandle();
431 }
432 }
433
434 FActiveTacticalEffectHandle MyHandle(INDEX_NONE, GetWorld());
435 bool FoundExistingStackableGE = false;
436
437 FActiveTacticalEffect* AppliedEffect = nullptr;
438 FTacticalEffectSpec* OurCopyOfSpec = nullptr; // 실제 연산 대상이 되는 스펙(지속형은 컨테이너 보관본, 순간형은 로컬 복사본)
439 TUniquePtr<FTacticalEffectSpec> StackSpec;
440 {
441 // 지속형(Has Duration/Infinite) 이펙트: 활성 이펙트 컨테이너에 등록하고 보관본 스펙을 참조
442 if (Spec.mEffectClass->mDurationPolicy != ETacticalEffectDurationType::Instant)
443 {
444 AppliedEffect = mActiveAttributeEffects.ApplyTacticalEffectSpec(Spec, FoundExistingStackableGE);
445 if (AppliedEffect == nullptr)
446 {
447 return FActiveTacticalEffectHandle();
448 }
449
450 MyHandle = AppliedEffect->mHandle;
451 OurCopyOfSpec = &(AppliedEffect->mSpec);
452 }
453
454 if (OurCopyOfSpec == nullptr)
455 {
456 /* 인스턴스 Effect */
457
458 // 순간형(Instant) 이펙트: 컨테이너에 보관하지 않으므로 로컬 복사본을 만들어 연산에 사용
459 StackSpec = MakeUnique<FTacticalEffectSpec>(Spec);
460 OurCopyOfSpec = StackSpec.Get();
461
462 UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(this);
463 check(TacticalFrameworkModel != nullptr);
464 // 전역 사전 처리 훅(시전자 기반 동적 모디파이어 캡처 등)
465 TacticalFrameworkModel->GlobalPreTacticalEffectSpecApply(*OurCopyOfSpec, this);
466 }
467 }
468
469 UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(this);
470 check(TacticalFrameworkModel != nullptr);
471 TacticalFrameworkModel->SetCurrentAppliedTE(OurCopyOfSpec); // 현재 적용 스펙을 프레임워크에 등록
472
473 // 순간형이면 즉시 모디파이어를 실행해 BaseValue에 영구 반영
474 if (Spec.mEffectClass->mDurationPolicy == ETacticalEffectDurationType::Instant)
475 {
476 ExecuteTacticalEffect(*OurCopyOfSpec);
477 }
478
479 Spec.mEffectClass->OnApplied(mActiveAttributeEffects, *OurCopyOfSpec); // 이펙트별 적용 후처리 콜백
480
481 // 시전자 측 컴포넌트를 컨텍스트에서 얻어 "자신에게 적용됨" 통지
482 UAttributeSetComponentModel* InstigatorASCModel = Spec.GetContext()->GetAttributeSetComponentModel();
483 OnTacticalEffectAppliedToSelf(InstigatorASCModel, *OurCopyOfSpec, MyHandle);
484
485 // 시전자가 존재하면 시전자 측에 "대상에게 적용됨" 통지(시전자 관점 콜백)
486 if (InstigatorASCModel != nullptr)
487 {
488 InstigatorASCModel->OnTacticalEffectAppliedToTarget(this, *OurCopyOfSpec, MyHandle);
489 }
490
491 return MyHandle;
492}

630void UAttributeSetComponentModel::UpdateTagMapSingle_Internal(const FGameplayTag& Tag, int32 CountDelta)
631{
632 if (CountDelta > 0)
633 {
634 // 카운트 증가: 0→양수로 새로 존재하게 되면 true 반환 → 추가 통지
635 if (mTacticalTagCountContainer.UpdateTagCount(Tag, CountDelta))
636 {
637 OnTagUpdated(Tag, true);
638 }
639 }
640 else if (CountDelta < 0)
641 {
642 // 카운트 감소: 부모 태그 제거를 지연 처리해 순회 중 변경을 피한다
643 TArray<FDeferredTagChangeDelegate> DeferredTagChangeDelegates;
644 if (mTacticalTagCountContainer.UpdateTagCount_DeferredParentRemoval(Tag, CountDelta, DeferredTagChangeDelegates))
645 {
646 mTacticalTagCountContainer.FillParentTags(); // 부모 태그 계층 재구성
647 OnTagUpdated(Tag, false); // 제거 통지
648
649 // 지연해 둔 부모 태그 변화 델리게이트들을 안전한 시점에 실행
650 for (FDeferredTagChangeDelegate& Delegate : DeferredTagChangeDelegates)
651 {
652 Delegate.Execute();
653 }
654 }

663void UAttributeSetComponentModel::UpdateTagMap_Internal(const FGameplayTagContainer& Container, int32 CountDelta)
664{
665 if (CountDelta > 0)
666 {
667 // 추가 경로: 각 태그를 증가시키고 새로 존재하게 된 태그마다 즉시 통지
668 for (TArray<FGameplayTag>::TConstIterator TagIt = Container.CreateConstIterator(); TagIt; ++TagIt)
669 {
670 const FGameplayTag& Tag = *TagIt;
671 if (mTacticalTagCountContainer.UpdateTagCount(Tag, CountDelta))
672 {
673 OnTagUpdated(Tag, true);
674 }
675 }
676 }
677 else if (CountDelta < 0)
678 {
679 // 제거 경로: 부모 태그 제거를 지연하고, 실제 사라진 태그를 모아 일괄 후처리
680 TArray<FGameplayTag> RemovedTags;
681 RemovedTags.Reserve(Container.Num());
682 TArray<FDeferredTagChangeDelegate> DeferredTagChangeDelegates;
683
684 for (TArray<FGameplayTag>::TConstIterator TagIt = Container.CreateConstIterator(); TagIt; ++TagIt)
685 {
686 const FGameplayTag& Tag = *TagIt;
687 if (mTacticalTagCountContainer.UpdateTagCount_DeferredParentRemoval(Tag, CountDelta, DeferredTagChangeDelegates))
688 {
689 RemovedTags.Add(Tag); // 0으로 떨어진 태그만 수집
690 }
691 }
692
693 if (RemovedTags.Num() > 0)
694 {
695 mTacticalTagCountContainer.FillParentTags(); // 부모 태그 계층을 한 번에 재구성
696 }
697
698 // 지연된 부모 태그 변화 델리게이트 실행
699 for (FDeferredTagChangeDelegate& Delegate : DeferredTagChangeDelegates)
700 {
701 Delegate.Execute();
702 }
703
704 // 사라진 태그들에 대한 제거 통지(계층 재구성 이후라 일관된 상태에서 통지됨)
705 for (FGameplayTag& Tag : RemovedTags)
706 {
707 OnTagUpdated(Tag, false);
708 }
709 }
710}
