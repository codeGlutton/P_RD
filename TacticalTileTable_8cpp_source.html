13void FTacticalTileTable::Build(
14 const UTileMapModel* TileMap,
15 const UBoardActorModel* Self,
16 const FTileIndex& Origin,
17 const TArray<FTileIndex>& TargetTiles,
18 const TArray<const UStaticUnitSkillData*>& Skills,
19 int32 MoveBudget,
20 int32 ActionPoint)
21{
22 // 재사용 대비 초기화
23 mTacticalTiles.Reset();
24 mOriginTargetDistances.Reset();
25 mSkillSlotCount = Skills.Num();
26 mTargetCount = TargetTiles.Num();
27
28 // 타일맵 또는 타겟이 없으면 계산이 무의미하므로 바로 리턴
29 if (TileMap == nullptr || mTargetCount == 0)
30 {
31 return;
32 }
33
34 // 이동가능한 타일 수집 (어쨌거나 이동가능한 타일에서 공격을 시작하니까)
35 TArray<FTileIndex> ReachableTiles = TileMap->GetReachableTiles(Origin, MoveBudget, Self);
36 ReachableTiles.Add(Origin);
37
38 // 원점에서 타일까지 이동거리장 (타일까지 이동하는데 소모하는 행동력)
39 const TArray<int32> MoveCostField = TileMap->GetDistanceField(Origin, Self);
40
41 // 타일에서 타겟까지 이동거리장 (이동성향에 따라 분기할 때 참조)
42 TArray<TArray<int32>> TargetDistanceFields;
43 TargetDistanceFields.Reserve(mTargetCount);
44 for (const FTileIndex& TargetTile : TargetTiles)
45 {
46 TargetDistanceFields.Add(TileMap->GetDistanceField(TargetTile, Self));
47 }
48
49 // 원점에서 타겟까지의 경로 거리 (타겟 선택할 때 '최근접' 비교값)
50 for (const FTileIndex& TargetTile : TargetTiles)
51 {
52 // 타겟이 서 있는 칸까지 들어선다고 칠 때의 걸음 수 = 경로 길이 - 1 (경로가 없으면 도달 불가)
53 const TArray<FTileIndex> Path = TileMap->FindPath(Origin, TargetTile, Self, /*bAllowOccupiedGoal*/true);
54 mOriginTargetDistances.Add(Path.IsEmpty() ? MAX_int32 : Path.Num() - 1);
55 }
56
57 // 타일마다 전술 정보 채움
58 mTacticalTiles.Reserve(ReachableTiles.Num());
59 for (const FTileIndex& Tile : ReachableTiles)
60 {
61 const int32 Linear = TileMap->TileIndexToLinearIndex(Tile);
62
63 // 이동비용 (원점 자신은 거리장에서 0)
64 const int32 MoveCost = MoveCostField[Linear];
65 checkf(MoveCost >= 0, TEXT("타일(%d,%d)이 이동가능 목록에는 있는데 거리장에서는 도달불가(%d) — GetReachableTiles()와 GetDistanceField()의 통과 판정이 어긋남"), Tile.mX, Tile.mY, MoveCost);
66
67 FTacticalTileInfo& Info = mTacticalTiles.AddDefaulted_GetRef();
68 Info.mIndex = Tile;
69 Info.mMoveCost = MoveCost;
70
71 // 타일에서 타겟까지의 경로 거리 (도달 불가는 MAX_int32로 통일)
72 Info.mTargetDistances.Reserve(mTargetCount);
73 for (int32 TargetIndex = 0; TargetIndex < mTargetCount; ++TargetIndex)
74 {
75 const int32 RawDistance = TargetDistanceFields[TargetIndex][Linear];
76 Info.mTargetDistances.Add(RawDistance >= 0 ? RawDistance : MAX_int32);
77 }
78
79 // 조준/시전 플래그 채우기
80 Info.mAimableFlags.Init(false, mSkillSlotCount * mTargetCount);
81 Info.mCastableFlags.Init(false, mSkillSlotCount * mTargetCount);
82 for (int32 SkillSlot = 0; SkillSlot < mSkillSlotCount; ++SkillSlot)
83 {
84 const UStaticUnitSkillData* Skill = Skills[SkillSlot];
85 // 빈 슬롯은 전부 false 유지
86 if (Skill == nullptr)
87 {
88 continue;
89 }
90
91 // 예산 판정: 이 타일까지 이동한 뒤 남는 행동력으로 시전비용을 감당할 수 있는가
92 const bool bBudgetOk = (Info.mMoveCost + Skill->mRequiredActionPoint <= ActionPoint);
93
94 // Single 판정 준비: 조준 타일이 시전자 자기 칸이라 타겟 칸 조준 여부로는 판정 불가.
95 // 이 타일을 조준 타일로 삼아 타겟 패턴 -> 영향 범위를 계산해 두고, 타겟 칸이 그 안에 드는지로 판정.
96 // 자기 자신은 이동으로 자리를 비울 예정이므로 확산 차단에서 제외
97 const bool bSingleAim = (Skill->mAimPattern == EAimPattern::Single);
98 TArray<FTileIndex> SingleEffectTiles;
99 if (bSingleAim)
100 {
101 const TArray<FTileIndex> CenterTiles = TileMap->GetTargetTiles(Tile, Tile, Skill->mTargetPattern);
102 for (const FTileIndex& Center : CenterTiles)
103 {
104 for (const FTileIndex& Effect : TileMap->GetEffectTiles(
105 Center,
106 Skill->mEffectPattern,
107 Skill->mEffectArea,
108 static_cast<ETileLayerFlag>(Skill->mEffectBlockerMask),
109 /*IgnoreBlocker*/Self))
110 {
111 SingleEffectTiles.AddUnique(Effect);
112 }
113 }
114 }
115
116 for (int32 TargetIndex = 0; TargetIndex < mTargetCount; ++TargetIndex)
117 {
118 bool bAimable = false;
119 if (bSingleAim)
120 {
121 // Single: 영향 범위에 타겟 칸이 들면 조준(=시전) 가능
122 bAimable = SingleEffectTiles.Contains(TargetTiles[TargetIndex]);
123 }
124 else
125 {
126 // 그 외 패턴: 자기 자신은 이동으로 자리를 비울 예정이므로 시야 차폐에서 제외
127 bAimable = TileMap->CanAim(
128 Tile,
129 TargetTiles[TargetIndex],
130 Skill->mAimRange,
131 Skill->mAimPattern,
132 Skill->mCanAimBoardActor,
133 static_cast<ETileLayerFlag>(Skill->mAimBlockerMask),
134 /*Incoming*/nullptr,
135 /*IgnoreBlocker*/Self);
136 }
137
138 const int32 Flag = FlagIndex(SkillSlot, TargetIndex);
139 Info.mAimableFlags[Flag] = bAimable;
140 Info.mCastableFlags[Flag] = bAimable && bBudgetOk;
141 }
142 }
143 }
144}

157{
158 // 어느 타일에서든, 어떤 스킬로든 시전 가능하면 참
159 for (const FTacticalTileInfo& Tile : mTacticalTiles)
160 {
161 for (int32 SkillSlot = 0; SkillSlot < mSkillSlotCount; ++SkillSlot)

219FTileIndex FTacticalTileTable::PickTile(
220 TFunctionRef<bool(const FTacticalTileInfo&)> Filter,
221 TFunctionRef<int64(const FTacticalTileInfo&)> PrimaryAxis,
222 TFunctionRef<int64(const FTacticalTileInfo&)> SecondaryAxis,
223 const FTileIndex& Fallback) const
224{
225 FTileIndex Best = Fallback;
226 bool HasBest = false;
227 int64 BestPrimary = 0;
228 int64 BestSecondary = 0;
229
230 for (const FTacticalTileInfo& Tile : mTacticalTiles)
231 {
232 // 필터 탈락 타일은 후보가 아님
233 if (Filter(Tile) == false)
234 {
235 continue;
236 }
237
238 const int64 Primary = PrimaryAxis(Tile);
239 const int64 Secondary = SecondaryAxis(Tile);
240
241 bool Better = false;
242 // 1) 첫 후보면 무조건 최선
243 if (HasBest == false)
244 {
245 Better = true;
246 }
247 // 2) 1순위 축이 다르면 작은 쪽이 최선
248 else if (Primary != BestPrimary)
249 {
250 Better = (Primary < BestPrimary);
251 }
252 // 3) 1순위가 같으면 2순위 축이 작은 쪽이 최선
253 // 완전 동률이면 먼저 나온 타일 유지
254 else
255 {
256 Better = (Secondary < BestSecondary);
257 }
258
259 if (Better == true)
260 {
261 Best = Tile.mIndex;
262 BestPrimary = Primary;
263 BestSecondary = Secondary;
264 HasBest = true;
265 }
266 }
267 return Best;
268}
