55 void Export(const TArray<FString>& Args)
56 {
57 const FString Label = Args.IsEmpty() ? TEXT("current") : Args[0];
58 const FString OutputDirectory = FPaths::ProjectSavedDir() / TEXT("UIAudit");
59 IFileManager::Get().MakeDirectory(*OutputDirectory, true);
60 const FString ProgressPath = OutputDirectory /
61 FString::Printf(TEXT("WidgetFonts_%s.progress.txt"), *Label);
62 FFileHelper::SaveStringToFile(TEXT("command-start\n"), *ProgressPath,
63 FFileHelper::EEncodingOptions::ForceUTF8WithoutBOM);
64 const FString OutputPath = OutputDirectory /
65 FString::Printf(TEXT("WidgetFonts_%s.tsv"), *Label);
66
67 FString Output = TEXT("AssetPath\tWidgetName\tSize\tTypeface\tFontObject\tLetterSpacing\tOutlineSize\tText\n");
68 int32 TextBlockCount = 0;
69 int32 BlueprintCount = 0;
70
71 const TArray<FAssetData> Assets = FindWidgetBlueprints();
72 FFileHelper::SaveStringToFile(
73 FString::Printf(TEXT("asset-count=%d\n"), Assets.Num()), *ProgressPath,
74 FFileHelper::EEncodingOptions::ForceUTF8WithoutBOM,
75 &IFileManager::Get(), FILEWRITE_Append);
76 for (const FAssetData& Asset : Assets)
77 {
78 FFileHelper::SaveStringToFile(
79 FString::Printf(TEXT("loading=%s\n"), *Asset.PackageName.ToString()),
80 *ProgressPath, FFileHelper::EEncodingOptions::ForceUTF8WithoutBOM,
81 &IFileManager::Get(), FILEWRITE_Append);
82 UWidgetBlueprint* Blueprint = Cast<UWidgetBlueprint>(Asset.GetAsset());
83 if (Blueprint == nullptr || Blueprint->WidgetTree == nullptr)
84 {
85 continue;
86 }
87
88 ++BlueprintCount;
89 Blueprint->WidgetTree->ForEachWidget([&Output, &Asset, &TextBlockCount](UWidget* Widget)
90 {
91 const UTextBlock* TextBlock = Cast<UTextBlock>(Widget);
92 if (TextBlock == nullptr)
93 {
94 return;
95 }
96
97 const FSlateFontInfo Font = TextBlock->GetFont();
98 const UObject* FontObject = Font.FontObject;
99 Output += FString::Printf(TEXT("%s\t%s\t%.3f\t%s\t%s\t%d\t%d\t%s\n"),
100 *Asset.PackageName.ToString(),
101 *TextBlock->GetName(),
102 Font.Size,
103 *SanitizeField(Font.TypefaceFontName.ToString()),
104 FontObject != nullptr ? *FontObject->GetPathName() : TEXT(""),
105 Font.LetterSpacing,
106 Font.OutlineSettings.OutlineSize,
107 *SanitizeField(TextBlock->GetText().ToString()));
108 ++TextBlockCount;
109 });
110 }
111
112 if (!FFileHelper::SaveStringToFile(Output, *OutputPath,
113 FFileHelper::EEncodingOptions::ForceUTF8WithoutBOM))
114 {
115 UE_LOG(LogTemp, Error, TEXT("RD_WIDGET_FONT_AUDIT export failed path=%s"), *OutputPath);
116 return;
117 }
118 FFileHelper::SaveStringToFile(TEXT("complete\n"), *ProgressPath,
119 FFileHelper::EEncodingOptions::ForceUTF8WithoutBOM,
120 &IFileManager::Get(), FILEWRITE_Append);
121
122 UE_LOG(LogTemp, Display,
123 TEXT("RD_WIDGET_FONT_AUDIT export success blueprints=%d text_blocks=%d path=%s"),
124 BlueprintCount, TextBlockCount, *OutputPath);
125 }

127 void Apply(const TArray<FString>& Args)
128 {
129 if (Args.IsEmpty())
130 {
131 UE_LOG(LogTemp, Error,
132 TEXT("RD_WIDGET_FONT_AUDIT usage: RD.Editor.ApplyWidgetFontSizes <absolute-tsv-path>"));
133 return;
134 }
135
136 FString Input;
137 if (!FFileHelper::LoadFileToString(Input, *Args[0]))
138 {
139 UE_LOG(LogTemp, Error, TEXT("RD_WIDGET_FONT_AUDIT cannot read %s"), *Args[0]);
140 return;
141 }
142
143 TMap<FString, TMap<FName, float>> RequestedSizes;
144 TArray<FString> Lines;
145 Input.ParseIntoArrayLines(Lines, true);
146 for (int32 LineIndex = 1; LineIndex < Lines.Num(); ++LineIndex)
147 {
148 TArray<FString> Fields;
149 Lines[LineIndex].ParseIntoArray(Fields, TEXT("\t"), false);
150 if (Fields.Num() < 3)
151 {
152 continue;
153 }
154 RequestedSizes.FindOrAdd(Fields[0]).Add(FName(Fields[1]), FCString::Atof(*Fields[2]));
155 }
156
157 int32 ChangedBlueprints = 0;
158 int32 ChangedTextBlocks = 0;
159 for (const TPair<FString, TMap<FName, float>>& AssetRequest : RequestedSizes)
160 {
161 UWidgetBlueprint* Blueprint = LoadObject<UWidgetBlueprint>(nullptr,
162 *FString::Printf(TEXT("%s.%s"), *AssetRequest.Key,
163 *FPackageName::GetLongPackageAssetName(AssetRequest.Key)));
164 if (Blueprint == nullptr || Blueprint->WidgetTree == nullptr)
165 {
166 UE_LOG(LogTemp, Warning, TEXT("RD_WIDGET_FONT_AUDIT missing blueprint %s"),
167 *AssetRequest.Key);
168 continue;
169 }
170
171 int32 BlueprintChanges = 0;
172 for (const TPair<FName, float>& WidgetRequest : AssetRequest.Value)
173 {
174 UTextBlock* TextBlock = Cast<UTextBlock>(
175 Blueprint->WidgetTree->FindWidget(WidgetRequest.Key));
176 if (TextBlock == nullptr)
177 {
178 UE_LOG(LogTemp, Warning,
179 TEXT("RD_WIDGET_FONT_AUDIT missing text block asset=%s widget=%s"),
180 *AssetRequest.Key, *WidgetRequest.Key.ToString());
181 continue;
182 }
183
184 FSlateFontInfo Font = TextBlock->GetFont();
185 if (Font.Size == WidgetRequest.Value)
186 {
187 continue;
188 }
189 Font.Size = WidgetRequest.Value;
190 TextBlock->SetFont(Font);
191 ++BlueprintChanges;
192 ++ChangedTextBlocks;
193 }
194
195 if (BlueprintChanges == 0)
196 {
197 continue;
198 }
199
200 FBlueprintEditorUtils::MarkBlueprintAsModified(Blueprint);
201 FKismetEditorUtilities::CompileBlueprint(Blueprint);
202 const FString Filename = FPackageName::LongPackageNameToFilename(
203 Blueprint->GetOutermost()->GetName(), FPackageName::GetAssetPackageExtension());
204 if (!UPackage::SavePackage(Blueprint->GetPackage(), Blueprint, *Filename,
205 FSavePackageArgs()))
206 {
207 UE_LOG(LogTemp, Error, TEXT("RD_WIDGET_FONT_AUDIT save failed asset=%s"),
208 *AssetRequest.Key);
209 continue;
210 }
211 ++ChangedBlueprints;
212 }
213
214 UE_LOG(LogTemp, Display,
215 TEXT("RD_WIDGET_FONT_AUDIT apply success blueprints=%d text_blocks=%d"),
216 ChangedBlueprints, ChangedTextBlocks);
217 }

383 bool WrapTextInDownOnlyScaleBox(UWidgetBlueprint* Blueprint,
384 const FName TextName)
385 {
386 UTextBlock* Text = Cast<UTextBlock>(Blueprint->WidgetTree->FindWidget(TextName));
387 if (Text == nullptr)
388 {
389 return false;
390 }
391 // 세로로는 자르지 않는다. 가로는 ScaleToFitX + DownOnly가 이미 막고
392 // 있어서, 자르기가 하는 일은 아래꼬리와 외곽선을 깎는 것뿐이다.
393 // 런타임의 URDUserWidget::NormalizeAutoFitTextClipping과 같은 값이다.
394 constexpr EWidgetClipping AutoFitClipping = EWidgetClipping::Inherit;
395
396 // 이 명령은 여러 번 실행해도 같은 결과가 나와야 한다. 예전 구현은 이미
397 // AutoFit 안에 든 TextBlock을 통째로 건너뛰어, 과거의 RenderTransform,
398 // Padding, 정렬 값이 그대로 남았다. 기존 래퍼도 신규 래퍼와 똑같이
399 // 표준화한다.
400 if (UScaleBox* ExistingScale = Cast<UScaleBox>(Text->GetParent()))
401 {
402 ExistingScale->Modify();
403 if (ExistingScale->Slot != nullptr)
404 {
405 ExistingScale->Slot->Modify();
406 }
407 Text->Modify();
408 if (Text->Slot != nullptr)
409 {
410 Text->Slot->Modify();
411 }
412 ExistingScale->SetStretch(EStretch::ScaleToFitX);
413 ExistingScale->SetStretchDirection(EStretchDirection::DownOnly);
414 ExistingScale->SetClipping(AutoFitClipping);
415 ExistingScale->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
416 if (UOverlaySlot* OuterOverlay = Cast<UOverlaySlot>(ExistingScale->Slot))
417 {
418 OuterOverlay->SetPadding(FMargin(0.f));
419 OuterOverlay->SetHorizontalAlignment(HAlign_Fill);
420 OuterOverlay->SetVerticalAlignment(VAlign_Fill);
421 }
422 else if (UButtonSlot* OuterButton = Cast<UButtonSlot>(ExistingScale->Slot))
423 {
424 OuterButton->SetPadding(FMargin(0.f));
425 OuterButton->SetHorizontalAlignment(HAlign_Fill);
426 OuterButton->SetVerticalAlignment(VAlign_Fill);
427 }
428 if (UScaleBoxSlot* TextSlot = Cast<UScaleBoxSlot>(Text->Slot))
429 {
430 TextSlot->SetHorizontalAlignment(HAlign_Center);
431 TextSlot->SetVerticalAlignment(VAlign_Center);
432 }
433 Text->SetMargin(FMargin(0.f));
434 Text->SetJustification(ETextJustify::Center);
435 Text->SetClipping(AutoFitClipping);
436 if (UWidget* Center = ExistingScale->GetParent())
437 {
438 Center->SetClipping(AutoFitClipping);
439 }
440 Text->SetRenderTransform(FWidgetTransform());
441 Text->SetRenderTransformPivot(FVector2D(.5f));
442 Text->SetAutoWrapText(false);
443 Text->SetMinDesiredWidth(0.f);
444 return true;
445 }
446
447 UPanelWidget* Parent = Text->GetParent();
448 UPanelSlot* OriginalSlot = Text->Slot;
449 if (Parent == nullptr || OriginalSlot == nullptr)
450 {
451 return false;
452 }
453
454 const int32 ChildIndex = Parent->GetChildIndex(Text);
455 UPanelSlot* SlotTemplate = DuplicateObject<UPanelSlot>(
456 OriginalSlot, GetTransientPackage());
457 if (ChildIndex == INDEX_NONE || SlotTemplate == nullptr)
458 {
459 return false;
460 }
461
462 const FName ScaleName(*FString::Printf(TEXT("%s_AutoFit"), *TextName.ToString()));
463 UScaleBox* Scale = Cast<UScaleBox>(Blueprint->WidgetTree->FindWidget(ScaleName));
464 if (Scale == nullptr)
465 {
466 Scale = Blueprint->WidgetTree->ConstructWidget<UScaleBox>(
467 UScaleBox::StaticClass(), ScaleName);
468 Blueprint->OnVariableAdded(ScaleName);
469 }
470
471 Parent->RemoveChildAt(ChildIndex);
472 UPanelSlot* NewOuterSlot = Parent->InsertChildAt(
473 ChildIndex, Scale, SlotTemplate);
474 if (NewOuterSlot == nullptr)
475 {
476 return false;
477 }
478 if (UOverlaySlot* OverlaySlot = Cast<UOverlaySlot>(NewOuterSlot))
479 {
480 // 중앙 정렬된 TextBlock의 desired size가 아니라 부모가 준 전체 상자를
481 // 축소 판단 기준으로 사용한다.
482 OverlaySlot->SetHorizontalAlignment(HAlign_Fill);
483 OverlaySlot->SetVerticalAlignment(VAlign_Fill);
484 }
485 else if (UButtonSlot* ButtonSlot = Cast<UButtonSlot>(NewOuterSlot))
486 {
487 ButtonSlot->SetPadding(FMargin(0.f));
488 ButtonSlot->SetHorizontalAlignment(HAlign_Fill);
489 ButtonSlot->SetVerticalAlignment(VAlign_Fill);
490 }
491
492 Scale->SetStretch(EStretch::ScaleToFitX);
493 Scale->SetStretchDirection(EStretchDirection::DownOnly);
494 Scale->SetClipping(AutoFitClipping);
495 Scale->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
496 Scale->SetContent(Text);
497 if (UScaleBoxSlot* TextSlot = Cast<UScaleBoxSlot>(Text->Slot))
498 {
499 TextSlot->SetHorizontalAlignment(HAlign_Center);
500 TextSlot->SetVerticalAlignment(VAlign_Center);
501 }
502 Text->SetMargin(FMargin(0.f));
503 Text->SetJustification(ETextJustify::Center);
504 Text->SetRenderTransform(FWidgetTransform());
505 Text->SetRenderTransformPivot(FVector2D(.5f));
506 Text->SetAutoWrapText(false);
507 Text->SetMinDesiredWidth(0.f);
508 return true;
509 }

521 bool EnsureMinimumCanvasSize(UWidgetBlueprint* Blueprint,
522 const FName WidgetName, const FVector2D MinimumSize)
523 {
524 UWidget* Host = Blueprint->WidgetTree->FindWidget(WidgetName);
525 while (Host != nullptr && Cast<UCanvasPanelSlot>(Host->Slot) == nullptr)
526 {
527 Host = Host->GetParent();
528 }
529 UCanvasPanelSlot* Slot = Host != nullptr
530 ? Cast<UCanvasPanelSlot>(Host->Slot) : nullptr;
531 if (Slot == nullptr)
532 {
533 return false;
534 }
535 const FVector2D OldSize = Slot->GetSize();
536 const FVector2D NewSize(FMath::Max(OldSize.X, MinimumSize.X),
537 FMath::Max(OldSize.Y, MinimumSize.Y));
538 if (OldSize.Equals(NewSize))
539 {
540 return false;
541 }
542 Slot->Modify();
543 // 현재 버튼 중심은 유지하고 판과 클릭/라벨 영역만 함께 확장한다.
544 Slot->SetPosition(Slot->GetPosition() - (NewSize - OldSize) * .5f);
545 Slot->SetSize(NewSize);
546 return true;
547 }
