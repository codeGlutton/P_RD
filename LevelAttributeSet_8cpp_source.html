101bool ULevelAttributeSet::GetRarityRates(const UObject* WorldContextObject, OUT TArray<FRarityRate>& OutRates)
102{
103 OutRates = TArray<FRarityRate>();
104
105 if (WorldContextObject == nullptr)
106 {
107 return false;
108 }
109
110 const UTacticalFrameworkModel* TacticalFrameworkModel = GetWorldSubsystemModel<UTacticalFrameworkModel>(WorldContextObject);
111 if (TacticalFrameworkModel == nullptr)
112 {
113 return false;
114 }
115
116 FTacticalAttributeSetInitter* Initter = const_cast<UTacticalFrameworkModel*>(TacticalFrameworkModel)->GetAttributeSetInitter();
117 if (Initter == nullptr)
118 {
119 return false;
120 }
121
122 const TArray<float> CommonWeights = Initter->GetAttributeSetValues(ULevelAttributeSet::StaticClass(), GetCommonSkillWeightAttribute().GetUProperty(), KEY_NAME);
123 const TArray<float> RareWeights = Initter->GetAttributeSetValues(ULevelAttributeSet::StaticClass(), GetRareSkillWeightAttribute().GetUProperty(), KEY_NAME);
124 const TArray<float> EpicWeights = Initter->GetAttributeSetValues(ULevelAttributeSet::StaticClass(), GetEpicSkillWeightAttribute().GetUProperty(), KEY_NAME);
125
126 const int32 LevelCount = FMath::Min(CommonWeights.Num(), FMath::Min(RareWeights.Num(), EpicWeights.Num()));
127 OutRates.Reserve(LevelCount);
128
129 for (int32 LevelIndex = 0; LevelIndex < LevelCount; ++LevelIndex)
130 {
131 FRarityRate Rate;
132 Rate.mWeights[static_cast<uint8>(ERarityType::Common)] = CommonWeights[LevelIndex];
133 Rate.mWeights[static_cast<uint8>(ERarityType::Rare)] = RareWeights[LevelIndex];
134 Rate.mWeights[static_cast<uint8>(ERarityType::Epic)] = EpicWeights[LevelIndex];
135 OutRates.Add(Rate);
136 }
137
138 return true;
139}
