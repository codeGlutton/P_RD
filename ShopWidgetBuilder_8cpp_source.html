48 FSlateBrush TextureBrush(UTexture2D* Source, const bool bNineSlice = false,
49 const FLinearColor& Tint = FLinearColor::White)
50 {
51 FSlateBrush Brush;
52 Brush.SetResourceObject(Source);
53 Brush.DrawAs = bNineSlice ? ESlateBrushDrawType::Box : ESlateBrushDrawType::Image;
54 Brush.TintColor = FSlateColor(Tint);
55 if (bNineSlice)
56 {
57 Brush.Margin = FMargin(.16f);
58 }
59 if (Source != nullptr)
60 {
61 const FIntPoint Size = Source->GetImportedSize();
62 Brush.ImageSize = FVector2D(Size.X, Size.Y);
63 }
64 return Brush;
65 }

75 void Place(UCanvasPanel* Parent, UWidget* Child, const FVector2D& Position,
76 const FVector2D& Size, const int32 ZOrder)
77 {
78 UCanvasPanelSlot* Slot = Parent->AddChildToCanvas(Child);
79 const FString ParentName = Parent->GetFName().ToString();
80 const bool bResponsiveParent = ParentName == TEXT("ShopDesignCanvas")
81 || ParentName == TEXT("mArtifactShopPanel")
82 || ParentName == TEXT("mSkillShopPanel")
83 || ParentName == TEXT("mRestShopPanel")
84 || ParentName == TEXT("mArtifactInventoryPanel")
85 || ParentName.StartsWith(TEXT("RestUnitRowHolder_"));
86 Slot->SetAlignment(FVector2D::ZeroVector);
87 Slot->SetAutoSize(false);
88 if (bResponsiveParent && Position.IsNearlyZero()
89 && Size.Equals(FVector2D(DesignWidth, DesignHeight)))
90 {
91 Slot->SetAnchors(FAnchors(0.f, 0.f, 1.f, 1.f));
92 Slot->SetOffsets(FMargin(0.f));
93 }
94 else if (bResponsiveParent)
95 {
96 const FVector2D AnchorPoint(Position.X / DesignWidth,
97 Position.Y / DesignHeight);
98 Slot->SetAnchors(FAnchors(AnchorPoint.X, AnchorPoint.Y));
99 Slot->SetOffsets(FMargin(0.f, 0.f, Size.X, Size.Y));
100 }
101 else
102 {
103 Slot->SetAnchors(FAnchors(0.f));
104 Slot->SetPosition(Position);
105 Slot->SetSize(Size);
106 }
107 Slot->SetZOrder(ZOrder);
108 }
