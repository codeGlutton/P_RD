27bool FTacticalEffectSpec::AttemptCalculateDurationFromDef(OUT int32& ResultDuration) const
28{
29 check(mEffectClass != nullptr);
30
31 bool IsCalculatedDuration = true;
32
33 const ETacticalEffectDurationType DurType = mEffectClass->mDurationPolicy;
34 if (DurType == ETacticalEffectDurationType::Infinite)
35 {
36 ResultDuration = FTacticalEffectConstants::INFINITE_DURATION;
37 }
38 else if (DurType == ETacticalEffectDurationType::Instant)
39 {
40 ResultDuration = FTacticalEffectConstants::NO_DURATION;
41 }
42 else
43 {
44 const int32 FinalDuration = mEffectClass->mDurationMagnitude * mDynamicDurationMagnitude;
45 if (FinalDuration > 0)
46 {
47 ResultDuration = FinalDuration;
48 }
49 else
50 {
51 IsCalculatedDuration = false;
52 }
53 }
54
55 return IsCalculatedDuration;
56}

119void FTacticalEffectSpec::CalculateModifierMagnitudes()
120{
121 for (int32 ModIdx = 0; ModIdx < mModifiers.Num(); ++ModIdx)
122 {
123 const FTacticalModifierInfo& ModDef = mEffectClass->mModifiers[ModIdx];
124 float& EvaluatedMagnitude = mModifiers[ModIdx];
125
126 // 정적 크기 x 동적 배율 = 스택 1개 기준 단일 모디파이어 크기.
127 EvaluatedMagnitude = ModDef.mModifierMagnitude * mDynamicMagnitude;
128 }
129}

131float FTacticalEffectSpec::GetStackedModifierMagnitude(int32 ModifierIdx) const
132{
133 check(mModifiers.IsValidIndex(ModifierIdx) && mEffectClass != nullptr && mEffectClass->mModifiers.IsValidIndex(ModifierIdx));
134
135 // CalculateModifierMagnitudes 가 캐시한 스택 1개 기준 단일 크기.
136 const float SingleEvaluatedMagnitude = mModifiers[ModifierIdx];
137 float ModMagnitude = SingleEvaluatedMagnitude;
138 if (mEffectClass->mFactorInStackCount == true)
139 {
140 /* 스택 반영 */
141
142 ModMagnitude = TacticalEffectUtilities::ComputeStackedModifierMagnitude(SingleEvaluatedMagnitude, GetStackCount(), mEffectClass->mModifiers[ModifierIdx].mModifierOp);
143 }
144 return ModMagnitude;
145}
