62void UUnitAttributeSet::PostAttributeChange(const FTacticalAttribute& Attribute, float OldValue, float NewValue)
63{
64 Super::PostAttributeChange(Attribute, OldValue, NewValue);
65
66 /* 체력 최댓값 변경 시, 체력 초과 방지 */
67 if (Attribute == GetMaxHPAttribute())

69 // MaxHP가 줄어들어 현재 HP가 새 상한을 넘어선 경우에만 HP를 끌어내린다.

72 UAttributeSetComponentModel* ASC = GetOwningAttributeSetComponentModel();
73 // [PR #191 enum 치환 지점] 구 EGameplayModOp::Override -> ETacticalModOp::Override(정수값 3).
74 // Override는 "덮어쓰기" 연산이므로, 기존 HP를 무시하고 새 MaxHP 값(NewValue)으로 강제 설정한다.
75 // (정수값 3은 구 GAS와 동일하게 유지되므로 직렬화/배열 인덱싱/CoreRedirect 호환이 보장된다.)
76 ASC->ApplyModToAttribute(GetHPAttribute(), ETacticalModOp::Override, NewValue);
77 }
78 }

80 if (Attribute == GetHPAttribute() && OldValue > 0.f && NewValue <= 0.f)
