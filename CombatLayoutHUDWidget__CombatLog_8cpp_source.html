121 bool AddDamageNumberGlyphs(UWidgetTree* WidgetTree, UHorizontalBox* Parent,
122 UTexture2D* Atlas, const FText& NumberText, const bool bCritical)
123 {
124 if (WidgetTree == nullptr || Parent == nullptr || Atlas == nullptr)
125 {
126 return false;
127 }
128
129 const FString Characters = NumberText.ToString();
130 if (Characters.IsEmpty())
131 {
132 return false;
133 }
134 for (const TCHAR Character : Characters)
135 {
136 if (DamageNumberGlyphIndex(Character) == INDEX_NONE)
137 {
138 return false;
139 }
140 }
141
142 const FIntPoint ImportedSize = Atlas->GetImportedSize();
143 const float CellAspect = ImportedSize.Y > 0
144 ? (static_cast<float>(ImportedSize.X) / DamageNumberAtlasColumns)
145 / (static_cast<float>(ImportedSize.Y) / DamageNumberAtlasRows)
146 : 0.8f;
147 const float GlyphHeight = bCritical ? 72.0f : 60.0f;
148 const float GlyphWidth = GlyphHeight * CellAspect;
149
150 for (const TCHAR Character : Characters)
151 {
152 const int32 GlyphIndex = DamageNumberGlyphIndex(Character);
153 const int32 Column = GlyphIndex % DamageNumberAtlasColumns;
154 const int32 Row = GlyphIndex / DamageNumberAtlasColumns;
155 const FVector2f UVMin(
156 static_cast<float>(Column) / DamageNumberAtlasColumns,
157 static_cast<float>(Row) / DamageNumberAtlasRows);
158 const FVector2f UVMax(
159 static_cast<float>(Column + 1) / DamageNumberAtlasColumns,
160 static_cast<float>(Row + 1) / DamageNumberAtlasRows);
161
162 UImage* Glyph = WidgetTree->ConstructWidget<UImage>(UImage::StaticClass());
163 if (Glyph == nullptr)
164 {
165 return false;
166 }
167 FSlateBrush Brush;
168 Brush.SetResourceObject(Atlas);
169 Brush.DrawAs = ESlateBrushDrawType::Image;
170 Brush.ImageSize = FVector2D(GlyphWidth, GlyphHeight);
171 Brush.SetUVRegion(FBox2f(UVMin, UVMax));
172 Glyph->SetBrush(Brush);
173 Glyph->SetDesiredSizeOverride(FVector2D(GlyphWidth, GlyphHeight));
174 Glyph->SetVisibility(ESlateVisibility::HitTestInvisible);
175 if (UHorizontalBoxSlot* GlyphSlot = Parent->AddChildToHorizontalBox(Glyph))
176 {
177 GlyphSlot->SetVerticalAlignment(VAlign_Center);
178 GlyphSlot->SetPadding(FMargin(-5.0f, 0.0f, -5.0f, 0.0f));
179 }
180 }
181 return true;
182 }

373void UCombatLayoutHUDWidget::UpdateFloatingCombatLogQueue(float InDeltaTime)
374{
375 // 지도(풀스크린) 열림 중에는 새 플로팅 로그를 스폰하지 않는다(탑바만 남기는 뷰).
376 if (GetVisibility() == ESlateVisibility::Collapsed)
377 {
378 return;
379 }
380
381 if (mPendingFloatingCombatLogs.Num() == 0)
382 {
383 mFloatingCombatLogQueueCooldown = 0.0f;
384 return;
385 }
386
387 mFloatingCombatLogQueueCooldown -= InDeltaTime;
388 if (mFloatingCombatLogQueueCooldown > 0.0f)
389 {
390 return;
391 }
392
393 const FCombatFloatingLogRequest Request = mPendingFloatingCombatLogs[0].mRequest;
394 mPendingFloatingCombatLogs.RemoveAt(0);
395 SpawnFloatingCombatLogAtWorld(Request);
396 mFloatingCombatLogQueueCooldown = FloatingLogQueueInterval;
397}

409{
410 if (mRootCanvas == nullptr || WidgetTree == nullptr)
411 {
412 return;
413 }
414
415 // 아이콘+텍스트 또는 이미지 숫자를 한 덩어리로 움직이게 가로 박스에 담는다.
416 UHorizontalBox* LogBox = WidgetTree->ConstructWidget<UHorizontalBox>(UHorizontalBox::StaticClass());
417 if (LogBox == nullptr)
418 {
419 return;
420 }
421 LogBox->SetVisibility(ESlateVisibility::HitTestInvisible);
422 LogBox->SetRenderTransformPivot(FVector2D(0.5f, 0.5f));
423
424 UTexture2D* DamageNumberAtlas = nullptr;
425 if (Request.mIsPreview == false && Request.mIconType == EFloatingLogIconType::HP)
426 {
427 if (Request.mColorType == EFloatingLogColorType::Heal)
428 {
429 DamageNumberAtlas = mDamageNumberHealAtlas;
430 }
431 else if (Request.mColorType == EFloatingLogColorType::Damage)
432 {
433 DamageNumberAtlas = Request.mIsCritical
434 ? mDamageNumberCriticalAtlas : mDamageNumberNormalAtlas;
435 }
436 }
437 const bool bUsesDamageNumberSkin = AddDamageNumberGlyphs(
438 WidgetTree, LogBox, DamageNumberAtlas, Request.mText, Request.mIsCritical);
439
440 if (bUsesDamageNumberSkin == false)
441 {
442 UTexture2D* Icon = ResolveFloatingLogIcon(Request.mIconType, Request.mColorType);
443 if (Icon != nullptr)
444 {
445 if (UImage* LogIcon = WidgetTree->ConstructWidget<UImage>(UImage::StaticClass()))
446 {
447 LogIcon->SetBrushFromTexture(Icon, false);
448 LogIcon->SetDesiredSizeOverride(FVector2D(26.0f, 26.0f));
449 if (UHorizontalBoxSlot* IconSlot = LogBox->AddChildToHorizontalBox(LogIcon))

451 IconSlot->SetVerticalAlignment(VAlign_Center);

456

458 if (LogText == nullptr)
459 {
460 return;
461 }
462 LogText->SetJustification(ETextJustify::Center);

464 LogText->SetColorAndOpacity(FSlateColor(ResolveFloatingLogColor(Request.mColorType)));

484 UImage* SlashCoreB = nullptr;

486 {
487 // 숫자는 전경에 고정하고, 짧게 터지는 X자 접촉광만 별도 레이어로 움직인다.
488 // 흰 사각 엔진 텍스처를 가늘게 늘여 쓰므로 별도 VFX 에셋 없이도 색/길이 조절이 가능하다.

515 SlashCoreA = AddDamageSlashLayer(WidgetTree, EffectRoot, WhiteTexture,

551 Entry.mIsHeal = bUsesDamageNumberSkin
552 && Request.mColorType == EFloatingLogColorType::Heal;
553

555 {
556 // 미리보기는 상승/페이드가 없으므로, 같은 위치에 이미 뜬 미리보기 수만큼 위로 고정 오프셋을 준다.
557 int32 StackCount = 0;
558 for (const FFloatingCombatLogEntry& Existing : mFloatingCombatLogs)

560 // 퇴장 중인 로그는 곧 사라지므로 쌓기 오프셋 계산에서 제외한다.

577 FFloatingCombatLogEntry& Existing = mFloatingCombatLogs[LogIndex];
578 if (Existing.mIsPreview == false && Existing.mWorldLocation.Equals(Entry.mWorldLocation, 1.0f))
579 {
580 Existing.mStackOffsetY += DamageNumberStackSpacing;
581 }
582 }
583 }

593void UCombatLayoutHUDWidget::UpdateFloatingCombatLogs(float InDeltaTime)
594{
595 if (mFloatingCombatLogs.Num() == 0)
596 {
597 return;
598 }
599
600 // 지도(풀스크린) 열림 중에는 떠 있는 로그를 숨긴다(수명/이동은 멈춤 — 지도 닫으면 재개).
601 if (GetVisibility() == ESlateVisibility::Collapsed)
602 {
603 for (FFloatingCombatLogEntry& Entry : mFloatingCombatLogs)
604 {
605 if (Entry.mRoot != nullptr) { Entry.mRoot->SetVisibility(ESlateVisibility::Collapsed); }
606 }
607 return;
608 }
609
610 APlayerController* PlayerController = GetOwningPlayer();
611
612 for (int32 LogIndex = mFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
613 {
614 FFloatingCombatLogEntry& Entry = mFloatingCombatLogs[LogIndex];
615 Entry.mElapsed += InDeltaTime;
616
617 UWidget* LogRoot = Entry.mRoot;
618 // 미리보기 로그는 수명으로 사라지지 않는다(MotionFinished/Clear로만). 실행 로그만 수명 만료 시 제거.
619 // 퇴장 연출 중이면 수명 만료보다 퇴장(슬라이드+페이드) 완료를 우선한다.
620 if (LogRoot == nullptr || (Entry.mIsPreview == false && Entry.mIsDismissing == false && Entry.mElapsed >= FloatingLogLifetime))
621 {
622 if (LogRoot != nullptr)
623 {
624 LogRoot->RemoveFromParent();
625 }
626 mFloatingCombatLogs.RemoveAt(LogIndex);
627 continue;
628 }
629
630 // 퇴장 중(모션 종료)이면 오른쪽으로 흐르며 페이드아웃하고 연출이 끝나면 제거한다.
631 if (Entry.mIsDismissing == true)
632 {
633 Entry.mDismissElapsed += InDeltaTime;
634 if (Entry.mDismissElapsed >= FloatingLogDismissDuration)
635 {
636 LogRoot->RemoveFromParent();
637 mFloatingCombatLogs.RemoveAt(LogIndex);
638 continue;
639 }
640 }
641
642 // HP바와 같은 투영. 화면 밖이면 숨기되 수명은 계속 흘려보낸다.
643 FVector2D ScreenPosition;
644 const bool bOnScreen = PlayerController != nullptr
645 && UWidgetLayoutLibrary::ProjectWorldLocationToWidgetPosition(
646 PlayerController, Entry.mWorldLocation, ScreenPosition, false);
647 if (bOnScreen == false)
648 {
649 LogRoot->SetVisibility(ESlateVisibility::Collapsed);
650 continue;
651 }
652 LogRoot->SetVisibility(ESlateVisibility::HitTestInvisible);
653
654 float DamageNumberOffsetX = 0.0f;
655 float DamageNumberOffsetY = 0.0f;
656
657 // 이미지 숫자는 타격점에서 팝→정착한 뒤 움직이지 않는다.
658 // 충격의 방향과 속도는 숫자가 아니라 뒤쪽 X자 접촉광이 담당한다.
659 if (Entry.mUsesDamageNumberSkin == true && Entry.mIsPreview == false)
660 {
661 UWidget* NumberRoot = Entry.mNumberRoot != nullptr ? Entry.mNumberRoot.Get() : LogRoot;
662 FVector2D NumberScale = FVector2D::UnitVector;
663 const float PeakScale = Entry.mIsCritical ? 1.18f : (Entry.mIsHeal ? 1.08f : 1.10f);
664 if (Entry.mElapsed < 0.065f)
665 {
666 const float Alpha = FMath::SmoothStep(0.0f, 1.0f, Entry.mElapsed / 0.065f);
667 const float ScaleValue = FMath::Lerp(0.82f, PeakScale, Alpha);
668 NumberScale = FVector2D(ScaleValue, ScaleValue);
669 }
670 else if (Entry.mElapsed < DamageNumberImpactEnd)
671 {
672 const float Alpha = FMath::SmoothStep(0.0f, 1.0f,
673 (Entry.mElapsed - 0.065f) / (DamageNumberImpactEnd - 0.065f));
674 const float ScaleValue = FMath::Lerp(PeakScale, 1.0f, Alpha);
675 NumberScale = FVector2D(ScaleValue, ScaleValue);
676 }
677
678 if (Entry.mElapsed >= DamageNumberExitStart)
679 {
680 const float ExitAlpha = FMath::Clamp((Entry.mElapsed - DamageNumberExitStart)
681 / (FloatingLogLifetime - DamageNumberExitStart), 0.0f, 1.0f);
682 const float ScaleValue = FMath::Lerp(1.0f, 0.96f, ExitAlpha);
683 NumberScale = FVector2D(ScaleValue, ScaleValue);
684 }
685 NumberRoot->SetRenderScale(NumberScale);
686 NumberRoot->SetRenderTransformAngle(0.0f);
687
688 // X자 참격은 0.06초에 터지고, 약 0.3초 동안 길이 잔상만 남긴다.
689 const float EffectEnd = Entry.mIsCritical ? 0.40f : (Entry.mIsHeal ? 0.46f : 0.32f);
690 float SlashLengthScale = 0.0f;
691 float SlashCoreOpacity = 0.0f;
692 float SlashGlowOpacity = 0.0f;
693 if (Entry.mElapsed < 0.06f)
694 {
695 const float Alpha = FMath::SmoothStep(0.0f, 1.0f, Entry.mElapsed / 0.06f);
696 SlashLengthScale = FMath::Lerp(0.05f, 1.15f, Alpha);
697 SlashCoreOpacity = Alpha;
698 SlashGlowOpacity = Alpha * (Entry.mIsHeal ? 0.65f : 1.0f);
699 }
700 else if (Entry.mElapsed < EffectEnd)
701 {
702 const float Alpha = FMath::Clamp((Entry.mElapsed - 0.06f)
703 / (EffectEnd - 0.06f), 0.0f, 1.0f);
704 SlashLengthScale = FMath::Lerp(1.15f, 1.0f, Alpha);
705 SlashCoreOpacity = 1.0f - FMath::SmoothStep(0.0f, 0.62f, Alpha);
706 SlashGlowOpacity = 1.0f - FMath::SmoothStep(0.18f, 1.0f, Alpha);
707 }
708
709 for (UWidget* Slash : { Entry.mSlashGlowA.Get(), Entry.mSlashGlowB.Get() })
710 {
711 if (Slash != nullptr)
712 {
713 Slash->SetRenderScale(FVector2D(SlashLengthScale, 1.0f));
714 Slash->SetRenderOpacity(SlashGlowOpacity);
715 }
716 }
717 for (UWidget* Slash : { Entry.mSlashCoreA.Get(), Entry.mSlashCoreB.Get() })
718 {
719 if (Slash != nullptr)
720 {
721 Slash->SetRenderScale(FVector2D(SlashLengthScale, 1.0f));
722 Slash->SetRenderOpacity(SlashCoreOpacity);
723 }
724 }
725
726 // 숫자 전체는 월드 타격점에 고정한다. 마지막까지 Y 이동을 주지 않는다.
727 DamageNumberOffsetY = -10.0f;
728 }
729
730 // 퇴장 중에는 진행률만큼 오른쪽으로 밀리는 X 오프셋이 붙는다.
731 const float DismissOffsetX = Entry.mIsDismissing == true && Entry.mUsesDamageNumberSkin == false
732 ? FloatingLogDismissSlideSpeed * Entry.mDismissElapsed
733 : 0.0f;
734
735 if (UCanvasPanelSlot* LogSlot = Cast<UCanvasPanelSlot>(LogRoot->Slot))
736 {
737 // 미리보기: 고정 / 이미지 숫자: 충돌 곡선 / 일반 텍스트: 기존 선형 상승.
738 const float OffsetY = Entry.mIsPreview == true
739 ? (FloatingLogBaseOffsetY - Entry.mStackOffsetY)
740 : (Entry.mUsesDamageNumberSkin
741 ? FloatingLogBaseOffsetY - Entry.mStackOffsetY + DamageNumberOffsetY
742 : FloatingLogBaseOffsetY - Entry.mStackOffsetY - FloatingLogRiseSpeed * Entry.mElapsed);
743 const FVector2D CanvasPosition = CombatWorldWidgetPosition::ViewportToCanvas(
744 ScreenPosition, UWidgetLayoutLibrary::GetViewportWidgetGeometry(this),
745 mRootCanvas->GetCachedGeometry());
746 LogSlot->SetPosition(CanvasPosition
747 + FVector2D(DismissOffsetX + DamageNumberOffsetX, OffsetY));
748 }
749
750 // 퇴장 중에는 진행률에 따라 서서히 투명해진다(미리보기/실행 공통).
751 if (Entry.mIsDismissing == true)
752 {
753 LogRoot->SetRenderOpacity(FMath::Clamp(1.0f - Entry.mDismissElapsed / FloatingLogDismissDuration, 0.0f, 1.0f));
754 continue;
755 }
756
757 // 미리보기는 페이드 없이 계속 또렷하게(모션 종료/클리어로만 사라짐).
758 if (Entry.mIsPreview == true)
759 {
760 LogRoot->SetRenderOpacity(1.0f);
761 continue;
762 }
763
764 // (실행 로그) 수명 후반부에 서서히 사라진다.
765 const float FadeStart = Entry.mUsesDamageNumberSkin
766 ? DamageNumberExitStart
767 : FloatingLogLifetime * FloatingLogFadePortion;
768 const float Opacity = Entry.mElapsed <= FadeStart
769 ? 1.0f
770 : 1.0f - (Entry.mElapsed - FadeStart) / (FloatingLogLifetime - FadeStart);
771 LogRoot->SetRenderOpacity(FMath::Clamp(Opacity, 0.0f, 1.0f));
772 }
773}

782void UCombatLayoutHUDWidget::RemoveFloatingCombatLogsByMotionIndex(int32 MotionIndex)
783{
784 if (MotionIndex == INDEX_NONE)
785 {
786 return;
787 }
788
789 for (int32 LogIndex = mPendingFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
790 {
791 if (mPendingFloatingCombatLogs[LogIndex].mRequest.mMotionIndex == MotionIndex)
792 {
793 mPendingFloatingCombatLogs.RemoveAt(LogIndex);
794 }
795 }
796
797 for (int32 LogIndex = mFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
798 {
799 FFloatingCombatLogEntry& Entry = mFloatingCombatLogs[LogIndex];
800 if (Entry.mMotionIndex != MotionIndex)
801 {
802 continue;
803 }
804
805 // 뿅 사라지지 않고 퇴장 연출(오른쪽으로 흐르며 페이드아웃)로 넘긴다. 실제 제거는 UpdateFloatingCombatLogs가 한다.
806 if (Entry.mRoot != nullptr)
807 {
808 Entry.mIsDismissing = true;
809 Entry.mDismissElapsed = 0.0f;
810 }
811 else
812 {
813 mFloatingCombatLogs.RemoveAt(LogIndex);
814 }
815 }
