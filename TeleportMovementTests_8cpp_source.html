94bool FTeleportMovementBasicTests::RunTest(const FString& Parameters)
95{
96 FTestObjectScope Scope;
97
98 UWorld* World = GetAnyGameWorldForTeleportTests();
99 if (World == nullptr)
100 {
101 World = GWorld;
102 }
103 if (TestNotNull(TEXT("유효한 UWorld"), World) == false)
104 {
105 return false;
106 }
107
108 FTeleportFixture Fixture = MakeTeleportFixture(World, Scope, FTileTransform(FTileIndex(2, 2), ETileActorDirection::Left));
109 UMockOverlapSensorModel* StartSensor = PlaceSensor(Fixture, Scope, FTileIndex(2, 2));
110 UMockOverlapSensorModel* EndSensor = PlaceSensor(Fixture, Scope, FTileIndex(5, 5));
111
112 UAttributeSetComponentModel* AttrComp = Fixture.Unit->GetAttributeComponentModel();
113 if (TestNotNull(TEXT("속성 컴포넌트"), AttrComp) == false)
114 {
115 return false;
116 }
117 const float BeforeAP = AttrComp->GetAttributeCurrentValue(UUnitAttributeSet::GetActionPointAttribute());
118
119 // 센서 배치로 생긴 진입 통지는 텔레포트와 무관하므로 기준값으로 기록
120 const int32 StartBeginBase = StartSensor->mBeginCount;
121 const int32 EndBeginBase = EndSensor->mBeginCount;
122
123 int32 FinishCount = 0;
124 const bool Started = Fixture.Movement->TeleportTo(
125 FTileTransform(FTileIndex(5, 5), ETileActorDirection::Left),
126 FOnBoardMoveFinished::CreateLambda([&FinishCount]()
127 {
128 ++FinishCount;
129 }));
130
131 TestTrue(TEXT("시작 성공"), Started);
132 TestTrue(TEXT("목표 타일 도착"), Fixture.Unit->GetTileTransform().mIndex == FTileIndex(5, 5));
133 TestTrue(TEXT("지정한 방향 유지 (Left)"), Fixture.Unit->GetTileTransform().mDirection == ETileActorDirection::Left);
134 TestEqual(TEXT("OnFinished 1회"), FinishCount, 1);
135 TestFalse(TEXT("이동 종료 상태"), Fixture.Movement->IsMoving());
136
137 const float AfterAP = AttrComp->GetAttributeCurrentValue(UUnitAttributeSet::GetActionPointAttribute());
138 TestEqual(TEXT("이동 AP 불변"), AfterAP, BeforeAP);
139
140 TestEqual(TEXT("출발 타일 이탈 통지 1회"), StartSensor->mEndCount, 1);
141 TestEqual(TEXT("도착 타일 진입 통지 1회"), EndSensor->mBeginCount - EndBeginBase, 1);
142 TestEqual(TEXT("출발 타일에 추가 진입 통지 없음"), StartSensor->mBeginCount - StartBeginBase, 0);
143
144 return true;
145}

158bool FTeleportMovementTrapTests::RunTest(const FString& Parameters)
159{
160 FTestObjectScope Scope;
161
162 UWorld* World = GetAnyGameWorldForTeleportTests();
163 if (World == nullptr)
164 {
165 World = GWorld;
166 }
167 if (TestNotNull(TEXT("유효한 UWorld"), World) == false)
168 {
169 return false;
170 }
171
172 FTeleportFixture Fixture = MakeTeleportFixture(World, Scope, FTileTransform(FTileIndex(2, 2), ETileActorDirection::Forward));
173
174 // (5,5) 함정: 진입 통지에서 +Y로 2칸 밀치기 시도 (밀치기 이펙트의 정지 대상 경로 흉내)
175 UMockOverlapSensorModel* TrapSensor = PlaceSensor(Fixture, Scope, FTileIndex(5, 5));
176 bool MovingAtOverlap = true;
177 bool PushStarted = false;
178 TrapSensor->mOnBeginOverlap.BindLambda([&]()
179 {
180 MovingAtOverlap = Fixture.Movement->IsMoving();
181 PushStarted = Fixture.Movement->PushAlongPath({ FTileIndex(5, 5), FTileIndex(5, 6), FTileIndex(5, 7) });
182 });
183
184 int32 FinishCount = 0;
185 Fixture.Movement->TeleportTo(
186 FTileTransform(FTileIndex(5, 5), ETileActorDirection::Forward),
187 FOnBoardMoveFinished::CreateLambda([&FinishCount]()
188 {
189 ++FinishCount;
190 }));
191
192 TestFalse(TEXT("진입 통지 시점에 이동 종료 상태"), MovingAtOverlap);
193 TestTrue(TEXT("함정 밀치기 즉시 시작"), PushStarted);
194 TestTrue(TEXT("밀치기 경로 끝에 도착"), Fixture.Unit->GetTileTransform().mIndex == FTileIndex(5, 7));
195 TestEqual(TEXT("텔레포트 완료 통지 1회"), FinishCount, 1);
196 TestFalse(TEXT("최종 이동 종료 상태"), Fixture.Movement->IsMoving());
197
198 return true;
199}

212bool FTeleportMovementGuardTests::RunTest(const FString& Parameters)
213{
214 FTestObjectScope Scope;
215
216 UWorld* World = GetAnyGameWorldForTeleportTests();
217 if (World == nullptr)
218 {
219 World = GWorld;
220 }
221 if (TestNotNull(TEXT("유효한 UWorld"), World) == false)
222 {
223 return false;
224 }
225
226 /* Case1: 막힌 타일 */
227 AddInfo(TEXT("=== Case1: 도착 타일 점유 -> 시작 거부, 제자리 ==="));
228 {
229 FTeleportFixture Fixture = MakeTeleportFixture(World, Scope, FTileTransform(FTileIndex(2, 2), ETileActorDirection::Forward));
230
231 UMockEnemyUnitModel* Blocker = Scope.New<UMockEnemyUnitModel>(World);
232 Blocker->Initialize();
233 Blocker->BeginPlay();
234 Fixture.TileMap->PlaceActor(FTileTransform(FTileIndex(5, 5), ETileActorDirection::Forward), Blocker);
235
236 int32 FinishCount = 0;
237 const bool Started = Fixture.Movement->TeleportTo(
238 FTileTransform(FTileIndex(5, 5), ETileActorDirection::Forward),
239 FOnBoardMoveFinished::CreateLambda([&FinishCount]()
240 {
241 ++FinishCount;
242 }));
243
244 TestFalse(TEXT("[Case1] 시작 거부"), Started);
245 TestTrue(TEXT("[Case1] 제자리 유지"), Fixture.Unit->GetTileTransform().mIndex == FTileIndex(2, 2));
246 TestTrue(TEXT("[Case1] 점유 유닛 그대로"), Blocker->GetTileTransform().mIndex == FTileIndex(5, 5));
247 TestEqual(TEXT("[Case1] 완료 통지 없음"), FinishCount, 0);
248 TestFalse(TEXT("[Case1] 이동 종료 상태"), Fixture.Movement->IsMoving());
249 }
250
251 /* Case2: 걷는 중 */
252 AddInfo(TEXT("=== Case2: 걷는 중 텔레포트 거부, 걷기 종료 후 허용 ==="));
253 {
254 FTeleportFixture Fixture = MakeTeleportFixture(World, Scope, FTileTransform(FTileIndex(2, 2), ETileActorDirection::Forward));
255
256 // 뷰 흉내: 걷기 스텝 배리어를 잡아 걷는 중 상태를 유지
257 TSharedPtr<FPresentationBarrier> HeldStepBarrier;
258 FDelegateHandle StepHandle = Fixture.Unit->OnStartMoveStep.AddLambda(
259 [&HeldStepBarrier](const FTileTransform&, const FTransform&, TSharedPtr<FPresentationBarrier> Barrier, float, EBoardMoveMode)
260 {
261 HeldStepBarrier = Barrier;
262 });
263
264 Fixture.Movement->MoveAlongPath({ FTileIndex(2, 2), FTileIndex(3, 2) });
265 TestTrue(TEXT("[Case2] 걷는 중 상태"), Fixture.Movement->IsMoving());
266 TestFalse(TEXT("[Case2] 걷는 중 텔레포트 거부"), Fixture.Movement->TeleportTo(FTileTransform(FTileIndex(5, 5), ETileActorDirection::Forward)));
267
268 // 걷기 종료 후에는 허용 (걷기 완주는 배리어 해제로 동기 진행)
269 Fixture.Unit->OnStartMoveStep.Remove(StepHandle);
270 HeldStepBarrier.Reset();
271 TestFalse(TEXT("[Case2] 걷기 종료 상태"), Fixture.Movement->IsMoving());
272 TestTrue(TEXT("[Case2] 걷기 종료 후 텔레포트 허용"), Fixture.Movement->TeleportTo(FTileTransform(FTileIndex(5, 5), ETileActorDirection::Forward)));
273 TestTrue(TEXT("[Case2] 목표 타일 도착"), Fixture.Unit->GetTileTransform().mIndex == FTileIndex(5, 5));
274 }
275
276 return true;
