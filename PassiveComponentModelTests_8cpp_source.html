34bool FPassiveComponentModelTests::RunTest(const FString& Parameters)
35{
36 const FGameplayTag StartTiming = PassiveTiming(TEXT("GameplayAbility.Passive.OnStartApplyingEffect"));
37 const FGameplayTag EndTiming = PassiveTiming(TEXT("GameplayAbility.Passive.OnEndApplyingEffect"));
38 const FGameplayTag UnusedTiming = PassiveTiming(TEXT("GameplayAbility.Passive.OnStartTurn"));
39
40 UPassiveComponentModel* Component = NewObject<UPassiveComponentModel>();
41 if (!TestNotNull(TEXT("컴포넌트 생성"), Component))
42 {
43 return false;
44 }
45
46 // 서로 다른 타입의 패시브 2개 추가 (무상태 / 상태형)
47 UTacticalPassive* AddPassive = Component->AddPassive(UTacticalPassive_AddStat::StaticClass());
48 UTacticalPassive* NthPassive = Component->AddPassive(UTacticalPassive_NthAddStat::StaticClass());
49 if (!TestNotNull(TEXT("AddStat 추가"), AddPassive) || !TestNotNull(TEXT("NthAddStat 추가"), NthPassive))
50 {
51 return false;
52 }
53
54 // 서로 다른 발동 시점 설정 (friend로 protected 접근)
55 // 단일 태그라 기본값은 비어있음 → 테스트용 시점만 지정 (결정적 버킷 검증)
56 AddPassive->mActivateTimingTag = StartTiming;
57 NthPassive->mActivateTimingTag = EndTiming;
58
59 // 보유 수 확인
60 TestEqual(TEXT("보유 패시브 2개"), Component->GetPassives().Num(), 2);
61
62 // 시점별 조회: 일치분만 반환
63 TArray<UTacticalPassive*> StartList = Component->GetPassivesByTiming(StartTiming);
64 TestEqual(TEXT("OnStartApplyingEffect_Add 1개"), StartList.Num(), 1);
65 if (StartList.Num() == 1)
66 {
67 TestEqual(TEXT("OnStartApplyingEffect_Add 대상 일치"), StartList[0], AddPassive);
68 }
69
70 TArray<UTacticalPassive*> EndList = Component->GetPassivesByTiming(EndTiming);
71 TestEqual(TEXT("OnEndApplyingEffect 1개"), EndList.Num(), 1);
72 if (EndList.Num() == 1)
73 {
74 TestEqual(TEXT("OnEndApplyingEffect 대상 일치"), EndList[0], NthPassive);
75 }
76
77 // 미설정 시점 조회: 0개
78 TArray<UTacticalPassive*> NoneList = Component->GetPassivesByTiming(UnusedTiming);
79 TestEqual(TEXT("미설정 시점 0개"), NoneList.Num(), 0);
80
81 // 제거: 성공 후 보유 수 감소, 같은 대상 재제거는 실패
82 TestTrue(TEXT("AddStat 제거 성공"), Component->RemovePassive(AddPassive));
83 TestEqual(TEXT("제거 후 1개"), Component->GetPassives().Num(), 1);
84 TestFalse(TEXT("이미 제거된 대상 재제거 실패"), Component->RemovePassive(AddPassive));
85
86 return true;
87}
