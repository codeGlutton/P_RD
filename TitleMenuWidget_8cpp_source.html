205void UTitleMenuWidget::RefreshResponsiveTitleLayout(const FVector2D& ViewportSize)
206{
207 if (TitleLayoutSwitcher == nullptr)
208 {
209 return;
210 }
211
212 const FName DesiredProfileName = SelectTitleLayoutProfile(ViewportSize);
213 if (DesiredProfileName.IsNone() || DesiredProfileName == mActiveTitleLayoutProfileName)
214 {
215 const bool bViewportChangedEnoughForLog =
216 FMath::Abs(ViewportSize.X - mLastLoggedTitleLayoutViewportSize.X) >= TitleLayoutLogViewportThreshold ||
217 FMath::Abs(ViewportSize.Y - mLastLoggedTitleLayoutViewportSize.Y) >= TitleLayoutLogViewportThreshold;
218 const bool bProfileChangedForLog = DesiredProfileName != mLastLoggedTitleLayoutProfileName;
219 if (DesiredProfileName.IsNone() || (bViewportChangedEnoughForLog == false && bProfileChangedForLog == false))
220 {
221 return;
222 }
223
224 LogResponsiveTitleLayoutMetrics(
225 ViewportSize,
226 DesiredProfileName,
227 TitleLayoutSwitcher->GetActiveWidget(),
228 TitleLayoutSwitcher->GetActiveWidgetIndex());
229 mLastLoggedTitleLayoutProfileName = DesiredProfileName;
230 mLastLoggedTitleLayoutViewportSize = ViewportSize;
231 return;
232 }
233
234 const FName DesiredScaleBoxWidgetName(*FString::Printf(TEXT("TitleLayoutScaleBox_%s"), *DesiredProfileName.ToString()));
235 const FName LegacyLayoutWidgetName(*FString::Printf(TEXT("TitleLayout_%s"), *DesiredProfileName.ToString()));
236 for (int32 WidgetIndex = 0; WidgetIndex < TitleLayoutSwitcher->GetNumWidgets(); ++WidgetIndex)
237 {
238 const UWidget* LayoutWidget = TitleLayoutSwitcher->GetWidgetAtIndex(WidgetIndex);
239 if (LayoutWidget != nullptr && LayoutWidget->GetFName() == DesiredScaleBoxWidgetName)
240 {
241 TitleLayoutSwitcher->SetActiveWidgetIndex(WidgetIndex);
242 mActiveTitleLayoutProfileName = DesiredProfileName;
243 LogResponsiveTitleLayoutMetrics(ViewportSize, DesiredProfileName, LayoutWidget, WidgetIndex);
244 mLastLoggedTitleLayoutProfileName = DesiredProfileName;
245 mLastLoggedTitleLayoutViewportSize = ViewportSize;
246 return;
247 }
248 }
249
250 for (int32 WidgetIndex = 0; WidgetIndex < TitleLayoutSwitcher->GetNumWidgets(); ++WidgetIndex)
251 {
252 const UWidget* LayoutWidget = TitleLayoutSwitcher->GetWidgetAtIndex(WidgetIndex);
253 if (LayoutWidget != nullptr && LayoutWidget->GetFName() == LegacyLayoutWidgetName)
254 {
255 TitleLayoutSwitcher->SetActiveWidgetIndex(WidgetIndex);
256 mActiveTitleLayoutProfileName = DesiredProfileName;
257 LogResponsiveTitleLayoutMetrics(ViewportSize, DesiredProfileName, LayoutWidget, WidgetIndex);
258 mLastLoggedTitleLayoutProfileName = DesiredProfileName;
259 mLastLoggedTitleLayoutViewportSize = ViewportSize;
260 return;
261 }
262 }

264 UE_LOG(
265 LogRD,
266 Warning,
267 TEXT("TitleMenuWidget LayoutMetrics: no layout widget for profile=%s viewport=%s aspect=%.3f expected=%s legacy=%s switcherChildren=%d"),
268 *DesiredProfileName.ToString(),
269 *FormatVec2(ViewportSize),
270 ViewportSize.Y > 0.0f ? ViewportSize.X / ViewportSize.Y : 0.0f,
271 *DesiredScaleBoxWidgetName.ToString(),

277void UTitleMenuWidget::LogResponsiveTitleLayoutMetrics(const FVector2D& ViewportSize, const FName ProfileName, const UWidget* ActiveLayoutWidget, const int32 ActiveWidgetIndex) const
278{
279 const FString ProfileString = ProfileName.ToString();
280 const UWidget* MutableSizeBoxWidget = const_cast<UTitleMenuWidget*>(this)->GetWidgetFromName(FName(*FString::Printf(TEXT("TitleLayoutSizeBox_%s"), *ProfileString)));
281 const USizeBox* ProfileSizeBox = Cast<USizeBox>(MutableSizeBoxWidget);
282 const UWidget* ProfileCanvas = const_cast<UTitleMenuWidget*>(this)->GetWidgetFromName(FName(*FString::Printf(TEXT("TitleLayoutCanvas_%s"), *ProfileString)));
283 const UWidget* LogoWidget = const_cast<UTitleMenuWidget*>(this)->GetWidgetFromName(MakeProfileWidgetName(TEXT("TitleLogoImage"), ProfileName));
284 const UWidget* StartButtonWidget = const_cast<UTitleMenuWidget*>(this)->GetWidgetFromName(MakeProfileWidgetName(TEXT("StartButton"), ProfileName));
285 const UWidget* VersionTextWidget = const_cast<UTitleMenuWidget*>(this)->GetWidgetFromName(MakeProfileWidgetName(TEXT("VersionText"), ProfileName));
286
287 FVector2D DesignSize = FVector2D::ZeroVector;
288 if (ProfileSizeBox != nullptr)
289 {
290 DesignSize.X = ProfileSizeBox->IsWidthOverride() ? ProfileSizeBox->GetWidthOverride() : ProfileSizeBox->GetDesiredSize().X;
291 DesignSize.Y = ProfileSizeBox->IsHeightOverride() ? ProfileSizeBox->GetHeightOverride() : ProfileSizeBox->GetDesiredSize().Y;
292 }
293
294 float CalculatedScale = 0.0f;
295 FVector2D FittedSize = FVector2D::ZeroVector;
296 FVector2D LetterboxInset = FVector2D::ZeroVector;
297 if (ViewportSize.X > 0.0f && ViewportSize.Y > 0.0f && DesignSize.X > 0.0f && DesignSize.Y > 0.0f)
298 {
299 CalculatedScale = FMath::Min(ViewportSize.X / DesignSize.X, ViewportSize.Y / DesignSize.Y);
300 FittedSize = DesignSize * CalculatedScale;
301 LetterboxInset = (ViewportSize - FittedSize) * 0.5f;
302 }
303
304 UE_LOG(
305 LogRD,
306 Display,
307 TEXT("TitleMenuWidget LayoutMetrics: viewport=%s aspect=%.3f profile=%s activeIndex=%d active=%s design=%s scale=%.4f fitted=%s inset=%s switcher=%s activeGeom=%s sizeBox=%s canvas=%s"),
308 *FormatVec2(ViewportSize),
309 ViewportSize.Y > 0.0f ? ViewportSize.X / ViewportSize.Y : 0.0f,
310 *ProfileString,
311 ActiveWidgetIndex,
312 ActiveLayoutWidget != nullptr ? *ActiveLayoutWidget->GetName() : TEXT("missing"),
313 *FormatVec2(DesignSize),
314 CalculatedScale,
315 *FormatVec2(FittedSize),
316 *FormatVec2(LetterboxInset),
317 *DescribeWidgetGeometry(TitleLayoutSwitcher),
318 *DescribeWidgetGeometry(ActiveLayoutWidget),
319 *DescribeWidgetGeometry(ProfileSizeBox),
320 *DescribeWidgetGeometry(ProfileCanvas));
321
322 UE_LOG(
323 LogRD,
324 Display,
325 TEXT("TitleMenuWidget LayoutMetrics Widgets: profile=%s logo={%s} startButton={%s} versionText={%s}"),
326 *ProfileString,
327 *DescribeWidgetGeometry(LogoWidget),
328 *DescribeWidgetGeometry(StartButtonWidget),
329 *DescribeWidgetGeometry(VersionTextWidget));

333FName UTitleMenuWidget::SelectTitleLayoutProfile(const FVector2D& ViewportSize) const
334{
335 if (ViewportSize.X <= 0.0f || ViewportSize.Y <= 0.0f)
336 {
337 return NAME_None;
338 }
339
340 const float AspectRatio = ViewportSize.X / ViewportSize.Y;
341 if (AspectRatio <= 1.35f)
342 {
343 return TitleLayoutProfileFoldInner;
344 }
345 if (AspectRatio <= 1.70f)
346 {
347 return TitleLayoutProfileTablet16x10;
348 }
349 if (AspectRatio <= 1.95f)
350 {
351 return TitleLayoutProfileBase16x9;
352 }
353 if (AspectRatio <= 2.25f)
354 {
355 return TitleLayoutProfilePhoneWide;
356 }
357
358 return TitleLayoutProfilePhoneUltraWide;
359}

447void UTitleMenuWidget::AlignMenuTextBlock(UTextBlock* TextBlock)
448{
449 if (TextBlock == nullptr)
450 {
451 return;
452 }
453
454 TextBlock->SetJustification(ETextJustify::Center);
455
456 if (UOverlaySlot* ExistingOverlaySlot = Cast<UOverlaySlot>(TextBlock->Slot))
457 {
458 ExistingOverlaySlot->SetHorizontalAlignment(HAlign_Fill);
459 ExistingOverlaySlot->SetVerticalAlignment(VAlign_Center);
460 return;
461 }
462
463 UCanvasPanelSlot* SourceCanvasSlot = Cast<UCanvasPanelSlot>(TextBlock->Slot);
464 UPanelWidget* SourceParent = TextBlock->GetParent();
465 if (WidgetTree == nullptr || SourceCanvasSlot == nullptr || SourceParent == nullptr)
466 {
467 return;
468 }
469
470 const FAnchors Anchors = SourceCanvasSlot->GetAnchors();
471 const FMargin Offsets = SourceCanvasSlot->GetOffsets();
472 const FVector2D Alignment = SourceCanvasSlot->GetAlignment();
473 const int32 ZOrder = SourceCanvasSlot->GetZOrder();
474 const bool bAutoSize = SourceCanvasSlot->GetAutoSize();
475
476 const FName OverlayName = MakeUniqueObjectName(this, UOverlay::StaticClass(), *FString::Printf(TEXT("%s_CenterOverlay"), *TextBlock->GetName()));
477 UOverlay* CenterOverlay = WidgetTree->ConstructWidget<UOverlay>(UOverlay::StaticClass(), OverlayName);
478 if (CenterOverlay == nullptr)
479 {
480 return;
481 }
482
483 SourceParent->RemoveChild(TextBlock);
484 if (UCanvasPanel* CanvasParent = Cast<UCanvasPanel>(SourceParent))
485 {
486 UCanvasPanelSlot* OverlayCanvasSlot = CanvasParent->AddChildToCanvas(CenterOverlay);
487 if (OverlayCanvasSlot == nullptr)
488 {
489 return;
490 }
491
492 OverlayCanvasSlot->SetAnchors(Anchors);
493 OverlayCanvasSlot->SetOffsets(Offsets);
494 OverlayCanvasSlot->SetAlignment(Alignment);
495 OverlayCanvasSlot->SetAutoSize(bAutoSize);
496 OverlayCanvasSlot->SetZOrder(ZOrder);
497 }
498 else
499 {
500 SourceParent->AddChild(CenterOverlay);
501 }
502
503 UOverlaySlot* TextOverlaySlot = CenterOverlay->AddChildToOverlay(TextBlock);
504 if (TextOverlaySlot != nullptr)
505 {
506 TextOverlaySlot->SetHorizontalAlignment(HAlign_Fill);
507 TextOverlaySlot->SetVerticalAlignment(VAlign_Center);
508 TextOverlaySlot->SetPadding(FMargin(0.0f));
509 }
