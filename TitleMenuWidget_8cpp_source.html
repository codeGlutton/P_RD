122 void ApplyResponsiveTitleLayout(UUserWidget* Owner, const FVector2D& ViewportSize,
123 const bool bCanContinueRun)
124 {
125 if (Owner == nullptr || ViewportSize.X <= 0.0f || ViewportSize.Y <= 0.0f)
126 {
127 return;
128 }
129
130 // Keep the authored canvas height; compact UI independently on Fold inner displays.
131 // The background retains its own cover fit.
132 if (UScaleBox* LayoutScaleBox = Cast<UScaleBox>(
133 Owner->GetWidgetFromName(TEXT("TitleLayoutScaleBox_base_16_9"))))
134 {
135 LayoutScaleBox->SetStretch(EStretch::ScaleToFitY);
136 LayoutScaleBox->SetStretchDirection(EStretchDirection::Both);
137 }
138
139 constexpr float DesignWidth = 1920.0f;
140 constexpr float DesignHeight = 1080.0f;
141 constexpr float ButtonLeftMargin = 60.0f;
142 const bool bFoldLayout = ViewportSize.X / ViewportSize.Y < 1.6f;
143 const float ElementScale = bFoldLayout ? 0.78f : 1.0f;
144 const float EdgeMargin = bFoldLayout ? 44.0f : ButtonLeftMargin;
145 const float VisibleDesignWidth = DesignHeight * (ViewportSize.X / ViewportSize.Y);
146 const float CroppedDesignMargin = FMath::Max(0.0f, (DesignWidth - VisibleDesignWidth) * 0.5f);
147 const float ButtonX = CroppedDesignMargin + EdgeMargin;
148
149 PositionProfileWidget(Owner, TEXT("TitleLogoImage"), FAnchors(0.5f, 0.0f),
150 FVector2D(0.5f, 0.0f), FVector2D(0.0f, bFoldLayout ? 40.0f : 24.0f), ElementScale);
151
152 // Use the visible right edge, including the sides cropped by ScaleToFitY.
153 const float RightInset = (DesignWidth - VisibleDesignWidth) * 0.5f + EdgeMargin;
154 for (const TCHAR* Name : { TEXT("VersionPlateImage"), TEXT("VersionText") })
155 {
156 PositionProfileWidget(Owner, Name, FAnchors(1.0f, 1.0f), FVector2D(1.0f, 1.0f),
157 FVector2D(-RightInset, -48.0f), bFoldLayout ? 0.9f : 1.0f);
158 }

160 struct FButtonRow

162 const TCHAR* Frame;
163 const TCHAR* Button;
164 const TCHAR* Text;

167 // 이어하기가 숨겨지면 새로 시작이 그 빈 슬롯으로 내려가야 한다.

172 { TEXT("StartButtonFrameImage"), TEXT("StartButton"), TEXT("StartButtonText"), StartBottomOffset },
173 { TEXT("ContinueButtonFrameImage"), TEXT("ContinueButton"), TEXT("ContinueButtonText"), 238.0f },
174 { TEXT("SettingsButtonFrameImage"), TEXT("SettingsButton"), TEXT("SettingsButtonText"), 143.0f },
175 { TEXT("ExitButtonFrameImage"), TEXT("ExitButton"), TEXT("ExitButtonText"), 48.0f },

180 PositionProfileWidget(Owner, Row.Frame, FAnchors(0.0f, 1.0f), FVector2D(0.0f, 1.0f), Position, ElementScale);
181 PositionProfileWidget(Owner, Row.Button, FAnchors(0.0f, 1.0f), FVector2D(0.0f, 1.0f), Position, ElementScale);

186 void ApplyConfiguredTitleLogo(UUserWidget* Owner)
187 {
188 if (Owner == nullptr)
189 {
190 return;
191 }
192
193 const UGamePlaySettings* Settings = GetDefault<UGamePlaySettings>();
194 UObject* LogoTexture = nullptr;
195 if (Settings)
196 {
197 if (FInternationalization::Get().GetCurrentLanguage()->GetTwoLetterISOLanguageName() == TEXT("ko"))
198 LogoTexture = Settings->mTitleLogoTexture.LoadSynchronous();
199 else
200 LogoTexture = Settings->mTitleLogoEnglishMaterial.LoadSynchronous();
201 }
202 if (LogoTexture == nullptr)
203 {
204 UE_LOG(LogRD, Warning, TEXT("TitleMenuWidget: configured title logo texture could not be loaded"));
205 return;
206 }
207
208 const FName LogoNames[] =
209 {
210 TEXT("TitleLogoImage"),
211 MakeProfileWidgetName(TEXT("TitleLogoImage"), TitleLayoutProfileBase16x9),
212 };
213 for (const FName LogoName : LogoNames)
214 {
215 if (UImage* LogoImage = Cast<UImage>(Owner->GetWidgetFromName(LogoName)))
216 {
217 FSlateBrush LogoBrush = LogoImage->GetBrush();
218 LogoBrush.SetResourceObject(LogoTexture);
219 LogoBrush.DrawAs = ESlateBrushDrawType::Image;
220 LogoBrush.ImageSize = FVector2D(1536.0, 1024.0);
221 LogoImage->SetBrush(LogoBrush);
222 LogoImage->SetColorAndOpacity(FLinearColor::White);
223 LogoImage->SetDesiredSizeOverride(FVector2D(600.0f, 400.0f));
224 if (UCanvasPanelSlot* LogoSlot = Cast<UCanvasPanelSlot>(LogoImage->Slot))
225 {
226 LogoSlot->SetAutoSize(false);
227 LogoSlot->SetSize(FVector2D(600.0f, 400.0f));
228 }
229 LogoImage->SetVisibility(ESlateVisibility::HitTestInvisible);
230 UE_LOG(LogRD, Display, TEXT("TitleMenuWidget: title logo applied to %s from %s"),
231 *LogoName.ToString(), *LogoTexture->GetPathName());
232 }
233 }
234 }

400void UTitleMenuWidget::EnsureExitButton()
401{
402 if (WidgetTree == nullptr)
403 {
404 return;
405 }
406
407 auto AddOverlayButton = [this](const FName ButtonName, UWidget* Frame) -> UButton*
408 {
409 UCanvasPanel* CanvasParent = Cast<UCanvasPanel>(Frame->GetParent());
410 UOverlay* OverlayParent = Cast<UOverlay>(Frame->GetParent());
411 if (CanvasParent == nullptr && OverlayParent == nullptr)
412 {
413 return nullptr;
414 }
415
416 UButton* Button = WidgetTree->ConstructWidget<UButton>(
417 UButton::StaticClass(), ButtonName);
418 FButtonStyle Style = Button->GetStyle();
419 Style.Normal.DrawAs = ESlateBrushDrawType::NoDrawType;
420 Style.Hovered.DrawAs = ESlateBrushDrawType::NoDrawType;
421 Style.Pressed.DrawAs = ESlateBrushDrawType::NoDrawType;
422 Style.Disabled.DrawAs = ESlateBrushDrawType::NoDrawType;
423 Button->SetStyle(Style);
424 Button->SetTouchMethod(EButtonTouchMethod::PreciseTap);
425 Button->SetClickMethod(EButtonClickMethod::PreciseClick);
426
427 if (CanvasParent != nullptr)
428 {
429 UCanvasPanelSlot* ButtonSlot = CanvasParent->AddChildToCanvas(Button);
430 if (const UCanvasPanelSlot* FrameSlot = Cast<UCanvasPanelSlot>(Frame->Slot))
431 {
432 ButtonSlot->SetAnchors(FrameSlot->GetAnchors());
433 ButtonSlot->SetAlignment(FrameSlot->GetAlignment());
434 ButtonSlot->SetAutoSize(false);
435 ButtonSlot->SetPosition(FrameSlot->GetPosition());
436 ButtonSlot->SetSize(FrameSlot->GetSize());
437 ButtonSlot->SetZOrder(FrameSlot->GetZOrder() + 20);
438 }
439 return Button;
440 }
441
442 if (UOverlaySlot* ButtonSlot = OverlayParent->AddChildToOverlay(Button))
443 {
444 ButtonSlot->SetPadding(FMargin(0.f));
445 ButtonSlot->SetHorizontalAlignment(HAlign_Fill);
446 ButtonSlot->SetVerticalAlignment(VAlign_Fill);
447 }
448 return Button;
449 };
450
451 for (const FName ProfileName : TitleLayoutProfiles)
452 {
453 const FName ButtonName = MakeProfileWidgetName(TEXT("ExitButton"), ProfileName);
454 if (WidgetTree->FindWidget(ButtonName) != nullptr)
455 {
456 continue;
457 }
458 UWidget* Frame = WidgetTree->FindWidget(
459 MakeProfileWidgetName(TEXT("ExitButtonFrameImage"), ProfileName));
460 if (Frame != nullptr)
461 {
462 AddOverlayButton(ButtonName, Frame);
463 }
464 }
465
466 if (ExitButton == nullptr && WidgetTree->FindWidget(TEXT("ExitButton")) == nullptr)
467 {
468 if (UWidget* Frame = WidgetTree->FindWidget(TEXT("ExitButtonFrameImage")))
469 {
470 ExitButton = AddOverlayButton(TEXT("ExitButton"), Frame);
471 }
472 }

577void UTitleMenuWidget::ApplyMainMenuTextOpticalAlignment()
578{
579 // UMG의 Center는 폰트 줄 박스를 중앙에 놓는다. 현재 번역 문자열을 Slate가
580 // 실제로 shaping/rasterize한 결과의 잉크 경계를 구해 남는 차이만 보정한다.
581 // 문자열/언어/대체 폰트가 달라져도 별도의 언어 조건값은 필요 없다.
582 auto ApplyOffset = [](UTextBlock* TextBlock)
583 {
584 RDTextOpticalAlignment::Apply(TextBlock, TitleMenuFontOpticalBiasY);
585 };
586
587 ApplyOffset(StartButtonText);
588 ApplyOffset(ContinueButtonText);
589 ApplyOffset(SettingsButtonText);
590 ApplyOffset(Cast<UTextBlock>(GetWidgetFromName(TEXT("ExitButtonText"))));
591 ApplyOffset(Cast<UTextBlock>(GetWidgetFromName(TEXT("VersionText"))));
592
593 for (const FName ProfileName : TitleLayoutProfiles)
594 {
595 ApplyOffset(Cast<UTextBlock>(GetWidgetFromName(MakeProfileWidgetName(TEXT("StartButtonText"), ProfileName))));

604void UTitleMenuWidget::AlignMenuTextBlock(UTextBlock* TextBlock)
605{
606 if (TextBlock == nullptr)
607 {
608 return;
609 }
610
611 TextBlock->SetJustification(ETextJustify::Center);
612
613 if (UOverlaySlot* ExistingOverlaySlot = Cast<UOverlaySlot>(TextBlock->Slot))
614 {
615 ExistingOverlaySlot->SetHorizontalAlignment(HAlign_Fill);
616 ExistingOverlaySlot->SetVerticalAlignment(VAlign_Center);
617 return;
618 }
619
620 UCanvasPanelSlot* SourceCanvasSlot = Cast<UCanvasPanelSlot>(TextBlock->Slot);
621 UPanelWidget* SourceParent = TextBlock->GetParent();
622 if (WidgetTree == nullptr || SourceCanvasSlot == nullptr || SourceParent == nullptr)
623 {
624 return;
625 }
626
627 const FAnchors Anchors = SourceCanvasSlot->GetAnchors();
628 const FMargin Offsets = SourceCanvasSlot->GetOffsets();
629 const FVector2D Alignment = SourceCanvasSlot->GetAlignment();
630 const int32 ZOrder = SourceCanvasSlot->GetZOrder();
631 const bool bAutoSize = SourceCanvasSlot->GetAutoSize();
632
633 const FName OverlayName = MakeUniqueObjectName(this, UOverlay::StaticClass(), *FString::Printf(TEXT("%s_CenterOverlay"), *TextBlock->GetName()));
634 UOverlay* CenterOverlay = WidgetTree->ConstructWidget<UOverlay>(UOverlay::StaticClass(), OverlayName);
635 if (CenterOverlay == nullptr)
636 {
637 return;
638 }
639
640 SourceParent->RemoveChild(TextBlock);
641 if (UCanvasPanel* CanvasParent = Cast<UCanvasPanel>(SourceParent))
642 {
643 UCanvasPanelSlot* OverlayCanvasSlot = CanvasParent->AddChildToCanvas(CenterOverlay);
644 if (OverlayCanvasSlot == nullptr)
645 {
646 return;
647 }
648
649 OverlayCanvasSlot->SetAnchors(Anchors);
650 OverlayCanvasSlot->SetOffsets(Offsets);
651 OverlayCanvasSlot->SetAlignment(Alignment);
652 OverlayCanvasSlot->SetAutoSize(bAutoSize);
653 OverlayCanvasSlot->SetZOrder(ZOrder);
654 }
655 else
656 {
657 SourceParent->AddChild(CenterOverlay);
658 }
659
660 UOverlaySlot* TextOverlaySlot = CenterOverlay->AddChildToOverlay(TextBlock);
661 if (TextOverlaySlot != nullptr)
662 {
663 TextOverlaySlot->SetHorizontalAlignment(HAlign_Fill);
664 TextOverlaySlot->SetVerticalAlignment(VAlign_Center);
665 TextOverlaySlot->SetPadding(FMargin(0.0f));
666 }
667}
