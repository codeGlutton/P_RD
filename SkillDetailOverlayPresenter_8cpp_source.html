87 {

90 return;
91 }

93 const float Height = static_cast<float>(Texture->GetSizeY());
94 if (Width <= 0.f || Height <= 0.f || Aspect <= 0.f)
95 {
96 Image->SetBrushFromTexture(Texture, false);
97 return;
98 }
99
100 FSlateBrush Brush = Image->GetBrush();

102 Brush.DrawAs = ESlateBrushDrawType::Image;

104 // 칸보다 세로로 길면 위에서 잘라 쓰고, 가로로 길면 가운데를 쓴다.
105 const float WantHeight = Width / Aspect;
106 if (WantHeight < Height)
107 {
108 Brush.ImageSize = FVector2D(Width, WantHeight);
109 Brush.SetUVRegion(FBox2f(FVector2f(0.f, 0.f),
110 FVector2f(1.f, WantHeight / Height)));
111 }
112 else
113 {
114 const float WantWidth = Height * Aspect;
115 const float Margin = (1.f - WantWidth / Width) * 0.5f;
116 Brush.ImageSize = FVector2D(WantWidth, Height);
117 Brush.SetUVRegion(FBox2f(FVector2f(Margin, 0.f),
118 FVector2f(1.f - Margin, 1.f)));
119 }
120 Image->SetBrush(Brush);
121 }

134 bool SelectCovers(ECombatSkillSelectShapeUI Shape, int32 Range, int32 dRow, int32 dColumn)
135 {
136 const int32 Manhattan = FMath::Abs(dRow) + FMath::Abs(dColumn);
137 const int32 Chebyshev = FMath::Max(FMath::Abs(dRow), FMath::Abs(dColumn));
138 switch (Shape)
139 {
140 case ECombatSkillSelectShapeUI::Single: return dRow == 0 && dColumn == 0;
141 case ECombatSkillSelectShapeUI::Cross: return (dRow == 0 || dColumn == 0) && Manhattan <= Range;
142 case ECombatSkillSelectShapeUI::Square: return Chebyshev <= Range;
143 case ECombatSkillSelectShapeUI::Diagonal: return Chebyshev <= Range
144 && (dRow == 0 || dColumn == 0 || FMath::Abs(dRow) == FMath::Abs(dColumn));
145 case ECombatSkillSelectShapeUI::Line: return (dRow == 0 || dColumn == 0) && Manhattan <= Range;
146 default: return false;
147 }
148 }
149
150 bool HitCovers(ECombatSkillHitShapeUI Shape, int32 Range, int32 dRow, int32 dColumn)
151 {
152 const int32 Manhattan = FMath::Abs(dRow) + FMath::Abs(dColumn);
153 const int32 Chebyshev = FMath::Max(FMath::Abs(dRow), FMath::Abs(dColumn));
154 switch (Shape)
155 {
156 case ECombatSkillHitShapeUI::Single: return Manhattan == 0;
157 case ECombatSkillHitShapeUI::Cross: return (dRow == 0 || dColumn == 0) && Manhattan <= Range;
158 // 8방향: 직교 축 또는 정확한 대각선. TileMapModel::GetEffectTiles의
159 // Star(직교+대각 8방향)와 같은 모양 -- 전술판(SkillTacticalDiagram)과 동일 판정.
160 case ECombatSkillHitShapeUI::Star: return Chebyshev <= Range
161 && (dRow == 0 || dColumn == 0 || FMath::Abs(dRow) == FMath::Abs(dColumn));
162 case ECombatSkillHitShapeUI::Circle: return Chebyshev <= Range;
163 default: return false;
164 }

200 FSlateBrush MakeRangeButtonBrush(UTexture2D* Texture,
201 const FLinearColor& Tint = FLinearColor::White)

203 FSlateBrush Brush;
204 if (Texture != nullptr)

209 Brush.DrawAs = ESlateBrushDrawType::Image;
210 // 소스의 투명 캔버스를 UV 단계에서 잘라, 그림과 라벨이 같은

221 else

224 Brush.OutlineSettings = FSlateBrushOutlineSettings(3.f,
225 FSlateColor(FLinearColor(.69f, .43f, .16f, 1.f)), 1.25f);
226 }

344bool USkillDetailOverlayPresenter::EnsureOverlayWidget(
345 APlayerController* OwningPlayer)
346{
347 if (mDetailOverlayWidget != nullptr)
348 {
349 return true;
350 }
351 if (mDetailOverlayWidgetClass == nullptr)
352 {
353 return false;
354 }
355 APlayerController* ResolvedOwningPlayer = OwningPlayer;
356 if (ResolvedOwningPlayer == nullptr)
357 {
358 if (UWorld* World = mWorld.Get())
359 {
360 ResolvedOwningPlayer = World->GetFirstPlayerController();
361 }
362 }
363 if (ResolvedOwningPlayer != nullptr)
364 {
365 mDetailOverlayWidget =
366 CreateWidget<UUserWidget>(ResolvedOwningPlayer,
367 mDetailOverlayWidgetClass);
368 }
369 else if (UWorld* World = mWorld.Get())
370 {
371 // 에디터 자동화처럼 로컬 플레이어가 없는 월드에서도 상세 계약은
372 // 검증할 수 있어야 한다. 실제 플레이에서는 늘 위의 소유 플레이어를 쓴다.
373 mDetailOverlayWidget =
374 CreateWidget<UUserWidget>(World, mDetailOverlayWidgetClass);
375 }
376 if (mDetailOverlayWidget == nullptr)
377 {
378 return false;
379 }
380 // 몬스터 도감(55)에서 스킬을 오래 눌러도 상세는 반드시 그 위에 온다.

502void USkillDetailOverlayPresenter::ApplyResponsiveSkillPanelScale(
503 const bool bSkillDetail)
504{
505 mResponsiveDetailActive = true;
506 mResponsiveDetailIsSkill = bSkillDetail;
507 if (mDetailOverlayWidget == nullptr)
508 {
509 return;
510 }
511
512 if (mDetailDesignCanvas == nullptr)
513 {
514 mDetailDesignCanvas = Cast<UCanvasPanel>(
515 mDetailOverlayWidget->GetWidgetFromName(TEXT("DetailResponsiveCanvas")));
516 }
517
518 if (mDetailDesignCanvas == nullptr)
519 {
520 return;
521 }
522 if (mDetailScaleDefaultsCached == false)
523 {
524 mDefaultDetailDesignScale =
525 mDetailDesignCanvas->GetRenderTransform().Scale;
526 mDefaultDetailDesignTranslation =
527 mDetailDesignCanvas->GetRenderTransform().Translation;
528 mDefaultDetailDesignPivot = mDetailDesignCanvas->GetRenderTransformPivot();
529 mDetailScaleDefaultsCached = true;
530 }
531
532 // 실제 반응형 배율은 WBP의 DetailResponsiveScale(ScaleToFit)이 담당한다.
533 // 여기서 루트 Canvas를 다시 RenderScale 하면 DPI 배율과 중복되어 폴드/태블릿에서
534 // 판이 커지고 닫기 버튼이 화면 밖으로 잘린다. 디자인 Canvas는 항상 원본 transform이다.
535 mDetailDesignCanvas->SetRenderTransformPivot(mDefaultDetailDesignPivot);
536 mDetailDesignCanvas->SetRenderTranslation(mDefaultDetailDesignTranslation);
537 mDetailDesignCanvas->SetRenderScale(mDefaultDetailDesignScale);
538
539 const FVector2D ViewportSize =
540 UWidgetLayoutLibrary::GetViewportSize(mDetailOverlayWidget);
541 if (mDetailOverlayWidget->GetOwningPlayer() == nullptr
542 || ViewportSize.X <= 0.f || ViewportSize.Y <= 0.f)
543 {
544 // WidgetRenderer 자동화처럼 소유 플레이어/GameViewport가 없는 경로에서는
545 // 위치 보정을 계산할 수 없다. authored 1920 위치와 배율을 그대로 쓴다.
546 return;
547 }
548
549 const float ViewportScale = FMath::Max(KINDA_SMALL_NUMBER,
550 UWidgetLayoutLibrary::GetViewportScale(mDetailOverlayWidget));
551 const float ResponsiveScale = bSkillDetail
552 ? CalculateResponsiveSkillPanelScale(ViewportSize, ViewportScale)
553 : CalculateResponsiveDetailPanelScale(ViewportSize, ViewportScale);
554 const FVector2D LogicalViewport = ViewportSize / ViewportScale;
555 const FVector2D DesignSize(1920.f, 1080.f);
556 const FVector2D PresentedDesignSize = DesignSize * ResponsiveScale;
557 mLastResponsiveViewportSize = ViewportSize;
558 mLastResponsiveViewportScale = ViewportScale;
559 mDetailOverlayWidget->ForceLayoutPrepass();
560 UE_LOG(LogTemp, Display,
561 TEXT("RD_DETAIL_RESPONSIVE viewport=%.0fx%.0f dpi=%.3f logical=%.0fx%.0f fit=%.3f letterbox=%.0f,%.0f skill=%d"),
562 ViewportSize.X, ViewportSize.Y, ViewportScale,
563 LogicalViewport.X, LogicalViewport.Y, ResponsiveScale,
564 (LogicalViewport.X - PresentedDesignSize.X) * .5f,
565 (LogicalViewport.Y - PresentedDesignSize.Y) * .5f,
566 bSkillDetail ? 1 : 0);
567}
568
569float USkillDetailOverlayPresenter::CalculateResponsiveDetailPanelScale(
570 const FVector2D ViewportSize, const float ViewportScale)
571{
572 if (ViewportSize.X <= 0.f || ViewportSize.Y <= 0.f
573 || ViewportScale <= KINDA_SMALL_NUMBER)
574 {
575 return 1.f;
576 }
577 const FVector2D LogicalViewport = ViewportSize / ViewportScale;
578 const float FitScale = FMath::Min(LogicalViewport.X / 1920.f,
579 LogicalViewport.Y / 1080.f);
580 // 4K에서 원화를 무한히 키우지는 않되 폴드의 2176 폭은 자연스럽게 채운다.
581 return FMath::Clamp(FitScale, .5f, 1.25f);
582}

585 const FVector2D ViewportSize, const float ViewportScale)
586{
587 // 스킬도 닫기 단추와 설명 스크롤까지 전부 보여야 한다. 판 내부만 기준으로
588 // 다시 키우면 폴드에서 1.5배가 되어 하단 조작부가 화면 밖으로 밀린다.
589 return CalculateResponsiveDetailPanelScale(ViewportSize, ViewportScale);
