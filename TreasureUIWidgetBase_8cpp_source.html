156void UTreasureUIWidgetBase::RefreshView()
157{
158 if (mUIModel == nullptr)
159 {
160 return;
161 }
162
163 const FTreasureUI& Treasure = mUIModel->GetTreasure();
164
165 // 개봉 버튼은 개봉 전에만 노출
166 if (mOpenButton != nullptr)
167 {
168 mOpenButton->SetVisibility(Treasure.mIsOpened
169 ? ESlateVisibility::Collapsed : ESlateVisibility::Visible);
170 }
171
172 if (mItemBox == nullptr)
173 {
174 return;
175 }
176 mItemBox->ClearChildren();
177
178 for (const FTreasureItemUI& Item : Treasure.mItems)
179 {
180 // 보상 카드 = 세로 박스(아이콘 + 이름 + 수량). 종류별 아이콘 자동 매핑
181 UVerticalBox* Card = WidgetTree->ConstructWidget<UVerticalBox>(UVerticalBox::StaticClass());
182 if (Card == nullptr)
183 {
184 continue;
185 }
186
187 if (UImage* Icon = WidgetTree->ConstructWidget<UImage>(UImage::StaticClass()))
188 {
189 if (UTexture2D* Tex = ResolveTreasureIcon(Item))
190 {
191 Icon->SetBrushFromTexture(Tex, false);
192 }
193 Icon->SetDesiredSizeOverride(FVector2D(112.f, 112.f));
194 Card->AddChildToVerticalBox(Icon);
195 }
196
197 if (UTextBlock* Name = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass()))
198 {
199 Name->SetText(Item.mName);
200 Name->SetJustification(ETextJustify::Center);
201 // 등급은 이름 색으로 표시 (골드 카드는 기본 흰색)
202 Name->SetColorAndOpacity(FSlateColor(Item.mRarityColor));
203 if (UVerticalBoxSlot* NameSlot = Card->AddChildToVerticalBox(Name))
204 {
205 NameSlot->SetPadding(FMargin(0.f, 6.f, 0.f, 0.f));
206 NameSlot->SetHorizontalAlignment(HAlign_Center);
207 }
208 }
209
210 // 수량 라벨 (골드처럼 수량이 있는 카드만)
211 if (Item.mAmount > 0)
212 {
213 if (UTextBlock* Amount = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass()))
214 {
215 Amount->SetText(FText::Format(LOCTEXT("{0} G", "{0} G"), FText::AsNumber(Item.mAmount)));
216 Amount->SetJustification(ETextJustify::Center);
217 Amount->SetColorAndOpacity(FSlateColor(FLinearColor(0.95f, 0.85f, 0.45f, 1.f)));
218 if (UVerticalBoxSlot* AmountSlot = Card->AddChildToVerticalBox(Amount))
219 {
220 AmountSlot->SetPadding(FMargin(0.f, 2.f, 0.f, 0.f));
221 AmountSlot->SetHorizontalAlignment(HAlign_Center);
222 }
223 }
224 }
225
226 if (UHorizontalBoxSlot* BoxSlot = mItemBox->AddChildToHorizontalBox(Card))
227 {
228 BoxSlot->SetPadding(FMargin(14.f, 0.f));
229 }
230 }
231}
