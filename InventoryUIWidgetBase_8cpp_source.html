129void UInventoryUIWidgetBase::FillSection(UHorizontalBox* Box, const TArray<FInventoryItemUI>& Items)
130{
131 if (Box == nullptr)
132 {
133 return;
134 }
135 Box->ClearChildren();
136
137 for (const FInventoryItemUI& Item : Items)
138 {
139 UVerticalBox* Card = WidgetTree->ConstructWidget<UVerticalBox>(UVerticalBox::StaticClass());
140 if (Card == nullptr)
141 {
142 continue;
143 }
144
145 if (UImage* Icon = WidgetTree->ConstructWidget<UImage>(UImage::StaticClass()))
146 {
147 if (UTexture2D* Tex = ResolveInventoryIcon(Item))
148 {
149 Icon->SetBrushFromTexture(Tex, false);
150 }
151 Icon->SetDesiredSizeOverride(FVector2D(96.f, 96.f));
152 if (Item.mRarityColor != FLinearColor::White)
153 {
154 Icon->SetColorAndOpacity(Item.mRarityColor);
155 }
156 Card->AddChildToVerticalBox(Icon);
157 }
158
159 if (UTextBlock* Name = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass()))
160 {
161 Name->SetText(Item.mName);
162 Name->SetJustification(ETextJustify::Center);
163 if (UVerticalBoxSlot* NameSlot = Card->AddChildToVerticalBox(Name))
164 {
165 NameSlot->SetPadding(FMargin(0.f, 4.f, 0.f, 0.f));
166 NameSlot->SetHorizontalAlignment(HAlign_Center);
167 }
168 }
169
170 if (UHorizontalBoxSlot* BoxSlot = Box->AddChildToHorizontalBox(Card))
171 {
172 BoxSlot->SetPadding(FMargin(10.f, 0.f));
173 }
174 }
175}

178void UInventoryUIWidgetBase::RefreshView()
179{
180 if (mUIModel == nullptr)
181 {
182 return;
183 }
184
185 const FInventoryUI& Inv = mUIModel->GetInventory();
186
187 if (mMetaText != nullptr)
188 {
189 mMetaText->SetText(FText::Format(
190 LOCTEXT("Gold {0} Lv {1} HP {2} / {3}", "Gold {0} Lv {1} HP {2} / {3}"),
191 FText::AsNumber(Inv.mGold),
192 FText::AsNumber(Inv.mLevel),
193 FText::AsNumber(FMath::RoundToInt(Inv.mHP)),
194 FText::AsNumber(FMath::RoundToInt(Inv.mMaxHP))));
195 }
196
197 FillSection(mDiceBox, Inv.mDice);
198 FillSection(mSkillBox, Inv.mSkills);
199 FillSection(mEquipBox, Inv.mEquipment);
200}
