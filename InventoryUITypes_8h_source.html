24// UI 필요값:

27// - mName/mIcon: 슬롯 기본 표시.
28// - mRarityColor: 희귀도 테두리/배경(어댑터가 enum→색 변환).
29// - mDetailText: 다이스면 "d20", 장비면 슬롯명 등 보조 한 줄.
30USTRUCT(BlueprintType)

32{
33 GENERATED_BODY()

35 UPROPERTY(BlueprintReadOnly) EInventoryItemKind mKind = EInventoryItemKind::Dice;

37 UPROPERTY(BlueprintReadOnly) FText mName;
38 UPROPERTY(BlueprintReadOnly) TObjectPtr<UTexture2D> mIcon = nullptr;
39 UPROPERTY(BlueprintReadOnly) FLinearColor mRarityColor = FLinearColor::White;
40 UPROPERTY(BlueprintReadOnly) FText mDetailText;
41};
