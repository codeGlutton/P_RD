43 bool BuildCurrentRunViews(UWorld* World, TArray<FMapRoomView>& OutRooms,
44 bool& bOutAtStageStart)
45 {
46 OutRooms.Reset();
47 bOutAtStageStart = false;
48 if (World == nullptr || World->GetGameInstance() == nullptr)
49 {
50 return false;
51 }
52
53 const UPersistentDataSubsystem* Persistent =
54 World->GetGameInstance()->GetSubsystem<UPersistentDataSubsystem>();
55 const URunPersistData* Run = Persistent != nullptr
56 ? Persistent->GetRunPersistData() : nullptr;
57 if (Run == nullptr || Run->IsActive() == false)
58 {
59 return false;
60 }
61
62 int32 CurrentRow = INDEX_NONE;
63 int32 CurrentColumn = INDEX_NONE;
64 Run->GetCurrentRoomIndex(OUT CurrentRow, OUT CurrentColumn);
65 const FStage& Stage = Run->GetStage();
66 bOutAtStageStart = CurrentRow == 0 && CurrentColumn == Stage.mStartColumn;
67
68 const FRoom* CurrentRoom = Stage.HasRoom(CurrentRow, CurrentColumn)
69 ? &Stage.mRoomRows[CurrentRow].mRooms[CurrentColumn].Get<FRoom>() : nullptr;
70 for (int32 RowIndex = 0; RowIndex < Stage.mRoomRows.Num(); ++RowIndex)
71 {
72 const FRoomRow& RoomRow = Stage.mRoomRows[RowIndex];
73 for (int32 ColumnIndex = 0; ColumnIndex < RoomRow.mRooms.Num(); ++ColumnIndex)
74 {
75 if (Stage.HasRoom(RowIndex, ColumnIndex) == false)
76 {
77 continue;
78 }
79
80 const FRoom& Room = RoomRow.mRooms[ColumnIndex].Get<FRoom>();
81 const bool bReady = CurrentRoom != nullptr
82 && RowIndex == CurrentRow + 1
83 && CurrentRoom->mNextRoomColumns.Contains(ColumnIndex);
84 const EMapRoomState State = Room.mWasSelected
85 ? EMapRoomState::Cleared
86 : (bReady ? EMapRoomState::Ready : EMapRoomState::Locked);
87
88 FMapRoomView& View = OutRooms.AddDefaulted_GetRef();
89 View.mRow = RowIndex;
90 View.mColumn = ColumnIndex;
91 View.mType = Room.mType;
92 View.mState = State;
93 View.mTitle = RowIndex == 0 && ColumnIndex == Stage.mStartColumn
94 ? NSLOCTEXT("FrontendMapPreview", "Start", "Start") : RoomTitle(Room.mType);
95 View.mNextRoomColumns = Room.mNextRoomColumns;
96 View.mPositionOffsetRate = Room.mPositionOffsetRate;
97 View.mSelectable = false;
98 View.mSelected = false;
99 View.mVisited = State == EMapRoomState::Cleared;
100 View.mCanEnter = false;
101 View.mIsStartPoint = RowIndex == 0 && ColumnIndex == Stage.mStartColumn;
102 }
103 }
104 return OutRooms.IsEmpty() == false;
105 }

107 void ShowCurrentRunData(UWorld* World)
108 {
109 if (World == nullptr || World->IsGameWorld() == false)
110 {
111 return;
112 }
113
114 TArray<FMapRoomView> Rooms;
115 bool bAtStageStart = false;
116 if (BuildCurrentRunViews(World, OUT Rooms, OUT bAtStageStart) == false)
117 {
118 UE_LOG(LogRD, Warning, TEXT("RD.MapPreview.Data: 복구된 런 데이터가 아직 없습니다."));
119 return;
120 }
121
122 UFrontendMapWidget* MapWidget = DataPreviewWidget.Get();
123 if (MapWidget == nullptr)
124 {
125 UClass* MapClass = LoadClass<UFrontendMapWidget>(nullptr,
126 TEXT("/Game/UI/WorldMapLandscape/WBP_FrontendMapLandscape."
127 "WBP_FrontendMapLandscape_C"));
128 MapWidget = MapClass != nullptr
129 ? CreateWidget<UFrontendMapWidget>(World, MapClass) : nullptr;
130 if (MapWidget == nullptr)
131 {
132 UE_LOG(LogRD, Warning, TEXT("RD.MapPreview.Data: 가로 지도 WBP 생성 실패."));
133 return;
134 }
135 MapWidget->AddToViewport(10000);
136 DataPreviewWidget = MapWidget;
137 }
138
139 MapWidget->SetRoomSelectionEnabled(false);
140 MapWidget->SetPreviewRoomsForDebug(Rooms, bAtStageStart);
141 MapWidget->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
142 MapWidget->RefreshMap();
143 UE_LOG(LogRD, Display, TEXT("RD.MapPreview.Data: 격리 런 지도 %d개 방을 표시했습니다."), Rooms.Num());
144 }

146 void CaptureCurrentRunData(const TArray<FString>& Args, UWorld* World)
147 {
148 ShowCurrentRunData(World);
149 UFrontendMapWidget* MapWidget = DataPreviewWidget.Get();
150 if (MapWidget == nullptr)
151 {
152 return;
153 }
154
155 int32 CaptureWidth = 1672;
156 int32 CaptureHeight = 941;
157 if (Args.Num() > 0)
158 {
159 if (Args.Num() < 2)
160 {
161 UE_LOG(LogRD, Warning,
162 TEXT("사용법: RD.MapPreview.DataCapture [폭 높이]"));
163 return;
164 }
165 CaptureWidth = FMath::Clamp(FCString::Atoi(*Args[0]), 320, 4096);
166 CaptureHeight = FMath::Clamp(FCString::Atoi(*Args[1]), 320, 4096);
167 }
168
169 const FVector2D CaptureSize(
170 StaticCast<float>(CaptureWidth), StaticCast<float>(CaptureHeight));
171 MapWidget->ForceLayoutPrepass();
172 FWidgetRenderer Renderer(true, true);
173 UTextureRenderTarget2D* RenderTarget = NewObject<UTextureRenderTarget2D>();
174 RenderTarget->RenderTargetFormat = RTF_RGBA8_SRGB;
175 RenderTarget->ClearColor = FLinearColor::Transparent;
176 RenderTarget->InitCustomFormat(
177 CaptureWidth, CaptureHeight, PF_B8G8R8A8, true);
178 RenderTarget->UpdateResourceImmediate(true);
179 Renderer.DrawWidget(RenderTarget, MapWidget->TakeWidget(), CaptureSize, 0.f);
180
181 const FString OutputDirectory = FPaths::Combine(
182 FPaths::ProjectSavedDir(), TEXT("Screenshots"), TEXT("WindowsEditor"));
183 IFileManager::Get().MakeDirectory(*OutputDirectory, true);
184 const FString OutputName = Args.Num() > 0
185 ? FString::Printf(TEXT("WBP_FrontendMapLandscape_RunData_%dx%d_20260812.png"),
186 CaptureWidth, CaptureHeight)
187 : TEXT("WBP_FrontendMapLandscape_RunData_20260812.png");
188 UKismetRenderingLibrary::ExportRenderTarget(
189 World, RenderTarget, OutputDirectory, OutputName);
190 UE_LOG(LogRD, Display, TEXT("RD.MapPreview.DataCapture: %s"),
191 *FPaths::Combine(OutputDirectory, OutputName));
192 }

194 void Toggle(UWorld* World)
195 {

235 MapWidget->OpenUI();
236 MapWidget->RefreshMap();

239 UE_LOG(LogRD, Display,
240 TEXT("RD.MapPreview: 저장 변경 없이 현재 런 지도를 열었습니다."));
