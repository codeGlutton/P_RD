45void FSkillEffectLayer_Spawn::CommitEffect(const FSkillEffectCommitParams& Params) const
46{
47 if (Params.mInstigator == nullptr || mSpawnDatas.IsEmpty() == true)
48 {
49 return;
50 }
51
52 UObject* InstigatorObject = Params.mInstigator.GetObject();
53 USRPGCombatModel* CombatModel = GetWorldSubsystemModel<USRPGCombatModel>(InstigatorObject);
54 UTileMapModel* TileMapModel = CombatModel->GetTileMap();
55
56 const FTileTransform& OriginTileTransform = Params.mInstigatorSnapshot->mTileTransform;
57 int32 mSpawnDataIndex = 0;
58
59 for (const FTileIndex& FinalTileIndex : Params.mFinalTileIndexes)
60 {
61 const FDynamicPlacementData& SpawnData = mSpawnDatas[mSpawnDataIndex];
62 const UStaticObstacleSpawnData* ObstacleSpawnData = SpawnData.mSpawnData.LoadSynchronous();
63 const TSubclassOf<UBoardActorModel> BoardActorModelClass = ObstacleSpawnData->mModelClass.LoadSynchronous();
64
65 if (TileMapModel->CanPlace(FinalTileIndex, GetDefault<UBoardActorModel>(BoardActorModelClass)) == true)
66 {
67 if (SpawnData.IsUnit())
68 {
69 FEnemyUnitPlacementData EnemyPlacement = SpawnData.MakeEnemyUnitPlacementData(FTileTransform(FinalTileIndex, OriginTileTransform.mDirection));
70 if (CombatModel != nullptr)
71 {
72 CombatModel->RegisterEnemyUnitModel(EnemyPlacement);
73 }
74 }
75 else
76 {
77 FObstaclePlacementData ObstaclePlacement = SpawnData.MakeObstaclePlacementData(FTileTransform(FinalTileIndex, OriginTileTransform.mDirection));
78 if (CombatModel != nullptr)
79 {
80 CombatModel->RegisterObstacleModel(ObstaclePlacement);
81 }
82 }
83 }
84
85 mSpawnDataIndex = (mSpawnDataIndex + 1) % mSpawnDatas.Num();
86 }
87}
