96 const auto CheckCulture = [this, World, SettingsClass, &Expected](
97 const TCHAR* Culture, const bool bKorean)
98 {
99 if (!FInternationalization::Get().SetCurrentCulture(Culture))
100 {
101 AddError(FString::Printf(TEXT("문화 변경 실패: %s"), Culture));
102 return;
103 }
104 // TakeWidget/NativeConstruct를 호출하지 않아 C++ SyncText가 값을
105 // 덮어쓰지 않는 순수 WBP 기본값을 검증한다.
106 USettingsPanelWidget* Panel =
107 CreateWidget<USettingsPanelWidget>(World, SettingsClass);
108 if (!TestNotNull(FString::Printf(TEXT("%s WBP 인스턴스"), Culture), Panel))
109 {
110 return;
111 }
112 USettingsPanelWidget* RuntimePanel =
113 CreateWidget<USettingsPanelWidget>(World, SettingsClass);
114 if (!TestNotNull(FString::Printf(TEXT("%s 런타임 인스턴스"), Culture),
115 RuntimePanel))
116 {
117 return;
118 }
119 RuntimePanel->ApplyValueModel(RuntimePanel->GetValueModel());
120 for (const FExpectedText& Entry : Expected)
121 {
122 UTextBlock* Text = Cast<UTextBlock>(
123 Panel->WidgetTree->FindWidget(FName(Entry.WidgetName)));
124 if (TestNotNull(FString::Printf(TEXT("%s/%s"), Culture, Entry.WidgetName), Text))
125 {
126 TestEqual(FString::Printf(TEXT("%s/%s 문구"), Culture, Entry.WidgetName),
127 Text->GetText().ToString(),
128 FString(bKorean ? Entry.Korean : Entry.English));
129 if (UTextBlock* RuntimeText = Cast<UTextBlock>(
130 RuntimePanel->WidgetTree->FindWidget(FName(Entry.WidgetName))))
131 {
132 TestEqual(FString::Printf(TEXT("%s/%s WBP-C++ 일치"),
133 Culture, Entry.WidgetName), Text->GetText().ToString(),
134 RuntimeText->GetText().ToString());
135 }
136 }
137 }
138 };
139
140 CheckCulture(TEXT("ko"), true);
141 CheckCulture(TEXT("en"), false);
142 return !HasAnyErrors();
143}
