37bool LevelUpSkillReward::IsEligibleForUnit(UPlayerUnitModel* Unit, UStaticUnitSkillData* Skill)
38{
39 if (!Unit || !Skill || Skill->mJobType == EUnitJobType::Common || Skill->mJobType == EUnitJobType::None
40 || Skill->mJobType != Unit->GetUnitJobType() || Skill->mSkillPhaseLayers.IsEmpty()
41 || !GameplayAssetPolicy::IsPlayerFacing(Skill->GetPrimaryAssetId())) return false;
42 const USkillComponentModel* Skills = Unit->GetSkillComponentModel();

46bool LevelUpSkillReward::RefreshOffer(FLevelUpSkillReward& Reward, UPlayerUnitModel* Unit,
47 const TArray<UStaticUnitSkillData*>& Pool, const FRarityRate& Rate, const FRandomStream& Stream)
48{
49 if (Reward.Completed || !Unit || !Unit->GetSkillComponentModel()) return false;
50 const int32 Removed = Reward.Candidates.RemoveAll([Unit](const FPrimaryAssetId& Id)
51 { return !IsEligibleForUnit(Unit, LoadSkill(Id)); });
52 if (Reward.Offered && Removed == 0) return false;
53 if (!Reward.Offered) Reward.Candidates.Reset();
54 TArray<UStaticUnitSkillData*> Eligible;
55 for (UStaticUnitSkillData* Skill : Pool)
56 if (IsEligibleForUnit(Unit, Skill) && !Reward.Candidates.Contains(Skill->GetPrimaryAssetId()))
57 Eligible.Add(Skill);
58 Reward.Candidates.Append(ChooseCandidates(Eligible, Rate, Stream, 3 - Reward.Candidates.Num()));
59 Reward.Offered = true;
60 return true;
61}

63TArray<FPrimaryAssetId> LevelUpSkillReward::ChooseCandidates(const TArray<UStaticUnitSkillData*>& Pool,
64 const FRarityRate& Rate, const FRandomStream& Stream, int32 MaxChoices)
65{
66 TArray<UStaticUnitSkillData*> Remaining;
67 for (UStaticUnitSkillData* Skill : Pool)
68 if (Skill && GameplayAssetPolicy::IsPlayerFacing(Skill->GetPrimaryAssetId())) Remaining.AddUnique(Skill);
69 Remaining.Sort([](const UStaticUnitSkillData& A, const UStaticUnitSkillData& B)
70 { return A.GetPrimaryAssetId().ToString() < B.GetPrimaryAssetId().ToString(); });
71 TArray<FPrimaryAssetId> Result;
72 while (!Remaining.IsEmpty() && Result.Num() < FMath::Clamp(MaxChoices, 0, 3))
73 {
74 TArray<ERarityType> Rarities;
75 for (const UStaticUnitSkillData* Skill : Remaining) Rarities.AddUnique(Skill->mRarityType);
76 float TotalWeight = 0.f;
77 for (ERarityType Rarity : Rarities) TotalWeight += FMath::Max(0.f, Rate.mWeights[static_cast<uint8>(Rarity)]);
78 ERarityType Rarity = Rarities.Last();
79 if (TotalWeight > 0.f)
80 {
81 float Roll = Stream.FRand() * TotalWeight;
82 for (ERarityType Candidate : Rarities)
83 {
84 Roll -= FMath::Max(0.f, Rate.mWeights[static_cast<uint8>(Candidate)]);
85 if (Roll < 0.f) { Rarity = Candidate; break; }
86 }
87 }
88 else Rarity = Rarities[Stream.RandRange(0, Rarities.Num() - 1)];
89 TArray<int32> Matching;
90 for (int32 Index = 0; Index < Remaining.Num(); ++Index)
91 if (Remaining[Index]->mRarityType == Rarity) Matching.Add(Index);
92 const int32 Pick = Matching[Stream.RandRange(0, Matching.Num() - 1)];
93 Result.Add(Remaining[Pick]->GetPrimaryAssetId());
94 Remaining.RemoveAt(Pick);
95 }
96 return Result;
97}

112bool LevelUpSkillReward::TryEquip(FLevelUpSkillReward& Reward, UPlayerUnitModel* Unit,
113 const FPrimaryAssetId& SkillId, int32 SkillSlot)
114{
115 if (Reward.Completed || !Reward.Offered || !GameplayAssetPolicy::IsPlayerFacing(SkillId) || !Reward.Candidates.Contains(SkillId)
116 || !Unit || SkillSlot > 4) return false;
117 USkillComponentModel* Skills = Unit->GetSkillComponentModel();
118 UStaticUnitSkillData* Skill = LoadSkill(SkillId);
119 if (!Skills || !IsEligibleForUnit(Unit, Skill) || !Skills->GetSkills().IsValidIndex(SkillSlot)
120 || IsOwned(Skills, SkillId) || !Skills->SetSkill(SkillSlot, Skill)) return false;
121 Reward.Completed = true;
122 return true;
123}

181void ULevelUpSkillRewardFlow::ShowNext(int32 PreferredUnit)
182{
183 TArray<FLevelUpSkillReward>& Rewards = mRun->GetRoomTransactionsMutable().LevelUpSkills;
184 mActiveReward = LevelUpSkillReward::FindPendingReward(Rewards, PreferredUnit);
185 if (mActiveReward == INDEX_NONE) mActiveReward = LevelUpSkillReward::FindPendingReward(Rewards);
186 if (mActiveReward == INDEX_NONE) { Close(); return; }
187 FLevelUpSkillReward& Reward = Rewards[mActiveReward];
188 const auto& Units = mParty->GetPlayerUnitModels();
189 UPlayerUnitModel* Unit = Units.IsValidIndex(Reward.UnitIndex) ? Units[Reward.UnitIndex].Get() : nullptr;
190 if (!Unit) { UE_LOG(LogTemp, Error, TEXT("Level-up reward target is missing")); return; }
191 USkillComponentModel* Skills = Unit->GetSkillComponentModel();
192 if (!Skills) return;
193 const bool NeedsOffer = !Reward.Offered || Reward.Candidates.ContainsByPredicate([Unit](const FPrimaryAssetId& Id)
194 { return !LevelUpSkillReward::IsEligibleForUnit(Unit, LoadSkill(Id)); });
195 if (NeedsOffer)
196 {
197 UAssetManager* Manager = UAssetManager::GetIfInitialized();
198 if (!Manager || !Skills) return;
199 TArray<FPrimaryAssetId> Ids;
200 Manager->GetPrimaryAssetIdList(SkillPrimaryAssetTypes::GetActiveType(), Ids);
201 TArray<UStaticUnitSkillData*> Pool;
202 for (const FPrimaryAssetId& Id : Ids)
203 {
204 UStaticUnitSkillData* Skill = LoadSkill(Id);
205 if (Skill) Pool.Add(Skill);
206 }
207 if (LevelUpSkillReward::RefreshOffer(Reward, Unit, Pool,
208 ULevelAttributeSet::GetRarityRate(mController, Reward.Level), mRun->GetEventStream())) Save();
209 }
210
211 FShopUI View;
212 View.mIsLevelUpReward = true;
213 View.mRewardTitle = FText::Format(NSLOCTEXT("LevelUpReward", "Target", "{0} · Lv.{1}"),
214 Unit->GetBoardActorDisplayName(), FText::AsNumber(Reward.Level));
215 View.mRewardOfferId = mActiveReward;
216 View.mRewardUnitIndex = Reward.UnitIndex;
217 for (int32 UnitIndex = 0; UnitIndex < Units.Num(); ++UnitIndex)
218 {
219 const int32 Pending = LevelUpSkillReward::FindPendingReward(Rewards, UnitIndex);
220 if (Pending == INDEX_NONE || !Units[UnitIndex]) continue;
221 FShopOwnedUnitUI Target;
222 Target.mUnitIndex = UnitIndex;
223 Target.mJobType = Units[UnitIndex]->GetUnitJobType();
224 Target.mLevel = Rewards[Pending].Level;
225 for (const FSkillEntry& Entry : Units[UnitIndex]->GetSkillComponentModel()->GetSkills())
226 {
227 FShopOwnedSkillSlotUI& Slot = Target.mSkillSlots.AddDefaulted_GetRef();
228 Slot.mIsEmpty = !Entry.mData;
229 if (Entry.mData)
230 {
231 Slot.mName = Entry.mData->mName;
232 Slot.mIcon = Entry.mData->mIcon.LoadSynchronous();
233 Slot.mDescription = Entry.mData->mDescription;
234 }
235 }
236 View.mOwnedUnits.Add(Target);
237 View.mSkillTargetUnits.Add(Target);
238 }
239 for (int32 Index = 0; Index < Reward.Candidates.Num(); ++Index)
240 {
241 const UStaticUnitSkillData* Skill = LoadSkill(Reward.Candidates[Index]);
242 if (!Skill) { UE_LOG(LogTemp, Error, TEXT("Level-up candidate failed to load")); return; }
243 FShopItemUI& Item = View.mItems.AddDefaulted_GetRef();
244 // Encode the offer as well, so a double tap from the previous offer cannot consume the next one.
245 Item.mSlotIndex = mActiveReward * 3 + Index;
246 Item.mRequiredJobType = Skill->mJobType;
247 Item.mName = Skill->mName;
248 Item.mIcon = Skill->mIcon.LoadSynchronous();
249 Item.mDescription = Skill->mDescription;
250 Item.mRarityColor = Skill->mRarityType == ERarityType::Epic ? FLinearColor(.72f, .46f, .92f)
251 : Skill->mRarityType == ERarityType::Rare ? FLinearColor(.42f, .66f, .95f) : FLinearColor(.72f, .78f, .75f);
252 }
253 if (!mWidget && mWidgetClass) mWidget = CreateWidget<UShopUIWidgetBase>(mController, mWidgetClass);
254 if (!mWidget) { UE_LOG(LogTemp, Error, TEXT("Level-up shop screen could not be created")); return; }
255 SuspendRewardScreens();
256 mUIModel->SetShop(View);
257 mWidget->BindUIModel(mUIModel);
258 mWidget->OpenUI();
259 mWidget->SetVisibility(ESlateVisibility::Visible);
260 mWidget->ShowFirstVisitGuide();
261}

263void ULevelUpSkillRewardFlow::Select(int32 Choice, int32 UnitIndex, int32 SkillSlot)
264{
265 if (!mRun || mActiveReward == INDEX_NONE || Choice < 0 || Choice / 3 != mActiveReward) return;
266 FLevelUpSkillReward& Reward = mRun->GetRoomTransactionsMutable().LevelUpSkills[mActiveReward];
267 if (Reward.UnitIndex != UnitIndex || !Reward.Candidates.IsValidIndex(Choice % 3)) return;
268 const auto& Units = mParty->GetPlayerUnitModels();
269 if (!Units.IsValidIndex(UnitIndex)) return;
