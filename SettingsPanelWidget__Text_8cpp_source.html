37 const TFunction<UTextBlock*(UWidget*)> FindFirstTextBlock = [&FindFirstTextBlock](UWidget* Root) -> UTextBlock*
38 {
39 if (Root == nullptr)
40 {
41 return nullptr;
42 }
43 if (UTextBlock* TextBlock = Cast<UTextBlock>(Root))
44 {
45 return TextBlock;
46 }
47 if (UContentWidget* ContentWidget = Cast<UContentWidget>(Root))
48 {
49 if (UTextBlock* TextBlock = FindFirstTextBlock(ContentWidget->GetContent()))
50 {
51 return TextBlock;
52 }
53 }
54 if (UPanelWidget* PanelWidget = Cast<UPanelWidget>(Root))
55 {
56 const int32 ChildCount = PanelWidget->GetChildrenCount();
57 for (int32 ChildIndex = 0; ChildIndex < ChildCount; ++ChildIndex)
58 {
59 if (UTextBlock* TextBlock = FindFirstTextBlock(PanelWidget->GetChildAt(ChildIndex)))
60 {
61 return TextBlock;
62 }
63 }
64 }
65 return nullptr;
66 };

68 {
69 if (WidgetTree == nullptr)
70 {
71 return;
72 }
73 if (UTextBlock* TextBlock = FindFirstTextBlock(WidgetTree->FindWidget(WidgetName)))
74 {
75 TextBlock->SetText(InText);
76 }
77 };
