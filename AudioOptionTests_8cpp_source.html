22 UWorld* World = GEditor->GetEditorWorldContext().World();

24 if (!TestNotNull(TEXT("Feedback controller"), PC)) return false;
25 // Editor worlds do not run the normal game controller registration lifecycle.

28 TStrongObjectPtr<UOptionPersistData> Options(NewObject<UOptionPersistData>(World));
29 Options->SetVibrationEnabled(false);
30 TestFalse(TEXT("Disabling vibration reaches the controller"), bool(PC->bForceFeedbackEnabled));
31 TArray<uint8> Bytes;
32 TestTrue(TEXT("Serialize saved vibration preference"), RDCheckpoint::Serialize(Options.Get(), Bytes));
33 TStrongObjectPtr<UOptionPersistData> Restored(NewObject<UOptionPersistData>(World));
34 TestTrue(TEXT("Restore vibration preference"), RDCheckpoint::Deserialize(Bytes, Restored.Get()));
35 TestFalse(TEXT("Vibration remains disabled after reload"), Restored->IsVibrationEnabled());
36 Restored->SetVibrationEnabled(true);
37 TestTrue(TEXT("Re-enabling vibration reaches the controller"), bool(PC->bForceFeedbackEnabled));
38 return !HasAnyErrors();
39}
