39bool FEquipmentComponentModelTests::RunTest(const FString& Parameters)
40{
41 const FGameplayTag StartTiming = PassiveTiming(TEXT("GameplayAbility.Passive.OnStartTurn"));
42
43 // 컴포넌트 (오너 없이 독립 생성, 내부 시드로 형제 컴포넌트를 직접 주입)
44 UEquipmentComponentModel* EquipComp = NewObject<UEquipmentComponentModel>();
45 UPassiveComponentModel* PassiveComp = NewObject<UPassiveComponentModel>();
46 if (!TestNotNull(TEXT("장비 컴포넌트 생성"), EquipComp) || !TestNotNull(TEXT("패시브 컴포넌트 생성"), PassiveComp))
47 {
48 return false;
49 }
50
51 // 패시브 DA (코드 구성): AddStat 패시브 + AttackPoint 이펙트 + 수치 5 + OnStartTurn 시점
52 UStaticPassiveData* PassiveData = NewObject<UStaticPassiveData>();
53 PassiveData->mPassiveClass = UTacticalPassive_AddStat::StaticClass();
54 PassiveData->mEffectClass = UTacticalEffect_AttackFactor_AddBase::StaticClass();
55 PassiveData->mMagnitude = 5.f;
56 PassiveData->mActivateTimingTag = StartTiming;
57
58 // 무기 DA (코드 구성): 위 패시브를 참조
59 UStaticWeaponEquipmentData* Weapon = NewObject<UStaticWeaponEquipmentData>();
60 Weapon->mStaticPassiveData.Add(TSoftObjectPtr<UStaticPassiveData>(PassiveData));
61
62 // 장착: 패시브가 설치돼야 함
63 const bool bEquipped = EquipComp->EquipInternal(Weapon, PassiveComp, nullptr);
64 TestTrue(TEXT("장착 성공"), bEquipped);
65 TestEqual(TEXT("설치된 패시브 1개"), PassiveComp->GetPassives().Num(), 1);
66
67 // 발동 시점 태그로 조회됨 + 정적 데이터가 주입됨
68 TArray<UTacticalPassive*> StartList = PassiveComp->GetPassivesByTiming(StartTiming);
69 TestEqual(TEXT("OnStartTurn 패시브 1개"), StartList.Num(), 1);
70 if (StartList.Num() == 1)
71 {
72 TestEqual(TEXT("주입된 정적 데이터 일치"), StartList[0]->GetStaticData(), PassiveData);
73 }
74
75 // 무기 슬롯 점유 확인
76 TestNotNull(TEXT("무기 슬롯 점유"), EquipComp->GetEquipped(EEquipmentType::Weapon));
77
78 // 해제: 패시브가 제거돼야 함
79 const bool bUnequipped = EquipComp->UnequipInternal(EEquipmentType::Weapon, PassiveComp, nullptr);
80 TestTrue(TEXT("해제 성공"), bUnequipped);
81 TestEqual(TEXT("해제 후 패시브 0개"), PassiveComp->GetPassives().Num(), 0);
82 TestNull(TEXT("무기 슬롯 비움"), EquipComp->GetEquipped(EEquipmentType::Weapon));
83
84 // 빈 슬롯 재해제는 실패
85 TestFalse(TEXT("빈 슬롯 재해제 실패"), EquipComp->UnequipInternal(EEquipmentType::Weapon, PassiveComp, nullptr));
86
87 return true;
88}
