229 AssetManager->GetPrimaryAssetObject<UStaticSkillData>(ItemId))

388 }

416void AShopGameMode::MarkOwnedSaleItems(FShopUI& ShopUIData) const
417{
418 UPartyModel* PartyModel = GetPartyModel();
419 if (PartyModel == nullptr)
420 {
421 return;
422 }
423
424 // 유닛 index -> 그 유닛이 가진 스킬 자산 집합.
425 TMap<int32, TSet<FPrimaryAssetId>> OwnedSkillIds;
426 const TArray<TObjectPtr<UPlayerUnitModel>>& UnitModels =
427 PartyModel->GetPlayerUnitModels();
428 for (int32 UnitIndex = 0; UnitIndex < UnitModels.Num(); ++UnitIndex)
429 {
430 const UPlayerUnitModel* UnitModel = UnitModels[UnitIndex];
431 const USkillComponentModel* SkillModel = UnitModel != nullptr
432 ? UnitModel->GetSkillComponentModel() : nullptr;
433 if (SkillModel == nullptr)
434 {
435 continue;
436 }
437 TSet<FPrimaryAssetId>& Owned = OwnedSkillIds.FindOrAdd(UnitIndex);
438 for (const FSkillEntry& Entry : SkillModel->GetSkills())
439 {
440 if (Entry.mData != nullptr)
441 {
442 Owned.Add(Entry.mData->GetPrimaryAssetId());
443 }
444 }
445 }
446
447 for (FShopItemUI& Item : ShopUIData.mItems)
448 {
449 if (Item.mKind != EShopItemKind::Skill)
450 {
451 continue;
452 }
453 const FShopSlotSource* Source = mSlotSources.Find(Item.mSlotIndex);
454 if (Source == nullptr || Source->mItemId.IsValid() == false)
455 {
456 continue;
457 }
458 for (const TPair<int32, TSet<FPrimaryAssetId>>& Pair : OwnedSkillIds)
459 {
460 if (Pair.Value.Contains(Source->mItemId))
461 {
462 Item.mOwnedByUnitIndices.Add(Pair.Key);
463 }
464 }
465 Item.mOwnedByUnitIndices.Sort();
466 }

700 && SkillSlotIndex < ReplaceableSkillStartIndex + ReplaceableSkillSlotCount;
701 if (!bReplaceableSkillSlot || Skills.IsValidIndex(SkillSlotIndex) == false)

703 UE_LOG(LogRD, Log,
704 TEXT("스킬 구매 거절: 교체 가능한 스킬 슬롯 범위 밖 (Unit=%d, SkillSlot=%d)"),

716 UE_LOG(LogRD, Log, TEXT("스킬 구매 거절: 스킬 데이터 로드 실패 (Slot=%d)"), SlotIndex);
717 return;
718 }
719 for (const FSkillEntry& ExistingSkill : Skills)
720 {
721 if (ExistingSkill.mData != nullptr
722 && ExistingSkill.mData->GetPrimaryAssetId() == Source->mItemId)
723 {
724 UE_LOG(LogRD, Log,
725 TEXT("스킬 구매 거절: 대상 유닛이 이미 보유한 스킬 (Slot=%d, Unit=%d)"),

728 }
729 }
730

736 }

741 // 과금 + 품절 처리
742 SpendPartyGold(Item->mPrice);
