69template<typename ElementType, typename ContainerType>
70class FActiveTacticalEffectIterator
71{
72public:
78 FActiveTacticalEffectIterator(const ContainerType& Container, int32 StartIdx = 0) :
79 mContainer(const_cast<ContainerType&>(Container)),
80 mIndex(StartIdx),
81 mPending(nullptr),
82 mCurrent(nullptr)
83 {
84 // 순회 동안 컨테이너 변형을 막기 위해 락 카운트 증가
85 mContainer.IncrementLock();
86 if (mIndex >= 0)
87 {
88 UpdateCurrent();
89 }
90 }
91
92 // 소멸자: 락 카운트를 해제하여 보류된 증감을 반영할 수 있게 한다.
93 ~FActiveTacticalEffectIterator()
94 {
95 mContainer.DecrementLock();
96 }
97
98public:
99 // 전위 증가: 다음 유효 요소로 이동
100 void operator++()
101 {
102 Next();
103 }
104
105 // 역참조: 현재 가리키는 요소 반환(유효성 단언)
106 ElementType& operator*() const
107 {
108 check(mCurrent != nullptr);
109 return *mCurrent;
110 }
111 // 멤버 접근: 현재 가리키는 요소 포인터 반환(유효성 단언)
112 ElementType* operator->() const
113 {
114 check(mCurrent != nullptr);
115 return mCurrent;
116 }
117
118 // bool 변환: 순회가 아직 유효한 요소를 가리키는지 여부
119 explicit operator bool() const
120 {
121 return (mCurrent != nullptr);
122 }
123
124 // 동등 비교: 같은 노드를 가리키는지(주로 end 반복자와 비교에 사용)
125 friend bool operator==(const FActiveTacticalEffectIterator& Lhs, const FActiveTacticalEffectIterator& Rhs)
126 {
127 return Lhs.mCurrent == Rhs.mCurrent;
128 }
129 friend bool operator!=(const FActiveTacticalEffectIterator& Lhs, const FActiveTacticalEffectIterator& Rhs)
130 {
131 return Lhs.mCurrent != Rhs.mCurrent;
132 }
133
134private:
140 ElementType* AdvancePending(ElementType** Next)
141 {
142 // tail은 "마지막 다음" 슬롯을 가리키므로, 거기에 닿으면 순회 종료(nullptr)
143 return (Next != const_cast<ElementType**>(mContainer.mPendingTacticalEffectTail)) ? *Next : nullptr;
144 }
145
150 void Next()
151 {
152 if (mIndex >= 0)
153 {
154 // 일반 이펙트는 해당 경로로 Iter 증가
155 ++mIndex;
156 }
157 else if (mPending != nullptr)
158 {
159 // 증감 이펙트는 해당 경로로 Iter 증가
160 mPending = AdvancePending(const_cast<ElementType**>(&mPending->mPendingNext));
161 }
162 UpdateCurrent();
163 }
164
170 void UpdateCurrent()
171 {
172 if (mIndex < 0)
173 {
174 // 이미 추가 예약된 객체 순회로 전환됨. Linked-List에 따라서 Pending을 현재 탐색 노드로 지정
175 mCurrent = mPending;
176 }
177 else if (mIndex >= mContainer.mTacticalEffects.Num())
178 {
179 // 기존 Effect를 모두 살폈다면, 추가 예약된 객체도 탐색
180 mPending = AdvancePending(const_cast<ElementType**>(&mContainer.mPendingTacticalEffectHead));
181 mCurrent = mPending;
182 mIndex = INDEX_NONE;
183 }
184 else
185 {
186 // 우선적으로 기존 Effect부터 살피기
187 mCurrent = &mContainer.mTacticalEffects[mIndex];
188 }
189
190 if (mCurrent != nullptr && mCurrent->mIsPendingRemove == true)
191 {
192 // 삭제될 객체는 반복에서 빠지기
193 Next();
194 }
195 }
196
197private:
198 ContainerType& mContainer;
199
200private:
201 // @brief 현재 Effect 배열 내 인덱스
202 int32 mIndex;
203 // @brief 예약 Effect의 노드
204 ElementType* mPending;
205 // @brief 현재 노드
206 ElementType* mCurrent;
207};
