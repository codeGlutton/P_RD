75template<typename ElementType, typename ContainerType>
76class FActiveTacticalEffectIterator
77{
78public:
84 FActiveTacticalEffectIterator(const ContainerType& Container, int32 StartIdx = 0) :
85 mContainer(const_cast<ContainerType&>(Container)),
86 mIndex(StartIdx),
87 mPending(nullptr),
88 mCurrent(nullptr)
89 {
90 // 순회 동안 컨테이너 변형을 막기 위해 락 카운트 증가
91 mContainer.IncrementLock();
92 if (mIndex >= 0)
93 {
94 UpdateCurrent();
95 }
96 }
97
98 ~FActiveTacticalEffectIterator()
99 {
100 // 락 카운트를 해제하여 보류된 증감을 반영
101 mContainer.DecrementLock();
102 }
103
104public:
105 void operator++()
106 {
107 Next();
108 }
109
110 ElementType& operator*() const
111 {
112 check(mCurrent != nullptr);
113 return *mCurrent;
114 }
115 ElementType* operator->() const
116 {
117 check(mCurrent != nullptr);
118 return mCurrent;
119 }
120
121 explicit operator bool() const
122 {
123 return (mCurrent != nullptr);
124 }
125
126 friend bool operator==(const FActiveTacticalEffectIterator& Lhs, const FActiveTacticalEffectIterator& Rhs)
127 {
128 return Lhs.mCurrent == Rhs.mCurrent;
129 }
130 friend bool operator!=(const FActiveTacticalEffectIterator& Lhs, const FActiveTacticalEffectIterator& Rhs)
131 {
132 return Lhs.mCurrent != Rhs.mCurrent;
133 }
134
135private:
141 ElementType* AdvancePending(ElementType** Next)
142 {
143 // tail은 "마지막 다음" 슬롯을 가리키므로, 거기에 닿으면 순회 종료(nullptr)
144 return (Next != const_cast<ElementType**>(mContainer.mPendingTacticalEffectTail)) ? *Next : nullptr;
145 }
146
151 void Next()
152 {
153 if (mIndex >= 0)
154 {
155 // 일반 이펙트는 해당 경로로 Iter 증가
156 ++mIndex;
157 }
158 else if (mPending != nullptr)
159 {
160 // 증감 이펙트는 해당 경로로 Iter 증가
161 mPending = AdvancePending(const_cast<ElementType**>(&mPending->mPendingNext));
162 }
163 UpdateCurrent();
164 }
165
172 void UpdateCurrent()
173 {
174 if (mIndex < 0)
175 {
176 // 이미 추가 예약된 객체 순회로 전환됨. Linked-List에 따라서 Pending을 현재 탐색 노드로 지정
177 mCurrent = mPending;
178 }
179 else if (mIndex >= mContainer.mTacticalEffects.Num())
180 {
181 // 기존 Effect를 모두 살폈다면, 추가 예약된 객체도 탐색
182 mPending = AdvancePending(const_cast<ElementType**>(&mContainer.mPendingTacticalEffectHead));
183 mCurrent = mPending;
184 mIndex = INDEX_NONE;
185 }
186 else
187 {
188 // 우선적으로 기존 Effect부터 살피기
189 mCurrent = &mContainer.mTacticalEffects[mIndex];
190 }
191
192 if (mCurrent != nullptr && mCurrent->mIsPendingRemove == true)
193 {
194 // 삭제될 객체는 반복에서 빠지기
195 Next();
196 }
197 }
198
199private:
200 ContainerType& mContainer;
201
202private:
203 // @brief 현재 Effect 배열 내 인덱스
204 int32 mIndex;
205 // @brief 예약 Effect의 노드
206 ElementType* mPending;
207 // @brief 현재 노드
208 ElementType* mCurrent;
209};
