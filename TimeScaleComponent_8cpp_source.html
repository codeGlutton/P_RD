67void UTimeScaleComponent::UpdateTimeScale(float DeltaTime)
68{
69 mCurrentTimeScale = 1.f;
70
71 for (auto& Pair : mActivateRequests)
72 {
73 Pair.Value.CurrentTimeScale = FMath::FInterpConstantTo(Pair.Value.CurrentTimeScale, Pair.Value.TargetTimeScale, DeltaTime, Pair.Value.BlendSpeed);
74 mCurrentTimeScale *= Pair.Value.CurrentTimeScale;
75 }
76
77 mCurrentTimeScale = FMath::Max(mCurrentTimeScale, mMinTimeScale);
78
79 UGameplayStatics::SetGlobalTimeDilation(GetWorld(), mCurrentTimeScale);
80}

82void UTimeScaleComponent::UpdateActivateRequests(float DeltaTime)
83{
84 // 제거 목록
85 TArray<int32> KeysToRemove;
86
87 for (auto& Pair : mActivateRequests)
88 {
89 FTimeScaleRequest& Req = Pair.Value;
90
91 if (!Req.bReleasing)
92 {
93 if (Req.Duration == -1.f)
94 {
95 // 무기한 요청은 Requester 무효화 시에만 종료 트리거
96 if (!Req.Requester.IsValid())
97 {
98 Req.bReleasing = true;
99 Req.TargetTimeScale = 1.f;
100 }
101 }
102 else
103 {
104 Req.RemaingTime -= DeltaTime;
105 if (Req.RemaingTime <= 0.f)
106 {
107 Req.bReleasing = true;
108 Req.TargetTimeScale = 1.f;
109 }
110 }
111 }
112 else if (FMath::IsNearlyEqual(Req.CurrentTimeScale, 1.f, KINDA_SMALL_NUMBER))
113 {
114 KeysToRemove.Add(Pair.Key);
115 }
116 }
117
118 // 실제 제거
119 RemoveExpiredActivateRequests(KeysToRemove);
120}

138bool UTimeScaleComponent::SetTimeScaleImmediately(const FTimeScaleHandle& Handle, float Scale)
139{
140 auto* Request = mActivateRequests.Find(Handle.ID);
141 if (!Handle.IsValid() || !Request || !FMath::IsFinite(Scale) || Scale <= 0.f) return false;
142 Request->CurrentTimeScale = Request->TargetTimeScale = Scale;
143 Request->bReleasing = false;
144 UpdateTimeScale(0.f);
145 return true;
146}
