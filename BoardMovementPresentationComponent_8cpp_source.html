66void UBoardMovementPresentationComponent::OnStartMovePath(const TArray<FVector>& PathWorldLocations, EBoardMoveMode MoveMode)
67{
68 // 이동 모드 기억 (속도 보고 여부 판단용)
69 mMoveMode = MoveMode;
70
71 // 밀치기/당기기는 코너링 없이 직선 이동 (폴리라인을 만들지 않으면 직선 이동모드로 동작)
72 // 회전도 목표 회전으로 보간하는데, 강제 이동은 모델이 방향을 유지하므로 자연스럽게 무회전

81
82 // 진행 상태 초기화
83 mCurrentMoveStep = 0;
84 mPolyLineTraveledDistance = 0.0f;

87void UBoardMovementPresentationComponent::OnStartMoveStep(const FTileTransform& NextTileTransform, const FTransform& TargetWorldTransform, TSharedPtr<FPresentationBarrier> Barrier, float RemainingPathDistance, EBoardMoveMode MoveMode)
88{
89 // 목표타일의 월드트랜스폼과 배리어 보관
90 mMoveTargetTransform = TargetWorldTransform;
91 mMoveBarrier = Barrier;
92 // 목표 이후 남은 거리 저장 (제동거리 계산에 사용)
93 mRemainingPathDistance = RemainingPathDistance;
94 // @note 현재 속도(mCurrentMoveSpeed)는 초기화하지 않음 (중간에 멈칫하지 않게)
95
96 // 타일 월드트랜스폼의 Z축은 타일 바닥이 기준이므로 액터의 중심이 타일 바닥에 파묻힘
97 // 바닥 오프셋만큼 더해서 바닥을 딛고 있도록 변경
98 FVector TargetLocation = mMoveTargetTransform.GetLocation();
99 TargetLocation.Z += mGroundZOffset;
100 mMoveTargetTransform.SetLocation(TargetLocation);
101
102 // 루트가 Movable이 아니면 SetActorLocation이 실패하므로 보정 (BP 설정 실수 방어)
103 AActor* Owner = GetOwner();
104 if (Owner != nullptr && Owner->GetRootComponent() != nullptr
105 && Owner->GetRootComponent()->Mobility != EComponentMobility::Movable)
106 {
107 Owner->GetRootComponent()->SetMobility(EComponentMobility::Movable);
108 }
109
110 // 폴리라인 이동모드면 경로 순번 진행
111 if (mPolyLinePoints.Num() >= 2)
112 {
113 // 현재 스텝(타일) 증가 -> 다음 타일 관련된 작업 진행
114 // @note 타일 하나에 여러 개의 점들이 포함될 수 있으므로 기준 타일을 설정하고 그 안에서 작업 진행
115 ++mCurrentMoveStep;
116 // 경로 통지와 스텝 수가 어긋나면 폴리라인 이동을 포기하고 직선 이동모드로 폴백
117 if (mStepMarkerDistances.IsValidIndex(mCurrentMoveStep) == false)
118 {
119 ResetPolyLineState();
120 }
121 }
122
123 // 틱 시작
124 SetComponentTickEnabled(true);

241void UBoardMovementPresentationComponent::TickStraight(float DeltaSeconds)
242{
243 AActor* Owner = GetOwner();
244 if (Owner == nullptr)
245 {
246 return;
247 }
248
249 // 목표 속도 결정
250 // - 기본은 최대속도
251 // - 제동거리에 들어가면 감속하면서 느려짐
252 const FVector CurrentLocation = Owner->GetActorLocation();
253 const float RemainingDistance = FVector::Dist(CurrentLocation, mMoveTargetTransform.GetLocation());
254
255 // 남은 거리에서 멈출 수 있는 속도 계산
256 float TargetSpeed = mMaxMoveSpeed;
257 if (mDeceleration > 0.0f)
258 {
259 // 제동 기준 거리 = 이번 스텝 남은 거리 + 이후 경로 거리 (최종 목적지 기준으로 감속)
260 // 목표가 많이 남으면 아래 BrakeSpeed가 최고속도보다 커지므로 최고속도로 순항하고
261 // 목표가 가까워져서 제동거리 안에 들어가면 BrakeSpeed가 최고속도보다 낮아지므로 BrakeSpeed로 느려짐
262 const float BrakeDistance = RemainingDistance + mRemainingPathDistance;
263 // @details
264 // 등가속도 공식은 `v^2 = v0^2 + 2as` 인데,
265 // 최종속도 v=0 이 되면서 멈추는 거니까 이 공식은 `0 = v0^2 + 2as`
266 // 이때 감속하니까 가속도는 -가 되고 2as를 좌변으로 넘기면 `2as = v0^2`
267 // 따라서 `v0 = sqrt(2as)`
268 const float BrakeSpeed = FMath::Sqrt(2.0f * mDeceleration * BrakeDistance);
269 TargetSpeed = FMath::Min(TargetSpeed, BrakeSpeed);
270 }
271
272 // 현재 속도를 목표 속도로 가감속 (올릴 땐 가속도, 내릴 땐 감속도 사용)
273 // 해당 가감속도가 0이면 즉시 목표 속도 도달
274 const float InterpRate = (TargetSpeed < mCurrentMoveSpeed)
275 ? mDeceleration
276 : mAcceleration;
277 mCurrentMoveSpeed = (InterpRate > 0.0f)
278 ? FMath::FInterpConstantTo(mCurrentMoveSpeed, TargetSpeed, DeltaSeconds, InterpRate)
279 : TargetSpeed;
280
281 // 위치: 현재벡터를, 현재위치에서 다음위치로 현재 속도만큼 이동한 값 계산
282 const FVector NewLocation = FMath::VInterpConstantTo(
283 CurrentLocation,
284 mMoveTargetTransform.GetLocation(),
285 DeltaSeconds, mCurrentMoveSpeed);
286 // 회전: 현재회전을, 다음 타일을 바라보도록 지정한 각도만큼 변경한 값 계산
287 const FRotator NewRotation = FMath::RInterpConstantTo(
288 Owner->GetActorRotation(),
289 mMoveTargetTransform.Rotator(),
290 DeltaSeconds, mRotationSpeed);
291 Owner->SetActorLocationAndRotation(NewLocation, NewRotation);
292
293 // 현재 속도 저장
294 // 애니메이션이 GetVelocity()로 읽어서 활용 가능
295 mCurrentMoveVelocity
296 = (DeltaSeconds > 0.0f) // 혹시 0으로 나누는 경우를 방지
297 ? (NewLocation - CurrentLocation) / DeltaSeconds
298 : FVector::ZeroVector;
299
300 // 위치와 회전이 모두 목표와 일치하면 -> 해당 타일에 도착
301 if (NewLocation.Equals(mMoveTargetTransform.GetLocation()) &&
302 NewRotation.Equals(mMoveTargetTransform.Rotator()))
303 {
304 // 최종 목적지 도착이면 정지 상태로 초기화 (다음 이동은 0부터 다시 가속)
305 if (mRemainingPathDistance <= KINDA_SMALL_NUMBER)
306 {
307 mCurrentMoveSpeed = 0.0f;
308 mCurrentMoveVelocity = FVector::ZeroVector;
309 }
310 // 도착했으면 틱 해제
311 SetComponentTickEnabled(false);
312 // 베리어 해제
313 // -> 이동 컴포넌트 모델의 OnStepPresentationFinished() 호출
314 // -> StartStep(NextTile) 해서 타일간 이동 반복
315 mMoveBarrier.Reset();
316 }

319void UBoardMovementPresentationComponent::TickPolyLine(float DeltaSeconds)
320{
321 AActor* Owner = GetOwner();
322 if (Owner == nullptr)
323 {
324 return;
325 }
326
327 // 전체 길이와 남은 거리 (모든 거리는 폴리라인 시작점 기준)
328 const float TotalDistance = mPolyLineDistances.Last();
329 const float RemainingDistance = TotalDistance - mPolyLineTraveledDistance;
330
331 // 남은 거리에서 멈출 수 있는 속도 계산 (직선이동모드와 동일한 등가속도 공식)
332 float TargetSpeed = mMaxMoveSpeed;
333 if (mDeceleration > 0.0f)
334 {
335 const float BrakeSpeed = FMath::Sqrt(2.0f * mDeceleration * RemainingDistance);
336 TargetSpeed = FMath::Min(TargetSpeed, BrakeSpeed);
337 }
338
339 // 현재 속도를 목표 속도로 가감속 (올릴 땐 가속도, 내릴 땐 감속도 사용)
340 // 해당 가감속도가 0이면 즉시 목표 속도 도달
341 const float InterpRate = (TargetSpeed < mCurrentMoveSpeed)
342 ? mDeceleration
343 : mAcceleration;
344 mCurrentMoveSpeed = (InterpRate > 0.0f)
345 ? FMath::FInterpConstantTo(mCurrentMoveSpeed, TargetSpeed, DeltaSeconds, InterpRate)
346 : TargetSpeed;
347
356 mPolyLineTraveledDistance = FMath::Min(
357 mPolyLineTraveledDistance + mCurrentMoveSpeed * DeltaSeconds,
358 TotalDistance);
359
360 // 진행거리에 해당하는 폴리라인 위치/접선 계산
361 FVector SampleLocation = FVector::ZeroVector;
362 FVector SampleTangent = FVector::ZeroVector;
363 if (GetPolyLinePoint(mPolyLineTraveledDistance, SampleLocation, SampleTangent) == false)
364 {
365 // 폴리라인이 비정상이면 직선이동모드로 폴백
366 ResetPolyLineState();
367 return;
368 }
369
370 // 폴리라인은 타일 바닥 기준이므로 바닥 오프셋만큼 보정
371 SampleLocation.Z += mGroundZOffset;
372
373 // 이 타일에서의 기본 회전값
374 FRotator TargetRotation = mMoveTargetTransform.Rotator();
375 FVector FlatTangent = SampleTangent;
376 // 접선벡터에서 요 회전만 바꾸기 위해서 Z축은 0으로 설정
377 FlatTangent.Z = 0.0f;
378 if (RemainingDistance > KINDA_SMALL_NUMBER && FlatTangent.IsNearlyZero() == false)
379 {
380 // 접선 벡터를 회전값으로 변환 -> 이 회전값만큼 액터가 회전한다
381 TargetRotation = FlatTangent.Rotation();
382 }
383
384 // 위치와 회전 적용
385 const FVector PreviousLocation = Owner->GetActorLocation();
386 const FRotator NewRotation = FMath::RInterpConstantTo(Owner->GetActorRotation(), TargetRotation, DeltaSeconds, mRotationSpeed);
387 Owner->SetActorLocationAndRotation(SampleLocation, NewRotation);
388
389 // 현재 속도 저장
390 // 애니메이션이 GetVelocity()로 읽어서 활용 가능
391 mCurrentMoveVelocity = (DeltaSeconds > 0.0f)
392 ? (SampleLocation - PreviousLocation) / DeltaSeconds
393 : FVector::ZeroVector;
394
395 // 해당 타일을 지났는 지 판정
396 // @note
397 // FPS가 느리면 한 프레임에 여러 마커를 지날 수 있으므로 반복 처리
398 while (mMoveBarrier.IsValid()
399 && mCurrentMoveStep < mStepMarkerDistances.Num() - 1
400 && mPolyLineTraveledDistance >= mStepMarkerDistances[mCurrentMoveStep])
401 {
402 mMoveBarrier.Reset();
403 }
404
405 // (이유는 몰라도) 폴리라인이 리셋됐으면 바로 종료
406 if (mPolyLinePoints.Num() < 2)
407 {
408 return;
409 }
410
411 // 최종 지점 도착 + 회전 완료면 이동 종료
412 if (mMoveBarrier.IsValid() &&
413 mCurrentMoveStep >= mStepMarkerDistances.Num() - 1 &&
414 mPolyLineTraveledDistance >= TotalDistance &&
415 NewRotation.Equals(mMoveTargetTransform.Rotator()))
416 {
417 // 정지 상태로 초기화 (다음 이동은 0부터 다시 가속)
418 mCurrentMoveSpeed = 0.0f;
419 mCurrentMoveVelocity = FVector::ZeroVector;
420 ResetPolyLineState();
421 // 도착했으면 틱 해제
422 SetComponentTickEnabled(false);
423 // 베리어 해제
424 // -> 이동 컴포넌트 모델의 OnStepPresentationFinished() 호출
425 // -> 마지막 스텝이므로 이동 완료 처리
426 mMoveBarrier.Reset();
427 }

430bool UBoardMovementPresentationComponent::GetPolyLinePoint(float Distance, FVector& OutLocation, FVector& OutTangent) const
431{
432 // 폴리라인이 없거나 배열 크기가 어긋나면 실패
433 if (mPolyLinePoints.Num() < 2 || mPolyLinePoints.Num() != mPolyLineDistances.Num())
434 {
435 return false;
436 }
437
438 // 진행거리가 속한 선분 검색
439 // @details
440 // 누적거리 오름차순이므로 이진 탐색 사용
441 int32 SegmentEndIndex = Algo::UpperBound(mPolyLineDistances, Distance);
442 SegmentEndIndex = FMath::Clamp(SegmentEndIndex, 1, mPolyLinePoints.Num() - 1);
443 const int32 SegmentStartIndex = SegmentEndIndex - 1;
444
445 // 선분 시작점과 끝점 사이에서 진행거리가 차지하는 비율 (접선공식)
446 const float SegmentLength = mPolyLineDistances[SegmentEndIndex] - mPolyLineDistances[SegmentStartIndex];
447 const float Alpha = (SegmentLength > KINDA_SMALL_NUMBER)
448 ? FMath::Clamp((Distance - mPolyLineDistances[SegmentStartIndex]) / SegmentLength, 0.0f, 1.0f)
449 : 1.0f;
450
451 // 위치: 선분 위 보간
452 // 접선: 선분의 단위 방향벡터
453 OutLocation = FMath::Lerp(mPolyLinePoints[SegmentStartIndex], mPolyLinePoints[SegmentEndIndex], Alpha);
454 OutTangent = (SegmentLength > KINDA_SMALL_NUMBER)
455 ? (mPolyLinePoints[SegmentEndIndex] - mPolyLinePoints[SegmentStartIndex]) / SegmentLength
456 : FVector::ZeroVector;
457 return true;
458}

470void UBoardMovementPresentationComponent::BakePolyLinePoints(
471 const TArray<FVector>& PathPoints,
472 float CornerCutRatio,
473 float CornerTension,
474 TArray<FVector>& OutPoints,
475 TArray<float>& OutDistances,
476 TArray<float>& OutStepMarkerDistances)
477{
478 OutPoints.Reset();
479 OutDistances.Reset();
480 OutStepMarkerDistances.Reset();
481
482 // 경로가 2칸 미만이면 폴리라인 불성립
483 if (PathPoints.Num() < 2)
484 {
485 return;
486 }
487
496 constexpr int32 CornerSampleCount = 8;
497 const float CutRatio = FMath::Clamp(CornerCutRatio, 0.0f, 0.5f);
498 const float Tension = FMath::Clamp(CornerTension, 0.0f, 2.0f);
499
500 // 점을 추가하고 그 점까지의 누적거리를 반환하는 헬퍼
501 auto AppendPoint = [&OutPoints, &OutDistances](const FVector& Point) -> float
502 {
503 // 이전 점과의 거리가 사실상 0이면
504 // -> 추가할 의미가 없고
505 // -> 0으로 나누기 하는 위험도 있으므로
506 // 추가하지 않고 넘어감
507 if (OutPoints.Num() > 0 && FVector::DistSquared(OutPoints.Last(), Point) < KINDA_SMALL_NUMBER)
508 {
509 return OutDistances.Last();
510 }
511
512 const float Distance = (OutPoints.Num() > 0)
513 ? OutDistances.Last() + FVector::Dist(OutPoints.Last(), Point)
514 : 0.0f;
515 OutPoints.Add(Point);
516 OutDistances.Add(Distance);
517 return Distance;
518 };
519
520 // 시작점 (시작 타일의 도착 판정 거리는 0)
521 OutStepMarkerDistances.Add(AppendPoint(PathPoints[0]));
522
523 // 경로의 중간 지점들(시작/도착 제외) 처리
524 // 직진이면 중점 그대로,
525 // 꺾이는 곳이면, 코너 컷(곡선의 시작과 끝) 후 베지어곡선 점들 추가
526 for (int32 i = 1; i < PathPoints.Num() - 1; ++i)
527 {
528 const FVector& PrevPoint = PathPoints[i - 1]; // 직전 타일 중점
529 const FVector& CornerPoint = PathPoints[i]; // 이번 타일 중점
530 const FVector& NextPoint = PathPoints[i + 1]; // 다음 타일 중점
531
532 // 진입 벡터(직전->이번), 진출 벡터(이번->다음)
533 const FVector InDelta = CornerPoint - PrevPoint;
534 const FVector OutDelta = NextPoint - CornerPoint;
535
536 // 코너 컷 거리 = 구간 길이 x 컷 비율
537 const float CutInDistance = InDelta.Size() * CutRatio;
538 const float CutOutDistance = OutDelta.Size() * CutRatio;
539
540 // 1) 직진: 진입/진출 방향이 같으면 커브 없이 중점 그대로 추가
541 const bool IsStraight =
542 // 평행하고
543 FVector::Parallel(InDelta.GetSafeNormal(), OutDelta.GetSafeNormal()) &&
544 // 두 벡터의 사이각이 90도 이내(내적이 0보다 큰)면 -> 방향이 같음
545 FVector::DotProduct(InDelta, OutDelta) > 0.0f;
546 if (IsStraight || CutInDistance < KINDA_SMALL_NUMBER || CutOutDistance < KINDA_SMALL_NUMBER)
547 {
548 // 중점의 누적거리가 곧 이 타일의 도착 판정 거리
549 OutStepMarkerDistances.Add(AppendPoint(CornerPoint));
550 continue;
551 }
552
553 // 2) 코너: 커브 시작점/끝점을 구함
554 // 커브 시작점은 코너 중점에서 코너 컷 비율만큼 먼저 시작
555 const FVector CurveStart = CornerPoint - InDelta.GetSafeNormal() * CutInDistance;
556 // 커브 끝점은 코너 중점에서 코너 컷 비율만큼 나중에 끝남
557 const FVector CurveEnd = CornerPoint + OutDelta.GetSafeNormal() * CutOutDistance;
558
559 // 베지어곡선의 컨트롤 포인트: 텐션에 따라 시작점과 끝점을 잇는 현의 중점과 코너 중점 사이에 위치
560 // 텐션 0 = 현의 중점 = 대각선 직진
561 // 텐션 1 = 코너 중점 = 표준 베지어
562 // 텐션 2 = 코너 중점 반대편 = 곡선이 코너 중점을 통과 (각진 코너링)
563 const FVector ChordMidPoint = (CurveStart + CurveEnd) * 0.5f;
564 const FVector ControlPoint = FMath::Lerp(ChordMidPoint, CornerPoint, Tension);
565
566 // 시작점과 끝점 사이를 베지어곡선을 n분할해서 만듦
567 float MarkerDistance = 0.0f;
568 AppendPoint(CurveStart);
569 for (int32 Sample = 1; Sample <= CornerSampleCount; ++Sample)
570 {
571 // T는 1/8, 2/8, ... 8/8 식으로 변화
572 const float T = static_cast<float>(Sample) / CornerSampleCount;
573 /*
574 * @brief 베지어곡선상의 점 Point는 다음 두 지점을 잇는 선분의 T/8 지점이다.
575 * @details
576 * 기준이 되는 선분은 두 개가 고정이다.
577 * 하나는 CurveStart에서 ControlPoint를 향하는 선분(진입선분)
578 * 또 하나는 ControlPoint에서 CurveEnd를 향하는 선분(진출선분)
579 *
580 * 앞의 선분을 L1, 뒤의 선분을 L2라고 하면,
581 * 첫번째 베지어곡선상의 점 S1은, L1의 1/8 지점과 L2의 1/8 지점을 잇는 선분의 1/8 지점
582 * 두번째 베지어곡선상의 점 S2는, L1의 2/8 지점과 L2의 2/8 지점을 잇는 선분의 2/8 지점
583 * ...
584 * 마지막 베지어곡선상의 점 S8은, L1의 8/8 지점(ControlPoint)과 L2의 8/8 지점(진출점) 을 잇는 선분의 8/8 지점(진출점)
585 * 이런식으로 분할 개수만큼 반복한다.
586 */
587 const FVector Point = FMath::Lerp(
588 FMath::Lerp(CurveStart, ControlPoint, T),
589 FMath::Lerp(ControlPoint, CurveEnd, T),
590 T);
591 const float Distance = AppendPoint(Point);
592
593 // 곡선의 중간 점을 이 타일의 도착 판정 거리로 기록
594 // @note 그래서 분할 개수는 짝수로 하는 게 좋음
595 if (Sample == CornerSampleCount / 2)
596 {
597 MarkerDistance = Distance;
598 }
599 }
600 OutStepMarkerDistances.Add(MarkerDistance);
601 }
602
603 // 도착점 설정
604 // 마지막 타일의 도착 판정 거리이며 폴리라인 전체 길이와 동일함
605 OutStepMarkerDistances.Add(AppendPoint(PathPoints.Last()));
606}
