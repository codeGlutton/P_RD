76bool URDUIInputSubsystem::RouteBack()
77{
78 UGameViewportSubsystem* Viewport = UGameViewportSubsystem::Get();
79 if (!Viewport) return false;
80 int32 TopIndex = INDEX_NONE, TopZ = MIN_int32;
81 for (int32 Index = 0; Index < Entries.Num(); ++Index)
82 {
83 const FEntry& Entry = Entries[Index];
84 if (!Entry.Owner.IsValid()) continue;
85 UUserWidget* Layer = Entry.Layer();
86 if (!Layer || !Layer->IsInViewport() || !Layer->IsVisible() || !Layer->GetIsEnabled()) continue;
87 const int32 Z = Viewport->GetWidgetSlot(Layer).ZOrder;
88 if (Z >= TopZ) { TopIndex = Index; TopZ = Z; }
89 }
90 if (TopIndex == INDEX_NONE) return false;
91 TArray<UUserWidget*> TopLevelWidgets;
92 UWidgetBlueprintLibrary::GetAllWidgetsOfClass(GetGameInstance(), TopLevelWidgets, UUserWidget::StaticClass(), true);
93 for (UUserWidget* Widget : TopLevelWidgets)
94 if (Widget && Widget->IsVisible() && Viewport->GetWidgetSlot(Widget).ZOrder > TopZ)
95 return true; // A cinematic/loading layer owns the screen above our navigation owners.
96 // The handler can remove/re-register widgets. Copy it before invoking it.
97 const TFunction<bool()> Handler = Entries[TopIndex].Handler;
98 Handler();
99 // Never let an unclosable reward/transition fall through to the screen underneath.
100 return true;
101}
