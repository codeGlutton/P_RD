60bool FSimulationPreviewIsolationTest::RunTest(const FString& Parameters)
61{
62 TStrongObjectPtr<UCombatUIModel> UIModel(NewObject<UCombatUIModel>());
63 TStrongObjectPtr<USimulationPreviewUIModelTestListener> Listener(NewObject<USimulationPreviewUIModelTestListener>());
64
65 USimulationPreviewUIModel* PreviewModel = UIModel->GetSimulationPreviewUIModel();
66 TestNotNull(TEXT("미리보기 모델이 지연 생성된다"), PreviewModel);
67 TestTrue(TEXT("미리보기 모델은 실전 모델이 소유한다"), PreviewModel->GetOuter() == UIModel.Get());
68
69 UIModel->OnCombatEventBatchChanged.AddDynamic(Listener.Get(), &USimulationPreviewUIModelTestListener::HandleLiveEventBatch);
70 PreviewModel->OnPreviewEventBatch.AddDynamic(Listener.Get(), &USimulationPreviewUIModelTestListener::HandlePreviewEventBatch);
71 PreviewModel->OnPreviewCleared.AddDynamic(Listener.Get(), &USimulationPreviewUIModelTestListener::HandlePreviewCleared);
72
73 // 실전 로그(-40)를 먼저 내린다.
74 UIModel->SetCombatEventBatch(ECombatEventDataSourceUI::LiveCombat,
75 { MakeSimulationPreviewTestRequest(TEXT("-40"), false) });
76 TestEqual(TEXT("실전 배치 revision"), UIModel->GetCombatEventBatch().mRevision, 1);
77
78 // 미리보기(-10)를 얹는다 — 실전 배치는 어떤 것도 변하면 안 된다.
79 const int32 Generation = PreviewModel->BeginPreview();
80 PreviewModel->SetPreviewEventBatch(Generation,
81 { MakeSimulationPreviewTestRequest(TEXT("-10"), true) });
82
83 TestEqual(TEXT("미리보기 배치 출처"), Listener->mLastPreviewBatch.mSource, ECombatEventDataSourceUI::SimulationPreview);
84 TestEqual(TEXT("미리보기 배치 로그 수"), Listener->mLastPreviewBatch.mFloatingLogs.Num(), 1);
85 TestEqual(TEXT("미리보기 set 후에도 실전 revision 불변"), UIModel->GetCombatEventBatch().mRevision, 1);
86 TestEqual(TEXT("미리보기 set 후에도 실전 로그 수 불변"), UIModel->GetCombatEventBatch().mFloatingLogs.Num(), 1);
87 TestEqual(TEXT("미리보기 set 후에도 실전 로그 내용 불변"),
88 UIModel->GetCombatEventBatch().mFloatingLogs[0].mText.ToString(), FString(TEXT("-40")));
89 TestEqual(TEXT("미리보기 set이 실전 배치 알림을 내지 않는다"), Listener->mLiveBatchCallCount, 1);
90
91 // 미리보기를 통째로 버려도 실전은 그대로다.
92 PreviewModel->ClearPreview();
93 TestEqual(TEXT("Clear 후에도 실전 revision 불변"), UIModel->GetCombatEventBatch().mRevision, 1);
94 TestEqual(TEXT("Clear 후에도 실전 로그 수 불변"), UIModel->GetCombatEventBatch().mFloatingLogs.Num(), 1);
95 TestEqual(TEXT("Clear 후에도 실전 배치 알림 수 불변"), Listener->mLiveBatchCallCount, 1);
96 TestEqual(TEXT("Clear 후 미리보기 payload는 빈다"), PreviewModel->GetPreviewEventBatch().mFloatingLogs.Num(), 0);
97 TestFalse(TEXT("Clear 후 미리보기는 닫힌다"), PreviewModel->IsPreviewActive());
98
99 return true;
100}

108bool FSimulationPreviewGenerationTest::RunTest(const FString& Parameters)

110 TStrongObjectPtr<USimulationPreviewUIModel> PreviewModel(NewObject<USimulationPreviewUIModel>());
111 TStrongObjectPtr<USimulationPreviewUIModelTestListener> Listener(NewObject<USimulationPreviewUIModelTestListener>());
112
113 PreviewModel->OnPreviewEventBatch.AddDynamic(Listener.Get(), &USimulationPreviewUIModelTestListener::HandlePreviewEventBatch);
114
115 // 조준을 연달아 바꾸면 세대가 겹친다 — 낡은 세대의 결과는 버려져야 한다.
116 const int32 StaleGeneration = PreviewModel->BeginPreview();
117 const int32 CurrentGeneration = PreviewModel->BeginPreview();
118 TestTrue(TEXT("BeginPreview마다 세대가 오른다"), CurrentGeneration > StaleGeneration);
119
120 PreviewModel->SetPreviewEventBatch(StaleGeneration,
121 { MakeSimulationPreviewTestRequest(TEXT("-99"), true) });
122 TestEqual(TEXT("낡은 세대 배치는 알림 없이 버려진다"), Listener->mPreviewBatchCallCount, 0);
123 TestEqual(TEXT("낡은 세대 배치는 payload에 남지 않는다"), PreviewModel->GetPreviewEventBatch().mFloatingLogs.Num(), 0);
124
125 PreviewModel->SetPreviewEventBatch(CurrentGeneration,
126 { MakeSimulationPreviewTestRequest(TEXT("-10"), true) });
127 TestEqual(TEXT("현재 세대 배치는 반영된다"), Listener->mPreviewBatchCallCount, 1);
128 TestEqual(TEXT("현재 세대 배치 로그 수"), PreviewModel->GetPreviewEventBatch().mFloatingLogs.Num(), 1);
129 TestEqual(TEXT("현재 세대 배치 내용"),
130 PreviewModel->GetPreviewEventBatch().mFloatingLogs[0].mText.ToString(), FString(TEXT("-10")));
131
132 // Clear도 세대를 올린다 — 비행 중이던 결과가 빈 자리를 다시 채우면 안 된다.
133 PreviewModel->ClearPreview();
134 PreviewModel->SetPreviewEventBatch(CurrentGeneration,
135 { MakeSimulationPreviewTestRequest(TEXT("-77"), true) });
136 TestEqual(TEXT("Clear 뒤 낡은 세대 배치는 버려진다"), Listener->mPreviewBatchCallCount, 1);
137 TestEqual(TEXT("Clear 뒤 payload는 빈 채로 남는다"), PreviewModel->GetPreviewEventBatch().mFloatingLogs.Num(), 0);
138
139 return true;
140}
