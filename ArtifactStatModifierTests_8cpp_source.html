44 }

56 FTestObjectScope Scope;

58 // 라이브 월드 확보 (TAS 속성 테스트와 동일 환경)
59 UWorld* World = FindAnyGameWorld();

68

77 {
78 return false;
79 }
80
81 // 전략 프레임워크 모델 등록 (없을 때만 Mock 등록 — 이펙트 컨텍스트/적용에 필요)
82 UTacticalFrameworkModel* FrameworkModel = Cast<UTacticalFrameworkModel>(RoomInstance->mAliveSubsystemModels.FindRef(UTacticalFrameworkModel::StaticClass()));

84 {

87 // 테스트가 등록한 것이므로 파괴 전에 등록 해제
88 Scope.Defer([RoomInstance]() { RoomInstance->mAliveSubsystemModels.Remove(UTacticalFrameworkModel::StaticClass()); });
89 }

91 // 속성 컴포넌트 Mock

94 MockActorModel->BeginPlay();

96 if (TestNotNull(TEXT("AttributeSetComponentModel"), AttrComp) == false)
97 {
98 return false;
99 }
100
101 // 공격력 기본값 10 세팅 (현재값도 10으로 전파됨)
102 const FTacticalAttribute AttackFactor = UCombatTargetAttributeSet::GetAttackFactorAttribute();
103 AttrComp->SetAttributeBaseValue(AttackFactor, 10.f);
104 TestEqual(TEXT("기본 공격력 10"), AttrComp->GetAttributeCurrentValue(AttackFactor), 10.f);
105
106 // 아티펙트 DA (코드 구성): 공격력 +5 고유 스탯
107 UStaticArtifactData* Artifact = Scope.New<UStaticArtifactData>();
108 FTacticalModifierInfo Mod;
109 Mod.mAttribute = AttackFactor;
110 Mod.mModifierOp = ETacticalModOp::AddBase;
111 Mod.mModifierMagnitude = 5.f;
112 Artifact->mStatModifiers.Add(Mod);

114 // 장착: 스탯 이펙트가 자신에게 적용 (패시브 컴포넌트는 불필요 → nullptr)
115 UArtifactComponentModel* ArtifactComp = Scope.New<UArtifactComponentModel>();

117 TestTrue(TEXT("장착 성공"), bEquipped);
118 TestEqual(TEXT("장착 후 공격력 15 (기본 10 + 스탯 5)"), AttrComp->GetAttributeCurrentValue(AttackFactor), 15.f);
119
120 // 스탯 이펙트 핸들이 유효해야 함
121 const FArtifactEntry* Entry = ArtifactComp->GetEquipped(Artifact);
122 if (TestNotNull(TEXT("장착 엔트리 조회"), Entry))
123 {
124 TestTrue(TEXT("스탯 이펙트 핸들 유효"), Entry->mStatEffectHandle.IsValid());
125 }
126
127 // 중복 장착: 같은 아티펙트 하나 더 → 스탯도 중첩 (10 + 5 + 5)
128 TestTrue(TEXT("중복 장착 성공"), ArtifactComp->EquipInternal(Artifact, nullptr, AttrComp));
129 TestEqual(TEXT("중복 장착 후 공격력 20"), AttrComp->GetAttributeCurrentValue(AttackFactor), 20.f);
130
131 // 해제: 하나 내리면 15, 마저 내리면 10 원복
132 TestTrue(TEXT("해제 성공"), ArtifactComp->UnequipInternal(Artifact, nullptr, AttrComp));
133 TestEqual(TEXT("해제 후 공격력 15"), AttrComp->GetAttributeCurrentValue(AttackFactor), 15.f);
134 TestTrue(TEXT("남은 엔트리 해제 성공"), ArtifactComp->UnequipInternal(Artifact, nullptr, AttrComp));
135 TestEqual(TEXT("전부 해제 후 공격력 10 원복"), AttrComp->GetAttributeCurrentValue(AttackFactor), 10.f);
136
137 return true;
138}
