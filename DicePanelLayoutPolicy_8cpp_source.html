49void RDDicePanelLayout::BuildOverviewLayouts(int32 DiceCount, TArray<FDicePanelSlotLayout>& OutLayouts)
50{
51 OutLayouts.Reset();
52 if (DiceCount <= 0)
53 {
54 return;
55 }
56
57 const int32 ColumnCount = FMath::Min(3, DiceCount);
58 const int32 RowCount = FMath::CeilToInt(StaticCast<float>(DiceCount) / StaticCast<float>(ColumnCount));
59 const float CardWidth = 0.135f;
60 const float CardHeight = 0.235f;
61 const float GapX = 0.040f;
62 const float GapY = 0.040f;
63 const float TotalWidth = CardWidth * StaticCast<float>(ColumnCount) + GapX * StaticCast<float>(FMath::Max(0, ColumnCount - 1));
64 const float TotalHeight = CardHeight * StaticCast<float>(RowCount) + GapY * StaticCast<float>(FMath::Max(0, RowCount - 1));
65 const float StartLeft = 0.5f - TotalWidth * 0.5f;
66 const float StartTop = 0.5f - TotalHeight * 0.5f + 0.025f;
67
68 for (int32 DiceIndex = 0; DiceIndex < DiceCount; ++DiceIndex)
69 {
70 const int32 Column = DiceIndex % ColumnCount;
71 const int32 Row = DiceIndex / ColumnCount;
72 const float Left = StartLeft + (CardWidth + GapX) * StaticCast<float>(Column);
73 const float Top = StartTop + (CardHeight + GapY) * StaticCast<float>(Row);
74
75 FDicePanelSlotLayout Layout;
76 Layout.mCardAnchors = FAnchors(Left, Top, Left + CardWidth, Top + CardHeight);
77 Layout.mImageAnchors = Layout.mCardAnchors;
78 Layout.mCardZOrder = 126 + DiceIndex;
79 Layout.mImageZOrder = 127 + DiceIndex;
80 Layout.mShowCard = true;
81 OutLayouts.Add(Layout);
82 }
83}

86bool RDDicePanelLayout::BuildCarouselLayout(int32 DiceIndex, int32 SelectedDiceIndex, int32 DiceCount, FDicePanelSlotLayout& OutLayout)
87{
88 if (DiceIndex < 0 || DiceIndex >= DiceCount || SelectedDiceIndex < 0 || SelectedDiceIndex >= DiceCount)
89 {
90 return false;
91 }
92
93 if (DiceIndex == SelectedDiceIndex)
94 {
95 OutLayout.mCardAnchors = FAnchors();
96 OutLayout.mImageAnchors = FAnchors(0.391f, 0.365f, 0.609f, 0.715f);
97 OutLayout.mCardZOrder = 0;
98 OutLayout.mImageZOrder = 170;
99 OutLayout.mShowCard = false;
100 return true;
101 }
102
103 // 원형 carousel이라 끝과 처음이 인접하도록 상대 index를 가장 짧은 방향으로 접는다.
104 int32 RelativeIndex = DiceIndex - SelectedDiceIndex;
105 if (RelativeIndex > DiceCount / 2)
106 {
107 RelativeIndex -= DiceCount;
108 }
109 else if (RelativeIndex < -DiceCount / 2)
110 {
111 RelativeIndex += DiceCount;
112 }
113
114 const float Direction = RelativeIndex < 0 ? -1.0f : 1.0f;
115 const float Distance = StaticCast<float>(FMath::Max(1, FMath::Abs(RelativeIndex)));
116 const float CenterX = 0.5f + Direction * (0.265f + (Distance - 1.0f) * 0.120f);
117 const float CardWidth = 0.120f;
118 const float CardHeight = 0.235f;
119 const float Top = 0.475f + FMath::Min(Distance - 1.0f, 2.0f) * 0.030f;
120 const int32 ZOrder = 150 - FMath::RoundToInt(Distance * 4.0f);
121
122 OutLayout.mCardAnchors = FAnchors(CenterX - CardWidth * 0.5f, Top, CenterX + CardWidth * 0.5f, Top + CardHeight);
123 OutLayout.mImageAnchors = OutLayout.mCardAnchors;
124 OutLayout.mCardZOrder = ZOrder;
125 OutLayout.mImageZOrder = ZOrder + 1;
126 OutLayout.mShowCard = true;
127 return true;
128}
