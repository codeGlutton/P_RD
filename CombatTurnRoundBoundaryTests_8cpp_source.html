76{
77 constexpr int32 InitialSeed = 712367;
78 constexpr int32 RequiredSpeedPointForTurn = 10;
79
80 // 두 후보의 잔여 속도가 매 라운드 정확히 같아 정렬 순서를 난수만으로
81 // 결정하게 한다. 예측 결과는 같은 초기 seed에서 항상 같은 값을 내야 한다.
82 TArray<FSRPGTurnCandidate> TiedCandidates;
83 for (int32 Index = 0; Index < 2; ++Index)
84 {
85 FSRPGTurnCandidate& Candidate = TiedCandidates.AddDefaulted_GetRef();
86 Candidate.mRemainSpeedPoint = 0;
87 Candidate.mRechargedSpeedPoint = RequiredSpeedPointForTurn;
88 }
89
90 const TArray<FSRPGPredictedRound> FirstPrediction =
91 USRPGCombatModel::PredictTurnRounds(TiedCandidates,
92 RequiredSpeedPointForTurn, InitialSeed, /*RoundCount=*/2);
93 const TArray<FSRPGPredictedRound> RepeatedPrediction =
94 USRPGCombatModel::PredictTurnRounds(TiedCandidates,
95 RequiredSpeedPointForTurn, InitialSeed, /*RoundCount=*/2);
96
97 if (!TestEqual(TEXT("동률 예측 라운드 수"), FirstPrediction.Num(), 2)
98 || !TestEqual(TEXT("반복 동률 예측 라운드 수"),
99 RepeatedPrediction.Num(), 2))
100 {
101 return false;
102 }
103
104 FRandomStream FirstRoundStream(InitialSeed);
105 TArray<float> ExpectedFirstRound = {
106 FirstRoundStream.FRand(), FirstRoundStream.FRand()
107 };
108 const int32 SecondRoundSeed = FirstRoundStream.GetCurrentSeed();
109 FRandomStream SecondRoundStream(SecondRoundSeed);
110 TArray<float> ExpectedSecondRound = {
111 SecondRoundStream.FRand(), SecondRoundStream.FRand()
112 };
113 ExpectedFirstRound.Sort(TGreater<float>());
114 ExpectedSecondRound.Sort(TGreater<float>());
115
116 for (int32 RoundIndex = 0; RoundIndex < 2; ++RoundIndex)
117 {
118 const FSRPGPredictedRound& Round = FirstPrediction[RoundIndex];
119 if (!TestEqual(*FString::Printf(TEXT("동률 라운드 %d 후보 수"),
120 RoundIndex + 1), Round.mCandidates.Num(), 2))
121 {
122 return false;
123 }
124 const TArray<float>& Expected = RoundIndex == 0
125 ? ExpectedFirstRound : ExpectedSecondRound;
126 for (int32 CandidateIndex = 0; CandidateIndex < 2; ++CandidateIndex)
127 {
128 TestEqual(*FString::Printf(
129 TEXT("동률 라운드 %d 난수 %d"), RoundIndex + 1,
130 CandidateIndex),
131 Round.mCandidates[CandidateIndex].mRandomTieBreaker,
132 Expected[CandidateIndex]);
133 TestEqual(*FString::Printf(
134 TEXT("같은 seed 반복 결과 라운드 %d 후보 %d"), RoundIndex + 1,
135 CandidateIndex),
136 RepeatedPrediction[RoundIndex].mCandidates[CandidateIndex]
137 .mRandomTieBreaker,
138 Round.mCandidates[CandidateIndex].mRandomTieBreaker);
139 }
140 }
141
142 // 두 번째 라운드가 초기 seed를 다시 사용했다면 첫 라운드 난수열과 같아진다.
143 // 정확히 다음 seed를 사용한 기대값을 위에서 비교했으므로, 이 검사는 회귀
144 // 원인을 메시지에서 바로 드러내기 위한 보조 단언이다.
145 TestTrue(TEXT("중간 라운드에서 갱신된 seed가 다음 라운드로 이어짐"),
146 !FMath::IsNearlyEqual(ExpectedFirstRound[0], ExpectedSecondRound[0])
147 || !FMath::IsNearlyEqual(ExpectedFirstRound[1], ExpectedSecondRound[1]));
148
149 return true;
150}
