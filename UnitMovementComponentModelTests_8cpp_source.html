63 // 유닛 생성 (스폰 데이터 없이 도는 플래너용 Mock 재사용)
64 UMockEnemyUnitModel* Unit = Scope.New<UMockEnemyUnitModel>(World);

68 UUnitMovementComponentModel* MovementCompModel = Cast<UUnitMovementComponentModel>(Unit->GetBoardMovementComponentModel());

72 }
73 UAttributeSetComponentModel* AttrComp = Unit->GetAttributeComponentModel();
74 if (TestNotNull(TEXT("속성 컴포넌트 모델"), AttrComp) == false)
75 {
76 return false;
77 }
78
79 /* [1] 기본 상태: 이동 가능 */

83 AttrComp->AddLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Root, 2);

87 AttrComp->RemoveLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Root, 1);
88 TestFalse(TEXT("1스택 제거(1스택): 이동 불가"), MovementCompModel->CanSelfMove());
89

91 AttrComp->RemoveLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Root, 1);
92 TestTrue(TEXT("스택 소진(0스택): 이동 가능"), MovementCompModel->CanSelfMove());
93
94 return true;
