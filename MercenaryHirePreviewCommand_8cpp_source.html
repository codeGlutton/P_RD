50 void Toggle(const TArray<FString>& Args)
51 {
52 if (Shown.IsValid())
53 {
54 Shown->RemoveFromParent();
55 Shown.Reset();
56 if (Args.Num() == 0)
57 {
58 UE_LOG(LogTemp, Display, TEXT("[고용] 닫음"));
59 return;
60 }
61 }
62
63 UWorld* World = PickWorld();
64 if (World == nullptr)
65 {
66 UE_LOG(LogTemp, Warning, TEXT("[고용] 월드를 못 찾음"));
67 return;
68 }
69
70 UClass* WidgetClass = LoadClass<UMercenaryHireWidget>(nullptr, WidgetPath);
71 if (WidgetClass == nullptr)
72 {
73 // 부모 클래스를 아직 안 바꿨으면 여기서 걸린다. 조용히 아무 일도
74 // 안 일어나면 원인을 찾는 데 한참 걸리므로 이유를 남긴다.
75 UE_LOG(LogTemp, Warning,
76 TEXT("[고용] %s 를 UMercenaryHireWidget 으로 못 읽음. "
77 "WBP 의 부모 클래스를 확인하라."), WidgetPath);
78 return;
79 }
80
81 APlayerController* Controller = World->GetFirstPlayerController();
82 UMercenaryHireWidget* Widget = Controller != nullptr
83 ? CreateWidget<UMercenaryHireWidget>(Controller, WidgetClass)
84 : CreateWidget<UMercenaryHireWidget>(World, WidgetClass);
85 if (Widget == nullptr)
86 {
87 UE_LOG(LogTemp, Warning, TEXT("[고용] 위젯 생성 실패"));
88 return;
89 }
90
91 // 프론트엔드에 있으면 실제 후보를 건다. 없으면 시안 글자가 남는다.
92 if (const AFrontendGameMode* Frontend =
93 Cast<AFrontendGameMode>(World->GetAuthGameMode()))
94 {
95 TArray<FFrontendCharacterOption> Options;
96 if (Frontend->GetCharacterOptions(Options))
97 {
98 Widget->SetCharacterOptions(Options);
99 UE_LOG(LogTemp, Display, TEXT("[고용] 후보 %d명 걸음"),
100 Options.Num());
101 }
102 }
103
104 Widget->AddToViewport(5000);
105 Shown = Widget;
106 UE_LOG(LogTemp, Display, TEXT("[고용] 띄움"));
107 }
