69void UCarouselPanelWidget::CacheCarouselWidgets()
70{
71 /*
72 * WBP마다 카드 개수를 코드로 직접 넘기지 않기 위해 이름 규칙으로 최대 8개까지 찾는다.
73 * CarouselItem_N이 있으면 그 위젯을 카드로 사용하고, 없으면 CarouselButton_N 자체를 카드로 사용한다.
74 * 이렇게 해두면 단순 버튼형 WBP와 카드 안에 버튼이 들어간 WBP를 같은 코드로 처리할 수 있다.
75 */
76 mCarouselItems.Reset();
77 mCarouselButtons.Reset();
78
79 if (CloseButton != nullptr)
80 {
81 CloseButton->OnClicked.AddUniqueDynamic(this, &UCarouselPanelWidget::HandleCloseButtonClicked);
82 }
83 if (CloseButtonText != nullptr)
84 {
85 CloseButtonText->SetText(NSLOCTEXT("CarouselPanelWidget", "CloseButtonText", "X"));
86 }
87
88 for (int32 Index = 0; Index < FCarouselLayoutPolicy::MaxItemCount; ++Index)
89 {
90 const FString ItemName = FString::Printf(TEXT("CarouselItem_%d"), Index);
91 UWidget* Item = FindWidgetByName(WidgetTree, ItemName);
92 const FString ButtonName = FString::Printf(TEXT("CarouselButton_%d"), Index);
93 UButton* Button = Cast<UButton>(FindWidgetByName(WidgetTree, ButtonName));
94 if (Item == nullptr)
95 {
96 Item = Button;
97 }
98 if (Item == nullptr)
99 {
100 continue;
101 }
102
103 if (Button != nullptr)
104 {
105 mCarouselButtons.Add(Button);
106 }
107 mCarouselItems.Add(Item);
108 }
109
110 mSelectedCarouselIndex = mCarouselItems.IsEmpty() ? INDEX_NONE : FMath::Clamp(mSelectedCarouselIndex, 0, mCarouselItems.Num() - 1);
111}

143{
144 /*

148 if (Button == nullptr)
