18IMPLEMENT_SIMPLE_AUTOMATION_TEST(FTutorialEnemyColdSkillsTest, "P_RD.Tutorial.Guided.EnemyColdSkills",
19 EAutomationTestFlags::EditorContext | EAutomationTestFlags::EngineFilter)
20bool FTutorialEnemyColdSkillsTest::RunTest(const FString&)
21{
22 // Run in a fresh editor process, before UI/roster tests load the enemy skill assets.
23 auto* Spawn = LoadObject<UStaticEnemyUnitSpawnData>(nullptr,
24 TEXT("/Game/BP/DataAsset/Unit/EnemyUnit/Stage1/DA_EagleUnit.DA_EagleUnit"));
25 if (!TestNotNull(TEXT("Actual scripted enemy spawn data exists"), Spawn)) return false;
26 UClass* EnemyClass = Spawn->mModelClass.LoadSynchronous();
27 if (!TestNotNull(TEXT("Authored concrete enemy model exists"), EnemyClass)) return false;
28 int32 Pending = 0;
29 for (const auto& Skill : Spawn->mSkillDatas) Pending += Skill.IsPending() ? 1 : 0;
30 TestTrue(TEXT("Regression starts with unloaded soft skill references"), Pending > 0);
31 for (int32 EnemyIndex = 0; EnemyIndex < 2; ++EnemyIndex)
32 {
33 auto* Enemy = NewObject<UEnemyUnitModel>(GetTransientPackage(), EnemyClass);
34 auto* Component = Enemy->GetSkillComponentModel();
35 if (!TestNotNull(TEXT("Real enemy skill component exists"), Component)) return false;
36 // Same assignment used by UEnemyUnitModel::PostInitializeComponentModels.
37 Component->SetSkillFrom(Spawn->mSkillDatas);
38 for (int32 Index = 0; Index < Spawn->mSkillDatas.Num(); ++Index)
39 {
40 const auto* Entry = Component->GetSkill(Index);
41 TestTrue(TEXT("Each original slot contains an executable skill"), Entry && Entry->mData && !Entry->mData->mSkillPhaseLayers.IsEmpty());
42 TestTrue(TEXT("Enemy receives the authored skill, not a fallback"), Entry && Entry->mData == Spawn->mSkillDatas[Index].Get());
43 }
44 }
45 AddInfo(FString::Printf(TEXT("Assigned %d authored skills to both tutorial enemies; %d were initially unloaded."), Spawn->mSkillDatas.Num(), Pending));
46 return !HasAnyErrors();
47}
