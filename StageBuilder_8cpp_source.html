74void FStageBuilder::Build(OUT FStage& NewStage) const
75{
76 InitStage(OUT NewStage);
77
78 TArray<FRoomEdge> Edges;
79 CreateStartRoom(OUT NewStage);
80 Edges = MakeStartRoutes(OUT NewStage);
81
82 const int32 RowCount = mParams.mRowCount;
83 for (int32 RowIndex = 1; RowIndex < RowCount - 1; ++RowIndex)
84 {
85 CreateRooms(OUT NewStage, RowIndex, Edges);
86 Edges = MakeNextRoutes(OUT NewStage, RowIndex, Edges);
87 }
88
89 CreateRooms(OUT NewStage, RowCount - 1, Edges);
90
91 /* 디버깅 */
92#if UE_BUILD_DEVELOPMENT || UE_BUILD_DEBUG
93 for (auto Iter = NewStage.mRoomRows.rbegin(); Iter != NewStage.mRoomRows.rend(); ++Iter)
94 {
95 FString RowStr = TEXT("");
96 for (const TInstancedStruct<FRoom>& RoomInst : Iter->mRooms)
97 {
98 ERoomType Type = ERoomType::None;
99 FString LinkRoomStr = TEXT("");
100 if (const FRoom* Room = RoomInst.GetPtr<FRoom>())
101 {
102 Type = Room->mType;
103 for (int32 NextColumn : Room->mNextRoomColumns)
104 {
105 LinkRoomStr += FString::FromInt(NextColumn) + TEXT(",");
106 }
107 }
108 RowStr.Append(FString::Printf(TEXT(" [%-12s](%-7s) "), *EnumToString(Type), *LinkRoomStr));
109 }
110 UE_LOG(LogStageBuilder, Log, TEXT("(%s)"), *RowStr);
111 }
112#endif
113}

115void FStageBuilder::LoadAllAssetIds()
116{
117 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
118 checkf(AssetManager != nullptr, TEXT("에셋 매니저 nullptr"));
119
120 /* 방 PrimaryAssetId 캐싱 */
121
122 auto StageFilter = [FilterStr = EnumToString(mParams.mStageLevel)](const FAssetData& AssetData) -> bool {
123 FString FoundStr;
124 bool IsFound = AssetData.GetTagValue(TEXT("mStageLevel"), OUT FoundStr) == true;
125 return IsFound == true && FoundStr == FilterStr;
126 };
127
128 mRoomAssetIds[static_cast<uint8>(ERoomType::Treasure)] = GetFilteredPrimaryAssets(RoomPrimaryAssetTypes::GetTreasureRoomType(), StageFilter);
129 mRoomAssetIds[static_cast<uint8>(ERoomType::Shop)] = GetFilteredPrimaryAssets(RoomPrimaryAssetTypes::GetShopRoomType(), StageFilter);
130 mRoomAssetIds[static_cast<uint8>(ERoomType::Monster)] = GetFilteredPrimaryAssets(RoomPrimaryAssetTypes::GetMonsterRoomType(), StageFilter);
131 mRoomAssetIds[static_cast<uint8>(ERoomType::EliteMonster)] = GetFilteredPrimaryAssets(RoomPrimaryAssetTypes::GetEliteMonsterRoomType(), StageFilter);
132 mRoomAssetIds[static_cast<uint8>(ERoomType::BossMonster)] = GetFilteredPrimaryAssets(RoomPrimaryAssetTypes::GetBossMonsterRoomType(), StageFilter);
133
134 /* 아티팩트 및 스킬 PrimaryAssetId 캐싱 */
135
136 const uint8 RarityTypeCount = StaticCast<uint8>(ERarityType::Count);
137 for (uint8 RarityTypeIndex = 0; RarityTypeIndex < RarityTypeCount; ++RarityTypeIndex)
138 {
139 auto RarityFilter = [FilterStr = EnumToString(StaticCast<ERarityType>(RarityTypeIndex))](const FAssetData& AssetData) -> bool {
140 FString FoundStr;
141 bool IsFound = AssetData.GetTagValue(TEXT("mRarityType"), OUT FoundStr) == true;
142 return IsFound == true && FoundStr == FilterStr;
143 };
144 mArtifactAssetIds[RarityTypeIndex] = GetFilteredPrimaryAssets(ArtifactPrimaryAssetTypes::GetArtifactType(), RarityFilter);
145 const TArray<FAssetData> SkillAssetDatas = GetFilteredPrimaryAssetDatas(SkillPrimaryAssetTypes::GetActiveType(), RarityFilter);
146
147 const uint8 JobTypeCount = StaticCast<uint8>(EUnitJobType::PlayerJobCount);
148 for (uint8 JobTypeIndex = 0; JobTypeIndex < JobTypeCount; ++JobTypeIndex)
149 {
150 auto JobFilter = [FilterStr = EnumToString(StaticCast<EUnitJobType>(JobTypeIndex))](const FAssetData& AssetData) -> bool {
151 FString FoundStr;
152 bool IsFound = AssetData.GetTagValue(TEXT("mJobType"), OUT FoundStr) == true;
153 return IsFound == true && FoundStr == FilterStr;
154 };
155
156 mJobSkillAssetIds[JobTypeIndex][RarityTypeIndex] = GetFilteredPrimaryAssets(SkillAssetDatas, JobFilter);
157 }
158
159 auto JobFilter = [FilterStr = EnumToString(EUnitJobType::Common)](const FAssetData& AssetData) -> bool {
160 FString FoundStr;
161 bool IsFound = AssetData.GetTagValue(TEXT("mJobType"), OUT FoundStr) == true;
162 return IsFound == true && FoundStr == FilterStr;
163 };
164 mCommonSkillAssetIds[RarityTypeIndex] = GetFilteredPrimaryAssets(SkillAssetDatas, JobFilter);
165 }
166
167 /* 용병 PrimaryAssetId 캐싱 */
168
169 AssetManager->GetPrimaryAssetIdList(UnitPrimaryAssetTypes::GetPlayerUnitType(), mMercenaryAssetIds);
170
171 mIsLoadedIds = true;
172}

187TArray<FRoomEdge> FStageBuilder::MakeStartRoutes(OUT FStage& Stage) const
188{
189 const int32 StartPointCount = mBuildStream.RandRange(mParams.mMinStartPointCount, mParams.mMaxStartPointCount);
190 const int32 MaxPathCount = mParams.mMaxPathCount;
191 const int32 ColumnCount = mParams.mColumnCount;
192
193 checkf(StartPointCount < ColumnCount, TEXT("시작점이 최대 가로 점보다 많아질 수 없습니다"));
194 checkf(StartPointCount < MaxPathCount, TEXT("시작점이 최대 경로보다 많아질 수 없습니다"));
195
196 TArray<int32> StartPoints;
197 StartPoints.Reserve(FMath::Max(ColumnCount, MaxPathCount));
198
199 /* 칼럼 인덱스 채우기 */
200 for (int32 ColumnIndex = 0; ColumnIndex < ColumnCount; ++ColumnIndex)
201 {
202 StartPoints.Push(ColumnIndex);
203 }
204 /* 하나씩 빼가면서 모든 시작방에서 도달할 수 있는 일반 방 결정 */
205 for (int32 i = StartPointCount; i < ColumnCount; ++i)
206 {
207 const int32 RemoveIndex = mBuildStream.RandRange(0, StartPoints.Num() - 1);
208 StartPoints.RemoveAtSwap(RemoveIndex, EAllowShrinking::No);
209 }
210
211 FRoom& StartRoom = Stage.mRoomRows[0].mRooms[Stage.mStartColumn].GetMutable<FRoom>();
212 StartRoom.mNextRoomColumns = StartPoints;
213
214 /* 시작 방 -> 일반 방의 모든 경로 결정 */
215 for (int32 i = StartPointCount; i < MaxPathCount; ++i)
216 {
217 const int32 PathStartPoint = StartPoints[mBuildStream.RandRange(0, StartPointCount - 1)];
218 StartPoints.Push(PathStartPoint);
219 }
220
221 /* 방 생성 데이터로 사용될 Edge 반환 데이터 채우기 */
222 TArray<FRoomEdge> NewEdges;
223 NewEdges.Init(FRoomEdge(), ColumnCount);
224 for (int32 i = 0; i < MaxPathCount; ++i)
225 {
226 NewEdges[StartPoints[i]].mPreRoomColumns.Push(Stage.mStartColumn);
227 }
228 return NewEdges;
229}

231TArray<FRoomEdge> FStageBuilder::MakeNextRoutes(OUT FStage& Stage, int32 CurRowIndex, const TArray<FRoomEdge>& CurEdges) const
232{
233 const int32 ColumnCount = mParams.mColumnCount;
234 const int32 RowCount = mParams.mRowCount;
235
236 TArray<FRoomEdge> NewEdges;
237 NewEdges.Init(FRoomEdge(), ColumnCount);
238
239 if (CurRowIndex == RowCount - 2)
240 {
241 /* 상점 방 -> 보스 방 연결 */
242
243 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
244 {
245 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
246 if (PreRoomColumns.IsEmpty() == true)
247 {
248 continue;
249 }
250
251 TInstancedStruct<FRoom>& CurRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
252 FRoom* CurRoomPtr = CurRoomInst.GetMutablePtr<FRoom>();
253 CurRoomPtr->mNextRoomColumns.Push(Stage.mStartColumn);
254
255 for (int32 PreRoomColumnIndex : PreRoomColumns)
256 {
257 NewEdges[Stage.mStartColumn].mPreRoomColumns.Push(CurColumnIndex);
258 }
259 }
260 return NewEdges;
261 }
262
263 /* 일반 방 -> 일반 방 연결 */
264
265 int32 MaxAllocatedColumnIndex = 0;
266
267 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
268 {
269 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
270 if (PreRoomColumns.IsEmpty() == true)
271 {
272 continue;
273 }
274
275 const int32 MinNewColumnIndex = FMath::Max(MaxAllocatedColumnIndex, CurColumnIndex - 1);
276 const int32 MaxNewColumnIndex = FMath::Min(CurColumnIndex + 1, ColumnCount - 1);
277
278 TInstancedStruct<FRoom>& CurRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
279 FRoom* CurRoomPtr = CurRoomInst.GetMutablePtr<FRoom>();
280 TSet<int32> NextRoomColumns;
281
282 for (int32 PreRoomColumnIndex : PreRoomColumns)
283 {
284 int32 NewColumnIndex = mBuildStream.RandRange(MinNewColumnIndex, MaxNewColumnIndex);
285 NewEdges[NewColumnIndex].mPreRoomColumns.Push(CurColumnIndex);
286 NextRoomColumns.Add(NewColumnIndex);
287
288 if (MaxAllocatedColumnIndex < NewColumnIndex)
289 {
290 MaxAllocatedColumnIndex = NewColumnIndex;
291 }
292 }
293
294 CurRoomPtr->mNextRoomColumns = NextRoomColumns.Array();
295 }
296
297 return NewEdges;
298}

300void FStageBuilder::CreateStartRoom(OUT FStage& Stage) const
301{
302 const int32 ColumnCount = mParams.mColumnCount;
303 const int32 StartColumn = Stage.mStartColumn = ColumnCount / 2;
304
305 ERoomType StartRoomType = ERoomType::None;
306 if (Stage.mStageLevel == EStageLevelType::Stage1)
307 {
308 StartRoomType = ERoomType::Monster;
309 }
310 else
311 {
312 StartRoomType = ERoomType::Shop;
313 }
314 FRoom& StartRoom = CreateRoom(StartRoomType, 0, StartColumn, Stage.mRoomRows[0].mRooms[StartColumn]);
315
316 if (mFirstRoomOverride.IsValid())
317 {
318 if (mFirstRoomOverride.PrimaryAssetType == RoomPrimaryAssetTypes::GetMonsterRoomType()
319 && UAssetManager::Get().GetPrimaryAssetPath(mFirstRoomOverride).IsValid())
320 StartRoom.mStaticRoomSpawnDataId = mFirstRoomOverride;
321 else
322 UE_LOG(LogStageBuilder, Error, TEXT("Invalid first-room override: %s"), *mFirstRoomOverride.ToString());
323 }
324 Stage.mCurRow = 0;
325 Stage.mCurColumn = StartColumn;
326}

328void FStageBuilder::CreateRooms(OUT FStage& Stage, int32 CurRowIndex, const TArray<FRoomEdge>& CurEdges) const
329{
330 const int32 ColumnCount = mParams.mColumnCount;
331 const int32 RowCount = mParams.mRowCount;
332
333 /* 특정 타입의 행 검사 */
334
335 ERoomType NewRoomType = ERoomType::None;
336 if (CurRowIndex == RowCount - 1)
337 {
338 NewRoomType = ERoomType::BossMonster;
339 }
340 else if (CurRowIndex == RowCount - 2)
341 {
342 NewRoomType = ERoomType::Shop;
343 }
344 else if (CurRowIndex % mGlobalSetting.mTreasureColumnInterval == 0)
345 {
346 NewRoomType = ERoomType::Treasure;
347 }
348
349 if (NewRoomType == ERoomType::None)
350 {
351 /* 특정 타입의 행이 아닌 경우, 각 방의 루트를 검사해서 적절한 방 타입 도입 */
352
353 const bool IsRowInShopRange = CurRowIndex > mGlobalSetting.mBottomShopColumnOffset && CurRowIndex < (RowCount - mGlobalSetting.mTopShopColumnOffset);
354 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
355 {
356 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
357 if (PreRoomColumns.IsEmpty() == true)
358 {
359 continue;
360 }
361
362 bool IsPreRoomShop = false;
363 bool IsPreRoomElite = false;
364 for (int32 PreRoomColumnIndex : PreRoomColumns)
365 {
366 TInstancedStruct<FRoom>& PreRoomInst = Stage.mRoomRows[CurRowIndex - 1].mRooms[PreRoomColumnIndex];
367 FRoom* PreRoomPtr = PreRoomInst.GetMutablePtr<FRoom>();
368 if (PreRoomPtr == nullptr)
369 {
370 continue;
371 }
372
373 if (PreRoomPtr->mType == ERoomType::Shop)
374 {
375 IsPreRoomShop = true;
376 }
377 if (PreRoomPtr->mType == ERoomType::EliteMonster)
378 {
379 IsPreRoomElite = true;
380 }
381 }
382
383 TArray<ERoomBuildType> CandidateRoomTypes;
384 CandidateRoomTypes.Push(ERoomBuildType::Monster);
385 if (IsRowInShopRange == true && IsPreRoomShop == false)
386 {
387 CandidateRoomTypes.Push(ERoomBuildType::Shop);
388 }
389 if (IsPreRoomElite == false)
390 {
391 CandidateRoomTypes.Push(ERoomBuildType::EliteMonster);
392 }
393
394 // 후보들 중에 랜덤 배정
395 NewRoomType = mParams.mRoomBuildRate.GetType(mBuildStream.FRand(), CandidateRoomTypes);
396
397 TInstancedStruct<FRoom>& NewRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
398 FRoom& NewRoom = CreateRoom(NewRoomType, CurRowIndex, CurColumnIndex, NewRoomInst);
399 }
400 }
401 else
402 {
403 /* 정해진 타입의 행인 경우 */
404
405 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
406 {
407 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
408 if (PreRoomColumns.IsEmpty() == true)
409 {
410 continue;
411 }
412
413 TInstancedStruct<FRoom>& NewRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
414 FRoom& NewRoom = CreateRoom(NewRoomType, CurRowIndex, CurColumnIndex, NewRoomInst);
415 }
416 }

419FRoom& FStageBuilder::CreateRoom(ERoomType Type, int32 Row, int32 Column, TInstancedStruct<FRoom>& Room) const
420{
421 FRoom* NewRoomPtr = nullptr;
422 switch (Type)
423 {
424 case ERoomType::Treasure:
425 {
426 /* 보상 아티팩트 */
427
428 Room.InitializeAs<FTreasureRoom>();
429 auto& NewRoom = Room.GetMutable<FTreasureRoom>();
430
431 // 개봉 보상 골드 선정
432 NewRoom.mRewardMoney = URandomStreamFunctionLibrary::GetRandomFromInterval(mBuildStream, mGlobalSetting.mTreasureRewardMoney);
433
434 // 개봉 보상 아티팩트 선정 (해당 희귀도 에셋이 없으면 골드만 지급)
435 const uint8 RarityTypeIndex = GetRandomRarityIndex(mParams.mArtifactRarityRate);
436 const TArray<FPrimaryAssetId>& ArtifactIdArray = mArtifactAssetIds[RarityTypeIndex];
437 NewRoom.mRewardArtifactDataIds = URandomStreamFunctionLibrary::GetRandomUniqueItemsUsingNonCopiedArray(mBuildStream, ArtifactIdArray, 3);
438
439 NewRoomPtr = &NewRoom;
440 break;
441 }
442 case ERoomType::Shop:
443 {
444 Room.InitializeAs<FShopRoom>();
445 auto& NewRoom = Room.GetMutable<FShopRoom>();
446
447 /* 판매 아이템들 */
448
449 const uint8 JobTypeCount = StaticCast<uint8>(EUnitJobType::PlayerJobCount);
450 for (uint8 JobTypeIndex = 0; JobTypeIndex < JobTypeCount; ++JobTypeIndex)
451 {
452 const uint8 RarityTypeIndex = GetRandomRarityIndex(mParams.mSkillRarityRate);
453 const TArray<FPrimaryAssetId>& SkillIdArray = mJobSkillAssetIds[JobTypeIndex][RarityTypeIndex];
454
455 NewRoom.mSaleJobSkillDataItems[JobTypeIndex].mSaleItemIds = URandomStreamFunctionLibrary::GetRandomUniqueItemsUsingNonCopiedArray(mBuildStream, SkillIdArray, 2);
456 }
457 {
458 const uint8 RarityTypeIndex = GetRandomRarityIndex(mParams.mSkillRarityRate);
459 const TArray<FPrimaryAssetId>& SkillIdArray = mCommonSkillAssetIds[RarityTypeIndex];
460 NewRoom.mSaleCommonSkillDataItems.mSaleItemIds = URandomStreamFunctionLibrary::GetRandomUniqueItemsUsingNonCopiedArray(mBuildStream, SkillIdArray, 3);
461 }
462
463 /* 판매 아티펙트 선정 */
464 {
465 const uint8 RarityTypeIndex = GetRandomRarityIndex(mParams.mArtifactRarityRate);
466 const TArray<FPrimaryAssetId>& ArtifactIdArray = mArtifactAssetIds[RarityTypeIndex];
467 NewRoom.mSaleArtifactDataItems.mSaleItemIds = URandomStreamFunctionLibrary::GetRandomUniqueItemsUsingNonCopiedArray(mBuildStream, ArtifactIdArray, 3);
468 }
469
470 /* 고용가능한 용병 (중복 허용) */
471
472 for (int32 i = 0; i < 3; ++i)
473 {
474 FMercenaryCandidate Candidate;
475 Candidate.mSaleMercenaryId = URandomStreamFunctionLibrary::GetRandomItem(mBuildStream, mMercenaryAssetIds);
476 Candidate.mLevel = URandomStreamFunctionLibrary::GetRandomFromInterval(mBuildStream, mParams.mMercenaryLevelRange);
477 Candidate.mPrice = mLevelCache.GetPrice(Candidate.mLevel);
478
479 const UEnum* StaticJobEnum = StaticEnum<EUnitJobType>();
480 const FString FoundJobStr = GetPropertyAssetData(Candidate.mSaleMercenaryId, TEXT("mJobType"));
481 const int64 FoundJobIndex = StaticJobEnum->GetValueByNameString(FoundJobStr);
482
483 /* 최대 레벨부터 내림차순으로 스킬 선택 (레벨 2 이상, 최대 5개) */
484
485 for (int32 SkillLevel = Candidate.mLevel; SkillLevel >= 2; --SkillLevel)
486 {
487 if (Candidate.mOwingSkillIds.Num() >= USkillComponentModel::DEFAULT_SKILL_POOL_SIZE)
488 {
489 break;
490 }
491
492 const FRarityRate RarityRate = mLevelCache.GetRarityRate(SkillLevel);
493 const uint8 RarityIndex = GetRandomRarityIndex(RarityRate);
494
495 const TArray<FPrimaryAssetId>& JobSkillArray = mJobSkillAssetIds[FoundJobIndex][RarityIndex];
496 Candidate.mOwingSkillIds.Push(URandomStreamFunctionLibrary::GetRandomItem(mBuildStream, JobSkillArray));
497 }
498
499 NewRoom.mSaleMercenaryDataCandidates.mCandidates.Add(Candidate);
500 }
501
502 NewRoomPtr = &NewRoom;
503 break;
504 }
505 case ERoomType::Monster:
506 {
507 Room.InitializeAs<FMonsterRoom>();
508 auto& NewRoom = Room.GetMutable<FMonsterRoom>();
509
510 NewRoom.mRewardMoney = URandomStreamFunctionLibrary::GetRandomFromInterval(mBuildStream, mGlobalSetting.mMonsterRewardMoney);
511 NewRoom.mRewardExp = mGlobalSetting.mMonsterRewardExp;
512
513 NewRoomPtr = &NewRoom;
514 break;
515 }
516 case ERoomType::EliteMonster:
517 {
518 /* 보상 아티팩트 */
519
520 Room.InitializeAs<FEliteMonsterRoom>();
521 auto& NewRoom = Room.GetMutable<FEliteMonsterRoom>();
522
523 NewRoom.mRewardMoney = URandomStreamFunctionLibrary::GetRandomFromInterval(mBuildStream, mGlobalSetting.mEliteRewardMoney);
524 NewRoom.mRewardExp = mGlobalSetting.mEliteRewardExp;
525
526 const uint8 RarityTypeIndex = GetRandomRarityIndex(mParams.mArtifactRarityRate);
527
528 const TArray<FPrimaryAssetId>& ArtifactIdArray = mArtifactAssetIds[RarityTypeIndex];
529 NewRoom.mRewardArtifactDataIds = URandomStreamFunctionLibrary::GetRandomUniqueItemsUsingNonCopiedArray(mBuildStream, ArtifactIdArray, 3);
530
531 NewRoomPtr = &NewRoom;
532 break;
533 }
534 case ERoomType::BossMonster:
535 {
536 /* 보상 아티팩트 (보스방은 항상 에픽만) */
537
538 Room.InitializeAs<FBossMonsterRoom>();
539 auto& NewRoom = Room.GetMutable<FBossMonsterRoom>();
540
541 NewRoom.mRewardMoney = URandomStreamFunctionLibrary::GetRandomFromInterval(mBuildStream, mGlobalSetting.mBossRewardMoney);
542 NewRoom.mRewardExp = mGlobalSetting.mBossRewardExp;
543
544 const TArray<FPrimaryAssetId>& ArtifactIdArray = mArtifactAssetIds[StaticCast<uint8>(ERarityType::Epic)];
545 NewRoom.mRewardArtifactDataIds = URandomStreamFunctionLibrary::GetRandomUniqueItemsUsingNonCopiedArray(mBuildStream, ArtifactIdArray, 3);
546
547 NewRoomPtr = &NewRoom;
548 break;
549 }
550 }
551 checkf(NewRoomPtr != nullptr, TEXT("알 수 없는 방 타입 오류"));
552
553 const TArray<FPrimaryAssetId>& RoomIdArray = mRoomAssetIds[static_cast<uint8>(Type)];
554 NewRoomPtr->mRow = Row;
555 NewRoomPtr->mColumn = Column;
556 NewRoomPtr->mPositionOffsetRate = FVector2D(mBuildStream.FRandRange(-1., 1.), mBuildStream.FRandRange(-1., 1.));
557 NewRoomPtr->mStaticRoomSpawnDataId = URandomStreamFunctionLibrary::GetRandomItem(mBuildStream, RoomIdArray);
558
559 // 디버깅용 몬스터방 고정: 지정 이름의 방 DA가 실제로 있으면 추첨 결과를 교체
560 // (추첨은 위에서 이미 소모했으므로 다른 방/보상의 시드 재현성은 유지됨)
561 if (Type == ERoomType::Monster && GFixedMonsterRoomNameForDebugging.IsEmpty() == false)
562 {
563 const FPrimaryAssetId FixedRoomId(RoomPrimaryAssetTypes::GetMonsterRoomType(), FName(GFixedMonsterRoomNameForDebugging));
564 if (UAssetManager::Get().GetPrimaryAssetPath(FixedRoomId).IsValid() == true)
565 {
566 NewRoomPtr->mStaticRoomSpawnDataId = FixedRoomId;
567 }
