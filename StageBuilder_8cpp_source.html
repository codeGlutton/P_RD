53void FStageBuilder::Build(OUT FStage& NewStage) const
54{
55 InitStage(OUT NewStage);
56
57 TArray<FRoomEdge> Edges;
58 CreateStartRoom(OUT NewStage);
59 Edges = MakeStartRoutes(OUT NewStage);
60
61 const int32 RowCount = mParams.mRowCount;
62 for (int32 RowIndex = 1; RowIndex < RowCount - 1; ++RowIndex)
63 {
64 CreateRooms(OUT NewStage, RowIndex, Edges);
65 Edges = MakeNextRoutes(OUT NewStage, RowIndex, Edges);
66 }
67
68 CreateRooms(OUT NewStage, RowCount - 1, Edges);
69
70 /* 디버깅 */
71#if UE_BUILD_DEVELOPMENT || UE_BUILD_DEBUG
72 for (auto Iter = NewStage.mRoomRows.rbegin(); Iter != NewStage.mRoomRows.rend(); ++Iter)
73 {
74 FString RowStr = TEXT("");
75 for (const TInstancedStruct<FRoom>& RoomInst : Iter->mRooms)
76 {
77 ERoomType Type = ERoomType::None;
78 FString LinkRoomStr = TEXT("");
79 if (const FRoom* Room = RoomInst.GetPtr<FRoom>())
80 {
81 Type = Room->mType;
82 for (int32 NextColumn : Room->mNextRoomColumns)
83 {
84 LinkRoomStr += FString::FromInt(NextColumn) + TEXT(",");
85 }
86 }
87 RowStr.Append(FString::Printf(TEXT(" [%-12s](%-7s) "), *EnumToString(Type), *LinkRoomStr));
88 }
89 UE_LOG(LogStageBuilder, Log, TEXT("(%s)"), *RowStr);
90 }
91#endif
92}

132TArray<FRoomEdge> FStageBuilder::MakeStartRoutes(OUT FStage& Stage) const
133{
134 const int32 StartPointCount = mBuildStream.RandRange(mParams.mMinStartPointCount, mParams.mMaxStartPointCount);
135 const int32 MaxPathCount = mParams.mMaxPathCount;
136 const int32 ColumnCount = mParams.mColumnCount;
137
138 checkf(StartPointCount < ColumnCount, TEXT("시작점이 최대 가로 점보다 많아질 수 없습니다"));
139 checkf(StartPointCount < MaxPathCount, TEXT("시작점이 최대 경로보다 많아질 수 없습니다"));
140
141 TArray<int32> StartPoints;
142 StartPoints.Reserve(FMath::Max(ColumnCount, MaxPathCount));
143
144 /* 칼럼 인덱스 채우기 */
145 for (int32 ColumnIndex = 0; ColumnIndex < ColumnCount; ++ColumnIndex)
146 {
147 StartPoints.Push(ColumnIndex);
148 }
149 /* 하나씩 빼가면서 모든 시작방에서 도달할 수 있는 일반 방 결정 */
150 for (int32 i = StartPointCount; i < ColumnCount; ++i)
151 {
152 const int32 RemoveIndex = mBuildStream.RandRange(0, StartPoints.Num() - 1);
153 StartPoints.RemoveAtSwap(RemoveIndex, EAllowShrinking::No);
154 }
155
156 FRoom& StartRoom = Stage.mRoomRows[0].mRooms[Stage.mStartColumn].GetMutable<FRoom>();
157 StartRoom.mNextRoomColumns = StartPoints;
158
159 /* 시작 방 -> 일반 방의 모든 경로 결정 */
160 for (int32 i = StartPointCount; i < MaxPathCount; ++i)
161 {
162 const int32 PathStartPoint = StartPoints[mBuildStream.RandRange(0, StartPointCount - 1)];
163 StartPoints.Push(PathStartPoint);
164 }
165
166 /* 방 생성 데이터로 사용될 Edge 반환 데이터 채우기 */
167 TArray<FRoomEdge> NewEdges;
168 NewEdges.Init(FRoomEdge(), ColumnCount);
169 for (int32 i = 0; i < MaxPathCount; ++i)
170 {
171 NewEdges[StartPoints[i]].mPreRoomColumns.Push(Stage.mStartColumn);
172 }
173 return NewEdges;
174}

176TArray<FRoomEdge> FStageBuilder::MakeNextRoutes(OUT FStage& Stage, int32 CurRowIndex, const TArray<FRoomEdge>& CurEdges) const
177{
178 const int32 ColumnCount = mParams.mColumnCount;
179 const int32 RowCount = mParams.mRowCount;
180
181 TArray<FRoomEdge> NewEdges;
182 NewEdges.Init(FRoomEdge(), ColumnCount);
183
184 if (CurRowIndex == RowCount - 2)
185 {
186 /* 상점 방 -> 보스 방 연결 */
187
188 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
189 {
190 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
191 if (PreRoomColumns.IsEmpty() == true)
192 {
193 continue;
194 }
195
196 TInstancedStruct<FRoom>& CurRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
197 FRoom* CurRoomPtr = CurRoomInst.GetMutablePtr<FRoom>();
198 CurRoomPtr->mNextRoomColumns.Push(Stage.mStartColumn);
199
200 for (int32 PreRoomColumnIndex : PreRoomColumns)
201 {
202 NewEdges[Stage.mStartColumn].mPreRoomColumns.Push(CurColumnIndex);
203 }
204 }
205 return NewEdges;
206 }
207
208 /* 일반 방 -> 일반 방 연결 */
209
210 int32 MaxAllocatedColumnIndex = 0;
211
212 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
213 {
214 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
215 if (PreRoomColumns.IsEmpty() == true)
216 {
217 continue;
218 }
219
220 const int32 MinNewColumnIndex = FMath::Max(MaxAllocatedColumnIndex, CurColumnIndex - 1);
221 const int32 MaxNewColumnIndex = FMath::Min(CurColumnIndex + 1, ColumnCount - 1);
222
223 TInstancedStruct<FRoom>& CurRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
224 FRoom* CurRoomPtr = CurRoomInst.GetMutablePtr<FRoom>();
225 TSet<int32> NextRoomColumns;
226
227 for (int32 PreRoomColumnIndex : PreRoomColumns)
228 {
229 int32 NewColumnIndex = mBuildStream.RandRange(MinNewColumnIndex, MaxNewColumnIndex);
230 NewEdges[NewColumnIndex].mPreRoomColumns.Push(CurColumnIndex);
231 NextRoomColumns.Add(NewColumnIndex);
232
233 if (MaxAllocatedColumnIndex < NewColumnIndex)
234 {
235 MaxAllocatedColumnIndex = NewColumnIndex;
236 }
237 }
238
239 CurRoomPtr->mNextRoomColumns = NextRoomColumns.Array();
240 }
241
242 return NewEdges;
243}

255void FStageBuilder::CreateRooms(OUT FStage& Stage, int32 CurRowIndex, const TArray<FRoomEdge>& CurEdges) const
256{
257 const int32 ColumnCount = mParams.mColumnCount;
258 const int32 RowCount = mParams.mRowCount;
259
260 /* 특정 타입의 행 검사 */
261
262 ERoomType NewRoomType = ERoomType::None;
263 if (CurRowIndex == RowCount - 1)
264 {
265 NewRoomType = ERoomType::BossMonster;
266 }
267 else if (CurRowIndex == RowCount - 2)
268 {
269 NewRoomType = ERoomType::Shop;
270 }
271 else if (CurRowIndex % mGlobalSetting.mTreasureColumnInterval == 0)
272 {
273 NewRoomType = ERoomType::Treasure;
274 }
275
276 if (NewRoomType == ERoomType::None)
277 {
278 /* 특정 타입의 행이 아닌 경우, 각 방의 루트를 검사해서 적절한 방 타입 도입 */
279
280 const bool IsRowInShopRange = CurRowIndex > mGlobalSetting.mBottomShopColumnOffset && CurRowIndex < (RowCount - mGlobalSetting.mTopShopColumnOffset);
281 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
282 {
283 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
284 if (PreRoomColumns.IsEmpty() == true)
285 {
286 continue;
287 }
288
289 bool IsPreRoomShop = false;
290 bool IsPreRoomElite = false;
291 for (int32 PreRoomColumnIndex : PreRoomColumns)
292 {
293 TInstancedStruct<FRoom>& PreRoomInst = Stage.mRoomRows[CurRowIndex - 1].mRooms[PreRoomColumnIndex];
294 FRoom* PreRoomPtr = PreRoomInst.GetMutablePtr<FRoom>();
295 if (PreRoomPtr == nullptr)
296 {
297 continue;
298 }
299
300 if (PreRoomPtr->mType == ERoomType::Shop)
301 {
302 IsPreRoomShop = true;
303 }
304 if (PreRoomPtr->mType == ERoomType::EliteMonster)
305 {
306 IsPreRoomElite = true;
307 }
308 }
309
310 TArray<ERoomBuildType> CandidateRoomTypes;
311 CandidateRoomTypes.Push(ERoomBuildType::Monster);
312 if (IsRowInShopRange == true && IsPreRoomShop == false)
313 {
314 CandidateRoomTypes.Push(ERoomBuildType::Shop);
315 }
316 if (IsPreRoomElite == false)
317 {
318 CandidateRoomTypes.Push(ERoomBuildType::EliteMonster);
319 }
320
321 // 후보들 중에 랜덤 배정
322 NewRoomType = mParams.mRoomBuildRate.GetType(mBuildStream.FRand(), CandidateRoomTypes);
323
324 TInstancedStruct<FRoom>& NewRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
325 FRoom& NewRoom = CreateRoom(NewRoomType, CurRowIndex, CurColumnIndex, NewRoomInst);
326 }
327 }
328 else
329 {
330 /* 정해진 타입의 행인 경우 */
331
332 for (int32 CurColumnIndex = 0; CurColumnIndex < ColumnCount; ++CurColumnIndex)
333 {
334 const TArray<int32>& PreRoomColumns = CurEdges[CurColumnIndex].mPreRoomColumns;
335 if (PreRoomColumns.IsEmpty() == true)
336 {
337 continue;
338 }
339
340 TInstancedStruct<FRoom>& NewRoomInst = Stage.mRoomRows[CurRowIndex].mRooms[CurColumnIndex];
341 FRoom& NewRoom = CreateRoom(NewRoomType, CurRowIndex, CurColumnIndex, NewRoomInst);
342 }
343 }
344}

382 NewRoom.mSaleEquipmentDataIds.Push(URandomStreamFunctionLibrary::GetRandomItem(mBuildStream, EquipmentIdArray));
