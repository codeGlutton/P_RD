194 {
195 return false;

205 if (LevelGeometry.GetLocalSize().X <= 0.0f || LevelGeometry.GetLocalSize().Y <= 0.0f)

207 return false;

290void UCombatTileMapHUDWidget::UpdateGoldValueLabel(int32 NewGold) const
291{
292 // 최초 표시(전투 진입)나 감소는 연출 없이 즉시 반영한다 — 차르륵은 '획득' 연출.
293 if (mDisplayedGold == INDEX_NONE || NewGold < mDisplayedGold)
294 {
295 if (GetWorld() != nullptr)
296 {
297 GetWorld()->GetTimerManager().ClearTimer(mGoldCountUpTimerHandle);
298 }
299 mDisplayedGold = NewGold;
300 SetGoldValueLabelText(NewGold);
301 return;
302 }
303
304 if (NewGold == mDisplayedGold || GetWorld() == nullptr)
305 {
306 return;
307 }
308
309 // 증가: 목표만 갱신하고 30Hz 반복 타이머로 수렴시킨다. 이미 도는 중이면 목표만 늘어난다.
310 mGoldCountUpTarget = NewGold;
311 if (GetWorld()->GetTimerManager().IsTimerActive(mGoldCountUpTimerHandle) == false)
312 {
313 if (mCoinGainSound != nullptr)
314 {
315 UGameplayStatics::PlaySound2D(this, mCoinGainSound);
316 }
317 GetWorld()->GetTimerManager().SetTimer(
318 mGoldCountUpTimerHandle,
319 FTimerDelegate::CreateUObject(this, &UCombatTileMapHUDWidget::TickGoldCountUp),
320 0.033f, true);
321 }

324void UCombatTileMapHUDWidget::TickGoldCountUp() const
325{
326 // 남은 차이의 ~18%씩(최소 1) 올려 자연스럽게 감속하며 0.5~0.8초 안에 목표에 붙는다.
327 const int32 Remaining = mGoldCountUpTarget - mDisplayedGold;
328 const int32 Step = FMath::Max(1, FMath::RoundToInt(StaticCast<float>(Remaining) * 0.18f));
329 mDisplayedGold = FMath::Min(mDisplayedGold + Step, mGoldCountUpTarget);
330 SetGoldValueLabelText(mDisplayedGold);
331
332 if (mDisplayedGold >= mGoldCountUpTarget && GetWorld() != nullptr)
333 {
334 GetWorld()->GetTimerManager().ClearTimer(mGoldCountUpTimerHandle);
335 }

435void UCombatTileMapHUDWidget::RebuildEquipmentBar()
436{
437 // 기존 칩 제거(개수가 바뀔 수 있어 매번 다시 만든다).
438 for (UBorder* Chip : mEquipmentChips)
439 {
440 if (Chip != nullptr) { Chip->RemoveFromParent(); }
441 }
442 for (UTextBlock* Text : mEquipmentChipTexts)
443 {
444 if (Text != nullptr) { Text->RemoveFromParent(); }
445 }
446 mEquipmentChips.Reset();
447 mEquipmentChipTexts.Reset();
448
449 // 디자이너 스킨(concept_02): 레거시 칩(Border+Text) 대신 탑바 레벨 아래에 장비 아이콘을 그린다.
450 if (IsDesignerSkinActive())
451 {
452 RebuildEquipmentIcons();
453 return;
454 }
455
456 UCanvasPanel* Canvas = RootCanvas.Get();
457 if (Canvas == nullptr || WidgetTree == nullptr || mCombatUIModel == nullptr)
458 {
459 return;
460 }
461
462 const TArray<FEquipmentUI>& Equips = mCombatUIModel->GetEquipmentUIs();
463
464 // 탑바(~0.11) 바로 아래, 스킬레일(0.144) 위의 얇은 좌측 줄.
465 const float ChipWidth = 0.050f;
466 const float ChipHeight = 0.030f;
467 const float Gap = 0.006f;
468 const float Left0 = 0.022f;
469 const float Top = 0.112f;
470 for (int32 Index = 0; Index < Equips.Num(); ++Index)
471 {
472 UBorder* Chip = WidgetTree->ConstructWidget<UBorder>(UBorder::StaticClass());
473 UTextBlock* Text = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass());
474 if (Chip == nullptr || Text == nullptr)
475 {
476 continue;
477 }
478
479 Chip->SetBrushColor(Equips[Index].mIsEquipped
480 ? FLinearColor(0.16f, 0.40f, 0.40f, 0.92f)
481 : FLinearColor(0.12f, 0.17f, 0.19f, 0.80f));
482 Chip->SetPadding(FMargin(2.0f, 1.0f));
483
484 Text->SetJustification(ETextJustify::Center);
485 Text->SetText(Equips[Index].mName.IsEmpty()
486 ? FText::Format(NSLOCTEXT("CombatTileMapHUDWidget", "EquipSlotFallback", "EQ{0}"), FText::AsNumber(Index + 1))
487 : Equips[Index].mName);
488 Text->SetColorAndOpacity(FSlateColor(FLinearColor(0.90f, 1.0f, 0.96f, 1.0f)));
489 SetChipFontSize(Text, 11);
490
491 Chip->AddChild(Text);
492 Canvas->AddChildToCanvas(Chip);
493
494 const float Left = Left0 + StaticCast<float>(Index) * (ChipWidth + Gap);
495 RDUILayout::ApplyAnchoredSlot(Chip, FAnchors(Left, Top, Left + ChipWidth, Top + ChipHeight), 31);
496
497 mEquipmentChips.Add(Chip);
498 mEquipmentChipTexts.Add(Text);
499 }

550void UCombatTileMapHUDWidget::RebuildTurnOrderBar()
551{
552 // 풀스크린 지도 뷰 동안엔 턴 순서를 다시 만들지 않는다(만들면 숨긴 칩이 되살아나 지도 위에 뜬다). 지도 닫힐 때 복원된다.
553 if (mCombatControlsHidden)
554 {
555 return;
556 }
557
558 for (UBorder* Chip : mTurnOrderChips)
559 {
560 if (Chip != nullptr) { Chip->RemoveFromParent(); }
561 }
562 for (UTextBlock* Text : mTurnOrderChipTexts)
563 {
564 if (Text != nullptr) { Text->RemoveFromParent(); }
565 }
566 mTurnOrderChips.Reset();
567 mTurnOrderChipTexts.Reset();
568
569 UCanvasPanel* Canvas = GetSkinTargetCanvas(); // 스킨 활성 시 DesignCanvas — 칩 줄이 레터박스 스킨과 함께 움직인다.
570 if (Canvas == nullptr || WidgetTree == nullptr || mCombatUIModel == nullptr)
571 {
572 return;
573 }
574
575 const TArray<FUnitUI>& Units = mCombatUIModel->GetUnitUIs();
576 const FTurnUI& Turn = mCombatUIModel->GetTurnUI();
577
578 int32 PlayerId = INDEX_NONE;
579 for (const FUnitUI& Unit : Units)
580 {
581 if (Unit.mIsPlayer) { PlayerId = Unit.mUnitId; break; }
582 }
583
584 // 순서는 무조건 플레이어(나)부터, 그 뒤 턴순서(없으면 유닛순서)에서 플레이어 제외.
585 TArray<int32> Order;
586 if (PlayerId != INDEX_NONE)
587 {
588 Order.Add(PlayerId);
589 }
590 if (Turn.mTurnOrderUnitIds.Num() > 0)
591 {
592 for (int32 Id : Turn.mTurnOrderUnitIds)
593 {
594 if (Id != PlayerId && !Order.Contains(Id)) { Order.Add(Id); }
595 }
596 }
597 else
598 {
599 for (const FUnitUI& Unit : Units)
600 {
601 if (Unit.mUnitId != PlayerId && !Order.Contains(Unit.mUnitId)) { Order.Add(Unit.mUnitId); }
602 }
603 }
604 if (Order.Num() == 0)
605 {
606 return;
607 }
608
609 // 전투 HUD 가운데 하단, 칩 줄을 가로 중앙 정렬.
610 const bool bChipSkin = IsDesignerSkinActive();
611 const float ChipWidth = 0.048f;
612 const float ChipHeight = 0.040f;
613 const float Gap = 0.007f;
614 const float Top = 0.108f;
615 const float Total = StaticCast<float>(Order.Num()) * ChipWidth + StaticCast<float>(Order.Num() - 1) * Gap;
616 const float Start = 0.5f - Total * 0.5f;
617 // 스킨(엣지 피닝): 중앙(0.5,0) 핀 + 디자인px 오프셋 — turn_order 요소의 center/top 핀을 따른다.
618 const float ChipWidthPx = ChipWidth * 1920.0f;
619 const float ChipHeightPx = ChipHeight * 1080.0f;
620 const float GapPx = Gap * 1920.0f;
621 const float TopPx = Top * 1080.0f + mFoldTurnOrderDeltaY; // 폴드 변형이면 턴 스트립과 함께 내려간다.
622
623 // 칩 폭: 초상화가 있으면 원본 비율(가로/세로)로 높이에 맞춘 폭, 없으면 기존 고정폭 — 초상화가 눌리거나 늘어나지 않게.
624 TArray<float> ChipWidthsPx;
625 ChipWidthsPx.Reserve(Order.Num());
626 for (int32 SlotIndex = 0; SlotIndex < Order.Num(); ++SlotIndex)
627 {
628 float SlotWidthPx = ChipWidthPx;
629 for (const FUnitUI& Unit : Units)
630 {
631 if (Unit.mUnitId == Order[SlotIndex] && Unit.mPortrait != nullptr && Unit.mPortrait->GetSizeY() > 0)
632 {
633 SlotWidthPx = ChipHeightPx
634 * (StaticCast<float>(Unit.mPortrait->GetSizeX()) / StaticCast<float>(Unit.mPortrait->GetSizeY()));
635 break;
636 }
637 }
638 ChipWidthsPx.Add(SlotWidthPx);
639 }
640 float TotalPx = StaticCast<float>(Order.Num() - 1) * GapPx;
641 for (const float SlotWidthPx : ChipWidthsPx)
642 {
643 TotalPx += SlotWidthPx;
644 }
645 float ChipCursorX = -TotalPx * 0.5f; // 스킨 배치: 중앙 정렬 시작점에서 칩 폭만큼 전진.
646
647 int32 EnemyOrdinal = 0;
648 for (int32 SlotIndex = 0; SlotIndex < Order.Num(); ++SlotIndex)
649 {
650 const int32 Id = Order[SlotIndex];
651 const bool bPlayer = (Id == PlayerId);
652 const bool bCurrent = (Id == Turn.mCurrentUnitId);
653
654 UBorder* Chip = WidgetTree->ConstructWidget<UBorder>(UBorder::StaticClass());
655 UTextBlock* Text = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass());
656 if (Chip == nullptr || Text == nullptr)
657 {
658 continue;
659 }
660
661 FLinearColor BrushColor = bPlayer
662 ? FLinearColor(0.10f, 0.34f, 0.22f, 0.92f)
663 : FLinearColor(0.34f, 0.12f, 0.10f, 0.90f);
664 if (bCurrent)
665 {
666 // 현재 턴 강조(밝게).
667 BrushColor = FMath::Lerp(BrushColor, FLinearColor(1.0f, 0.92f, 0.50f, 1.0f), 0.45f);
668 }
669 Chip->SetBrushColor(BrushColor);
670 Chip->SetPadding(FMargin(2.0f, 1.0f));
671
672 // 초상화가 있으면 초상화 이미지, 없으면 "나"/"적N" 텍스트 폴백.
673 UTexture2D* Portrait = nullptr;
674 for (const FUnitUI& Unit : Units)
675 {
676 if (Unit.mUnitId == Id)
677 {
678 Portrait = Unit.mPortrait;
679 break;
680 }
681 }
682 if (bPlayer == false)
683 {
684 ++EnemyOrdinal; // 폴백 라벨 번호는 초상화 유무와 무관하게 순서를 유지한다.
685 }
686
687 if (Portrait != nullptr)
688 {
689 if (UImage* PortraitImage = WidgetTree->ConstructWidget<UImage>(UImage::StaticClass()))
690 {
691 PortraitImage->SetBrushFromTexture(Portrait, false);
692 PortraitImage->SetVisibility(ESlateVisibility::HitTestInvisible);
693 Chip->AddChild(PortraitImage);
694 }
695 }
696 else
697 {
698 // "나" / "적N" — 소스 인코딩에 흔들리지 않게 UTF-8 바이트를 명시 변환.
699 const FText Label = bPlayer
700 ? FText::FromString(UTF8_TO_TCHAR("\xEB\x82\x98"))
701 : FText::Format(FText::FromString(UTF8_TO_TCHAR("\xEC\xA0\x81{0}")), FText::AsNumber(EnemyOrdinal));
702 Text->SetJustification(ETextJustify::Center);
703 Text->SetText(Label);
704 Text->SetColorAndOpacity(FSlateColor(FLinearColor(0.98f, 1.0f, 0.94f, 1.0f)));
705 SetChipFontSize(Text, 13);
706 Chip->AddChild(Text);
707 }
708 Canvas->AddChildToCanvas(Chip);
709
710 if (bChipSkin)
711 {
712 FAnchorData ChipSlot;
713 ChipSlot.Anchors = FAnchors(0.5f, 0.0f, 0.5f, 0.0f);
714 ChipSlot.Alignment = FVector2D::ZeroVector;
715 ChipSlot.Offsets = FMargin(ChipCursorX, TopPx, ChipWidthsPx[SlotIndex], ChipHeightPx);
716 ChipCursorX += ChipWidthsPx[SlotIndex] + GapPx;
717 RDUILayout::ApplyDesignerSlotData(Chip, ChipSlot, 32);
718 }
719 else
720 {
721 const float Left = Start + StaticCast<float>(SlotIndex) * (ChipWidth + Gap);
722 RDUILayout::ApplyAnchoredSlot(Chip, FAnchors(Left, Top, Left + ChipWidth, Top + ChipHeight), 32);
723 }
724
725 mTurnOrderChips.Add(Chip);
726 mTurnOrderChipTexts.Add(Text);
727 }
728}
