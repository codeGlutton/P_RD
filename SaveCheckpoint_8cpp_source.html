20 bool ReadBank(const FString& Slot, TArray<uint8>& OutData, uint64& OutGeneration)
21 {
22 if (!UGameplayStatics::DoesSaveGameExist(Slot, 0)) return false;
23 TArray<uint8> Bytes;
24 if (!UGameplayStatics::LoadDataFromSlot(Bytes, Slot, 0)
25 || Bytes.Num() < HeaderSize || Bytes.Num() > HeaderSize + MaxPayloadSize)
26 {
27 return false;
28 }
29 FMemoryReader Reader(Bytes);
30 uint32 Magic = 0, Version = 0, Size = 0, Checksum = 0;
31 uint64 Generation = 0;
32 Reader << Magic << Version << Generation << Size << Checksum;
33 if (Reader.IsError() || Magic != SaveMagic || Version != SaveVersion
34 || Size == 0 || Size > MaxPayloadSize || Size != Bytes.Num() - HeaderSize
35 || FCrc::MemCrc32(Bytes.GetData() + HeaderSize, Size,
36 FCrc::MemCrc32(&Generation, sizeof(Generation))) != Checksum)
37 {
38 return false;
39 }
40 OutData.Reset(Size);
41 OutData.Append(Bytes.GetData() + HeaderSize, Size);
42 OutGeneration = Generation;
43 return true;
44 }

46 bool WriteSlot(const FString& Slot, const TArray<uint8>& Data)
47 {
48 if (Slot.IsEmpty() || Data.IsEmpty() || Data.Num() > MaxPayloadSize)
49 {
50 return false;
51 }
52 TArray<uint8> Existing;
53 uint64 GenerationA = 0, GenerationB = 0;
54 const bool bHasA = ReadBank(Slot + TEXT("_A"), Existing, GenerationA);
55 const bool bHasB = ReadBank(Slot + TEXT("_B"), Existing, GenerationB);
56 const FString Destination = Slot + ((!bHasA || (bHasB && GenerationA <= GenerationB))
57 ? TEXT("_A") : TEXT("_B"));
58 uint32 Magic = SaveMagic, Version = SaveVersion, Size = Data.Num();
59 uint64 Generation = FMath::Max(GenerationA, GenerationB) + 1;
60 uint32 Checksum = FCrc::MemCrc32(Data.GetData(), Data.Num(), FCrc::MemCrc32(&Generation, sizeof(Generation)));
61 TArray<uint8> Bytes;
62 FMemoryWriter Writer(Bytes);
63 Writer << Magic << Version << Generation << Size << Checksum;
64 Writer.Serialize(const_cast<uint8*>(Data.GetData()), Data.Num());
65 if (Writer.IsError() || !UGameplayStatics::SaveDataToSlot(Bytes, Destination, 0))
66 {
67 return false;
68 }
69 uint64 VerifiedGeneration = 0;
70 return ReadBank(Destination, Existing, VerifiedGeneration)
71 && VerifiedGeneration == Generation && Existing == Data;
72 }
73}

113bool RDCheckpoint::LoadSlot(const FString& Slot, TArray<uint8>& OutData)
114{
115 if (Slot.IsEmpty()) return false;
116 bool bFound = false;
117 SavePipe.Launch(UE_SOURCE_LOCATION, [&]()
118 {
119 TArray<uint8> A, B;
120 uint64 GenerationA = 0, GenerationB = 0;
121 const bool bHasA = ReadBank(Slot + TEXT("_A"), A, GenerationA);
122 const bool bHasB = ReadBank(Slot + TEXT("_B"), B, GenerationB);
123 bFound = bHasA || bHasB;
124 if (bFound) OutData = bHasA && (!bHasB || GenerationA >= GenerationB) ? MoveTemp(A) : MoveTemp(B);
125 }).Wait();
126 if (bFound) return true;
127 // Preserve installations made before checked, alternating slots were introduced.
128 if (!UGameplayStatics::DoesSaveGameExist(Slot, 0)) return false;
129 if (UBinarySaveGame* Legacy = Cast<UBinarySaveGame>(UGameplayStatics::LoadGameFromSlot(Slot, 0)))
130 {
131 OutData = Legacy->mData;
132 return !OutData.IsEmpty();
133 }
134 return false;
135}
