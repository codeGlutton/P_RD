52void UUnitModel::OnBeginRoom()

55
56 // 전투 시작 패시브 처리
57 TArray<UTacticalPassive*> Passives = mPassiveCompModel->GetPassivesByTiming(AbilityTags::GameplayAbility_Passive_OnStartRoom);
58
59 FBoardCombatTargetSnapshotData UnitSnapshot = MakeSnapshotData();
60
61 FPassiveActivateContext PassiveContext;
62 PassiveContext.mOwner = this;
63 PassiveContext.mOwnerSnapshot = &UnitSnapshot;
64 PassiveContext.mTargets.Add(this);
65 PassiveContext.mTargetSnapshots.Add(&UnitSnapshot);
66
67 for (UTacticalPassive*& Passive : Passives)
68 {
69 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;
70 Passive->ActivatePassive(AbilityTags::GameplayAbility_Passive_OnStartRoom, PassiveContext, OUT DynamicPassiveData);

72 }
