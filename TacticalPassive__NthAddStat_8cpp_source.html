14bool UTacticalPassive_NthAddStat::EvaluateActivate(
15 const FPassiveActivateContext& Ctx,
16 TInstancedStruct<FDynamicPassiveData>& PassiveState,
17 float& OutMagnitude)
18{
19 // 데이터/이펙트가 없으면 적용 안 함
20 if (mStaticData == nullptr || mEffectClass == nullptr)
21 {
22 return false;
23 }
24
25 // 내부상태 복사본이 없으면 복사하거나 새로 생성 (복사본을 리턴해야 하니까)
26 if (!PassiveState.IsValid())
27 {
28 if (mState.IsValid())
29 {
30 PassiveState = mState;
31 }
32 else
33 {
34 PassiveState.InitializeAs<FDynamicPassiveData_NthCounter>();
35 }
36 }
37
38 // 내부상태 복사본을 구체화
39 FDynamicPassiveData_NthCounter& Running = PassiveState.GetMutable<FDynamicPassiveData_NthCounter>();
40
41 // 완료 카운트를 읽고, 현재 카운트는 +1 해서 사용
42 const int32 CompletedCount = Running.mCount;
43 const int32 Ordinal = CompletedCount + 1;
44
45 // 이번 카운트가 임계치에 도달했으면 이펙트 발동!
46 const bool bTriggered = (Ordinal >= mStaticData->mThreshold);
47 if (bTriggered)
48 {
49 // DA의 매그니튜드 값을 그대로 사용
50 OutMagnitude = mStaticData->mMagnitude;
51 }
52
53 // 다음 내부상태 기록: 발동했으면 리셋해야 하니까 0, 아니면 증가
54 Running.mCount = bTriggered ? 0 : Ordinal;
55
56 // 이번 카운트가 임계치에 도달했으면 적용, 아니면 스킵 회신
57 return bTriggered;
58}
