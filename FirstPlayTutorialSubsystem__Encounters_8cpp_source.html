16bool ProjectTrap(UFirstPlayTutorialSubsystem* Context, const UGimmickModel* Trap,
17 UCombatLayoutHUDWidget* HUD, TArray<FVector2D>& Corners)
18{
19 Corners.Reset();
20 const AActor* View = Trap ? Trap->GetView<AActor>() : nullptr;
21 auto* PC = Context->GetWorld()->GetFirstPlayerController();
22 if (!View || View->IsHidden() || !PC || Trap->IsDead()) return false;
23 FVector Center, Extent;
24 View->GetActorBounds(false, Center, Extent);
25 if (Extent.IsNearlyZero()) return false;
26 FVector2D Min(DBL_MAX, DBL_MAX), Max(-DBL_MAX, -DBL_MAX);
27 int32 Width = 0, Height = 0;
28 PC->GetViewportSize(Width, Height);
29 for (int32 I = 0; I < 8; ++I)
30 {
31 FVector2D Pixel;
32 const FVector Point = Center + Extent * FVector(I & 1 ? 1 : -1, I & 2 ? 1 : -1, I & 4 ? 1 : -1);
33 if (!PC->ProjectWorldLocationToScreen(Point, Pixel)) return false;
34 Min.X = FMath::Min(Min.X, Pixel.X); Min.Y = FMath::Min(Min.Y, Pixel.Y);
35 Max.X = FMath::Max(Max.X, Pixel.X); Max.Y = FMath::Max(Max.Y, Pixel.Y);
36 }
37 // Never block controls for a target outside the visible playfield.
38 if (Min.X < 0 || Min.Y < 0 || Max.X > Width || Max.Y > Height) return false;
39 FVector2D AbsoluteCenter;
40 USlateBlueprintLibrary::ScreenToWidgetAbsolute(Context, (Min + Max) * .5, AbsoluteCenter);
41 if (!HUD->IsGuidedBoardInputAt(AbsoluteCenter)) return false;
42 for (const auto Pixel : {Min, FVector2D(Max.X, Min.Y), Max, FVector2D(Min.X, Max.Y)})
43 {
44 FVector2D Absolute;
45 USlateBlueprintLibrary::ScreenToWidgetAbsolute(Context, Pixel, Absolute);
46 Corners.Add(Absolute);
47 }
48 return true;
49}

79bool UFirstPlayTutorialSubsystem::UpdateEncounterHint(UCombatLayoutHUDWidget* HUD)
80{
81 const auto* Data = GetUserMutableData();
82 const auto& Flow = Data->GuidedTutorial;
83 const bool BasicGuideActive = Flow.Enrolled && !Data->TutorialProgress.Skipped &&
84 (Flow.CanGuideCombat() || Flow.SystemsActive || Flow.Lesson != EGuidedLesson::None);
85 if (!HUD || !bInCombat || !bPlayerTurn || bExecutingAction || BasicGuideActive ||
86 !HUD->CanShowEncounterHint())
87 {
88 HideEncounterHint();
89 return false;
90 }
91 TArray<FVector2D> Corners;
92 if (EncounterWidget)
93 {
94 if (!ProjectTrap(this, EncounterTarget.Get(), HUD, Corners))
95 {
96 HideEncounterHint();
97 return false;
98 }
99 EncounterWidget->SetWorldFocus(Corners);
100 return true;
101 }
102 if (FPlatformTime::Seconds() < EncounterReadyAt) return false;
103 const auto* Combat = GetWorldSubsystemModel<USRPGCombatModel>(GetWorld());
104 if (!Combat) return false;
105 for (const auto& Obstacle : Combat->GetObstacles())
106 {
107 auto* Trap = Cast<UGimmickModel>(Obstacle);
108 if (!Trap || Data->SeenTrapHints.Contains(Trap->GetStaticSpawnDataId())) continue;
109 const auto* Spawn = Cast<UStaticGimmickSpawnData>(Trap->GetStaticSpawnData());
110 // An unconfigured future gimmick must not show an empty, blocking notice.
111 if (!Spawn || Spawn->mDisplayName.IsEmpty() || Spawn->mDescription.IsEmpty() ||
112 !ProjectTrap(this, Trap, HUD, Corners)) continue;
113 EncounterWidget = CreateWidget<UGuidedTutorialWidget>(GetGameInstance());
114 if (!EncounterWidget) return false;
115 EncounterTarget = Trap;
116 HUD->SetEncounterPresentationActive(true);
117 EncounterWidget->PresentEncounter(Spawn->mDisplayName, Spawn->mDescription);
118 EncounterWidget->SetWorldFocus(Corners);
119 EncounterWidget->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
120 EncounterWidget->AddToViewport(75);
121 EncounterWidget->GetContinueButton()->OnClicked.AddUniqueDynamic(this, &UFirstPlayTutorialSubsystem::EncounterConfirmed);
122 UE_LOG(LogTemp, Display, TEXT("Trap hint shown: %s"), *Trap->GetStaticSpawnDataId().ToString());
123 return true;
124 }
125 return false;
126}
