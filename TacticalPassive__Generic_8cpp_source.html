34void UTacticalPassive_Generic::OnActivate(const FPassiveActivateContext& InCtx, TInstancedStruct<FDynamicPassiveData>& PassiveState)
35{
36 // 데이터나 상태가 없으면 발동할 수 없음
37 if (mStaticData == nullptr)
38 {
39 return;
40 }
41 FDynamicPassiveData_Generic* State = PassiveState.GetMutablePtr<FDynamicPassiveData_Generic>();
42 if (State == nullptr)
43 {
44 return;
45 }
46
47 // 캡처한 타겟에게 발동하는 패시브면, 모아 둔 타겟을 대상으로 삼고 목록은 비움
48 FPassiveActivateContext CapturedCtx;
49 if (mStaticData->mActivateOnCapturedTargets)
50 {
51 CapturedCtx = InCtx;
52 CapturedCtx.mTargets = State->mCapturedTargets;
53 CapturedCtx.mTargetSnapshots.Reset();
54
55 // 캡처값과 비교할 수 있도록 현재값 저장. 사라진 타겟은 nullptr
56 for (const TWeakObjectPtr<UBoardActorModel>& Target : CapturedCtx.mTargets)
57 {
58 const IBoardCombatTarget* CombatTarget = Cast<IBoardCombatTarget>(Target.Get());
59 CapturedCtx.mTargetSnapshots.Add(CombatTarget != nullptr ? CombatTarget->MakeSnapshotData() : nullptr);
60 }
61 State->mCapturedTargets.Reset();
62 }
63
64 // 캡처한 타겟을 대상으로 하면 CapturedCtx를, 아니면 기본값인 InCtx를 사용
65 const FPassiveActivateContext& Ctx = mStaticData->mActivateOnCapturedTargets ? CapturedCtx : InCtx;
66
67 // 카운터는 트리거 횟수이므로 조건 통과 여부와 무관하게 증가
68 State->mCounter += 1;
69
70 UE_LOG(LogPassive, Verbose, TEXT("발동 시점 진입: %s (소유자 %s, 카운터 %d)"), *GetNameSafe(mStaticData), *GetNameSafe(Ctx.mOwner.Get()), State->mCounter);
71
72 // 수량 조건 게이트
73 if (PassesTargetQuantifier(Ctx, PassiveState) == false)
74 {
75 UE_LOG(LogPassive, Verbose, TEXT("수량 조건 탈락: %s"), *GetNameSafe(mStaticData));
76 return;
77 }
78
79 // 효과 수치를 먼저 전부 계산해 실제 적용할 효과만 추림
80 struct FResolvedEffect
81 {
82 TSubclassOf<UTacticalEffect> mEffectClass;
83 float mMagnitude = 0.f;
84 };
85 TArray<FResolvedEffect> ResolvedEffects;
86 for (const FPassiveEffectEntry& Entry : mStaticData->mEffects)
87 {
88 float Magnitude = 0.f;
89 if (Entry.mMagnitude.Resolve(Ctx, 0, *State, Magnitude) == false)
90 {
91 UE_LOG(LogPassive, Warning, TEXT("패시브 효과 수치 계산 실패: %s"), *GetNameSafe(mStaticData));
92 continue;
93 }
94 if (FMath::IsNearlyZero(Magnitude))
95 {
96 continue;
97 }
98 TSubclassOf<UTacticalEffect> EffectClass = Entry.mEffectClass.LoadSynchronous();
99 if (EffectClass == nullptr)
100 {
101 continue;
102 }
103 ResolvedEffects.Add({ EffectClass, Magnitude });
104 }
105
106 // 적용할 효과가 없으면 이전 배치도 그대로 유지
107 if (ResolvedEffects.Num() == 0)
108 {
109 UE_LOG(LogPassive, Verbose, TEXT("적용할 효과 없음: %s"), *GetNameSafe(mStaticData));
110 return;
111 }
112
113 // 재발동은 갱신이므로 이전 배치를 먼저 제거
114 DeactivatePassive();
115
116 // 효과마다 이펙트 적용
117 for (const FResolvedEffect& Resolved : ResolvedEffects)
118 {
119 NotifyPassive(Ctx, Resolved.mEffectClass, Resolved.mMagnitude, mStaticData->mEffectTarget, false);
120 }
121}

123bool UTacticalPassive_Generic::IsTargetQualified(const FPassiveActivateContext& Ctx, int32 TargetIndex, const TInstancedStruct<FDynamicPassiveData>& State) const
124{
125 // 데이터가 없으면 조건도 없으므로 통과
126 if (mStaticData == nullptr)
127 {
128 return true;
129 }
130
131 // 상태가 아직 없으면 빈 상태로 판정
132 static const FDynamicPassiveData_Generic EmptyState;
133 const FDynamicPassiveData_Generic* Generic = State.GetPtr<FDynamicPassiveData_Generic>();
134
135 // DA에 정의된 조건으로 자격 판정
136 return PassiveConditionUtils::EvaluateAll(mStaticData->mConditions, Ctx, TargetIndex, Generic != nullptr ? *Generic : EmptyState);
137}

139void UTacticalPassive_Generic::OnCapture(const FPassiveActivateContext& Ctx, TInstancedStruct<FDynamicPassiveData>& PassiveState)
140{
141 // 데이터나 상태가 없으면 캡처할 수 없음
142 if (mStaticData == nullptr)
143 {
144 return;
145 }
146 FDynamicPassiveData_Generic* State = PassiveState.GetMutablePtr<FDynamicPassiveData_Generic>();
147 if (State == nullptr)
148 {
149 return;
150 }
151
152 // 캡처한 타겟에게 발동하는 패시브는, 페이즈마다 캡처되는 타겟을 발동 전까지 모음
153 const bool bAccumulate = mStaticData->mActivateOnCapturedTargets;
154 const bool bFirstCapture = (bAccumulate == false) || State->mCapturedTargets.Num() == 0;
155
156 // 첫 캡처: 키마다 Self 값을 새로 저장. 타겟 값은 아래에서 뒤에 붙임
157 // Self와 타겟 값을 모두 저장해서, 읽는 쪽에서 선택해서 사용하게 함
158 if (bFirstCapture)
159 {
160 for (const FPassiveCaptureEntry& Entry : mStaticData->mCaptureOperands)
161 {
162 // 출처를 Self로 바꿔 평가하기 위한 복사본
163 FPassiveOperand Operand = Entry.mOperand;
164 Operand.mSource = EPassiveOperandSource::Self;
165
166 // 실패하면 키 자체를 지움 -> 발동 시 Captured 조회 실패: 조건 실패
167 FPassiveCaptureSlot Slot;
168 if (Operand.Resolve(Ctx, INDEX_NONE, *State, Slot.mSelf))
169 {
170 State->mCaptures.Add(Entry.mKey, Slot);
171 }
172 else
173 {
174 State->mCaptures.Remove(Entry.mKey);
175 UE_LOG(LogPassive, Warning, TEXT("패시브 캡처 실패: %s (키 %s)"), *GetNameSafe(mStaticData), *Entry.mKey.ToString());
176 }
177 }
178
179 // 타일 위치는 항상 저장 (이동 거리 판정용). 스냅샷 없으면 Invalid
180 State->mCapturedSelfTile = (Ctx.mOwnerSnapshot != nullptr) ? Ctx.mOwnerSnapshot->mTileTransform.mIndex : FTileIndex::Invalid;
181 State->mCapturedTargetTiles.Reset();
182 State->mCapturedTargets.Reset();
183 }
184
185 // 타겟마다 캡처값과 타일을 뒤에 붙임. 이미 수집된 타겟은 처음 캡처값 유지
186 for (int32 Index = 0; Index < Ctx.mTargets.Num(); ++Index)
187 {
188 const TWeakObjectPtr<UBoardActorModel>& Target = Ctx.mTargets[Index];
189 if (bAccumulate && State->mCapturedTargets.Contains(Target))
190 {
191 continue;
192 }
193
194 // 키마다 타겟 값 추가. 오류로 계산에 실패하면 키 자체를 지워서 조건 실패하게 함
195 for (const FPassiveCaptureEntry& Entry : mStaticData->mCaptureOperands)
196 {
197 FPassiveCaptureSlot* Slot = State->mCaptures.Find(Entry.mKey);
198 if (Slot == nullptr)
199 {
200 continue;
201 }
202
203 FPassiveOperand Operand = Entry.mOperand;
204 Operand.mSource = EPassiveOperandSource::Target;
205
206 float Value = 0.f;
207 if (Operand.Resolve(Ctx, Index, *State, Value))
208 {
209 Slot->mTargets.Add(Value);
210 }
211 else
212 {
213 State->mCaptures.Remove(Entry.mKey);
214 UE_LOG(LogPassive, Warning, TEXT("패시브 캡처 실패: %s (키 %s)"), *GetNameSafe(mStaticData), *Entry.mKey.ToString());
215 }
216 }
217
218 // 타일 위치는 항상 저장 (이동 거리 판정용). 스냅샷 없으면 Invalid
219 const UBoardCombatTargetSnapshotData* Snapshot = Ctx.mTargetSnapshots.IsValidIndex(Index) ? Ctx.mTargetSnapshots[Index] : nullptr;
220 State->mCapturedTargetTiles.Add(Snapshot != nullptr ? Snapshot->mTileTransform.mIndex : FTileIndex::Invalid);
221 State->mCapturedTargets.Add(Target);
222 }
223}
