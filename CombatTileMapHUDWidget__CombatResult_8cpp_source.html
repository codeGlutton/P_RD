47void UCombatTileMapHUDWidget::StartCombatResultCinematic()
48{
49 // 결과 징글: 승리/패배에 맞는 짧은 음악을 결과 연출 시작과 동시에 1회 재생한다.
50 USoundBase* ResultJingle = mCombatResult == ESRPGCombatResult::PlayerWin ? mVictoryJingleSound.Get() : mDefeatJingleSound.Get();

71 {

83
84 if (mCombatResultOverlayWidget == nullptr)

87 }
88

95 mCombatRewardUIModel->OnRewardClaimRequested.RemoveDynamic(this, &UCombatTileMapHUDWidget::HandleCombatRewardClaimRequested);

98
99 if (mCombatRewardWidget == nullptr && mRewardWidgetClass != nullptr)
100 {
101 mCombatRewardWidget = CreateWidget<URewardUIWidgetBase>(GetOwningPlayer(), mRewardWidgetClass);
102 }
103

105 {
106 mCombatRewardWidget->OnClosed.RemoveDynamic(this, &UCombatTileMapHUDWidget::HandleCombatResultRewardConfirmed);
107 mCombatRewardWidget->OnClosed.AddUniqueDynamic(this, &UCombatTileMapHUDWidget::HandleCombatResultRewardConfirmed);
108 mCombatRewardWidget->BindUIModel(mCombatRewardUIModel);

112void UCombatTileMapHUDWidget::HandleCombatResultVideoFinished(UCinematicWidget* CinematicWidget)
113{
114 if (mCombatResultFlowActive == false)
115 {
116 return;
117 }
118
119 EnsureCombatResultWidgets();
120 if (mCombatResultOverlayWidget == nullptr)
121 {
122 SetCombatResultViewActive(false, true);
123 mCombatResultBarrier.Reset();
124 return;
125 }
126
127 if (mCombatResult == ESRPGCombatResult::PlayerWin)
128 {
129 FRewardUI RewardUI;
130 TArray<FRewardChoiceUI> RewardChoices;
131 if (ACombatGameMode* CombatGameMode = GetWorld() != nullptr ? GetWorld()->GetAuthGameMode<ACombatGameMode>() : nullptr)
132 {
133 RewardUI = CombatGameMode->MakeCombatRewardUI();
134 RewardChoices = CombatGameMode->MakeCombatRewardChoicesUI();
135 }
136
137 if (mCombatRewardWidget != nullptr && mCombatRewardUIModel != nullptr)
138 {
139 mCombatRewardWidget->BindUIModel(mCombatRewardUIModel);
140 mCombatRewardUIModel->SetReward(RewardUI);
141 mCombatRewardUIModel->SetRewardChoices(RewardChoices);
142 mCombatRewardWidget->OpenUI();
143 return;
144 }
145
146 mCombatResultOverlayWidget->ShowVictoryReward(
147 RewardUI,
148 FSimpleDelegate::CreateUObject(this, &UCombatTileMapHUDWidget::HandleCombatResultRewardConfirmed));
149 mCombatResultOverlayWidget->OpenUI();
150 return;
151 }
152
153 mCombatResultOverlayWidget->ShowDefeatContinue(
154 FSimpleDelegate::CreateUObject(this, &UCombatTileMapHUDWidget::HandleCombatResultContinueConfirmed));
155 mCombatResultOverlayWidget->OpenUI();
156}

257void UCombatTileMapHUDWidget::SetRoomNameClusterVisible(bool bVisible)
258{
259 if (WidgetTree == nullptr)
260 {
261 return;
262 }
263
264 static const FName RoomNameWidgets[] = {
265 FName(TEXT("R_room_name_room_name_7")),
266 FName(TEXT("HUD_M_room_name_text"))
267 };
268
269 for (const FName WidgetName : RoomNameWidgets)
270 {
271 UWidget* Widget = WidgetTree->FindWidget(WidgetName);
272 if (Widget == nullptr)
273 {
274 continue;
275 }
276
277 if (bVisible)
278 {
279 if (ESlateVisibility* PreviousVisibility = mCombatResultRoomNameVisibilities.Find(WidgetName))
280 {
281 Widget->SetVisibility(*PreviousVisibility);
282 }
283 else
284 {
285 Widget->SetVisibility(ESlateVisibility::HitTestInvisible);
286 }
287 continue;
288 }
289
290 if (mCombatResultRoomNameVisibilities.Contains(WidgetName) == false)
291 {
292 mCombatResultRoomNameVisibilities.Add(WidgetName, Widget->GetVisibility());
293 }
294 Widget->SetVisibility(ESlateVisibility::Collapsed);
295 }
296
297 if (bVisible)
298 {
299 mCombatResultRoomNameVisibilities.Reset();
300 }
301}

303FString UCombatTileMapHUDWidget::GetCombatResultVideoPath(ESRPGCombatResult Result) const
304{
305 const UGamePlaySettings* GamePlaySettings = GetDefault<UGamePlaySettings>();
306 if (Result == ESRPGCombatResult::PlayerWin)
307 {
308 return GamePlaySettings != nullptr && GamePlaySettings->mCombatVictoryVideoPath.IsEmpty() == false
309 ? GamePlaySettings->mCombatVictoryVideoPath
310 : FString(FallbackVictoryVideoPath);
311 }
312
313 return GamePlaySettings != nullptr && GamePlaySettings->mCombatDefeatVideoPath.IsEmpty() == false
314 ? GamePlaySettings->mCombatDefeatVideoPath
315 : FString(FallbackDefeatVideoPath);
316}
