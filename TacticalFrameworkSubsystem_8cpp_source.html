76void UTacticalFrameworkSubsystem::SpawnGlobalInfiniteVFXEffect(const FTacticalEffectSpec& Spec, FActiveTacticalEffectHandle ActiveHandle, const UAttributeSetComponentModel* Model)
77{
78 const UGamePlaySettings* GamePlaySettings = GetDefault<UGamePlaySettings>();
79 checkf(GamePlaySettings != nullptr, TEXT("게임 플레이 세팅 nullptr 오류"));
80
81 /* VFX 생성 */
82
83 const FSoftVFXSpawnData* VFXSpawnData = GamePlaySettings->mGlobalStatusEffectVFXSetting.mInfiniteEffectVFXs.Find(Spec.mEffectClass->GetClass());
84
85 const UActorModel* Instigator = Model->GetOwnerModel();
86 const AActor* ActorView = Instigator->GetView<AActor>();
87 if (ActorView != nullptr && VFXSpawnData != nullptr)
88 {
89 const IBoardCombatTargetView* CombatTargetView = Instigator->GetView<IBoardCombatTargetView>();
90 TArray<TObjectPtr<UNiagaraComponent>> SpawnedComponents;
91 if (CombatTargetView != nullptr)
92 {
93 UPrimitiveComponent* TargetMeshComp = CombatTargetView->GetTargetMeshComponent();
94 const FVFXTimelineEventTarget EventTarget = UVFXFunctionLibrary::MakeTimelineEventTarget(TargetMeshComp);
95
96 SpawnedComponents = UVFXFunctionLibrary::SpawnAndExecuteVFX(*VFXSpawnData, TargetMeshComp, CombatTargetView->GetCombatTargetVFXTimelineComponent(), EventTarget);
97 }
98 else
99 {
100 SpawnedComponents.Reserve(VFXSpawnData->mNiagaraSpawnDatas.Num());
101 for (const FSoftNiagaraSpawnData& NiagaraSpawnData : VFXSpawnData->mNiagaraSpawnDatas)
102 {
103 SpawnedComponents.Add(UVFXFunctionLibrary::SpawnNiagaraEffect(NiagaraSpawnData, ActorView));
104 }
105 }
106
107 FSpawnedNiagaraComponentList& NiagaraComponentList = mSpawnedVFXComponents.FindOrAdd(ActiveHandle);
108 for (const TObjectPtr<UNiagaraComponent>& SpawnedComponent : SpawnedComponents)
109 {
110 NiagaraComponentList.mComponents.Add(SpawnedComponent);
111 }
112 }
