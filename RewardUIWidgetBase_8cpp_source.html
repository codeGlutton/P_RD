194void URewardUIWidgetBase::RefreshRows()
195{
196 mRewardClaimRows.Reset();
197 mRewardRowWidgets.Reset();
198
199 if (mUIModel == nullptr || mRewardRowsBox == nullptr)
200 {
201 UpdateCloseButtonVisibility();
202 return;
203 }
204
205 mRewardRowsBox->ClearChildren();
206
207 auto AddRow = [this](ERewardClaimKind ClaimKind, int32 ChoiceIndex, const FText& MainText, const FText& SubText, UTexture2D* IconTexture)
208 {
209 if (mRewardRowWidgetClass == nullptr)
210 {
211 mRewardRowWidgetClass = LoadClass<URewardRowWidgetBase>(nullptr, TEXT("/Game/BP/UI/WBP_RewardRow.WBP_RewardRow_C"));
212 }
213 if (mRewardRowWidgetClass == nullptr)
214 {
215 UE_LOG(LogRD, Warning, TEXT("WBP_RewardRow 클래스를 찾지 못해 보상 행을 생성하지 못했습니다."));
216 return;
217 }
218
219 URewardRowWidgetBase* RowWidget = CreateWidget<URewardRowWidgetBase>(GetWorld(), mRewardRowWidgetClass);
220 if (RowWidget == nullptr)
221 {
222 return;
223 }
224
225 const int32 RewardRowIndex = mRewardClaimRows.Add(FRewardClaimRow{ ClaimKind, ChoiceIndex, false });
226
227 RowWidget->SetRewardRow(MainText, SubText, IconTexture);
228 RowWidget->SetRewardIndex(RewardRowIndex);
229 RowWidget->OnRewardRowClicked.AddUniqueDynamic(this, &URewardUIWidgetBase::HandleRewardRowClicked);
230 mRewardRowWidgets.Add(RowWidget);
231
232 if (UVerticalBoxSlot* RowSlot = mRewardRowsBox->AddChildToVerticalBox(RowWidget))
233 {
234 RowSlot->SetHorizontalAlignment(HAlign_Center);
235 RowSlot->SetVerticalAlignment(VAlign_Center);
236 RowSlot->SetPadding(mRewardRowPadding);
237 }
238 };
239
240 const FRewardUI& Reward = mUIModel->GetReward();
241 UTexture2D* SummaryIcon = mGoldIcon != nullptr ? mGoldIcon.Get() : mRewardGoldIconTexture.Get();
242
243 if (Reward.mGoldGained != 0)
244 {
245 AddRow(
246 ERewardClaimKind::Gold,
247 INDEX_NONE,
248 FText::Format(LOCTEXT("GoldRewardValue", "골드 +{0}"), FText::AsNumber(Reward.mGoldGained)),
249 FText::GetEmpty(),
250 SummaryIcon);
251 }
252 if (Reward.mExpGained != 0)
253 {
254 // 경험치 행은 전용 아이콘(없으면 골드 아이콘 폴백).
255 UTexture2D* ExpIcon = mRewardExpIconTexture != nullptr ? mRewardExpIconTexture.Get() : SummaryIcon;
256 AddRow(
257 ERewardClaimKind::Exp,
258 INDEX_NONE,
259 FText::Format(LOCTEXT("ExpRewardValue", "경험치 +{0}"), FText::AsNumber(Reward.mExpGained)),
260 FText::GetEmpty(),
261 ExpIcon);
262 }
263
264 const TArray<FRewardChoiceUI>& Items = mUIModel->GetRewardChoices();
265 for (const FRewardChoiceUI& Item : Items)
266 {
267 UTexture2D* IconTexture = Item.mIcon != nullptr ? Item.mIcon.Get() : GetRewardIcon(Item.mKind);
268 AddRow(ERewardClaimKind::Choice, Item.mChoiceIndex, MakeRewardChoiceText(Item), Item.mDescription, IconTexture);
269 }
270
271 UpdateCloseButtonVisibility();
272}

274void URewardUIWidgetBase::HandleRewardRowClicked(int32 RewardRowIndex)
275{
276 if (mRewardClaimRows.IsValidIndex(RewardRowIndex) == false || mRewardClaimRows[RewardRowIndex].mClaimed)
277 {
278 return;
279 }
280
281 // 지급 의도만 게임플레이로 보낸다. 실제 지급 성공 여부와 행 제거는 게임플레이 확정(NotifyRewardClaimed)이 결정한다.
282 // RequestClaimReward는 동기적으로 게임플레이를 거쳐 NotifyRewardClaimed까지 재진입할 수 있으므로,
283 // 값을 복사해 호출하고 이후 이 행 인덱스를 다시 건드리지 않는다(재진입 중 배열 상태 변경 대비).
284 const FRewardClaimRow ClaimRow = mRewardClaimRows[RewardRowIndex];
285 if (mUIModel != nullptr)
286 {
287 mUIModel->RequestClaimReward(ClaimRow.mKind, ClaimRow.mChoiceIndex);
288 }
289}

292void URewardUIWidgetBase::NotifyRewardClaimed(ERewardClaimKind ClaimKind, int32 ChoiceIndex)
293{
294 for (int32 RowIndex = 0; RowIndex < mRewardClaimRows.Num(); ++RowIndex)
295 {
296 FRewardClaimRow& ClaimRow = mRewardClaimRows[RowIndex];
297 if (ClaimRow.mClaimed || ClaimRow.mKind != ClaimKind || ClaimRow.mChoiceIndex != ChoiceIndex)
298 {
299 continue;
300 }
301
302 // 행 인덱스(=위젯 클릭 시 넘어오는 값) 정렬을 유지하려고 배열은 줄이지 않고, 표시 위젯만 박스에서 제거한다.
303 ClaimRow.mClaimed = true;
304 if (mRewardRowWidgets.IsValidIndex(RowIndex) && mRewardRowWidgets[RowIndex] != nullptr)
305 {
306 mRewardRowWidgets[RowIndex]->RemoveFromParent();
307 }
308 break;
309 }
310
311 UpdateCloseButtonVisibility();
312}
