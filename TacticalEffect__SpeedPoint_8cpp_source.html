37void UTacticalEffectExecutionCalculation_GetSpeedPoint::Execute(const FTacticalEffectCustomExecutionParameters& ExecutionParams, FTacticalEffectCustomExecutionOutput& OutExecutionOutput) const
38{
39 Super::Execute(ExecutionParams, OutExecutionOutput);
40
41 const UGameBalanceSettings* GameBalanceSettings = GetDefault<UGameBalanceSettings>();
42 checkf(GameBalanceSettings != nullptr, TEXT("게임 밸런스 세팅 nullptr"));
43
44 UBoardCombatTargetSnapshotData* SourceSnapshotData = ExecutionParams.GetOwningSpec().GetInstigatorSnapshotData();
45 checkf(SourceSnapshotData != nullptr, TEXT("소스 스냅샷 nullptr"));
46
47 UBoardCombatTargetSnapshotData* TargetSnapshotData = ExecutionParams.GetOwningSpec().GetTargetSnapshotData();
48 checkf(TargetSnapshotData != nullptr, TEXT("타겟 스냅샷 nullptr"));
49
50 // 신속
51 const bool IsTargetHaste = TargetSnapshotData->mEffectCounts.Contains(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Buff_Haste);
52 const float TargetHasteRatio = IsTargetHaste == true ? GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_RoundDuration_Buff_Haste] : 1.f;
53
54 // 둔화
55 const bool IsTargetSlow = TargetSnapshotData->mEffectCounts.Contains(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Slow);
56 const float TargetSlowRatio = IsTargetSlow == true ? GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Slow] : 1.f;
57
58 const float TotalSpeed =
59 ExecutionParams.GetOwningSpec().GetStackCount() *
60 ExecutionParams.GetOwningSpec().mDynamicMagnitude *
61 TargetHasteRatio *
62 TargetSlowRatio *
63 SourceSnapshotData->mAttributes[UUnitAttributeSet::GetSpeedPointFactorAttribute()];
64 float SpeedDiff = FMath::Floor(TotalSpeed);
65
66 OnPreGetSpeedPoint(ExecutionParams, OutExecutionOutput, OUT SpeedDiff);
67 if (SpeedDiff > 0.f)
68 {
69 /* 속도 포인트 증가 적용 */
70 OutExecutionOutput.AddOutputModifier(FTacticalModifierEvaluatedData(UUnitAttributeSet::GetSpeedPointAttribute(), ETacticalModOp::AddBase, SpeedDiff));
71 OnPostGetSpeedPoint(ExecutionParams, OutExecutionOutput, SpeedDiff);
72
73 /* 로그 적용 */
74 FSRPGAttributeEffectEventLog Log;
75 Log.mEffectAttribute = UUnitAttributeSet::GetSpeedPointAttribute();
76 Log.mMagnitude = SpeedDiff;
77
78 UAttributeSetComponentModel* TargetAttributeSetCompModel = ExecutionParams.GetTargetAttributeSetComponentModel();
79 const UActorModel* Target = TargetAttributeSetCompModel->GetOwnerModel();
80 GetWorldEventLogger(Target)->LogAttributeEffect(Target->GetModelId(), Target->GetClass(), Log);
81 }
82
83 OutExecutionOutput.MarkDynamicMagnitudeHandledManually();
84 OutExecutionOutput.MarkStackCountHandledManually();
85}

106bool UTacticalEffect_GetSpeedPoint::CanApply(const FActiveTacticalEffectsContainer& ActiveTEContainer, const FTacticalEffectSpec& TESpec) const
107{
108 if (Super::CanApply(ActiveTEContainer, TESpec) == false)
109 {
110 return false;
111 }
112
113 const UBoardCombatTargetSnapshotData* SourceSnapshotData = TESpec.GetInstigatorSnapshotData();
114 const UBoardCombatTargetSnapshotData* TargetSnapshotData = TESpec.GetTargetSnapshotData();
115 if (SourceSnapshotData == nullptr || TargetSnapshotData == nullptr)
116 {
117 return false;
118 }
119
120 if (SourceSnapshotData->mAttributes[UUnitAttributeSet::GetSpeedPointFactorAttribute()] * TESpec.mDynamicMagnitude <= 0)
121 {
122 return false;
123 }
124
125 return true;
126}
