42 // 타일 간격 (cm)

45 TArray<FVector> OutPoints;
46 TArray<float> OutDistances;

49 /* [케이스 1] 컷 비율 0: 폴리라인이 원본 경로와 완전히 같아야 함 (직각 이동과 동일) */
50 {
51 const TArray<FVector> Path = {
52 FVector(0.0, 0.0, 0.0),
53 FVector(TileSize, 0.0, 0.0),
54 FVector(TileSize, TileSize, 0.0) };
55 UBoardMovementPresentationComponent::BakePolyLinePoints(Path, 0.0f, 1.0f, OutPoints, OutDistances, OutMarkers);

63 TestEqual(TEXT("컷0: 마커 개수 = 경로 개수"), OutMarkers.Num(), Path.Num());
64 }

67 {

70 FVector(TileSize, 0.0, 0.0),

72 UBoardMovementPresentationComponent::BakePolyLinePoints(Path, 0.35f, 1.0f, OutPoints, OutDistances, OutMarkers);

78 }
79 }

81 /* [케이스 3] L자 코너 (컷 비율 0.25): 정확 지점 + 성질 검증 */
82 {
83 // A(0,0) -> B(300,0) -> C(300,300). B에서 직각으로 꺾임
84 const TArray<FVector> Path = {
85 FVector(0.0, 0.0, 0.0),
86 FVector(TileSize, 0.0, 0.0),
87 FVector(TileSize, TileSize, 0.0) };
88 constexpr float CutRatio = 0.25f;
89 UBoardMovementPresentationComponent::BakePolyLinePoints(Path, CutRatio, 1.0f, OutPoints, OutDistances, OutMarkers);
90
91 // 컷 거리 = 300 x 0.25 = 75
92 const FVector CurveStart(TileSize - TileSize * CutRatio, 0.0, 0.0); // P1 = (225, 0)
93 const FVector CurveEnd(TileSize, TileSize * CutRatio, 0.0); // P2 = (300, 75)
94 const FVector CornerPoint = Path[1]; // C(컨트롤 포인트) = (300, 0)
95
96 /* 정확 비교 */
97
98 // 시작/끝점은 타일 중점과 정확히 일치
99 TestTrue(TEXT("L자: 시작점 = 시작 타일 중점"), OutPoints[0].Equals(Path[0], Tolerance));
100 TestTrue(TEXT("L자: 끝점 = 도착 타일 중점"), OutPoints.Last().Equals(Path.Last(), Tolerance));
101
102 // 커브 시작점(P1)/끝점(P2)은 컷 비율에서 유도한 좌표와 일치
103 TestTrue(TEXT("L자: 커브 시작점 = P1"), OutPoints[1].Equals(CurveStart, Tolerance));
104 TestTrue(TEXT("L자: 커브 끝점 = P2"), OutPoints[OutPoints.Num() - 2].Equals(CurveEnd, Tolerance));
105
106 // 베지어 t=0.5 지점 = (P1 + 2C + P2) / 4 (2차 베지어 공식의 해석값)
107 const FVector BezierMidPoint = (CurveStart + CornerPoint * 2.0f + CurveEnd) / 4.0f;
108 bool ContainsMidPoint = false;
109 for (const FVector& Point : OutPoints)
110 {
111 if (Point.Equals(BezierMidPoint, Tolerance))
112 {
113 ContainsMidPoint = true;
114 break;
115 }
116 }
117 TestTrue(TEXT("L자: 베지어 t=0.5 지점 포함"), ContainsMidPoint);
118
119 /* 성질 검증 */
120
121 // 누적거리/마커는 순증가, 마커 개수 = 경로 개수
122 TestTrue(TEXT("L자: 누적거리 순증가"), IsMonotonicIncreasing(OutDistances));
123 TestTrue(TEXT("L자: 마커 순증가"), IsMonotonicIncreasing(OutMarkers));
124 TestEqual(TEXT("L자: 마커 개수 = 경로 개수"), OutMarkers.Num(), Path.Num());
125
126 // 곡선 전체 길이: 최단 하한(직선 구간 + P1~P2 직선 거리) < 길이 < 직각 경로 길이
127 const float StraightSections = (TileSize - TileSize * CutRatio) * 2.0f; // A~P1 + P2~C = 450
128 const float CurveChord = FVector::Dist(CurveStart, CurveEnd); // P1~P2 직선 거리
129 const float TotalLength = OutDistances.Last();
130 TestTrue(TEXT("L자: 길이 > 최단 하한"), TotalLength > StraightSections + CurveChord - Tolerance);
131 TestTrue(TEXT("L자: 길이 < 직각 경로"), TotalLength < TileSize * 2.0f);
132
133 // 코너 타일 마커는 커브 구간(P1~P2) 안에 위치
134 TestTrue(TEXT("L자: 코너 마커 >= P1 거리"), OutMarkers[1] >= OutDistances[1] - Tolerance);
135 TestTrue(TEXT("L자: 코너 마커 <= P2 거리"), OutMarkers[1] <= OutDistances[OutDistances.Num() - 2] + Tolerance);
136
137 // 모든 커브 점이 삼각형 P1-C-P2 내부 (경로 이탈 없음)
138 // 이 지오메트리에서 내부 조건: 225 <= X <= 300, 0 <= Y <= X - 225
139 for (int32 i = 1; i < OutPoints.Num() - 1; ++i)
140 {
141 const FVector& Point = OutPoints[i];
142 TestTrue(TEXT("L자: 커브 점 X 범위"), Point.X >= CurveStart.X - Tolerance && Point.X <= CornerPoint.X + Tolerance);
143 TestTrue(TEXT("L자: 커브 점 Y 범위"), Point.Y >= 0.0 - Tolerance && Point.Y <= Point.X - CurveStart.X + Tolerance);
144 }
145 }

147 /* [케이스 4] 180도 반전 경로: 크래시 없이 성질 유지 (베지어가 되돌아오기 곡선으로 퇴화) */
148 {

171 /* [케이스 6] 코너 텐션: 곡선을 코너 중점 쪽으로 당기는 정도 검증 (L자, 컷 비율 0.5) */
172 {
173 const TArray<FVector> Path = {
174 FVector(0.0, 0.0, 0.0),

179 // 텐션 2: 곡선의 t=0.5 지점이 정확히 코너 중점을 통과 (해석값)

192 // 텐션 0: 커브 구간이 P1~P2 직선 (대각선 직진) -> 곡선 길이 = 직선 구간 + 현의 길이
193 UBoardMovementPresentationComponent::BakePolyLinePoints(Path, 0.5f, 0.0f, OutPoints, OutDistances, OutMarkers);

196 const float ExpectedLength = TileSize + FVector::Dist(CurveStart, CurveEnd); // 직선 150x2 + 현
197 TestEqual(TEXT("텐션0: 전체 길이 = 직선 + 현"), OutDistances.Last(), ExpectedLength, Tolerance);
198
199 // 텐션 0 < 1 < 2 순으로 곡선이 코너 중점에 가까워짐 (t=0.5 지점의 중점 거리 비교)
200 float ApexDistances[3] = { 0.0f, 0.0f, 0.0f };
201 const float Tensions[3] = { 0.0f, 1.0f, 2.0f };
202 for (int32 i = 0; i < 3; ++i)
203 {
204 UBoardMovementPresentationComponent::BakePolyLinePoints(Path, 0.5f, Tensions[i], OutPoints, OutDistances, OutMarkers);
205 // 곡선 점들 중 코너 중점과 가장 가까운 거리
206 float MinDistance = TNumericLimits<float>::Max();
207 for (const FVector& Point : OutPoints)
208 {
209 MinDistance = FMath::Min(MinDistance, static_cast<float>(FVector::Dist(Point, CornerPoint)));
210 }
211 ApexDistances[i] = MinDistance;
212 }
213 TestTrue(TEXT("텐션 증가 -> 코너 중점 접근"), ApexDistances[0] > ApexDistances[1] && ApexDistances[1] > ApexDistances[2]);
214 }
