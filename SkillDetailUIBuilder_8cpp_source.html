46 return ECombatSkillHitShapeUI::None;

76void SkillDetailUIBuilder::FillFromSkillData(const UStaticSkillData* SkillData,
77 FSkillDetailUI& OutDetail)
78{
79 if (SkillData == nullptr)
80 {
81 return;
82 }
83
84 OutDetail.mName = SkillData->mName;
85 // 0823 확정: 구워 둔 설명은 번역 키가 없어(생성 스냅샷) 언어를 안 탄다.
86 // 같은 내용을 런타임에 다시 생성해 현재 언어의 LOCTEXT 로 조립한다.
87 // 레이어가 없어 생성이 비면 구워 둔 설명(수기 작성분)을 그대로 쓴다.
88 {
89 const FText Generated = SkillData->MakeDescription();
90 OutDetail.mDescription = Generated.IsEmpty()
91 ? SkillData->mDescription : Generated;
92 }
93 OutDetail.mIcon = SkillData->mIcon.LoadSynchronous();
94 if (const UStaticUnitSkillData* UnitSkill = Cast<UStaticUnitSkillData>(SkillData))
95 {
96 OutDetail.mActionPointCost = FMath::Max(UnitSkill->mRequiredActionPoint, 0);
97 }
98 OutDetail.mCooldownTurns = FMath::Max(SkillData->mCooldownDuration, 0);
99 for (const FSkillPhaseLayer& MotionLayer : SkillData->mSkillPhaseLayers)
100 {
101 for (const TInstancedStruct<FSkillEffectLayer>& EffectLayer :
102 MotionLayer.mSkillEffectLayers)
103 {
104 if (const FSkillEffectLayer_Attack* Attack =
105 EffectLayer.GetPtr<FSkillEffectLayer_Attack>())
106 {
107 OutDetail.mDamageMin += Attack->mMinDamage;
108 OutDetail.mDamageMax += Attack->mMaxDamage;
109 }
110 if (const FSkillEffectLayer_GetActionPoint* Gain =
111 EffectLayer.GetPtr<FSkillEffectLayer_GetActionPoint>())
112 {
113 OutDetail.mActionPointGain += Gain->mActionPointGain;
114 }
115 }
116 }
117 OutDetail.mCriticalDamage = FMath::RoundToInt(OutDetail.mDamageMax * 1.5f);
118 OutDetail.mTargeting.mSelectShape = ToSelectShape(SkillData->mAimPattern);
119 OutDetail.mTargeting.mSelectRange = StaticCast<float>(SkillData->mAimRange);
120 OutDetail.mTargeting.mHitShape = ToHitShape(SkillData->mEffectPattern);
121 OutDetail.mTargeting.mHitRange = StaticCast<float>(SkillData->mEffectArea);
122 OutDetail.mTargeting.mAimBlockerMask = SkillData->mAimBlockerMask;
123 OutDetail.mTargeting.mEffectBlockerMask = SkillData->mEffectBlockerMask;
124 OutDetail.mTargeting.mTargetPattern = ToTargetPattern(SkillData->mTargetPattern);
125}

183void SkillDetailUIBuilder::FillFromArtifactData(const UStaticArtifactData* ArtifactData,
184 FCombatArtifactUI& OutDetail)
185{
186 if (ArtifactData == nullptr)
187 {
188 return;
189 }
190
191 // 이름 미설정 DA(placeholder)는 에셋 이름으로 폴백한다 -- 상점 목록과 같은 규칙.
192 OutDetail.mName = ArtifactData->mName.IsEmpty()
193 ? FText::FromName(ArtifactData->GetFName()) : ArtifactData->mName;
194 OutDetail.mIcon = ArtifactData->mIcon.LoadSynchronous();
195 OutDetail.mPrice = ArtifactData->mPrice;
196 OutDetail.mRarityColor = ToRarityColor(ArtifactData->mRarityType);
197 OutDetail.mRarityName = StaticEnum<ERarityType>() != nullptr
198 ? StaticEnum<ERarityType>()->GetDisplayNameTextByValue(
199 StaticCast<int64>(ArtifactData->mRarityType))
200 : FText::GetEmpty();
201 OutDetail.mRarityLevel = StaticCast<int32>(ArtifactData->mRarityType);
202
203 OutDetail.mEffectDescriptions.Reset();
204 const FText Description = ArtifactData->GetDisplayDescription();
205 if (!Description.IsEmpty())
206 {
207 OutDetail.mEffectDescriptions.Add(Description);
208 }
