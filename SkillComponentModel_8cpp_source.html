45
46 const FSoftObjectPath AssetPath = AssetManager->GetPrimaryAssetPath(SkillId);

138void USkillComponentModel::ActivateSkill(UTileMapModel* MapModel, int32 SkillIndex, const FTileIndex& TargetIndex, int32 DiceSum, FOnEndSkillUI Callback)
139{
140 checkf(mSkillEntries.IsValidIndex(SkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
141
142 FSkillEntry& SkillEntry = mSkillEntries[SkillIndex];
143 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
144
145 const UStaticSkillData* SkillData = SkillEntry.mData;
146 checkf(SkillData != nullptr, TEXT("빈 스킬 시전 오류"));
147
148 UUnitModel* OwnerUnitModel = GetOwnerModel<UUnitModel>();
149 checkf(OwnerUnitModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
150
151 UPassiveComponentModel* PassiveComponentModel = OwnerUnitModel->GetPassiveComponentModel();
152 checkf(PassiveComponentModel != nullptr, TEXT("패시브 컴포넌트 nullptr"));
153
154 /* 활성화 스킬 데이터 채우기 */
155
156 mActiveSkillContext.mDiceSum = DiceSum;
157 mActiveSkillContext.mMapModel = MapModel;
158 mActiveSkillContext.mSelfTileIndex = OwnerUnitModel->GetTileTransform().mIndex;
159 mActiveSkillContext.mTargetTileIndex = TargetIndex;
160 mActiveSkillContext.mEffectTileIndexes = GetEffectTiles(MapModel, SkillIndex, TargetIndex, DiceSum);
161 mActiveSkillContext.mSkillIndex = SkillIndex;
162 mActiveSkillContext.mMotionIndex = 0;
163 mActiveSkillContext.mEndCallback = MoveTemp(Callback);
164
165 /* 스킬 실행 콜백 */
166
167 OnPlaySkillUI.Broadcast(mActiveSkillContext, SkillData);
168
169 /* 카메라 줌인 */
170
171 FVector MinTileLocation;
172 FVector MaxTileLocation;
173 FVector ZoomInLocation = MapModel->TileToWorldLocation(mActiveSkillContext.mSelfTileIndex);
174 MinTileLocation = MaxTileLocation = ZoomInLocation;
175
176 for (const FTileIndex& EffectTileIndex : mActiveSkillContext.mEffectTileIndexes)
177 {
178 const FVector TileLocation = MapModel->TileToWorldLocation(EffectTileIndex);
179 ZoomInLocation += TileLocation;
180
181 MinTileLocation.X = FMath::Min(MinTileLocation.X, TileLocation.X);
182 MinTileLocation.Y = FMath::Min(MinTileLocation.Y, TileLocation.Y);
183 MaxTileLocation.X = FMath::Max(MaxTileLocation.X, TileLocation.X);
184 MaxTileLocation.Y = FMath::Max(MaxTileLocation.Y, TileLocation.Y);
185 }
186 ZoomInLocation /= (1 + mActiveSkillContext.mEffectTileIndexes.Num());
187 ZoomInLocation.Z = OwnerUnitModel->GetWorldTransform().GetLocation().Z;
188
189 const float ZoomInSize = FMath::Max(MaxTileLocation.X - MinTileLocation.X, MaxTileLocation.Y - MinTileLocation.Y);
190
191 UWorldCameraModel* WorldCameraModel = GetWorldSubsystemModel<UWorldCameraModel>(this);
192 checkf(WorldCameraModel != nullptr, TEXT("월드 카메라 모델 nullptr"));
193
194 const UGamePlaySettings* GamePlaySettings = GetDefault<UGamePlaySettings>();
195 checkf(GamePlaySettings != nullptr, TEXT("게임 플레이 세팅 nullptr"));
196
197 const float FinalZoomInSize = FMath::Clamp(GamePlaySettings->mSkillZoomDefaultSize + ZoomInSize * GamePlaySettings->mSkillZoomSizeRatio, GamePlaySettings->mSkillMinZoomSize, GamePlaySettings->mSkillMaxZoomSize);
198 WorldCameraModel->RequestZoomInMainCamera(ZoomInLocation, FinalZoomInSize);
199
200 /* 스킬 사용 시 패시브 발동 */
201
202 TArray<UTacticalPassive*> Passives = PassiveComponentModel->GetPassivesByTiming(AbilityTags::GameplayAbility_Passive_OnStartUsingSkill);
203
204 FBoardCombatTargetSnapshotData OwnerSnapshot = OwnerUnitModel->MakeSnapshotData();
205
206 FPassiveActivateContext PassiveContext;
207 PassiveContext.mOwner = OwnerUnitModel;
208 PassiveContext.mOwnerSnapshot = &OwnerSnapshot;
209 PassiveContext.mTargets.Add(OwnerUnitModel);
210 PassiveContext.mTargetSnapshots.Add(&OwnerSnapshot);
211
212 for (UTacticalPassive*& Passive : Passives)
213 {
214 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;
215 Passive->ActivatePassive(AbilityTags::GameplayAbility_Passive_OnStartUsingSkill, PassiveContext, OUT DynamicPassiveData);
216 Passive->CommitPassive(DynamicPassiveData);
217 }
218
219
220 // 모션 레이어가 하나도 없는(미저작) 스킬은 시전을 무동작으로 즉시 종료
221 if (SkillData->mSkillMotionLayers.Num() == 0)
222 {
223 UE_LOG(LogRD, Warning, TEXT("스킬(index %d)에 모션 레이어가 없어 시전을 건너뜁니다 — DA에 mSkillMotionLayers 미설정"), SkillIndex);
224 DeactivateSkill();
225 return;
226 }
227
228 PlayMotionLayer();
229}

231void USkillComponentModel::PlayMotionLayer()
232{
233 checkf(mSkillEntries.IsValidIndex(mActiveSkillContext.mSkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
234
235 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
236 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
237
238 const UStaticSkillData* SkillData = SkillEntry.mData;
239 checkf(SkillData != nullptr, TEXT("빈 스킬 시전 오류"));
240
241 UUnitModel* OwnerUnitModel = GetOwnerModel<UUnitModel>();
242 checkf(OwnerUnitModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
243
244 IBoardCombatTarget* OwnerCombatTarget = Cast<IBoardCombatTarget>(OwnerUnitModel);
245 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
246
247 UPassiveComponentModel* PassiveComponentModel = OwnerUnitModel->GetPassiveComponentModel();
248 checkf(PassiveComponentModel != nullptr, TEXT("패시브 컴포넌트 nullptr"));
249
250 /* 모션 로그 시작 */
251
252 GetWorldEventLogger(this)->BeginMotionLog();
253 const FSkillMotionLayer& MotionLayer = SkillData->mSkillMotionLayers[mActiveSkillContext.mMotionIndex];
254
255 /* 활성화 모션 데이터 채우기 */
256
257 mActiveSkillContext.mTargetTileIndexes = MotionLayer.FilterTileIndexes(mActiveSkillContext.mSelfTileIndex, mActiveSkillContext.mEffectTileIndexes);
258 mActiveSkillContext.mOtherCombatTargets = MotionLayer.FilterCombatTargets(mActiveSkillContext.mMapModel.Get(), OwnerCombatTarget, mActiveSkillContext.mTargetTileIndexes);
259 mActiveSkillContext.mMotionTileMapDir = mActiveSkillContext.mMapModel->TileDeltaToDirection(
260 mActiveSkillContext.mSelfTileIndex,
261 mActiveSkillContext.mTargetTileIndex,
262 OwnerUnitModel->GetTileTransform().mDirection
263 );
264 mActiveSkillContext.mIsMotionTriggered = false;
265
266 /* Effect 기본 값부터 참고용으로 적용 */
267
268 for (const TInstancedStruct<FSkillEffectLayer>& EffectLayer : MotionLayer.mSkillEffectLayers)
269 {
270 EffectLayer.Get().ApplyPointEffect(OwnerCombatTarget, mActiveSkillContext.mDiceSum);
271 }
272
273 /* 이펙트 가격 전 패시브 적용 */
274
275 {
276 FBoardCombatTargetSnapshotData OwnerSnapshot = OwnerCombatTarget->MakeSnapshotData();
277
278 TArray<UTacticalPassive*> Passives = PassiveComponentModel->GetPassivesByTiming(AbilityTags::GameplayAbility_Passive_OnStartApplyingEffect);
279
280 FPassiveActivateContext PassiveContext;
281 PassiveContext.mOwner = OwnerUnitModel;
282 PassiveContext.mOwnerSnapshot = &OwnerSnapshot;
283
284 TArray<FBoardCombatTargetSnapshotData> OtherSnapshots;
285 OtherSnapshots.Reserve(mActiveSkillContext.mOtherCombatTargets.Num());
286 for (IBoardCombatTarget* OtherCombatTarget : mActiveSkillContext.mOtherCombatTargets)
287 {
288 UBoardActorModel* OtherActorModel = Cast<UBoardActorModel>(OtherCombatTarget);
289 checkf(OtherActorModel != nullptr, TEXT("스킬을 받는 타겟이 유효하지 않음"));
290 OtherSnapshots.Add(OtherCombatTarget->MakeSnapshotData());
291
292 PassiveContext.mTargets.Add(OtherActorModel);
293 PassiveContext.mTargetSnapshots.Add(&OtherSnapshots.Last());
294 }
295
296 for (UTacticalPassive* Passive : Passives)
297 {
298 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;
299
300 Passive->ActivatePassive(AbilityTags::GameplayAbility_Passive_OnStartApplyingEffect, PassiveContext, OUT DynamicPassiveData);
301 Passive->CommitPassive(DynamicPassiveData);
302 }
303 }
304
305 /* 이펙트 피격 전 패시브 적용(선택적) */
306 {
307 FBoardCombatTargetSnapshotData OwnerSnapshot = OwnerCombatTarget->MakeSnapshotData();
308
309 for (IBoardCombatTarget* OtherCombatTarget : mActiveSkillContext.mOtherCombatTargets)
310 {
311 UBoardActorModel* OtherActorModel = Cast<UBoardActorModel>(OtherCombatTarget);
312 checkf(OtherActorModel != nullptr, TEXT("스킬을 받는 타겟이 유효하지 않음"));
313
314 UPassiveComponentModel* OtherPassiveComponentModel = OtherActorModel->FindComponentModelByClass<UPassiveComponentModel>();
315 if (OtherPassiveComponentModel == nullptr)
316 {
317 continue;
318 }
319
320 TArray<UTacticalPassive*> OtherPassives = OtherPassiveComponentModel->GetPassivesByTiming(AbilityTags::GameplayAbility_Passive_OnStartReceivingEffect);
321
322 FBoardCombatTargetSnapshotData OtherSnapshot = OtherCombatTarget->MakeSnapshotData();
323
324 FPassiveActivateContext PassiveContext;
325 PassiveContext.mOwner = OtherActorModel;
326 PassiveContext.mOwnerSnapshot = &OtherSnapshot;
327 PassiveContext.mTargets.Add(OwnerUnitModel);
328 PassiveContext.mTargetSnapshots.Add(&OwnerSnapshot);
329
330 for (UTacticalPassive* OtherPassive : OtherPassives)
331 {
332 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;
333
334 OtherPassive->ActivatePassive(AbilityTags::GameplayAbility_Passive_OnStartReceivingEffect, PassiveContext, OUT DynamicPassiveData);
335 OtherPassive->CommitPassive(DynamicPassiveData);
336 }
337 }
338 }
339
340 /* 애니메이션 시작 */
341
342 // 자동 회전
343 if (MotionLayer.mAutoRotateTowardTarget == true)
344 {
345 // 회전이 끝나면 베리어가 소멸하면서 PlayMotionLayerAnimation() 함수 호출.
346 // 이때 방향은 로컬방향이라서 Forward는 정면을 향하고 있음
347 TSharedPtr<FPresentationBarrier> RotateBarrier = FPresentationBarrier::Make(
348 FOnFinishPresentation::CreateWeakLambda(this, [this]() {
349 PlayMotionLayerAnimation(ETileActorDirection::Forward);
350 }));
351
352 // 타일맵의 절대적 방향으로 회전
353 mActiveSkillContext.mMapModel->RotateActor(mActiveSkillContext.mMotionTileMapDir, OwnerUnitModel, RotateBarrier);
354 }
355 // 자동 회전 안 함
356 else
357 {
358 // 회전하지 않으므로 즉시 모션 시작
359 PlayMotionLayerAnimation(TileMapToLocalDirection(mActiveSkillContext.mMotionTileMapDir, OwnerUnitModel->GetTileTransform().mDirection));
360 }
361}

363void USkillComponentModel::PlayMotionLayerAnimation(ETileActorDirection LocalDirectionToTarget)
364{
365 checkf(mSkillEntries.IsValidIndex(mActiveSkillContext.mSkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
366
367 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
368 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
369
370 const UStaticSkillData* SkillData = SkillEntry.mData;
371 checkf(SkillData != nullptr, TEXT("빈 스킬 시전 오류"));
372
373 UUnitModel* OwnerUnitModel = GetOwnerModel<UUnitModel>();
374 checkf(OwnerUnitModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
375
376 const FSkillMotionLayer& MotionLayer = SkillData->mSkillMotionLayers[mActiveSkillContext.mMotionIndex];
377
378 auto MotionEndBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
379 TriggerMotionLayer(nullptr);
380 EndMotionLayer();
381 }));
382
383 FOnRequestReceiveAnimation TriggerCallback;
384 TriggerCallback.BindWeakLambda(this, [this](const FApplyEventTriggerPayload* Payload) {
385 TriggerMotionLayer(Payload);
386 });
387
388 mActiveSkillContext.mMotionEndBarrier = MotionEndBarrier;
389
390 OwnerUnitModel->OnPlayApplyAnimationUI.Broadcast(MotionEndBarrier, MoveTemp(TriggerCallback), MotionLayer.mApplyMotionTag, LocalDirectionToTarget);
391 OnPlayMotionLayerUI.Broadcast(mActiveSkillContext.mMotionIndex, MotionEndBarrier, MotionLayer.mApplyMotionTag, LocalDirectionToTarget);
392}

507void USkillComponentModel::EndMotionLayer()
508{
509 checkf(mSkillEntries.IsValidIndex(mActiveSkillContext.mSkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
510
511 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
512 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
513
514 const UStaticSkillData* SkillData = SkillEntry.mData;
515 checkf(SkillData != nullptr, TEXT("빈 스킬 시전 오류"));
516
517 UUnitModel* OwnerUnitModel = GetOwnerModel<UUnitModel>();
518 checkf(OwnerUnitModel != nullptr, TEXT("스킬을 시전한 Owner가 유효하지 않음"));

520 IBoardCombatTarget* OwnerCombatTarget = Cast<IBoardCombatTarget>(OwnerUnitModel);

532

534 OnEndMotionLayerUI.Broadcast(mActiveSkillContext.mMotionIndex);

537
538 ++mActiveSkillContext.mMotionIndex;
539 if (SkillData->mSkillMotionLayers.Num() == mActiveSkillContext.mMotionIndex)

549void USkillComponentModel::DeactivateSkill()
550{
551 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
552 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
553
554 const UStaticSkillData* SkillData = SkillEntry.mData;
555 checkf(SkillData != nullptr, TEXT("빈 스킬 시전 오류"));
556
557 UUnitModel* OwnerUnitModel = GetOwnerModel<UUnitModel>();
558 checkf(OwnerUnitModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
559

561 checkf(PassiveComponentModel != nullptr, TEXT("패시브 컴포넌트 nullptr"));
562
563 /* 스킬 종료 시 패시브 발동 */
564
565 TArray<UTacticalPassive*> Passives = PassiveComponentModel->GetPassivesByTiming(AbilityTags::GameplayAbility_Passive_OnEndUsingSkill);
566
567 FBoardCombatTargetSnapshotData OwnerSnapshot = OwnerUnitModel->MakeSnapshotData();
568
569 FPassiveActivateContext PassiveContext;
570 PassiveContext.mOwner = OwnerUnitModel;
571 PassiveContext.mOwnerSnapshot = &OwnerSnapshot;
572 PassiveContext.mTargets.Add(OwnerUnitModel);
573 PassiveContext.mTargetSnapshots.Add(&OwnerSnapshot);
574
575 for (UTacticalPassive*& Passive : Passives)
576 {
577 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;
578 Passive->ActivatePassive(AbilityTags::GameplayAbility_Passive_OnEndUsingSkill, PassiveContext, OUT DynamicPassiveData);
579 Passive->CommitPassive(DynamicPassiveData);
580 }
581
582 /* 카메라 줌아웃 */
583
584 UWorldCameraModel* WorldCameraModel = GetWorldSubsystemModel<UWorldCameraModel>(this);
585 checkf(WorldCameraModel != nullptr, TEXT("월드 카메라 모델 nullptr"));
586
587 WorldCameraModel->RequestZoomOutMainCamera();
588
589 /* 스킬 종료 콜백 */
590
591 mActiveSkillContext.mEndCallback.Broadcast(mActiveSkillContext, SkillData);
592 OnEndSkillUI.Broadcast(mActiveSkillContext, SkillData);
593
594 /* 활성화 스킬 데이터 비우기 */
595
596 mActiveSkillContext.Clear();
597}
