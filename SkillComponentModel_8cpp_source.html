28 UStaticSkillData* LoadStaticSkillData(const FPrimaryAssetId& SkillId)

30 if (SkillId.IsValid() == false)

32 return nullptr;

37 {
38 return nullptr;

40

42 {
43 return Loaded;
44 }
45
46 const FSoftObjectPath AssetPath = AssetManager->GetPrimaryAssetPath(SkillId);
47 return Cast<UStaticSkillData>(AssetPath.TryLoad());

105 *Skill.ToSoftObjectPath().ToString(), *GetNameSafe(GetOwnerModel()), NextSkillIndex - 1,
106 *GetNameSafe(SkillData));
107 }

144bool USkillComponentModel::SetSkill(int32 SkillIndex, UStaticSkillData* SkillData)
145{
146 checkf(mSkillEntries.IsValidIndex(SkillIndex) == true, TEXT("잘못된 스킬 인덱스 범위"));
147
148 if (IsAcquirableSkill(SkillData) == false)
149 {
150 return false;
151 }
152
153 const UStaticSkillData* PreSkillData = mSkillEntries[SkillIndex].mData;
154 mSkillEntries[SkillIndex] = FSkillEntry(SkillData);
155 if (SkillData->mStartsOnCooldown == true)
156 {
157 /* 쿨다운 처리 */
158
159 IBoardCombatTarget* OwnerCombatTarget = GetOwnerModel<IBoardCombatTarget>();
160 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
161 UAttributeSetComponentModel* AttributeSetCompModel = OwnerCombatTarget->GetAttributeComponentModel();
162 checkf(AttributeSetCompModel != nullptr, TEXT("속성 컴포넌트 nullptr"));
163
164 UTacticalEffectContext* EffectContext = AttributeSetCompModel->MakeEffectContext();
165 TSharedPtr<FTacticalEffectSpec> EffectSpec = AttributeSetCompModel->MakeOutgoingSpec(SkillData->mCooldownEffectClass.LoadSynchronous(), EffectContext);
166 EffectSpec->mDynamicDurationMagnitude = GetStaticCooldownDuration(SkillIndex);
167 mSkillEntries[SkillIndex].mCooldownHandle = AttributeSetCompModel->ApplyTacticalEffectSpecToSelf(*EffectSpec);
168 }
169
170 OnChangeSkillUI.Broadcast(SkillIndex, PreSkillData, SkillData);
171 return true;
172}

316{
317 checkf(mSkillEntries.IsValidIndex(SkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
318
319 FSkillEntry& SkillEntry = mSkillEntries[SkillIndex];
320 const UStaticSkillData* SkillData = SkillEntry.mData;
321
322 UBoardActorModel* OwnerBoardActorModel = GetOwnerModel<UBoardActorModel>();
323 checkf(OwnerBoardActorModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
324
325 IBoardCombatTarget* OwnerCombatTarget = GetOwnerModel<IBoardCombatTarget>();
326 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
327
328 UAttributeSetComponentModel* AttributeSetCompModel = OwnerCombatTarget->GetAttributeComponentModel();
329 checkf(AttributeSetCompModel != nullptr, TEXT("속성 컴포넌트 nullptr"));
330
331 /* 활성화 스킬 데이터 채우기 */
332
333 auto SkillEndBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
334 DeactivateSkill();
335 }));
336 {
337 mActiveSkillContext.mInstigator = OwnerBoardActorModel;
338 mActiveSkillContext.mMapModel = MapModel;
339 mActiveSkillContext.mSelfTileIndex = OwnerBoardActorModel->GetTileTransform().mIndex;
340 mActiveSkillContext.mAimedTileIndex = AimedTileIndex;
341 mActiveSkillContext.mTargetTileIndexes = GetTargetTiles(MapModel, SkillIndex, AimedTileIndex);
342 mActiveSkillContext.mEffectTileIndexes = GetEffectTiles(MapModel, SkillIndex, mActiveSkillContext.mTargetTileIndexes);
343 mActiveSkillContext.mSkillEndBarrier = SkillEndBarrier;
344 mActiveSkillContext.mSkillIndex = SkillIndex;
345 mActiveSkillContext.mAnimationIndex = 0;
346 mActiveSkillContext.mPhaseIndex = 0;
347 mActiveSkillContext.mEndCallback = MoveTemp(Callback);
348 }
349
350 auto SkillPlayerBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
351
352 checkf(mSkillEntries.IsValidIndex(mActiveSkillContext.mSkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
353
354 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
355 const UStaticSkillData* SkillData = SkillEntry.mData;
356 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
357
358 UBoardActorModel* OwnerBoardActorModel = GetOwnerModel<UBoardActorModel>();
359 checkf(OwnerBoardActorModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
360
361 IBoardCombatTarget* OwnerCombatTarget = GetOwnerModel<IBoardCombatTarget>();
362 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
363
364 /* 스킬 실행 콜백 */
365
366 OnPlaySkillUI.Broadcast(mActiveSkillContext, SkillData, mActiveSkillContext.mSkillEndBarrier);
367 OwnerCombatTarget->OnStartUsingSkill(mActiveSkillContext, mActiveSkillContext.mSkillIndex);
368
369 /* 스킬 페이즈 시작 */
370
371 PreparePhaseLayer();
372
373 /* 애니메이션 시작 */
374
375 const ETileActorDirection MotionTileMapDir = mActiveSkillContext.mMapModel->TileDeltaToDirection(
376 mActiveSkillContext.mSelfTileIndex,
377 mActiveSkillContext.mAimedTileIndex,
378 OwnerBoardActorModel->GetTileTransform().mDirection
379 );
380
381 if (SkillData->mSkillAnimationSet.mAutoRotateTowardTarget == true)
382 {
383 // 자동 회전
384
385 TSharedPtr<FPresentationBarrier> RotateBarrier = FPresentationBarrier::Make(
386 FOnFinishPresentation::CreateWeakLambda(this, [this]() {
387 // 회전 완료 후 로컬 정면방향으로 실행

390 }));

396 {
397 // 자동 회전 안 함
398
399 mActiveSkillContext.mMotionLocalDir = TileMapToLocalDirection(MotionTileMapDir, OwnerBoardActorModel->GetTileTransform().mDirection);
400 PlaySkillAnimation();
401 }
402 }));
403 OnPrePlaySkillUI.Broadcast(mActiveSkillContext, SkillData, SkillPlayerBarrier);
404}

429void USkillComponentModel::TriggerPhaseLayer(const FEventTriggerPayloadBase* Payload)
430{
431 checkf(mSkillEntries.IsValidIndex(mActiveSkillContext.mSkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
432
433 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
434 const UStaticSkillData* SkillData = SkillEntry.mData;
435 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
436
437 UBoardActorModel* OwnerBoardActorModel = GetOwnerModel<UBoardActorModel>();
438 checkf(OwnerBoardActorModel != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
439
440 IBoardCombatTarget* OwnerCombatTarget = GetOwnerModel<IBoardCombatTarget>();
441 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
442
443 // 앞 페이즈가 밀당 종료를 기다리는 중이거나 남은 페이즈가 없으면 노티파이 무시
444 if (IsWaitingForcedMove() == true || mActiveSkillContext.mPhaseIndex >= SkillData->mSkillPhaseLayers.Num())
445 {
446 return;
447 }
448
449 const FSkillPhaseLayer& PhaseLayer = SkillData->mSkillPhaseLayers[mActiveSkillContext.mPhaseIndex];
450
451 /* 페이즈 배리어 */
452
453 // 이 배리어가 풀리면 EndPhaseLayer가 호출되어 페이즈가 끝남
454 // 밀당이 있으면 대기열이 배리어를 잡고 있다가 모든 대상이 다 밀린 뒤에 풀고, 밀당이 없으면 이 함수 끝에서 바로 풀림
455 TSharedPtr<FPresentationBarrier> PhaseBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
456 EndPhaseLayer();
457 }));

465 struct FFactorEffectHandleContainer
466 {
467 public:
468 FFactorEffectHandleContainer(TArray<FActiveTacticalEffectHandle> Handle) : mHandles(Handle)
469 {
470 }
471
472 public:
473 TArray<FActiveTacticalEffectHandle> mHandles;
474 };
475
476 const int32 EffectLayerNum = PhaseLayer.mSkillEffectLayers.Num();
477 TArray<FFactorEffectHandleContainer> FactorEffectHandleContainers;
478 FactorEffectHandleContainers.Reserve(PhaseLayer.mSkillEffectLayers.Num());
479
480 /* Effect 기본 값부터 적용 */
481
482 {
483 UBoardCombatTargetSnapshotData* OwnerSnapshot = OwnerCombatTarget->MakeSnapshotData();
484
485 for (int32 i = 0; i < EffectLayerNum; ++i)
486 {
487 const TInstancedStruct<FSkillEffectLayer>& EffectLayer = PhaseLayer.mSkillEffectLayers[i];
488 FactorEffectHandleContainers.Add(EffectLayer.Get().ApplyFactorEffect(OwnerCombatTarget, OwnerSnapshot));
489 }
490 }
491
492 /* 이펙트 전 이벤트들 */
493
494 OwnerCombatTarget->OnStartApplyingEffects(mActiveSkillContext, mActiveSkillContext.mPhaseIndex);
495 {
496 UBoardCombatTargetSnapshotData* OwnerSnapshot = OwnerCombatTarget->MakeSnapshotData();
497
498 for (IBoardCombatTarget* OtherCombatTarget : mActiveSkillContext.mFinalCombatTargets)
499 {
500 OtherCombatTarget->OnStartReceivingEffects(OwnerSnapshot, mActiveSkillContext, mActiveSkillContext.mPhaseIndex);
501 }
502 }
503
504 /* 실제 Effect 적용 (밀당은 대기열에 등록만 됨) */
505
506 {
507 UBoardCombatTargetSnapshotData* OwnerSnapshot = OwnerCombatTarget->MakeSnapshotData();
508 TArray<TScriptInterface<IBoardCombatTarget>> OtherCombatTargets;
509 OtherCombatTargets.Reserve(mActiveSkillContext.mFinalCombatTargets.Num());
510 TArray<TObjectPtr<UBoardCombatTargetSnapshotData>> OtherSnapshots;
511 OtherSnapshots.Reserve(mActiveSkillContext.mFinalCombatTargets.Num());
512 for (IBoardCombatTarget* OtherCombatTarget : mActiveSkillContext.mFinalCombatTargets)
513 {
514 UBoardActorModel* OtherActorModel = Cast<UBoardActorModel>(OtherCombatTarget);
515 checkf(OtherActorModel != nullptr, TEXT("스킬을 받는 타겟이 유효하지 않음"));
516
517 OtherCombatTargets.Add(OtherActorModel);
518 OtherSnapshots.Add(OtherCombatTarget->MakeSnapshotData());
519 }
520
521 FSkillEffectCommitParams Params(
522 OwnerBoardActorModel,
523 OwnerSnapshot,
524 OtherCombatTargets,
525 OtherSnapshots,
526 mActiveSkillContext.mFinalTileIndexes,
527 mActiveSkillContext.mAimedTileIndex
528 );
529
530 // 이펙트를 적용하는 중에 들어온 밀당 요청은 대기열에 모임
531 mShouldEnqueueForcedMove = true;
532 for (int32 i = 0; i < EffectLayerNum; ++i)
533 {
534 const TInstancedStruct<FSkillEffectLayer>& EffectLayer = PhaseLayer.mSkillEffectLayers[i];
535 EffectLayer.Get().CommitEffect(Params);
536 }
537 mShouldEnqueueForcedMove = false;
538 }
539
540 /* Effect 포인트 수치 비우기 */
541
542 {
543 for (int32 i = 0; i < EffectLayerNum; ++i)
544 {
545 const TInstancedStruct<FSkillEffectLayer>& EffectLayer = PhaseLayer.mSkillEffectLayers[i];
546 EffectLayer.Get().ClearFactorEffect(OwnerCombatTarget, FactorEffectHandleContainers[i].mHandles);
547 }
548 }
549
550 /* 미뤄놨던 밀당 요청 순차 처리 */
551
552 // 이펙트 적용 중에 모아둔 밀당 요청이 있으면 지금부터 하나씩 처리
553 // 대기열이 페이즈 배리어를 들고 있으므로, 마지막 요청까지 다 처리된 뒤에야 배리어가 풀리고 페이즈가 끝난다
554 if (mPendingForcedMoves.IsEmpty() == false)
555 {
556 mForcedMovePhaseBarrier = PhaseBarrier;
557 StartNextForcedMove();
558 }
559
560 // 이 함수가 들고 있던 배리어를 놓는다
561 // 밀당이 없었으면 이게 마지막 참조라서 여기서 바로 EndPhaseLayer가 호출된다
562 // EndPhaseLayer 안에서 mPhaseIndex 등 컨텍스트가 바뀌므로 이 줄 뒤에 다른 코드를 두면 안 된다
563 PhaseBarrier.Reset();
564}

566void USkillComponentModel::EndPhaseLayer()
567{
568 checkf(mSkillEntries.IsValidIndex(mActiveSkillContext.mSkillIndex) == true, TEXT("잘못된 사용 스킬 인덱스"));
569
570 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
571 const UStaticSkillData* SkillData = SkillEntry.mData;
572 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
573
574 IBoardCombatTarget* OwnerCombatTarget = GetOwnerModel<IBoardCombatTarget>();
575 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
576
577 /* 이펙트 후 이벤트들 */
578
579 {
580 UBoardCombatTargetSnapshotData* OwnerSnapshot = OwnerCombatTarget->MakeSnapshotData();
581
582 for (IBoardCombatTarget* OtherCombatTarget : mActiveSkillContext.mFinalCombatTargets)
583 {
584 OtherCombatTarget->OnEndReceivingEffects(OwnerSnapshot, mActiveSkillContext, mActiveSkillContext.mPhaseIndex);
585 }
586 }
587 OwnerCombatTarget->OnEndApplyingEffects(mActiveSkillContext, mActiveSkillContext.mPhaseIndex);
588
589 /* 모션 로그 종료 */
590
591 GetWorldEventLogger(this)->EndMotionLog();
592
593 /* 페이즈 종료 시 대리자 호출 */
594
595 OnEndPhaseLayerUI.Broadcast(mActiveSkillContext.mPhaseIndex);
596
597 /* 종료 판정 */
598
599 ++mActiveSkillContext.mPhaseIndex;
600 if (mActiveSkillContext.mPhaseIndex < SkillData->mSkillPhaseLayers.Num())
601 {
602 /* 다음 스킬 준비 */
603
604 PreparePhaseLayer();
605 }
606
607 /* 보류했던 스킬 종료 재개 */
608
609 // 밀당을 기다리는 동안 시전자 애니메이션이 먼저 끝났으면 DeactivateSkill이 보류하고 돌아갔으므로 여기서 다시 부른다
610 // 보류된 적 없으면 아무것도 안 함
611 // - 시전자 애니메이션이 아직 진행 중이면, 애니메이션이 끝날 때 DeactivateSkill이 불림
612 // - DeactivateSkill이 이미 실행 중이면, 그쪽이 계속 진행함
613 if (mActiveSkillContext.mIsDeactivationPending == true)
614 {
615 DeactivateSkill();
616 }
617}

619void USkillComponentModel::FlushRemainingPhaseLayers()
620{
621 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
622 const UStaticSkillData* SkillData = SkillEntry.mData;
623 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
624
625 /* 남은 스킬 페이즈 털어내기 */
626
627 while (mActiveSkillContext.mPhaseIndex < SkillData->mSkillPhaseLayers.Num())
628 {
629 // 밀당을 기다리는 페이즈가 생기면 멈춤. 밀당이 끝나면 DeactivateSkill이 다시 불려서 나머지를 이어감
630 if (IsWaitingForcedMove() == true)
631 {
632 break;
633 }
634 TriggerPhaseLayer(nullptr);
635 }
636}

638void USkillComponentModel::DeactivateSkill()
639{
640 FSkillEntry& SkillEntry = mSkillEntries[mActiveSkillContext.mSkillIndex];
641 const UStaticSkillData* SkillData = SkillEntry.mData;
642 checkf(SkillEntry.IsValid() == true, TEXT("빈 스킬 시전 오류"));
643
644 IBoardCombatTarget* OwnerCombatTarget = GetOwnerModel<IBoardCombatTarget>();
645 checkf(OwnerCombatTarget != nullptr, TEXT("스킬을 시전할 Owner가 유효하지 않음"));
646
647 /* 밀당 대기 중이면 스킬 종료 보류 */
648
649 // 보류 표시를 켜두고 나가면, 밀당이 끝날 때 EndPhaseLayer가 이 함수를 다시 부름
650 mActiveSkillContext.mIsDeactivationPending = false;
651 if (IsWaitingForcedMove() == true)
652 {
653 mActiveSkillContext.mIsDeactivationPending = true;
654 return;
655 }
656
657 /* 미호출한 Phase 단계들 처리 */
658
659 FlushRemainingPhaseLayers();
660
661 // 털어내던 페이즈에 밀당이 있어서 기다리게 됐으면 여기서도 보류
662 if (IsWaitingForcedMove() == true)
663 {
664 mActiveSkillContext.mIsDeactivationPending = true;
665 return;
666 }
667
668 /* 스킬 종료 콜백 */
669
670 OwnerCombatTarget->OnEndUsingSkill(mActiveSkillContext.mSkillIndex);
671 mActiveSkillContext.mEndCallback.Broadcast(mActiveSkillContext, SkillData);
672 OnEndSkillUI.Broadcast(mActiveSkillContext, SkillData);
673
674 /* 활성화 스킬 데이터 비우기 */
675
676 mActiveSkillContext.Clear();
677}

691void USkillComponentModel::StartNextForcedMove()
692{
693 // 남은 요청이 없으면 페이즈 배리어를 놓음 -> EndPhaseLayer 호출
694 if (mPendingForcedMoves.IsEmpty() == true)
695 {
696 mForcedMovePhaseBarrier.Reset();
697 return;
698 }
699
700 FOnStartForcedMove Start = MoveTemp(mPendingForcedMoves[0]);
701 mPendingForcedMoves.RemoveAt(0);
702
703 // 이 대상의 이동(함정 연쇄 포함)이 끝나면 풀리는 배리어. 풀리면 다음 요청 출발
704 TSharedPtr<FPresentationBarrier> MoveEndBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
705 StartNextForcedMove();
706 }));
707
708 // 출발. 이동을 시작했으면 이동 컴포넌트가 배리어를 들고 있다가 이동이 끝날 때 놓음
709 // 시작하지 못했으면(면역, 밀 곳 없음, 이미 이동 중) 아무도 안 들고 있으므로 이 함수가 끝날 때 바로 풀려서 다음 요청으로 넘어감
710 Start.ExecuteIfBound(MoveEndBarrier);
711}
