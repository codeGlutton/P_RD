38 FIntPoint NativeTextureSize(UTexture2D* Source)
39 {
40 if (Source == nullptr)
41 {
42 return FIntPoint::ZeroValue;
43 }
44
45 const FIntPoint ImportedSize = Source->GetImportedSize();
46 if (ImportedSize.X > 0 && ImportedSize.Y > 0)
47 {
48 return ImportedSize;
49 }
50 return FIntPoint(Source->GetSizeX(), Source->GetSizeY());
51 }

60 FSlateBrush TextureBrush(UTexture2D* Source, const FBox2f& UV, bool bBox = false)
61 {
62 FSlateBrush Brush;
63 Brush.SetResourceObject(Source);
64 Brush.DrawAs = bBox ? ESlateBrushDrawType::Box : ESlateBrushDrawType::Image;
65 Brush.Margin = bBox ? FMargin(.16f) : FMargin(0.f);
66 const FIntPoint NativeSize = NativeTextureSize(Source);
67 Brush.ImageSize = NativeSize.X > 0 && NativeSize.Y > 0
68 ? FVector2D(NativeSize) : FVector2D::ZeroVector;
69 Brush.SetUVRegion(UV);
70 return Brush;
71 }

73 void StyleText(UTextBlock* Text, int32 Size, const FLinearColor& Color,
74 ETextJustify::Type Justification = ETextJustify::Center)
75 {
76 FSlateFontInfo Font = UIFont::MakeProjectExact(Text->GetFont(), Size);
77 const bool bLightText = Color.R > .5f;
78 Font.OutlineSettings.OutlineSize = bLightText ? (Size >= 60 ? 3 : 2) : 0;
79 Font.OutlineSettings.OutlineColor = FLinearColor(0.03f, 0.015f, 0.005f, 1.f);
80 Text->SetFont(Font);
81 Text->SetColorAndOpacity(FSlateColor(Color));
82 Text->SetShadowOffset(bLightText ? FVector2D(1.5f, 1.5f) : FVector2D::ZeroVector);
83 Text->SetShadowColorAndOpacity(FLinearColor(0.f, 0.f, 0.f, .62f));
84 Text->SetJustification(Justification);
85 Text->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
86 }

111 UTextBlock* AddText(UWidgetBlueprint* Blueprint, UCanvasPanel* Parent, const FName Name,
112 const FText& Value, int32 FontSize, const FLinearColor& Color,
113 const FVector2D Position, const FVector2D Size, int32 ZOrder,
114 ETextJustify::Type Justification = ETextJustify::Center)
115 {
116 // UTextBlock does not expose a vertical justification option. Giving the text
117 // the full Canvas slot therefore leaves glyphs top-aligned inside taller rows.
118 // A dedicated overlay keeps the semantic TextBlock name intact while centering
119 // its desired height inside the exact design rectangle at every ScaleBox size.
120 const FName MountName(*FString::Printf(TEXT("%s_CenterMount"), *Name.ToString()));
121 UOverlay* Mount = Blueprint->WidgetTree->ConstructWidget<UOverlay>(
122 UOverlay::StaticClass(), MountName);
123 Mount->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
124 Place(Parent, Mount, Position, Size, ZOrder);
125
126 UTextBlock* Text = Blueprint->WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass(), Name);
127 Text->SetText(Value);
128 StyleText(Text, FontSize, Color, Justification);
129 Mount->AddChildToOverlay(Text);
130 UOverlaySlot* TextSlot = CastChecked<UOverlaySlot>(Text->Slot);
131 TextSlot->SetHorizontalAlignment(HAlign_Fill);
132 TextSlot->SetVerticalAlignment(VAlign_Center);
133 return Text;
134 }

168 {
169 // Resolve every new hard dependency before touching the existing WidgetTree.
170 // A missing import must fail without leaving the currently open asset empty.
171 UTexture2D* Artwork = Texture(TEXT("/Game/SVN/OutSideAsset/AICreation/UI/DefeatClassic_20260916/T_DefeatClassicBlank_v1.T_DefeatClassicBlank_v1"));
172
173 UWidgetBlueprint* Blueprint = FindOrCreateBlueprint();
174 if (Blueprint == nullptr || Blueprint->WidgetTree == nullptr)
175 {
176 UE_LOG(LogTemp, Error, TEXT("RD_COMBAT_DEFEAT_BUILD could not create %s"), AssetPath);
177 return;
178 }
179
180 Blueprint->Modify();
181 Blueprint->WidgetTree->Modify();
182 // The 0809 board changes some legacy mount classes. Removing the old root
183 // releases all prior widget objects before names are reused by another class.
184 // DeleteWidgets structurally compiles immediately, so use the neutral UUserWidget
185 // parent during that one transient compile to avoid false BindWidget errors.
186 if (UWidget* PreviousRoot = Blueprint->WidgetTree->RootWidget)
187 {
188 Blueprint->ParentClass = UUserWidget::StaticClass();
189 TSet<UWidget*> PreviousWidgets;
190 PreviousWidgets.Add(PreviousRoot);
191 FWidgetBlueprintEditorUtils::DeleteWidgets(
192 Blueprint,
193 MoveTemp(PreviousWidgets),
194 FWidgetBlueprintEditorUtils::EDeleteWidgetWarningType::DeleteSilently);
195 }
196 Blueprint->ParentClass = UCombatResultOverlayWidget::StaticClass();
197
198 UOverlay* Root = Blueprint->WidgetTree->ConstructWidget<UOverlay>(UOverlay::StaticClass(), TEXT("DefeatViewportRoot"));
199 Blueprint->WidgetTree->RootWidget = Root;
200
201 UBorder* Dimmer = Blueprint->WidgetTree->ConstructWidget<UBorder>(UBorder::StaticClass(), TEXT("BattlefieldDimmer"));
202 Dimmer->SetBrushColor(FLinearColor(.022f, .032f, .025f, 1.f));
203 Root->AddChildToOverlay(Dimmer);
204 CastChecked<UOverlaySlot>(Dimmer->Slot)->SetHorizontalAlignment(HAlign_Fill);
205 CastChecked<UOverlaySlot>(Dimmer->Slot)->SetVerticalAlignment(VAlign_Fill);
206
207 UScaleBox* Scale = Blueprint->WidgetTree->ConstructWidget<UScaleBox>(UScaleBox::StaticClass(), TEXT("DefeatResponsiveScale"));
208 Scale->SetStretch(EStretch::ScaleToFit);
209 Scale->SetStretchDirection(EStretchDirection::Both);
210 Scale->SetClipping(EWidgetClipping::ClipToBoundsAlways);
211 Root->AddChildToOverlay(Scale);
212 CastChecked<UOverlaySlot>(Scale->Slot)->SetHorizontalAlignment(HAlign_Fill);
213 CastChecked<UOverlaySlot>(Scale->Slot)->SetVerticalAlignment(VAlign_Fill);
214
215 USizeBox* DesignSize = Blueprint->WidgetTree->ConstructWidget<USizeBox>(USizeBox::StaticClass(), TEXT("DefeatDesignSize"));
216 DesignSize->SetWidthOverride(1672.f);
217 DesignSize->SetHeightOverride(941.f);
218 Scale->AddChild(DesignSize);
219 UCanvasPanel* Canvas = Blueprint->WidgetTree->ConstructWidget<UCanvasPanel>(UCanvasPanel::StaticClass(), TEXT("DefeatDesignCanvas"));
220 DesignSize->SetContent(Canvas);
221
222 const FBox2f FullUV(FVector2f::ZeroVector, FVector2f(1.f, 1.f));
223
224 // Approved composition is a text-free art plate. All copy and values are native
225 // widgets, so localized labels and live run data never become baked pixels.
226 AddImage(Blueprint, Canvas, TEXT("DefeatArtwork"), Artwork, FullUV,

232 38, InkColor, FVector2D(376.f, 247.f), FVector2D(920.f, 67.f), 1);

234 NSLOCTEXT("CombatDefeat", "Location", "현재 전투 지역"), 31, DefeatTextColor,
235 FVector2D(609.f, 469.f), FVector2D(454.f, 59.f), 1);
236 AddText(Blueprint, Canvas, TEXT("DefeatRoundLabel"),
237 NSLOCTEXT("CombatDefeat", "RoundLabel", "진행 라운드"), 25, InkColor,
238 FVector2D(522.f, 558.f), FVector2D(271.f, 40.f), 1);

240 NSLOCTEXT("CombatDefeat", "EnemiesDefeated", "처치한 적"), 25, InkColor,
241 FVector2D(878.f, 558.f), FVector2D(271.f, 40.f), 1);
242 UTextBlock* RoundText = AddText(Blueprint, Canvas, TEXT("mRoundText"),
243 FText::AsNumber(1), 46, InkColor, FVector2D(535.f, 595.f), FVector2D(245.f, 77.f), 1);

245 FText::AsNumber(0), 46, InkColor, FVector2D(891.f, 595.f), FVector2D(245.f, 77.f), 1);
246 Blueprint->OnVariableAdded(LocationText->GetFName());
247 Blueprint->OnVariableAdded(RoundText->GetFName());
248 Blueprint->OnVariableAdded(EnemyText->GetFName());
249 const FVector2D TitleButtonPosition(548.f, 734.f);
250 const FVector2D TitleButtonBounds(576.f, 120.f);
251 AddText(Blueprint, Canvas, TEXT("mTitleButtonText"),
252 NSLOCTEXT("CombatDefeat", "BackToTitle", "타이틀로 돌아가기"),

255 TitleButtonPosition, TitleButtonBounds, 3);
256
257 // UE 5.7 expects every live widget to have a stable variable GUID. Previous
258 // widget GUIDs are removed by DeleteWidgets without touching animation GUIDs.
259 Blueprint->WidgetTree->ForEachWidget([Blueprint](UWidget* Widget)
260 {
261 if (Widget == nullptr)
262 {
263 return;
264 }
265 if (Blueprint->WidgetVariableNameToGuidMap.Contains(Widget->GetFName()) == false)
266 {
267 Blueprint->OnVariableAdded(Widget->GetFName());
268 }
269 });
270
271 FBlueprintEditorUtils::MarkBlueprintAsStructurallyModified(Blueprint);
272 FKismetEditorUtilities::CompileBlueprint(Blueprint);
273 const FString Filename = FPackageName::LongPackageNameToFilename(
274 Blueprint->GetOutermost()->GetName(), FPackageName::GetAssetPackageExtension());
275 if (!UPackage::SavePackage(Blueprint->GetPackage(), Blueprint, *Filename, FSavePackageArgs()))
276 {
277 UE_LOG(LogTemp, Error, TEXT("RD_COMBAT_DEFEAT_BUILD save failed"));
278 return;
279 }
280 UE_LOG(LogTemp, Display, TEXT("RD_COMBAT_DEFEAT_BUILD success asset=%s responsive=1672x941"), AssetPath);
281 }
