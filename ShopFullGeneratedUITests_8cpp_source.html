84 bool ContainsBytes(const TArray<uint8>& Haystack, const TArray<uint8>& Needle)
85 {
86 if (Needle.IsEmpty() || Haystack.Num() < Needle.Num())
87 {
88 return false;
89 }
90
91 for (int32 Offset = 0; Offset <= Haystack.Num() - Needle.Num(); ++Offset)
92 {
93 if (FMemory::Memcmp(Haystack.GetData() + Offset,
94 Needle.GetData(), Needle.Num()) == 0)
95 {
96 return true;
97 }
98 }
99 return false;
100 }

103 bool PackageContainsString(const TArray<uint8>& PackageBytes, const FString& Value)
104 {
105 FTCHARToUTF8 Utf8(*Value);
106 TArray<uint8> AnsiNeedle;
107 AnsiNeedle.Append(reinterpret_cast<const uint8*>(Utf8.Get()), Utf8.Length());
108 if (ContainsBytes(PackageBytes, AnsiNeedle))
109 {
110 return true;
111 }
112
113 TArray<uint8> Utf16Needle;
114 Utf16Needle.Reserve(Value.Len() * 2);
115 for (const TCHAR Character : Value)
116 {
117 Utf16Needle.Add(static_cast<uint8>(Character & 0xff));
118 Utf16Needle.Add(static_cast<uint8>((Character >> 8) & 0xff));
119 }
120 return ContainsBytes(PackageBytes, Utf16Needle);
121 }

129 bool SourceContainsTransparentPixel(UTexture2D& Texture)
130 {
131 FImage SourceImage;
132 if (!Texture.Source.GetMipImage(SourceImage, 0, 0, 0))
133 {
134 return false;
135 }
136
137 for (int32 Y = 0; Y < SourceImage.SizeY; ++Y)
138 {
139 for (int32 X = 0; X < SourceImage.SizeX; ++X)
140 {
141 if (SourceImage.GetOnePixelLinear(X, Y).A < 0.999f)
142 {
143 return true;
144 }
145 }
146 }
147 return false;
148 }

155bool FShopFullGeneratedTextureContractTest::RunTest(const FString& Parameters)
156{
157 using namespace ShopFullGeneratedUITests;
158 bool bAllValid = true;
159 TestEqual(TEXT("전용 생성 상점 텍스처 수"),
160 static_cast<int32>(UE_ARRAY_COUNT(ExpectedTextures)), 22);
161
162 for (const FExpectedTexture& Expected : ExpectedTextures)
163 {
164 const FString ObjectPath = TextureObjectPath(Expected);
165 UTexture2D* Texture = LoadObject<UTexture2D>(nullptr, *ObjectPath);
166 if (!TestNotNull(*FString::Printf(TEXT("전용 생성 텍스처: %s"),
167 *ObjectPath), Texture))
168 {
169 bAllValid = false;
170 continue;
171 }
172
173 const FIntPoint ImportedSize = Texture->GetImportedSize();
174 bAllValid &= TestEqual(*FString::Printf(TEXT("%s 폭"), Expected.Name),
175 ImportedSize.X, Expected.Width);
176 bAllValid &= TestEqual(*FString::Printf(TEXT("%s 높이"), Expected.Name),
177 ImportedSize.Y, Expected.Height);
178 bAllValid &= TestEqual(*FString::Printf(TEXT("%s UI LOD"), Expected.Name),
179 Texture->LODGroup, TEnumAsByte<TextureGroup>(TEXTUREGROUP_UI));
180 bAllValid &= TestEqual(*FString::Printf(TEXT("%s NoMipmaps"), Expected.Name),
181 Texture->MipGenSettings,
182 TEnumAsByte<TextureMipGenSettings>(TMGS_NoMipmaps));
183 bAllValid &= TestEqual(*FString::Printf(TEXT("%s UI 압축"), Expected.Name),
184 Texture->CompressionSettings,
185 TEnumAsByte<TextureCompressionSettings>(TC_EditorIcon));
186 bAllValid &= TestTrue(*FString::Printf(TEXT("%s sRGB"), Expected.Name),
187 Texture->SRGB);
188 bAllValid &= TestTrue(*FString::Printf(TEXT("%s NeverStream"), Expected.Name),
189 Texture->NeverStream);
190 bAllValid &= TestFalse(*FString::Printf(TEXT("%s 가상 텍스처 비활성"), Expected.Name),
191 Texture->VirtualTextureStreaming);
192 bAllValid &= TestEqual(*FString::Printf(TEXT("%s X Clamp"), Expected.Name),
193 Texture->AddressX, TEnumAsByte<TextureAddress>(TA_Clamp));
194 bAllValid &= TestEqual(*FString::Printf(TEXT("%s Y Clamp"), Expected.Name),
195 Texture->AddressY, TEnumAsByte<TextureAddress>(TA_Clamp));
196 if (Expected.bRequiresAlpha)
197 {
198 // HasAlphaChannel()은 플랫폼 GPU 포맷의 알파 지원 여부만 반환한다.
199 // UI 압축/NullRHI에서도 실제 소스 투명 픽셀이 보존됐는지 직접 확인한다.
200 bAllValid &= TestTrue(*FString::Printf(TEXT("%s 소스 투명 픽셀"),
201 Expected.Name), SourceContainsTransparentPixel(*Texture));
202 }
203 }
204 return bAllValid;
205}
