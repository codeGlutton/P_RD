67 TestTrue(TEXT("우세 축 Y(-1,-4)는 Left"), UTileMapModel::TileDeltaToDirection(FTileIndex(0, 0), FTileIndex(-1, -4), ETileActorDirection::Forward) == ETileActorDirection::Left);

70
71 return true;
72}

85bool FBoardActorMoveStepBarrierTests::RunTest(const FString& Parameters)
86{
87 UWorld* World = GetAnyGameWorld();
88 if (World == nullptr)
89 {
90 World = GWorld;
91 }
92 if (TestNotNull(TEXT("유효한 UWorld"), World) == false)
93 {
94 return false;
95 }
96
97 // OnStartMoveStep() 구독자를 없게 해서 시뮬레이션모드 테스트
98 AddInfo(TEXT("=== Case1: 구독자 없음 -> 시뮬레이션모드 -> 즉시 완료 ==="));
99 {
100 UMockPlayerUnitModel* Unit = NewObject<UMockPlayerUnitModel>(World);
101
102 bool bFinished = false;
103 {
104 TSharedPtr<FPresentationBarrier> Barrier = FPresentationBarrier::Make(
105 FOnFinishPresentation::CreateLambda([&bFinished]()
106 {
107 bFinished = true;
108 }));
109 Unit->OnStartMoveStep.Broadcast(FTileTransform(), FTransform::Identity, Barrier, 0.0f);
110
111 TestFalse(TEXT("배리어 참조가 남아있으면 미완료"), bFinished);
112 }
113 TestTrue(TEXT("스코프를 벗어나서 베리어 참조가 없으면 참조 소멸 시 즉시 완료"), bFinished);
114 }
115
116 // OnStartMoveStep 구독자를 만들어서 라이브모드 테스트
117 AddInfo(TEXT("=== Case2: 구독자 있음 -> 라이브모드 -> 구독자가 놓으면 완료 ==="));
118 {
119 UMockPlayerUnitModel* Unit = NewObject<UMockPlayerUnitModel>(World);
120
121 // 뷰 역할의 구독자 (Barrier를 보관하고 있어서 참조가 0이 안 되게 유지)
122 TSharedPtr<FPresentationBarrier> HeldBarrier;
123 Unit->OnStartMoveStep.AddLambda(
124 [&HeldBarrier](const FTileTransform&, const FTransform&, TSharedPtr<FPresentationBarrier> Barrier, float) {
125 HeldBarrier = Barrier;
126 });
127
128 bool bFinished = false;
129 {
130 TSharedPtr<FPresentationBarrier> Barrier = FPresentationBarrier::Make(
131 FOnFinishPresentation::CreateLambda([&bFinished]()
132 {
133 bFinished = true;
134 }));
135 Unit->OnStartMoveStep.Broadcast(FTileTransform(), FTransform::Identity, Barrier, 0.0f);
136 }
137
138 TestTrue(TEXT("구독자가 배리어 수신"), HeldBarrier.IsValid());
139 TestFalse(TEXT("발행자는 사라졌지만, 구독자가 잡고있는 동안은 미완료"), bFinished);
140
141 // 구독자도 베리어를 놓음
142 HeldBarrier.Reset();
143 TestTrue(TEXT("구독자도 베리어 놓으면 콜백 호출되면서 완료로 변경"), bFinished);
144 }
145
146 return true;
147}
