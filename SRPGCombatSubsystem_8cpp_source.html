163bool USRPGCombatSubsystem::UnregisterTurnImmediately(TSharedRef<FSRPGTurnContext> TurnContext)
164{
165 auto* CurNode = mTurnContexts.FindNode(TurnContext);
166 if (CurNode != nullptr)
167 {
168 // 현재 진행 중인 턴을 삭제하게 되어 노드 전환이 필요한지 여부
169 const bool NeedToChangeTurnContext = mCurTurnContextNode->GetValue() == TurnContext;
170
171 auto* NextNode = mTurnContexts.RemoveNode(CurNode);
172 if (NeedToChangeTurnContext == true)
173 {
174 mCurTurnContextNode = NextNode;
175 }
176 return true;
177 }
178 return false;

181int32 USRPGCombatSubsystem::UnregisterTurnsImmediately(const AUnit* Owner)
182{
183 auto* CurNode = mTurnContexts.GetHead();
184
185 int32 UnregisterCount = 0;
186 const int32 TotalContextNum = mTurnContexts.Num();
187 for (int32 i = 0; i < TotalContextNum; ++i)
188 {
189 auto* NextNode = CurNode->GetNextNode();
190
191 TSharedPtr<FSRPGTurnContext> CurTurnContext = CurNode->GetValue();
192 if (CurTurnContext->GetOwner() == Owner)
193 {
194 // 현재 진행 중인 턴을 삭제하게 되어 노드 전환이 필요한지 여부
195 const bool NeedToChangeTurnContext = mCurTurnContextNode->GetValue() == CurTurnContext;
196
197 NextNode = mTurnContexts.RemoveNode(CurNode);
198 if (NeedToChangeTurnContext == true)
199 {
200 mCurTurnContextNode = NextNode;
201 }
202 ++UnregisterCount;
203 }
204
205 CurNode = NextNode;
206 }
207
208 return UnregisterCount;
209}

211void USRPGCombatSubsystem::FlushPendingTurnRequests()
212{
213 while (mPendingTurnRequests.IsEmpty() == false)
214 {
215 FSRPGTurnUnregisterRequest* Request = mPendingTurnRequests.Peek();
216
217 TSharedPtr<FSRPGTurnContext>& TargetTurnContext = Request->mTargetTurnContext;
218 if (TargetTurnContext != nullptr)
219 {
220 UnregisterTurnImmediately(TargetTurnContext.ToSharedRef());
221 }
222 const AUnit* TargetOwner = Request->mTargetOwner;
223 if (TargetOwner != nullptr)
224 {
225 UnregisterTurnsImmediately(TargetOwner);
226 }
227
228 mPendingTurnRequests.Pop();
229 }
230}

373void USRPGCombatSubsystem::EvaluateCombatEndState()
374{
375 checkf(mPlayerUnit != nullptr, TEXT("플레이어 미동록 오류"));
376
377 /* 이미 중단 */
378
379 if (mPhase == ECombatRoomPhase::CombatAbort)
380 {
381 return;
382 }
383
384 /* 플레이어가 죽어서 전투가 종료되는가? */
385
386 if (mPlayerUnit->IsDead() == true)
387 {
388 mResult = ECombatResult::PlayerLose;
389 mPhase = ECombatRoomPhase::CombatAbort;
390 return;
391 }
392
393 /* 적군이 모두 죽어 전투가 종료되는가? */
394
395 bool AnyEnemyAlive = false;
396 for (const TObjectPtr<AUnit>& Unit : mUnits)
397 {
398 if (Unit->GetTeamAttitudeTowards(*mPlayerUnit) == ETeamAttitude::Hostile && Unit->IsDead() == false)
399 {
400 AnyEnemyAlive = true;
401 }
402 }
403 if (AnyEnemyAlive == false)
404 {
405 mResult = ECombatResult::PlayerWin;
406 mPhase = ECombatRoomPhase::CombatAbort;
407 return;
408 }
