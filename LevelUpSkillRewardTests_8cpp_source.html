12IMPLEMENT_SIMPLE_AUTOMATION_TEST(FLevelUpSkillCandidateTest,
13 "P_RD.Reward.LevelUp.Candidates", EAutomationTestFlags::EditorContext | EAutomationTestFlags::ProductFilter)
14bool FLevelUpSkillCandidateTest::RunTest(const FString& Parameters)
15{
16 TArray<UStaticUnitSkillData*> Pool;
17 for (int32 Index = 0; Index < 6; ++Index)
18 {
19 UStaticUnitSkillData* Skill = NewObject<UStaticUnitSkillData>(GetTransientPackage(),
20 *FString::Printf(TEXT("RewardCandidate%d"), Index));
21 Skill->mRarityType = Index < 3 ? ERarityType::Common : ERarityType::Rare;
22 Pool.Add(Skill);
23 }
24 FRarityRate Rate;
25 Rate.mWeights[static_cast<uint8>(ERarityType::Common)] = 1.f;
26 UStaticUnitSkillData* Fixture = NewObject<UStaticUnitSkillData>(GetTransientPackage(), TEXT("DA_TestKnight_Spell_Common"));
27 TestTrue(TEXT("Prototype skills cannot return through the restored level-up reward pool"),
28 LevelUpSkillReward::ChooseCandidates(TArray<UStaticUnitSkillData*>{Fixture}, Rate, FRandomStream(17)).IsEmpty());
29 const auto First = LevelUpSkillReward::ChooseCandidates(Pool, Rate, FRandomStream(17));
30 TestEqual(TEXT("Three choices"), First.Num(), 3);
31 TSet<FPrimaryAssetId> Unique(First);
32 TestEqual(TEXT("No duplicate candidate"), Unique.Num(), 3);
33 for (const auto& Id : First)
34 TestTrue(TEXT("Available weighted rarity honored"), Id == Pool[0]->GetPrimaryAssetId()
35 || Id == Pool[1]->GetPrimaryAssetId() || Id == Pool[2]->GetPrimaryAssetId());
36 TestTrue(TEXT("Same seed restores identical offer"), First == LevelUpSkillReward::ChooseCandidates(Pool, Rate, FRandomStream(17)));
37 Pool.SetNum(2);
38 TestEqual(TEXT("Small pool never fabricates or repeats a skill"),
39 LevelUpSkillReward::ChooseCandidates(Pool, FRarityRate(), FRandomStream(1)).Num(), 2);
40 Pool.Reset();
41 TestTrue(TEXT("Exhausted pool is safe"), LevelUpSkillReward::ChooseCandidates(Pool, Rate, FRandomStream(1)).IsEmpty());
42 return true;
43}

45IMPLEMENT_SIMPLE_AUTOMATION_TEST(FLevelUpSkillEquipTest,
46 "P_RD.Reward.LevelUp.Equip", EAutomationTestFlags::EditorContext | EAutomationTestFlags::ProductFilter)
47bool FLevelUpSkillEquipTest::RunTest(const FString& Parameters)
48{
49 TStrongObjectPtr<ULevelUpSkillTestPlayer> Unit(NewObject<ULevelUpSkillTestPlayer>());
50 USkillComponentModel* Skills = Unit->GetSkillComponentModel();
51 Skills->SetSkillFrom(TArray<FPrimaryAssetId>());
52 UAssetManager& Manager = UAssetManager::Get();
53 TArray<FPrimaryAssetId> Ids;
54 Manager.GetPrimaryAssetIdList(SkillPrimaryAssetTypes::GetActiveType(), Ids);
55 TArray<UStaticUnitSkillData*> Eligible;
56 UStaticUnitSkillData* OtherJob = nullptr;
57 UStaticUnitSkillData* CommonJob = nullptr;
58 for (const auto& Id : Ids)
59 {
60 auto* Skill = Cast<UStaticUnitSkillData>(Manager.GetPrimaryAssetPath(Id).TryLoad());
61 if (!Skill || Skill->mSkillPhaseLayers.IsEmpty()) continue;
62 if (LevelUpSkillReward::IsEligibleForUnit(Unit.Get(), Skill)) Eligible.Add(Skill);
63 else if (Skill->mJobType == EUnitJobType::Common) CommonJob = Skill;
64 else if (!Skills->IsAcquirableSkill(Skill)) OtherJob = Skill;
65 }
66 if (!TestTrue(TEXT("Real knight skill pool available"), Eligible.Num() >= 3)
67 || !TestNotNull(TEXT("Other-job skill available"), OtherJob)
68 || !TestNotNull(TEXT("Common-job skill available"), CommonJob)) return false;
69 Skills->SetSkill(0, Eligible[0]);
70 FLevelUpSkillReward Reward;
71 Reward.Offered = true;
72 Reward.Candidates = { CommonJob->GetPrimaryAssetId() };
73 TestTrue(TEXT("Common skills remain acquirable outside level-up rewards"), Skills->IsAcquirableSkill(CommonJob));
74 TestFalse(TEXT("Old common-job offer cannot be equipped"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[0], 1));
75 TestFalse(TEXT("Rejected common skill does not consume reward"), Reward.Completed);
76 Reward.Candidates = { Eligible[1]->GetPrimaryAssetId(), Eligible[2]->GetPrimaryAssetId(), OtherJob->GetPrimaryAssetId() };
77 TestFalse(TEXT("Basic attack cannot be overwritten"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[0], 0));
78 TestFalse(TEXT("Out-of-range slot rejected"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[0], 6));
79 TestFalse(TEXT("Other-job candidate rejected by gameplay"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[2], 1));
80 TestFalse(TEXT("Unoffered skill rejected"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Eligible[0]->GetPrimaryAssetId(), 1));
81 TestTrue(TEXT("Chosen skill actually equipped"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[0], 1));
82 TestTrue(TEXT("Correct skill in model"), Skills->GetSkill(1)->mData == Eligible[1]);
83 TestFalse(TEXT("Same reward cannot grant twice"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[1], 2));
84 Reward.Completed = false;
85 TestFalse(TEXT("Next reward cannot duplicate an owned skill"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[0], 2));
86 TestTrue(TEXT("Full slot can be explicitly replaced"), LevelUpSkillReward::TryEquip(Reward, Unit.Get(), Reward.Candidates[1], 1));
87 TestTrue(TEXT("Basic attack preserved"), Skills->GetSkill(0)->mData == Eligible[0]);
88 return true;
89}
