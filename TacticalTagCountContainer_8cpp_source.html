192 checkf(Tag.IsValid() == true, TEXT("변동되는 태그가 없음"));
193 checkf(CountDelta != 0, TEXT("태그 변동 수가 0개"));

250bool FTacticalTagCountContainer::GatherTagChangeDelegates(const FGameplayTag& Tag, int32 CountDelta, TArray<FDeferredTagChangeDelegate>& TagChangeDelegates)
251{
252 FGameplayTagContainer TagAndParentsContainer = Tag.GetGameplayTagParents();
253 TagAndParentsContainer.AddTag(Tag);
254 bool CreatedSignificantChange = false;
255 for (auto CompleteTagIt = TagAndParentsContainer.CreateConstIterator(); CompleteTagIt; ++CompleteTagIt)
256 {
257 /* 부모 자식 모두 태그 갱신 */
258
259 const FGameplayTag& CurTag = *CompleteTagIt;
260 int32& TagCountRef = mTagCountMap.FindOrAdd(CurTag);
261
262 const int32 OldCount = TagCountRef;
263 int32 NewTagCount = FMath::Max(OldCount + CountDelta, 0);
264
265 TagCountRef = NewTagCount;
266
267 /* 대리자 호출 부분 */
268
269 const bool SignificantChange = (OldCount == 0 || NewTagCount == 0);
270 CreatedSignificantChange |= SignificantChange;
271 if (SignificantChange)
272 {
273 /* 추가 삭제 대리자 호출 필요 */
274
275 TagChangeDelegates.AddDefaulted();
276 TagChangeDelegates.Last().BindLambda([Delegate = OnAnyTagChangeDelegate, CurTag, NewTagCount]()
277 {
278 Delegate.Broadcast(CurTag, NewTagCount);
279 });
280 }
281
282 FDelegateInfo* DelegateInfo = mTagEventMap.Find(CurTag);
283 if (DelegateInfo != nullptr)
284 {
285 TagChangeDelegates.AddDefaulted();
286 TagChangeDelegates.Last().BindLambda([Delegate = DelegateInfo->OnAnyChange, CurTag, NewTagCount]()
287 {
288 Delegate.Broadcast(CurTag, NewTagCount);
289 });
290
291 if (SignificantChange == true)
292 {
293 /* 추가 삭제 대리자 호출 필요 */
294
295 TagChangeDelegates.AddDefaulted();
296 TagChangeDelegates.Last().BindLambda([Delegate = DelegateInfo->OnNewOrRemove, CurTag, NewTagCount]()
297 {
298 Delegate.Broadcast(CurTag, NewTagCount);
299 });
300 }
301 }
302 }
303
304 return CreatedSignificantChange;
