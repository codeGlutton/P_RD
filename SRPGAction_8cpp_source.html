26void USRPGAction::BeginAction()
27{

30
31 UE_LOG(LogSRPGCombat, Log, TEXT("액션 시작: %s (%s)"),
32 *GetClass()->GetName(),
33 mInstigator.IsValid() ? *mInstigator->GetName() : TEXT("?"));
34
35 // 시작 연출 시작
36 TSharedPtr<FPresentationBarrier> PresentationBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
37 checkf(mActionPhase == ESRPGActionPhase::ActionStart, TEXT("액션 실행 절차 오류"));
38 mActionPhase = ESRPGActionPhase::ActionPlay;
39
40 // 핸들러 등록
41 USRPGCommandRouterModel* CommandRouterModel = GetWorldSubsystemModel<USRPGCommandRouterModel>(this);
42 checkf(CommandRouterModel != nullptr, TEXT("명령 라우터 모델 nullptr"));
43 CommandRouterModel->RegisterCommandHandler(this);
44
45 // 로그 작성
46 GetWorldEventLogger(this)->BeginActionLog(mInstigator->GetTileTransform().mIndex);
47
48 // 예약된 초기화 커맨드 시작
49 if (mInitializeCommand.IsValid() == true)

52 mInitializeCommand.Reset();
53 }
54
55 // 액션 시작 로직

57 }));
