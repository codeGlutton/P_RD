59void FSRPGAction::EndAction(ESRPGActionResult Result)
60{
61 if (mActionPhase == ESRPGActionPhase::ActionPlay)
62 {
63 /* 정상 종료 시, 원하는 결과로 반영 */
64
65 mActionPhase = ESRPGActionPhase::ActionAbort;
66 mActionResult = Result;
67 }
68
69 checkf(mActionPhase == ESRPGActionPhase::ActionAbort, TEXT("액션 종료 절차 오류"));
70 mActionPhase = ESRPGActionPhase::ActionEnd;
71
72 OnEndAction();
73
74 USRPGCombatSubsystem* CombatSubsystem = GetWorld()->GetSubsystem<USRPGCombatSubsystem>();
75 checkf(CombatSubsystem != nullptr, TEXT("전투 서브시스템 nullptr"));
76
77 // 전투 상태 평가
78 CombatSubsystem->EvaluateCombatStates();
79
80 UPresentationSyncSubsystem* PresentationSyncSubsystem = GetWorld()->GetSubsystem<UPresentationSyncSubsystem>();
81 checkf(PresentationSyncSubsystem != nullptr, TEXT("연출 동기화 서브시스템 nullptr"));
82
83 auto PresentationBarrier = PresentationSyncSubsystem->MakePresentationBarrier(FOnFinishPresentation::CreateSPLambda(AsShared(), [this]() {
84 TSharedPtr<FSRPGTurnContext> TurnContext = mParent.Pin();
85 checkf(TurnContext != nullptr, TEXT("이미 제거된 턴에서 Action 종료 명령 오류"));
86 TurnContext->OnEndCurrentAction(AsShared(), mActionResult);
87 }));
88 OnEndActionUI.Broadcast(PresentationBarrier, *this, mActionResult);
89}

141{
142 Actor = nullptr;
143 TileIndex = FTileIndex::Invalid;

145 /* 마우스 포인트 지점 아래로 Raycast 검사 */

149
150 APlayerController* PlayerController = GetWorld()->GetFirstPlayerController();
151 if (PlayerController != nullptr)
152 {
153 FHitResult HitResult;
154 if (PlayerController->GetHitResultUnderCursor(Channel, false, HitResult) == true)

161 else

164 if (TileActor != nullptr)

166 TileIndex = TileActor->GetTileTransform().mIndex;
