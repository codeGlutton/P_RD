6ACombatCameraPlane::ACombatCameraPlane()

10 mPlaneMesh = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("PlaneMesh"));
11 mPlaneMesh->SetRelativeScale3D(FVector(10000, 10000, 0));

14 // 기본 메시 에셋 로드 (엔진 기본 제공 Plane)
15 static ConstructorHelpers::FObjectFinder<UStaticMesh> PlaneMeshAsset(
16 TEXT("/Engine/BasicShapes/Plane.Plane")
17 );
18 if (PlaneMeshAsset.Succeeded())
19 {
20 mPlaneMesh->SetStaticMesh(PlaneMeshAsset.Object);
21 }
22
23 // 평소엔 화면에 안 보이게, 콜리전도 꺼둠 (필요시 CameraMovementComponent에서 켬)
24 mPlaneMesh->SetVisibility(false);
25
26 // 1. 콜리전 활성화 방식
27 mPlaneMesh->SetCollisionEnabled(ECollisionEnabled::QueryOnly); // 물리 시뮬레이션 없이 쿼리(트레이스)만
28
29 // 2. Object Type 지정 (커스텀 채널)

31
32 // 3. 모든 채널에 대한 기본 반응을 Ignore로 초기화 (깔끔하게 시작)

34
35 // 4. CameraMove 채널에 대해서만 Block으로 오버라이드
