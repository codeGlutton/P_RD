70void ULoadingNotifyWidget::NativeTick(const FGeometry& MyGeometry, float InDeltaTime)
71{
72 Super::NativeTick(MyGeometry, InDeltaTime);
73
74 // New room rails can be created during the Completed/closing interval too.
75 if (mLoadingState != ELoadingNotifyState::None)
76 {
77 SuppressRunOptionsRails();
78 }
79
80 if (mLoadingState != ELoadingNotifyState::Loading)
81 {
82 return;
83 }
84
85 if (mLoadingIndicator == nullptr)
86 {
87 return;
88 }
89
90 mIndicatorElapsedSeconds += InDeltaTime;
91 const float BlinkRate = FMath::Max(0.01f, mIndicatorBlinkSpeed);
92 const float WaveAlpha = 0.5f + 0.5f * FMath::Sin(mIndicatorElapsedSeconds * BlinkRate * 2.0f * UE_PI);
93 ApplyIndicatorAlpha(0.35f + 0.65f * WaveAlpha);
94}

102void ULoadingNotifyWidget::PlayOpenUIAnimation_Implementation()

128void ULoadingNotifyWidget::PlayCloseUIAnimation_Implementation()
129{
130 UWorld* World = GetWorld();
131 if (World == nullptr)
132 {
133 ShowCompletedState();
134 return;
135 }
136
137 const double ElapsedSeconds = FMath::Max(0.0, World->GetTimeSeconds() - mOpenedTimeSeconds);
138 const float RemainingSeconds = FMath::Max(0.0f, mMinimumVisibleSeconds - StaticCast<float>(ElapsedSeconds));
139 if (RemainingSeconds > 0.0f)
140 {
141 World->GetTimerManager().SetTimer(
142 mMinimumVisibleTimerHandle,
143 this,
144 &ULoadingNotifyWidget::ShowCompletedState,
145 RemainingSeconds,
146 false);
147 return;
148 }
149
150 ShowCompletedState();
151}

284void ULoadingNotifyWidget::SuppressRunOptionsRails()
285{
286 UWorld* ThisWorld = GetWorld();
287 for (TObjectIterator<URunOptionsRailWidget> It; It; ++It)
288 {
289 URunOptionsRailWidget* Rail = *It;
290 if (Rail == nullptr || Rail->GetWorld() != ThisWorld
291 || Rail->GetVisibility() == ESlateVisibility::Collapsed)
292 {
293 continue;
294 }
295
296 if (!mSuppressedRunOptionsRails.Contains(Rail))
297 {
298 mSuppressedRunOptionsRails.Add(Rail, Rail->GetVisibility());
299 }
300 Rail->SetVisibility(ESlateVisibility::Collapsed);
301 }
302}
303
304void ULoadingNotifyWidget::RestoreRunOptionsRails()
305{
306 for (const TPair<TWeakObjectPtr<URunOptionsRailWidget>, ESlateVisibility>& Entry
307 : mSuppressedRunOptionsRails)
308 {
309 if (URunOptionsRailWidget* Rail = Entry.Key.Get())
310 {
311 Rail->SetVisibility(Entry.Value);
312 }
313 }
314 mSuppressedRunOptionsRails.Reset();
315}
