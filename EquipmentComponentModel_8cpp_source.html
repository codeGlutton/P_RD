23 UStaticEquipmentData* LoadStaticEquipmentData(const FPrimaryAssetId& EquipmentId)
24 {
25 if (EquipmentId.IsValid() == false)
26 {
27 return nullptr;
28 }
29
30 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
31 if (AssetManager == nullptr)
32 {
33 return nullptr;
34 }
35
36 if (UStaticEquipmentData* Loaded = AssetManager->GetPrimaryAssetObject<UStaticEquipmentData>(EquipmentId))
37 {
38 return Loaded;
39 }
40
41 const FSoftObjectPath AssetPath = AssetManager->GetPrimaryAssetPath(EquipmentId);
42 return Cast<UStaticEquipmentData>(AssetPath.TryLoad());
43 }

106bool UEquipmentComponentModel::EquipInternal(UStaticEquipmentData* Data, UPassiveComponentModel* PassiveComp, UAttributeSetComponentModel* AttrComp)
107{
108 if (Data == nullptr)
109 {
110 return false;
111 }
112
113 // 슬롯 결정, 이미 점유 중이면 먼저 해제(교체)
114 const EEquipmentType Slot = Data->GetEquipmentType();
115 if (Slot == EEquipmentType::Count)
116 {
117 return false;
118 }
119 if (mEquipped.Contains(Slot))
120 {
121 UnequipInternal(Slot, PassiveComp, AttrComp);
122 }
123
124 FEquippedEntry Entry;
125 Entry.mData = Data;
126
127 // 패시브 설치: 장비가 참조하는 패시브 DA를 로드해 PassiveComponent에 설치
128 if (PassiveComp != nullptr)
129 {
130 for (const TSoftObjectPtr<UStaticPassiveData>& PassivePtr : Data->mStaticPassiveData)
131 {
132 UStaticPassiveData* PassiveData = PassivePtr.LoadSynchronous();
133 if (PassiveData == nullptr)
134 {
135 continue;
136 }
137 if (UTacticalPassive* Installed = PassiveComp->AddPassiveFromData(PassiveData))
138 {
139 Entry.mInstalledPassives.Add(Installed);
140 }
141 }
142 }
143
144 // 장비 고유 스탯이 있으면 무한 이펙트로 자신에게 적용 (해제 시 핸들로 제거)
145 if (AttrComp != nullptr && Data->mStatModifiers.Num() > 0)
146 {
147 // 데이터의 수정자들을 런타임 이펙트 인스턴스에 담아 무한 정책으로 구성
148 UTacticalEffect* StatEffect = NewObject<UTacticalEffect>(this);
149 StatEffect->mModifiers = Data->mStatModifiers;
150 StatEffect->mDurationPolicy = ETacticalEffectDurationType::Infinite;
151
152 // 소유자를 시전자로 컨텍스트 구성 후 self 적용, 인스턴스·핸들 보관
153 UTacticalEffectContext* EffectContext = AttrComp->MakeEffectContext();
154 EffectContext->SetAttributeSetComponentModel(AttrComp);
155
156 FTacticalEffectSpec Spec(StatEffect, EffectContext);
157 Entry.mStatEffect = StatEffect;
158 Entry.mStatEffectHandle = AttrComp->ApplyTacticalEffectSpecToSelf(Spec);
159 }
160
161 mEquipped.Add(Slot, MoveTemp(Entry));
162 return true;
163}

165bool UEquipmentComponentModel::UnequipInternal(EEquipmentType Slot, UPassiveComponentModel* PassiveComp, UAttributeSetComponentModel* AttrComp)
166{
167 FEquippedEntry* Entry = mEquipped.Find(Slot);
168 if (Entry == nullptr)
169 {
170 return false;
171 }
172
173 // 설치한 패시브 해제 (적용 중 이펙트가 있으면 내리고 보유 목록에서 제거)
174 if (PassiveComp != nullptr)
175 {
176 for (const TObjectPtr<UTacticalPassive>& Installed : Entry->mInstalledPassives)
177 {
178 if (Installed != nullptr)
179 {
180 Installed->DeactivatePassive();
181 PassiveComp->RemovePassive(Installed);
182 }
183 }
184 }
185
186 // 장비 고유 스탯 이펙트 제거 (적용돼 있던 경우)
187 if (Entry->mStatEffectHandle.IsValid() && AttrComp != nullptr)
188 {
189 AttrComp->RemoveActiveTacticalEffect(Entry->mStatEffectHandle);
190 }
191 Entry->mStatEffectHandle.Reset();
192
193 mEquipped.Remove(Slot);
194 return true;
195}
