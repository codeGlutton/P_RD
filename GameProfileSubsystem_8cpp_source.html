34 TArray<uint8> PreviousRun, PreviousUser;

37 const FRunLog CompletedLog = RunMutableData->GetRunLog();
38 RunMutableData->ClearRun();
39 RunMutableData->QueueCompletedRunLog(CompletedLog);
40 USaveGameSubsystem* Saver = GetGameInstance()->GetSubsystem<USaveGameSubsystem>();
41 if (!Saver->SaveEndedRun(PreviousRun))

43 RDCheckpoint::Deserialize(PreviousRun, RunMutableData);

46 }
47 Saver->CommitPendingRunLogs();
