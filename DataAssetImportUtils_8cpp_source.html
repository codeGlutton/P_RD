29FString FImportCsvTable::NormalizeColumn(const FString& Name)
30{
31 // 괄호 안은 삭제하고 공백도 전부 제거
32 FString Result;
33 int32 Depth = 0;
34 for (const TCHAR Ch : Name)
35 {
36 if (Ch == TEXT('('))
37 {
38 ++Depth;
39 }
40 else if (Ch == TEXT(')'))
41 {
42 Depth = FMath::Max(0, Depth - 1);
43 }
44 else if (Depth == 0 && FChar::IsWhitespace(Ch) == false)
45 {
46 Result.AppendChar(Ch);
47 }
48 }
49 return Result;
50}

52bool DataAssetImportUtils::LoadCsv(const FString& FilePath, FImportCsvTable& OutTable)
53{
54 FString Content;
55 if (FFileHelper::LoadFileToString(Content, *FilePath) == false)
56 {
57 UE_LOG(LogDataAssetImport, Error, TEXT("CSV 파일을 열 수 없음: %s"), *FilePath);
58 return false;
59 }
60
61 // 구분자: .tsv면 탭, 아니면 쉼표
62 const TCHAR Delimiter = FilePath.EndsWith(TEXT(".tsv")) ? TEXT('\t') : TEXT(',');
63
64 // 문자 단위로 검사해서 따옴표를 기준으로 셀과 행으로 구분
65 TArray<TArray<FString>> Records;
66 TArray<FString> Cells;
67 FString Cell;
68 bool bInQuotes = false;
69 for (int32 Index = 0; Index < Content.Len(); ++Index)
70 {
71 const TCHAR Ch = Content[Index];
72
73 // 따옴표 안: 쉼표와 줄바꿈도 셀 값. 따옴표 두 개는 따옴표 문자 하나, 홀로 있으면 닫힘
74 if (bInQuotes)
75 {
76 if (Ch != TEXT('"'))
77 {
78 Cell.AppendChar(Ch);
79 }
80 else if (Index + 1 < Content.Len() && Content[Index + 1] == TEXT('"'))
81 {
82 Cell.AppendChar(TEXT('"'));
83 ++Index;
84 }
85 else
86 {
87 bInQuotes = false;
88 }
89 continue;
90 }
91
92 // 따옴표 밖: 여는 따옴표, 구분자, 줄바꿈을 처리. 줄바꿈은 \n만 쓰고 \r은 건너뜀
93 if (Ch == TEXT('"'))
94 {
95 bInQuotes = true;
96 }
97 else if (Ch == Delimiter)
98 {
99 Cells.Add(Cell);
100 Cell.Reset();
101 }
102 else if (Ch == TEXT('\n'))
103 {
104 Cells.Add(Cell);
105 Cell.Reset();
106 Records.Add(MoveTemp(Cells));
107 Cells.Reset();
108 }
109 else if (Ch != TEXT('\r'))
110 {
111 Cell.AppendChar(Ch);
112 }
113 }
114
115 // 파일 끝에 줄바꿈이 없으면 남은 셀을 마지막 행으로 마감
116 if (Cell.Len() > 0 || Cells.Num() > 0)
117 {
118 Cells.Add(Cell);
119 Records.Add(MoveTemp(Cells));
120 }
121
122 // 첫 행이 헤더니까 이름을 확인해서 컬럼 번호와 매핑
123 if (Records.Num() == 0)
124 {
125 UE_LOG(LogDataAssetImport, Error, TEXT("CSV에 헤더가 없음: %s"), *FilePath);
126 return false;
127 }
128 OutTable.mColumnIndex.Reset();
129 for (int32 Index = 0; Index < Records[0].Num(); ++Index)
130 {
131 OutTable.mColumnIndex.Add(FImportCsvTable::NormalizeColumn(Records[0][Index]), Index);
132 }
133
134 // 나머지 행은 데이터만 있음. 모든 셀이 빈 행은 버림
135 OutTable.mRows.Reset();
136 for (int32 Index = 1; Index < Records.Num(); ++Index)
137 {
138 const bool bHasValue = Records[Index].ContainsByPredicate([](const FString& Cell) { return Cell.TrimStartAndEnd().Len() > 0; });
139 if (bHasValue)
140 {
141 OutTable.mRows.Add(MoveTemp(Records[Index]));
142 }
143 }
144 return true;
145}

147TMap<FString, FString> DataAssetImportUtils::ParseLabels(const FString& Text)
148{
149 TMap<FString, FString> Labels;
150
151 // 줄마다 "라벨 : 값"으로 해석
152 TArray<FString> Lines;
153 Text.ParseIntoArrayLines(Lines);
154 for (const FString& RawLine : Lines)
155 {
156 const FString Line = RawLine.TrimStartAndEnd();
157 if (Line.IsEmpty())
158 {
159 continue;
160 }
161
162 // 콜론이 있으면 콜론, 없으면 첫 공백을 기준으로 라벨과 값을 나눔. 둘 다 없으면 숫자 하나로 보고 고정값
163 FString Label;
164 FString Value;
165 if (Line.Split(TEXT(":"), &Label, &Value) == false && Line.Split(TEXT(" "), &Label, &Value) == false)
166 {
167 Label = TEXT("고정값");
168 Value = Line;
169 }
170 Labels.Add(Label.TrimStartAndEnd(), Value.TrimStartAndEnd());
171 }
172 return Labels;
173}

205bool DataAssetImportUtils::ParseOperand(const FString& KindText, const FString& LabelText, FPassiveOperand& OutOperand)
206{
207 OutOperand = FPassiveOperand();
208
209 // 종류 이름 -> enum. 노션 표기가 "캡처"와 "캡쳐"로 섞여 있어 둘 다 허용
210 static const TMap<FString, EPassiveOperandKind> KindByName = {
211 { TEXT("고정값"), EPassiveOperandKind::Const },
212 { TEXT("속성값"), EPassiveOperandKind::Attribute },
213 { TEXT("태그 개수"), EPassiveOperandKind::TagCount },
214 { TEXT("카운터"), EPassiveOperandKind::Counter },
215 { TEXT("거리"), EPassiveOperandKind::Distance },
216 { TEXT("타겟 수"), EPassiveOperandKind::TargetCount },
217 { TEXT("캡처값"), EPassiveOperandKind::Captured },
218 { TEXT("캡쳐값"), EPassiveOperandKind::Captured },
219 { TEXT("이동 거리"), EPassiveOperandKind::MovedDistance },
220 { TEXT("팀 관계"), EPassiveOperandKind::Team },
221 };
222 const EPassiveOperandKind* Kind = KindByName.Find(KindText.TrimStartAndEnd());
223 if (Kind == nullptr)
224 {
225 UE_LOG(LogDataAssetImport, Error, TEXT("알 수 없는 피연산자 종류: %s"), *KindText);
226 return false;
227 }
228 OutOperand.mKind = *Kind;
229
230 // 공통 라벨: 출처(자신/대상), 배수
231 const TMap<FString, FString> Labels = ParseLabels(LabelText);
232 if (const FString* Source = Labels.Find(TEXT("출처")))
233 {
234 OutOperand.mSource = (*Source == TEXT("대상")) ? EPassiveOperandSource::Target : EPassiveOperandSource::Self;
235 }
236 if (const FString* Multiplier = Labels.Find(TEXT("배수")))
237 {
238 OutOperand.mMultiplier = FCString::Atof(**Multiplier);
239 }
240
241 // 종류별 필드: 고정값 / 속성 / 태그 / 캡처 키. 나머지 종류는 공통 라벨만으로 충분
242 switch (OutOperand.mKind)
243 {
244 case EPassiveOperandKind::Const:
245 {
246 const FString* Value = Labels.Find(TEXT("고정값"));
247 if (Value == nullptr)
248 {
249 UE_LOG(LogDataAssetImport, Error, TEXT("고정값이 없음: %s"), *LabelText);
250 return false;
251 }
252 OutOperand.mConst = FCString::Atof(**Value);
253 }
254 break;
255
256 case EPassiveOperandKind::Attribute:
257 {
258 const FString* Value = Labels.Find(TEXT("속성"));
259 if (Value == nullptr || ParseAttribute(*Value, OutOperand.mAttribute) == false)
260 {
261 UE_LOG(LogDataAssetImport, Error, TEXT("속성이 없거나 잘못됨: %s"), *LabelText);
262 return false;
263 }
264 }
265 break;
266
267 case EPassiveOperandKind::TagCount:
268 {
269 const FString* Value = Labels.Find(TEXT("태그"));
270 OutOperand.mTag = (Value != nullptr) ? FGameplayTag::RequestGameplayTag(FName(**Value), false) : FGameplayTag();
271 if (OutOperand.mTag.IsValid() == false)
272 {
273 UE_LOG(LogDataAssetImport, Error, TEXT("태그가 없거나 잘못됨: %s"), *LabelText);
274 return false;
275 }
276 }
277 break;
278
279 case EPassiveOperandKind::Captured:
280 {
281 const FString* Value = Labels.Find(TEXT("캡처 키"));
282 if (Value == nullptr)
283 {
284 Value = Labels.Find(TEXT("캡쳐 키"));
285 }
286 if (Value == nullptr)
287 {
288 UE_LOG(LogDataAssetImport, Error, TEXT("캡처 키가 없음: %s"), *LabelText);
289 return false;
290 }
291 OutOperand.mCaptureKey = FName(**Value);
292 }
293 break;
294
295 default:
296 break;
297 }
298 return true;
299}

340bool DataAssetImportUtils::ParseAttribute(const FString& Text, FTacticalAttribute& OutAttribute)
341{
342 // 예) "CombatTargetAttributeSet.HP"면 클래스와 속성으로 나누고, "HP"만 있으면 클래스는 비움
343 FString ClassName;
344 FString PropertyName;
345 if (Text.TrimStartAndEnd().Split(TEXT("."), &ClassName, &PropertyName) == false)
346 {
347 PropertyName = Text.TrimStartAndEnd();
348 }
349
350 // 등록된 속성 전체에서 이름이 맞는 것을 찾음. 클래스가 없으면 속성명만 맞는 게 하나여야 함
351 // 노션에 UCombatTargetAttributeSet처럼 U를 붙여 적어도 되게, 리플렉션 이름과 U를 붙인 이름 둘 다 허용
352 TArray<FProperty*> Properties;
353 FTacticalAttribute::GetAllAttributeProperties(Properties);
354 TArray<FProperty*> Matched;
355 for (FProperty* Property : Properties)
356 {
357 const FString OwnerName = GetNameSafe(Property->GetOwnerStruct());
358 const bool bClassMatched = ClassName.IsEmpty() || ClassName == OwnerName || ClassName == TEXT("U") + OwnerName;
359 if (bClassMatched && Property->GetName() == PropertyName)
360 {
361 Matched.Add(Property);
362 }
363 }
364 if (Matched.Num() != 1)
365 {
366 UE_LOG(LogDataAssetImport, Error, TEXT("속성을 찾을 수 없거나 여러 개 일치: %s (%d개)"), *Text, Matched.Num());
367 return false;
368 }
369 OutAttribute.SetUProperty(Matched[0]);
370 return true;
371}
