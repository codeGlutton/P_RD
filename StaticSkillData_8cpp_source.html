100FText UStaticSkillData::MakeDescription() const
101{
102 TArray<FText> DescriptionLines;
103
104 /* 1. 모션 및 이펙트 레이어 순회 */
105 const int32 PhaseCount = mSkillPhaseLayers.Num();
106 for (int32 PhaseIndex = 0; PhaseIndex < PhaseCount; ++PhaseIndex)
107 {
108 const FSkillPhaseLayer& MotionLayer = mSkillPhaseLayers[PhaseIndex];
109
110 const FText TargetFilterText = GetTargetIndexFilterText(MotionLayer.mTargetIndexFilter);
111 const FText TeamFilterText = GetTeamAttitudeFilterText(MotionLayer.mTeamAttitudeFilter);
112
113 FText MotionHeader = FText::Format(
114 LOCTEXT("MotionHeaderFormat", "■ {0}타 모션 ({1}, {2}):"),
115 FText::AsNumber(PhaseIndex + 1),
116 TargetFilterText,
117 TeamFilterText
118 );
119 DescriptionLines.Add(MotionHeader);
120
121 for (const TInstancedStruct<FSkillEffectLayer>& InstancedEffect : MotionLayer.mSkillEffectLayers)
122 {
123 if (InstancedEffect.IsValid() == true)
124 {
125 const FSkillEffectLayer& EffectLayer = InstancedEffect.Get();
126 FText EffectText = EffectLayer.MakeDescription();
127 if (EffectText.IsEmpty() == false)
128 {
129 FText FormattedEffect = FText::Format(
130 LOCTEXT("EffectLineFormat", "- {0}"),
131 EffectText
132 );
133 DescriptionLines.Add(FormattedEffect);
134 }
135 }
136 }
137 }
138
139 /* 3. 줄바꿈 조합 반환 (FText::Join 활용) */
140 return FText::Join(FText::FromString(TEXT("\n")), DescriptionLines);
141}

144EDataValidationResult UStaticSkillData::IsDataValid(FDataValidationContext& Context) const
145{
146 EDataValidationResult SuperResult = Super::IsDataValid(Context);
147 EDataValidationResult ThisResult = EDataValidationResult::Valid;
148
149 if (mName.IsEmpty() == true)
150 {
151 Context.AddError(FText::FromString(TEXT("스킬 이름 미지정")));
152 ThisResult = EDataValidationResult::Invalid;
153 }
154 if (mCooldownEffectClass.ToSoftObjectPath().IsValid() == false)
155 {
156 Context.AddError(FText::FromString(TEXT("쿨다운 타입 미지정")));
157 ThisResult = EDataValidationResult::Invalid;
158 }
159 if (mSkillPhaseLayers.IsEmpty() == true)
160 {
161 Context.AddError(FText::FromString(TEXT("스킬 모션 미지정")));
162 ThisResult = EDataValidationResult::Invalid;
163 }
164
165 // 밀치기/당기기 배치 순서 검증
166 // 밀당이 중간에 끼면 시전자와 피격자의 모션이 어긋나서 어색할 수 있으므로 의도가 맞는 지 한 번 더 확인해보라는 의미에서 경고
167 const int32 LastPhaseIndex = mSkillPhaseLayers.Num() - 1;
168 for (int32 PhaseIndex = 0; PhaseIndex <= LastPhaseIndex; ++PhaseIndex)
169 {
170 const TArray<TInstancedStruct<FSkillEffectLayer>>& Layers = mSkillPhaseLayers[PhaseIndex].mSkillEffectLayers;
171 bool HasPassedForcedMovement = false;
172 for (int32 LayerIndex = 0; LayerIndex < Layers.Num(); ++LayerIndex)
173 {
174 if (Layers[LayerIndex].IsValid() == false)
175 {
176 continue;
177 }
178
179 // 마지막 페이즈가 아닌 곳의 밀당
180 const bool HasForced = Layers[LayerIndex].Get().HasForcedMovement();
181 if (HasForced == true && PhaseIndex != LastPhaseIndex)
182 {
183 Context.AddWarning(FText::FromString(FString::Printf(TEXT("밀치기/당기기를 마지막 페이즈 아닌 곳에 배치 (%d타)"), PhaseIndex + 1)));
184 }
185
186 // 같은 페이즈에서 밀당 뒤에 오는 다른 이펙트
187 if (HasForced == true)
188 {
189 HasPassedForcedMovement = true;
190 }
191 else if (HasPassedForcedMovement == true)
192 {
193 Context.AddWarning(FText::FromString(FString::Printf(TEXT("밀치기/당기기 뒤에 다른 이펙트 배치 (%d타 %d번째)"), PhaseIndex + 1, LayerIndex + 1)));
194 }
195 }
196 }
197
198 return CombineDataValidationResults(SuperResult, ThisResult);
199}
