111void USkillAnimationComponent::OnHandleHitAnimationEvent(const FBoardActorAnimationContext& Context, UObject* EndAnim, const FEventTriggerPayloadBase* Payload)
112{
113 const FApplyAnimationEventTriggerPayload* AnimationPayload = StaticCast<const FApplyAnimationEventTriggerPayload*>(Payload);
114
115 const USkillComponentModel* OwnerSkillComp = Context.mMetaData.Get<FSkillAnimationMetaData>().mInstigator.Get();
116 if (OwnerSkillComp == nullptr)
117 {
118 return;
119 }
120
121 const UBoardActorModel* OwnerActorModel = OwnerSkillComp->GetOwnerModel<UBoardActorModel>();
122 const FActiveSkillContext& ActiveSkillContext = OwnerSkillComp->GetActiveSkillContext();
123 for (IBoardCombatTarget* OtherCombatTarget : ActiveSkillContext.mFinalCombatTargets)
124 {
125 UBoardActorModel* OtherActorModel = Cast<UBoardActorModel>(OtherCombatTarget);
126 if (OtherActorModel == nullptr || OwnerActorModel == OtherActorModel)
127 {
128 continue;
129 }
130
131 IBoardCombatTargetView* OtherActorView = OtherActorModel->GetView<IBoardCombatTargetView>();
132 if (OtherActorView == nullptr)
133 {
134 continue;
135 }
136
137 const ETileActorDirection OtherAnimationDir = LocalToOtherLocalDirection(
138 Context.mMontageDir,
139 OwnerActorModel->GetTileTransform().mDirection,
140 OtherActorModel->GetTileTransform().mDirection
141 );
142 const USkillAnimationComponent* OtherSkillAnimComp = OtherActorView->GetSkillAnimationComponent();
143 OtherSkillAnimComp->PlayHitAnimation(ActiveSkillContext.mSkillEndBarrier, AnimationPayload->mAnimationTag, OtherAnimationDir);
144 }

147void USkillAnimationComponent::OnHandleHitVFXEvent(const FBoardActorAnimationContext& Context, UObject* EndAnim, const FEventTriggerPayloadBase* Payload)
148{
149 const FApplyVFXEventTriggerPayload* VFXPayload = StaticCast<const FApplyVFXEventTriggerPayload*>(Payload);
150
151 const USkillComponentModel* OwnerSkillComp = Context.mMetaData.Get<FSkillAnimationMetaData>().mInstigator.Get();
152 if (OwnerSkillComp == nullptr)
153 {
154 return;
155 }
156
157 const UBoardActorModel* OwnerActorModel = OwnerSkillComp->GetOwnerModel<UBoardActorModel>();
158 const FActiveSkillContext& ActiveSkillContext = OwnerSkillComp->GetActiveSkillContext();
159
160 switch (VFXPayload->mTargetType)
161 {
162 case EApplyNiagaraTargetType::Actor:
163 {
164 for (IBoardCombatTarget* OtherCombatTarget : ActiveSkillContext.mFinalCombatTargets)
165 {
166 UBoardActorModel* OtherActorModel = Cast<UBoardActorModel>(OtherCombatTarget);
167 if (OtherActorModel == nullptr)
168 {
169 continue;
170 }
171
172 IBoardCombatTargetView* OtherActorView = OtherActorModel->GetView<IBoardCombatTargetView>();
173 if (OtherActorView == nullptr)
174 {
175 continue;
176 }
177
178 const ETileActorDirection OtherVFXDir = LocalToOtherLocalDirection(
179 Context.mMontageDir,
180 OwnerActorModel->GetTileTransform().mDirection,
181 OtherActorModel->GetTileTransform().mDirection
182 );
183 const USkillAnimationComponent* OtherSkillAnimComp = OtherActorView->GetSkillAnimationComponent();
184 OtherSkillAnimComp->SpawnHitVFXOnSelf(ActiveSkillContext.mSkillEndBarrier, VFXPayload->mVFXSpawnData, OtherVFXDir);
185 }
186 break;
187 }
188 case EApplyNiagaraTargetType::EffectTile:
189 {
190 for (const FTileIndex& EffectTileIndex : ActiveSkillContext.mEffectTileIndexes)
191 {
192 const FTileTransform EffectTileTransform(EffectTileIndex, Context.mMontageDir);
193 const FTransform EffectTransform = ActiveSkillContext.mMapModel->TileToWorldTransform(EffectTileTransform);
194 SpawnHitVFXOnTile(ActiveSkillContext.mSkillEndBarrier, VFXPayload->mVFXSpawnData, EffectTransform);
195 }
196 break;
197 }
198 case EApplyNiagaraTargetType::TargetTile:
199 {
200 for (const FTileIndex& TargetTileIndex : ActiveSkillContext.mTargetTileIndexes)
201 {
202 const FTileTransform TargetTileTransform(TargetTileIndex, Context.mMontageDir);
203 const FTransform TargetTransform = ActiveSkillContext.mMapModel->TileToWorldTransform(TargetTileTransform);
204 SpawnHitVFXOnTile(ActiveSkillContext.mSkillEndBarrier, VFXPayload->mVFXSpawnData, TargetTransform);
205 }
206 break;
207 }
208 case EApplyNiagaraTargetType::Self:
209 {
210 SpawnHitVFXOnSelf(ActiveSkillContext.mSkillEndBarrier, VFXPayload->mVFXSpawnData, Context.mMontageDir);
211 break;
212 }
213 }
214}

223void USkillAnimationComponent::OnHandleCameraZoomEvent(const FBoardActorAnimationContext& Context, UObject* EndAnim, const FEventTriggerPayloadBase* Payload)
224{
225 const FCameraZoomEventTriggerPayload* CameraZoomPayload = StaticCast<const FCameraZoomEventTriggerPayload*>(Payload);
226
227 const USkillComponentModel* OwnerSkillComp = Context.mMetaData.Get<FSkillAnimationMetaData>().mInstigator.Get();
228 if (OwnerSkillComp == nullptr)
229 {
230 return;
231 }
232
233 const UBoardActorModel* OwnerActorModel = OwnerSkillComp->GetOwnerModel<UBoardActorModel>();
234 const FActiveSkillContext& ActiveSkillContext = OwnerSkillComp->GetActiveSkillContext();
235
236 bool IsZoomTargetSelfActor = false;
237 bool ContainSelfTileLocation = false;
238 bool ContainTargetTileLocations = false;
239
240 const FCameraZoomEventData& ZoomEventData = CameraZoomPayload->mCameraZoomEventData;
241 switch (ZoomEventData.mTargetType)
242 {
243 case ECameraZoomTargetType::SelfActor:
244 {
245 IsZoomTargetSelfActor = true;
246 ContainSelfTileLocation = true;
247 ContainTargetTileLocations = false;
248 break;
249 }
250 case ECameraZoomTargetType::SelfLocation:
251 {
252 IsZoomTargetSelfActor = false;
253 ContainSelfTileLocation = true;
254 ContainTargetTileLocations = false;
255 break;
256 }
257 case ECameraZoomTargetType::TargetLocation:
258 {
259 IsZoomTargetSelfActor = false;
260 ContainSelfTileLocation = false;
261 ContainTargetTileLocations = true;
262 break;
263 }
264 case ECameraZoomTargetType::AllLocation:
265 {
266 IsZoomTargetSelfActor = false;
267 ContainSelfTileLocation = true;
268 ContainTargetTileLocations = true;
269 break;
270 }
271 }
272
273 TArray<FVector> CollectedLocations;
274 if (ContainSelfTileLocation == true)
275 {
276 CollectedLocations.Add(ActiveSkillContext.mMapModel->TileToWorldLocation(ActiveSkillContext.mSelfTileIndex));
277 }
278 if (ContainTargetTileLocations == true)
279 {
280 for (const FTileIndex& EffectTileIndex : ActiveSkillContext.mEffectTileIndexes)
281 {
282 CollectedLocations.Add(ActiveSkillContext.mMapModel->TileToWorldLocation(EffectTileIndex));
283 }
284 }
285
286 float FinalZoomScale = 0.f;
287 FVector FinalZoomLocation = FVector::ZeroVector;
288 ZoomEventData.CalculateCameraZoomScaleAndLocation(CollectedLocations, OUT FinalZoomScale, OUT FinalZoomLocation);
289
290 if (IsZoomTargetSelfActor == true)
291 {
292 ZoomInCamera(CameraZoomPayload->OnEndDurationEventTrigger, FinalZoomScale, GetOwner());
293 }
294 else
295 {
296 ZoomInCamera(CameraZoomPayload->OnEndDurationEventTrigger, FinalZoomScale, GetOwner());
297 }
298}
