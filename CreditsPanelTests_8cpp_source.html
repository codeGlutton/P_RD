24 TSharedPtr<SButton> FindCreditsButton(const TSharedRef<SWidget>& Widget, const FString& Label, TSharedPtr<SButton> ParentButton = nullptr)
25 {
26 if (Widget->GetTypeAsString() == TEXT("SButton")) { ParentButton = StaticCastSharedRef<SButton>(Widget); }
27 if (Widget->GetTypeAsString() == TEXT("STextBlock") && StaticCastSharedRef<STextBlock>(Widget)->GetText().ToString() == Label) { return ParentButton; }
28 FChildren* Children = Widget->GetChildren();
29 for (int32 Index = 0; Index < Children->Num(); ++Index)
30 {
31 if (TSharedPtr<SButton> Found = FindCreditsButton(Children->GetChildAt(Index), Label, ParentButton)) { return Found; }
32 }
