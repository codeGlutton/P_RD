50ESRPGCommandResult USRPGSkillBuildAction::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
51{
52 ESRPGCommandResult Result = Super::HandleCommand(Command);
53 if (Result == ESRPGCommandResult::Handled)
54 {
55 return Result;
56 }
57
58 switch (Command.Get().GetCommandType())
59 {
60 case ESRPGCommandType::SkillSelect:
61 {
62 /* 새롭게 스킬 선택 시 제거 */
63
64 const FSRPGSkillSelectCommand& SkillSelectCommand = Command.Get<FSRPGSkillSelectCommand>();
65
66 OnSelectSkill = SkillSelectCommand.OnSelectSkill;
67 OnChangeSkillBuildPhase = SkillSelectCommand.OnChangeSkillBuildPhase;
68 OnPostSimulateSkillAction = SkillSelectCommand.OnPostSimulateSkillAction;
69 OnCancelSimulateSkillAction = SkillSelectCommand.OnCancelSimulateSkillAction;
70 if (SkillSelectCommand.mSkillIndex != mSelectedSkillIndex)
71 {
72 /* 다르면 변경 */
73
74 // 이전 스킬의 조준/효과 하이라이트를 먼저 지운다. ResetTargetTile은 데이터(mEffectTileIndexes)만
75 // 비우고 화면 하이라이트는 안 지워서, 안 지우면 이전 스킬의 효과 범위가 남아 "취소가 안 된 것"처럼 보인다.
76 ClearAllTileHighlights();
77 ResetTargetTile();
78 ResetDice();
79 ResetSkill();
80 /*
81 * 조준 중 다른 스킬을 고르면 처음부터 다시 시작한다.
82 * 페이즈도 None으로 돌려야 다음 SetSkill이 안전하다.
83 */
84 SetBuildPhase(ESRPGSkillBuildPhase::None);
85 SetSkill(SkillSelectCommand.mSkillIndex);
86 RefreshAimableTileHighlights();
87 SetBuildPhase(ESRPGSkillBuildPhase::AimSelection);
88 }
89 else
90 {
91 /* 같으면 취소 */
92
93 MarkActionCompleted(ESRPGActionResult::Cancelled);
94 SetBuildPhase(ESRPGSkillBuildPhase::None);
95 }
96 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
97 }
98 case ESRPGCommandType::MoveSelect:
99 case ESRPGCommandType::TurnEnd:
100 {
101 /* 다른 동작 요구 시 취소 */
102
103 ClearAllTileHighlights();
104 ResetTargetTile();
105 ResetDice();
106 ResetSkill();
107 MarkActionCompleted(ESRPGActionResult::Cancelled);
108 SetBuildPhase(ESRPGSkillBuildPhase::None);
109 return ESRPGCommandResult::Ignored;
110 }
111 case ESRPGCommandType::DiceSelect:
112 {
113 /* 주사위 변경 시 타겟부터 재설정 */
114
115 const FSRPGDiceSelectCommand& DiceSelectCommand = Command.Get<FSRPGDiceSelectCommand>();
116
117 /*
118 * 주사위 변경은 스킬을 고른 뒤(조준/프리뷰 단계)에만 의미가 있다.
119 * - 스킬 미선택(None)/시전 완료(Build) 상태의 주사위 클릭은 무시한다(억지로 진행하면
120 * 선택 스킬이 없어 ChangeDices에서 nullptr 참조로 죽는다).
121 * - 프리뷰에서 주사위를 바꾸면 주사위 합이 달라져 조준이 무효가 되므로, ChangeDices 전에
122 * 조준 단계로 되돌린다. ChangeDices는 AimSelection 전제(checkf)라 안 되돌리면 어설션 크래시.
123 * (스킬 조준 -> 주사위 재선택 크래시 수정)
124 */
125 const bool IsSkillSelectedForDice = mSkillBuildPhase == ESRPGSkillBuildPhase::AimSelection
126 || mSkillBuildPhase == ESRPGSkillBuildPhase::Preview;
127 if (DiceSelectCommand.mDiceIndex != INDEX_NONE && IsSkillSelectedForDice == true)
128 {
129 ResetTargetTile();
130 SetBuildPhase(ESRPGSkillBuildPhase::AimSelection);
131 ChangeDices(DiceSelectCommand.mDiceIndex);
132 RefreshAimableTileHighlights();
133 }
134 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
135 }
136 case ESRPGCommandType::WorldTrace:
137 {
138 /* 월드 공간 터치 시 선택 위치에 따라서 결정 */
139
140 return CombineSRPGCommandResult(HandleWorldTraceCommand(Command), Result);
141 }
142 }
143
144 return ESRPGCommandResult::Ignored;
145}

147ESRPGCommandResult USRPGSkillBuildAction::HandleWorldTraceCommand(const TInstancedStruct<FSRPGCommand>& Command)
148{
149 ESRPGCommandResult Result = ESRPGCommandResult::Ignored;
150
151 const FSRPGWorldTraceCommand& WorldTraceCommand = Command.Get<FSRPGWorldTraceCommand>();
152 if (WorldTraceCommand.mIsLongPress == true)
153 {
154 return Result;
155 }
156
157 UTileMapModel* TileMap = GetTileMap();
158 checkf(TileMap != nullptr, TEXT("타일 맵 nullptr"));
159
160 AActor* TargetActor = nullptr;
161 FTileIndex TargetTileIndex = FTileIndex::Invalid;
162 GetTileActorUnderCursor(GetWorld(), RDTraceChannels::TileOnlyTrace, WorldTraceCommand.mScreenPosition, OUT TargetActor, OUT TargetTileIndex);
163
164 IActorView* ActorView = Cast<IActorView>(TargetActor);
165 const bool IsContactedTileMap = ActorView != nullptr && ActorView->GetModel() == TileMap;
166 const bool IsContactedBoard = IsContactedTileMap == true || TargetTileIndex != FTileIndex::Invalid;
167 // 보드를 아예 벗어난 탭(IsContactedBoard=false)도 무효 타일과 똑같이 "타일 밖" 취소로 취급한다
168 // (기획: 스킬 조준 중 타일 밖을 누르면 취소). TargetTileIndex가 유효하면 정상 처리로 간다.
169 if (IsContactedBoard == true || TargetTileIndex == FTileIndex::Invalid)
170 {
171 if (TargetTileIndex == FTileIndex::Invalid)
172 {
173 /* 한단계 취소작업 (무효 타일 또는 보드 밖 탭) */
174
175 switch (mSkillBuildPhase)
176 {
177 case ESRPGSkillBuildPhase::Preview:
178 {
179 /* 프리뷰 단계에서 한단계 취소 시, 조준 대상 취소 처리 */
180
181 ResetTargetTile();
182 RefreshAimableTileHighlights();
183 SetBuildPhase(ESRPGSkillBuildPhase::AimSelection);
184
185 Result = ESRPGCommandResult::Handled;
186 break;
187 }
188 case ESRPGSkillBuildPhase::AimSelection:
189 {
190 /* 조준 대상 설정 단계에서 한단계 취소 시, 빌드 자체 종료 */
191
192 MarkActionCompleted(ESRPGActionResult::Cancelled);
193 SetBuildPhase(ESRPGSkillBuildPhase::None);
194
195 Result = ESRPGCommandResult::Handled;
196 break;
197 }
198 }
199 }
200 else
201 {
202 /* 한단계 처리작업 */
203
204 switch (mSkillBuildPhase)
205 {
206 case ESRPGSkillBuildPhase::Preview:
207 {
208 /* 확정 칸 클릭 시, 스킬 캐스팅 */
209
210 if (mTargetIndex == TargetTileIndex)
211 {
212 BuildSkill();
213 SetBuildPhase(ESRPGSkillBuildPhase::Build);
214 MarkActionCompleted(ESRPGActionResult::Succeeded);
215
216 Result = ESRPGCommandResult::Handled;
217 break;
218 }
219 [[fallthrough]];
220 }
221 case ESRPGSkillBuildPhase::AimSelection:
222 {
223 /* 조준 가능한 칸 클릭 시, 프리뷰 단계까지 보여주기 */
224
225 if (CanSelectTargetTile(TargetTileIndex) == true)
226 {
227 ResetTargetTile();
228 SetBuildPhase(ESRPGSkillBuildPhase::AimSelection);
229 SetTargetTile(TargetTileIndex);
230 RefreshEffectTileHighlights();
231 SetBuildPhase(ESRPGSkillBuildPhase::Preview);
232
233 Result = ESRPGCommandResult::Handled;
234 break;
235 }
236 break;
237 }
238 }
239 }
240 }
241 return Result;
242}

245{

272void USRPGSkillBuildAction::ChangeDices(int32 RequestedDiceIndex)
273{
274 checkf(mSkillBuildPhase == ESRPGSkillBuildPhase::AimSelection, TEXT("스킬 빌드 순서 오류"));
275
276 UPlayerUnitModel* PlayerUnit = Cast<UPlayerUnitModel>(mInstigator.Get());
277 checkf(PlayerUnit != nullptr, TEXT("주사위를 굴릴 수 있는 플레이어 유닛이 아님"));
278
279 UDicePoolModel* DicePoolModel = PlayerUnit->GetDicePoolModel();
280 checkf(DicePoolModel != nullptr, TEXT("주사위 컴포넌트를 들고 있지 않음"));
281
282 if (DicePoolModel->IsSelectedDice(RequestedDiceIndex) == true)
283 {
284 // 이전 주사위 제거
285 DicePoolModel->MarkDiceUnselected(RequestedDiceIndex);
286 }
287 else if (DicePoolModel->GetSelectedDiceNum() < mSelectedSkill->mRequiredDiceCount)
288 {
289 // 새로운 주사위 추가 할당
290 DicePoolModel->MarkDiceSelected(RequestedDiceIndex);
291 }
292}

295{
296 checkf(mSkillBuildPhase == ESRPGSkillBuildPhase::AimSelection, TEXT("스킬 빌드 순서 오류"));
297
298 UPlayerUnitModel* PlayerUnit = Cast<UPlayerUnitModel>(mInstigator.Get());
299 checkf(PlayerUnit != nullptr, TEXT("주사위를 굴릴 수 있는 플레이어 유닛이 아님"));
300
301 UDicePoolModel* DicePoolModel = PlayerUnit->GetDicePoolModel();
302 checkf(DicePoolModel != nullptr, TEXT("주사위 컴포넌트를 들고 있지 않음"));
303
304 mTargetIndex = TargetIndex;
305
306 UTileMapModel* TileMap = GetTileMap();
307 checkf(TileMap != nullptr, TEXT("타일 맵 nullptr"));
308
309 TArray<TObjectPtr<UBoardActorModel>> AllEffectActors;
310 for (const FTileIndex& EffectTileIndex : mEffectTileIndexes)
311 {
312 TArray<UBoardActorModel*> EffectActors = TileMap->GetActorsOnTile(EffectTileIndex);
313 AllEffectActors.Append(EffectActors);
314 }
315
316 USimulationSubsystem* SimulationSubsystem = GetWorld()->GetSubsystem<USimulationSubsystem>();
317 checkf(SimulationSubsystem != nullptr, TEXT("시뮬레이션 서브시스템 모델 nullptr"));
318
319 TInstancedStruct<FSRPGCommand> SkillCastCommand;
320 SkillCastCommand.InitializeAs<FSRPGSkillCastCommand>();
321 SkillCastCommand.GetMutable<FSRPGSkillCastCommand>().mSkillIndex = mSelectedSkillIndex;
322 SkillCastCommand.GetMutable<FSRPGSkillCastCommand>().mTargetIndex = mTargetIndex;
323 SkillCastCommand.GetMutable<FSRPGSkillCastCommand>().mDiceSum = DicePoolModel->GetSelectedDiceSum();
324
325 TArray<FSRPGTurnEventLog> TurnEventLogs = SimulationSubsystem->SimulateUntilNextAction(MoveTemp(SkillCastCommand));
326 OnPostSimulateSkillAction.Broadcast(TurnEventLogs);
327}
