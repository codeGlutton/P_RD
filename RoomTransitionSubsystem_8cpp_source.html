155bool URoomTransitionSubsystem::MakeStageAndPreloadRoomAsync(EStageLevelType StageLevel, FOnReadyToTransition ReadyToTransitionCallback, FOnPreTransitNextRoom PreTransitionCallback, bool RequireExternalReady, bool IsAutoTransition)
156{
157 if (EnumHasAnyFlags(mTransitionState, ERoomTransitionStateFlag::AllTaskRequested) == true)
158 {
159 UE_LOG(LogTransition, Log, TEXT("다른 룸 데이터 처리 중으로 Preload 불가"));
160 return false;
161 }
162 EnumAddFlags(mTransitionState, ERoomTransitionStateFlag::NewStageRequested);
163
164 // 스테이지 먼저 생성
165 GetRunMutableData()->MakeStageAsync(StageLevel, FOnCreateStage::CreateLambda([this, ReadyToTransitionCallback, PreTransitionCallback, RequireExternalReady, IsAutoTransition](const FStage& NewStage) {
166 EnumRemoveFlags(mTransitionState, ERoomTransitionStateFlag::NewStageRequested);
167
168 // 이후 스테이지의 배정된 첫 방으로 Preload 시작
169 const FRoom& StartRoom = NewStage.GetStartRoom();
170 checkf(PreloadRoomAsync(StartRoom.mRow, StartRoom.mColumn, ReadyToTransitionCallback, PreTransitionCallback, RequireExternalReady, IsAutoTransition), TEXT("만들어진 Stage로 Preload 시도 실패"));
171 }));
172
173 return true;
