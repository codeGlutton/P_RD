138bool URoomTransitionSubsystem::MakeStageAndPreloadRoomAsync(EStageLevelType StageLevel, FOnReadyToTransition ReadyToTransitionCallback, FOnPreTransitNextRoom PreTransitionCallback, bool RequireExternalReady, bool IsAutoTransition)
139{
140 if (EnumHasAnyFlags(mTransitionState, ERoomTransitionStateFlag::AllTaskRequested) == true)
141 {
142 UE_LOG(LogTransition, Log, TEXT("다른 룸 데이터 처리 중으로 Preload 불가"));
143 return false;
144 }
145 EnumAddFlags(mTransitionState, ERoomTransitionStateFlag::NewStageRequested);
146
147 // 스테이지 먼저 생성
148 GetRunMutableData()->MakeStageAsync(StageLevel, FOnCreateStage::CreateLambda([this, ReadyToTransitionCallback, PreTransitionCallback, RequireExternalReady, IsAutoTransition](const FStage& NewStage) {
149 EnumRemoveFlags(mTransitionState, ERoomTransitionStateFlag::NewStageRequested);
150
151 // 이후 스테이지의 배정된 첫 방으로 Preload 시작
152 const FRoom& StartRoom = NewStage.GetStartRoom();
153 checkf(PreloadRoomAsync(StartRoom.mRow, StartRoom.mColumn, ReadyToTransitionCallback, PreTransitionCallback, RequireExternalReady, IsAutoTransition), TEXT("만들어진 Stage로 Preload 시도 실패"));
154 }));
155
156 return true;
157}
