28bool FRewardSelectionPolicyContractTest::RunTest(const FString& Parameters)
29{
30 TStrongObjectPtr<URewardUIModel> Model(NewObject<URewardUIModel>());
31 TStrongObjectPtr<URewardPolicyTestListener> Listener(
32 NewObject<URewardPolicyTestListener>());
33 const FPrimaryAssetId ArtifactA(TEXT("Artifact"), TEXT("PolicyA"));
34 const FPrimaryAssetId ArtifactB(TEXT("Artifact"), TEXT("PolicyB"));
35
36 FRewardSelectionOfferUI Offer;
37 for (int32 Index = 0; Index < 2; ++Index)
38 {
39 FRewardChoiceUI& Choice = Offer.mOptions.AddDefaulted_GetRef();
40 Choice.mChoiceIndex = Index;
41 Choice.mKind = ERewardChoiceKind::Artifact;
42 Choice.mSourceAssetId = Index == 0 ? ArtifactA : ArtifactB;
43 }
44 Model->SetSelectionOffer(Offer);
45 Model->OnRewardSelectionRequested.AddDynamic(
46 Listener.Get(), &URewardPolicyTestListener::HandleSelectionRequested);
47
48 TestEqual(TEXT("SelectOne 정책으로 설정됨"),
49 Model->GetAcquisitionPolicy(), ERewardAcquisitionPolicy::SelectOne);
50 TestTrue(TEXT("후보의 PrimaryAssetId 선택 요청 수락"),
51 Model->RequestSelectReward(ArtifactB));
52 TestEqual(TEXT("선택 요청은 한 번만 전달됨"),
53 Listener->SelectionRequestCount, 1);
54 TestEqual(TEXT("선택 요청 ID 보존"),
55 Listener->LastSelectedRewardId, ArtifactB);
56
57 const FPrimaryAssetId Unknown(TEXT("Artifact"), TEXT("Unknown"));
58 TestFalse(TEXT("후보 풀 밖의 선택 ID 거절"),
59 Model->RequestSelectReward(Unknown));
60 TestEqual(TEXT("잘못된 선택은 delegate를 호출하지 않음"),
61 Listener->SelectionRequestCount, 1);
62 return true;
63}
