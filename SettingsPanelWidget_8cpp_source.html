172 const FVector2D MetricsViewport = MyGeometry.GetLocalSize();
173 if (MetricsViewport.X > 1.0f && MetricsViewport.Y > 1.0f

179 }
180 if (mMetricsCountdown > 0 && --mMetricsCountdown == 0)
181 {
182 LogSettingsMetrics(MetricsViewport);

186void USettingsPanelWidget::ApplyFoldScaleVariant(const FVector2D& ViewportSize)
187{
188 if (ViewportSize.X <= 1.0f || ViewportSize.Y <= 1.0f || WidgetTree == nullptr)
189 {
190 return;
191 }
192 // 레터박스 WBP(SettingsModalCanvas 존재)면 ScaleBox가 전 화면비를 전담한다 — 폴드 스케일 이중 적용 금지.
193 if (WidgetTree->FindWidget(TEXT("SettingsModalCanvas")) != nullptr)
194 {
195 return;
196 }
197 // 컷은 concept/전투 HUD와 동기(1.68). 상태가 안 바뀌면 아무것도 하지 않는다.
198 constexpr float FoldAspectCut = 1.68f;
199 const bool bFold = (ViewportSize.X / ViewportSize.Y) < FoldAspectCut;
200 if (bFold == mFoldScaleActive)
201 {
202 return;
203 }
204
205 // 스케일 비율 = 폴드 마커 폭 / 기준 마커 폭 (디자이너 데이터). 마커 없는 구 WBP는 변형 생략.
206 auto GetMarkerWidth = [this](const TCHAR* MarkerName) -> float
207 {
208 UWidget* Marker = WidgetTree->FindWidget(FName(MarkerName));
209 UCanvasPanelSlot* MarkerSlot = Marker != nullptr ? Cast<UCanvasPanelSlot>(Marker->Slot) : nullptr;
210 return MarkerSlot != nullptr ? MarkerSlot->GetSize().X : 0.0f;
211 };
212 const float BaseWidth = GetMarkerWidth(TEXT("Set_grid_base"));
213 const float FoldWidth = GetMarkerWidth(TEXT("Set_grid_fold"));
214 if (BaseWidth <= 0.0f || FoldWidth <= 0.0f)
215 {
216 return;
217 }
218 const float Scale = FoldWidth / BaseWidth;
219
220 // 중앙(0.5,0.5) 점앵커 = 시안 모달 그리드 콘텐츠. 화면 중앙 기준 균일 스케일을 렌더 트랜스폼으로 얹는다.
221 // 슬롯은 불변이라 복원은 항등 트랜스폼이고, 텍스트/아트가 함께 스케일되며 히트테스트도 Slate가 변환을 반영한다.
222 WidgetTree->ForEachWidget([Scale, bFold](UWidget* Widget)
223 {
224 UCanvasPanelSlot* CanvasSlot = Widget != nullptr ? Cast<UCanvasPanelSlot>(Widget->Slot) : nullptr;
225 if (CanvasSlot == nullptr)
226 {
227 return;
228 }
229 const FAnchors SlotAnchors = CanvasSlot->GetAnchors();
230 if (SlotAnchors.Minimum != FVector2D(0.5f, 0.5f) || SlotAnchors.Maximum != FVector2D(0.5f, 0.5f))
231 {
232 return;
233 }
234 if (bFold)
235 {
236 const FVector2D CenterOffset = CanvasSlot->GetPosition();
237 Widget->SetRenderTransformPivot(FVector2D::ZeroVector);
238 Widget->SetRenderTransform(FWidgetTransform(
239 FVector2D(CenterOffset.X * (Scale - 1.0f), CenterOffset.Y * (Scale - 1.0f)),
240 FVector2D(Scale, Scale), FVector2D::ZeroVector, 0.0f));
241 }
242 else
243 {
244 Widget->SetRenderTransform(FWidgetTransform());
245 Widget->SetRenderTransformPivot(FVector2D(0.5f, 0.5f));
246 }
247 });
248
249 mFoldScaleActive = bFold;
250}
