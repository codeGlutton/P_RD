94UTexture2D* UCombatTileMapHUDWidget::ResolveFloatingLogIcon(EFloatingLogIconType IconType, EFloatingLogColorType ColorType) const
95{
96 switch (IconType)
97 {
98 case EFloatingLogIconType::HP:
99 // HP는 색으로 피해/회복을 가른다(Heal=회복, 그 외=피해).
100 return ColorType == EFloatingLogColorType::Heal ? mLogIconHpRecovery : mLogIconHpDamage;
101 case EFloatingLogIconType::GetMove: return mLogIconGetMove;
102 case EFloatingLogIconType::GetDefense: return mLogIconGetDefense;
103 case EFloatingLogIconType::Agility: return mLogIconAgility;
104 case EFloatingLogIconType::Fortification: return mLogIconFortification;
105 case EFloatingLogIconType::Vulnerability: return mLogIconVulnerability;
106 case EFloatingLogIconType::Weakness: return mLogIconWeakness;
107 default:
108 // None + 아직 전용 아이콘이 없는 Poison/Fire/Move는 아이콘 없이 텍스트만 띄운다.
109 return nullptr;

148void UCombatTileMapHUDWidget::HandleCombatFloatingLogsCleared()
149{
150 // 아직 안 뜬 대기분을 먼저 비운다(다음 프레임에 되살아나지 않게).
151 mPendingFloatingCombatLogs.Reset();
152 mFloatingCombatLogQueueCooldown = 0.0f;
153 mNextFloatingCombatLogArrivalOrder = 0;
154
155 // 화면에 떠 있는 로그는 즉시 떼지 않고 퇴장 연출(오른쪽으로 흐르며 페이드아웃)로 보낸다.
156 // (연출이 끝나면 UpdateFloatingCombatLogs가 제거한다. 애니 종료 후 "뿅" 사라짐 방지.)
157 for (int32 LogIndex = mFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
158 {
159 FFloatingCombatLogEntry& Entry = mFloatingCombatLogs[LogIndex];
160 if (Entry.mRoot == nullptr)
161 {
162 mFloatingCombatLogs.RemoveAt(LogIndex);
163 continue;
164 }
165 if (Entry.mIsDismissing == false)
166 {
167 Entry.mIsDismissing = true;
168 Entry.mDismissElapsed = 0.0f;
169 }
170 }

179void UCombatTileMapHUDWidget::UpdateFloatingCombatLogQueue(float InDeltaTime)
180{
181 // 지도(풀스크린) 열림 중에는 새 플로팅 로그를 스폰하지 않는다(탑바만 남기는 뷰).
182 if (mCombatControlsHidden)
183 {
184 return;
185 }
186
187 if (mPendingFloatingCombatLogs.Num() == 0)
188 {
189 mFloatingCombatLogQueueCooldown = 0.0f;
190 return;
191 }
192
193 mFloatingCombatLogQueueCooldown -= InDeltaTime;
194 if (mFloatingCombatLogQueueCooldown > 0.0f)
195 {
196 return;
197 }
198
199 const FCombatFloatingLogRequest Request = mPendingFloatingCombatLogs[0].mRequest;
200 mPendingFloatingCombatLogs.RemoveAt(0);
201 SpawnFloatingCombatLogAtWorld(Request);
202 mFloatingCombatLogQueueCooldown = FloatingLogQueueInterval;
203}

214void UCombatTileMapHUDWidget::SpawnFloatingCombatLogAtWorld(const FCombatFloatingLogRequest& Request)
215{
216 if (RootCanvas == nullptr || WidgetTree == nullptr)
217 {
218 return;
219 }
220
221 // 아이콘+텍스트를 한 덩어리로 움직이게 가로 박스에 담는다(아이콘 없으면 텍스트만).
222 UHorizontalBox* LogBox = WidgetTree->ConstructWidget<UHorizontalBox>(UHorizontalBox::StaticClass());
223 UTextBlock* LogText = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass());
224 if (LogBox == nullptr || LogText == nullptr)
225 {
226 return;
227 }
228 LogBox->SetVisibility(ESlateVisibility::HitTestInvisible);
229
230 UTexture2D* Icon = ResolveFloatingLogIcon(Request.mIconType, Request.mColorType);
231 if (Icon != nullptr)
232 {
233 if (UImage* LogIcon = WidgetTree->ConstructWidget<UImage>(UImage::StaticClass()))
234 {
235 LogIcon->SetBrushFromTexture(Icon, false);
236 LogIcon->SetDesiredSizeOverride(FVector2D(26.0f, 26.0f));
237 if (UHorizontalBoxSlot* IconSlot = LogBox->AddChildToHorizontalBox(LogIcon))
238 {
239 IconSlot->SetVerticalAlignment(VAlign_Center);
240 IconSlot->SetPadding(FMargin(0.0f, 0.0f, 4.0f, 0.0f));
241 }
242 }
243 }
244
245 LogText->SetJustification(ETextJustify::Center);
246 LogText->SetText(Request.mText);
247 LogText->SetColorAndOpacity(FSlateColor(ResolveFloatingLogColor(Request.mColorType)));
248 // 기본 폰트에서 크기만 키운다(전투 화면 위에서 읽히는 최소 크기).
249 FSlateFontInfo LogFont = LogText->GetFont();
250 LogFont.Size = 22;
251 // 밝은 배경(모래/눈밭) 위에서도 읽히도록 검은 윤곽선을 두른다.
252 LogFont.OutlineSettings.OutlineSize = 2;
253 LogFont.OutlineSettings.OutlineColor = FLinearColor::Black;
254 LogText->SetFont(LogFont);
255 if (UHorizontalBoxSlot* TextSlot = LogBox->AddChildToHorizontalBox(LogText))
256 {
257 TextSlot->SetVerticalAlignment(VAlign_Center);
258 }
259
260 if (UCanvasPanelSlot* LogSlot = RootCanvas->AddChildToCanvas(LogBox))
261 {
262 LogSlot->SetAutoSize(true);
263 LogSlot->SetAlignment(FVector2D(0.5f, 1.0f)); // 바닥-중앙 기준으로 머리 위에 선다.
264 LogSlot->SetZOrder(30); // HP바보다 위에 그린다.
265 }
266
267 FFloatingCombatLogEntry Entry;
268 Entry.mRoot = LogBox;
269 Entry.mWorldLocation = Request.mWorldLocation;
270 Entry.mMotionIndex = Request.mMotionIndex;
271 Entry.mElapsed = 0.0f;
272 Entry.mIsPreview = Request.mIsPreview;
273
274 if (Request.mIsPreview == true)
275 {
276 // 미리보기는 상승/페이드가 없으므로, 같은 위치에 이미 뜬 미리보기 수만큼 위로 고정 오프셋을 준다.
277 int32 StackCount = 0;
278 for (const FFloatingCombatLogEntry& Existing : mFloatingCombatLogs)
279 {
280 // 퇴장 중인 로그는 곧 사라지므로 쌓기 오프셋 계산에서 제외한다.
281 if (Existing.mIsPreview == true && Existing.mIsDismissing == false
282 && Existing.mWorldLocation.Equals(Request.mWorldLocation, 1.0f))
283 {
284 ++StackCount;
285 }
286 }
287 Entry.mStackOffsetY = StackCount * FloatingLogPreviewRowSpacing;
288 }
289 mFloatingCombatLogs.Add(Entry);
290
291 // (실행 로그 전용) 같은 위치에 로그가 연달아 뜰 때(예: 데미지+쓰러짐) 겹치지 않게 기존 로그를 위로 민다.
292 // 미리보기는 위 StackOffset으로 처리하므로 여기서 건드리지 않는다.
293 if (Request.mIsPreview == false)
294 {
295 for (int32 LogIndex = 0; LogIndex < mFloatingCombatLogs.Num() - 1; ++LogIndex)
296 {
297 FFloatingCombatLogEntry& Existing = mFloatingCombatLogs[LogIndex];
298 if (Existing.mIsPreview == false && Existing.mWorldLocation.Equals(Entry.mWorldLocation, 1.0f))
299 {
300 Existing.mElapsed = FMath::Max(Existing.mElapsed, 0.35f);
301 }
302 }
303 }

313void UCombatTileMapHUDWidget::UpdateFloatingCombatLogs(float InDeltaTime)
314{
315 if (mFloatingCombatLogs.Num() == 0)
316 {
317 return;
318 }
319
320 // 지도(풀스크린) 열림 중에는 떠 있는 로그를 숨긴다(수명/이동은 멈춤 — 지도 닫으면 재개).
321 if (mCombatControlsHidden)
322 {
323 for (FFloatingCombatLogEntry& Entry : mFloatingCombatLogs)
324 {
325 if (Entry.mRoot != nullptr) { Entry.mRoot->SetVisibility(ESlateVisibility::Collapsed); }
326 }
327 return;
328 }
329
330 APlayerController* PlayerController = GetOwningPlayer();
331
332 for (int32 LogIndex = mFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
333 {
334 FFloatingCombatLogEntry& Entry = mFloatingCombatLogs[LogIndex];
335 Entry.mElapsed += InDeltaTime;
336
337 UWidget* LogRoot = Entry.mRoot;
338 // 미리보기 로그는 수명으로 사라지지 않는다(MotionFinished/Clear로만). 실행 로그만 수명 만료 시 제거.
339 // 퇴장 연출 중이면 수명 만료보다 퇴장(슬라이드+페이드) 완료를 우선한다.
340 if (LogRoot == nullptr || (Entry.mIsPreview == false && Entry.mIsDismissing == false && Entry.mElapsed >= FloatingLogLifetime))
341 {
342 if (LogRoot != nullptr)
343 {
344 LogRoot->RemoveFromParent();
345 }
346 mFloatingCombatLogs.RemoveAt(LogIndex);
347 continue;
348 }
349
350 // 퇴장 중(모션 종료)이면 오른쪽으로 흐르며 페이드아웃하고 연출이 끝나면 제거한다.
351 if (Entry.mIsDismissing == true)
352 {
353 Entry.mDismissElapsed += InDeltaTime;
354 if (Entry.mDismissElapsed >= FloatingLogDismissDuration)
355 {
356 LogRoot->RemoveFromParent();
357 mFloatingCombatLogs.RemoveAt(LogIndex);
358 continue;
359 }
360 }
361
362 // HP바와 같은 투영. 화면 밖이면 숨기되 수명은 계속 흘려보낸다.
363 FVector2D ScreenPosition;
364 const bool bOnScreen = PlayerController != nullptr
365 && UWidgetLayoutLibrary::ProjectWorldLocationToWidgetPosition(
366 PlayerController, Entry.mWorldLocation, ScreenPosition, false);
367 if (bOnScreen == false)
368 {
369 LogRoot->SetVisibility(ESlateVisibility::Collapsed);
370 continue;
371 }
372 LogRoot->SetVisibility(ESlateVisibility::HitTestInvisible);
373
374 // 퇴장 중에는 진행률만큼 오른쪽으로 밀리는 X 오프셋이 붙는다.
375 const float DismissOffsetX = Entry.mIsDismissing == true
376 ? FloatingLogDismissSlideSpeed * Entry.mDismissElapsed
377 : 0.0f;
378
379 if (UCanvasPanelSlot* LogSlot = Cast<UCanvasPanelSlot>(LogRoot->Slot))
380 {
381 // 미리보기: 고정 위치(겹치면 위로 쌓기) / 실행: 시간에 따라 상승.
382 const float OffsetY = Entry.mIsPreview == true
383 ? (FloatingLogBaseOffsetY - Entry.mStackOffsetY)
384 : (FloatingLogBaseOffsetY - FloatingLogRiseSpeed * Entry.mElapsed);
385 LogSlot->SetPosition(ScreenPosition + FVector2D(DismissOffsetX, OffsetY));
386 }
387
388 // 퇴장 중에는 진행률에 따라 서서히 투명해진다(미리보기/실행 공통).
389 if (Entry.mIsDismissing == true)
390 {
391 LogRoot->SetRenderOpacity(FMath::Clamp(1.0f - Entry.mDismissElapsed / FloatingLogDismissDuration, 0.0f, 1.0f));
392 continue;
393 }
394
395 // 미리보기는 페이드 없이 계속 또렷하게(모션 종료/클리어로만 사라짐).
396 if (Entry.mIsPreview == true)
397 {
398 LogRoot->SetRenderOpacity(1.0f);
399 continue;
400 }
401
402 // (실행 로그) 수명 후반부에 서서히 사라진다.
403 const float FadeStart = FloatingLogLifetime * FloatingLogFadePortion;
404 const float Opacity = Entry.mElapsed <= FadeStart
405 ? 1.0f
406 : 1.0f - (Entry.mElapsed - FadeStart) / (FloatingLogLifetime - FadeStart);
407 LogRoot->SetRenderOpacity(FMath::Clamp(Opacity, 0.0f, 1.0f));
408 }
409}

418void UCombatTileMapHUDWidget::RemoveFloatingCombatLogsByMotionIndex(int32 MotionIndex)
419{
420 if (MotionIndex == INDEX_NONE)
421 {
422 return;
423 }
424
425 for (int32 LogIndex = mPendingFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
426 {
427 if (mPendingFloatingCombatLogs[LogIndex].mRequest.mMotionIndex == MotionIndex)
428 {
429 mPendingFloatingCombatLogs.RemoveAt(LogIndex);
430 }
431 }
432
433 for (int32 LogIndex = mFloatingCombatLogs.Num() - 1; LogIndex >= 0; --LogIndex)
434 {
435 FFloatingCombatLogEntry& Entry = mFloatingCombatLogs[LogIndex];
436 if (Entry.mMotionIndex != MotionIndex)
437 {
438 continue;
439 }
440
441 // 뿅 사라지지 않고 퇴장 연출(오른쪽으로 흐르며 페이드아웃)로 넘긴다. 실제 제거는 UpdateFloatingCombatLogs가 한다.
442 if (Entry.mRoot != nullptr)
443 {
444 Entry.mIsDismissing = true;
445 Entry.mDismissElapsed = 0.0f;
446 }
447 else
448 {
449 mFloatingCombatLogs.RemoveAt(LogIndex);
450 }
451 }
