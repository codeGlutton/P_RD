56 TestEqual(TEXT("Combat starts incomplete"), Recreated->GetStage().mClearData.mIsCleared, false);
57 TestEqual(TEXT("Room index restored"), Recreated->GetStage().mCurColumn, 0);
58 TestEqual(TEXT("Event stream resumes before combat"), Recreated->GetEventStream().FRand(), ExpectedEvent.FRand());
59 TestEqual(TEXT("Stage stream resumes before combat"), Recreated->GetStageBuildStream().FRand(), ExpectedStage.FRand());
60 TestTrue(TEXT("Restore in-memory run before frontend autosave"), Checkpoint.Restore(Live.Get()));
61 TestEqual(TEXT("Frontend cannot overwrite entry with combat damage"), Live->HP(0), 80.f);
