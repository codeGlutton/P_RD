20bool UStaticTutorialRoomSpawnData::ValidateLayout(TArray<FText>& Errors) const
21{
22 auto Error = [&Errors](const FString& Text) { Errors.Add(FText::FromString(Text)); };
23 if (mStageLevel != EStageLevelType::None) Error(TEXT("Tutorial StageLevel must be None; otherwise it enters normal random rooms."));
24 if (mUseRandomSpawnSetting || mDefaultSpawnSettingName.IsNone()) Error(TEXT("Tutorial requires a named, fixed background spawn setting."));
25 if (mPlayerTransforms.Num() != 3) Error(TEXT("Tutorial needs three party placements."));
26 if (mEnemyUnitPlacementDatas.Num() < 2) Error(TEXT("Tutorial needs at least two enemies so one attack cannot skip End Turn."));
27 if (GetTargetTile() == FTileIndex::Invalid) Error(TEXT("Tutorial TargetEnemyIndex must select an authored enemy."));
28 TSet<FTileIndex> Occupied;
29 auto Claim = [&Occupied, &Error](const FTileIndex& Tile)
30 {
31 if (Occupied.Contains(Tile)) Error(FString::Printf(TEXT("Tutorial overlapping occupancy at (%d,%d)."), Tile.mX, Tile.mY));
32 Occupied.Add(Tile);
33 };
34 for (const auto& Player : mPlayerTransforms) Claim(Player.mIndex);
35 for (const auto& Enemy : mEnemyUnitPlacementDatas) Claim(Enemy.mTransform.mIndex);
36 for (const auto& Placement : mObstaclePlacementDatas)
37 {
38 Claim(Placement.mTransform.mIndex);
39 const auto* Obstacle = Placement.mSpawnData.LoadSynchronous();
40 if (!Obstacle) { Error(TEXT("Tutorial obstacle asset cannot be loaded.")); continue; }
41 for (const FTileIndex& Relative : Obstacle->mRequiredEmptyTiles)
42 Claim(LocalToTileMapTransform(FTileTransform(Relative), Placement.mTransform).mIndex);
43 }
44 if (MoveDestination == FTileIndex::Invalid || Occupied.Contains(MoveDestination)) Error(TEXT("Tutorial move destination must be an empty tile."));
45 return Errors.IsEmpty();
46}
