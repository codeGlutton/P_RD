228 int32 MakeBrushTexturesResident(UShopUIWidgetBase& ShopWidget)
229 {
230 if (ShopWidget.WidgetTree == nullptr)
231 {
232 return 0;
233 }
234
235 TSet<UTexture2D*> Textures;
236 auto CollectBrush = [&Textures](const FSlateBrush& Brush)
237 {
238 if (UTexture2D* Texture = Cast<UTexture2D>(Brush.GetResourceObject()))
239 {
240 Textures.Add(Texture);
241 }
242 };
243 auto CollectStyle = [&CollectBrush](const FSlateWidgetStyle& Style)
244 {
245 TArray<const FSlateBrush*> Brushes;
246 Style.GetResources(Brushes);
247 for (const FSlateBrush* Brush : Brushes)
248 {
249 if (Brush != nullptr)
250 {
251 CollectBrush(*Brush);
252 }
253 }
254 };
255
256 ShopWidget.WidgetTree->ForEachWidget(
257 [&CollectBrush, &CollectStyle](UWidget* Widget)
258 {
259 if (const UImage* Image = Cast<UImage>(Widget))
260 {
261 CollectBrush(Image->GetBrush());
262 }
263 else if (const UBorder* Border = Cast<UBorder>(Widget))
264 {
265 CollectBrush(Border->Background);
266 }
267 else if (const UButton* Button = Cast<UButton>(Widget))
268 {
269 CollectStyle(Button->GetStyle());
270 }
271 });
272
273 for (UTexture2D* Texture : Textures)
274 {
275 Texture->UpdateResource();
276 Texture->SetForceMipLevelsToBeResident(30.f);
277 Texture->WaitForStreaming();
278 }
279 FlushRenderingCommands();
280 return Textures.Num();
281 }

283 bool Capture(UShopUIWidgetBase& ShopWidget, const TSharedRef<SWidget>& ShopSlate,
284 const TCHAR* FileName, TArray<FColor>& OutPixels, FString& OutError)
285 {
286 ShopWidget.SetVisibility(ESlateVisibility::SelfHitTestInvisible);
287 ShopWidget.ForceLayoutPrepass();
288 const int32 TextureCount = MakeBrushTexturesResident(ShopWidget);
289 if (TextureCount == 0)
290 {
291 OutError = TEXT("상점 WBP 브러시에 Texture2D가 하나도 없음");
292 return false;
293 }
294
295 const TSharedRef<SWidget> CaptureRoot =
296 SNew(SOverlay)
297 + SOverlay::Slot()
298 [
299 SNew(SColorBlock).Color(FLinearColor(0.008f, 0.012f, 0.018f, 1.f))
300 ]
301 + SOverlay::Slot()
302 [
303 ShopSlate
304 ];
305
306 FWidgetRenderer Renderer(true, true);
307 Renderer.SetIsPrepassNeeded(true);
308 UTextureRenderTarget2D* RenderTarget = Renderer.DrawWidget(
309 CaptureRoot, FVector2D(CaptureWidth, CaptureHeight));
310 if (RenderTarget == nullptr)
311 {
312 OutError = TEXT("FWidgetRenderer가 상점 렌더 타깃을 만들지 못함");
313 return false;
314 }
315
316 FlushRenderingCommands();
317 FReadSurfaceDataFlags ReadFlags(RCM_UNorm);
318 ReadFlags.SetLinearToGamma(false);
319 if (!RenderTarget->GameThread_GetRenderTargetResource()->ReadPixels(
320 OutPixels, ReadFlags)
321 || OutPixels.Num() != CaptureWidth * CaptureHeight)
322 {
323 OutError = TEXT("1672x941 상점 렌더 결과를 읽지 못함");
324 return false;
325 }
326
327 uint8 MinChannel = 255;
328 uint8 MaxChannel = 0;
329 for (FColor& Pixel : OutPixels)
330 {
331 // RGBA8 WidgetRenderer 경로에서 중복 적용된 감마를 한 번 되돌린다.
332 Pixel.R = uint8(FMath::RoundToInt(
333 255.f * FMath::Pow(Pixel.R / 255.f, 2.2f)));
334 Pixel.G = uint8(FMath::RoundToInt(
335 255.f * FMath::Pow(Pixel.G / 255.f, 2.2f)));
336 Pixel.B = uint8(FMath::RoundToInt(
337 255.f * FMath::Pow(Pixel.B / 255.f, 2.2f)));
338 MinChannel = FMath::Min3(MinChannel, Pixel.R,
339 FMath::Min(Pixel.G, Pixel.B));
340 MaxChannel = FMath::Max3(MaxChannel, Pixel.R,
341 FMath::Max(Pixel.G, Pixel.B));
342 }
343 if (int32(MaxChannel) - int32(MinChannel) < 16)
344 {
345 OutError = TEXT("상점 캡처가 단색이다. WBP가 그려지지 않음");
346 return false;
347 }
348
349 TArray64<uint8> PngData;
350 FImageUtils::PNGCompressImageArray(
351 CaptureWidth, CaptureHeight, OutPixels, PngData);
352 IFileManager::Get().MakeDirectory(*OutputDirectory(), true);
353 const FString OutputPath = FPaths::Combine(OutputDirectory(), FileName);
354 if (!FFileHelper::SaveArrayToFile(PngData, *OutputPath))
355 {
356 OutError = FString::Printf(TEXT("상점 캡처 저장 실패: %s"), *OutputPath);
357 return false;
358 }
359
360 UE_LOG(LogTemp, Display, TEXT("[ShopUI] %s (%d textures) -> %s"),
361 FileName, TextureCount, *OutputPath);
362 return true;
363 }

365 int32 CountChangedPixels(const TArray<FColor>& A, const TArray<FColor>& B)
366 {
367 if (A.Num() != B.Num())
368 {
369 return 0;
370 }
371
372 int32 ChangedPixels = 0;
373 for (int32 Index = 0; Index < A.Num(); ++Index)
374 {
375 if (FMath::Abs(int32(A[Index].R) - int32(B[Index].R))
376 + FMath::Abs(int32(A[Index].G) - int32(B[Index].G))
377 + FMath::Abs(int32(A[Index].B) - int32(B[Index].B)) > 12)
378 {
379 ++ChangedPixels;
380 }
381 }
382 return ChangedPixels;
383 }
