58 void Show(UWorld* World)
59 {
60 // 이미 떠 있으면 닫고 새로 연다
61 if (ShownWidget.IsValid())
62 {
63 ShownWidget->RemoveFromParent();
64 ShownWidget.Reset();
65 }
66
67 UShopUIModel* Model = FindLiveModel(World);
68 if (Model == nullptr)
69 {
70 return;
71 }
72
73 UClass* WidgetClass = LoadClass<UShopUIWidgetBase>(nullptr, ShopWidgetPath);
74 if (WidgetClass == nullptr)
75 {
76 UE_LOG(LogRD, Warning,
77 TEXT("RD.ShopPreview: %s 클래스를 찾지 못했습니다."), ShopWidgetPath);
78 return;
79 }
80
81 APlayerController* Controller = World->GetFirstPlayerController();
82 UShopUIWidgetBase* Widget = Controller != nullptr
83 ? CreateWidget<UShopUIWidgetBase>(Controller, WidgetClass)
84 : CreateWidget<UShopUIWidgetBase>(World, WidgetClass);
85 if (Widget == nullptr)
86 {
87 UE_LOG(LogRD, Warning, TEXT("RD.ShopPreview: 위젯 생성에 실패했습니다."));
88 return;
89 }
90
91 Widget->BindUIModel(Model);
92 Widget->OpenUI();
93 ShownWidget = Widget;
94
95 UE_LOG(LogRD, Display, TEXT("RD.ShopPreview: 상점 화면을 열었습니다."));
96 }
