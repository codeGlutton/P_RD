97 // @param BlockerIndex 길/시야를 막는 제3의 유닛 배치 좌표 (Invalid면 미배치. 우회 접근 검증용)
98 TArray<TInstancedStruct<FSRPGCommand>> Plan(
99 UWorld* World,
100 TArray<UObject*>& KeepAlive,
101 EMoveTendency Tendency,
102 int32 MoveRange,
103 int32 AimRange,
104 int32 TileMapWidth,
105 int32 TileMapHeight,
106 FTileIndex EnemyIndex,
107 FTileIndex PlayerIndex,
108 EAimPattern AimPattern = EAimPattern::Square,
109 int32 SecondSkillAimRange = 0,
110 FTileIndex BlockerIndex = FTileIndex::Invalid)
111 {
112 // 타일맵 생성

114 TileMap->SetDimensions(TileMapWidth, TileMapHeight);

117 UMockPlayerUnitModel* Player = NewObject<UMockPlayerUnitModel>(World);
118
119 // 적유닛 생성
120 UMockEnemyUnitModel* Enemy = NewObject<UMockEnemyUnitModel>(World);
121 Enemy->Initialize();

123 Enemy->SetMoveTendency(Tendency);
124 Enemy->SetMovePoint(MoveRange);
125
126 // 스킬 추가: 일반공격 계열
127 Enemy->GetSkillComponentModel()->SetSkill(0, MakeSkill(World, KeepAlive, AimPattern, AimRange));
128 // 두 번째 스킬(옵션): 스킬 랜덤 선택 검증용
129 if (SecondSkillAimRange > 0)
130 {
131 Enemy->GetSkillComponentModel()->SetSkill(1, MakeSkill(World, KeepAlive, EAimPattern::Square, SecondSkillAimRange));
132 }
133
134 // 플레이어유닛과 적유닛 배치
135 TileMap->PlaceActor(FTileTransform(PlayerIndex), Player);
136 TileMap->PlaceActor(FTileTransform(EnemyIndex), Enemy);
137
138 // 차단 유닛(옵션): 길/시야를 막는 제3의 유닛 (Unit 레이어 점유만 필요하니 플레이어 Mock 재사용)
139 if (TileMap->IsValidIndex(BlockerIndex))

142 TileMap->PlaceActor(FTileTransform(BlockerIndex), Blocker);
143 }

146 const FRandomStream Stream(20260710);
147
148 // AI 돌려서(ㅋㅋ) 적유닛의 예상커맨드 획득
149 return USRPGEnemyTurnPlanner::PlanTurn(Enemy, Player, TileMap, Stream);
150 }

152 // @brief 커맨드가 비어있지 않고, 마지막은 항상 턴 종료인 지 검증

163 // @brief 이동해서도 플레이어를 조준 불가능한 케이스 검증
164 void CheckApproachNoCast(
165 FAutomationTestBase& Test,
166 const TArray<TInstancedStruct<FSRPGCommand>>& Commands,
167 const TCHAR* Label,
168 FTileIndex ExpectedDest)
169 {
170 CheckTail(Test, Commands, Label);
171
172 // 이동커맨드가 있고 경로가 2칸 이상이어야 목적지 검증이 의미 있음
173 const FSRPGMoveCommand* Move = FindMoveCommand(Commands);
174 if (Test.TestTrue(FString::Printf(TEXT("[%s] 이동커맨드 존재"), Label), Move != nullptr) &&
175 Test.TestTrue(FString::Printf(TEXT("[%s] 경로는 2칸 이상"), Label), Move->mPathTileIndexes.Num() >= 2))
176 {
177 // 목적지 = 경로 마지막 (FindPath가 goal을 마지막 원소로 넣음)
178 const FTileIndex Dest = Move->mPathTileIndexes.Last();
179 Test.TestTrue(FString::Printf(TEXT("[%s] 목적지=(%d,%d)"), Label, ExpectedDest.mX, ExpectedDest.mY),
180 Dest == ExpectedDest);
181 }
182
183 // 도착해서도 조준범위 밖이라 스킬캐스팅 없어야 함
184 Test.TestTrue(FString::Printf(TEXT("[%s] 스킬커맨드 없음(이동 후에도 조준 불가능)"), Label), FindCast(Commands) == nullptr);
185 }
