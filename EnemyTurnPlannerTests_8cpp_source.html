87 // @brief 테스트용 일반공격 스킬 생성 (KeepAlive에 등록해 GC 방지)
88 // @param EffectPattern 영향 범위 패턴 (Single 조준 스킬은 이 범위로 타격 여부가 결정됨)
89 // @param EffectArea 영향 범위 크기
90 // @param SkillCost 시전 비용 (플래너 AP 판정용, 0이면 비용 없음)
91 // @param SkillType 스킬 타입 (Attack은 공격 후보, Spell은 자기 버프)
92 UStaticSkillData* MakeSkill(UWorld* World, TArray<UObject*>& KeepAlive, EAimPattern AimPattern, int32 AimRange,
93 EEffectPattern EffectPattern = EEffectPattern::Single, int32 EffectArea = 0, int32 SkillCost = 0,
94 ESkillType SkillType = ESkillType::Attack)
95 {
96 UStaticUnitSkillData* Skill = NewObject<UStaticUnitSkillData>(World);
97 Skill->mJobType = EUnitJobType::Common; // 직업 무관 (직업 불일치면 SetSkill이 장착 거부)
98 Skill->mSkillType = SkillType;
99 Skill->mRequiredActionPoint = SkillCost;
100 Skill->mAimPattern = AimPattern;
101 Skill->mAimRange = AimRange;
102 Skill->mCanAimBoardActor = true;
103 Skill->mAimBlockerMask = static_cast<int32>(ETileLayerFlag::Obstacle | ETileLayerFlag::Unit);
104 Skill->mEffectPattern = EffectPattern;
105 Skill->mEffectArea = EffectArea;
106 Skill->mEffectBlockerMask = static_cast<int32>(ETileLayerFlag::Obstacle | ETileLayerFlag::Unit);
107 // 모션 레이어 없으면 FSkillEntry::IsValid()가 미장착으로 판정하므로 더미 1개 추가
108 Skill->mSkillPhaseLayers.AddDefaulted();
109 Skill->AddToRoot();
110 KeepAlive.Add(Skill);
111 return Skill;
112 }

114 // @brief 적 유닛의 한 턴을 계획해 커맨드 목록 반환
115 // @param KeepAlive SkillComponent가 약참조라 GC 안당하도록 붙잡아두는 장치
116 // @param AimPattern 스킬 조준 패턴 (Cross/Star 직사는 시야 검사 경로를 태움)
117 // @param Slot1AimRange 슬롯 1에 장착할 Square 스킬의 사거리 (0이면 미장착. 다중 스킬 선택 검증용)
118 // @param BlockerIndex 길/시야를 막는 제3의 유닛 배치 좌표 (Invalid면 미배치. 우회 접근 검증용)
119 // @param SecondPlayerIndex 두 번째 플레이어 배치 좌표 (Invalid면 미배치. 다중 타겟 선택 검증용)
120 // @param Rooted 속박 상태 여부 (이동 불가 계획 검증용)
121 // @param EffectPattern 슬롯 0 스킬의 영향 범위 패턴 (Single 조준 검증용)
122 // @param EffectArea 슬롯 0 스킬의 영향 범위 크기
123 // @param SkillCost 슬롯 0 스킬의 시전 비용 (이동과 시전의 AP 분배 검증용)
124 // @param Slot2SpellCost 슬롯 2에 장착할 자기 버프의 시전 비용 (0이면 미장착. 버프 턴 검증용)
125 // @param HasAttack 슬롯 0 공격 스킬 장착 여부 (거짓이면 버프만 가진 적. 성향 이동 검증용)
126 // @param Slot1Priority 슬롯 1의 선택 우선순위 (우선순위 선택 검증용)
127 // @param Slot2Priority 슬롯 2 버프의 선택 우선순위 (공격/버프 우선순위 검증용)
128 // @param Slot1Cooldown 슬롯 1을 쿨다운 상태로 둘지 (쿨다운이면 차순위 검증용)
129 TArray<TInstancedStruct<FSRPGCommand>> Plan(
130 UWorld* World,
131 TArray<UObject*>& KeepAlive,
132 EMoveTendency Tendency,
133 int32 MoveRange,
134 int32 AimRange,
135 int32 TileMapWidth,
136 int32 TileMapHeight,
137 FTileIndex EnemyIndex,
138 FTileIndex PlayerIndex,
139 EAimPattern AimPattern = EAimPattern::Square,
140 int32 Slot1AimRange = 0,
141 FTileIndex BlockerIndex = FTileIndex::Invalid,
142 FTileIndex SecondPlayerIndex = FTileIndex::Invalid,
143 bool Rooted = false,
144 EEffectPattern EffectPattern = EEffectPattern::Single,
145 int32 EffectArea = 0,
146 int32 SkillCost = 0,
147 int32 Slot2SpellCost = 0,
148 bool HasAttack = true,
149 ESkillPriority Slot1Priority = ESkillPriority::Normal,
150 ESkillPriority Slot2Priority = ESkillPriority::Normal,
151 bool Slot1Cooldown = false)
152 {
153 // 타일맵 생성
154 UTileMapModel* TileMap = NewObject<UTileMapModel>(World);
155 TileMap->SetDimensions(TileMapWidth, TileMapHeight);
156
157 // 플레이어유닛 생성 (배열 순서가 곧 타겟 인덱스)
158 TArray<UUnitModel*> Players;
159 UMockPlayerUnitModel* Player = NewObject<UMockPlayerUnitModel>(World);
160 Players.Add(Player);
161
162 // 적유닛 생성
163 UMockEnemyUnitModel* Enemy = NewObject<UMockEnemyUnitModel>(World);
164 Enemy->Initialize();
165 Enemy->BeginPlay();
166 Enemy->SetMoveTendency(Tendency);
167 Enemy->GetAttributeComponentModel()->ApplyModToAttribute(UEnemyUnitAttributeSet::GetActionPointAttribute(), ETacticalModOp::Override, MoveRange);
168
169 // 속박 옵션: 이동불가 상태의 계획 검증용
170 if (Rooted == true)
171 {
172 Enemy->GetAttributeComponentModel()->AddLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Root, 1);
173 }
174
175 // 스킬 슬롯 풀 할당: Mock은 스폰 데이터 초기화를 건너뛰므로 빈 목록으로 슬롯만 확보
176 Enemy->GetSkillComponentModel()->SetSkillFrom(TArray<TSoftObjectPtr<UStaticSkillData>>());
177 // 스킬 추가: 일반공격 계열 (버프만 가진 적 검증 시 생략)
178 if (HasAttack == true)
179 {
180 Enemy->GetSkillComponentModel()->SetSkill(0, MakeSkill(World, KeepAlive, AimPattern, AimRange, EffectPattern, EffectArea, SkillCost));
181 }
182 // 슬롯 1 스킬(옵션): 다중 스킬 선택 검증용 (우선순위/쿨다운 지정 가능)
183 if (Slot1AimRange > 0)
184 {
185 Enemy->GetSkillComponentModel()->SetSkill(1, MakeSkill(World, KeepAlive, EAimPattern::Square, Slot1AimRange));
186 Enemy->SetSkillPriority(1, Slot1Priority);
187 if (Slot1Cooldown == true)
188 {
189 Enemy->SetSkillCooldown(1);
190 }
191 }
192 // 슬롯 2 자기 버프(옵션): Single 조준, 영향 범위 자기 칸
193 if (Slot2SpellCost > 0)
194 {
195 Enemy->GetSkillComponentModel()->SetSkill(2, MakeSkill(World, KeepAlive, EAimPattern::Single, 0,
196 EEffectPattern::Single, 0, Slot2SpellCost, ESkillType::Spell));
197 Enemy->SetSkillPriority(2, Slot2Priority);
198 }
199
200 // 플레이어유닛과 적유닛 배치
201 TileMap->PlaceActor(FTileTransform(PlayerIndex), Player);
202 TileMap->PlaceActor(FTileTransform(EnemyIndex), Enemy);
203
204 // 두 번째 플레이어(옵션): 다중 타겟 선택 검증용
205 if (TileMap->IsValidIndex(SecondPlayerIndex))
206 {
207 UMockPlayerUnitModel* SecondPlayer = NewObject<UMockPlayerUnitModel>(World);
208 Players.Add(SecondPlayer);
209 TileMap->PlaceActor(FTileTransform(SecondPlayerIndex), SecondPlayer);
210 }
211
212 // 차단 유닛(옵션): 길/시야를 막는 제3의 유닛 (Unit 레이어 점유만 필요하니 플레이어 Mock 재사용)
213 if (TileMap->IsValidIndex(BlockerIndex))
214 {
215 UMockPlayerUnitModel* Blocker = NewObject<UMockPlayerUnitModel>(World);
216 TileMap->PlaceActor(FTileTransform(BlockerIndex), Blocker);
217 }
218
219 // 스킬 랜덤 선택용 고정 시드 스트림 (테스트 결정성 보장)
220 const FRandomStream Stream(20260710);
221
222 // AI 돌려서(ㅋㅋ) 적유닛의 예상커맨드 획득
223 return USRPGEnemyTurnPlanner::PlanTurn(Enemy, Players, TileMap, Stream);
224 }

226 // @brief 커맨드가 비어있지 않고, 마지막은 항상 턴 종료인 지 검증
227 void CheckTail(FAutomationTestBase& Test, const TArray<TInstancedStruct<FSRPGCommand>>& Commands, const TCHAR* Label)
228 {
229 if (Test.TestTrue(FString::Printf(TEXT("[%s] 커맨드 존재"), Label),
230 Commands.Num() > 0))
231 {
232 Test.TestTrue(FString::Printf(TEXT("[%s] 마지막은 턴 종료"), Label),
233 Commands.Last().Get().GetCommandType() == ESRPGCommandType::TurnEnd);
234 }
235 }

237 // @brief 이동해서도 플레이어를 조준 불가능한 케이스 검증
238 void CheckApproachNoCast(
239 FAutomationTestBase& Test,
240 const TArray<TInstancedStruct<FSRPGCommand>>& Commands,
241 const TCHAR* Label,
242 FTileIndex ExpectedDest)
243 {
244 CheckTail(Test, Commands, Label);
245
246 // 이동커맨드가 있고 경로가 2칸 이상이어야 목적지 검증이 의미 있음
247 const FSRPGMoveCommand* Move = FindMoveCommand(Commands);
248 if (Test.TestTrue(FString::Printf(TEXT("[%s] 이동커맨드 존재"), Label), Move != nullptr) &&
249 Test.TestTrue(FString::Printf(TEXT("[%s] 경로는 2칸 이상"), Label), Move->mPathTileIndexes.Num() >= 2))
250 {
251 // 목적지 = 경로 마지막 (FindPath가 goal을 마지막 원소로 넣음)
252 const FTileIndex Dest = Move->mPathTileIndexes.Last();
253 Test.TestTrue(FString::Printf(TEXT("[%s] 목적지=(%d,%d)"), Label, ExpectedDest.mX, ExpectedDest.mY),
254 Dest == ExpectedDest);
255 }
256
257 // 도착해서도 조준범위 밖이라 스킬캐스팅 없어야 함
258 Test.TestTrue(FString::Printf(TEXT("[%s] 스킬커맨드 없음(이동 후에도 조준 불가능)"), Label), FindCast(Commands) == nullptr);
259 }
