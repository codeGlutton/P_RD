122 // @param EffectPattern 슬롯 0 스킬의 영향 범위 패턴 (Single 조준 검증용)
123 // @param EffectArea 슬롯 0 스킬의 영향 범위 크기
124 // @param SkillCost 슬롯 0 스킬의 시전 비용 (이동과 시전의 AP 분배 검증용)
125 // @param Slot2SpellCost 슬롯 2에 장착할 자기 버프의 시전 비용 (0이면 미장착. 버프 턴 검증용)
126 // @param HasAttack 슬롯 0 공격 스킬 장착 여부 (거짓이면 버프만 가진 적. 성향 이동 검증용)

130 TArray<TInstancedStruct<FSRPGCommand>> Plan(
131 UWorld* World,
132 FTestObjectScope& Scope,
133 EMoveTendency Tendency,
134 int32 MoveRange,
135 int32 AimRange,
136 int32 TileMapWidth,
137 int32 TileMapHeight,
138 FTileIndex EnemyIndex,
139 FTileIndex PlayerIndex,
140 EAimPattern AimPattern = EAimPattern::Square,
141 int32 Slot1AimRange = 0,
142 FTileIndex BlockerIndex = FTileIndex::Invalid,
143 FTileIndex SecondPlayerIndex = FTileIndex::Invalid,
144 bool Rooted = false,
145 EEffectPattern EffectPattern = EEffectPattern::Single,
146 int32 EffectArea = 0,
147 int32 SkillCost = 0,
148 int32 Slot2SpellCost = 0,
149 bool HasAttack = true,
150 ESkillPriority Slot1Priority = ESkillPriority::Normal,
151 ESkillPriority Slot2Priority = ESkillPriority::Normal,
152 bool Slot1Cooldown = false)

156 TileMap->SetDimensions(TileMapWidth, TileMapHeight);
157
158 // 플레이어유닛 생성 (배열 순서가 곧 타겟 인덱스)
159 TArray<UUnitModel*> Players;
160 UMockPlayerUnitModel* Player = Scope.New<UMockPlayerUnitModel>(World);
161 Players.Add(Player);

163 // 적유닛 생성
164 UMockEnemyUnitModel* Enemy = Scope.New<UMockEnemyUnitModel>(World);
165 Enemy->Initialize();

168 Enemy->GetAttributeComponentModel()->ApplyModToAttribute(UEnemyUnitAttributeSet::GetActionPointAttribute(), ETacticalModOp::Override, MoveRange);
169
170 // 속박 옵션: 이동불가 상태의 계획 검증용
171 if (Rooted == true)
172 {
173 Enemy->GetAttributeComponentModel()->AddLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Root, 1);
174 }
175
176 // 스킬 슬롯 풀 할당: Mock은 스폰 데이터 초기화를 건너뛰므로 빈 목록으로 슬롯만 확보
177 Enemy->GetSkillComponentModel()->SetSkillFrom(TArray<TSoftObjectPtr<UStaticSkillData>>());
178 // 스킬 추가: 일반공격 계열 (버프만 가진 적 검증 시 생략)

183 // 슬롯 1 스킬(옵션): 다중 스킬 선택 검증용 (우선순위/쿨다운 지정 가능)
184 if (Slot1AimRange > 0)
185 {
186 Enemy->GetSkillComponentModel()->SetSkill(1, MakeSkill(World, Scope, EAimPattern::Square, Slot1AimRange));
187 Enemy->SetSkillPriority(1, Slot1Priority);
188 if (Slot1Cooldown == true)
189 {
190 Enemy->SetSkillCooldown(1);
191 }
192 }
193 // 슬롯 2 자기 버프(옵션): Single 조준, 영향 범위 자기 칸

195 {

202 TileMap->PlaceActor(FTileTransform(PlayerIndex), Player);

204
205 // 두 번째 플레이어(옵션): 다중 타겟 선택 검증용

214 if (TileMap->IsValidIndex(BlockerIndex))

223 // AI 돌려서(ㅋㅋ) 적유닛의 예상커맨드 획득

227 // @brief 이동 스킬 하나만 가진 적 유닛의 한 턴을 계획해 커맨드 목록 반환
228 // @param Scope 생성 객체 등록 (GC 방지 + 종료 시 파괴)
229 // @param ActionPoint 적 유닛의 행동력
230 // @param AimRange 이동 스킬의 조준 사거리 (Square 패턴, 빈 타일만 조준)
231 // @param SkillCost 이동 스킬의 시전 비용
232 // @param PlayerIndexes 플레이어 배치 좌표들 (배열 순서가 곧 타겟 인덱스)
233 // @param BlockerIndexes 타일을 점유하는 제3의 유닛 배치 좌표들 (타겟 아님. 점유 제외와 경로 차단 검증용)
234 // @param Priority 이동 스킬의 대상우선순위 (기준 타겟 선정 검증용)
235 TArray<TInstancedStruct<FSRPGCommand>> PlanMoveSkill(
236 UWorld* World,
237 FTestObjectScope& Scope,
238 int32 ActionPoint,
239 int32 AimRange,
240 int32 SkillCost,
241 int32 TileMapWidth,
242 int32 TileMapHeight,
243 FTileIndex EnemyIndex,
244 const TArray<FTileIndex>& PlayerIndexes,
245 const TArray<FTileIndex>& BlockerIndexes = TArray<FTileIndex>(),
246 EEnemyTargetPriority Priority = EEnemyTargetPriority::Nearest)
247 {
248 // 타일맵 생성
249 UTileMapModel* TileMap = Scope.New<UTileMapModel>(World);
250 TileMap->SetDimensions(TileMapWidth, TileMapHeight);
251
252 // 적유닛 생성 (시전 불가 시 접근 이동 검증을 위해 근거리 성향)
253 UMockEnemyUnitModel* Enemy = Scope.New<UMockEnemyUnitModel>(World);
254 Enemy->Initialize();
255 Enemy->BeginPlay();
256 Enemy->SetMoveTendency(EMoveTendency::MoveClose);
257 Enemy->GetAttributeComponentModel()->ApplyModToAttribute(UEnemyUnitAttributeSet::GetActionPointAttribute(), ETacticalModOp::Override, ActionPoint);
258
259 // 이동 스킬 생성: 점유 타일은 조준 불가, 대상우선순위는 스킬 DA 값으로 지정
260 UStaticUnitSkillData* MoveSkill = CastChecked<UStaticUnitSkillData>(MakeSkill(World, Scope, EAimPattern::Square, AimRange,
261 EEffectPattern::Single, 0, SkillCost, ESkillType::Move));
262 MoveSkill->mCanAimBoardActor = false;
263 MoveSkill->mTargetPolicy.mPriority = Priority;
264
265 // 스킬 슬롯 풀 할당 후 슬롯 0에 장착
266 Enemy->GetSkillComponentModel()->SetSkillFrom(TArray<TSoftObjectPtr<UStaticSkillData>>());
267 Enemy->GetSkillComponentModel()->SetSkill(0, MoveSkill);
268
269 // 적유닛 배치
270 TileMap->PlaceActor(FTileTransform(EnemyIndex), Enemy);
271
272 // 플레이어유닛 생성 및 배치 (배열 순서가 곧 타겟 인덱스)
273 TArray<UUnitModel*> Players;
274 for (const FTileIndex& PlayerIndex : PlayerIndexes)
275 {
276 UMockPlayerUnitModel* Player = Scope.New<UMockPlayerUnitModel>(World);
277 Players.Add(Player);
278 TileMap->PlaceActor(FTileTransform(PlayerIndex), Player);
279 }
280
281 // 차단 유닛 배치 (Unit 레이어 점유만 필요하니 플레이어 Mock 재사용, 타겟 목록에는 넣지 않음)
282 for (const FTileIndex& BlockerIndex : BlockerIndexes)
283 {
284 UMockPlayerUnitModel* Blocker = Scope.New<UMockPlayerUnitModel>(World);
285 TileMap->PlaceActor(FTileTransform(BlockerIndex), Blocker);
286 }
287
288 // 고정 시드 스트림 (테스트 결정성 보장)
289 const FRandomStream Stream(20260710);
290
291 return USRPGEnemyTurnPlanner::PlanTurn(Enemy, Players, TileMap, Stream);
292 }

305 // @brief 이동해서도 플레이어를 조준 불가능한 케이스 검증
306 void CheckApproachNoCast(
307 FAutomationTestBase& Test,
308 const TArray<TInstancedStruct<FSRPGCommand>>& Commands,
309 const TCHAR* Label,
310 FTileIndex ExpectedDest)
311 {
312 CheckTail(Test, Commands, Label);
313
314 // 이동커맨드가 있고 경로가 2칸 이상이어야 목적지 검증이 의미 있음
315 const FSRPGMoveCommand* Move = FindMoveCommand(Commands);
316 if (Test.TestTrue(FString::Printf(TEXT("[%s] 이동커맨드 존재"), Label), Move != nullptr) &&
317 Test.TestTrue(FString::Printf(TEXT("[%s] 경로는 2칸 이상"), Label), Move->mPathTileIndexes.Num() >= 2))
318 {
319 // 목적지 = 경로 마지막 (FindPath가 goal을 마지막 원소로 넣음)
320 const FTileIndex Dest = Move->mPathTileIndexes.Last();
321 Test.TestTrue(FString::Printf(TEXT("[%s] 목적지=(%d,%d)"), Label, ExpectedDest.mX, ExpectedDest.mY),
322 Dest == ExpectedDest);
323 }
324
325 // 도착해서도 조준범위 밖이라 스킬캐스팅 없어야 함
326 Test.TestTrue(FString::Printf(TEXT("[%s] 스킬커맨드 없음(이동 후에도 조준 불가능)"), Label), FindCast(Commands) == nullptr);
327 }
