388 WidgetTree->ForEachWidget([this, bApplyPressVisual](UWidget* Widget)

397 {
398 ApplyCommonButtonPressSound(Button);

401 /*
402 * 호버 피드백(0811 점검). 마우스를 올려도 아무 변화가 없어
403 * "반응이 없다" 로 읽혔다 -- 모든 화면 공통으로, 호버 브러시가

408 const FSlateBrush& Hovered = Style.Hovered;

410 Hovered.DrawAs == ESlateBrushDrawType::NoDrawType

415 FSlateBrush HoverBrush;
416 HoverBrush.TintColor = FSlateColor(FLinearColor(1.0f, 1.0f, 1.0f, 0.12f));

423 if (bApplyPressVisual == false)

439void URDUserWidget::CollectButtonCompanions(const UButton* Button, TArray<UWidget*>& OutCompanions) const
440{
441 if (Button == nullptr || WidgetTree == nullptr)
442 {
443 return;
444 }
445
446 // 버튼명을 "<Base>__<Profile>"로 보고 Base/접미사를 분리한다(접미사 없으면 Base=전체, 접미사="").
447 const FString FullName = Button->GetName();
448 FString Base = FullName;
449 FString Suffix;
450 const int32 ProfileSep = FullName.Find(TEXT("__"), ESearchCase::CaseSensitive, ESearchDir::FromStart);
451 if (ProfileSep != INDEX_NONE)
452 {
453 Base = FullName.Left(ProfileSep); // 예: "StartButton"
454 Suffix = FullName.Mid(ProfileSep); // 예: "__base_16_9"
455 }
456
457 // 같은 트리에서 Base로 시작 + 같은 접미사로 끝나는 Image/Text 형제를 짝으로 모은다.
458 WidgetTree->ForEachWidget([&Base, &Suffix, Button, &OutCompanions](UWidget* Widget)
459 {
460 if (Widget == nullptr || Widget == Button)
461 {
462 return;
463 }
464 if (Widget->IsA<UImage>() == false && Widget->IsA<UTextBlock>() == false)
465 {
466 return;
467 }
468 const FString Name = Widget->GetName();
469 if (Name.StartsWith(Base) == true && Name.EndsWith(Suffix) == true)
470 {
471 OutCompanions.Add(Widget);
472 }
473 });
474}

476void URDUserWidget::HandleAnyButtonPressed()
477{
478 // OnPressed는 발신 버튼을 알려주지 않으므로, 실제로 눌린 버튼만 골라 처리한다(동시에 하나만 눌리는 게 보통).
479 for (int32 Index = 0; Index < mFeedbackButtons.Num(); ++Index)
480 {
481 UButton* Button = mFeedbackButtons[Index];
482 if (Button == nullptr || Button->IsPressed() == false)
483 {
484 continue;
485 }
486
487 // (1) 버튼 자체 — 버튼 브러시가 시각요소인 경우(클래스 선택 등)에 배경색을 어둡게 + 축소.
488 FLinearColor Pressed = mFeedbackButtonBaseColors.IsValidIndex(Index) ? mFeedbackButtonBaseColors[Index] : FLinearColor::White;
489 Pressed.R *= ButtonPressColorMul;
490 Pressed.G *= ButtonPressColorMul;
491 Pressed.B *= ButtonPressColorMul; // 알파는 유지, RGB만 어둡게.
492 Button->SetBackgroundColor(Pressed);
493 Button->SetRenderScale(FVector2D(ButtonPressScale, ButtonPressScale));
494
495 // (2) 동반 시각 위젯 — 투명 버튼 위에 얹힌 프레임/텍스트(타이틀 등). 이쪽을 어둡게+축소해야 실제로 보인다.
496 // 이전 잔여가 있으면 먼저 복원 후 새로 적용(한 번에 한 버튼만 눌린다는 전제).
497 RestoreActivePressCompanions();
498 TArray<UWidget*> Companions;
499 CollectButtonCompanions(Button, Companions);
500 for (UWidget* Companion : Companions)
501 {
502 Companion->SetRenderTransformPivot(FVector2D(0.5f, 0.5f));
503 Companion->SetRenderScale(FVector2D(ButtonPressScale, ButtonPressScale));
504
505 FLinearColor CompanionBase = FLinearColor::White;
506 if (UImage* Image = Cast<UImage>(Companion))
507 {
508 CompanionBase = Image->GetColorAndOpacity();
509 FLinearColor Dark = CompanionBase;
510 Dark.R *= ButtonPressColorMul;
511 Dark.G *= ButtonPressColorMul;
512 Dark.B *= ButtonPressColorMul;
513 Image->SetColorAndOpacity(Dark);
514 }
515 else if (UTextBlock* Text = Cast<UTextBlock>(Companion))
516 {
517 CompanionBase = Text->GetColorAndOpacity().GetSpecifiedColor();
518 FLinearColor Dark = CompanionBase;
519 Dark.R *= ButtonPressColorMul;
520 Dark.G *= ButtonPressColorMul;
521 Dark.B *= ButtonPressColorMul;
522 Text->SetColorAndOpacity(FSlateColor(Dark));
523 }
524
525 mActivePressCompanions.Add(Companion);
526 mActivePressCompanionBaseColors.Add(CompanionBase);
527 }
528
529 break; // 한 번에 한 버튼만.
530 }
