203void URDUserWidget::SetupCommonButtonFeedback()
204{
205 mFeedbackButtons.Reset();
206 mFeedbackButtonBaseColors.Reset();
207
208 if (WidgetTree == nullptr)
209 {
210 return;
211 }
212
213 USoundBase* PressSound = mCommonButtonPressSound;
214 const bool bApplyPressVisual = ShouldApplyButtonFeedback();
215
216 // 이 위젯 트리 안의 모든 UButton을 순회한다(자식 UserWidget은 각자 이 베이스를 상속하므로 스스로 처리).
217 WidgetTree->ForEachWidget([this, PressSound, bApplyPressVisual](UWidget* Widget)
218 {
219 UButton* Button = Cast<UButton>(Widget);
220 if (Button == nullptr)
221 {
222 return;
223 }
224
225 // 클릭 사운드: 스타일의 PressedSlateSound에만 주입한다. 브러시/색은 그대로 둬 디자이너 스킨을 보존한다.
226 if (PressSound != nullptr)
227 {
228 FButtonStyle Style = Button->GetStyle();
229 Style.PressedSlateSound.SetResourceObject(PressSound);
230 Button->SetStyle(Style);
231 }
232
233 // 누름 시각 피드백은 opt-in 화면(타이틀/클래스 선택 등)만 — 전투/주사위엔 걸지 않는다.
234 if (bApplyPressVisual == false)
235 {
236 return;
237 }
238
239 // 누름 피드백: 배경색을 어둡게(주 효과) + 살짝 축소(보조). 브러시 자체는 안 건드려 디자이너 스킨 보존.
240 // 원래 배경색을 저장해 뒀다가 떼면 그대로 복원한다(버튼별 커스텀 틴트도 안전).
241 Button->SetRenderTransformPivot(FVector2D(0.5f, 0.5f));
242 Button->OnPressed.AddUniqueDynamic(this, &URDUserWidget::HandleAnyButtonPressed);
243 Button->OnReleased.AddUniqueDynamic(this, &URDUserWidget::HandleAnyButtonReleased);
244
245 mFeedbackButtons.Add(Button);
246 mFeedbackButtonBaseColors.Add(Button->GetBackgroundColor());
247 });

250void URDUserWidget::CollectButtonCompanions(const UButton* Button, TArray<UWidget*>& OutCompanions) const
251{
252 if (Button == nullptr || WidgetTree == nullptr)
253 {
254 return;
255 }
256
257 // 버튼명을 "<Base>__<Profile>"로 보고 Base/접미사를 분리한다(접미사 없으면 Base=전체, 접미사="").
258 const FString FullName = Button->GetName();
259 FString Base = FullName;
260 FString Suffix;
261 const int32 ProfileSep = FullName.Find(TEXT("__"), ESearchCase::CaseSensitive, ESearchDir::FromStart);
262 if (ProfileSep != INDEX_NONE)
263 {
264 Base = FullName.Left(ProfileSep); // 예: "StartButton"
265 Suffix = FullName.Mid(ProfileSep); // 예: "__base_16_9"
266 }
267
268 // 같은 트리에서 Base로 시작 + 같은 접미사로 끝나는 Image/Text 형제를 짝으로 모은다.
269 WidgetTree->ForEachWidget([&Base, &Suffix, Button, &OutCompanions](UWidget* Widget)
270 {
271 if (Widget == nullptr || Widget == Button)
272 {
273 return;
274 }
275 if (Widget->IsA<UImage>() == false && Widget->IsA<UTextBlock>() == false)
276 {
277 return;
278 }
279 const FString Name = Widget->GetName();
280 if (Name.StartsWith(Base) == true && Name.EndsWith(Suffix) == true)
281 {
282 OutCompanions.Add(Widget);
283 }
284 });
285}

287void URDUserWidget::HandleAnyButtonPressed()
288{
289 // OnPressed는 발신 버튼을 알려주지 않으므로, 실제로 눌린 버튼만 골라 처리한다(동시에 하나만 눌리는 게 보통).
290 for (int32 Index = 0; Index < mFeedbackButtons.Num(); ++Index)
291 {
292 UButton* Button = mFeedbackButtons[Index];
293 if (Button == nullptr || Button->IsPressed() == false)
294 {
295 continue;
296 }
297
298 // (1) 버튼 자체 — 버튼 브러시가 시각요소인 경우(클래스 선택 등)에 배경색을 어둡게 + 축소.
299 FLinearColor Pressed = mFeedbackButtonBaseColors.IsValidIndex(Index) ? mFeedbackButtonBaseColors[Index] : FLinearColor::White;
300 Pressed.R *= ButtonPressColorMul;
301 Pressed.G *= ButtonPressColorMul;
302 Pressed.B *= ButtonPressColorMul; // 알파는 유지, RGB만 어둡게.
303 Button->SetBackgroundColor(Pressed);
304 Button->SetRenderScale(FVector2D(ButtonPressScale, ButtonPressScale));
305
306 // (2) 동반 시각 위젯 — 투명 버튼 위에 얹힌 프레임/텍스트(타이틀 등). 이쪽을 어둡게+축소해야 실제로 보인다.
307 // 이전 잔여가 있으면 먼저 복원 후 새로 적용(한 번에 한 버튼만 눌린다는 전제).
308 RestoreActivePressCompanions();
309 TArray<UWidget*> Companions;
310 CollectButtonCompanions(Button, Companions);
311 for (UWidget* Companion : Companions)
312 {
313 Companion->SetRenderTransformPivot(FVector2D(0.5f, 0.5f));
314 Companion->SetRenderScale(FVector2D(ButtonPressScale, ButtonPressScale));
315
316 FLinearColor CompanionBase = FLinearColor::White;
317 if (UImage* Image = Cast<UImage>(Companion))
318 {
319 CompanionBase = Image->GetColorAndOpacity();
320 FLinearColor Dark = CompanionBase;
321 Dark.R *= ButtonPressColorMul;
322 Dark.G *= ButtonPressColorMul;
323 Dark.B *= ButtonPressColorMul;
324 Image->SetColorAndOpacity(Dark);
325 }
326 else if (UTextBlock* Text = Cast<UTextBlock>(Companion))
327 {
328 CompanionBase = Text->GetColorAndOpacity().GetSpecifiedColor();
329 FLinearColor Dark = CompanionBase;
330 Dark.R *= ButtonPressColorMul;
331 Dark.G *= ButtonPressColorMul;
332 Dark.B *= ButtonPressColorMul;
333 Text->SetColorAndOpacity(FSlateColor(Dark));
334 }
335
336 mActivePressCompanions.Add(Companion);
337 mActivePressCompanionBaseColors.Add(CompanionBase);
338 }
339
340 break; // 한 번에 한 버튼만.
341 }
342}
