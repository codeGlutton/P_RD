31bool UBoardMovementComponentModel::TeleportTo(const FTileTransform& NextTransform, FOnBoardMoveFinished OnFinished)
32{
33 // 경로 이동 중이거나 텔레포트 중이면 시작 안 함
34 if (IsMoving() == true)
35 {
36 return false;
37 }
38
39 UBoardActorModel* Owner = GetOwnerModel<UBoardActorModel>();
40 UTileMapModel* TileMap = GetTileMap();
41
42 // 도착 타일이 막혀 있으면 시작 안 함
43 if (TileMap->IsBlocked(NextTransform.mIndex, Owner) == true)
44 {
45 return false;
46 }
47
48 // 논리 좌표 변경 -> 출발 타일의 이탈 오버랩 통지
49 TileMap->StartActorMovement(NextTransform, Owner);
50 // 도착 타일 진입 오버랩 통지
51 TileMap->CompleteActorMovement(GetOwnerModel<UBoardActorModel>());
52
53 // 완료 통지
54 OnFinished.ExecuteIfBound();
55
56 // OnTeleport를 구독하고 있던 뷰가 사라짐/나타남 동기 연출 시작
57 Owner->OnTeleport.Broadcast(NextTransform, TileMap->TileToWorldTransform(NextTransform));
58 return true;
59}

61bool UBoardMovementComponentModel::TryRegisterPendingPush(const FTileIndex& TrapTileIndex, const TArray<FTileIndex>& PushPathTileIndexes)
62{
63 // 이동 루프 밖에서는 소비 지점이 없으므로 등록 불가 (정지 대상은 PushAlongPath 사용)
64 if (IsMoving() == false)
65 {
66 return false;
67 }
68
69 // 한 연쇄 안에서 같은 함정은 1회만 발동 (무한 연쇄 방지)
70 if (mChainedTrapTiles.Contains(TrapTileIndex) == true)
71 {
72 return false;
73 }
74
75 // 빈 경로만 거부. 밀 곳이 없어 제자리 1칸인 경로도 등록해서 잔여 걷기를 끊음 (막힘 판정은 경로 생성 쪽 책임)
76 if (PushPathTileIndexes.IsEmpty() == true)
77 {
78 return false;
79 }
80
81 // 보류 경로의 시작은 피격자의 현재 타일이어야 함
82 checkf(PushPathTileIndexes[0] == GetOwnerModel<UBoardActorModel>()->GetTileTransform().mIndex,
83 TEXT("보류 밀치기 경로의 시작 타일 불일치"));
84
85 // 한 타일에 함정이 여러개 있을 경우 마지막 등록이 덮어 씀
86 mChainedTrapTiles.Add(TrapTileIndex);
87 mPendingPushPath = PushPathTileIndexes;
88 return true;
89}

91bool UBoardMovementComponentModel::StartPathInternal(const TArray<FTileIndex>& PathTileIndexes, EBoardMoveMode MoveMode, FOnBoardMoveFinished OnFinished, TSharedPtr<FPresentationBarrier> MoveEndBarrier)
92{
93 // 이동 중 재호출이거나 경로가 2칸 미만이면 시작 거부
94 if (IsMoving() == true || PathTileIndexes.Num() < 2)
95 {
96 return false;
97 }
98
99 mPathTileIndexes = PathTileIndexes;
100 mMoveMode = MoveMode;
101 mOnFinished = OnFinished;
102 mMoveEndBarrier = MoveEndBarrier;
103 mCancelRequested = false;
104
105 // 첫 칸이 이미 막혔으면 시작 안 함 (경로 계산 뒤 다른 유닛이 먼저 차지한 경우 등)
106 if (CanEnterStep(1) == false)
107 {
108 ResetMoveState();
109 return false;
110 }
111
112 // 외부 요청 시작 = 새 연쇄 시작 (연쇄 기록/보류 경로/깊이 초기화)
113 mChainedTrapTiles.Empty();
114 mPendingPushPath.Empty();
115 mAdoptedPushCount = 0;
116
117 // 전체 경로를 뷰에 통지
118 BroadcastStartMovePath();
119
120 // 1번 타일로 이동하는 것부터 시작 (0번은 현재 타일).
121 // 해당 타일로 이동이 완료되면, 베리어가 끝나고 다시 다음 스텝으로 가는 걸 마지막 타일까지 반복.
122 // 시뮬레이션모드에서는 베리어가 없으므로 즉각 다음 스텝으로 진행하므로 문제 없음
123 StartStep(1);
124 return true;
125}

168void UBoardMovementComponentModel::StartStep(int32 StepIndex)
169{
170 checkf(mPathTileIndexes.IsValidIndex(StepIndex) == true, TEXT("이동 경로 인덱스 오류"));
171 mCurrentStepIndex = StepIndex;
172
173 // 파생 훅 (유닛의 AP 차감 등)
174 OnStartStep(StepIndex, mMoveMode);
175
176 UBoardActorModel* Owner = GetOwnerModel<UBoardActorModel>();
177 UTileMapModel* TileMap = GetTileMap();
178
179 // 직전 타일에서 이번 타일을 바라볼때의 방향 계산
180 // 직전->현재와 현재->다음 방향을 보간해서 자연스럽게 코너링 할 계획
181 // 밀치기는 밀려나는 것이므로 바라보는 방향 유지 (뒤로 밀려도 몸은 그대로)
182 const ETileActorDirection Direction = (mMoveMode == EBoardMoveMode::Normal)
183 ? UTileMapModel::TileDeltaToDirection(
184 mPathTileIndexes[StepIndex - 1],
185 mPathTileIndexes[StepIndex],
186 Owner->GetTileTransform().mDirection)
187 : Owner->GetTileTransform().mDirection;
188
189 // 다음 타일로 이동 (모델의 논리적 위치 변경)
190 // 점유는 즉시 하고, 도착 오버랩 통지는 CompleteStep가 함
191 const FTileTransform NextTransform(mPathTileIndexes[StepIndex], Direction);
192 TileMap->StartActorMovement(NextTransform, Owner);
193
194 // 이동 후 받을 베리어 생성
195 // 컴포넌트 모델이 먼저 파괴될 수 있으므로 WeakLambda로 보호.
196 TSharedPtr<FPresentationBarrier> Barrier = FPresentationBarrier::Make(
197 FOnFinishPresentation::CreateWeakLambda(this, [this]() {
198 OnStepPresentationFinished();
199 }));
200
201 // 이번 스텝 도착 후 최종 목적지까지 남은 경로 거리 계산
202 // -> 제동거리에 들어가면 감속할 때 사용
203 float RemainingPathDistance = 0.0f;
204 for (int32 i = StepIndex; i < mPathTileIndexes.Num() - 1; ++i)
205 {
206 RemainingPathDistance += FVector::Dist(
207 TileMap->TileToWorldLocation(mPathTileIndexes[i]),
208 TileMap->TileToWorldLocation(mPathTileIndexes[i + 1]));
209 }
210
211 // OnStartMoveStep을 구독하고 있던 뷰가 이동 시작 (뷰의 물리적 위치 변경)
212 Owner->OnStartMoveStep.Broadcast(NextTransform, TileMap->TileToWorldTransform(NextTransform), Barrier, RemainingPathDistance, mMoveMode);
213}

221void UBoardMovementComponentModel::OnStepPresentationFinished()
222{
223 // 취소 요청됐으면 완료 통지 없이 정지 (뷰가 걷기 연출을 멈추도록 경로 종료는 통지)
224 if (mCancelRequested == true)
225 {
226 // 이동 종료 배리어가 ResetMoveState 실행에도 살아남도록 로컬 변수에 저장
227 TSharedPtr<FPresentationBarrier> MoveEndBarrier = MoveTemp(mMoveEndBarrier);
228 ResetMoveState();
229 BroadcastEndMovePath();
230 return;
231 }
232
233 // 현재 타일 도착 처리 -> 함정/장판 등 오버랩 관련된 처리
234 CompleteStep();
235
236 // OnEndMoveStep을 구독하고 있던 뷰가 도착 확인 (UI 변경 등)
237 UBoardActorModel* Owner = GetOwnerModel<UBoardActorModel>();
238 Owner->OnEndMoveStep.Broadcast(Owner->GetTileTransform(), Owner->GetWorldTransform());
239
240 // 이 스텝 도중 함정이나 스킬에 밀쳐졌으면 남은 경로를 버리고 밀치기 경로로 갈아탐
241 // 완료 통지는 밀치기까지 다 끝난 뒤 한 번만 하므로 mOnFinished는 그대로 둠
242 if (mPendingPushPath.Num() > 0)
243 {
244 // 같은 함정은 한 번만 발동하므로 갈아탄 횟수가 상한을 넘으면 연쇄 기록 버그
245 ++mAdoptedPushCount;
246 checkf(mAdoptedPushCount <= MaxPushChainDepth, TEXT("밀치기 연쇄 깊이 상한 초과"));
247
248 mPathTileIndexes = MoveTemp(mPendingPushPath);
249 mPendingPushPath.Empty();
250
251 // 실제로 밀리는 경우에만 밀치기 모드로 전환하고 뷰에 새 경로 통지 (걷기 곡선은 버리고 밀치기 연출로 전환)
252 // 밀 곳이 없는 제자리 1칸 경로는 연출 없이 아래에서 바로 완료
253 if (mPathTileIndexes.Num() >= 2)
254 {
255 mMoveMode = EBoardMoveMode::Push;
256 BroadcastStartMovePath();
257 }
258
259 // 바뀐 새 경로를 시작하도록 인덱스 초기화
260 mCurrentStepIndex = 0;
261 }
262
263 // 스스로 하는 이동 중 상태이상에 걸렸으면 남은 경로를 버리고 여기서 종료
264 // 강제 이동은 상태이상과 무관하게 끝까지 진행
265 const bool IsStoppedByStatus = IsSelfMove(mMoveMode) == true && CanSelfMove() == false;
266
267 // 다음 칸이 남아 있고 들어갈 수 있으면 계속 이동
268 const int32 NextStepIndex = mCurrentStepIndex + 1;
269 if (IsStoppedByStatus == false && mPathTileIndexes.IsValidIndex(NextStepIndex) && CanEnterStep(NextStepIndex))
270 {
271 StartStep(NextStepIndex);
272 return;
273 }
274
275 // 마지막 칸 도착, 다음 칸 막힘, 상태이상 정지 중 하나면 여기서 이동 완료
276 // 통지를 받은 쪽이 바로 새 이동을 시작할 수 있으므로 상태를 먼저 비우고 호출 (비우면 mOnFinished도 지워지니 복사해 둠)
277 // 뷰의 경로 종료 정리가 완료 통지보다 먼저 끝나야 다음 이동 연출과 섞이지 않음
278 FOnBoardMoveFinished Finished = mOnFinished;
279 // 이동 종료 배리어가 ResetMoveState 실행에도 살아남도록 로컬 변수에 저장
280 TSharedPtr<FPresentationBarrier> MoveEndBarrier = MoveTemp(mMoveEndBarrier);
281 ResetMoveState();
282 BroadcastEndMovePath();
283 Finished.ExecuteIfBound();
284}
