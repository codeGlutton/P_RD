50bool FTacticalEffectQuery::Matches(const FTacticalEffectSpec& Spec) const
51{
52 if (Spec.mEffectClass == nullptr)
53 {
54 return false;
55 }
56
57 /* ASC 태그 검사 */
58
59 if (mOwningTagQuery.IsEmpty() == false)
60 {
61 check(IsInGameThread() == true);
62
63 static FGameplayTagContainer TargetTags;
64 TargetTags.Reset();
65
66 TargetTags.AppendTags(Spec.mEffectClass->GetAssetTags());
67 TargetTags.AppendTags(Spec.mEffectClass->GetGrantedTags());
68
69 if (mOwningTagQuery.Matches(TargetTags) == false)
70 {
71 return false;
72 }
73 }
74
75 /* 활성 Effect 태그 검사 */
76
77 if (mEffectTagQuery.IsEmpty() == false)
78 {
79 check(IsInGameThread());
80
81 static FGameplayTagContainer TETags;
82 TETags.Reset();
83
84 TETags.AppendTags(Spec.mEffectClass->GetAssetTags());
85
86 if (mEffectTagQuery.Matches(TETags) == false)
87 {
88 return false;
89 }
90 }
91
92 /* 수정 속성 검사 */
93
94 if (mModifyingAttribute.IsValid() == true)
95 {
96 bool IsEffectModifiesThisAttribute = false;
97
98 for (int32 ModIdx = 0; ModIdx < Spec.mModifiers.Num(); ++ModIdx)
99 {
100 const FTacticalModifierInfo& ModDef = Spec.mEffectClass->mModifiers[ModIdx];
101 if (ModDef.mAttribute == mModifyingAttribute)
102 {
103 IsEffectModifiesThisAttribute = true;
104 break;
105 }
106 }
107 if (IsEffectModifiesThisAttribute == false)
108 {
109 return false;
110 }
111 }
112
113 /* 소스 오브젝트 검사 */
114
115 if (mEffectSource != nullptr)
116 {
117 if (Spec.GetContext()->GetSourceObject() != mEffectSource)
118 {
119 return false;
120 }
121 }
122
123 /* 활성 이펙트 클래스 검사 */
124
125 if (mEffectDefinition != nullptr)
126 {
127 if (Spec.mEffectClass != mEffectDefinition.GetDefaultObject())
128 {
129 return false;
130 }
131 }
132
133 return true;
134}
