40void USRPGMoveAction::OnBeginAction()
41{
42 Super::OnBeginAction();
43
44 // 경로가 없거나 시작 타일만 있으면 이동 없이 즉시 종료
45 if (mPathTileIndexes.Num() < 2)
46 {
47 MarkActionCompleted(ESRPGActionResult::Succeeded);
48 return;
49 }
50
51 // 실제 이동은 컴포넌트 모델이 처리, 액션은 완료를 기다렸다가 턴 시스템에 보고
52 // 액션이 먼저 파괴될 수 있으므로 WeakLambda로 보호.
53 UBoardMovementComponentModel* MovementCompModel = mInstigator->GetBoardMovementComponentModel();
54 checkf(MovementCompModel != nullptr, TEXT("이동 컴포넌트 모델 nullptr"));
55
56 const bool Started = MovementCompModel->MoveAlongPath(
57 mPathTileIndexes,
58 FOnBoardMoveFinished::CreateWeakLambda(this, [this]() {
59 FSimulationOption Option;
60 Option.mDuration = ESimulationDurtaion::AllPlayerTurnEnd;
61 Option.mSkipAIActions = true;
62 Option.mReservedCommand.InitializeAs<FSRPGTurnEndCommand>();
63 GetCombatModel()->OnSimulateAllPlayerTurn.Broadcast(Option);
64
65 MarkActionCompleted(ESRPGActionResult::Succeeded);
66 }));
67 checkf(Started == true, TEXT("이동 시작 실패 (이미 이동 중이거나 경로 오류)"));
68}

70void USRPGMoveAction::OnEndAction()

75 UBoardMovementComponentModel* MovementCompModel = mInstigator.IsValid() ? mInstigator->GetBoardMovementComponentModel() : nullptr;
