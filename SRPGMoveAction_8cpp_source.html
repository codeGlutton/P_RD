24ESRPGCommandResult USRPGMoveAction::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
25{
26 ESRPGCommandResult Result = Super::HandleCommand(Command);
27 if (Result == ESRPGCommandResult::Handled)
28 {
29 return Result;
30 }
31
32 /* 생성 시 예약된 이동 명령에서 확정 경로를 수신 (빌드 액션이 실어 보낸 경로) */
33 if (Command.Get().GetCommandType() == ESRPGCommandType::MoveCast)

42void USRPGMoveAction::OnBeginAction()
43{
44 Super::OnBeginAction();
45
46 // 경로가 없거나 시작 타일만 있으면 이동 없이 즉시 종료
47 if (mPathTileIndexes.Num() < 2)
48 {
49 MarkActionCompleted(ESRPGActionResult::Succeeded);
50 return;
51 }
52
53 // 1번 타일로 이동하는 것부터 시작 (0번은 현재 타일).
54 // 해당 타일로 이동이 완료되면, 베리어가 끝나고 다시 다음 스텝으로 가는 걸 마지막 타일까지 반복.
55 // 시뮬레이션모드에서는 베리어가 없으므로 즉각 다음 스텝으로 진행하므로 문제 없음
56 StartStep(1);
57}

59void USRPGMoveAction::OnEndAction()
60{
61 Super::OnEndAction();
62
63 /* 이동을 정상 완료한 경우, 사용한 이동 포인트를 이동 유닛에게 차감 통지한다 */
64
65 if (mActionResult == ESRPGActionResult::Succeeded && mPathTileIndexes.Num() >= 2)
66 {
67 // 소모 이동 포인트 = 밟은 칸 수 (경로 칸 수 - 시작 타일)
68 const int32 SpentPoint = mPathTileIndexes.Num() - 1;
69
70 // [로컬/임시] 이동력 차감. 정본 API(UseMovePoint)가 아직 미구현 스텁이라, MOVE 표시/범위와 같은
71 // Movement 어트리뷰트를 직접 차감해 카운터가 줄게 한다. [주의] 정본 차감 API가 생기면 교체(Mo).
72 if (UAttributeSetComponentModel* AttrComp = mInstigator->GetAttributeComponentModel())
73 {
74 AttrComp->ApplyModToAttribute(UUnitAttributeSet::GetMovementAttribute(), ETacticalModOp::Additive, -static_cast<float>(SpentPoint));
75 }
76 }
77}

79void USRPGMoveAction::StartStep(int32 StepIndex)
80{
81 checkf(mPathTileIndexes.IsValidIndex(StepIndex) == true, TEXT("이동 경로 인덱스 오류"));
82 mCurrentStepIndex = StepIndex;
83
84 UTileMapModel* TileMap = GetTileMap();
85
86 // 직전 타일에서 이번 타일을 바라볼때의 방향 계산
87 // 직전->현재와 현재->다음 방향을 보간해서 자연스럽게 코너링 할 계획
88 const ETileActorDirection Direction = UTileMapModel::TileDeltaToDirection(
89 mPathTileIndexes[StepIndex - 1],
90 mPathTileIndexes[StepIndex],
91 mInstigator->GetTileTransform().mDirection);
92
93 // 다음 타일로 이동 (모델의 논리적 위치 변경)
94 // 점유는 즉시 하고, 도착 오버랩 통지는 CompleteStep가 함
95 const FTileTransform NextTransform(mPathTileIndexes[StepIndex], Direction);
96 TileMap->StartActorMovement(NextTransform, mInstigator.Get());
97
98 // 이동 후 받을 베리어 생성
99 // 액션이 먼저 파괴될 수 있으므로 WeakLambda로 보호.
100 TSharedPtr<FPresentationBarrier> Barrier = FPresentationBarrier::Make(
101 FOnFinishPresentation::CreateWeakLambda(this, [this]() {
102 OnStepPresentationFinished();
103 }));
104
105 // 이번 스텝 도착 후 최종 목적지까지 남은 경로 거리 계산
106 // -> 제동거리에 들어가면 감속할 때 사용
107 float RemainingPathDistance = 0.0f;
108 for (int32 i = StepIndex; i < mPathTileIndexes.Num() - 1; ++i)
109 {
110 RemainingPathDistance += FVector::Dist(
111 TileMap->TileToWorldLocation(mPathTileIndexes[i]),
112 TileMap->TileToWorldLocation(mPathTileIndexes[i + 1]));
113 }
114
115 // OnStartMoveStep을 구독하고 있던 뷰가 이동 시작 (뷰의 물리적 위치 변경)
116 mInstigator->OnStartMoveStep.Broadcast(NextTransform, TileMap->TileToWorldTransform(NextTransform), Barrier, RemainingPathDistance);
117}

125void USRPGMoveAction::OnStepPresentationFinished()
126{
127 // ActionPlay 상태가 아니면 중단됐다고 보고 바로 리턴
128 if (mActionPhase != ESRPGActionPhase::ActionPlay)
129 {
130 return;
131 }
132
133 // 현재 타일 도착 처리 -> 함정/장판 등 오버랩 관련된 처리
134 CompleteStep();
135
136 // 1) 마지막 타일이면 이동 완료.
137 if (mCurrentStepIndex >= mPathTileIndexes.Num() - 1)
138 {
139 MarkActionCompleted(ESRPGActionResult::Succeeded);
140 }
141 // 2) 마지막 타일이 아니면 다음 타일로 이동
142 else
143 {
144 StartStep(mCurrentStepIndex + 1);
145 }
146}
