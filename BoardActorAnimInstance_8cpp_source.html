43bool UBoardActorAnimInstance::PlayMontageUsingTag_Internal(FBoardActorAnimationContext Context)
44{
45 if (mTagAnimMontageSets.Contains(Context.mMontageTag) == false)
46 {
47 // 존재하지 않는 애님 몽타쥬
48 return false;
49 }
50
51 TObjectPtr<UAnimMontage>& TargetAnimMontage = mTagAnimMontageSets[Context.mMontageTag].mAnimMontages[StaticCast<int32>(Context.mMontageDir)];
52 if (TargetAnimMontage == nullptr)
53 {
54 // nullptr 애님 몽타쥬
55 return false;
56 }
57
58 if (IsPlayingMontageUsingTag() == true && GetPlayingMontageSetUsingTag()->mPlayPriority > mTagAnimMontageSets[Context.mMontageTag].mPlayPriority)
59 {
60 // 우선순위가 낮은 몽타쥬
61 return false;
62 }
63
64 if (Montage_Play(TargetAnimMontage) <= 0.f)
65 {
66 // 애님 몽타쥬 실행 시간이 0초 이하
67 return false;
68 }
69
70 /* 성공적으로 몽타쥬 실행 시 */
71
72 mActiveAnimationContext = MoveTemp(Context);
73
74 FOnMontageEnded EndDelegate;
75 EndDelegate.BindUObject(this, &UBoardActorAnimInstance::OnEndMontageUsingTag);
76 Montage_SetEndDelegate(EndDelegate, GetPlayingMontageUsingTag());
77
78 return true;

87bool UBoardActorAnimInstance::TriggerMontageTagEvent(const FGameplayTag& EventTag, const FEventTriggerPayloadBase* Payload)
88{
89 if (IsPlayingMontageUsingTag() == false)

91 // 태그 기반 몽타쥬 실행 중이 아님

93 }
94
95 const bool HasMontageEvent = mActiveAnimationContext.mMontageEvents.Contains(EventTag);
96 const bool HasAllMontageEvent = mAllMontageEvents.Contains(EventTag);
97 if (HasMontageEvent == false && HasAllMontageEvent == false)
98 {
99 // 존재하지 않는 이벤트
100 return false;
101 }
102
103 if (HasMontageEvent == true)
104 {
105 /* 현 몽타쥬 대상 이벤트 실행 */
106
107 FBoardActorAnimationEvent& TargetEvent = mActiveAnimationContext.mMontageEvents[EventTag];
108 if (TargetEvent.OnTriggerAnimationEvent.IsBound() == true)
109 {
110 TargetEvent.OnTriggerAnimationEvent.Broadcast(mActiveAnimationContext, GetPlayingMontageUsingTag(), Payload);
111 }
112
113 if (TargetEvent.mIsOneTimeEvent == true)
114 {
115 UnregisterTagEventOnMontage(EventTag);
116 }
117 }
118 if (HasAllMontageEvent == true)
119 {
120 /* 모든 몽타쥬 대상 이벤트 실행 */
121
122 if (mAllMontageEvents[EventTag].OnTriggerAnimationEvent.IsBound() == true)
123 {
124 mAllMontageEvents[EventTag].OnTriggerAnimationEvent.Broadcast(mActiveAnimationContext, GetPlayingMontageUsingTag(), Payload);
125 }
126 }

128 return true;
