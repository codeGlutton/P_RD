88 inline FBox2D Inner(const TCHAR* Name, const FVector2D Position,
89 const FVector2D Size, const bool bSliced = true, const int32 Which = 0)
90 {
91 for (const FPart& Part : Parts)
92 {
93 if (FCString::Strcmp(Part.Name, Name) != 0 || Part.Index != Which)
94 {
95 continue;
96 }
97 if (bSliced == false)
98 {
99 return FBox2D(
100 Position + FVector2D(Size.X * Part.Left, Size.Y * Part.Top),
101 Position + FVector2D(Size.X * Part.Right, Size.Y * Part.Bottom));
102 }
103 FVector2D Near(Part.Left * Part.SourceW, Part.Top * Part.SourceH);
104 FVector2D Far((1.f - Part.Right) * Part.SourceW,
105 (1.f - Part.Bottom) * Part.SourceH);
106 // 놓을 자리가 원본보다 좁으면 테두리끼리 겹친다. 엔진도 그때는
107 // 반씩으로 자르므로 여기서도 같은 비율로 줄인다.
108 if (Near.X + Far.X > Size.X * 0.8f)
109 {
110 const float Shrink = Size.X * 0.8f / (Near.X + Far.X);
111 Near.X *= Shrink;
112 Far.X *= Shrink;
113 }
114 if (Near.Y + Far.Y > Size.Y * 0.8f)
115 {
116 const float Shrink = Size.Y * 0.8f / (Near.Y + Far.Y);
117 Near.Y *= Shrink;
118 Far.Y *= Shrink;
119 }
120 return FBox2D(Position + Near, Position + Size - Far);
121 }
122 const FVector2D Guess = Size * 0.12f;
123 return FBox2D(Position + Guess, Position + Size - Guess);
124 }

152 inline FBox2D Cell(const TCHAR* Name, const FVector2D Position,
153 const FVector2D Size, const bool bSliced, const int32 Which,
154 const int32 Total, const bool bHorizontal = true)
155 {
156 if (Count(Name) >= Total)
157 {
158 return Inner(Name, Position, Size, bSliced, Which);
159 }
160 const FBox2D Whole = Inner(Name, Position, Size, bSliced, 0);
161 const FVector2D Span = Whole.GetSize();
162 if (bHorizontal)
163 {
164 const float Step = Span.X / Total;
165 return FBox2D(Whole.Min + FVector2D(Step * Which, 0.f),
166 Whole.Min + FVector2D(Step * (Which + 1), Span.Y));
167 }
168 const float Step = Span.Y / Total;
169 return FBox2D(Whole.Min + FVector2D(0.f, Step * Which),
170 Whole.Min + FVector2D(Span.X, Step * (Which + 1)));
171 }
