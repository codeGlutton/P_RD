90void UCharacterSelectWidget::ApplyButtonStyles() const
91{
92 /*
93 * 이 함수는 위젯을 생성하지 않고, WBP가 제공한 버튼 인스턴스의 스타일만 보정한다.
94 * 버튼의 배치/텍스트/상호작용 대상은 WBP_CharacterSelect가 계속 소유한다.
95 * 모바일에서는 hover/pressed 틴트가 눌린 상태처럼 남아 보일 수 있어 동일한 브러시를 사용한다.
96 */
97 struct FButtonTexture { UButton* Button; UTexture2D* Asset; };
98 const FButtonTexture Targets[] = {
99 { mConfirmButton, mConfirmButtonTexture },
100 { mBackToMainButton, mBackButtonTexture },
101 };
102
103 for (const FButtonTexture& Target : Targets)
104 {
105 if (Target.Button == nullptr)
106 {
107 continue;
108 }
109 UTexture2D* Texture = Target.Asset;
110 if (Texture == nullptr)
111 {
112 continue;
113 }
114
115 FSlateBrush NormalBrush;
116 NormalBrush.DrawAs = ESlateBrushDrawType::Image;
117 NormalBrush.ImageSize = FVector2D(static_cast<float>(Texture->GetSizeX()), static_cast<float>(Texture->GetSizeY()));
118 NormalBrush.SetResourceObject(Texture);
119
120 FButtonStyle Style = Target.Button->GetStyle();
121 Style.SetNormal(NormalBrush);
122 Style.SetHovered(NormalBrush);
123 Style.SetPressed(NormalBrush);
124 Style.SetDisabled(NormalBrush);
125 Style.SetNormalPadding(FMargin(0.f));
126 Style.SetPressedPadding(FMargin(0.f));
127 Target.Button->SetStyle(Style);
128 }
129}
