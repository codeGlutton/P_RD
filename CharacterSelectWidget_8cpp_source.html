175void UCharacterSelectWidget::RebuildCharacterCards()
176{
177 if (CharacterCardContainer == nullptr || CharacterCardWidgetClass == nullptr)
178 {
179 return;
180 }
181
182 while (mCharacterCardWidgets.Num() > mCharacterOptions.Num())
183 {
184 UCharacterCardWidget* RemovedCharacterCardWidget = mCharacterCardWidgets.Pop(EAllowShrinking::No);
185 if (RemovedCharacterCardWidget != nullptr)
186 {
187 RemovedCharacterCardWidget->OnCharacterCardClicked.RemoveDynamic(this, &UCharacterSelectWidget::HandleCharacterCardClicked);
188 CharacterCardContainer->RemoveChild(RemovedCharacterCardWidget);
189 }
190 }
191
192 while (mCharacterCardWidgets.Num() < mCharacterOptions.Num())
193 {
194 UCharacterCardWidget* NewCharacterCardWidget = CreateWidget<UCharacterCardWidget>(this, CharacterCardWidgetClass);
195 if (NewCharacterCardWidget == nullptr)
196 {
197 return;
198 }
199
200 NewCharacterCardWidget->OnCharacterCardClicked.AddUniqueDynamic(this, &UCharacterSelectWidget::HandleCharacterCardClicked);
201 UPanelSlot* NewCardSlot = CharacterCardContainer->AddChild(NewCharacterCardWidget);
202 if (UHorizontalBoxSlot* HorizontalCardSlot = Cast<UHorizontalBoxSlot>(NewCardSlot))
203 {
204 HorizontalCardSlot->SetPadding(FMargin(8.0f));
205 HorizontalCardSlot->SetSize(FSlateChildSize(ESlateSizeRule::Fill));
206 HorizontalCardSlot->SetHorizontalAlignment(HAlign_Center);
207 HorizontalCardSlot->SetVerticalAlignment(VAlign_Center);
208 }
209 mCharacterCardWidgets.Add(NewCharacterCardWidget);
210 }
211
212 for (int32 CharacterIndex = 0; CharacterIndex < mCharacterOptions.Num(); ++CharacterIndex)
213 {
214 UCharacterCardWidget* CharacterCardWidget = mCharacterCardWidgets[CharacterIndex];
215 if (CharacterCardWidget == nullptr)
216 {
217 continue;
218 }
219
220 CharacterCardWidget->SetCharacterOption(mCharacterOptions[CharacterIndex], CharacterIndex);
221 CharacterCardWidget->SetSelected(CharacterIndex == mSelectedCharacterIndex);
222 }
223}

332void UCharacterSelectWidget::SetPortraitImage(const TSoftObjectPtr<UTexture2D>& Portrait)
333{
334 if (SelectedCharacterPortraitImage == nullptr)
335 {
336 CancelPortraitLoad();
337 return;
338 }
339
340 if (Portrait.IsNull() == true)
341 {
342 CancelPortraitLoad();
343 ClearPortraitImage();
344 return;
345 }
346
347 const FSoftObjectPath PortraitPath = Portrait.ToSoftObjectPath();
348 if (Portrait.IsValid() == true)
349 {
350 CancelPortraitLoad();
351 SelectedCharacterPortraitImage->SetBrushFromTexture(Portrait.Get());
352 SelectedCharacterPortraitImage->SetVisibility(ESlateVisibility::Visible);
353 return;
354 }
355
356 if (mPortraitLoadHandle.IsValid() == true && mPendingPortraitPath == PortraitPath)
357 {
358 ClearPortraitImage();
359 return;
360 }
361
362 CancelPortraitLoad();
363 ClearPortraitImage();
364
365 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
366 if (AssetManager == nullptr)
367 {
368 UE_LOG(LogRD, Warning, TEXT("CharacterSelectWidget: AssetManager is not ready."));
369 return;
370 }
371
372 mPendingPortraitPath = PortraitPath;
373 mPortraitLoadHandle = AssetManager->GetStreamableManager().RequestAsyncLoad(
374 PortraitPath,
375 FStreamableDelegate::CreateUObject(this, &UCharacterSelectWidget::HandlePortraitLoaded, PortraitPath),
376 FStreamableManager::DefaultAsyncLoadPriority,
377 false,
378 false,
379 TEXT("CharacterSelectPortrait")
380 );
381}

383void UCharacterSelectWidget::HandlePortraitLoaded(FSoftObjectPath PortraitPath)
384{
385 if (mPendingPortraitPath != PortraitPath)
386 {
387 return;
388 }
389
390 UTexture2D* PortraitTexture = Cast<UTexture2D>(PortraitPath.ResolveObject());
391 if (SelectedCharacterPortraitImage != nullptr && PortraitTexture != nullptr)
392 {
393 SelectedCharacterPortraitImage->SetBrushFromTexture(PortraitTexture);
394 SelectedCharacterPortraitImage->SetVisibility(ESlateVisibility::Visible);
395 }
396 else
397 {
398 ClearPortraitImage();
399 }
400
401 mPortraitLoadHandle.Reset();
402 mPendingPortraitPath.Reset();
