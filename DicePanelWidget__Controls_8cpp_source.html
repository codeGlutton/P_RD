18 UTextBlock* FindOrCreateButtonText(UWidgetTree* WidgetTree, UButton* Button, FName Name, const FText& Text)
19 {
20 if (WidgetTree == nullptr || Button == nullptr)
21 {
22 return nullptr;
23 }
24
25 UTextBlock* ButtonText = Cast<UTextBlock>(WidgetTree->FindWidget(Name));
26 if (ButtonText == nullptr)
27 {
28 ButtonText = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass(), Name);
29 }
30 if (ButtonText != nullptr)
31 {
32 ButtonText->SetJustification(ETextJustify::Center);
33 ButtonText->SetColorAndOpacity(FSlateColor(RDPanelNavigationStyle::GetButtonTextColor()));
34 ButtonText->SetText(Text);
35 if (ButtonText->GetParent() == nullptr)
36 {
37 Button->AddChild(ButtonText);
38 }
39 }
40 return ButtonText;

163void UDicePanelWidget::RefreshDiceCarouselControlWidgets()
164{
165 EnsureDiceCarouselControlWidgets();
166
167 const bool bShowCarouselControls = mDicePanelCarouselActivated && mDicePanelViews.IsValidIndex(mSelectedDicePanelIndex);
168 const ESlateVisibility ControlVisibility = bShowCarouselControls ? ESlateVisibility::Visible : ESlateVisibility::Collapsed;
169 if (mDiceCarouselTitleText != nullptr)
170 {
171 mDiceCarouselTitleText->SetVisibility(ControlVisibility);
172 }
173 if (mDiceCarouselFaceText != nullptr)
174 {
175 mDiceCarouselFaceText->SetVisibility(ControlVisibility);
176 }
177 const int32 FaceButtonCount = mDicePanelViews.IsValidIndex(mSelectedDicePanelIndex)
178 ? FMath::Clamp(mDicePanelViews[mSelectedDicePanelIndex].mFaceCount, 0, DicePanelMaxFaceButtonCount)
179 : 0;
180 for (int32 FaceIndex = 0; FaceIndex < mDiceCarouselFaceButtons.Num(); ++FaceIndex)
181 {
182 UIndexedButtonWidget* FaceButton = mDiceCarouselFaceButtons[FaceIndex];
183 if (FaceButton != nullptr)
184 {
185 FaceButton->SetVisibility(bShowCarouselControls && FaceIndex < FaceButtonCount ? ESlateVisibility::Visible : ESlateVisibility::Collapsed);
186 }
187 }
188 if (mDiceCarouselPreviousButton != nullptr)
189 {
190 mDiceCarouselPreviousButton->SetVisibility(bShowCarouselControls && mDicePanelViews.Num() > 1 ? ESlateVisibility::Visible : ESlateVisibility::Collapsed);
191 }
192 if (mDiceCarouselNextButton != nullptr)
193 {
194 mDiceCarouselNextButton->SetVisibility(bShowCarouselControls && mDicePanelViews.Num() > 1 ? ESlateVisibility::Visible : ESlateVisibility::Collapsed);
195 }
196 if (mDiceCarouselBackButton != nullptr)
197 {
198 mDiceCarouselBackButton->SetVisibility(ControlVisibility);
199 }
200
201 if (bShowCarouselControls == false)
202 {
203 return;
204 }
205
206 const FDiceViewData& DiceView = mDicePanelViews[mSelectedDicePanelIndex];
207 if (mDiceCarouselTitleText != nullptr)
208 {
209 mDiceCarouselTitleText->SetText(FText::Format(
210 LOCTEXT("DICE {0} / {1}", "DICE {0} / {1}"),
211 FText::AsNumber(mSelectedDicePanelIndex + 1),
212 FText::AsNumber(mDicePanelViews.Num())
213 ));
214 }
215 if (mDiceCarouselFaceText != nullptr)
216 {
217 mDiceCarouselFaceText->SetText(FText::Format(
218 LOCTEXT("RESULT {0} / d{1}", "RESULT {0} / d{1}"),
219 BuildDiceFaceText(DiceView),
220 FText::AsNumber(DiceView.mFaceCount)
221 ));
222 }
223
224 for (int32 FaceIndex = 0; FaceIndex < FaceButtonCount; ++FaceIndex)
225 {
226 if (mDiceCarouselFaceButtonTexts.IsValidIndex(FaceIndex) && mDiceCarouselFaceButtonTexts[FaceIndex] != nullptr)
227 {
228 const int32 FaceDisplayValue = DiceView.mFaceValues.IsValidIndex(FaceIndex) ? DiceView.mFaceValues[FaceIndex] : FaceIndex + 1;
229 mDiceCarouselFaceButtonTexts[FaceIndex]->SetText(FText::AsNumber(FaceDisplayValue));
230 }
231 }
232 RefreshDiceCarouselFaceButtonStyles();

235void UDicePanelWidget::RefreshDiceCarouselFaceButtonStyles()
236{
237 const int32 SelectedFaceValue = mDicePanelSelectedFaceValues.IsValidIndex(mSelectedDicePanelIndex)
238 ? mDicePanelSelectedFaceValues[mSelectedDicePanelIndex]
239 : INDEX_NONE;
240 const int32 FaceButtonCount = mDicePanelViews.IsValidIndex(mSelectedDicePanelIndex)
241 ? FMath::Clamp(mDicePanelViews[mSelectedDicePanelIndex].mFaceCount, 0, DicePanelMaxFaceButtonCount)
242 : 0;
243
244 for (int32 FaceIndex = 0; FaceIndex < mDiceCarouselFaceButtons.Num(); ++FaceIndex)
245 {
246 UIndexedButtonWidget* FaceButton = mDiceCarouselFaceButtons[FaceIndex];
247 if (FaceButton == nullptr)
248 {
249 continue;
250 }
251
252 if (FaceIndex >= FaceButtonCount)
253 {
254 FaceButton->SetVisibility(ESlateVisibility::Collapsed);
255 continue;
256 }
257
258 const bool bIsSelectedFace = SelectedFaceValue == FaceIndex + 1;
259 FaceButton->SetBackgroundColor(bIsSelectedFace
260 ? RDPanelNavigationStyle::GetSelectedButtonColor()
261 : RDPanelNavigationStyle::GetOptionButtonColor());
262
263 if (mDiceCarouselFaceButtonTexts.IsValidIndex(FaceIndex) && mDiceCarouselFaceButtonTexts[FaceIndex] != nullptr)
264 {
265 mDiceCarouselFaceButtonTexts[FaceIndex]->SetColorAndOpacity(FSlateColor(bIsSelectedFace
266 ? RDPanelNavigationStyle::GetSelectedButtonTextColor()
267 : RDPanelNavigationStyle::GetButtonTextColor()));
268 }
269 }
