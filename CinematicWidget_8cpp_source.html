74 virtual int32 OnPaint(const FPaintArgs& Args, const FGeometry& AllottedGeometry, const FSlateRect& MyCullingRect,
75 FSlateWindowElementList& OutDrawElements, int32 LayerId, const FWidgetStyle& InWidgetStyle, bool bParentEnabled) const override
76 {
77 // 그릴 브러시가 없거나 그리지 않는 타입이면 아무것도 안 하고 현재 레이어 반환
78 if (Image == nullptr || Image->DrawAs == ESlateBrushDrawType::NoDrawType)
79 {
80 return LayerId;
81 }
82
83 const FVector2D ScreenSize = AllottedGeometry.GetLocalSize(); // 위젯이 차지하는 실제 화면 크기
84 if (ScreenSize.X <= 0.0f || ScreenSize.Y <= 0.0f)
85 {
86 return LayerId; // 레이아웃 미확정(크기 0) 프레임은 스킵
87 }
88
89 // 비율 계산 기준 = 영상 원본 해상도. 미확보 시 브러시 ImageSize → 최종적으로 1x1로 단계 폴백
90 FVector2D VideoSize = NativeSize.Get();
91 if (VideoSize.X <= 0.0f || VideoSize.Y <= 0.0f)
92 {
93 VideoSize = Image->ImageSize;
94 }
95 if (VideoSize.X <= 0.0f || VideoSize.Y <= 0.0f)
96 {
97 VideoSize = FVector2D(1.0f, 1.0f); // 0 나눗셈 방지용 안전 폴백
98 }
99
100 // cover 스케일: 두 축 비율 중 큰 값을 택해 화면을 빈틈없이 덮음(작은 값을 쓰면 contain=레터박스가 됨)
101 const float CoverScale = FMath::Max(ScreenSize.X / VideoSize.X, ScreenSize.Y / VideoSize.Y);
102 const FVector2D DrawSize = VideoSize * CoverScale; // 실제로 그릴 영상 크기(화면보다 한 축 큼)
103 const FVector2D DrawOffset = (ScreenSize - DrawSize) * 0.5f; // 넘치는 양을 절반씩 나눠 중앙 정렬(상하/좌우 균등 크롭)
104 const ESlateDrawEffect DrawEffects = ShouldBeEnabled(bParentEnabled) ? ESlateDrawEffect::None : ESlateDrawEffect::DisabledEffect;
105
106 OutDrawElements.PushClip(FSlateClippingZone(AllottedGeometry)); // 위젯 경계 밖으로 넘친 영상을 잘라냄
107 FSlateDrawElement::MakeBox(
108 OutDrawElements,
109 LayerId,
110 AllottedGeometry.ToPaintGeometry(DrawSize, FSlateLayoutTransform(DrawOffset)), // 계산한 크기+오프셋으로 페인트 지오메트리 구성
111 Image,
112 DrawEffects,
113 InWidgetStyle.GetColorAndOpacityTint() * Image->GetTint(InWidgetStyle) // 위젯 틴트와 브러시 틴트를 곱해 최종 색 적용
114 );
115 OutDrawElements.PopClip();
116
117 return LayerId;
118 }
119
120 private:
121 const FSlateBrush* Image = nullptr; // 그릴 영상 브러시(소유하지 않는 약참조 포인터)
122 TAttribute<FVector2D> NativeSize; // 영상 원본 해상도 핸들(매 페인트 시 최신 값을 lazy 조회)
123 };

161TSharedRef<SWidget> UCinematicWidget::RebuildWidget()
162{
163 EnsureCinematicMediaObjects(); // 미디어 플레이어/텍스처 등 재생에 필요한 오브젝트를 선확보
164
165 // 영상 출력용 브러시를 매 Rebuild마다 초기화하고 이미지 드로우로 설정
166 mCinematicVideoBrush = FSlateBrush();
167 mCinematicVideoBrush.DrawAs = ESlateBrushDrawType::Image;
168 mCinematicVideoNativeSize = ReadCinematicVideoFileDimensions(ResolveCinematicVideoPath()); // 파일 헤더에서 원본 해상도 파싱(비율 계산 기준)
169 // 원본 해상도를 못 읽으면 640x640 정사각으로 폴백(브러시 기본 크기)
170 mCinematicVideoBrush.ImageSize = mCinematicVideoNativeSize.X > 0.0f && mCinematicVideoNativeSize.Y > 0.0f
171 ? mCinematicVideoNativeSize
172 : FVector2D(640.0f, 640.0f);
173 mCinematicVideoBrush.SetResourceObject(mCinematicMediaTexture); // 루프 재생되는 미디어 텍스처를 브러시 리소스로 연결
174
175 return SNew(SOverlay)
176 // [레이어 1] 맨 아래 검은 배경 — 영상이 못 채우는 영역/로딩 전 화면을 검게 덮음
177 + SOverlay::Slot()
178 .HAlign(HAlign_Fill)
179 .VAlign(VAlign_Fill)
180 [
181 SNew(SBorder)
182 .BorderImage(FCoreStyle::Get().GetBrush(TEXT("WhiteBrush")))
183 .BorderBackgroundColor(FLinearColor::Black)
184 ]
185 // [레이어 2] 가운데 cover 영상 이미지
186 + SOverlay::Slot()
187 .HAlign(HAlign_Fill)
188 .VAlign(VAlign_Fill)
189 [
190 // 화면 전체에 직접 그리되 원본 비율을 유지한 cover rect를 계산한다.
191 // SImage는 할당 영역에 리소스를 그대로 늘려 그릴 수 있어 Android에서 정사각 영상이 4:3으로 찌그러졌다.
192 SAssignNew(mCinematicVideoImage, SCinematicCoverImage)
193 .Image(&mCinematicVideoBrush)
194 .NativeSize(TAttribute<FVector2D>::CreateLambda([this]()
195 {
196 return mCinematicVideoNativeSize;
197 }))
198 .Clipping(EWidgetClipping::ClipToBounds)
199 ]
200 // [레이어 3] 맨 위 로딩 대기 레이어 — 페이드 아웃 후 다음 씬 로딩을 가리는 검은 막. 평소엔 Collapsed
201 + SOverlay::Slot()
202 .HAlign(HAlign_Fill)
203 .VAlign(VAlign_Fill)
204 [
205 SAssignNew(mLoadingWaitLayer, SBorder)
206 .Visibility(EVisibility::Collapsed) // 초기엔 숨김(FadeToLoadingWaitScreen 시 표시)
207 .BorderImage(FCoreStyle::Get().GetBrush(TEXT("WhiteBrush")))

209 [
210 SNew(SVerticalBox)
211 // 상단 여백을 채워 텍스트를 화면 하단 쪽으로 밀어냄
212 + SVerticalBox::Slot()
213 .FillHeight(1.0f)
214 [
215 SNullWidget::NullWidget
216 ]
217 // 하단 중앙 "로딩중" 텍스트(아래 96px 패딩으로 바닥에서 살짝 띄움)
218 + SVerticalBox::Slot()
219 .AutoHeight()

252void UCinematicWidget::PlayCinematic(FOnEndCinematicAnimation Callback)
253{
254 if (IsOpened() == false)
255 {
256 return; // 위젯이 열려 있지 않으면 재생하지 않음
257 }
258
259 OnEndCinematicAnimation = MoveTemp(Callback); // 종료 콜백 등록(불필요 복사 방지 위해 이동)
260 mCinematicFinished = false; // 종료 플래그 리셋
261 ClearDefaultCinematicTimer();
262 ClearCinematicFadeTimer();
263 HideLoadingWaitScreen(); // 이전 재생에서 남은 로딩막 제거
264 PlayCinematicAnimation();
265}

271void UCinematicWidget::FinishCinematic()
272{
273 if (mCinematicFinished)
274 {
275 return; // 이미 종료됐으면 재진입 차단(콜백 중복 실행 방지)
276 }
277
278 mCinematicFinished = true;
279 ClearDefaultCinematicTimer();
280
281 if (OnEndCinematicAnimation.IsBound())
282 {
283 OnEndCinematicAnimation.Execute(this);
284 OnEndCinematicAnimation.Unbind(); // 1회성 콜백이므로 실행 후 즉시 해제
285 }
286}
