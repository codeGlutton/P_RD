162void UCameraMovementComponent::ZoomCamera_InstantAndMoveToViewportPosition_Instant(float ZoomDelta, FVector2D ViewPortPos)
163{
164 if (mCamerControlState != ECameraControlState::Normal)
165 return;
166
167 checkf(mCameraComponent.IsValid(), TEXT("카메라가 유효하지 않습니다"));
168
169 // 확대 전 ViewPortPos 구하기
170 // =========================================
171 // ViewPortPos를 World 좌표로 변환합니다.
172 FVector WorldLocation;
173 FVector WorldDirection;
174
175 GetWorld()->GetFirstPlayerController()->DeprojectScreenPositionToWorld(ViewPortPos.X, ViewPortPos.Y, WorldLocation, WorldDirection);
176
177 FHitResult ViewPortHitResult;
178 FVector StartTrace = WorldLocation; // 시작 지점
179 FVector EndTrace = StartTrace + (WorldDirection * 100000.0f);
180
181 // CameraMove Channel을 가진 오브젝트와 충돌 체크를 진행합니다.
182 FCollisionObjectQueryParams ObjectParams;
183 ObjectParams.AddObjectTypesToQuery(ECC_GameTraceChannel3);
184
185 FCollisionQueryParams QueryParams;
186 QueryParams.AddIgnoredActor(GetOwner()); // 자기 자신은 충돌에서 제외
187
188 bool bViewPortHit = GetWorld()->LineTraceSingleByObjectType(
189 ViewPortHitResult,
190 StartTrace,
191 EndTrace,
192 ObjectParams,
193 QueryParams
194 );
195
196 // 충돌되지 않았다면 취소
197 if (!bViewPortHit)
198 return;
199
200 FVector PrePos = ViewPortHitResult.ImpactPoint;
201 PrePos.Z = 0;
202
203 // 카메라를 줌 한다.
204 ZoomCamera_Instant(ZoomDelta);
205
206 // 카메라를 갱신하여 ProjMat을 갱신해줍니다.
207 GetWorld()->GetFirstPlayerController()->PlayerCameraManager->UpdateCamera(0.0f);
208
209 // 확대 후 ViewPortPos 구하기
210 // =========================================
211 // ViewPortPos를 World 좌표로 변환합니다.
212 GetWorld()->GetFirstPlayerController()->DeprojectScreenPositionToWorld(ViewPortPos.X, ViewPortPos.Y, WorldLocation, WorldDirection);
213
214 StartTrace = WorldLocation; // 시작 지점
215 EndTrace = StartTrace + (WorldDirection * 100000.0f);
216
217 bViewPortHit = GetWorld()->LineTraceSingleByObjectType(
218 ViewPortHitResult,
219 StartTrace,
220 EndTrace,
221 ObjectParams,
222 QueryParams
223 );
224
225 if (!bViewPortHit)
226 return;
227
228 FVector NextPos = ViewPortHitResult.ImpactPoint;
229 NextPos.Z = 0;
230
231 // 위치의 차를 이용하여 카메라의 위치를 옮깁니다.
232 GetOwner()->AddActorWorldOffset(PrePos - NextPos);
233}

276void UCameraMovementComponent::MoveToViewportPosition_Smooth(FVector2D ViewPortPos)
277{
278 if (mCamerControlState != ECameraControlState::Normal)
279 return;
280
281 checkf(mCameraComponent.IsValid(), TEXT("카메라가 유효하지 않습니다"));
282
283 //==============================
284 // ViewPortPos를 WorldLocation으로 변환하고 Lay를 쏩니다.
285 FVector WorldLocation;
286 FVector WorldDirection;
287
288 GetWorld()->GetFirstPlayerController()->DeprojectScreenPositionToWorld(ViewPortPos.X, ViewPortPos.Y, WorldLocation, WorldDirection);
289
290 FHitResult ViewPortHitResult;
291 FVector StartTrace = WorldLocation; // 시작 지점
292 FVector EndTrace = StartTrace + (WorldDirection * 100000.0f);
293
294 // CameraMove Channel을 가진 오브젝트와 충돌 체크를 진행합니다.
295 FCollisionObjectQueryParams ObjectParams;
296 ObjectParams.AddObjectTypesToQuery(ECC_GameTraceChannel3);
297
298 FCollisionQueryParams QueryParams;
299 QueryParams.AddIgnoredActor(GetOwner()); // 자기 자신은 충돌에서 제외
300
301 bool bViewPortHit = GetWorld()->LineTraceSingleByObjectType(
302 ViewPortHitResult,
303 StartTrace,
304 EndTrace,
305 ObjectParams, // 충돌 채널
306 QueryParams
307 );
308
309 if (!bViewPortHit)
310 return;
311
312 // ============================================
313 // 카메라의 시선으로로 Ray를 쏜다.
314 FHitResult CameraCenterRayCastHitResult;
315 bool bCameraHit = GetCameraRayHitPoint(CameraCenterRayCastHitResult);
316
317 // 카메라의 중심에서 Ray가 적중하지 않았다면 이동시키지 않는다.
318 if (!bCameraHit)
319 return;
320
321 // ==========================================
322 // 카메라의 위치를 이동시킨다.
323 // Z 축은 반영하지 않으므로 0으로 만든다.
324 FVector LocationPos = ViewPortHitResult.ImpactPoint;
325 FVector CameraPos = CameraCenterRayCastHitResult.ImpactPoint; // 카메라가 바라보고 있는 땅의 위치를 가져온다.
326 LocationPos.Z = 0;
327 CameraPos.Z = 0;
328
329 // 0.01초 간격으로 호출할 예정이므로 (Duration / 0.01)번 반복
330 mStartLocation = CameraPos;
331 mCurLocation = CameraPos;
332 mEndLocation = LocationPos;
333
334 // 0.01초 간격으로 호출할 예정이므로 (Duration / 0.01)번 반복
335 mCurrentMoveAlpha = 0.0f;
336
337 // Timer로 이동을 수행합니다.
338 GetWorld()->GetTimerManager().SetTimer(mTimerHandle_Move, this, &UCameraMovementComponent::MoveSmooth, 0.01f, true);
339}

574void UCameraMovementComponent::MoveSmooth()
575{
576 checkf(mCameraComponent.IsValid(), TEXT("카메라가 유효하지 않습니다"));
577
578 // mMoveDuration 시간동안 이동 합니다.
579 mCurrentMoveAlpha += 0.01f / mMoveDuration; // 시간에 따른 진행도 증가
580
581 if (mCurrentMoveAlpha >= 1.0f)
582 {
583 mCurrentMoveAlpha = 1.0f;
584 GetWorld()->GetTimerManager().ClearTimer(mTimerHandle_Move); // 타이머 종료
585 }
586
587 // 선형 보간 적용
