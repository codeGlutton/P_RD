17 void FitClassActionImageSlotToViewportWidthCrop(const UCharacterSelectWidget* Owner, UImage* Image, const UTexture2D* Illustration)
18 {
19 if (Owner == nullptr || Image == nullptr || Illustration == nullptr)
20 {
21 return;
22 }
23
24 UCanvasPanelSlot* CanvasSlot = Cast<UCanvasPanelSlot>(Image->Slot);
25 if (CanvasSlot == nullptr)
26 {
27 return;
28 }
29
30 const UWorld* World = Owner->GetWorld();
31 const UGameViewportClient* GameViewport = World != nullptr ? World->GetGameViewport() : nullptr;
32 FVector2D ViewportSize = FVector2D::ZeroVector;
33 if (GameViewport != nullptr)
34 {
35 GameViewport->GetViewportSize(OUT ViewportSize);
36 }
37 if (ViewportSize.X <= 0.0f || ViewportSize.Y <= 0.0f)
38 {
39 return;
40 }
41
42 const float ViewportScale = FMath::Max(UWidgetLayoutLibrary::GetViewportScale(Owner), KINDA_SMALL_NUMBER);
43 const FVector2D LogicalViewportSize = ViewportSize / ViewportScale;
44 const float IllustrationAspect = static_cast<float>(Illustration->GetSizeX()) / FMath::Max(1.0f, static_cast<float>(Illustration->GetSizeY()));
45 if (IllustrationAspect <= 0.0f)
46 {
47 return;
48 }
49
50 const FVector2D TargetSize(LogicalViewportSize.X, LogicalViewportSize.X / IllustrationAspect);
51
52 CanvasSlot->SetAnchors(FAnchors(0.5f, 0.5f));
53 CanvasSlot->SetAlignment(FVector2D(0.5f, 0.5f));
54 CanvasSlot->SetPosition(FVector2D::ZeroVector);
55 CanvasSlot->SetSize(TargetSize);
56 }
57}

60void UCharacterSelectWidget::RefreshCharacterOptions()
61{
62 const int32 PreviousSelectedCharacterIndex = mSelectedCharacterIndex;
63 mCharacterOptions.Reset();
64 mSelectedPlayerUnitId = FPrimaryAssetId();
65 mSelectedCharacterIndex = INDEX_NONE;
66
67 AFrontendGameMode* FrontendGameMode = GetFrontendGameMode();
68 if (FrontendGameMode == nullptr || !FrontendGameMode->GetCharacterOptions(OUT mCharacterOptions))
69 {
70 RebuildCharacterCards();
71 ClearSelectedCharacter();
72 SetStatusText(mNoCharacterStatusText);
73 SetConfirmButtonText(mConfirmText);
74 return;
75 }
76
77 const FFrontendCharacterOption* PreservedOption = mCharacterOptions.FindByPredicate([PreviousSelectedCharacterIndex](const FFrontendCharacterOption& Option)
78 {
79 return Option.mIndex == PreviousSelectedCharacterIndex;
80 });
81 const FFrontendCharacterOption* FirstOption = mCharacterOptions.IsEmpty() ? nullptr : &mCharacterOptions[0];
82 const FFrontendCharacterOption* FirstEnabledOption = mCharacterOptions.FindByPredicate([](const FFrontendCharacterOption& Option)
83 {
84 return Option.mSelectable;
85 });
86 // 새로 받은 후보 목록에서도 이전 선택을 유지한다. 없으면 플레이 가능한 첫 후보, 그것도 없으면 첫 항목을 보여준다.
87 const FFrontendCharacterOption* SelectedOption = PreservedOption != nullptr
88 ? PreservedOption
89 : (FirstEnabledOption != nullptr ? FirstEnabledOption : FirstOption);
90
91 if (SelectedOption != nullptr)
92 {
93 mSelectedCharacterIndex = SelectedOption->mIndex;
94 mSelectedPlayerUnitId = SelectedOption->mSelectable ? SelectedOption->mPlayerUnitId : FPrimaryAssetId();
95 }
96
97 SetConfirmButtonText(mConfirmText);
98 RebuildCharacterCards();
99 SyncSelectedCharacter();
100}

240void UCharacterSelectWidget::SyncSelectedCharacterArt(EPlayerJobType JobType)
241{
242 /*
243 * 직업별 액션 이미지는 WBP에 이미 배치돼 있지만 브러시가 비어 있다.
244 * 아트 원본은 팀 정책상 git이 아니라 SVN(Content/SVN)에만 있으므로, 타이틀 로고와 동일하게
245 * 런타임에 PNG를 로드해 각 이미지에 채우고(캐시), 선택된 직업의 이미지만 표시한다.
246 */
247 const auto ApplyJobImage = [this](UImage* Image, EPlayerJobType ImageJob, EPlayerJobType SelectedJob)
248 {
249 if (Image == nullptr)
250 {
251 return;
252 }
253
254 if (ImageJob != SelectedJob)
255 {
256 Image->SetVisibility(ESlateVisibility::Collapsed);
257 return;
258 }
259
260 if (UTexture2D* Illustration = GetJobIllustration(ImageJob))
261 {
262 FitClassActionImageSlotToViewportWidthCrop(this, Image, Illustration);
263
264 FSlateBrush Brush;
265 Brush.DrawAs = ESlateBrushDrawType::Image;
266 Brush.ImageSize = FVector2D(
267 static_cast<float>(Illustration->GetSizeX()),
268 static_cast<float>(Illustration->GetSizeY()));
269 Brush.SetResourceObject(Illustration);
270 Image->SetBrush(Brush);
271 Image->SetVisibility(ESlateVisibility::HitTestInvisible);
272 }
273 else
274 {
275 Image->SetVisibility(ESlateVisibility::Collapsed);
276 }
277 };
278
279 ApplyJobImage(mKnightActionImage, EPlayerJobType::Knight, JobType);
280 // [합의필요] 현재 WBP/아트 파일명은 Rogue지만 게임 enum은 Archer라 매핑을 코드에서 고정한다.
281 ApplyJobImage(mRogueActionImage, EPlayerJobType::Archer, JobType);
282 ApplyJobImage(mMageActionImage, EPlayerJobType::Mage, JobType);
283
284 const bool bHasClassActionImages = mKnightActionImage != nullptr || mRogueActionImage != nullptr || mMageActionImage != nullptr;
285 if (mSelectedCharacterPortraitImage == nullptr)
286 {
287 return;
288 }
289
290 // 직업별 액션 이미지를 쓰는 WBP에서는 공용 포트레이트 슬롯을 숨긴다. 구형 WBP fallback일 때만 포트레이트에 직접 채운다.
291 if (bHasClassActionImages)
292 {
293 mSelectedCharacterPortraitImage->SetVisibility(ESlateVisibility::Collapsed);

295 }
296

303 static_cast<float>(Illustration->GetSizeY()));

305 mSelectedCharacterPortraitImage->SetBrush(PortraitBrush);
306 mSelectedCharacterPortraitImage->SetVisibility(ESlateVisibility::HitTestInvisible);
307
308 if (mSelectedCharacterPortraitFallbackText != nullptr)
309 {
310 mSelectedCharacterPortraitFallbackText->SetVisibility(ESlateVisibility::Collapsed);
311 }
312 }
313 else

315 mSelectedCharacterPortraitImage->SetVisibility(ESlateVisibility::Collapsed);
