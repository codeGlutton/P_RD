65 // 이름으로 스킬 ID 검색 (정확 일치 우선, 부분 일치는 하나일 때만)
66 bool FindSkillId(const FString& Query, FPrimaryAssetId& OutId)
67 {
68 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
69 if (AssetManager == nullptr)
70 {
71 return false;
72 }
73
74 TArray<FPrimaryAssetId> Ids;
75 AssetManager->GetPrimaryAssetIdList(SkillPrimaryAssetTypes::GetActiveType(), Ids);
76
77 TArray<FPrimaryAssetId> Matched;
78 for (const FPrimaryAssetId& Id : Ids)
79 {
80 const FString Name = Id.PrimaryAssetName.ToString();
81 if (Name.Equals(Query, ESearchCase::IgnoreCase))
82 {
83 OutId = Id;
84 return true;
85 }
86 if (Name.Contains(Query, ESearchCase::IgnoreCase))
87 {
88 Matched.Add(Id);
89 }
90 }
91
92 if (Matched.Num() == 1)
93 {
94 OutId = Matched[0];
95 return true;
96 }
97 for (const FPrimaryAssetId& Id : Matched)
98 {
99 UE_LOG(LogRD, Warning, TEXT("후보: %s"), *Id.PrimaryAssetName.ToString());
100 }
101 UE_LOG(LogRD, Warning, TEXT("스킬 검색 실패: '%s' (일치 %d개)"), *Query, Matched.Num());
102 return false;
103 }

120 // 파티 유닛 슬롯에 스킬 장착
121 void AddSkill(const TArray<FString>& Args, UWorld* World)
122 {
123 int32 UnitIndex = 0;
124 int32 SlotIndex = 0;
125 if (GetIntArg(Args, 0, UnitIndex) == false || GetIntArg(Args, 1, SlotIndex) == false || Args.Num() < 3)
126 {
127 UE_LOG(LogRD, Warning, TEXT("사용법: RD.AddSkill <유닛번호> <스킬슬롯> <에셋이름>"));
128 return;
129 }
130
131 USkillComponentModel* Skills = FindUnitSkills(World, UnitIndex);
132 if (Skills == nullptr)
133 {
134 return;
135 }
136 if (Skills->GetSkills().IsValidIndex(SlotIndex) == false)
137 {
138 UE_LOG(LogRD, Warning, TEXT("스킬슬롯 범위 밖: %d (슬롯 %d개)"), SlotIndex, Skills->GetSkills().Num());
139 return;
140 }
141
142 FPrimaryAssetId SkillId;
143 if (FindSkillId(Args[2], SkillId) == false)
144 {
145 return;
146 }
147 UStaticSkillData* SkillData = LoadSkillData(SkillId);
148 if (SkillData == nullptr)
149 {
150 UE_LOG(LogRD, Warning, TEXT("스킬 로드 실패: %s"), *SkillId.ToString());
151 return;
152 }
153
154 // 직업 불일치 등은 SetSkill이 거부
155 if (Skills->SetSkill(SlotIndex, SkillData) == false)
156 {
157 UE_LOG(LogRD, Warning, TEXT("스킬 장착 거부: %s (유닛 %d, 직업 불일치 등)"), *SkillId.PrimaryAssetName.ToString(), UnitIndex);
158 return;
159 }
160 UE_LOG(LogRD, Log, TEXT("스킬 장착: 유닛 %d 슬롯 %d <- %s"), UnitIndex, SlotIndex, *SkillId.PrimaryAssetName.ToString());
161 }

215 if (Skills == nullptr)
216 {
217 UE_LOG(LogRD, Log, TEXT(" [%d] %s: 스킬 컴포넌트 없음"), UnitIndex, *GetNameSafe(Units[UnitIndex]));
218 continue;
219 }
220
221 UE_LOG(LogRD, Log, TEXT(" [%d] %s"), UnitIndex, *GetNameSafe(Units[UnitIndex]));
222 const TArray<FSkillEntry>& Entries = Skills->GetSkills();
223 for (int32 SlotIndex = 0; SlotIndex < Entries.Num(); ++SlotIndex)
224 {
225 UE_LOG(LogRD, Log, TEXT(" 슬롯 %d: %s"), SlotIndex, Entries[SlotIndex].mData != nullptr ? *GetNameSafe(Entries[SlotIndex].mData) : TEXT("(빈 슬롯)"));
226 }
227 }
