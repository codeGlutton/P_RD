6void UTacticalEffectExecutionCalculation_ReduceCooldown::Execute(const FTacticalEffectCustomExecutionParameters& ExecutionParams, FTacticalEffectCustomExecutionOutput& OutExecutionOutput) const
7{
8 Super::Execute(ExecutionParams, OutExecutionOutput);
9
10 UAttributeSetComponentModel* TargetAttributeSetCompModel = ExecutionParams.GetTargetAttributeSetComponentModel();
11 checkf(TargetAttributeSetCompModel != nullptr, TEXT("타겟 컴포넌트 모델 nullptr"));
12
13 const FTacticalEffectSpec& Spec = ExecutionParams.GetOwningSpec();
14 const int32 ReductionAmount = FMath::FloorToInt(Spec.mDynamicMagnitude);
15
16 /* 쿨다운 감소 대상 탐색 쿼리 구성 */
17
18 FTacticalEffectQuery Query;
19 Query.mCustomMatchDelegate.BindLambda([](const FActiveTacticalEffect& Effect) -> bool
20 {
21 const UTacticalEffect* EffectDef = Effect.mSpec.mEffectClass.Get();
22 if (EffectDef == nullptr)
23 {
24 return false;
25 }
26
27 return EffectDef->IsA(UTacticalEffect_Cooldown::StaticClass()) == true;
28 });
29
30 const TArray<FActiveTacticalEffectHandle> CooldownHandles = TargetAttributeSetCompModel->GetActiveEffects(Query);
31
32 /* 쿨다운 이펙트들의 잔여 시간 감소 */
33
34 for (const FActiveTacticalEffectHandle& Handle : CooldownHandles)
35 {
36 const int32 CurrentTimeRemaining = TargetAttributeSetCompModel->GetActiveEffectTimeRemaining(Handle);
37 if (CurrentTimeRemaining <= 0)
38 {
39 continue;
40 }
41
42 const int32 NewTimeRemaining = FMath::Max(0, CurrentTimeRemaining - ReductionAmount);
43 TargetAttributeSetCompModel->SetActiveEffectTimeRemaining(NewTimeRemaining, Handle);
44 }
45
46 OutExecutionOutput.MarkDynamicMagnitudeHandledManually();
47}
