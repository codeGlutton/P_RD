36 {
37 if ((Context.WorldType == EWorldType::PIE || Context.WorldType == EWorldType::Game) && Context.World() != nullptr)

40 }
41 }
42 return nullptr;
43 }
44}

54{
55 // 라이브 월드 확보 (TAS 속성 테스트와 동일 환경)
56 UWorld* World = FindAnyGameWorld();

58 {
59 World = GWorld;

62 {
63 return false;
64 }
65
66 // 시뮬레이션 서브시스템 + 룸 인스턴스 확보

70 return false;

74 {
75 return false;
76 }
77
78 // 전략 프레임워크 모델 등록 (없을 때만 모크 등록 — 이펙트 컨텍스트/적용에 필요)
79 UTacticalFrameworkModel* FrameworkModel = Cast<UTacticalFrameworkModel>(RoomInstance->mAliveSubsystemModels.FindRef(UTacticalFrameworkModel::StaticClass()));
80 if (FrameworkModel == nullptr)
81 {
82 FrameworkModel = NewObject<UTacticalFrameworkModel>(RoomInstance);
83 RoomInstance->mAliveSubsystemModels.Add(UTacticalFrameworkModel::StaticClass(), FrameworkModel);
84 }
85
86 // 속성 컴포넌트 모크 (UUnitAttributeSet 부착)
87 UTASActorModelMock* MockActorModel = NewObject<UTASActorModelMock>(World);
88 MockActorModel->Initialize();
89 MockActorModel->BeginPlay();

92 {
93 return false;
94 }
95
96 // 공격력 기본값 10 세팅 (현재값도 10으로 전파됨)
97 const FTacticalAttribute AttackPoint = UUnitAttributeSet::GetAttackPointAttribute();
98 AttrComp->SetAttributeBaseValue(AttackPoint, 10.f);
99 TestEqual(TEXT("기본 공격력 10"), AttrComp->GetAttributeCurrentValue(AttackPoint), 10.f);
100
101 // 무기 DA (코드 구성): 공격력 +5 고유 스탯
102 UStaticWeaponEquipmentData* Weapon = NewObject<UStaticWeaponEquipmentData>();
103 FTacticalModifierInfo Mod;
104 Mod.mAttribute = AttackPoint;
105 Mod.mModifierOp = ETacticalModOp::Additive;
106 Mod.mModifierMagnitude = 5.f;
107 Weapon->mStatModifiers.Add(Mod);
108
109 // 장착: 스탯 이펙트가 자신에게 적용 (패시브 컴포넌트는 불필요 → nullptr)
110 UEquipmentComponentModel* EquipComp = NewObject<UEquipmentComponentModel>();
111 const bool bEquipped = EquipComp->EquipInternal(Weapon, nullptr, AttrComp);
112 TestTrue(TEXT("장착 성공"), bEquipped);
113 TestEqual(TEXT("장착 후 공격력 15 (기본 10 + 스탯 5)"), AttrComp->GetAttributeCurrentValue(AttackPoint), 15.f);
114
115 // 스탯 이펙트 핸들이 유효해야 함
116 const FEquippedEntry* Entry = EquipComp->GetEquipped(EEquipmentType::Weapon);
117 if (TestNotNull(TEXT("무기 슬롯 점유"), Entry))
118 {
119 TestTrue(TEXT("스탯 이펙트 핸들 유효"), Entry->mStatEffectHandle.IsValid());
120 }
121
122 // 해제: 스탯이 원복돼야 함
123 const bool bUnequipped = EquipComp->UnequipInternal(EEquipmentType::Weapon, nullptr, AttrComp);
124 TestTrue(TEXT("해제 성공"), bUnequipped);
125 TestEqual(TEXT("해제 후 공격력 10 원복"), AttrComp->GetAttributeCurrentValue(AttackPoint), 10.f);
126
127 return true;
128}
