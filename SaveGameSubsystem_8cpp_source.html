38void USaveGameSubsystem::ReportSave(const FString& Slot, uint64 Revision, const bool bSuccess) const
39{
40 if (mRevisions.FindRef(Slot) != Revision) return;
41 if (bSuccess)

43 mFailedSlots.Remove(Slot);
44 mRetryPayloads.Remove(Slot);
45 }
46 else

54bool USaveGameSubsystem::SaveObject(const FString& Slot, UObject* Object) const
55{
56 const uint64 Revision = ++mRevisions.FindOrAdd(Slot);
57 TArray<uint8> Data;
58 const bool bPrepared = !mBlockedSlots.Contains(Slot) && MakePayload(Slot, Object, Data);
59 if (bPrepared) mRetryPayloads.Add(Slot, Data);
60 const bool bSaved = bPrepared && RDCheckpoint::SaveSlot(Slot, Data);
61 ReportSave(Slot, Revision, bSaved);
62 return bSaved;
63}

94bool USaveGameSubsystem::ApplyLoadedObject(const FString& Slot, UObject* Object, bool bFound, const TArray<uint8>& Data) const
95{
96 mLoadedSlots.Add(Slot);
97 if (bFound)
98 {
99 // Failed migrations must not partly change live progress.
100 TStrongObjectPtr<UObject> Candidate(NewObject<UObject>(GetTransientPackage(), Object->GetClass()));
101 if (RDCheckpoint::Deserialize(Data, Candidate.Get()) && RDCheckpoint::Deserialize(Data, Object))
102 {
103 mBlockedSlots.Remove(Slot);
104 mFailedSlots.Remove(Slot);
105 return true;
106 }
107 }
108 if (RDCheckpoint::DoesSlotExist(Slot))
109 {
110 // Never replace an unreadable installation with blank frontend autosaves.
111 mBlockedSlots.Add(Slot);
112 mFailedSlots.Add(Slot);
113 UE_LOG(LogSave, Warning, TEXT("No valid checkpoint could be loaded for %s; existing files preserved."), *Slot);
114 OnSaveStatusChanged.Broadcast();
115 }
116 return false;
117}

171bool USaveGameSubsystem::SaveEndedRun(const TArray<uint8>& PreviousRunData)
172{
173 FRunRoomCheckpoint PreviousCheckpoint = mRoomCheckpoint;
174 ResetRunCheckpoint();
175 if (SaveRun()) return true;
176 // The caller rolls back RAM. Do not retry a rejected abandonment later in the background.
177 mRoomCheckpoint = MoveTemp(PreviousCheckpoint);
178 ++mRevisions.FindOrAdd(RUN_SLOT_NAME);
179 TArray<uint8> RetryData = PreviousRunData;
180 if (mRoomCheckpoint.IsCombatEntry()) mRoomCheckpoint.MakeSaveData(nullptr, RetryData);
181 mRetryPayloads.Add(RUN_SLOT_NAME, MoveTemp(RetryData));
182 return false;
183}

185void USaveGameSubsystem::CommitPendingRunLogs()
186{
187 URunPersistData* Run = GetRunMutableData();
188 if (Run->GetPendingCompletedRuns().IsEmpty() || mBlockedSlots.Contains(USER_SLOT_NAME)) return;
189 UUserPersistData* User = GetUserMutableData();
190 for (const FPendingCompletedRun& Pending : Run->GetPendingCompletedRuns())
191 {
192 User->ApplyRunLogOnce(Pending.TransactionId, Pending.Log);
193 }
194 // Keep the journal across future runs when User cannot be written. Replaying is idempotent.
195 if (SaveUser())
196 {
197 Run->ClearCompletedRunLogs();
198 RequestRunAutosave();
199 }

227 const TSet<FString> Failed = mFailedSlots;

230 const TArray<uint8>* Payload = mRetryPayloads.Find(Slot);
231 if (!Payload || mBlockedSlots.Contains(Slot) || mRetryPending.Contains(Slot)) continue;

233 mRetryPending.Add(Slot);
