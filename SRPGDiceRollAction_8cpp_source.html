40 case ESRPGCommandType::DicePrepare:

44 UPlayerUnitModel* PlayerUnit = Cast<UPlayerUnitModel>(mInstigator.Get());

60 UPlayerUnitModel* PlayerUnit = Cast<UPlayerUnitModel>(mInstigator.Get());

62 UDicePoolModel* DicePoolModel = PlayerUnit->GetDicePoolModel();
63 checkf(DicePoolModel != nullptr, TEXT("주사위 컴포넌트 모델 nullptr"));

66

68 {
69 TArray<UTacticalPassive*> Passives = PassiveComponentModel->GetPassivesByTiming(AbilityTags::GameplayAbility_Passive_OnStartRollingDice);

71
72 FBoardCombatTargetSnapshotData PlayerUnitSnapshot = PlayerUnit->MakeSnapshotData();

74 FPassiveActivateContext PassiveContext;

78 PassiveContext.mTargetSnapshots.Add(&PlayerUnitSnapshot);

82 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;

84 Passive->CommitPassive(DynamicPassiveData);
85 }
86 }
87
88 const FSRPGDiceRollCommand& DiceRollCommand = Command.Get<FSRPGDiceRollCommand>();

95 // 물리 굴림 연출이 보여준 윗면을 그대로 기록한다("보이는 면 = 기록 숫자").
96 DicePoolModel->ApplyRolledFaceIndices(DiceRollCommand.mRolledFaceIndices);

102 const int32 PassiveNum = Passives.Num();
103
104 FBoardCombatTargetSnapshotData PlayerUnitSnapshot = PlayerUnit->MakeSnapshotData();
105
106 FPassiveActivateContext PassiveContext;
107 PassiveContext.mOwner = PlayerUnit;
108 PassiveContext.mOwnerSnapshot = &PlayerUnitSnapshot;
109 PassiveContext.mTargets.Add(PlayerUnit);
110 PassiveContext.mTargetSnapshots.Add(&PlayerUnitSnapshot);
111
112 for (UTacticalPassive*& Passive : Passives)
113 {
114 TInstancedStruct<FDynamicPassiveData> DynamicPassiveData;
115 Passive->ActivatePassive(AbilityTags::GameplayAbility_Passive_OnEndRollingDice, PassiveContext, OUT DynamicPassiveData);
116 Passive->CommitPassive(DynamicPassiveData);
117 }
118 }
119
120 MarkActionCompleted(ESRPGActionResult::Succeeded);
121 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
122 }
123 }
