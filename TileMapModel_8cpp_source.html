36 void BuildGoalOrderedSteps(const FTileIndex& Current, const FTileIndex& Goal, TArray<FTileIndex>& OutSteps)
37 {
38 OutSteps.Reset();
39
40 // 목표까지 각 축의 남은 변위
41 const int32 DeltaX = Goal.mX - Current.mX;
42 const int32 DeltaY = Goal.mY - Current.mY;
43
44 // 목표 쪽 부호 (0이면 그 축은 목표와 같은 줄)
45 const int32 SignX = (DeltaX > 0) - (DeltaX < 0);
46 const int32 SignY = (DeltaY > 0) - (DeltaY < 0);
47
48 // 남은 거리가 큰 축을 먼저 밟도록 우선순위 결정 (계단식 근접)
49 const bool bXFirst = FMath::Abs(DeltaX) >= FMath::Abs(DeltaY);
50
51 // 1·2순위: 목표 쪽 스텝 (해당 축이 0이 아닐 때만), 먼 축 먼저
52 if (bXFirst)
53 {
54 if (SignX != 0) OutSteps.Add(FTileIndex(SignX, 0));
55 if (SignY != 0) OutSteps.Add(FTileIndex(0, SignY));
56 }
57 else
58 {
59 if (SignY != 0) OutSteps.Add(FTileIndex(0, SignY));
60 if (SignX != 0) OutSteps.Add(FTileIndex(SignX, 0));
61 }
62
63 // 3·4순위: 목표에서 멀어지는 스텝(우회용). 같은 줄(부호 0)인 축은 양방향 모두 후보
64 if (SignX <= 0) OutSteps.AddUnique(FTileIndex(1, 0));
65 if (SignX >= 0) OutSteps.AddUnique(FTileIndex(-1, 0));
66 if (SignY <= 0) OutSteps.AddUnique(FTileIndex(0, 1));
67 if (SignY >= 0) OutSteps.AddUnique(FTileIndex(0, -1));
68 }

160FTileIndex UTileMapModel::TileDeltaToStep(const FTileIndex& From, const FTileIndex& To)
161{
162 // From→To 벡터
163 const FTileIndex Delta = To - From;
164
165 // 제자리면 방향 없음
166 if (Delta == FTileIndex::Zero)
167 {
168 return FTileIndex::Zero;
169 }
170
171 // 벡터를 각도로 변경. 음수 각도(아래쪽 방향)는 한 바퀴를 더해 0°~360°로 통일
172 float Angle = FMath::Atan2(static_cast<float>(Delta.mY), static_cast<float>(Delta.mX));
173 if (Angle < 0.0f)
174 {
175 Angle += 2.0f * PI;
176 }
177
178 // 45° 단위로 반올림 = 가장 가까운 방향 번호 (360°로 반올림된 경우만 0°와 같은 방향이므로 %8)
179 const int32 Octant = FMath::RoundToInt(Angle / (PI / 4.0f)) % 8;
180
181 // 8방향 단위 스텝 (0번=+X에서 시작해 +Y쪽으로 45°씩)
182 static const FTileIndex Steps[8] =
183 {
184 FTileIndex(1, 0),
185 FTileIndex(1, 1),
186 FTileIndex(0, 1),
187 FTileIndex(-1, 1),
188 FTileIndex(-1, 0),
189 FTileIndex(-1, -1),
190 FTileIndex(0, -1),
191 FTileIndex(1, -1)
192 };
193 return Steps[Octant];
194}

385void UTileMapModel::BresenhamLine(const FTileIndex& From, const FTileIndex& To, TArray<FTileIndex>& Out) const
386{
387 // 동점(2*Error == ±Delta) 처리가 진행 방향에 따라 다른 칸을 고르지 않도록,
388 // 항상 사전순(X 우선, 같으면 Y)으로 작은 쪽에서 큰 쪽으로만 그린다
389 const bool bSwapped = (To.mX < From.mX) || (To.mX == From.mX && To.mY < From.mY);
390 const FTileIndex& Start = bSwapped ? To : From;
391 const FTileIndex& End = bSwapped ? From : To;
392
393 // 이번 호출이 추가하는 구간의 시작 위치 (Out은 누적 배열이므로 이 구간만 뒤집어야 함)
394 const int32 FirstIndex = Out.Num();
395
396 // 시작점에서 끝점까지의 정수 좌표 (Start부터 한 칸씩 전진)
397 int32 X = Start.mX;
398 int32 Y = Start.mY;
399 const int32 X1 = End.mX;
400 const int32 Y1 = End.mY;
401
402 // 각 축별로 전체이동량과 전진방향 결정
403 const int32 DeltaX = FMath::Abs(X1 - X);
404 const int32 DeltaY = FMath::Abs(Y1 - Y);
405 const int32 StepX = (X < X1) ? 1 : -1;
406 const int32 StepY = (Y < Y1) ? 1 : -1;
407
408 // 직선에서 타일이 얼마나 멀리 있는 지 나타내는 오차 누적값 (dx - dy 기준)
409 // 이 오차를 줄이는 방향, 즉 직선과 가까운 방향으로 움직이는 게 기본 아이디어
410 // 단, dx - dy 이므로 x축으로 편향된 오차이므로, 오차가 클 수록 x축으로, 오차가 작을수록 y축으로 이동하는 압력이 세진다.
411 int32 Error = DeltaX - DeltaY;
412
413 while (true)
414 {
415 // 현재 칸 수집 (정규화된 방향이므로 Start가 첫 원소, End가 마지막 원소)
416 Out.Add(FTileIndex(X, Y));
417
418 // 끝점에 도달하면 종료
419 if (X == X1 && Y == Y1)
420 break;
421
422 // 오차에 따라 x축/y축 전진 결정
423 // 양쪽 동시에 넘으면 대각선 이동
424 // @note 나눗셈 계산을 생략하기 위해 x2 값과 비교
425 const int32 DoubleError = 2 * Error;
426
427 // x축으로 이동하는 게 이동하지 않는 것보다 오차를 줄여주는 경우 -> x축으로 이동
428 // @note 원래는 DeltaY의 절반을 기준으로 판단해야 하는데, 나눗셈을 피하기 위해 오차를 두 배 해서 비교한다.
429 if (DoubleError > -DeltaY)
430 {
431 // x축으로 이동하는 게 y축 오차를 줄여주니까 dy만큼 오차 감소
432 Error -= DeltaY;
433 X += StepX;
434 }
435 // y축으로 이동하는 게 이동하지 않는 것보다 오차를 줄여주는 경우 -> y축으로 이동
436 // @note 원래는 DeltaX의 절반을 기준으로 판단해야 하는데, 나눗셈을 피하기 위해 오차를 두 배 해서 비교한다.
437 if (DoubleError < DeltaX)
438 {
439 // y축으로 이동하면 그 다음부터는 x축 이동 압력이 커질 수 있도록 Error 수치 조정
440 Error += DeltaX;
441 Y += StepY;
442 }
443 }
444
445 // 뒤집어 그린 경우 이번에 추가한 구간만 반전해 "From이 첫 원소, To가 마지막 원소" 되도록 변경
446 if (bSwapped)
447 Algo::Reverse(MakeArrayView(Out.GetData() + FirstIndex, Out.Num() - FirstIndex));
448}

472 }
473 }
474 }

532TArray<FTileIndex> UTileMapModel::GetReachableTiles(const FTileIndex& Origin, int32 MoveDistance, const UBoardActorModel* IgnoreBlocker) const
533{
534 TArray<FTileIndex> Result;
535
536 // 시작점이 맵 밖이거나 이동력이 없으면 도달 가능 타일 없음
537 if (!IsValidIndex(Origin) || MoveDistance <= 0)
538 return Result;
539
540 // 칸별 최단 도달 거리 기록 (-1=미방문), 1차원 인덱스로 접근
541 TArray<int32> Distance;
542 Distance.Init(-1, mWidth * mHeight);
543 Distance[TileIndexToLinearIndex(Origin)] = 0;
544
545 // BFS 큐: 가까운 칸부터 한 겹씩 확장 (인덱스로 순회해 pop 비용 회피)
546 TArray<FTileIndex> Frontier;
547 Frontier.Add(Origin);
548
549 for (int32 Head = 0; Head < Frontier.Num(); ++Head)
550 {
551 const FTileIndex Current = Frontier[Head];
552 const int32 CurrentDistance = Distance[TileIndexToLinearIndex(Current)];
553
554 // 이동력을 모두 쓴 칸에서는 더 뻗지 않음
555 if (CurrentDistance >= MoveDistance)
556 continue;
557
558 // 직교 4방향 이웃 검사
559 for (const FTileIndex& Step : Orthogonal4)
560 {
561 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
562
563 // 맵 밖이면 제외 (INDEX_NONE)
564 const int32 LinearIndex = TileIndexToLinearIndex(Next);
565 if (LinearIndex == INDEX_NONE)
566 continue;
567
568 // 이미 방문한 칸은 건너뜀 (BFS라 먼저 방문한 경로가 최단)
569 if (Distance[LinearIndex] != -1)
570 continue;
571
572 // 장애물이나 유닛이 있는 타일은 통과나 도착 불가 (무시 대상인 경우는 가능)
573 // (확장 지점: 투명화 등 통과 규칙이 생기면 이 한 줄만 교체)
574 if (IsOccupied(Next, IgnoreBlocker))
575 continue;
576
577 // 거리 확정 후 결과·큐에 추가
578 Distance[LinearIndex] = CurrentDistance + 1;
579 Result.Add(Next);
580 Frontier.Add(Next);
581 }
582 }
583
584 return Result;
585}

587TArray<int32> UTileMapModel::GetDistanceField(const FTileIndex& Target, const UBoardActorModel* IgnoreBlocker) const
588{
589 // 칸별 최단 경로 거리
590 TArray<int32> Distance;
591 Distance.Init(-1, mWidth * mHeight);
592
593 // 목표가 맵 밖이면 도달 불가
594 const int32 TargetLinear = TileIndexToLinearIndex(Target);
595 if (TargetLinear == INDEX_NONE)
596 return Distance;
597
598 // 목표 칸 자신은 점유(유닛 등)여도 거리 0의 시작점으로 허용
599 Distance[TargetLinear] = 0;
600
601 // BFS 큐: 가까운 칸부터 한 겹씩 확장
602 TArray<FTileIndex> Frontier;
603 Frontier.Add(Target);
604
605 for (int32 Head = 0; Head < Frontier.Num(); ++Head)
606 {
607 const FTileIndex Current = Frontier[Head];
608 const int32 NextDistance = Distance[TileIndexToLinearIndex(Current)] + 1;
609
610 // 직교 4방향 이웃 검사
611 for (const FTileIndex& Step : Orthogonal4)
612 {
613 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
614
615 // 맵 밖이면 제외 (INDEX_NONE)
616 const int32 LinearIndex = TileIndexToLinearIndex(Next);
617 if (LinearIndex == INDEX_NONE)
618 continue;
619
620 // 이미 방문한 칸은 건너뜀 (BFS라 먼저 방문한 경로가 최단)
621 if (Distance[LinearIndex] != -1)
622 continue;
623
624 // 장애물/유닛이 점유한 칸은 통과/도착 불가
625 // 무시 대상 액터만 있으면 통과 가능
626 if (IsOccupied(Next, IgnoreBlocker))
627 continue;
628
629 // 거리 확정 후 큐에 추가
630 Distance[LinearIndex] = NextDistance;
631 Frontier.Add(Next);
632 }
633 }
634
635 return Distance;

644TArray<FTileIndex> UTileMapModel::FindPath(const FTileIndex& Start, const FTileIndex& Goal, const UBoardActorModel* IgnoreBlocker, bool bAllowOccupiedGoal) const
645{
646 TArray<FTileIndex> Result;
647
648 // 양 끝 중 하나라도 맵 밖이면 경로 없음
649 if (!IsValidIndex(Start) || !IsValidIndex(Goal))
650 return Result;
651
652 // 시작=목표면 그 한 칸이 곧 경로
653 if (Start == Goal)
654 {
655 Result.Add(Start);
656 return Result;
657 }
658
659 // 이미 점유하는 게 있으면 통과나 도착 불가 (무시 대상이거나 점유 도착 허용이면 가능)
660 if (!bAllowOccupiedGoal && IsOccupied(Goal, IgnoreBlocker))
661 return Result;
662
663 // 칸별 직전 칸(부모)을 1차원 인덱스로 기록 — 방문표시(INDEX_NONE=미방문) + 경로 복원을 겸함
664 TArray<int32> Parent;
665 Parent.Init(INDEX_NONE, mWidth * mHeight);
666
667 // 시작 칸은 자기 자신을 부모로 둬서 방문 표시 겸 복원 종료 기준으로 삼음
668 const int32 StartLinear = TileIndexToLinearIndex(Start);
669 Parent[StartLinear] = StartLinear;
670
671 // BFS 큐: 가까운 칸부터 한 겹씩 확장 (인덱스 순회로 pop 비용 회피)
672 TArray<FTileIndex> Frontier;
673 Frontier.Add(Start);
674
675 // 매 칸의 목표 방향 우선 정렬 스텝을 담을 버퍼 (루프 밖 선언해 재할당 회피)
676 TArray<FTileIndex> OrderedSteps;
677
678 // 목표 도달 여부 (도달 시 바깥 루프까지 즉시 종료)
679 bool bReached = false;
680
681 for (int32 Head = 0; Head < Frontier.Num() && !bReached; ++Head)
682 {
683 const FTileIndex Current = Frontier[Head];
684
685 // 목표 쪽(먼 축 먼저) → 우회 순으로 4방향 스텝 정렬
686 BuildGoalOrderedSteps(Current, Goal, OrderedSteps);
687
688 for (const FTileIndex& Step : OrderedSteps)
689 {
690 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
691
692 // 맵 밖이면 제외 (INDEX_NONE)
693 const int32 NextLinear = TileIndexToLinearIndex(Next);
694 if (NextLinear == INDEX_NONE)
695 continue;
696
697 // 이미 방문한 칸은 건너뜀 (BFS라 먼저 방문한 경로가 최단)
698 if (Parent[NextLinear] != INDEX_NONE)
699 continue;
700
701 // 이미 점유하는 게 있으면 통과나 도착 불가 (무시 대상이거나 점유 도착 허용된 목표면 가능)
702 if (IsOccupied(Next, IgnoreBlocker) && !(bAllowOccupiedGoal && Next == Goal))
703 continue;
704
705 // 직전 칸을 부모로 기록
706 Parent[NextLinear] = TileIndexToLinearIndex(Current);
707
708 // 목표에 닿았으면 더 넓힐 필요 없이 종료 (BFS라 이게 최단)
709 if (Next == Goal)
710 {
711 bReached = true;
712 break;
713 }
714
715 // 다음 겹 확장을 위해 큐에 추가
716 Frontier.Add(Next);
717 }
718 }
719
720 // 목표에 끝내 닿지 못했으면 경로 없음
721 if (Parent[TileIndexToLinearIndex(Goal)] == INDEX_NONE)
722 return Result;
723
724 // 목표→시작으로 부모를 거슬러 올라가며 경로를 모음 (역순으로 쌓임)
725 for (int32 Linear = TileIndexToLinearIndex(Goal); ; Linear = Parent[Linear])
726 {
727 // 1차원 인덱스를 (x,y)로 환원 (y*Width + x의 역변환)
728 Result.Add(FTileIndex(Linear % mWidth, Linear / mWidth));
729
730 // 시작 칸까지 모았으면 종료 (시작은 부모가 자기 자신)
731 if (Linear == StartLinear)
732 break;
733 }
734
735 // 시작→목표 순서가 되도록 뒤집기
736 Algo::Reverse(Result);
737 return Result;
738}

802TArray<FTileIndex> UTileMapModel::GetAimableTiles(const FTileIndex& Origin, int32 Range, EAimPattern Pattern, bool bIncludeOccupied, ETileLayerFlag BlockerLayers, const UBoardActorModel* Incoming, const UBoardActorModel* IgnoreBlocker) const
803{
804 TArray<FTileIndex> Result;
805
806 // 시작점이 맵 밖이면 조준 불가
807 if (!IsValidIndex(Origin))
808 return Result;
809
810 /*
811 * 페이즈1: 패턴에 따라 후보타일 수집
812 */
813
814 // Single은 기준 타일 한 칸만 (Range 무시). 그 외 패턴은 기준 타일 제외
815 if (Pattern == EAimPattern::Single)
816 {
817 Result.Add(Origin);
818 }
819 else
820 {
821 // Single 제외한 다른 패턴은 사거리가 없으면 후보타일도 없음
822 if (Range <= 0)
823 return Result;
824
825 // 패턴별 후보 타일 수집 (최종 필터링은 페이즈2에서 수행)
826 switch (Pattern)
827 {
828 case EAimPattern::Cross:
829 // 직교 4방향 직선
830 for (const FTileIndex& Step : Orthogonal4)
831 AppendRayTiles(Origin, Step, Range, Result);
832 break;
833
834 case EAimPattern::Star:
835 // 직교 + 대각 8방향 직선
836 for (const FTileIndex& Step : Orthogonal4)
837 AppendRayTiles(Origin, Step, Range, Result);
838 for (const FTileIndex& Step : Diagonal4)
839 AppendRayTiles(Origin, Step, Range, Result);
840 break;
841
842 case EAimPattern::Square:
843 // 중심 기준 사각형 범위 (하늘 낙하 개념 — 시야 무시)
844 AppendSquareTiles(Origin, Range, Result);
845 break;
846
847 default:
848 break;
849 }
850 }
851
856 // 시야/점유로 막힌 타일들은 후보에서 삭제
857 Result.RemoveAll([&](const FTileIndex& Candidate)
858 {
859 return IsAimBlocked(Origin, Candidate, Pattern, bIncludeOccupied, BlockerLayers, Incoming, IgnoreBlocker);
860 });
861
862 return Result;
863}

879bool UTileMapModel::IsInAimPattern(const FTileIndex& Origin, const FTileIndex& Target, int32 Range, EAimPattern Pattern) const
880{
881 // 기준 좌표에서 검사 좌표까지의 축별 변위
882 const int32 DeltaX = Target.mX - Origin.mX;
883 const int32 DeltaY = Target.mY - Origin.mY;
884
885 switch (Pattern)
886 {
887 case EAimPattern::Single:
888 // 기준 타일 한 칸만 (Range 무시)
889 return DeltaX == 0 && DeltaY == 0;
890
891 case EAimPattern::Cross:
892 // 직교 4방향 직선: 같은 행 또는 같은 열이면서 사거리 이내 (기준 타일 제외)
893 if (DeltaX == 0 && DeltaY == 0)
894 return false;
895 return (DeltaX == 0 && FMath::Abs(DeltaY) <= Range)
896 || (DeltaY == 0 && FMath::Abs(DeltaX) <= Range);
897
898 case EAimPattern::Star:
899 // 직교 4방향 + 정대각 4방향 직선, 사거리 이내 (기준 타일 제외)
900 if (DeltaX == 0 && DeltaY == 0)
901 return false;
902 if ((DeltaX == 0 && FMath::Abs(DeltaY) <= Range)
903 || (DeltaY == 0 && FMath::Abs(DeltaX) <= Range))
904 return true;
905 return FMath::Abs(DeltaX) == FMath::Abs(DeltaY) && FMath::Abs(DeltaX) <= Range;
906
907 case EAimPattern::Square:
908 {
909 // 중심 기준 사각형 범위: 체비셰프 거리가 사거리 이내 (기준 타일 제외)
910 const int32 Chebyshev = FTileIndex::ChebyshevDistance(Origin, Target);
911 return Chebyshev != 0 && Chebyshev <= Range;
912 }
913
914 default:
915 return false;
916 }
917}

919bool UTileMapModel::IsAimBlocked(const FTileIndex& Origin, const FTileIndex& Target, EAimPattern Pattern, bool bIncludeOccupied, ETileLayerFlag BlockerLayers, const UBoardActorModel* Incoming, const UBoardActorModel* IgnoreBlocker) const
920{
921 // 차폐 검사 여부: 직선패턴 AND 차폐 레이어 지정 (Square 공격은 하늘에서 내리는 공격이니까 차폐 무시)
922 const bool bApplyLineOfSight = (BlockerLayers != ETileLayerFlag::None) && (Pattern == EAimPattern::Cross || Pattern == EAimPattern::Star);
923
924 // 직사인데 시야가 막히면 조준 불가
925 if (bApplyLineOfSight && !HasLineOfSight(Origin, Target, IgnoreBlocker, BlockerLayers))
926 return true;
927
928 // Single은 시전자 자기 칸을 조준하므로 점유 검사 의미 없음 (자기 칸은 항상 점유 상태)
929 if (Pattern == EAimPattern::Single)
930 return false;
931
932 // 점유 타일을 포함하지 않으면 조준 불가
933 if (!bIncludeOccupied && IsOccupied(Target))
934 // Incoming으로 교체 불가능하면 조준 불가
935 if (!((Incoming != nullptr) && GetReplaceableActors(Target, Incoming).Num() > 0))
936 return true;
937
938 return false;
939}

941TArray<FTileIndex> UTileMapModel::GetTargetTiles(const FTileIndex& Caster, const FTileIndex& Target, ETargetPattern Pattern) const
942{
943 TArray<FTileIndex> Result;
944
945 // 조준 타일이 맵 밖이면 타겟 없음
946 if (!IsValidIndex(Target))
947 return Result;
948
949 switch (Pattern)
950 {
951 case ETargetPattern::TargetOnly:
952 // 조준 타일 한 칸만 타겟
953 Result.Add(Target);
954 break;
955
956 case ETargetPattern::LineToTarget:
957 {
958 // 시전자가 맵 밖이거나 조준 타일과 같으면 조준 타일만 타겟
959 if (!IsValidIndex(Caster) || Caster == Target)
960 {
961 Result.Add(Target);
962 break;
963 }
964
965 // 시전자→조준 타일 직선이 지나는 칸들을 래스터화 (첫 원소=시전자, 마지막 원소=조준 타일 보장)
966 TArray<FTileIndex> LineTiles;
967 RasterizeLine(Caster, Target, LineTiles);
968
969 // 시전자 칸을 제외하고 가까운 순으로 타겟에 추가
970 for (int32 Index = 1; Index < LineTiles.Num(); ++Index)
971 Result.Add(LineTiles[Index]);
972 }
973 break;
974
975 default:
976 break;
977 }
978
979 return Result;
980}

982TArray<FTileIndex> UTileMapModel::GetEffectTiles(const FTileIndex& Target, EEffectPattern Pattern, int32 Size, ETileLayerFlag BlockerLayers, const UBoardActorModel* IgnoreBlocker) const

1123TArray<FTileIndex> UTileMapModel::GetPullPath(const FTileIndex& Puller, const FTileIndex& Pulled, int32 MaxDistance) const
1124{
1125 // 경로는 최소한 당겨지는 칸 자신을 포함 (못 당기면 제자리 한 칸)
1126 TArray<FTileIndex> Path;
1127 Path.Add(Pulled);
1128
1129 // 당겨지는 칸이 맵 밖이거나 당길 거리가 없으면 그대로 둔다
1130 if (!IsValidIndex(Pulled) || MaxDistance <= 0)
1131 return Path;
1132
1133 // 당기는 쪽과 당겨지는 쪽 사이에 장애물이 있으면 당길 수 없음 (유닛은 관통, 밀치기와 같은 규칙)

1137 // 당겨지는 쪽→당기는 쪽 방향을 가장 가까운 8방향 단위 스텝으로 양자화

1139
1140 // 같은 칸이라 방향이 없으면 당길 수 없음

1145 FTileIndex Current = Pulled;
1146 while (Path.Num() <= MaxDistance && IsPullAdjacent(Current, Puller) == false)
1147 {
1148 const FTileIndex Next = Current + Step;

1157

1162 break;

1167 }
