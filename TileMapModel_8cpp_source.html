35 void BuildGoalOrderedSteps(const FTileIndex& Current, const FTileIndex& Goal, TArray<FTileIndex>& OutSteps)
36 {
37 OutSteps.Reset();
38
39 // 목표까지 각 축의 남은 변위
40 const int32 DeltaX = Goal.mX - Current.mX;
41 const int32 DeltaY = Goal.mY - Current.mY;
42
43 // 목표 쪽 부호 (0이면 그 축은 목표와 같은 줄)
44 const int32 SignX = (DeltaX > 0) - (DeltaX < 0);
45 const int32 SignY = (DeltaY > 0) - (DeltaY < 0);
46
47 // 남은 거리가 큰 축을 먼저 밟도록 우선순위 결정 (계단식 근접)
48 const bool bXFirst = FMath::Abs(DeltaX) >= FMath::Abs(DeltaY);
49
50 // 1·2순위: 목표 쪽 스텝 (해당 축이 0이 아닐 때만), 먼 축 먼저
51 if (bXFirst)
52 {
53 if (SignX != 0) OutSteps.Add(FTileIndex(SignX, 0));
54 if (SignY != 0) OutSteps.Add(FTileIndex(0, SignY));
55 }
56 else
57 {
58 if (SignY != 0) OutSteps.Add(FTileIndex(0, SignY));
59 if (SignX != 0) OutSteps.Add(FTileIndex(SignX, 0));
60 }
61
62 // 3·4순위: 목표에서 멀어지는 스텝(우회용). 같은 줄(부호 0)인 축은 양방향 모두 후보
63 if (SignX <= 0) OutSteps.AddUnique(FTileIndex(1, 0));
64 if (SignX >= 0) OutSteps.AddUnique(FTileIndex(-1, 0));
65 if (SignY <= 0) OutSteps.AddUnique(FTileIndex(0, 1));
66 if (SignY >= 0) OutSteps.AddUnique(FTileIndex(0, -1));
67 }
68}

139ETileActorDirection UTileMapModel::TileDeltaToDirection(const FTileIndex& From, const FTileIndex& To, ETileActorDirection Fallback)
140{
141 const int32 DeltaX = To.mX - From.mX;
142 const int32 DeltaY = To.mY - From.mY;
143
144 // 제자리면 폴백 방향 리턴
145 if (DeltaX == 0 && DeltaY == 0)
146 {
147 return Fallback;
148 }
149
150 // 델타값이 큰 쪽을 바라보게 되므로 그쪽으로 방향 결정.
151 // X, Y축 델타가 같을때는 X 우선.
152 if (FMath::Abs(DeltaX) >= FMath::Abs(DeltaY))
153 {
154 return DeltaX > 0 ? ETileActorDirection::Forward : ETileActorDirection::Backward;
155 }
156 return DeltaY > 0 ? ETileActorDirection::Right : ETileActorDirection::Left;
157}

187{

205 {
206 return true;
207 }
208 }
209

213TArray<UBoardActorModel*> UTileMapModel::GetReplaceableActors(const FTileIndex& TileIndex, const UBoardActorModel* Incoming) const
214{
215 TArray<UBoardActorModel*> Result;
216
217 // 진입 액터가 없으면 교체 대상 없음
218 if (Incoming == nullptr)
219 return Result;
220
221 // 타일 위 모든 액터를 받아서 교체 조건으로 거름
222 for (UBoardActorModel* Actor : GetActorsOnTile(TileIndex, ETileLayerFlag::All))
223 {
224 if (Actor == nullptr)
225 continue;
226
227 // 기존 액터의 레이어가 진입 액터의 레이어와 다르면 제외
228 if (!EnumHasAnyFlags(Actor->GetTileLayerFlags(), Incoming->GetTileLayerFlags()))
229 continue;
230
231 // 기존 액터가 진입 액터의 교체를 허용하지 않으면 제외
232 if (!EnumHasAnyFlags(Actor->GetReplaceLayerFlags(), Incoming->GetTileLayerFlags()))
233 continue;
234
235 // 진입 액터의 우선순위가 낮으면 제외
236 if (Incoming->GetOverlayLayerPriority() < Actor->GetOverlayLayerPriority())
237 continue;
238
239 // 모든 조건을 충족하면 교체대상
240 Result.Add(Actor);
241 }
242 return Result;
243}

245void UTileMapModel::AppendRayTiles(const FTileIndex& Origin, const FTileIndex& Step, int32 Range, TArray<FTileIndex>& Out) const
246{
247 // 원점에서 Step 방향으로 한 칸씩 전진하며 수집 (원점 자신은 제외)
248 FTileIndex Current = Origin;
249 for (int32 Distance = 0; Distance < Range; ++Distance)
250 {
251 // 다음 칸으로 전진
252 Current.mX += Step.mX;
253 Current.mY += Step.mY;
254
255 // 맵 밖으로 나가면 이 방향은 더 진행하지 않고 종료
256 if (!IsValidIndex(Current))
257 break;
258
259 // 맵 안의 칸만 후보로 누적
260 Out.Add(Current);
261 }

299void UTileMapModel::AppendSquareTiles(const FTileIndex& Center, const int32 Radius, TArray<FTileIndex>& Out) const
300{
301 // 중심 기준 [-Radius, Radius] 정사각형 격자를 순회한다.
302 for (int32 OffsetY = -Radius; OffsetY <= Radius; ++OffsetY)
303 {
304 for (int32 OffsetX = -Radius; OffsetX <= Radius; ++OffsetX)
305 {
306 // 중심 자신은 본체가 별도 처리하므로 제외 (중복 방지)
307 if (OffsetX == 0 && OffsetY == 0)
308 continue;
309
310 // ReSharper disable once CppTooWideScopeInitStatement
311 const FTileIndex Candidate(Center.mX + OffsetX, Center.mY + OffsetY);
312
313 // 맵 안의 칸만 누적 (장애물/시야는 보지 않음 — 하늘 낙하 개념)
314 if (IsValidIndex(Candidate))
315 Out.Add(Candidate);
316 }
317 }

320void UTileMapModel::BresenhamLine(const FTileIndex& From, const FTileIndex& To, TArray<FTileIndex>& Out) const
321{
322 // 동점(2*Error == ±Delta) 처리가 진행 방향에 따라 다른 칸을 고르지 않도록,
323 // 항상 사전순(X 우선, 같으면 Y)으로 작은 쪽에서 큰 쪽으로만 그린다
324 const bool bSwapped = (To.mX < From.mX) || (To.mX == From.mX && To.mY < From.mY);
325 const FTileIndex& Start = bSwapped ? To : From;
326 const FTileIndex& End = bSwapped ? From : To;
327
328 // 이번 호출이 추가하는 구간의 시작 위치 (Out은 누적 배열이므로 이 구간만 뒤집어야 함)
329 const int32 FirstIndex = Out.Num();
330
331 // 시작점에서 끝점까지의 정수 좌표 (Start부터 한 칸씩 전진)
332 int32 X = Start.mX;
333 int32 Y = Start.mY;
334 const int32 X1 = End.mX;
335 const int32 Y1 = End.mY;
336
337 // 각 축별로 전체이동량과 전진방향 결정
338 const int32 DeltaX = FMath::Abs(X1 - X);
339 const int32 DeltaY = FMath::Abs(Y1 - Y);
340 const int32 StepX = (X < X1) ? 1 : -1;
341 const int32 StepY = (Y < Y1) ? 1 : -1;
342
343 // 직선에서 타일이 얼마나 멀리 있는 지 나타내는 오차 누적값 (dx - dy 기준)
344 // 이 오차를 줄이는 방향, 즉 직선과 가까운 방향으로 움직이는 게 기본 아이디어
345 // 단, dx - dy 이므로 x축으로 편향된 오차이므로, 오차가 클 수록 x축으로, 오차가 작을수록 y축으로 이동하는 압력이 세진다.
346 int32 Error = DeltaX - DeltaY;
347
348 while (true)
349 {
350 // 현재 칸 수집 (정규화된 방향이므로 Start가 첫 원소, End가 마지막 원소)
351 Out.Add(FTileIndex(X, Y));
352
353 // 끝점에 도달하면 종료
354 if (X == X1 && Y == Y1)
355 break;
356
357 // 오차에 따라 x축/y축 전진 결정
358 // 양쪽 동시에 넘으면 대각선 이동
359 // @note 나눗셈 계산을 생략하기 위해 x2 값과 비교
360 const int32 DoubleError = 2 * Error;
361
362 // x축으로 이동하는 게 이동하지 않는 것보다 오차를 줄여주는 경우 -> x축으로 이동
363 // @note 원래는 DeltaY의 절반을 기준으로 판단해야 하는데, 나눗셈을 피하기 위해 오차를 두 배 해서 비교한다.
364 if (DoubleError > -DeltaY)
365 {
366 // x축으로 이동하는 게 y축 오차를 줄여주니까 dy만큼 오차 감소
367 Error -= DeltaY;
368 X += StepX;
369 }
370 // y축으로 이동하는 게 이동하지 않는 것보다 오차를 줄여주는 경우 -> y축으로 이동
371 // @note 원래는 DeltaX의 절반을 기준으로 판단해야 하는데, 나눗셈을 피하기 위해 오차를 두 배 해서 비교한다.
372 if (DoubleError < DeltaX)
373 {
374 // y축으로 이동하면 그 다음부터는 x축 이동 압력이 커질 수 있도록 Error 수치 조정
375 Error += DeltaX;
376 Y += StepY;
377 }
378 }
379
380 // 뒤집어 그린 경우 이번에 추가한 구간만 반전해 "From이 첫 원소, To가 마지막 원소" 되도록 변경
381 if (bSwapped)
382 Algo::Reverse(MakeArrayView(Out.GetData() + FirstIndex, Out.Num() - FirstIndex));
383}

391bool UTileMapModel::HasLineOfSight(const FTileIndex& From, const FTileIndex& To, const UBoardActorModel* IgnoreBlocker) const
392{
393 // From→To 직선이 지나는 칸들을 래스터화 (첫 원소=From, 마지막 원소=To 보장)
394 TArray<FTileIndex> LineTiles;
395 RasterizeLine(From, To, LineTiles);
396
397 // 양 끝(From, To)을 제외한 중간 칸만 검사
398 for (int32 Index = 1; Index < LineTiles.Num() - 1; ++Index)
399 {
400 // 중간 칸에 시야를 막는 액터(Obstacle 또는 Unit)가 있으면 시야가 막힘(=LoS:false)
401 for (const UBoardActorModel* Actor : GetActorsOnTile(LineTiles[Index], ETileLayerFlag::Obstacle | ETileLayerFlag::Unit))
402 {
403 // 무시 대상(자리를 비울 예정인 유닛 등)은 차폐로 치지 않음
404 if (Actor != IgnoreBlocker)
405 {
406 return false;
407 }
408 }
409 }
410
411 // 시야를 막는 액터가 없음 (=LoS:true)
412 return true;
413}

463TArray<FTileIndex> UTileMapModel::GetReachableTiles(const FTileIndex& Origin, int32 MoveDistance) const
464{
465 TArray<FTileIndex> Result;
466
467 // 시작점이 맵 밖이거나 이동력이 없으면 도달 가능 타일 없음
468 if (!IsValidIndex(Origin) || MoveDistance <= 0)
469 return Result;
470
471 // 칸별 최단 도달 거리 기록 (-1=미방문), 1차원 인덱스로 접근
472 TArray<int32> Distance;
473 Distance.Init(-1, mWidth * mHeight);
474 Distance[TileIndexToLinearIndex(Origin)] = 0;
475
476 // BFS 큐: 가까운 칸부터 한 겹씩 확장 (인덱스로 순회해 pop 비용 회피)
477 TArray<FTileIndex> Frontier;
478 Frontier.Add(Origin);
479
480 for (int32 Head = 0; Head < Frontier.Num(); ++Head)
481 {
482 const FTileIndex Current = Frontier[Head];
483 const int32 CurrentDistance = Distance[TileIndexToLinearIndex(Current)];
484
485 // 이동력을 모두 쓴 칸에서는 더 뻗지 않음
486 if (CurrentDistance >= MoveDistance)
487 continue;
488
489 // 직교 4방향 이웃 검사
490 for (const FTileIndex& Step : Orthogonal4)
491 {
492 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
493
494 // 맵 밖이면 제외 (INDEX_NONE)
495 const int32 LinearIndex = TileIndexToLinearIndex(Next);
496 if (LinearIndex == INDEX_NONE)
497 continue;
498
499 // 이미 방문한 칸은 건너뜀 (BFS라 먼저 방문한 경로가 최단)
500 if (Distance[LinearIndex] != -1)
501 continue;
502
503 // 장애물·유닛이 점유한 칸은 통과·도착 불가
504 // (확장 지점: 투명화 등 통과 규칙이 생기면 이 한 줄만 교체)
505 if (IsOccupied(Next))
506 continue;
507
508 // 거리 확정 후 결과·큐에 추가
509 Distance[LinearIndex] = CurrentDistance + 1;
510 Result.Add(Next);
511 Frontier.Add(Next);
512 }
513 }
514
515 return Result;
516}
517
518TArray<int32> UTileMapModel::GetDistanceField(const FTileIndex& Target, const UBoardActorModel* IgnoreBlocker) const
519{
520 // 칸별 최단 경로 거리
521 TArray<int32> Distance;
522 Distance.Init(-1, mWidth * mHeight);
523
524 // 목표가 맵 밖이면 도달 불가
525 const int32 TargetLinear = TileIndexToLinearIndex(Target);
526 if (TargetLinear == INDEX_NONE)
527 return Distance;
528
529 // 목표 칸 자신은 점유(유닛 등)여도 거리 0의 시작점으로 허용
530 Distance[TargetLinear] = 0;
531
532 // BFS 큐: 가까운 칸부터 한 겹씩 확장
533 TArray<FTileIndex> Frontier;
534 Frontier.Add(Target);
535
536 for (int32 Head = 0; Head < Frontier.Num(); ++Head)
537 {
538 const FTileIndex Current = Frontier[Head];
539 const int32 NextDistance = Distance[TileIndexToLinearIndex(Current)] + 1;
540
541 // 직교 4방향 이웃 검사
542 for (const FTileIndex& Step : Orthogonal4)
543 {
544 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
545
546 // 맵 밖이면 제외 (INDEX_NONE)
547 const int32 LinearIndex = TileIndexToLinearIndex(Next);
548 if (LinearIndex == INDEX_NONE)
549 continue;
550
551 // 이미 방문한 칸은 건너뜀 (BFS라 먼저 방문한 경로가 최단)
552 if (Distance[LinearIndex] != -1)
553 continue;
554
555 // 장애물/유닛이 점유한 칸은 통과/도착 불가
556 // 무시 대상 액터만 있으면 통과 가능
557 if (IsOccupied(Next, IgnoreBlocker))
558 continue;
559
560 // 거리 확정 후 큐에 추가
561 Distance[LinearIndex] = NextDistance;
562 Frontier.Add(Next);
563 }
564 }
565
566 return Distance;
567}
568
575TArray<FTileIndex> UTileMapModel::FindPath(const FTileIndex& Start, const FTileIndex& Goal) const
576{
577 TArray<FTileIndex> Result;
578
579 // 양 끝 중 하나라도 맵 밖이면 경로 없음
580 if (!IsValidIndex(Start) || !IsValidIndex(Goal))
581 return Result;
582
583 // 시작=목표면 그 한 칸이 곧 경로
584 if (Start == Goal)
585 {
586 Result.Add(Start);
587 return Result;
588 }
589
590 // 목표 칸이 점유돼 있으면 도착 불가 (GetReachableTiles와 동일 규칙)
591 if (IsOccupied(Goal))
592 return Result;
593
594 // 칸별 직전 칸(부모)을 1차원 인덱스로 기록 — 방문표시(INDEX_NONE=미방문) + 경로 복원을 겸함
595 TArray<int32> Parent;
596 Parent.Init(INDEX_NONE, mWidth * mHeight);
597
598 // 시작 칸은 자기 자신을 부모로 둬서 방문 표시 겸 복원 종료 기준으로 삼음
599 const int32 StartLinear = TileIndexToLinearIndex(Start);
600 Parent[StartLinear] = StartLinear;
601
602 // BFS 큐: 가까운 칸부터 한 겹씩 확장 (인덱스 순회로 pop 비용 회피)
603 TArray<FTileIndex> Frontier;
604 Frontier.Add(Start);
605
606 // 매 칸의 목표 방향 우선 정렬 스텝을 담을 버퍼 (루프 밖 선언해 재할당 회피)
607 TArray<FTileIndex> OrderedSteps;
608
609 // 목표 도달 여부 (도달 시 바깥 루프까지 즉시 종료)
610 bool bReached = false;
611
612 for (int32 Head = 0; Head < Frontier.Num() && !bReached; ++Head)
613 {
614 const FTileIndex Current = Frontier[Head];
615
616 // 목표 쪽(먼 축 먼저) → 우회 순으로 4방향 스텝 정렬
617 BuildGoalOrderedSteps(Current, Goal, OrderedSteps);
618
619 for (const FTileIndex& Step : OrderedSteps)
620 {
621 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
622
623 // 맵 밖이면 제외 (INDEX_NONE)
624 const int32 NextLinear = TileIndexToLinearIndex(Next);
625 if (NextLinear == INDEX_NONE)
626 continue;
627
628 // 이미 방문한 칸은 건너뜀 (BFS라 먼저 방문한 경로가 최단)
629 if (Parent[NextLinear] != INDEX_NONE)
630 continue;
631
632 // 장애물·유닛이 점유한 칸은 통과·도착 불가
633 if (IsOccupied(Next))
634 continue;
635
636 // 직전 칸을 부모로 기록
637 Parent[NextLinear] = TileIndexToLinearIndex(Current);
638
639 // 목표에 닿았으면 더 넓힐 필요 없이 종료 (BFS라 이게 최단)
640 if (Next == Goal)
641 {
642 bReached = true;
643 break;
644 }
645
646 // 다음 겹 확장을 위해 큐에 추가
647 Frontier.Add(Next);
648 }
649 }
650
651 // 목표에 끝내 닿지 못했으면 경로 없음
652 if (Parent[TileIndexToLinearIndex(Goal)] == INDEX_NONE)
653 return Result;
654
655 // 목표→시작으로 부모를 거슬러 올라가며 경로를 모음 (역순으로 쌓임)
656 for (int32 Linear = TileIndexToLinearIndex(Goal); ; Linear = Parent[Linear])
657 {
658 // 1차원 인덱스를 (x,y)로 환원 (y*Width + x의 역변환)
659 Result.Add(FTileIndex(Linear % mWidth, Linear / mWidth));
660
661 // 시작 칸까지 모았으면 종료 (시작은 부모가 자기 자신)
662 if (Linear == StartLinear)
663 break;
664 }
665
666 // 시작→목표 순서가 되도록 뒤집기
667 Algo::Reverse(Result);
668 return Result;
669}

704TArray<FTileIndex> UTileMapModel::GetAimableTiles(const FTileIndex& Origin, int32 Range, EAimPattern Pattern, bool bIncludeOccupied, bool bIndirect, const UBoardActorModel* Incoming, const UBoardActorModel* IgnoreBlocker) const
705{
706 TArray<FTileIndex> Result;
707
708 // 시작점이 맵 밖이면 조준 불가
709 if (!IsValidIndex(Origin))
710 return Result;
711
712 /*
713 * 페이즈1: 패턴에 따라 후보타일 수집
714 */
715
716 // Single은 기준 타일 한 칸만 (Range 무시). 그 외 패턴은 기준 타일 제외
717 if (Pattern == EAimPattern::Single)
718 {
719 Result.Add(Origin);
720 }
721 else
722 {
723 // Single 제외한 다른 패턴은 사거리가 없으면 후보타일도 없음
724 if (Range <= 0)
725 return Result;
726
727 // 패턴별 후보 타일 수집 (최종 필터링은 페이즈2에서 수행)
728 switch (Pattern)
729 {
730 case EAimPattern::Cross:
731 // 직교 4방향 직선
732 for (const FTileIndex& Step : Orthogonal4)
733 AppendRayTiles(Origin, Step, Range, Result);
734 break;
735
736 case EAimPattern::Star:
737 // 직교 + 대각 8방향 직선
738 for (const FTileIndex& Step : Orthogonal4)
739 AppendRayTiles(Origin, Step, Range, Result);
740 for (const FTileIndex& Step : Diagonal4)
741 AppendRayTiles(Origin, Step, Range, Result);
742 break;
743
744 case EAimPattern::Square:
745 // 중심 기준 사각형 범위 (하늘 낙하 개념 — 시야 무시)
746 AppendSquareTiles(Origin, Range, Result);
747 break;
748
749 default:
750 break;
751 }
752 }
753
758 // 장애물 막힘 검사 여부: 직선패턴 AND 직사공격 (Square 공격은 하늘에서 내리는 공격이니까 장애물 무시)
759 const bool bApplyLineOfSight = !bIndirect && (Pattern == EAimPattern::Cross || Pattern == EAimPattern::Star);
760
761 // 후보를 시야/점유 조건으로 거름
762 // @note 인덱스를 뒤에서 앞으로 오면서 제거하면 배열 재할당 이슈 없음
763 for (int32 Index = Result.Num() - 1; Index >= 0; --Index)
764 {
765 const FTileIndex& Candidate = Result[Index];
766
767 // 직사인데 시야가 막히면 조준 불가
768 if (bApplyLineOfSight && !HasLineOfSight(Origin, Candidate, IgnoreBlocker))
769 {
770 Result.RemoveAt(Index);
771 continue;
772 }
773
774 // 점유 타일을 포함하지 않으면 조준 불가
775 if (!bIncludeOccupied && IsOccupied(Candidate))
776 // Incoming으로 교체 불가능하면 조준 불가
777 if (!((Incoming != nullptr) && GetReplaceableActors(Candidate, Incoming).Num() > 0))
778 Result.RemoveAt(Index);
779 }
780
781 return Result;
782}

784TArray<FTileIndex> UTileMapModel::GetEffectTiles(const FTileIndex& Caster, const FTileIndex& Target, EEffectPattern Pattern, int32 Size, bool bPenetrate) const
785{
786 TArray<FTileIndex> Result;
787
788 // 영향 중심(Target)이 맵 밖이면 영향 없음
789 if (!IsValidIndex(Target))
790 return Result;
791
792 // 영향 중심은 모든 패턴에서 항상 포함 (광역도 중심은 맞음)
793 Result.Add(Target);
794
795 // Single이거나 범위가 없으면 중심 한 칸만 영향
796 if (Pattern == EEffectPattern::Single || Size <= 0)
797 return Result;
798
799 // 패턴별 영향 타일을 중심 둘레에 덧붙임 (직선은 관통 아니면 점유 칸에서 멈춤)
800 switch (Pattern)
801 {
802 case EEffectPattern::Cross:
803 // 중심에서 직교 4방향
804 for (const FTileIndex& Step : Orthogonal4)
805 AppendBlockableRay(Target, Step, Size, bPenetrate, Result);
806 break;
807
808 case EEffectPattern::Star:
809 // 중심에서 직교 + 대각 8방향
810 for (const FTileIndex& Step : Orthogonal4)
811 AppendBlockableRay(Target, Step, Size, bPenetrate, Result);
812 for (const FTileIndex& Step : Diagonal4)
813 AppendBlockableRay(Target, Step, Size, bPenetrate, Result);
814 break;
815
816 case EEffectPattern::Square:
817 // 중심 기준 사각형 범위 (하늘 낙하 개념 — 관통 무관)
818 AppendSquareTiles(Target, Size, Result);
819 break;
820
821 case EEffectPattern::Beam:
822 {
823 // 시전자→타겟 방향을 부호로 단위 스텝화 (8방향 중에 하나로 강제 매핑)
824 const FTileIndex Step(
825 FMath::Sign(Target.mX - Caster.mX),
826 FMath::Sign(Target.mY - Caster.mY)
827 );
828
829 // Target(클릭 지점)을 시작으로 그 방향으로 뻗음 — 빔 길이는 Target 포함 총 Size칸
830 // (Target은 상단에서 이미 추가했으므로 너머로 Size-1칸만 더 뻗음)
831 // 비관통 빔이 점유 칸을 직접 조준한 경우 거기서 맞고 멈춤 — 너머로 뻗지 않음
832 if ((Step.mX != 0 || Step.mY != 0) && (bPenetrate || !IsOccupied(Target)))
833 AppendBlockableRay(Target, Step, Size - 1, bPenetrate, Result);
834 }
835 break;
836
837 default:
838 break;
839 }
840
841 return Result;
842}

844FTileIndex UTileMapModel::GetPushDestination(const FTileIndex& Pusher, const FTileIndex& Pushed, int32 MaxDistance) const
845{
846 // 밀리는 칸이 맵 밖이거나 밀칠 거리가 없으면 그대로 둔다
847 if (!IsValidIndex(Pushed) || MaxDistance <= 0)
848 return Pushed;
849
850 // 미는 쪽→밀리는 쪽 방향을 각 축 부호로 8방향 단위 스텝화 (대각 포함)
851 const FTileIndex Step(
852 FMath::Sign(Pushed.mX - Pusher.mX),
853 FMath::Sign(Pushed.mY - Pusher.mY)
854 );
855
856 // 같은 칸이라 방향이 없으면 밀 수 없음
857 if (Step.mX == 0 && Step.mY == 0)
858 return Pushed;
859
860 // 밀리는 칸에서부터 한 칸씩 전진하며 멈출 지점을 찾음
861 FTileIndex Current = Pushed;
862 for (int32 Distance = 0; Distance < MaxDistance; ++Distance)
863 {
864 const FTileIndex Next(Current.mX + Step.mX, Current.mY + Step.mY);
865
866 // 맵 밖으로는 밀리지 않음
867 if (!IsValidIndex(Next))
868 break;
869
870 // 장애물/유닛에 걸리면 더 밀리지 않고 직전 칸에서 멈춤
871 if (IsOccupied(Next))
872 break;
873
874 // 빈 칸이면 거기까지 밀려남
875 Current = Next;
876 }
877
878 return Current;
879}

963void UTileMapModel::PlaceActor(const FTileTransform& NextTransform, UBoardActorModel* Actor)
964{
965 checkf(Actor != nullptr, TEXT("Actor가 nullptr"));
966 checkf(CanPlace(NextTransform.mIndex, Actor), TEXT("배치할 수 없는 타일"));
967
968 FTile* Tile = GetTile(NextTransform.mIndex);
969

971 for (UBoardActorModel* Victim : GetReplaceableActors(NextTransform.mIndex, Actor))
972 {
973 RemoveActor(Victim);
974 Victim->OnReplaced(Tile, Actor);
975 }
976
977 // 논리 좌표 갱신
978 Actor->SetTileTransform(NextTransform);
979 Actor->OnPlaceTileTransform.Broadcast(NextTransform, TileToWorldTransform(NextTransform));
980
981 // 다음 타일에 등록 후 진입 오버랩 통지 (등록 → Begin 순서)
982 RegisterActorToTile(Tile, Actor);
983 NotifyBeginOverlap(Tile, Actor);
984}

1001void UTileMapModel::RefreshActorPlacements()
1002{
1003 // 모든 타일의 보드 액터를 순회하며 현재 논리 좌표를 새 기준으로 변환해 브로드캐스팅
1004 for (const FTile& Tile : mTiles)
1005 {
1006 for (const TWeakObjectPtr<UBoardActorModel>& Actor : Tile.mBoardActors)
1007 {
1008 if (Actor.IsValid())
