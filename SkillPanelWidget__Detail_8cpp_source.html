42 UTextBlock* FindOrCreateSkillButtonText(UWidgetTree* WidgetTree, UButton* Button, FName Name, const FText& Text)
43 {
44 if (WidgetTree == nullptr || Button == nullptr)
45 {
46 return nullptr;
47 }
48
49 UTextBlock* ButtonText = Cast<UTextBlock>(WidgetTree->FindWidget(Name));
50 if (ButtonText == nullptr)
51 {
52 ButtonText = WidgetTree->ConstructWidget<UTextBlock>(UTextBlock::StaticClass(), Name);
53 }
54 if (ButtonText != nullptr)
55 {
56 ButtonText->SetJustification(ETextJustify::Center);
57 ButtonText->SetColorAndOpacity(FSlateColor(RDPanelNavigationStyle::GetButtonTextColor()));

61 Button->AddChild(ButtonText);
62 }
63 }
