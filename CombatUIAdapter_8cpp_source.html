33 // 스킬 레일 index (UI와 합의): BASIC=0, STEP=5.
34 // [합의필요] 실제 스킬 데이터가 연결되면 레일 index 하드코딩 대신 SkillId/BuildAction 매핑으로 교체해야 한다.
35 constexpr int32 SkillIndexBasic = 0;
36 constexpr int32 SkillIndexStep = 5;

38 // [합의필요] HP/Gold/이동력 진짜 소스 = UUnitData(GAS 폐기 후). 아래는 임시 placeholder —

40 // (다이스는 이미 진짜 — APlayerUnit::UDicePoolModel에서 읽음. HP/Gold만 미연결.)
41 constexpr float PlayerStartHP = 100.0f;
42 constexpr float EnemyStartHP = 30.0f;

46 constexpr int32 PlayerStartMovePoint = 0;

58void UCombatUIAdapter::Build(USRPGCombatSubsystem* InCombat, const URunPersistData* InRun)
59{
60 // 재빌드 대비: 이전 subsystem 턴 종료 구독을 먼저 해제한다.
61 // 전환/teardown 시 모델이 이미 파괴돼 GetModel()이 null일 수 있으므로 모델까지 null 가드한다.
62 if (mCombat != nullptr && mEndTurnHandle.IsValid())
63 {
64 if (USRPGCombatModel* PrevModel = mCombat->GetModel<USRPGCombatModel>())
65 {
66 PrevModel->OnEndAnyTurnUI.Remove(mEndTurnHandle);
67 }
68 mEndTurnHandle.Reset();
69 }
70
71 mCombat = InCombat;
72 mPlayerLevel = InRun != nullptr ? InRun->GetPlayerLevel() : 1;
73 mUnitStates.Reset();
74 mNextUnitId = 0;
75
76 // 플레이어: 실제 스폰된 유닛에서 타일/액터를 가져온다.
77 USRPGCombatModel* CombatModel = (mCombat != nullptr) ? mCombat->GetModel<USRPGCombatModel>() : nullptr;
78 if (CombatModel != nullptr)
79 {
80 // 턴이 끝날 때마다 이번 턴에 쓴 주사위 잠금을 해제하도록 훅을 건다(계약 D: Begin/EndTurn 리셋).
81 mEndTurnHandle = CombatModel->OnEndAnyTurnUI.AddUObject(this, &UCombatUIAdapter::HandleEndAnyTurn);
82
83 for (const TObjectPtr<UUnitModel>& Unit : CombatModel->GetUnits())
84 {
85 if (Unit == nullptr || Unit->IsPlayerUnitModel() == false)
86 {
87 continue;
88 }
89 FCombatUnitState State;
90 State.mId = mNextUnitId++;
91 State.mIsPlayer = true;
92 State.mTile = Unit->GetTileTransform().mIndex;
93 // 타일 트랜스폼이 신뢰 안 될 수 있어((0,0) 등) 액터 실제 위치 기준으로 보정.
94 if (UTileMapModel* TileMap = mCombat->GetModel<USRPGCombatModel>()->GetTileMap())
95 {
96 const FTileIndex ActorTile = TileMap->WorldToTileIndex(Unit->GetView<AActor>()->GetActorLocation());
97 if (TileMap->IsValidIndex(ActorTile))
98 {
99 State.mTile = ActorTile;
100 }
101 }
102 // 진짜 속성값(TAS AttributeSet, 커브테이블 초기화)에서 HP를 읽는다.
103 // 단 속성 초기화가 안 된 경우(MaxHP=0/비정상)는 placeholder로 폴백.
104 float RealMaxHP = 0.f;
105 float RealHP = 0.f;
106 if (UAttributeSetComponentModel* AttrComp = Unit->GetAttributeComponentModel())
107 {
108 bool bFound = false;
109 const float MaxHPValue = AttrComp->GetAttributeCurrentValue(UUnitAttributeSet::GetMaxHPAttribute(), bFound);
110 if (bFound)
111 {
112 RealMaxHP = MaxHPValue;
113 RealHP = AttrComp->GetAttributeCurrentValue(UUnitAttributeSet::GetHPAttribute(), bFound);
114 }
115 }
116 if (RealMaxHP > 0.f && RealMaxHP < 100000.f)
117 {
118 State.mMaxHP = RealMaxHP;
119 State.mHP = (RealHP > 0.f && RealHP < 100000.f) ? RealHP : RealMaxHP;
120 }
121 else
122 {
123 State.mMaxHP = PlayerStartHP;
124 State.mHP = PlayerStartHP;
125 }
126 State.mMovePoint = PlayerStartMovePoint;
127 State.mActor = Unit->GetView<AUnit>();
128 mUnitStates.Add(State);
129 }
130 }
131
132 // 적: 액터 스폰 대신 가상 유닛으로 타일 위에 둔다(스폰 크래시 회피).
133 for (const FTileIndex& Tile : VirtualEnemyTiles)
134 {
135 FCombatUnitState State;
136 State.mId = mNextUnitId++;
137 State.mIsPlayer = false;
138 State.mTile = Tile;
139 State.mMaxHP = EnemyStartHP;
140 State.mHP = EnemyStartHP;
141 mUnitStates.Add(State);
142 }
143
144 PushAll();
145}

169void UCombatUIAdapter::HandleCombatCommand(ECombatInputType Type, int32 IntPayload)
170{
171 switch (Type)
172 {
173 case ECombatInputType::SelectSkill:
174 mSelectedSkillIndex = IntPayload;
175 mPendingAttackDamage = -1;
176 mMovePending = false;
177 break;
178
179 case ECombatInputType::ToggleDice:
180 {
181 // 이미 쓴 주사위는 무시(턴 종료/다음 턴 시작까지 잠금).
182 if (mUIModel != nullptr)
183 {
184 const TArray<FDiceSlotUI>& Dice = mUIModel->GetDiceUIs();
185 if (Dice.IsValidIndex(IntPayload) && Dice[IntPayload].mIsUsed)
186 {
187 break;
188 }
189 }
190
191 const int32 DiceValue = GetRolledDiceValue(IntPayload);
192 if (DiceValue <= 0)
193 {
194 break;
195 }
196 mPendingDiceIndex = IntPayload; // 확정 시 '사용됨' 처리할 주사위.
197 if (mSelectedSkillIndex == SkillIndexStep)
198 {
199 // STEP: 주사위 배치 → 본인 타일을 회색(Aim)으로. 탭마다 노랑→빨강(확정).
200 mPendingStepValue = DiceValue;
201 mStepStage = 0;
202 mStepTile = GetPlayerTile();
203 SetSingleTileHighlight(mStepTile, ETileHighlightFlag::Aim);
204 }
205 else if (mSelectedSkillIndex == SkillIndexBasic)
206 {
207 // BASIC: 주사위 배치 → 사거리(회색 Aim)를 깐다. 그 안에서 타깃 탭=노랑, 재탭=빨강+실행.
208 mPendingAttackDamage = DiceValue;
209 mAttackTargetTile = FTileIndex::Invalid;
210 if (UTileMapModel* TileMap = mCombat != nullptr ? mCombat->GetModel<USRPGCombatModel>()->GetTileMap() : nullptr)
211 {
212 // 사거리 기준점 = 플레이어 타일. 타일 트랜스폼이 신뢰 안 될 수 있어((0,0) 등) 액터 실제 위치로 잡는다.
213 FTileIndex Origin = GetPlayerTile();
214 if (const FCombatUnitState* PlayerState = FindPlayerState())
215 {
216 if (PlayerState->mActor.IsValid())
217 {
218 Origin = TileMap->WorldToTileIndex(PlayerState->mActor->GetActorLocation());
219 }
220 }
221 // 사거리 = Square 범위(타일맵 GetAimableTiles, pyramidmine 구현). 점유 타일 포함(적 조준).
222 mAimTiles = TileMap->GetAimableTiles(Origin, BasicAttackRange, EAimPattern::Square, true, false);
223 ClearAllHighlight();
224 TileMap->GetView<ATileMap>()->SetTileHighlight(mAimTiles, ETileHighlightFlag::Aim); // 회색 사거리
225 }
226 }
227 break;
228 }
229
230 case ECombatInputType::RollDice:
231 // 굴림 요청: 컴포넌트(데이터)에서 굴리고 결과 뷰를 push한다.
232 RollDice();
233 break;
234
235 case ECombatInputType::Move:
236 {
237 // MOVE 모드: 도달 가능 범위(BFS 경로 기반, 이동력 기준)를 회색으로 깔고 그 안만 이동.
238 mMovePending = true;
239 mSelectedSkillIndex = INDEX_NONE;
240 mPendingAttackDamage = -1;
241 mAttackTargetTile = FTileIndex::Invalid;
242 if (UTileMapModel* TileMap = mCombat != nullptr ? mCombat->GetModel<USRPGCombatModel>()->GetTileMap() : nullptr)
243 {
244 mAimTiles = TileMap->GetReachableTiles(GetPlayerTile(), GetPlayerMovePoint());
245 ClearAllHighlight();
246 TileMap->GetView<ATileMap>()->SetTileHighlight(mAimTiles, ETileHighlightFlag::Aim);
247 }
248 break;
249 }
250
251 case ECombatInputType::Cancel:
252 ClearPendingAction();
253 break;
254
255 default:
256 break;
257 }
258}

261void UCombatUIAdapter::HandleWorldTouch(FVector2D ScreenPosition, bool bLongPress)
262{
263 // 롱프레스 여부는 후속 상세 패널 계약을 위해 받지만, 이 임시 어댑터는 일반 탭 액션만 처리한다.
264 FTileIndex Tile;
265 if (ResolveTileFromScreen(ScreenPosition, Tile) == false)
266 {
267 // 타일맵 밖 탭 = 진행 중 스킬/이동 취소.
268 ClearPendingAction();
269 return;
270 }
271
272 // STEP 단계 확정: 본인 타일을 탭할 때마다 회색→노랑→빨강(확정).
273 if (mPendingStepValue >= 0)
274 {
275 if (Tile.mX == mStepTile.mX && Tile.mY == mStepTile.mY)
276 {
277 ++mStepStage;
278 if (mStepStage == 1)
279 {
280 SetSingleTileHighlight(mStepTile, ETileHighlightFlag::Select); // 노랑
281 }
282 else if (mStepStage >= 2)
283 {
284 SetSingleTileHighlight(mStepTile, ETileHighlightFlag::Effect); // 빨강 = 확정
285 ApplyStep(mPendingStepValue); // 이동력 += 주사위값
286 if (mDicePool != nullptr && mPendingDiceIndex != INDEX_NONE)
287 {
288 mDicePool->MarkDiceUsed(mPendingDiceIndex); // 쓴 주사위 잠금
289 PushDiceUIs(); // 잠금 상태를 UI에 반영
290 }
291 ClearPendingAction();
292 }
293 }
294 else
295 {
296 // 다른 곳 탭 = 취소.
297 ClearPendingAction();
298 }
299 return;
300 }
301
302 // BASIC 평타: 사거리(회색) 안에서 타깃 탭=노랑, 같은 칸 재탭=빨강+실행.
303 if (mPendingAttackDamage >= 0)
304 {
305 UTileMapModel* TileMap = mCombat != nullptr ? mCombat->GetModel<USRPGCombatModel>()->GetTileMap() : nullptr;
306
307 // 사거리 밖 탭 = 취소.
308 if (mAimTiles.Contains(Tile) == false)
309 {
310 ClearPendingAction();
311 return;
312 }
313
314 const bool bSameTile = (mAttackTargetTile.mX == Tile.mX && mAttackTargetTile.mY == Tile.mY);
315 if (bSameTile == false)
316 {
317 // 1차 선택(또는 다른 칸 재선택): 회색 사거리는 유지하고 선택칸만 노랑(Select 최우선).
318 mAttackTargetTile = Tile;
319 if (TileMap != nullptr)
320 {
321 ATileMap* TileMapView = TileMap->GetView<ATileMap>();
322
323 TileMapView->ClearTileHighlight(ETileHighlightFlag::Select);
324 TileMapView->ClearTileHighlight(ETileHighlightFlag::Effect);
325 TileMapView->SetTileHighlight(mAimTiles, ETileHighlightFlag::Aim);
326 TArray<FTileIndex> Picked;
327 Picked.Add(Tile);
328 TileMapView->SetTileHighlight(Picked, ETileHighlightFlag::Select); // 노랑
329 }
330 return;
331 }
332
333 // 2차(같은 칸 재탭): 빨강 + 실행.
334 if (TileMap != nullptr)
335 {
336 ATileMap* TileMapView = TileMap->GetView<ATileMap>();
337
338 TileMapView->ClearTileHighlight(ETileHighlightFlag::Select);
339 TArray<FTileIndex> Picked;
340 Picked.Add(Tile);
341 TileMapView->SetTileHighlight(Picked, ETileHighlightFlag::Effect); // 빨강
342 }
343 const int32 TargetId = FindUnitIdAtTile(Tile);
344 FCombatUnitState* Target = FindStateById(TargetId);
345 if (Target != nullptr && Target->mIsPlayer == false)
346 {
347 ApplyBasicAttack(TargetId, mPendingAttackDamage);
348 if (mDicePool != nullptr && mPendingDiceIndex != INDEX_NONE)
349 {
350 mDicePool->MarkDiceUsed(mPendingDiceIndex); // 적을 친 경우에만 주사위 소모.
351 PushDiceUIs(); // 잠금 상태를 UI에 반영
352 }
353 }
354 ClearPendingAction();
355 return;
356 }
357
358 // MOVE: 도달 범위 안에서 칸 탭=노랑(선택), 같은 칸 재탭=빨강(확정)+이동. (BASIC과 동일 흐름)
359 if (mMovePending)
360 {
361 // 도달 범위 밖 탭 = 취소.
362 if (mAimTiles.Contains(Tile) == false)
363 {
364 ClearPendingAction();
365 return;
366 }
367
368 UTileMapModel* TileMap = mCombat != nullptr ? mCombat->GetModel<USRPGCombatModel>()->GetTileMap() : nullptr;
369 const bool bSameTile = (mAttackTargetTile.mX == Tile.mX && mAttackTargetTile.mY == Tile.mY);
370 if (bSameTile == false)
371 {
372 // 1차 선택: 회색 범위 유지 + 선택칸만 노랑.
373 mAttackTargetTile = Tile;
374 if (TileMap != nullptr)
375 {
376 ATileMap* TileMapView = TileMap->GetView<ATileMap>();
377
378 TileMapView->ClearTileHighlight(ETileHighlightFlag::Select);
379 TileMapView->ClearTileHighlight(ETileHighlightFlag::Effect);
380 TileMapView->SetTileHighlight(mAimTiles, ETileHighlightFlag::Aim);
381 TArray<FTileIndex> Picked;
382 Picked.Add(Tile);
383 TileMapView->SetTileHighlight(Picked, ETileHighlightFlag::Select); // 노랑
384 }
385 return;
386 }
387
388 // 2차(같은 칸 재탭): 빨강 + 이동.
389 if (TileMap != nullptr)
390 {
391 ATileMap* TileMapView = TileMap->GetView<ATileMap>();
392
393 TileMapView->ClearTileHighlight(ETileHighlightFlag::Select);
394 TArray<FTileIndex> Picked;
395 Picked.Add(Tile);
396 TileMapView->SetTileHighlight(Picked, ETileHighlightFlag::Effect); // 빨강
397 }
398 if (TryMovePlayer(Tile))
399 {
400 // 이동 후 남은 이동력으로 도달범위 재계산(다시 선택 가능), 없으면 종료.
401 if (GetPlayerMovePoint() > 0 && TileMap != nullptr)
402 {
403 ATileMap* TileMapView = TileMap->GetView<ATileMap>();
404
405 mAttackTargetTile = FTileIndex::Invalid;
406 mAimTiles = TileMap->GetReachableTiles(GetPlayerTile(), GetPlayerMovePoint());
407 ClearAllHighlight();
408 TileMapView->SetTileHighlight(mAimTiles, ETileHighlightFlag::Aim);
409 }
410 else
411 {
412 ClearPendingAction();
413 }
414 }
415 else
416 {
417 ClearPendingAction();
418 }
419 return;
420 }
421}

497void UCombatUIAdapter::RollDice()
498{
499 if (mDicePool == nullptr)
500 {
501 return;
502 }
503
504 /*
505 * 어댑터 단독 단계라 임시 난수 스트림을 쓴다.
506 * 게임플레이/런 통합 시 run의 RandomStream(URandomStreamFunctionLibrary)을 주입해 결정론을 맞춘다.

510 mDicePool->RollAll(Stream);

515void UCombatUIAdapter::PushDiceUIs() const
516{
517 if (mUIModel == nullptr || mDicePool == nullptr)
518 {
519 return;
520 }
521
522 const TArray<TObjectPtr<UDiceModel>>& Dice = mDicePool->GetDices();
523
524 TArray<FDiceSlotUI> Views;
525 Views.Reserve(Dice.Num());
526
527 for (const TObjectPtr<UDiceModel>& DicePtr : Dice)
528 {
529 if (DicePtr == nullptr)
530 {
531 continue;

572bool UCombatUIAdapter::ResolveTileFromScreen(const FVector2D& ScreenPosition, FTileIndex& OutTile) const
573{
574 ATileMap* TileMap = mCombat != nullptr ? mCombat->GetModel<USRPGCombatModel>()->GetTileMap()->GetView<ATileMap>() : nullptr;
575 UWorld* World = mCombat != nullptr ? mCombat->GetWorld() : nullptr;
576 if (TileMap == nullptr || World == nullptr)
577 {
578 return false;
579 }
580 APlayerController* PlayerController = World->GetFirstPlayerController();
581 if (PlayerController == nullptr)
582 {
583 return false;
584 }
585
586 // 라인 트레이스(터치 전용 트레이스 채널 TileOnlyTrace, PR #72)로 타일 판정.
587 // 손가락/커서 위치를 엔진이 직접 추적하므로 화면좌표 변환(평면 역투영)이 필요 없어
588 // 고DPI 기기에서도 어긋나지 않는다(정석 — RDCollision.h가 명시한 GetHitResultUnderFinger 경로).
589 // ScreenPosition은 계약상 받지만, 트레이스가 실제 입력 위치를 직접 쓰므로 여기선 사용하지 않는다.
590 FHitResult HitResult;
591 bool bHit = PlayerController->GetHitResultUnderFinger(ETouchIndex::Touch1, RDTraceChannels::TileOnlyTrace, false, HitResult);
592 if (bHit == false)
593 {
594 // 폴백: 손가락 추적이 비어있으면(또는 마우스), 전달받은 화면좌표로 직접 트레이스한다.
595 // GetHitResultAtScreenPosition은 뷰포트 픽셀을 기대하므로 Slate 절대좌표를 보정해 넘긴다.
596 FVector2D ViewportPixel;
597 FVector2D ViewportDPIScaled;
598 USlateBlueprintLibrary::AbsoluteToViewport(World, ScreenPosition, ViewportPixel, ViewportDPIScaled);
599 bHit = PlayerController->GetHitResultAtScreenPosition(ViewportPixel, RDTraceChannels::TileOnlyTrace, false, HitResult);
600 }
601 if (bHit == false)
602 {
603 return false; // 타일맵 밖 또는 입력 없음.
604 }
605
606 const FTileIndex Tile = TileMap->WorldToTileIndex(HitResult.ImpactPoint);
607 if (TileMap->IsValidIndex(Tile) == false)
608 {
609 return false; // 타일맵 밖.
610 }
611 OutTile = Tile;
612 return true;
613}

706void UCombatUIAdapter::RefreshUnitMarkers()
707{
708 UWorld* World = mCombat != nullptr ? mCombat->GetWorld() : nullptr;
709 if (World == nullptr)
710 {
711 return;
712 }
713
714 // 이전 마커 제거(이동/사망 반영을 위해 매 push마다 재생성).
715 for (const TObjectPtr<AActor>& Marker : mUnitMarkers)
716 {
717 if (Marker != nullptr)
718 {
719 Marker->Destroy();
720 }
721 }
722 mUnitMarkers.Reset();
723

731 FVector Loc = TileToWorld(State.mTile);

738 {
739 continue;
740 }
741

743 MeshComp->SetMobility(EComponentMobility::Movable);
744 MeshComp->SetStaticMesh(State.mIsPlayer ? PlayerMesh : EnemyMesh);
745 MeshComp->SetCollisionEnabled(ECollisionEnabled::NoCollision); // 타일 트레이스 통과
746 Marker->SetActorScale3D(State.mIsPlayer ? FVector(0.45f, 0.45f, 0.9f) : FVector(0.5f));

748 if (ShapeMat != nullptr)
749 {
750 UMaterialInstanceDynamic* DynMat = UMaterialInstanceDynamic::Create(ShapeMat, Marker);
751 if (DynMat != nullptr)
752 {
753 DynMat->SetVectorParameterValue(TEXT("Color"), State.mIsPlayer ? FLinearColor(0.2f, 0.45f, 1.0f) : FLinearColor(1.0f, 0.2f, 0.2f));
754 MeshComp->SetMaterial(0, DynMat);
755 }
756 }
757
758 mUnitMarkers.Add(Marker);
759 }

807bool UCombatUIAdapter::TryMovePlayer(const FTileIndex& TargetTile)
808{
809 FCombatUnitState* Player = mUnitStates.FindByPredicate([](const FCombatUnitState& State) { return State.mIsPlayer; });
810 if (Player == nullptr || Player->mMovePoint <= 0)
811 {
812 return false;
813 }
814
815 // 가상 적이 있는 타일로는 이동 불가.
816 const int32 OccupantId = FindUnitIdAtTile(TargetTile);
817 if (OccupantId != INDEX_NONE)
818 {
819 return false;
820 }
821
822 // 이동 비용 = 현재 칸→목표 칸 거리(Chebyshev 근사). 이동력이 모자라면 불가.
823 const int32 Cost = FMath::Max(FMath::Abs(TargetTile.mX - Player->mTile.mX), FMath::Abs(TargetTile.mY - Player->mTile.mY));
824 if (Cost <= 0 || Cost > Player->mMovePoint)
825 {
826 return false;
827 }
828
829 Player->mTile = TargetTile;
830 Player->mMovePoint -= Cost;
831
832 // 실제 플레이어 액터도 해당 타일 위로 옮긴다(StartActorMovement는 현재 스텁이라 위치를 직접 설정).
833 if (Player->mActor.IsValid())
834 {
835 if (UTileMapModel* TileMap = mCombat != nullptr ? mCombat->GetModel<USRPGCombatModel>()->GetTileMap() : nullptr)
836 {
837 FVector NewLoc = TileMap->TileToWorldLocation(TargetTile);
838 NewLoc.Z = Player->mActor->GetActorLocation().Z; // 높이 유지
839 // 폰 루트가 Static mobility면 SetActorLocation이 실패(moved=0)한다. 이동 전 Movable로 바꾼다.
840 if (USceneComponent* Root = Player->mActor->GetRootComponent())
841 {
842 Root->SetMobility(EComponentMobility::Movable);
843 }
844 Player->mActor->SetActorLocation(NewLoc, /*bSweep=*/false, nullptr, ETeleportType::TeleportPhysics);
845 }
846 }
847
848 PushAll();
849 return true;
850}
