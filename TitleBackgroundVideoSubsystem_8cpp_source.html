45void UTitleBackgroundVideoSubsystem::PreloadTitleBackgroundVideo(const FString& RelativeContentPath)
46{
47 // 명시 경로면 단일 루프, 빈 경로면 설정의 30종 셔플 목록을 사용한다.
48 TArray<FString> RequestedPaths = RelativeContentPath.IsEmpty()
49 ? GetDefaultTitleBackgroundVideoPaths()
50 : TArray<FString> { RelativeContentPath };
51 RequestedPaths.RemoveAll([](const FString& Path) { return Path.IsEmpty(); });
52 if (RequestedPaths.IsEmpty())
53 {
54 return;
55 }
56
57 // 같은 재생 목록이 이미 열려 있으면 위젯의 중복 Prime/Construct 호출로 재시작하지 않는다.
58 if (mOpenRequested && RequestedPaths == mCandidateRelativePaths)
59 {
60 return;
61 }
62
63 // MediaPlayer/MediaTexture/MediaSource 3종이 준비돼 있는지 보장(최초 1회 생성·연결).
64 EnsureMediaObjects();
65 if (mMediaPlayer == nullptr || mMediaSource == nullptr)
66 {
67 return;
68 }
69
70 mCandidateRelativePaths = MoveTemp(RequestedPaths);
71 mShuffleBag.Reset();
72 mShuffleCursor = 0;
73 mCurrentCandidateIndex = INDEX_NONE;
74 OpenNextTitleBackgroundVideo();
75}

86void UTitleBackgroundVideoSubsystem::StopTitleBackgroundVideo()
87{
88 if (mMediaPlayer != nullptr)
89 {
90 // 종료 후 콜백이 죽은 this를 건드리지 않도록 먼저 해제한다.
91 mMediaPlayer->OnMediaOpened.RemoveAll(this);
92 mMediaPlayer->OnMediaOpenFailed.RemoveAll(this);
93 mMediaPlayer->OnEndReached.RemoveAll(this);
94 mMediaPlayer->Close();
95 }
96 if (mMediaTexture != nullptr)
97 {
98 mMediaTexture->SetMediaPlayer(nullptr);
99 mMediaTexture->UpdateResource();
100 }
101
102 mCurrentRelativePath.Reset();
103 mCandidateRelativePaths.Reset();
104 mShuffleBag.Reset();
105 mShuffleCursor = 0;
106 mCurrentCandidateIndex = INDEX_NONE;
107 mOpenRequested = false;
108 mMediaOpened = false;

111void UTitleBackgroundVideoSubsystem::RebuildShuffleBag()
112{
113 mShuffleBag.Reset(mCandidateRelativePaths.Num());
114 for (int32 Index = 0; Index < mCandidateRelativePaths.Num(); ++Index)
115 {
116 mShuffleBag.Add(Index);
117 }
118
119 for (int32 Index = mShuffleBag.Num() - 1; Index > 0; --Index)
120 {
121 mShuffleBag.Swap(Index, FMath::RandRange(0, Index));
122 }
123 if (mShuffleBag.Num() > 1 && mShuffleBag[0] == mCurrentCandidateIndex)
124 {
125 mShuffleBag.Swap(0, 1);
126 }
127 mShuffleCursor = 0;
128}

130void UTitleBackgroundVideoSubsystem::OpenNextTitleBackgroundVideo()
131{
132 if (mMediaPlayer == nullptr || mMediaSource == nullptr || mCandidateRelativePaths.IsEmpty())
133 {
134 return;
135 }
136
137 for (int32 Attempt = 0; Attempt < mCandidateRelativePaths.Num(); ++Attempt)
138 {
139 if (mShuffleCursor >= mShuffleBag.Num())
140 {
141 RebuildShuffleBag();
142 }
143
144 const int32 CandidateIndex = mShuffleBag[mShuffleCursor++];
145 const FString& RequestedPath = mCandidateRelativePaths[CandidateIndex];
146 const FString VideoPath = RDUITexture::ResolveContentFilePath(RequestedPath);
147 if (FPaths::FileExists(VideoPath) == false)
148 {
149 UE_LOG(LogRD, Warning, TEXT("TitleBackgroundVideoSubsystem: title background video missing: %s"), *VideoPath);
150 continue;
151 }
152
153 mCurrentCandidateIndex = CandidateIndex;
154 mCurrentRelativePath = RequestedPath;
155 mOpenRequested = true;
156 mMediaOpened = false;
157 mMediaPlayer->PlayOnOpen = true;
158 mMediaPlayer->SetLooping(mCandidateRelativePaths.Num() == 1);
159 mMediaPlayer->Close();
160 mMediaSource->SetFilePath(VideoPath);
161 if (mMediaPlayer->OpenSource(mMediaSource))
162 {
163 return;
164 }
165
166 mOpenRequested = false;
167 UE_LOG(LogRD, Warning, TEXT("TitleBackgroundVideoSubsystem: failed to open title background video: %s"), *VideoPath);
168 }
169
170 UE_LOG(LogRD, Error, TEXT("TitleBackgroundVideoSubsystem: no playable title background video in configured list"));
171}

178void UTitleBackgroundVideoSubsystem::EnsureMediaObjects()
179{
180 // 1) 재생 엔진(MediaPlayer).
181 if (mMediaPlayer == nullptr)
182 {
183 mMediaPlayer = NewObject<UMediaPlayer>(this, TEXT("SharedTitleBackgroundMediaPlayer"));
184#if PLATFORM_WINDOWS || PLATFORM_MAC
185 mMediaPlayer->SetDesiredPlayerName(FName(TEXT("ElectraProtron")));
186#elif PLATFORM_ANDROID
187 // Electra가 자동 선택되면 본 프로젝트의 로컬 H.264 mp4를 "No playable streams"로 거부한다(5.7 실측).
188 // OS MediaPlayer 경로(AndroidMedia)가 로컬 파일 재생에 검증돼 있어 명시 지정한다.
189 mMediaPlayer->SetDesiredPlayerName(FName(TEXT("AndroidMedia")));
190#endif
191 }
192 if (mMediaPlayer != nullptr)
193 {
194 mMediaPlayer->PlayOnOpen = true;
195 mMediaPlayer->SetLooping(true);
196 // AddUnique: 중복 등록 방지(EnsureMediaObjects가 여러 번 불려도 콜백은 1개로 유지).
197 mMediaPlayer->OnMediaOpened.AddUniqueDynamic(this, &UTitleBackgroundVideoSubsystem::HandleMediaOpened);
198 mMediaPlayer->OnMediaOpenFailed.AddUniqueDynamic(this, &UTitleBackgroundVideoSubsystem::HandleMediaOpenFailed);
199 mMediaPlayer->OnEndReached.AddUniqueDynamic(this, &UTitleBackgroundVideoSubsystem::HandleMediaEndReached);
200 }
201
202 // 2) 출력 텍스처(MediaTexture) — 위젯이 브러시로 그릴 대상.
203 if (mMediaTexture == nullptr)
204 {
205 mMediaTexture = NewObject<UMediaTexture>(this, TEXT("SharedTitleBackgroundMediaTexture"));
206 mMediaTexture->AutoClear = false; // 다음 파일 open 중에는 직전 마지막 정지 프레임을 유지한다.
207 mMediaTexture->ClearColor = FLinearColor::Black; // 첫 프레임 전/끊김 시 검은색.
208 }
209 if (mMediaTexture != nullptr)
210 {
211 // 텍스처가 이 플레이어의 프레임을 받도록 연결한 뒤 GPU 리소스를 갱신한다.
212 mMediaTexture->SetMediaPlayer(mMediaPlayer);
213 mMediaTexture->UpdateResource();
214 }
215
216 // 3) 파일 소스(FileMediaSource) — 실제 mp4 경로는 PreloadTitleBackgroundVideo에서 SetFilePath로 지정.
217 if (mMediaSource == nullptr)
218 {
219 mMediaSource = NewObject<UFileMediaSource>(this, TEXT("SharedTitleBackgroundMediaSource"));
220 mMediaSource->PrecacheFile = true; // 파일을 미리 캐시해 첫 재생 지연을 줄인다.
221 }
222}
