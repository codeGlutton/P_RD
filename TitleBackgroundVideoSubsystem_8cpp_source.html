38void UTitleBackgroundVideoSubsystem::PreloadTitleBackgroundVideo(const FString& RelativeContentPath)
39{
40 // 빈 경로면 게임 설정의 기본 영상을 사용.
41 const FString RequestedPath = RelativeContentPath.IsEmpty()
42 ? GetDefaultTitleBackgroundVideoPath()
43 : RelativeContentPath;
44
45 // 같은 영상을 이미 열도록 요청했으면 다시 열지 않는다(GameInstance라 화면 재진입 시 중복 호출될 수 있음).
46 if (mOpenRequested && RequestedPath == mCurrentRelativePath)
47 {
48 return;
49 }
50
51 // MediaPlayer/MediaTexture/MediaSource 3종이 준비돼 있는지 보장(최초 1회 생성·연결).
52 EnsureMediaObjects();
53 if (mMediaPlayer == nullptr || mMediaSource == nullptr)
54 {
55 return;
56 }
57
58 // Content 상대 경로 → 실제 디스크 절대 경로로 해석(에디터/패키지 환경 모두 대응).
59 const FString VideoPath = RDUITexture::ResolveContentFilePath(RequestedPath);
60 if (FPaths::FileExists(VideoPath) == false)
61 {
62 UE_LOG(LogRD, Warning, TEXT("TitleBackgroundVideoSubsystem: title background video missing: %s"), *VideoPath);
63 return;
64 }
65
66 mCurrentRelativePath = RequestedPath;
67 mOpenRequested = true;
68 mMediaOpened = false;
69
70 // PlayOnOpen: 열리는 즉시 자동 재생 / SetLooping: 끝나면 처음으로 반복.
71 mMediaPlayer->PlayOnOpen = true;
72 mMediaPlayer->SetLooping(true);
73 // 이전 재생을 닫아 상태를 초기화하고, 소스에 새 파일 경로를 지정한 뒤 연다.
74 mMediaPlayer->Close();
75 mMediaSource->SetFilePath(VideoPath);
76
77 // OpenSource는 "열기 시작" 비동기 요청 — 성공/실패는 OnMediaOpened/OnMediaOpenFailed 콜백으로 통지된다.
78 if (mMediaPlayer->OpenSource(mMediaSource) == false)
79 {
80 // 즉시 거부된 경우(소스 무효 등) — 요청 플래그를 되돌린다.
81 mOpenRequested = false;
82 UE_LOG(LogRD, Warning, TEXT("TitleBackgroundVideoSubsystem: failed to open title background video: %s"), *VideoPath);
83 }
84 else
85 {
86 // 열기 시작 성공 — PlayOnOpen이 있어도 명시적으로 Play를 한 번 더 호출(안전).
87 mMediaPlayer->Play();
88 }
89}

100void UTitleBackgroundVideoSubsystem::StopTitleBackgroundVideo()
101{
102 if (mMediaPlayer != nullptr)
103 {
104 // 종료 후 콜백이 죽은 this를 건드리지 않도록 먼저 해제한다.
105 mMediaPlayer->OnMediaOpened.RemoveAll(this);
106 mMediaPlayer->OnMediaOpenFailed.RemoveAll(this);
107 mMediaPlayer->Close();
108 }
109 if (mMediaTexture != nullptr)
110 {
111 mMediaTexture->SetMediaPlayer(nullptr);
112 mMediaTexture->UpdateResource();
113 }
114
115 mCurrentRelativePath.Reset();
116 mOpenRequested = false;
117 mMediaOpened = false;
118}

125void UTitleBackgroundVideoSubsystem::EnsureMediaObjects()
126{
127 // 1) 재생 엔진(MediaPlayer).
128 if (mMediaPlayer == nullptr)
129 {
130 mMediaPlayer = NewObject<UMediaPlayer>(this, TEXT("SharedTitleBackgroundMediaPlayer"));
131#if PLATFORM_WINDOWS || PLATFORM_MAC
132 mMediaPlayer->SetDesiredPlayerName(FName(TEXT("ElectraProtron")));
133#elif PLATFORM_ANDROID
134 // Electra가 자동 선택되면 본 프로젝트의 로컬 H.264 mp4를 "No playable streams"로 거부한다(5.7 실측).
135 // OS MediaPlayer 경로(AndroidMedia)가 로컬 파일 재생에 검증돼 있어 명시 지정한다.
136 mMediaPlayer->SetDesiredPlayerName(FName(TEXT("AndroidMedia")));
137#endif
138 }
139 if (mMediaPlayer != nullptr)
140 {
141 mMediaPlayer->PlayOnOpen = true;
142 mMediaPlayer->SetLooping(true);
143 // AddUnique: 중복 등록 방지(EnsureMediaObjects가 여러 번 불려도 콜백은 1개로 유지).
144 mMediaPlayer->OnMediaOpened.AddUniqueDynamic(this, &UTitleBackgroundVideoSubsystem::HandleMediaOpened);
145 mMediaPlayer->OnMediaOpenFailed.AddUniqueDynamic(this, &UTitleBackgroundVideoSubsystem::HandleMediaOpenFailed);
146 }
147
148 // 2) 출력 텍스처(MediaTexture) — 위젯이 브러시로 그릴 대상.
149 if (mMediaTexture == nullptr)
150 {
151 mMediaTexture = NewObject<UMediaTexture>(this, TEXT("SharedTitleBackgroundMediaTexture"));
152 mMediaTexture->AutoClear = true; // 프레임이 없을 때 ClearColor로 비운다.
153 mMediaTexture->ClearColor = FLinearColor::Black; // 첫 프레임 전/끊김 시 검은색.
154 }
155 if (mMediaTexture != nullptr)
156 {
157 // 텍스처가 이 플레이어의 프레임을 받도록 연결한 뒤 GPU 리소스를 갱신한다.
158 mMediaTexture->SetMediaPlayer(mMediaPlayer);
159 mMediaTexture->UpdateResource();
160 }
161
162 // 3) 파일 소스(FileMediaSource) — 실제 mp4 경로는 PreloadTitleBackgroundVideo에서 SetFilePath로 지정.
163 if (mMediaSource == nullptr)
164 {
165 mMediaSource = NewObject<UFileMediaSource>(this, TEXT("SharedTitleBackgroundMediaSource"));
166 mMediaSource->PrecacheFile = true; // 파일을 미리 캐시해 첫 재생 지연을 줄인다.
167 }
168}

176void UTitleBackgroundVideoSubsystem::HandleMediaOpened(FString OpenedUrl)
177{
178 mMediaOpened = true;
179 if (mMediaPlayer != nullptr)
180 {
181 mMediaPlayer->SetLooping(true);
182 mMediaPlayer->Play();
183
184 // 선택된 비디오 트랙 → 그 트랙의 포맷 → 포맷의 해상도 순으로 원본 크기를 조회한다.
185 const int32 VideoTrack = mMediaPlayer->GetSelectedTrack(EMediaPlayerTrack::Video);
186 const int32 VideoFormat = mMediaPlayer->GetTrackFormat(EMediaPlayerTrack::Video, VideoTrack);
187 const FIntPoint VideoDimensions = mMediaPlayer->GetVideoTrackDimensions(VideoTrack, VideoFormat);
188 if (VideoDimensions.X > 0 && VideoDimensions.Y > 0)
189 {
190 mVideoNativeSize = FVector2D(VideoDimensions.X, VideoDimensions.Y);
191 }
192 }
193
194 UE_LOG(LogRD, Display, TEXT("TitleBackgroundVideoSubsystem: media opened: %s"), *OpenedUrl);
195}
