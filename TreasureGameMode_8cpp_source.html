154bool ATreasureGameMode::OpenRewardPresentation()
155{
156 UWorld* World = GetWorld();
157 APlayerController* PlayerController = World != nullptr
158 ? World->GetFirstPlayerController() : nullptr;
159 const URunPersistData* RunPersistData = GetRunPersistData();
160 if (World == nullptr || PlayerController == nullptr || RunPersistData == nullptr)
161 {
162 return false;
163 }
164 const FRoom& CurrentRoom = RunPersistData->GetCurrentRoom();
165 if (CurrentRoom.mType != ERoomType::Treasure)
166 {
167 return false;
168 }
169 const FTreasureRoom& TreasureRoom =
170 static_cast<const FTreasureRoom&>(CurrentRoom);
171 const TArray<FPrimaryAssetId>& ArtifactIds =
172 TreasureRoom.mRewardArtifactDataIds;
173 const bool bHasArtifact = ArtifactIds.IsEmpty() == false;
174 UClass* WidgetClass = bHasArtifact ? mRewardWidgetClass.Get() : mRewardWidgetClassNoArtifact.Get();
175 if (WidgetClass == nullptr)
176 {
177 UE_LOG(LogRD, Warning, TEXT("보물방 공용 보상 WBP 클래스 미설정 (bHasArtifact=%d)"), bHasArtifact);
178 return false;
179 }
180
181 mRewardUIModel = NewObject<URewardUIModel>(this, TEXT("TreasureRewardUIModel"));
182 FRewardUI Reward;
183 Reward.mTitle = LOCTEXT("TreasureRewardTitle", "보상");
184 Reward.mGoldGained = FMath::Max(0, TreasureRoom.mRewardMoney);
185 Reward.mGoldBalance = GetPartyGold() + (mGoldRewardGranted ? 0 : Reward.mGoldGained);
186 mRewardUIModel->SetReward(Reward);
187
188 TArray<FRewardChoiceUI> Choices;
189 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
190 for (const FPrimaryAssetId& ArtifactId : ArtifactIds)
191 {
192 if (mOpened && !mGrantedArtifactIds.Contains(ArtifactId)) continue;
193 FRewardChoiceUI Choice;
194 Choice.mChoiceIndex = Choices.Num();
195 Choice.mKind = ERewardChoiceKind::Artifact;
196 Choice.mSourceAssetId = ArtifactId;
197 Choice.mName = FText::FromName(ArtifactId.PrimaryAssetName);
198 if (AssetManager != nullptr)
199 {
200 if (const UStaticArtifactData* Data =
201 AssetManager->GetPrimaryAssetObject<UStaticArtifactData>(ArtifactId))
202 {
203 if (!Data->mName.IsEmpty())
204 {
205 Choice.mName = Data->mName;
206 }
207 Choice.mIcon = Data->mIcon.LoadSynchronous();
208 Choice.mDescription = GetTreasureArtifactDescription(Data);
209 Choice.mRarityColor = GetTreasureRarityColor(Data->mRarityType);
210 Choice.mRarityName = StaticEnum<ERarityType>() != nullptr
211 ? StaticEnum<ERarityType>()->GetDisplayNameTextByValue(
212 StaticCast<int64>(Data->mRarityType))
213 : FText::GetEmpty();
214 Choice.mRarityLevel = StaticCast<int32>(Data->mRarityType);
215 }
216 }
217 Choices.Add(Choice);
218 }
219 mRewardUIModel->SetRewardChoices(Choices);
220 mRewardUIModel->OnRewardClaimRequested.AddUniqueDynamic(
221 this, &ATreasureGameMode::HandleRewardClaimRequested);
222 mRewardUIModel->OnRewardSelectionRequested.AddUniqueDynamic(
223 this, &ATreasureGameMode::HandleRewardSelectionRequested);
224
225 mRewardWidget = CreateWidget<URewardConcept03Widget>(
226 PlayerController, WidgetClass);
227 if (mRewardWidget == nullptr)
228 {
229 mRewardUIModel = nullptr;
230 return false;
231 }
232 mRewardWidget->BindUIModel(mRewardUIModel);
233 mRewardWidget->OnRewardFlowCompleted.AddUniqueDynamic(
234 this, &ATreasureGameMode::HandleRewardPresentationCompleted);
235 mRewardWidget->AddToViewport(60);
236 mRewardWidget->ResetRewardFlow();
237 // 보물방은 EXP가 없으므로 첫 '다음' 입력을 자동 수행해 상자 단계부터 시작한다.
238 mRewardWidget->AdvanceRewardFlow();
239
240 if (UWorldWidgetSubsystem* WorldWidgetSubsystem =
241 World->GetSubsystem<UWorldWidgetSubsystem>())
242 {
243 if (URDUserWidget* TreasureHUD =
244 WorldWidgetSubsystem->GetHUD<URDUserWidget>())
245 {
246 TreasureHUD->SetVisibility(ESlateVisibility::Collapsed);
247 }
248 }
249 return true;

266 FTreasureUI TreasureUIData;

310 FTreasureItemUI Item;

318 {
319 // 데이터 이름이 비어 있으면 에셋명 폴백 유지
320 if (Data->mName.IsEmpty() == false)
321 {
322 Item.mName = Data->mName;
323 }
324 Item.mIcon = Data->mIcon.LoadSynchronous();
325 Item.mRarityColor = GetTreasureRarityColor(Data->mRarityType);
326 }
327 }
328 TreasureUIData.mItems.Add(Item);
329 }
330
331 mTreasureUIModel->SetTreasure(TreasureUIData);
332}

364void ATreasureGameMode::HandleRewardSelectionRequested(FPrimaryAssetId RewardId)
365{
366 if (mOpened)
367 {
368 if (mRewardUIModel && mGrantedArtifactIds.Contains(RewardId)) mRewardUIModel->ConfirmSelectedReward(RewardId);
369 return;
370 }
371 auto Reject = [this, RewardId]()
372 {
373 if (mRewardUIModel) mRewardUIModel->OnRewardSelectionRejected.Broadcast(RewardId);
374 };
375 const URunPersistData* Run = GetRunPersistData();
376 if (!Run || Run->GetCurrentRoom().mType != ERoomType::Treasure)
377 {
378 Reject();
379 return;
380 }
381 const FTreasureRoom& Room = static_cast<const FTreasureRoom&>(Run->GetCurrentRoom());
382 FPrimaryAssetId SelectedId;
383 if (!ArtifactRewardPolicy::TrySelectOne(Room.mRewardArtifactDataIds, RewardId, SelectedId))
384 {
385 Reject();
386 return;
387 }
388 UPartyModel* Party = GetPartyModel();
389 UPartyArtifactComponentModel* Artifacts = Party ? Party->GetPartyArtifactComponentModel() : nullptr;
390 if (!Artifacts || !GrantTreasureGold())
391 {
392 Reject();
393 return;
394 }
395 // 지급 중 재진입도 차단한다. 실패하면 선택 화면에서 다시 시도할 수 있다.
396 mOpened = true;
397 const FRewardGrantBundleResultUI Result = ArtifactRewardPolicy::GrantOne(SelectedId,
398 [Artifacts](const FPrimaryAssetId& Id) { return Artifacts->AddArtifact(Id); });
399 if (Result.mGrantedItemIds.IsEmpty())
400 {
401 mOpened = false;
402 Reject();
403 return;
404 }
405 mGrantedArtifactIds = Result.mGrantedItemIds;
406 mFailedArtifactIds.Reset();
407 FRoomTransactionState& Transactions = GetRunPersistData()->GetRoomTransactionsMutable();
408 Transactions.TreasureOpened = true;
409 Transactions.SelectedArtifact = SelectedId;
410 GetGameInstance()->GetSubsystem<USaveGameSubsystem>()->RequestRunAutosave();
411 PushTreasureUIData();
412 if (mRewardUIModel) mRewardUIModel->ConfirmSelectedReward(SelectedId);
413}
