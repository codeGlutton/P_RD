19TArray<TInstancedStruct<FSRPGCommand>> USRPGEnemyTurnPlanner::PlanTurn(
20 UEnemyUnitModel* Enemy,
21 UUnitModel* Player,
22 const UTileMapModel* TileMap,
23 const FRandomStream& EventStream)
24{
25 TArray<TInstancedStruct<FSRPGCommand>> Commands;
26 {
27 // 종료 커맨드는 무조건 필요하므로 일단 추가하고 시작
28 TInstancedStruct<FSRPGCommand> TurnEnd;
29 TurnEnd.InitializeAs<FSRPGTurnEndCommand>();
30 Commands.Add(MoveTemp(TurnEnd));
31 }
32
33 // @note 종료커맨드를 추가하고 시작했으므로 이후 커맨드는 종료커맨드 앞에 넣기 위해 가드 설정.
34 // 가드 없으면 실수로 맨 뒤에 넣을 수 있으니까.
35 auto AddAction = [&Commands](TInstancedStruct<FSRPGCommand>&& Command)
36 {
37 Commands.Insert(MoveTemp(Command), Commands.Num() - 1);
38 };
39
40 // 가드: 필수 입력이 없으면 행동 없이 턴만 종료
41 if (Enemy == nullptr || Player == nullptr || TileMap == nullptr)
42 {
43 return Commands;
44 }
45
46 // 스킬 컴포넌트 확인: 없으면 할 게 없으므로 턴 종료
47 USkillComponentModel* SkillComp = Enemy->GetSkillComponentModel();
48 if (SkillComp == nullptr)
49 {
50 return Commands;
51 }
52
53 // 장착된 스킬 슬롯 인덱스 수집
54 // @note 스킬 슬롯은 고정 크기로 미리 확보되므로, 개수가 아니라 슬롯의 데이터 유무로 판단
55 const TArray<FSkillEntry>& Skills = SkillComp->GetSkills();
56 TArray<int32> EquippedIndexes;
57 for (int32 Index = 0; Index < Skills.Num(); ++Index)
58 {
59 if (Skills[Index].mData != nullptr)
60 {
61 EquippedIndexes.Add(Index);
62 }
63 }
64 // 모든 슬롯이 비어있으면(스킬 미장착) 할 게 없으므로 턴 종료
65 if (EquippedIndexes.Num() == 0)
66 {
67 return Commands;
68 }
69
70 // 사용할 스킬을 랜덤으로 하나 선택. 이후 이동/시전 판단은 이 스킬의 사거리 기준
71 // @note 시뮬/라이브 동일 결과 보장을 위해 반드시 룸의 이벤트 스트림에서 뽑아야 함
72 const int32 SkillIndex = EquippedIndexes[EventStream.RandRange(0, EquippedIndexes.Num() - 1)];
73 const UStaticSkillData* Skill = Skills[SkillIndex].mData;
74
75 // 스킬이 여러 개일 때만 어떤 스킬이 뽑혔는지 확인용 로그
76 if (EquippedIndexes.Num() >= 2)
77 {
78 UE_LOG(LogRD, Log, TEXT("적 AI 스킬 랜덤 선택: %s(ID=%d), 후보 %d개 중 슬롯 %d (%s)"),
79 *GetNameSafe(Enemy), Enemy->GetModelId(), EquippedIndexes.Num(), SkillIndex, *GetNameSafe(Skill));
80 }
81
82 const FTileIndex Origin = Enemy->GetTileTransform().mIndex;
83 const FTileIndex PlayerTile = Player->GetTileTransform().mIndex;
84
85 // 스폰 데이터(DA)에서 로드된 이동포인트만큼 이동 가능
86 const int32 MoveRange = FMath::Max(Enemy->GetMovePoint(), 0);
87
88 // 조준거리
89 // @note 몹은 주사위가 없어 DiceSum=0 → 사거리는 기본값(mAimRangeDefaultValue) 그대로
90 const int32 AimRange = Skill->mAimRangeDefaultValue;
91
92 // 목적지 결정 (이동 성향 기반)
93 bool CanCast = false;
94 const FTileIndex Dest = ChooseDestination(Origin, PlayerTile, MoveRange, AimRange, Skill, TileMap, Enemy->GetMoveTendency(), Enemy, OUT CanCast);
95
101 if (Dest != Origin)
102 {
103 TArray<FTileIndex> Path = TileMap->FindPath(Origin, Dest);
104 // 경로는 출발지와 목적지가 포함되므로 2 이상이어야 실제 이동 가능.
105 if (Path.Num() >= 2)
106 {
107 // 이동커맨드 생성
108 TInstancedStruct<FSRPGCommand> Move;
109 Move.InitializeAs<FSRPGMoveCommand>();
110 Move.GetMutable<FSRPGMoveCommand>().mPathTileIndexes = MoveTemp(Path);
111 AddAction(MoveTemp(Move));
112 }
113 }
114
121 if (CanCast == true)
122 {
123 // 시전 커맨드에는 타겟 타일과 주사위 합만 담는다.
124 // 실제 효과 타일은 실행 시점에 스킬 컴포넌트가 목적지(시전 원점) 기준으로 계산한다.
125 TInstancedStruct<FSRPGCommand> Cast;
126 Cast.InitializeAs<FSRPGSkillCastCommand>();
127 FSRPGSkillCastCommand& CastRef = Cast.GetMutable<FSRPGSkillCastCommand>();
128 CastRef.mSkillIndex = SkillIndex;
129 // 타겟은 플레이어 위치
130 CastRef.mTargetIndex = PlayerTile;
131 // 몹은 주사위가 없다고 가정
132 CastRef.mDiceSum = 0;
133 AddAction(MoveTemp(Cast));
134 }
135
136 return Commands;
137}

139FTileIndex USRPGEnemyTurnPlanner::ChooseDestination(
140 const FTileIndex& Origin,
141 const FTileIndex& PlayerTile,
142 int32 MoveRange,
143 int32 AimRange,
144 const UStaticSkillData* Skill,
145 const UTileMapModel* TileMap,
146 EMoveTendency Tendency,
147 const UBoardActorModel* Self,
148 OUT bool& OutCanCast)
149{
150 // 도달 가능한 모든 타일 집합 (현재 위치한 타일도 포함)
151 TArray<FTileIndex> Candidates = TileMap->GetReachableTiles(Origin, MoveRange);
152 Candidates.Add(Origin);
153
154 // 이동 후 플레이어를 조준 가능한 타일만 추림
155 // @note 자기 자신(Self)은 이동으로 자리를 비울 예정이므로 시야 차폐에서 제외.
156 // 제외하지 않으면 직선 후퇴 타일이 전부 자기 몸에 막혀 제자리 사격이 됨.
157 TArray<FTileIndex> Feasible;
158 for (const FTileIndex& Candidate : Candidates)
159 {
160 TArray<FTileIndex> Aimable = TileMap->GetAimableTiles(Candidate, AimRange, Skill->mAimPattern, Skill->mCanAimBoardActor, Skill->mIsIndirect, /*Incoming*/nullptr, /*IgnoreBlocker*/Self);
161 if (Aimable.Contains(PlayerTile) == true)
162 {
163 Feasible.Add(Candidate);
164 }
165 }
166
167 // 이동 성향에 따라 타일 탐색
168 switch (Tendency)
169 {
170 case EMoveTendency::MoveClose:
171 // 1) 조준 가능한 타일이 있다면
172 if (Feasible.Num() > 0)
173 {
174 // 조준 가능한 타일이 있으니까 스킬 시전도 가능하게 설정
175 OutCanCast = true;
176 // 조준 가능한 타일들 중에서 플레이어와 가장 가까운 타일 선택
177 return PickByPlayerDistance(Feasible, Origin, PlayerTile, /*Closest*/true);
178 }
179 // 2) 조준 가능한 타일이 없다면
180 else
181 {
182 // 조준 가능한 타일이 없음 -> 어느 타일에서도 플레이어를 조준할 수 없음 -> 스킬 시전 불가능 설정
183 OutCanCast = false;
184 // '도달' 가능한 타일들 중에서 경로 거리 기준으로 플레이어와 가장 가까워지는 타일 선택
185 return PickApproachTile(Candidates, Origin, PlayerTile, TileMap, Self);
186 }
187 case EMoveTendency::MoveAway:
188 // 1) 조준 가능한 타일이 있다면
189 if (Feasible.Num() > 0)
190 {
191 // 조준 가능한 타일이 있으니까 스킬 시전도 가능하게 설정
192 OutCanCast = true;
193 // 조준 가능한 타일들 중에서 플레이어와 가장 먼 타일 선택
194 return PickByPlayerDistance(Feasible, Origin, PlayerTile, /*Closest*/false);
195 }
196 else
197 {
198 // 조준 가능한 타일이 없음 -> 어느 타일에서도 플레이어를 조준할 수 없음 -> 스킬 시전 불가능 설정
199 OutCanCast = false;
200 // '도달' 가능한 타일들 중에서 경로 거리 기준으로 플레이어와 가장 가까워지는 타일 선택 -> 다음번에 조준 가능성을 높임
201 return PickApproachTile(Candidates, Origin, PlayerTile, TileMap, Self);
202 }
203 case EMoveTendency::HoldRange:
204 default:
205 // 1) 이미 조준할 수 있다면 제자리 유지
206 if (Feasible.Contains(Origin) == true)
207 {
208 OutCanCast = true;
209 return Origin;
210 }
211 // 2) 제자리에서 조준할 수 없다면 -> 이동거리가 가장 작은 조준 가능한 타일이 최선
212 if (Feasible.Num() > 0)
213 {
214 OutCanCast = true;
215 return PickByMoveCost(Feasible, Origin);
216 }
217 // 3) 조준할 수 있는 타일이 없다면 -> 도달 가능한 타일들 중에서 경로 거리 기준으로 가장 가까워지는 타일이 최선
218 else
219 {
220 OutCanCast = false;
221 return PickApproachTile(Candidates, Origin, PlayerTile, TileMap, Self);
222 }
223 }

226FTileIndex USRPGEnemyTurnPlanner::PickByPlayerDistance(
227 const TArray<FTileIndex>& Tiles,
228 const FTileIndex& Origin,
229 const FTileIndex& PlayerTile,
230 bool Closest)
231{
232 FTileIndex Best = Origin;
233 bool HasBest = false;
234 int32 BestPlayerDist = 0;
235 int32 BestMoveCost = 0;
236 for (const FTileIndex& Tile : Tiles)
237 {
238 // 해당 타일과 플레이어 사이의 거리
239 const int32 PlayerDist = TileDistance(Tile, PlayerTile);
240 // 해당 타일과 적 사이의 거리
241 const int32 MoveCost = TileDistance(Tile, Origin);
242
243 bool Better = false;
244
248 // 1) 아직 최선 타일이 없으면 이 타일이 바로 최선!
249 if (HasBest == false)
250 {
251 Better = true;
252 }
253 // 2) 최선 타일이 있다면, 1순위 비교
254 else if (PlayerDist != BestPlayerDist)
255 {
256 Better = Closest
257 // 근거리를 좋아하면, 플레이어와의 거리가 가까우면 최선
258 ? (PlayerDist < BestPlayerDist)
259 // 원거리를 좋아하면, 플레이어와의 거리가 멀어지면 최선
260 : (PlayerDist > BestPlayerDist);
261 }
262 // 3) 1순위가 같다면, 2순위 비교
263 else
264 {
265 // 2순위는 이동거리가 작으면 최선
266 Better = (MoveCost < BestMoveCost);
267 }
268
269 // 신규 최선 타일이 있다면 그걸 최선으로 설정해서 리턴할 때 사용
270 if (Better == true)
271 {
272 Best = Tile;
273 BestPlayerDist = PlayerDist;
274 BestMoveCost = MoveCost;
275 HasBest = true;
276 }
277 }
278 return Best;
279}

281FTileIndex USRPGEnemyTurnPlanner::PickByMoveCost(
282 const TArray<FTileIndex>& Tiles,
283 const FTileIndex& Origin)
284{
285 FTileIndex Best = Origin;
286 bool HasBest = false;
287 int32 BestMoveCost = 0;
288 for (const FTileIndex& Tile : Tiles)
289 {
290 const int32 MoveCost = TileDistance(Tile, Origin);
291 // 최선 타일이 없으면 이게 최선!
292 // 최선 타일이 있으면 -> 이동비용이 작으면 최선
293 if (HasBest == false || MoveCost < BestMoveCost)
294 {
295 Best = Tile;
296 BestMoveCost = MoveCost;
297 HasBest = true;
298 }
299 }
300 return Best;
301}

303FTileIndex USRPGEnemyTurnPlanner::PickApproachTile(
304 const TArray<FTileIndex>& Tiles,
305 const FTileIndex& Origin,
306 const FTileIndex& PlayerTile,
307 const UTileMapModel* TileMap,
308 const UBoardActorModel* Self)
309{
310 // 플레이어 기준 경로 거리 표 (자기 자신은 자리를 비울 예정이므로 통과 판정에서 제외)
311 const TArray<int32> DistanceField = TileMap->GetDistanceField(PlayerTile, Self);
312
313 FTileIndex Best = Origin;
314 bool HasBest = false;
315 int32 BestPathDist = 0;
316 int32 BestMoveCost = 0;
317 for (const FTileIndex& Tile : Tiles)
318 {
319 // 해당 타일과 플레이어 사이의 경로 거리
320 const int32 Linear = TileMap->TileIndexToLinearIndex(Tile);
321 const int32 RawDist = (Linear != INDEX_NONE) ? DistanceField[Linear] : INDEX_NONE;
322 const int32 PathDist = (RawDist >= 0) ? RawDist : TNumericLimits<int32>::Max();
323 // 해당 타일과 적 사이의 거리
324 const int32 MoveCost = TileDistance(Tile, Origin);
325
326 bool Better = false;
327
331 // 1) 아직 최선 타일이 없으면 이 타일이 바로 최선!
332 if (HasBest == false)
333 {
334 Better = true;
335 }
336 // 2) 최선 타일이 있다면, 1순위 비교: 경로 거리가 가까우면 최선
337 else if (PathDist != BestPathDist)
338 {
339 Better = (PathDist < BestPathDist);
340 }
341 // 3) 1순위가 같다면, 2순위 비교
342 else
343 {
344 // 2순위는 이동거리가 작으면 최선 (동률이면 제자리 우선 -> 의미 없는 이동 방지)
345 Better = (MoveCost < BestMoveCost);
346 }
347
348 // 신규 최선 타일이 있다면 그걸 최선으로 설정해서 리턴할 때 사용
349 if (Better == true)
350 {
351 Best = Tile;
352 BestPathDist = PathDist;
353 BestMoveCost = MoveCost;
354 HasBest = true;
355 }
356 }
357 return Best;
358}
