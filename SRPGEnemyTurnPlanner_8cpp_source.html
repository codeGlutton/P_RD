31 // Evaluate actual effect footprints, including empty aimed tiles and each phase's team filter.
32 // This is geometric coverage, not damage simulation; it consumes no gameplay randomness.
33 int32 CountCoveredTargets(const UTileMapModel* Map, const UEnemyUnitModel* Enemy,
34 const UStaticUnitSkillData* Skill, const FTileIndex& From, const FTileIndex& Aim,
35 const TArray<const UUnitModel*>& Targets)
36 {
37 TArray<FTileIndex> EffectTiles;
38 for (const FTileIndex& Center : Map->GetTargetTiles(From, Aim, Skill->mTargetPattern))
39 {
40 for (const FTileIndex& Tile : Map->GetEffectTiles(Center, Skill->mEffectPattern,
41 Skill->mEffectArea, static_cast<ETileLayerFlag>(Skill->mEffectBlockerMask), Enemy))
42 {
43 EffectTiles.AddUnique(Tile);
44 }
45 }
46 TSet<const UUnitModel*> Covered;
47 for (const FSkillPhaseLayer& Phase : Skill->mSkillPhaseLayers)
48 {
49 const auto Tiles = Phase.FilterTileIndexes(From, EffectTiles);
50 const auto CombatTargets = Phase.FilterCombatTargets(Map, Enemy, Tiles);
51 for (const UUnitModel* Target : Targets)
52 {
53 if (CombatTargets.Contains(const_cast<UUnitModel*>(Target))) Covered.Add(Target);
54 }
55 }
56 return Covered.Num();
57 }

59 bool ChooseMostTargets(const FTacticalTileTable& Table, const UTileMapModel* Map,
60 const UEnemyUnitModel* Enemy, const UStaticUnitSkillData* Skill,
61 const TArray<const UUnitModel*>& Targets, int32 ActionPoint,
62 TMap<FTileIndex, FTileIndex>& BestAims)
63 {
64 int32 BestCount = 0;
65 for (const FTacticalTileInfo& Tile : Table.GetTacticalTiles())
66 {
67 if (Tile.mMoveCost + Skill->mRequiredActionPoint > ActionPoint) continue;
68 const auto AimTiles = Map->GetAimableTiles(Tile.mIndex, Skill->mAimRange,
69 Skill->mAimPattern, Skill->mCanAimBoardActor,
70 static_cast<ETileLayerFlag>(Skill->mAimBlockerMask), nullptr, Enemy);
71 for (const FTileIndex& Aim : AimTiles)
72 {
73 const int32 Count = CountCoveredTargets(Map, Enemy, Skill, Tile.mIndex, Aim, Targets);
74 if (Count == 0 || Count < BestCount) continue;
75 if (Count > BestCount)
76 {
77 BestCount = Count;
78 BestAims.Reset();
79 }
80 // Stable first aim per destination; movement tendency breaks destination ties.
81 if (!BestAims.Contains(Tile.mIndex)) BestAims.Add(Tile.mIndex, Aim);
82 }
83 }
84 return BestCount > 0;
85 }

99 void LogNoCastDetails(
100 const FString& LogPrefix,
101 const FTileIndex& EnemyTile,
102 int32 ActionPoint,
103 EMoveTendency Tendency,
104 int32 ChosenSkillSlot,
105 const UStaticUnitSkillData* ChosenSkill,
106 const TArray<const UUnitModel*>& TargetModels,
107 const TArray<FTileIndex>& TargetTiles,
108 const FTacticalTileTable& Table)
109 {
110 // 계획
111 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 계획: @(%d,%d) AP=%d 성향=%s 타겟수=%d"),
112 *LogPrefix, EnemyTile.mX, EnemyTile.mY, ActionPoint,
113 *StaticEnum<EMoveTendency>()->GetNameStringByValue(static_cast<int64>(Tendency)),
114 TargetTiles.Num());
115
116 // 확정 스킬 정보
117 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 스킬[%d]=%s 비용=%d 사거리=%d 패턴=%s %s"),
118 *LogPrefix, ChosenSkillSlot, *ChosenSkill->GetName(), ChosenSkill->mRequiredActionPoint, ChosenSkill->mAimRange,
119 *StaticEnum<EAimPattern>()->GetNameStringByValue(static_cast<int64>(ChosenSkill->mAimPattern)),
120 (ChosenSkill->mAimBlockerMask == 0) ? TEXT("곡사") : TEXT("직사"));
121
122 // 타겟별 시전불가 사유 탐색 (왜 이 타겟을 때리지 못했나)
123 for (int32 TargetIndex = 0; TargetIndex < TargetTiles.Num(); ++TargetIndex)
124 {
125 const int32 Distance = Table.GetDistanceToTarget(TargetIndex);
126 const FString DistanceText = (Distance == MAX_int32) ? FString(TEXT("도달불가")) : FString::FromInt(Distance);
127 const FString TargetLabel = MakeUnitLabel(TargetModels[TargetIndex]);
128
129 // 조준가능한 타일 중 최소행동력 탐색 (없으면: 조준이 안 됨, 있으면: 행동력 부족)
130 int32 MinNeed = MAX_int32;
131 int32 MinMoveCost = 0;
132 FTileIndex MinTile = FTileIndex::Invalid;
133 for (const FTacticalTileInfo& Tile : Table.GetTacticalTiles())
134 {
135 if (Table.IsAimable(Tile, ChosenSkillSlot, TargetIndex) == false)
136 {
137 continue;
138 }
139
140 const int32 Need = Tile.mMoveCost + ChosenSkill->mRequiredActionPoint;
141 if (Need < MinNeed)
142 {
143 MinNeed = Need;
144 MinMoveCost = Tile.mMoveCost;
145 MinTile = Tile.mIndex;
146 }
147 }
148
149 if (MinNeed == MAX_int32)
150 {
151 // 도달가능한 어떤 타일에서도 조준 불가
152 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 타겟[%d]=%s@(%d,%d) 경로거리=%s → 시전불가: 조준가능타일 없음(사거리/시야 밖)"),
153 *LogPrefix, TargetIndex, *TargetLabel, TargetTiles[TargetIndex].mX, TargetTiles[TargetIndex].mY, *DistanceText);
154 }
155 else if (MinNeed <= ActionPoint)
156 {
157 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 타겟[%d]=%s → 조준/AP 가능하지만 대상 정책 또는 효과 범위 조건으로 시전하지 않음"),
158 *LogPrefix, TargetIndex, *TargetLabel);
159 }
160 else
161 {
162 // 조준가능하지만 행동력 부족
163 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 타겟[%d]=%s@(%d,%d) 경로거리=%s → 시전불가: 최소소요 = 이동%d(→(%d,%d)) + 시전%d = %d > AP%d"),
164 *LogPrefix, TargetIndex, *TargetLabel, TargetTiles[TargetIndex].mX, TargetTiles[TargetIndex].mY, *DistanceText,
165 MinMoveCost, MinTile.mX, MinTile.mY, ChosenSkill->mRequiredActionPoint, MinNeed, ActionPoint);
166 }
167 }
168 }

171TArray<TInstancedStruct<FSRPGCommand>> USRPGEnemyTurnPlanner::PlanTurn(
172 UEnemyUnitModel* Enemy,
173 const TArray<UUnitModel*>& Players,
174 const UTileMapModel* TileMap,
175 const FRandomStream& EventStream,
176 const FString& LogTag)
177{
178 TArray<TInstancedStruct<FSRPGCommand>> Commands;
179 {
180 // 종료 커맨드는 무조건 필요하므로 일단 추가하고 시작
181 TInstancedStruct<FSRPGCommand> TurnEnd;
182 TurnEnd.InitializeAs<FSRPGTurnEndCommand>();
183 Commands.Add(MoveTemp(TurnEnd));
184 }
185
186 // @note 종료커맨드를 추가하고 시작했으므로 이후 커맨드는 종료커맨드 앞에 넣기 위해 가드 설정.
187 // 가드 없으면 실수로 맨 뒤에 넣을 수 있으니까.
188 auto AddAction = [&Commands](TInstancedStruct<FSRPGCommand>&& Command)
189 {
190 Commands.Insert(MoveTemp(Command), Commands.Num() - 1);
191 };
192
193 // 가드: 필수 입력이 없으면 행동 없이 턴만 종료
194 if (Enemy == nullptr || TileMap == nullptr)
195 {
196 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("[%s] 결정: 행동없음 — 적 또는 타일맵 없음"), *LogTag);
197 return Commands;
198 }
199
200 // 로그 헤더 문자열: [라운드/턴][유닛키#모델ID]
201 const FString LogPrefix = FString::Printf(TEXT("%s[%s]"),
202 LogTag.IsEmpty() ? TEXT("") : *FString::Printf(TEXT("[%s]"), *LogTag),
203 *MakeUnitLabel(Enemy));
204
205 // 타겟 타일 수집
206 TArray<FTileIndex> TargetTiles;
207 TArray<const UUnitModel*> TargetModels;
208 for (const UUnitModel* Player : Players)
209 {
210 if (Player != nullptr &&
211 (!Player->GetAttributeComponentModel() || !Player->GetAttributeComponentModel()->HasMatchingGameplayTag(
212 EffectTags::GameplayEffect_ActorState_Dead)))
213 {
214 TargetTiles.Add(Player->GetTileTransform().mIndex);
215 TargetModels.Add(Player);
216 }
217 }
218 // 타겟이 하나도 없으면 할 게 없으므로 턴 종료
219 if (TargetTiles.IsEmpty())
220 {
221 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 행동없음 — 타겟 없음"), *LogPrefix);
222 return Commands;
223 }
224
225 // 스킬 컴포넌트 확인: 없으면 할 게 없으므로 턴 종료
226 USkillComponentModel* SkillComp = Enemy->GetSkillComponentModel();
227 if (SkillComp == nullptr)
228 {
229 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 행동없음 — 스킬 컴포넌트 없음"), *LogPrefix);
230 return Commands;
231 }
232
233 // 스킬 확정: 배치를 보기 전에 우선순위와 사용 가능 여부만으로 이번 턴의 스킬을 정함
234 // (이후 배치 판단은 이 스킬 하나로만 하고, 시전이 안 돼도 다른 스킬로 바꾸지 않음)
235 const TArray<FSkillEntry>& Skills = SkillComp->GetSkills();
236 const int32 ChosenSkillSlot = ChooseSkillByPriority(Enemy, SkillComp, EventStream);
237 // 사용 가능한 스킬이 없으면 할 게 없으므로 턴 종료
238 if (ChosenSkillSlot == INDEX_NONE)
239 {
240 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 행동없음 — 사용가능 스킬 없음(전부 쿨다운/행동력 부족 또는 미장착)"), *LogPrefix);
241 return Commands;
242 }
243
244 // 로그 작성
245
246 UEventLogger* Logger = GetWorldEventLogger(SkillComp);
247 if (Logger != nullptr)
248 {
249 FSRPGAIPlanLog Log;
250 Log.mSkillIndex = ChosenSkillSlot;
251
252 Logger->LogAIPlan(Log);
253 }
254
255 const UStaticUnitSkillData* ChosenSkill = StaticCast<const UStaticUnitSkillData*>(Skills[ChosenSkillSlot].mData);
256 const ESkillPriority ChosenPriority = Enemy->GetSkillPriority(ChosenSkillSlot);
257 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 스킬확정: 스킬[%d]=%s 우선순위=%s"),
258 *LogPrefix, ChosenSkillSlot, *ChosenSkill->GetName(),
259 *StaticEnum<ESkillPriority>()->GetNameStringByValue(static_cast<int64>(ChosenPriority)));
260
261 // 판단근거 로그: 나머지 슬롯이 밀린 이유 (쿨다운 / 행동력 부족 / 기절 / 우선순위 낮음 / 동순위 추첨 탈락)
262 {
263 const UUnitSkillComponentModel* UnitSkillComp = Cast<UUnitSkillComponentModel>(SkillComp);
264 for (int32 Slot = 0; Slot < Skills.Num(); ++Slot)
265 {
266 if (Slot == ChosenSkillSlot || Skills[Slot].IsValid() == false)
267 {
268 continue;
269 }
270
271 const TCHAR* Reason = nullptr;
272 if (SkillComp->IsCooldown(Slot) == true)
273 {
274 Reason = TEXT("쿨다운");
275 }
276 else if (UnitSkillComp != nullptr && UnitSkillComp->HasRequiredActionPoint(Slot) == false)
277 {
278 Reason = TEXT("행동력 부족");
279 }
280 else if (SkillComp->CanActiveSkill(Slot) == false)
281 {
282 Reason = TEXT("기절 등 사용 불가");
283 }
284 else if (Enemy->GetSkillPriority(Slot) == ChosenPriority)
285 {
286 Reason = TEXT("동순위 추첨 탈락");
287 }
288 else
289 {
290 Reason = TEXT("우선순위 낮음");
291 }
292
293 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 스킬[%d]=%s 우선순위=%s → 제외: %s"),
294 *LogPrefix, Slot, *Skills[Slot].mData->GetName(),
295 *StaticEnum<ESkillPriority>()->GetNameStringByValue(static_cast<int64>(Enemy->GetSkillPriority(Slot))),
296 Reason);
297 }
298 }
299
300 // 속성 컴포넌트 확인: 없으면 할 게 없으므로 턴 종료
301 UAttributeSetComponentModel* AttributeSetComp = Enemy->GetAttributeComponentModel();
302 if (AttributeSetComp == nullptr)
303 {
304 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 행동없음 — 속성 컴포넌트 없음"), *LogPrefix);
305 return Commands;
306 }
307
308 // 사용 가능한 액션포인트
309 const int32 ActionPoint = FMath::Max(
310 AttributeSetComp->GetAttributeCurrentValue(UUnitAttributeSet::GetActionPointAttribute()),
311 0
312 );
313
314 const FTileIndex EnemyTile = Enemy->GetTileTransform().mIndex;
315
316 // 속박 등으로 이동 불가면 이동 예산을 0으로 -> 도달 범위가 제자리로 줄어 제자리 스킬만 계획
317 // (시전 예산은 ActionPoint를 따로 넘기므로 이동 불가여도 스킬 비용 판정에는 영향 없음)
318 const UUnitMovementComponentModel* MovementCompModel = Cast<UUnitMovementComponentModel>(Enemy->GetBoardMovementComponentModel());
319 const int32 MoveBudget = (MovementCompModel != nullptr && MovementCompModel->CanSelfMove() == false)
320 ? 0
321 : ActionPoint;
322
323 // 전술 타일 테이블 구성: 이후 판단은 전부 테이블 조회로 처리
324 // 확정 스킬 하나만 올리되, 슬롯 인덱스를 그대로 쓰기 위해 배열 크기는 슬롯 수 유지
325 // 자기 버프는 조준 판단이 없으므로 비워 두고, 시전 비용을 먼저 뗀 예산으로만 이동
326 const bool bSpell = (ChosenSkill->mSkillType == ESkillType::Spell);
327 TArray<const UStaticUnitSkillData*> SkillDatas;
328 SkillDatas.Init(nullptr, Skills.Num());
329 if (bSpell == false)
330 {
331 SkillDatas[ChosenSkillSlot] = ChosenSkill;
332 }
333 const int32 TableMoveBudget = bSpell
334 ? FMath::Max(MoveBudget - ChosenSkill->mRequiredActionPoint, 0)
335 : MoveBudget;
336 FTacticalTileTable Table;
337 Table.Build(TileMap, Enemy, EnemyTile, TargetTiles, SkillDatas, TableMoveBudget, ActionPoint);
338
339 // 목적지 이동비용 조회 (판단근거 로그용)
340 auto GetTableMoveCost = [&Table](const FTileIndex& Tile)
341 {
342 for (const FTacticalTileInfo& Info : Table.GetTacticalTiles())
343 {
344 if (Info.mIndex == Tile)
345 {
346 return Info.mMoveCost;
347 }
348 }
349 return 0;
350 };
351
352 bool CanCast = false;
353 int32 ChosenTarget = INDEX_NONE;
354 FTileIndex Dest = EnemyTile;
355 FTileIndex AreaAim = FTileIndex::Invalid;
356
357 const bool bTargetPolicyOverrided = Enemy->GetTargetPolicyOverride(ChosenSkillSlot).mIsOverrided;
358 const FEnemyTargetPolicy& ChosenTargetPolicy =
359 bSpell == true ?
360 FEnemyTargetPolicy::Default :
361 (bTargetPolicyOverrided == true ? Enemy->GetTargetPolicyOverride(ChosenSkillSlot).mTargetPolicy : ChosenSkill->mTargetPolicy);
362
363 const bool bChasePreferred = !bSpell && ChosenTargetPolicy.mPriority != EEnemyTargetPriority::MostTargets &&
364 ChosenTargetPolicy.mUnreachableBehavior == EEnemyUnreachableTargetBehavior::ChasePreferred;
365
366 // 이동 판단의 기준 타겟도 DA 우선순위를 사용한다. 자기 버프는 기존 거리 기준 유지.
367 TArray<int32> AllTargets;
368 for (int32 TargetIndex = 0; TargetIndex < TargetTiles.Num(); ++TargetIndex)
369 {
370 AllTargets.Add(TargetIndex);
371 }
372 const int32 ReferenceTarget = ChooseTargetByPriority(
373 bSpell ? FEnemyTargetPolicy() : ChosenTargetPolicy,
374 Table, TargetModels, AllTargets, EventStream);
375
376 // 확정 스킬로 시전 가능한 타겟 후보 수집 (버프는 테이블에 스킬이 없으므로 항상 비어 있음)
377 TArray<int32> CastableTargets;
378 for (int32 TargetIndex = 0; TargetIndex < TargetTiles.Num(); ++TargetIndex)
379 {
380 if (Table.CanCastToTarget(TargetIndex))
381 {
382 CastableTargets.Add(TargetIndex);
383 }
384 }
385
386 TMap<FTileIndex, FTileIndex> BestAreaAims;
387 const bool bAreaCast = !bSpell && ChosenTargetPolicy.mPriority == EEnemyTargetPriority::MostTargets &&
388 ChooseMostTargets(Table, TileMap, Enemy, ChosenSkill, TargetModels, ActionPoint, BestAreaAims);
389 if (bChasePreferred)
390 {
391 // Rank all living targets first, then only permit a cast at that target.
392 CastableTargets.RemoveAll([ReferenceTarget](int32 Index) { return Index != ReferenceTarget; });
393 }
394
395 if (bSpell == true)
396 {
397 //
398 // 자기 버프: 남는 예산으로 이동 성향대로 자리를 잡고 거기서 시전 (도달 가능한 모든 타일이 후보)
399 //
400 Dest = ChooseDestinationByTendency(Table, Enemy->GetMoveTendency(), ReferenceTarget, EnemyTile,
401 [](const FTacticalTileInfo&)
402 {
403 return true;
404 });
405 CanCast = true;
406
407 // 판단근거 로그: 버프 결정
408 if (Dest != EnemyTile)
409 {
410 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 이동+버프: 스킬[%d]=%s 비용%d, 이동 (%d,%d)→(%d,%d) 비용%d (이동예산 %d)"),
411 *LogPrefix, ChosenSkillSlot, *ChosenSkill->GetName(), ChosenSkill->mRequiredActionPoint,
412 EnemyTile.mX, EnemyTile.mY, Dest.mX, Dest.mY, GetTableMoveCost(Dest), TableMoveBudget);
413 }
414 else
415 {
416 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 제자리버프: 스킬[%d]=%s 비용%d (이동예산 %d)"),
417 *LogPrefix, ChosenSkillSlot, *ChosenSkill->GetName(), ChosenSkill->mRequiredActionPoint, TableMoveBudget);
418 }
419 }
420 else if (bAreaCast)
421 {
422 Dest = ChooseDestinationByTendency(Table, Enemy->GetMoveTendency(), ReferenceTarget, EnemyTile,
423 [&BestAreaAims](const FTacticalTileInfo& Tile) { return BestAreaAims.Contains(Tile.mIndex); });
424 AreaAim = BestAreaAims.FindChecked(Dest);
425 CanCast = true;
426 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 밀집공격: 이동(%d,%d) 조준(%d,%d)"),
427 *LogPrefix, Dest.mX, Dest.mY, AreaAim.mX, AreaAim.mY);
428 }
429 else if (CastableTargets.IsEmpty() == false && ChosenTargetPolicy.mPriority != EEnemyTargetPriority::MostTargets)
430 {
431 //
432 // 공격 시전: 시전 가능한 타겟이 있으면 [타겟 -> 목적지] 순서로 확정 (스킬은 이미 확정)
433 //
434 // 높은 우선순위 대상이 사거리/AP 조건 밖이어도 공격 가능한 다른 대상에게 시전한다.
435 ChosenTarget = bChasePreferred ? ReferenceTarget :
436 ChooseTargetByPriority(ChosenTargetPolicy, Table, TargetModels, CastableTargets, EventStream);
437 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 대상우선순위=%s 선택=%s"), *LogPrefix,
438 *StaticEnum<EEnemyTargetPriority>()->GetNameStringByValue(static_cast<int64>(ChosenTargetPolicy.mPriority)),
439 *MakeUnitLabel(TargetModels[ChosenTarget]));
440 // 목적지: 확정 스킬로 그 타겟에게 시전 가능한 타일 중 이동성향대로
441 Dest = ChooseDestinationByTendency(Table, Enemy->GetMoveTendency(), ChosenTarget, EnemyTile,
442 [&Table, ChosenSkillSlot, ChosenTarget](const FTacticalTileInfo& Tile)
443 {
444 return Table.IsCastable(Tile, ChosenSkillSlot, ChosenTarget);
445 });
446 CanCast = true;
447
448 // 판단근거 로그: 시전 턴은 결정 1줄로 요약
449 {
450 const int32 MoveCost = GetTableMoveCost(Dest);
451 const int32 CastCost = ChosenSkill->mRequiredActionPoint;
452 const FString TargetLabel = MakeUnitLabel(TargetModels[ChosenTarget]);
453 if (Dest != EnemyTile)
454 {
455 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 이동+시전: 타겟[%d]=%s@(%d,%d) 스킬[%d]=%s, 이동 (%d,%d)→(%d,%d) 비용%d + 시전%d = %d ≤ AP%d"),
456 *LogPrefix, ChosenTarget, *TargetLabel, TargetTiles[ChosenTarget].mX, TargetTiles[ChosenTarget].mY,
457 ChosenSkillSlot, *ChosenSkill->GetName(),
458 EnemyTile.mX, EnemyTile.mY, Dest.mX, Dest.mY, MoveCost, CastCost, MoveCost + CastCost, ActionPoint);
459 }
460 else
461 {
462 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 제자리시전: 타겟[%d]=%s@(%d,%d) 스킬[%d]=%s, 이동0 + 시전%d = %d ≤ AP%d"),
463 *LogPrefix, ChosenTarget, *TargetLabel, TargetTiles[ChosenTarget].mX, TargetTiles[ChosenTarget].mY,
464 ChosenSkillSlot, *ChosenSkill->GetName(),
465 CastCost, CastCost, ActionPoint);
466 }
467 }
468 }
469 else
470 {
471 //
472 // 공격 불가: 다른 스킬로 바꾸지 않고 이동만 (예상 스킬과 실제 스킬 일치 보장)
473 //
474 // 판단근거 로그: 시전하지 못한 턴은 근거를 상세히 남김
475 LogNoCastDetails(LogPrefix, EnemyTile, ActionPoint, Enemy->GetMoveTendency(), ChosenSkillSlot, ChosenSkill, TargetModels, TargetTiles, Table);
476
477 // Keep legacy Nearest/AttackAvailable movement and RNG behaviour exactly intact.
478 const bool bLegacyMovement = ChosenTargetPolicy.mPriority == EEnemyTargetPriority::Nearest && !bChasePreferred;
479 const auto CanPrepareCast = [&Table, ChosenSkillSlot, ReferenceTarget, bLegacyMovement](const FTacticalTileInfo& Tile)
480 {
481 return bLegacyMovement ? Tile.mAimableFlags.Contains(true) :
482 Table.IsAimable(Tile, ChosenSkillSlot, ReferenceTarget);
483 };
484 if (Table.GetTacticalTiles().ContainsByPredicate(CanPrepareCast))
485 {
486 // 이번 턴에는 못 때리므로, 다음 턴 시전을 노리고 조준 가능한 타일로 이동
487 Dest = ChooseDestinationByTendency(Table, Enemy->GetMoveTendency(), ReferenceTarget, EnemyTile,
488 CanPrepareCast);
489
490 // 판단근거 로그: 선점이동 결정
491 if (Dest != EnemyTile)
492 {
493 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 선점이동 → (%d,%d) 이동%d/AP%d, 다음턴 시전 노림"),
494 *LogPrefix, Dest.mX, Dest.mY, GetTableMoveCost(Dest), ActionPoint);
495 }
496 else
497 {
498 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 제자리대기 — 이동해도 개선 없음"), *LogPrefix);
499 }
500 }
501 else
502 {
503 // 어디로 가든 조준이 안 되면, DA로 선택한 타겟에게 최대한 접근
504 Dest = ChooseApproachDestination(Table, ReferenceTarget, EnemyTile);
505
506 // 판단근거 로그: 접근이동 결정
507 if (Dest != EnemyTile)
508 {
509 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 접근이동 → 타겟[%d]=%s 방향 (%d,%d) 이동%d/AP%d"),
510 *LogPrefix, ReferenceTarget, *MakeUnitLabel(TargetModels[ReferenceTarget]), Dest.mX, Dest.mY, GetTableMoveCost(Dest), ActionPoint);
511 }
512 else
513 {
514 UE_LOG(LogSRPGEnemyPlanner, Log, TEXT("%s 결정: 제자리대기 — 접근 경로 없음"), *LogPrefix);
515 }
516 }
517 }
518
524 if (Dest != EnemyTile)
525 {
526 TArray<FTileIndex> Path = TileMap->FindPath(EnemyTile, Dest);
527 // 경로는 출발지와 목적지가 포함되므로 2 이상이어야 실제 이동 가능.
528 if (Path.Num() >= 2)
529 {
530 // 이동커맨드 생성
531 TInstancedStruct<FSRPGCommand> Move;
532 Move.InitializeAs<FSRPGMoveCommand>();
533 Move.GetMutable<FSRPGMoveCommand>().mPathTileIndexes = MoveTemp(Path);
534 AddAction(MoveTemp(Move));
535 }
536 }
537
544 if (CanCast == true)
545 {
546 // 시전 커맨드에는 조준 타일과 스킬 슬롯만 담는다.
547 // 실제 효과 타일은 실행 시점에 스킬 컴포넌트가 조준 타일 기준으로 계산한다.
548 // Single 패턴은 조준 타일이 시전자 자기 칸이므로 타겟 칸이 아닌 목적지를 넣는다.
549 // 자기 버프는 타겟 없이 시전하므로 역시 목적지를 넣는다.
550 const bool bAimSelf = (ChosenSkill->mAimPattern == EAimPattern::Single) || (ChosenTarget == INDEX_NONE);
551 TInstancedStruct<FSRPGCommand> Cast;
552 Cast.InitializeAs<FSRPGSkillCastCommand>();
553 FSRPGSkillCastCommand& CastRef = Cast.GetMutable<FSRPGSkillCastCommand>();
554 CastRef.mSkillIndex = ChosenSkillSlot;
555 CastRef.mTargetIndex = AreaAim != FTileIndex::Invalid ? AreaAim :
556 (bAimSelf ? Dest : TargetTiles[ChosenTarget]);
557 AddAction(MoveTemp(Cast));
558 }
559
560 return Commands;
561}

563int32 USRPGEnemyTurnPlanner::ChooseSkillByPriority(
564 const UEnemyUnitModel* Enemy,
565 const USkillComponentModel* SkillComp,
566 const FRandomStream& EventStream)
567{
568 // 사용 가능한 슬롯 중 최상위 우선순위 후보 수집 (열거값이 작을수록 높음)
569 const TArray<FSkillEntry>& Skills = SkillComp->GetSkills();
570 ESkillPriority BestPriority = ESkillPriority::Lowest;
571 TArray<int32> Candidates;
572 for (int32 Slot = 0; Slot < Skills.Num(); ++Slot)
573 {
574 if (Skills[Slot].IsValid() == false || SkillComp->CanActiveSkill(Slot) == false)
575 {
576 continue;
577 }
578
579 const ESkillPriority Priority = Enemy->GetSkillPriority(Slot);
580 if (Priority < BestPriority)
581 {
582 // 더 높은 우선순위 발견: 후보 교체
583 BestPriority = Priority;
584 Candidates.Reset();
585 Candidates.Add(Slot);
586 }
587 else if (Priority == BestPriority)
588 {
589 Candidates.Add(Slot);
590 }
591 }
592
593 // 후보가 없으면 INDEX_NONE, 동순위가 여럿일 때만 랜덤
594 if (Candidates.IsEmpty())
595 {
596 return INDEX_NONE;
597 }
598 return (Candidates.Num() == 1)
599 ? Candidates[0]
600 : Candidates[EventStream.RandRange(0, Candidates.Num() - 1)];
601}

603int32 USRPGEnemyTurnPlanner::ChooseTargetByPriority(
604 const FEnemyTargetPolicy& Policy,
605 const FTacticalTileTable& Table,
606 const TArray<const UUnitModel*>& Targets,
607 const TArray<int32>& Candidates,
608 const FRandomStream& EventStream)
609{
610 const EEnemyTargetPriority Priority = Policy.mPriority;
611 if (Candidates.IsEmpty()) return INDEX_NONE;
612 if (Priority == EEnemyTargetPriority::Random)
613 return Candidates.Num() == 1 ? Candidates[0] : Candidates[EventStream.RandRange(0, Candidates.Num() - 1)];
614 if (Priority == EEnemyTargetPriority::Nearest || Priority == EEnemyTargetPriority::MostTargets ||
615 (Policy.IsStatusPolicy() && !Policy.mStatusTag.IsValid()))
616 return ChooseNearestTarget(Table, Candidates, EventStream);
617
618 double BestScore = TNumericLimits<double>::Max();
619 TArray<int32> BestTargets;
620 for (int32 Index : Candidates)
621 {
622 const auto* Attributes = Targets[Index]->GetAttributeComponentModel();
623 if (!Attributes && Priority != EEnemyTargetPriority::Farthest) continue;
624 bool HasValues = true;
625 const auto Read = [Attributes, &HasValues](const FTacticalAttribute& Attribute)
626 {
627 bool Found = false;
628 const float Value = Attributes->GetAttributeCurrentValue(Attribute, Found);
629 HasValues &= Found;
630 return Value;
631 };
632 double Score = 0.;
633 switch (Priority)
634 {
635 case EEnemyTargetPriority::LowestHP:
636 Score = Read(UCombatTargetAttributeSet::GetHPAttribute());
637 break;
638 case EEnemyTargetPriority::LowestHPPercent:
639 {
640 const double MaxHP = Read(UCombatTargetAttributeSet::GetMaxHPAttribute());
641 if (MaxHP <= 0.) continue;
642 Score = Read(UCombatTargetAttributeSet::GetHPAttribute()) / MaxHP;
643 break;
644 }
645 case EEnemyTargetPriority::HighestMaxHP:
646 Score = -Read(UCombatTargetAttributeSet::GetMaxHPAttribute());
647 break;
648 case EEnemyTargetPriority::Farthest:
649 if (Table.GetDistanceToTarget(Index) == MAX_int32) continue;
650 Score = -static_cast<double>(Table.GetDistanceToTarget(Index));
651 break;
652 case EEnemyTargetPriority::WithStatus:
653 Score = Attributes->HasMatchingGameplayTag(Policy.mStatusTag) ? 0. : 1.;
654 break;
655 case EEnemyTargetPriority::WithoutStatus:
656 Score = Attributes->HasMatchingGameplayTag(Policy.mStatusTag) ? 1. : 0.;
657 break;
658 default:
659 return ChooseNearestTarget(Table, Candidates, EventStream);
660 }
661 if (!HasValues || !FMath::IsFinite(Score)) continue;
662 if (Score < BestScore)
663 {
664 BestScore = Score;
665 BestTargets.Reset();
666 BestTargets.Add(Index);
667 }
668 else if (Score == BestScore)
669 {
670 BestTargets.Add(Index);
671 }
672 }
673 // Missing attributes must not stop the turn. Preserve the established seeded distance tie-break.
674 return ChooseNearestTarget(Table, BestTargets.IsEmpty() ? Candidates : BestTargets, EventStream);
675}

677int32 USRPGEnemyTurnPlanner::ChooseNearestTarget(
678 const FTacticalTileTable& Table,
679 const TArray<int32>& CandidateTargets,
680 const FRandomStream& EventStream)
681{
682 // 후보가 없으면 선택할 것도 없음
683 if (CandidateTargets.IsEmpty())
684 {
685 return INDEX_NONE;
686 }
687
688 // 후보 중 가장 짧은 경로 거리 탐색
689 int32 BestDistance = MAX_int32;
690 for (const int32 TargetIndex : CandidateTargets)
691 {
692 BestDistance = FMath::Min(BestDistance, Table.GetDistanceToTarget(TargetIndex));
693 }
694
695 // 최근접 동률 후보 수집
696 TArray<int32> NearestTargets;
697 for (const int32 TargetIndex : CandidateTargets)
698 {
699 if (Table.GetDistanceToTarget(TargetIndex) == BestDistance)
700 {
701 NearestTargets.Add(TargetIndex);
702 }
703 }
704
705 // 동률이 있을때만 랜덤 돌린다.
706 return (NearestTargets.Num() == 1)
707 ? NearestTargets[0]
708 : NearestTargets[EventStream.RandRange(0, NearestTargets.Num() - 1)];
709}

711FTileIndex USRPGEnemyTurnPlanner::ChooseDestinationByTendency(
712 const FTacticalTileTable& Table,
713 EMoveTendency Tendency,
714 int32 ReferenceTarget,
715 const FTileIndex& Origin,
716 TFunctionRef<bool(const FTacticalTileInfo&)> Filter)
717{
718 switch (Tendency)
719 {
720 case EMoveTendency::MoveClose:
721 // 근접 성향: 기준 타겟과의 거리 최소 -> 이동비용 최소
722 return Table.PickTile(
723 Filter,
724 [ReferenceTarget](const FTacticalTileInfo& Tile)
725 {
726 return static_cast<int64>(Tile.mTargetDistances[ReferenceTarget]);
727 },
728 [](const FTacticalTileInfo& Tile)
729 {
730 return static_cast<int64>(Tile.mMoveCost);
731 },
732 Origin);
733
734 case EMoveTendency::MoveAway:
735 // 원거리 성향: 무조건 먼 게 좋은 게 아니라, '최근접' 타겟이 '최대한' 멀어야 함
736 return Table.PickTile(
737 Filter,
738 [&Table](const FTacticalTileInfo& Tile)
739 {
740 return -static_cast<int64>(Table.GetNearestTargetDistance(Tile));
741 },
742 [](const FTacticalTileInfo& Tile)
743 {
744 return static_cast<int64>(Tile.mMoveCost);
745 },
746 Origin);
747
748 case EMoveTendency::HoldRange:
749 default:
750 // 등거리 성향: 최대한 제자리를 유지하려고 하지만, 조준이 안된다면 최근접 타겟과의 거리를 최대로 하는 지점으로 이동
751 return Table.PickTile(
752 Filter,
753 [](const FTacticalTileInfo& Tile)
754 {
755 return static_cast<int64>(Tile.mMoveCost);
756 },
757 [&Table](const FTacticalTileInfo& Tile)
758 {
759 return -static_cast<int64>(Table.GetNearestTargetDistance(Tile));
760 },
761 Origin);
762 }
763}
