233 int32 MakeBrushTexturesResident(UShopUIWidgetBase& ShopWidget)
234 {
235 if (ShopWidget.WidgetTree == nullptr)
236 {
237 return 0;
238 }
239
240 TSet<UTexture2D*> Textures;
241 auto CollectBrush = [&Textures](const FSlateBrush& Brush)
242 {
243 if (UTexture2D* Texture = Cast<UTexture2D>(Brush.GetResourceObject()))
244 {
245 Textures.Add(Texture);
246 }
247 };
248 auto CollectStyle = [&CollectBrush](const FSlateWidgetStyle& Style)
249 {
250 TArray<const FSlateBrush*> Brushes;
251 Style.GetResources(Brushes);
252 for (const FSlateBrush* Brush : Brushes)
253 {
254 if (Brush != nullptr)
255 {
256 CollectBrush(*Brush);
257 }
258 }
259 };
260
261 ShopWidget.WidgetTree->ForEachWidget(
262 [&CollectBrush, &CollectStyle](UWidget* Widget)
263 {
264 if (const UImage* Image = Cast<UImage>(Widget))
265 {
266 CollectBrush(Image->GetBrush());
267 }
268 else if (const UBorder* Border = Cast<UBorder>(Widget))
269 {
270 CollectBrush(Border->Background);
271 }
272 else if (const UButton* Button = Cast<UButton>(Widget))
273 {
274 CollectStyle(Button->GetStyle());
275 }
276 });
277
278 for (UTexture2D* Texture : Textures)
279 {
280 FTextureCompilingManager::Get().FinishCompilation({ Texture });
281 if (Texture->GetResource() == nullptr)
282 {
283 Texture->UpdateResource();
284 }
285 Texture->SetForceMipLevelsToBeResident(30.f);
286 Texture->WaitForStreaming();
287 }
288 FlushRenderingCommands();
289 return Textures.Num();
290 }

292 bool Capture(UShopUIWidgetBase& ShopWidget,
293 const TSharedRef<SWidget>& ShopSlate, const TCHAR* FileName,
294 TArray<FColor>& OutPixels, FString& OutError)
295 {
296 ShopWidget.SetVisibility(ESlateVisibility::SelfHitTestInvisible);
297 ShopWidget.ForceLayoutPrepass();
298 const int32 TextureCount = MakeBrushTexturesResident(ShopWidget);
299 if (TextureCount == 0)
300 {
301 OutError = TEXT("새 상점 WBP 브러시에 Texture2D가 하나도 없음");
302 return false;
303 }
304
305 const TSharedRef<SOverlay> CaptureRoot =
306 SNew(SOverlay)
307 + SOverlay::Slot()
308 [
309 SNew(SColorBlock).Color(FLinearColor(0.008f, 0.012f, 0.018f, 1.f))
310 ]
311 + SOverlay::Slot()
312 [
313 ShopSlate
314 ];
315 FWidgetRenderer Renderer(true, true);
316 Renderer.SetIsPrepassNeeded(true);
317 // 처음 로드한 Slate brush와 Texture2D resource가 각각 한 프레임씩
318 // 늦게 준비될 수 있으므로 캡처 전 두 번 그려 첫 화면도 안정화한다.
319 for (int32 WarmupFrame = 0; WarmupFrame < 6; ++WarmupFrame)
320 {
321 Renderer.DrawWidget(
322 CaptureRoot, FVector2D(CaptureWidth, CaptureHeight));
323 FlushRenderingCommands();
324 }
325 UTextureRenderTarget2D* RenderTarget = Renderer.DrawWidget(
326 CaptureRoot, FVector2D(CaptureWidth, CaptureHeight));
327 if (RenderTarget == nullptr)
328 {
329 OutError = TEXT("FWidgetRenderer가 새 상점 렌더 타깃을 만들지 못함");
330 return false;
331 }
332
333 FlushRenderingCommands();
334 FReadSurfaceDataFlags ReadFlags(RCM_UNorm);
335 ReadFlags.SetLinearToGamma(false);
336 if (!RenderTarget->GameThread_GetRenderTargetResource()->ReadPixels(
337 OutPixels, ReadFlags)
338 || OutPixels.Num() != CaptureWidth * CaptureHeight)
339 {
340 OutError = TEXT("새 상점 2176x1812 렌더 결과를 읽지 못함");
341 return false;
342 }
343
344 int32 PixelsDifferentFromFirst = 0;
345 uint8 MinLuma = 255;
346 uint8 MaxLuma = 0;
347 const FColor FirstPixel = OutPixels[0];
348 for (FColor& Pixel : OutPixels)
349 {
350 // RGBA8 WidgetRenderer 경로의 중복 감마를 한 번 되돌린다.
351 Pixel.R = uint8(FMath::RoundToInt(
352 255.f * FMath::Pow(Pixel.R / 255.f, 2.2f)));
353 Pixel.G = uint8(FMath::RoundToInt(
354 255.f * FMath::Pow(Pixel.G / 255.f, 2.2f)));
355 Pixel.B = uint8(FMath::RoundToInt(
356 255.f * FMath::Pow(Pixel.B / 255.f, 2.2f)));
357
358 const uint8 Luma = uint8((54 * int32(Pixel.R)
359 + 183 * int32(Pixel.G) + 19 * int32(Pixel.B)) >> 8);
360 MinLuma = FMath::Min(MinLuma, Luma);
361 MaxLuma = FMath::Max(MaxLuma, Luma);
362 if (FMath::Abs(int32(Pixel.R) - int32(FirstPixel.R))
363 + FMath::Abs(int32(Pixel.G) - int32(FirstPixel.G))
364 + FMath::Abs(int32(Pixel.B) - int32(FirstPixel.B)) > 12)
365 {
366 ++PixelsDifferentFromFirst;
367 }
368 }
369 if (int32(MaxLuma) - int32(MinLuma) < 16
370 || PixelsDifferentFromFirst < CaptureWidth * CaptureHeight / 100)
371 {
372 OutError = TEXT("새 상점 캡처가 단색이다. WBP가 그려지지 않음");
373 return false;
374 }
375
376 TArray64<uint8> PngData;
377 FImageUtils::PNGCompressImageArray(
378 CaptureWidth, CaptureHeight, OutPixels, PngData);
379 IFileManager::Get().MakeDirectory(*OutputDirectory(), true);
380 const FString OutputPath = FPaths::Combine(OutputDirectory(), FileName);
381 if (!FFileHelper::SaveArrayToFile(PngData, *OutputPath))
382 {
383 OutError = FString::Printf(TEXT("새 상점 캡처 저장 실패: %s"),
384 *OutputPath);
385 return false;
386 }
387
388 UE_LOG(LogTemp, Display,
389 TEXT("[ShopFullGeneratedUI] %s (%d textures) -> %s"),
390 FileName, TextureCount, *OutputPath);
391 return true;
392 }
393
394 int32 CountChangedPixels(const TArray<FColor>& A, const TArray<FColor>& B)
395 {
396 if (A.Num() != B.Num())
397 {
398 return 0;
399 }
400
401 int32 ChangedPixels = 0;
402 for (int32 Index = 0; Index < A.Num(); ++Index)
403 {
404 if (FMath::Abs(int32(A[Index].R) - int32(B[Index].R))
405 + FMath::Abs(int32(A[Index].G) - int32(B[Index].G))
406 + FMath::Abs(int32(A[Index].B) - int32(B[Index].B)) > 12)
407 {
408 ++ChangedPixels;
409 }
410 }
411 return ChangedPixels;
412 }
