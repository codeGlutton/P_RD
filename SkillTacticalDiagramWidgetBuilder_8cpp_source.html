141 void AddRangeButton(UWidgetBlueprint* Blueprint, UCanvasPanel* Parent,

144 UTexture2D* SelectedButtonTexture,

162 // 눌림 상태에서도 라벨의 중앙 좌표를 바꾸지 않는다.
163 Style.SetPressedPadding(FMargin(0.f));
164 Button->SetStyle(Style);
165 Button->SetBackgroundColor(FLinearColor::White);
166 Button->SetTouchMethod(EButtonTouchMethod::PreciseTap);
167 Button->SetClickMethod(EButtonClickMethod::PreciseClick);
168 Button->SetVisibility(ESlateVisibility::Visible);
169 Place(Parent, Button, Position, Size, 300);
170
171 UTextBlock* Text = Blueprint->WidgetTree->ConstructWidget<UTextBlock>(
172 UTextBlock::StaticClass(), FName(*FString::Printf(TEXT("%sText"), Stem)));
173 Text->SetText(TextValue);
174 StyleText(Text, 27, TextColor);
175 Button->SetContent(Text);
176 UButtonSlot* Slot = CastChecked<UButtonSlot>(Text->Slot);
177 Slot->SetHorizontalAlignment(HAlign_Fill);
178 Slot->SetVerticalAlignment(VAlign_Center);
179 Slot->SetPadding(FMargin(0.f));
180 }
