32void UCombatLayoutHUDWidget::BeginCombatResultPresentation(TSharedPtr<FPresentationBarrier> Barrier, bool IsPlayerWin)
33{
34 mIsPlayerWin = IsPlayerWin;
35 mCombatResultBarrier = MoveTemp(Barrier);
36 mVictoryWorldMapLocked = false;
37 mVictoryWorldMap = nullptr;
38 mBossCollapsePlayed = false;
39 mResultRewardsOpened = false;
40 if (mBossCollapseWidget) { mBossCollapseWidget->Cancel(); mBossCollapseWidget = nullptr; }
41 if (UWorld* World = GetWorld()) World->GetTimerManager().ClearTimer(mFinalRunCompletionTimerHandle);
42
43 // 마지막 타격 직후에는 쓰러짐 애니메이션과 디졸브를 먼저 보여 준다.
44 // 배리어를 쥔 채 기다리므로 보상/패배 UI가 전장을 덮지 않는다.
45 if (UWorld* World = GetWorld(); World != nullptr
46 && mDeathAnimationResultDelaySeconds > 0.0f)
47 {
48 World->GetTimerManager().ClearTimer(mCombatResultStartDelayTimerHandle);
49 World->GetTimerManager().SetTimer(mCombatResultStartDelayTimerHandle,
50 this, &UCombatLayoutHUDWidget::StartCombatResultCinematic,
51 mDeathAnimationResultDelaySeconds, false);
52 return;
53 }
54 StartCombatResultCinematic();
55}

74void UCombatLayoutHUDWidget::EnsureCombatResultWidgets()
75{
76 if (mCombatResultOverlayWidget == nullptr)
77 {
78 // 생성자에서 하드 레퍼런스로 든 클래스를 우선 쓴다. 문자열 로드는
79 // 테스트 픽스처 등 클래스가 비어 있는 환경의 폴백이다.
80 UClass* DefeatWidgetClass = mDefeatWidgetClass.Get();
81 if (DefeatWidgetClass == nullptr)
82 {
83 static const TCHAR* DefeatWidgetClassPath =
84 TEXT("/Game/UI/CombatResult/WBP_CombatDefeat.WBP_CombatDefeat_C");
85 DefeatWidgetClass = LoadClass<UCombatResultOverlayWidget>(nullptr, DefeatWidgetClassPath);
86 if (DefeatWidgetClass == nullptr)
87 {
88 UE_LOG(LogRD, Warning, TEXT("Combat defeat WBP is unavailable: %s"), DefeatWidgetClassPath);
89 DefeatWidgetClass = UCombatResultOverlayWidget::StaticClass();
90 }
91 }
92 mCombatResultOverlayWidget = CreateWidget<UCombatResultOverlayWidget>(GetOwningPlayer(), DefeatWidgetClass);
93 }
94
95 if (mCombatRewardUIModel == nullptr)
96 {
97 return;
98 }
99
100 const bool bHasArtifactChoices =
101 mCombatRewardUIModel->GetRewardChoices().Num() > 0;
102 if (mCombatRewardConceptWidget != nullptr
103 && mCombatRewardConceptWidget->HasArtifactReward() != bHasArtifactChoices)
104 {
105 mCombatRewardConceptWidget->RemoveFromParent();
106 mCombatRewardConceptWidget = nullptr;
107 }
108
109 if (mCombatRewardConceptWidget == nullptr)
110 {
111 static const TCHAR* FourStepClassPath =
112 TEXT("/Game/UI/RewardConcept03New/WBP_RewardConcept03_Frameless.WBP_RewardConcept03_Frameless_C");
113 static const TCHAR* ThreeStepClassPath =
114 TEXT("/Game/UI/RewardConcept03New/WBP_RewardConcept03_Frameless_NoArtifact.WBP_RewardConcept03_Frameless_NoArtifact_C");
115 const TCHAR* ClassPath = bHasArtifactChoices
116 ? FourStepClassPath : ThreeStepClassPath;
117 UClass* RewardClass = LoadClass<URewardConcept03Widget>(nullptr, ClassPath);
118 if (RewardClass == nullptr)
119 {
120 UE_LOG(LogRD, Error,
121 TEXT("Combat reward Concept03 WBP is unavailable: %s"), ClassPath);
122 return;
123 }
124 mCombatRewardConceptWidget =
125 CreateWidget<URewardConcept03Widget>(GetOwningPlayer(), RewardClass);
126 }
127
128 if (mCombatRewardConceptWidget != nullptr)
129 {
130 mCombatRewardConceptWidget->OnRewardFlowCompleted.RemoveDynamic(
131 this, &UCombatLayoutHUDWidget::HandleRewardConcept03Completed);
132 mCombatRewardConceptWidget->OnRewardFlowCompleted.AddUniqueDynamic(
133 this, &UCombatLayoutHUDWidget::HandleRewardConcept03Completed);
134 mCombatRewardConceptWidget->BindUIModel(mCombatRewardUIModel);
135 }

153void UCombatLayoutHUDWidget::HandleCombatResultOpenRequested()
154{
155 if (mBossCollapseWidget || mResultRewardsOpened) return;
156 if (mIsPlayerWin == true)
157 {
158 if (!mBossCollapsePlayed)
159 {
160 mBossCollapsePlayed = true;
161 auto* Mode = GetWorld() ? GetWorld()->GetAuthGameMode<ARoomGameModeBase>() : nullptr;
162 const int32 Stage = Mode && Mode->GetRunPersistData()
163 ? static_cast<int32>(Mode->GetRunPersistData()->GetStage().mStageLevel) : 0;
164 if (UBossCollapseWidget::ShouldPlay(mIsPlayerWin, mUIModel && mUIModel->GetCombatResultUI().mIsClearStage, Stage))
165 {
166 mBossCollapseWidget = UBossCollapseWidget::Show(GetOwningPlayer(), Stage,
167 FSimpleDelegate::CreateWeakLambda(this, [this]()
168 {
169 mBossCollapseWidget = nullptr;
170 HandleCombatResultOpenRequested();
171 }));
172 if (mBossCollapseWidget) return;
173 UE_LOG(LogRD, Warning, TEXT("RD_BOSS_COLLAPSE unavailable stage=%d; continuing to rewards"), Stage);
174 }
175 }
176 EnsureCombatResultWidgets();
177 if (mCombatRewardConceptWidget != nullptr
178 && mCombatRewardUIModel != nullptr)
179 {
180 mResultRewardsOpened = true;
181 if (USoundBase* Jingle = SelectCombatResultJingle(true)) UGameplayStatics::PlaySound2D(this, Jingle);
182 mCombatRewardConceptWidget->BindUIModel(mCombatRewardUIModel);
183 mCombatRewardConceptWidget->ResetRewardFlow();
184 mCombatRewardConceptWidget->AddToViewport(10000);
185 mCombatRewardConceptWidget->SetStageClearBackground(0);
186 if (mUIModel && mUIModel->GetCombatResultUI().mIsClearStage)
187 {
188 auto* Mode = GetWorld() ? GetWorld()->GetAuthGameMode<ARoomGameModeBase>() : nullptr;
189 const int32 Stage = Mode && Mode->GetRunPersistData()
190 ? static_cast<int32>(Mode->GetRunPersistData()->GetStage().mStageLevel) : 0;
191 mCombatRewardConceptWidget->SetStageClearBackground(Stage);
192 }
193
194 if (APlayerController* PlayerController = GetOwningPlayer())
195 {
196 FInputModeUIOnly InputMode;
197 InputMode.SetWidgetToFocus(
198 mCombatRewardConceptWidget->TakeWidget());
199 InputMode.SetLockMouseToViewportBehavior(
200 EMouseLockMode::DoNotLock);
201 PlayerController->SetInputMode(InputMode);
202 PlayerController->SetShowMouseCursor(true);
203 }
204 UE_LOG(LogRD, Display,
205 TEXT("RD_REWARD_CONCEPT03_COMBAT shown artifact=%d choices=%d"),
206 mCombatRewardConceptWidget->HasArtifactReward() ? 1 : 0,
207 mCombatRewardUIModel->GetRewardChoices().Num());
208 }
209 return;
210 }
211
212 mCombatResultOverlayWidget->ShowDefeatResult(
213 mUIModel->GetCombatResultUI(),
214 FSimpleDelegate::CreateUObject(this, &UCombatLayoutHUDWidget::HandleCombatResultContinueConfirmed));
215 mCombatResultOverlayWidget->OpenUI();
216}

268void UCombatLayoutHUDWidget::CompleteFinalRunAfterRewards()
269{
270 UWorld* World = GetWorld();
271 if (!World) return;
272 World->GetTimerManager().ClearTimer(mFinalRunCompletionTimerHandle);
273 auto* Mode = World->GetAuthGameMode<ARoomGameModeBase>();
274 if (!Mode) return;
275 if (!Mode->CompleteRunFromRoom())
276 {
277 // A transient save failure must not lose final settlement or replay the comic.
278 World->GetTimerManager().SetTimer(mFinalRunCompletionTimerHandle, this,
279 &UCombatLayoutHUDWidget::CompleteFinalRunAfterRewards, 3.f, false);
280 }

386void UCombatLayoutHUDWidget::OpenWorldMapForNextRoom()
387{
388 UWorldWidgetSubsystem* WorldWidgetSubsystem = GetWorld() != nullptr
389 ? GetWorld()->GetSubsystem<UWorldWidgetSubsystem>() : nullptr;
390 UFrontendMapWidget* MapWidget = mVictoryWorldMap;
391 if (MapWidget == nullptr && WorldWidgetSubsystem != nullptr)
392 {
393 MapWidget = Cast<UFrontendMapWidget>(
394 WorldWidgetSubsystem->GetWorldWidget(EWorldWidgetType::WorldMap));
395 }
396 if (MapWidget == nullptr && WorldWidgetSubsystem != nullptr)
397 {
398 WorldWidgetSubsystem->InitWorldWidget(EWorldWidgetType::WorldMap);
399 MapWidget = WorldWidgetSubsystem->GetWorldWidget<UFrontendMapWidget>(
400 EWorldWidgetType::WorldMap);
401 }
402 if (MapWidget == nullptr && GetWorld() != nullptr
403 && mWorldMapWidgetClass != nullptr)
404 {
405 MapWidget = CreateWidget<UFrontendMapWidget>(
406 GetWorld(), mWorldMapWidgetClass);
407 }
408 if (MapWidget == nullptr)
409 {
410 UE_LOG(LogRD, Warning, TEXT("CombatLayoutHUDWidget: WorldMap widget is not configured."));
411 RestorePostVictoryHUDAndInput();
412 return;
413 }
414
415 mVictoryWorldMap = MapWidget;
416 mVictoryWorldMapLocked = true;
417 MapWidget->OnCloseRequested.AddUniqueDynamic(
418 this, &UCombatLayoutHUDWidget::HandleWorldMapCloseRequested);
419 MapWidget->SetRoomSelectionEnabled(true);
420 MapWidget->ClearMapStatusOverride();
421 // The first post-victory open inherits a collapsed HUD from the reward flow.
422 // After BACK, however, RestorePostVictoryHUDAndInput makes the HUD visible
423 // again. Reopening without collapsing it leaves the full-screen combat HUD
424 // above the map and makes the MAP button appear to do nothing.
425 SetCombatControlsShown(false);
426 MapWidget->OpenUI(FOnEndUIOpenAnimation::CreateWeakLambda(MapWidget, [](UUserWidget* OpenedWidget)
427 {
428 if (UFrontendMapWidget* OpenedMapWidget = Cast<UFrontendMapWidget>(OpenedWidget))
429 {
430 OpenedMapWidget->RefreshMap();
431 }
432 }));
433 if (APlayerController* PlayerController = GetOwningPlayer())
434 {
435 FInputModeUIOnly InputMode;
436 InputMode.SetWidgetToFocus(MapWidget->TakeWidget());
437 InputMode.SetLockMouseToViewportBehavior(EMouseLockMode::DoNotLock);
438 PlayerController->SetInputMode(InputMode);
439 PlayerController->SetShowMouseCursor(true);
440 }
441 // OpenUI가 이미 열린 위젯에서 즉시 반환하는 경우도 있으므로 여기서도
442 // 한 번 갱신한다. 갱신은 선택/ENTER 활성 상태까지 함께 다시 계산한다.
443 MapWidget->RefreshMap();
444}

517void UCombatLayoutHUDWidget::OpenWorldMapForCombatReview()
518{
519 if (mCombatResultFlowActive || mVictoryWorldMapLocked)
520 {
521 return;
522 }
523
524 UWorld* World = GetWorld();
525 UWorldWidgetSubsystem* WorldWidgetSubsystem = World != nullptr
526 ? World->GetSubsystem<UWorldWidgetSubsystem>() : nullptr;
527 if (WorldWidgetSubsystem == nullptr)
528 {
529 UE_LOG(LogRD, Warning,
530 TEXT("CombatLayoutHUDWidget: WorldWidgetSubsystem is unavailable for combat map review."));
531 return;
532 }
533
534 UFrontendMapWidget* MapWidget = WorldWidgetSubsystem->GetWorldWidget<UFrontendMapWidget>(
535 EWorldWidgetType::WorldMap);
536 if (MapWidget == nullptr)
537 {
538 // 정상 방 진입에서는 RoomGameModeBase가 미리 만든다. 자동화/직접 로드처럼
539 // 초기화 순서가 다른 경로에서도 MAP 단추가 조용히 죽지 않게 한 번 보완한다.
540 WorldWidgetSubsystem->InitWorldWidget(EWorldWidgetType::WorldMap);
541 MapWidget = WorldWidgetSubsystem->GetWorldWidget<UFrontendMapWidget>(
542 EWorldWidgetType::WorldMap);
543 }
544 if (MapWidget == nullptr && mWorldMapWidgetClass != nullptr)
545 {
546 // 서브시스템 생성은 OwningPlayer를 요구한다. 플레이어 컨트롤러보다 HUD가
547 // 먼저 구성되는 테스트/직접 레벨 로드에서는 World를 소유자로 직접 만든다.
548 // 정상 방에서는 위 분기에서 공용 인스턴스를 얻으므로 중복 생성되지 않는다.
549 MapWidget = CreateWidget<UFrontendMapWidget>(
550 World, mWorldMapWidgetClass);
551 }
552 if (MapWidget == nullptr)
553 {
554 UE_LOG(LogRD, Warning,
555 TEXT("CombatLayoutHUDWidget: WorldMap widget is not configured for combat map review."));
556 return;
557 }
558
559 // 다른 전투 팝업과 조준 상태를 남겨 둔 채 지도를 열지 않는다.
560 SetMercenaryPanelShown(false);
561 HideDetailOverlay(/*bNotifyGameplay=*/true);
562 if (mUIModel != nullptr && IsAiming())
563 {
564 mUIModel->RequestCancel();
565 }
566
567 mCombatReviewWorldMap = MapWidget;
568 mCombatReviewWorldMapOpen = true;
569 MapWidget->OnCloseRequested.AddUniqueDynamic(
570 this, &UCombatLayoutHUDWidget::HandleWorldMapCloseRequested);
571 MapWidget->SetRoomSelectionEnabled(false);
572 MapWidget->ClearMapStatusOverride();
573
574 if (APlayerController* PlayerController = GetOwningPlayer())
575 {
576 mShowMouseCursorBeforeCombatReviewMap = PlayerController->bShowMouseCursor;
577 FInputModeUIOnly InputMode;
578 InputMode.SetWidgetToFocus(MapWidget->TakeWidget());
579 InputMode.SetLockMouseToViewportBehavior(EMouseLockMode::DoNotLock);
580 PlayerController->SetInputMode(InputMode);
581 PlayerController->SetShowMouseCursor(true);
582 }
583
584 SetCombatControlsShown(false);
585 MapWidget->OpenUI(FOnEndUIOpenAnimation::CreateWeakLambda(
586 MapWidget, [](UUserWidget* OpenedWidget)
587 {
588 if (UFrontendMapWidget* OpenedMapWidget =
589 Cast<UFrontendMapWidget>(OpenedWidget))
590 {
591 OpenedMapWidget->RefreshMap();
592 }
593 }));
594 MapWidget->RefreshMap();
595}

684void UCombatLayoutHUDWidget::SetCombatControlsShown(const bool bShown)
685{
686 // 상세 패널은 뷰포트에 따로 얹혀 있어 HUD 를 접어도 같이 안 접힌다.
687 // 결과 화면 위에 남으면 안 되니 여기서 닫는다.
688 if (bShown == false)
689 {
690 HideDetailOverlay(/*bNotifyGameplay=*/false);
691 }
692
693 // 결과 영상과 보상 화면은 별개 위젯으로 화면 위에 얹힌다. 겹 하나하나를
694 // 접는 것보다 HUD 를 통째로 접는 편이 확실하다 -- 카드·판·아군 칸이
695 // 저마다 제 조건으로 다시 켜지는 일이 없다.
696 //
697 // 되살릴 때는 **Visible** 이어야 한다. 이 HUD 는 화면 전체로 눌림을 받아
698 // 판 탭(카드 여닫기·적 살펴보기)을 처리한다(OpenUI 주석 참고).
699 // SelfHitTestInvisible 로 되돌리면 자식 버튼만 눌리고 판 탭이 죽는다 --
700 // 전투 중 지도를 닫고 돌아오면 빈 땅 탭이 전부 무시되던 원인이다.
701 SetVisibility(bShown ? ESlateVisibility::Visible
702 : ESlateVisibility::Collapsed);
703}
