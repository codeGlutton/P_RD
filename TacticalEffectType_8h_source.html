81 AddBase UMETA(DisplayName = "Add (Base)"),

83 MultiplyAdditive UMETA(DisplayName = "Multiply (Additive)"),

87 // 값 3은 Override가 차지하므로(아래 별칭 참조) 여기서는 건너뛰고 값 4를 명시 고정한다.
88 // 값 4: 배율을 곱셈(거듭제곱)으로 누적 — 스택 시 base^StackCount.
89 MultiplyCompound = 4 UMETA(DisplayName = "Multiply (Compound)"),
90
91 // 값 5: 모든 곱/나눗셈 이후 최종 단계의 합산 보정.
92 AddFinal UMETA(DisplayName = "Add (Final)"),
93 // 값 6: 유효 연산 개수이자 무효 센티넬. mMods[ETacticalModOp::Max]의 배열 크기로 사용.
94 Max UMETA(Hidden, DisplayName = "Invalid"),
95
96 // ── 하위호환 별칭(구 EGameplayModOp와 "동일 정수 값") ────────────────────────
97 // 구 GAS 시절의 enum 이름을 그대로 컴파일/리다이렉트되게 하기 위한 별칭이다.
98 // 같은 값을 가리키므로 신규 이름과 완전히 상호 교환 가능하다.
99 Additive = 0 UMETA(Hidden), // == AddBase
100 Multiplicitive = 1 UMETA(Hidden), // == MultiplyAdditive (구 GAS 철자 오타까지 보존)
101 Division = 2 UMETA(Hidden), // == DivideAdditive
102 // 값 3: 덮어쓰기. 별칭이지만 신규 enum에 대응 항목이 없어 이 이름이 정식으로 노출된다.
103 Override = 3 UMETA(DisplayName = "Override"),
104 };
