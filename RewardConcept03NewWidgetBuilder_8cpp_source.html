103 auto QueueTextureImport = [&SourceFiles, &ImportedNames](
104 const FString& Name, const FString& SourceFile)
105 {
106 const FString PackageName = FString::Printf(
107 TEXT("%s/%s"), ArtPackagePath, *Name);
108 const FString PackageFile = FPackageName::LongPackageNameToFilename(
109 PackageName, FPackageName::GetAssetPackageExtension());
110 IFileManager& FileManager = IFileManager::Get();
111 const bool bMissingAsset = !FileManager.FileExists(*PackageFile);
112 const bool bHasSource = FPaths::FileExists(SourceFile);
113 // 커밋된 uasset만 있는 기존 그림은 그대로 쓸 수 있다. 새 그림을
114 // 추가하거나 uasset이 없을 때만 원본 PNG를 필수로 요구한다.
115 checkf(bHasSource || !bMissingAsset,
116 TEXT("Missing RewardConcept03 texture source and asset: %s"),
117 *SourceFile);
118 if (!bHasSource)
119 {
120 return;
121 }
122 const bool bSourceIsNewer = !bMissingAsset
123 && FileManager.GetTimeStamp(*SourceFile)
124 > FileManager.GetTimeStamp(*PackageFile);
125 if (bMissingAsset || bSourceIsNewer)
126 {
127 SourceFiles.Add(SourceFile);
128 ImportedNames.Add(Name);
129 }
130 };
131 for (const TCHAR* Name : TextureNames)
132 {
133 const FString SourceFile = FPaths::Combine(
134 SourceDirectory(), FString(Name) + TEXT(".png"));
135 QueueTextureImport(Name, SourceFile);
136 }

178 auto ConfigureTexture = [&ImportedNames](const FString& Name)
179 {
180 UTexture2D* Texture = LoadObject<UTexture2D>(
181 nullptr, *TextureObjectPath(*Name));
182 checkf(Texture != nullptr,
183 TEXT("Failed to import RewardConcept03 texture %s"), *Name);
184 const bool bSettingsChanged = Texture->LODGroup != TEXTUREGROUP_UI
185 || Texture->MipGenSettings != TMGS_NoMipmaps
186 || Texture->CompressionSettings != TC_BC7
187 || !Texture->NeverStream || !Texture->SRGB;
188 if (bSettingsChanged)
189 {
190 Texture->Modify();
191 Texture->LODGroup = TEXTUREGROUP_UI;
192 Texture->MipGenSettings = TMGS_NoMipmaps;
193 Texture->CompressionSettings = TC_BC7;
194 Texture->NeverStream = true;
195 Texture->SRGB = true;
196 Texture->PostEditChange();
197 }
198 if (bSettingsChanged || ImportedNames.Contains(Name))
199 {
200 SaveObject(Texture);
201 }
202 };
203 for (const TCHAR* Name : TextureNames)
204 {
205 ConfigureTexture(Name);
206 }
207 ConfigureTexture(ChestTripleBurstTextureName(
208 ChestTripleBurstFrameCount - 1));
209 for (const TCHAR* Name : EffectTextureNames)
210 {
211 ConfigureTexture(Name);
212 }
213 UE_LOG(LogTemp, Display,
214 TEXT("RD_REWARD_CONCEPT03_TEXTURES_READY imported=%d"),
215 ImportedNames.Num());
216 }
