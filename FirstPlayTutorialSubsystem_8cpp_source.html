42void UFirstPlayTutorialSubsystem::Initialize(FSubsystemCollectionBase& Collection)
43{
44 Super::Initialize(Collection);
45 if (!FSlateApplication::IsInitialized()) return;
46 InputGate = MakeShared<FTutorialInputGate>();
47 InputGate->IsActive = [this] { return IsInputRestricted(); };
48 InputGate->IsAllowed = [this](const FVector2D& Position) { return IsPointerAllowed(Position); };
49 InputGate->IsInViewport = [this](const FVector2D& Position)
50 {
51 const auto* Client = GetGameInstance()->GetGameViewportClient();
52 const auto Viewport = Client ? Client->GetGameViewportWidget() : nullptr;
53 return Viewport.IsValid() && Viewport->GetCachedGeometry().IsUnderLocation(Position);
54 };
55 InputGate->IsGameFocused = [this]
56 {
57 const auto* Client = GetGameInstance()->GetGameViewportClient();
58 const auto Viewport = Client ? Client->GetGameViewportWidget() : nullptr;
59 return Viewport.IsValid() && (Viewport->HasKeyboardFocus() || Viewport->HasFocusedDescendants());
60 };
61 FSlateApplication::Get().RegisterInputPreProcessor(InputGate, 0);
62}

74bool UFirstPlayTutorialSubsystem::IsInputRestricted() const

84bool UFirstPlayTutorialSubsystem::IsPointerAllowed(const FVector2D& Position) const
85{
86 if (!IsInputRestricted()) return true;
87 auto* HUD = LastHUD.Get();
88 if (!bPlayerTurn || bExecutingAction || !HUD || HUD->IsGuidedOverlayObscured()) return false;
89 if (IsEncounterHintVisible())
90 {
91 const auto* Button = EncounterWidget->GetContinueButton();
92 if (!Button || !Button->GetCachedGeometry().IsUnderLocation(Position)) return false;
93 auto& Slate = FSlateApplication::Get();
94 const FWidgetPath Path = Slate.LocateWindowUnderMouse(Position, Slate.GetInteractiveTopLevelWindows());
95 for (int32 Index = 0; Index < Path.Widgets.Num(); ++Index)
96 if (Path.Widgets[Index].Widget == Button->GetCachedWidget()) return true;
97 return false;
98 }
99 const auto Stage = GetUserMutableData()->GuidedTutorial.Stage;
100 auto Next = Stage;
101 bool Board = false, Retry = false;
102 const auto* Target = HUD->ResolveGuidedTarget(Stage, Next, Board, Retry);
103 // Wait for the next HUD update to present the new instruction before accepting input.
104 if (Next != Stage) return false;
105 if (FGuidedTutorialProgress::IsReadingStep(Stage))
106 Target = GuidedWidget ? GuidedWidget->GetContinueButton() : nullptr;
107 auto& Slate = FSlateApplication::Get();
108 const FWidgetPath Path = Slate.LocateWindowUnderMouse(Position, Slate.GetInteractiveTopLevelWindows());
109 if (Board)
110 {
111 if (!HUD->IsGuidedBoardInputAt(Position) || !IsScenarioPointerAllowed(Position)) return false;
112 bool InHUD = false;
113 for (int32 Index = 0; Index < Path.Widgets.Num(); ++Index)
114 {
115 const auto& Entry = Path.Widgets[Index];
116 InHUD |= Entry.Widget == HUD->GetCachedWidget();
117 if (Entry.Widget->GetType() == FName(TEXT("SButton"))) return false;
118 }
119 return InHUD;
120 }
121 if (!Target || !Target->IsVisible() || !Target->GetIsEnabled() ||
122 !Target->GetCachedGeometry().IsUnderLocation(Position)) return false;
123 for (int32 Index = 0; Index < Path.Widgets.Num(); ++Index)
124 if (Path.Widgets[Index].Widget == Target->GetCachedWidget()) return true;
125 return false;
126}

148void UFirstPlayTutorialSubsystem::TurnStarted(bool PlayerTurn)
149{
150 const bool WasInCombat = bInCombat;
151 bInCombat = true;
152 bPlayerTurn = PlayerTurn;
153 bExecutingAction = false;
154 HideEncounterHint();
155 auto* Data = GetUserMutableData();
156 if (!Data->GuidedTutorial.Enrolled || (Data->GuidedTutorial.NeedsFirstRoom(Data->TutorialProgress.Skipped) && !HasScenario()))
157 {
158 HideGuided();
159 return;
160 }
161 if (!WasInCombat)
162 {
163 Data->GuidedTutorial.ResumeInNewBattle();
164 LastLessonContext = EGuidedLesson::None;
165 }
166 bInCombat = true;
167 bPlayerTurn = PlayerTurn;
168 bExecutingAction = false;
169 if (PlayerTurn && !Data->TutorialProgress.Skipped && Data->GuidedTutorial.BeginSystems())
170 Save();
171 if (!Data->TutorialProgress.WelcomeSeen && !Data->TutorialProgress.Skipped)
172 {
173 Data->TutorialProgress.WelcomeSeen = true;
174 Save();
175 }
176 if (!PlayerTurn)
177 HideGuided();
178}
179void UFirstPlayTutorialSubsystem::TurnEnded(bool PlayerTurn, bool Succeeded)
180{
181 HideEncounterHint();
182 if (bInCombat && PlayerTurn && Succeeded)
183 {
184 AdvanceGuided(EGuidedStage::EndTurn);
185 // An exhausted/disabled unit must never trap the next ally in a stale target-selection step.
186 auto& Flow = GetUserMutableData()->GuidedTutorial;
187 if (Flow.Stage == EGuidedStage::MoveTile || Flow.Stage == EGuidedStage::ConfirmMove)
188 {
189 Flow.Stage = EGuidedStage::SelectMove;
190 Save();
191 }
192 if (Flow.Stage == EGuidedStage::SkillTarget || Flow.Stage == EGuidedStage::ConfirmSkill)
193 {
194 Flow.Stage = EGuidedStage::SelectSkill;
195 Save();
196 }
197 }
198 bPlayerTurn = false;
199 HideGuided();
200}
201void UFirstPlayTutorialSubsystem::ActionEnded(bool PlayerAction, const USRPGAction* Action, bool Succeeded)
202{
203 if (!Action || Action->GetActionType() == ESRPGActionType::BuildAction) return;
204 bExecutingAction = false;
205 if (!bInCombat || !PlayerAction || !Succeeded || !Action)
206 return;
207 if (IsScenarioGuiding())
208 {
209 if (Action->GetInstigator() != ScenarioUnit.Get()) return;
210 if (Action->IsA<USRPGMoveAction>() && ScenarioUnit->GetTileTransform().mIndex != GetScenarioMoveTile()) return;
211 }
212 bool Changed = false;
213 if (Action->IsA<USRPGMoveAction>())
214 Changed = GetUserMutableData()->GuidedTutorial.CompleteAction(true, PlayerAction, Succeeded);
215 else if (Action->IsA<USRPGSkillAction>())
216 Changed = GetUserMutableData()->GuidedTutorial.CompleteAction(false, PlayerAction, Succeeded);
217 if (Changed)
218 {
219 Save();
220 HideGuided();
221 }
222}

274void UFirstPlayTutorialSubsystem::UpdateGuidedHUD(UCombatLayoutHUDWidget* HUD)
275{
276 if (LastHUD.Get() != HUD) HideEncounterHint();
277 LastHUD = HUD;
278 if (UpdateEncounterHint(HUD)) { HideGuided(); return; }
279 auto* Data = GetUserMutableData();
280 auto& Flow = Data->GuidedTutorial;
281 if (!Flow.Enrolled)
282 {
283 HideGuided();
284 return;
285 }
286 LastHUD = HUD;
287 const EGuidedLesson Context = HUD->GuidedLessonContext();
288 const bool LeavingLesson = !Flow.SystemsActive && (Flow.CoreComplete || Flow.FirstBattleFinished) &&
289 Flow.Lesson != EGuidedLesson::None &&
290 LastLessonContext != EGuidedLesson::None && Context != LastLessonContext;
291 const bool ArtifactTransit = Flow.Lesson == EGuidedLesson::Artifact &&
292 LastLessonContext == EGuidedLesson::Mercenary &&
293 Context == EGuidedLesson::Artifact;
294 if (LeavingLesson && !ArtifactTransit)
295 {
296 Flow.Lesson = EGuidedLesson::None;
297 Flow.Stage = EGuidedStage::LessonMenu;
298 Save();
299 HideGuided();
300 }
301 if (!Flow.SystemsActive && (Flow.CoreComplete || Flow.FirstBattleFinished) && Flow.Stage == EGuidedStage::LessonMenu &&
302 Context != LastLessonContext && Context != EGuidedLesson::None && !Data->TutorialProgress.Skipped)
303 {
304 if (Flow.BeginLesson(Context, HUD->HasGuidedArtifact()))
305 {
306 Save();
307 HideGuided();
308 }
309 }
310 LastLessonContext = Context;
311 if (HUD->IsGuidedOverlayObscured() || !bInCombat || !bPlayerTurn || bExecutingAction ||
312 Data->TutorialProgress.Skipped || Flow.Stage == EGuidedStage::Done ||
313 Flow.Stage == EGuidedStage::LessonMenu)
314 {
315 HideGuided();
316 return;
317 }
318 // A battle may have ended before the enemy-information step could start.
319 if (Flow.SystemsActive && Flow.Stage == EGuidedStage::OpenMonster && !HUD->HasGuidedMonster())
320 {
321 Flow.FinishSystems(); Save(); HideGuided(); return;
322 }
323 bool Board = false, Retry = false;
324 EGuidedStage Next = Flow.Stage;
325 UWidget* Target = HUD->ResolveGuidedTarget(Flow.Stage, Next, Board, Retry);
326 if (Next != Flow.Stage)
327 {
328 if (Next == FGuidedTutorialProgress::NextStage(Flow.Stage))
329 Flow.Advance(Flow.Stage);
330 else
331 Flow.Stage = Next;
332 Save();
333 HideGuided();
334 return;
335 }
336 bool TargetVisible = Target != nullptr;
337 for (UWidget* Parent = Target; Parent; Parent = Parent->GetParent())
338 if (!Parent->IsVisible())
339 {
340 TargetVisible = false;
341 break;
342 }
343 if (Flow.Stage != EGuidedStage::LessonMenu && !Board &&
344 (!TargetVisible || Target->GetCachedGeometry().GetLocalSize().IsNearlyZero()))
345 {
346 HideGuided();
347 return;
348 }
349 if (!GuidedWidget)
350 {
351 GuidedWidget = CreateWidget<UGuidedTutorialWidget>(GetGameInstance());
352 if (!GuidedWidget)
353 return;
354 GuidedWidget->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
355 GuidedWidget->AddToViewport(75);
356 }
357 GuidedWidget->Present(Flow.Stage, Target, Board, Retry);
358 TArray<FVector2D> WorldCorners;
359 GetScenarioFocus(WorldCorners);
360 GuidedWidget->SetWorldFocus(WorldCorners);
361 if (IsScenarioGuiding() && !Retry)
362 {
363 if (Flow.Stage == EGuidedStage::MoveTile)
364 GuidedWidget->SetNotice(NSLOCTEXT("Guided", "FixedMove", "빛나는 칸을 누르세요.\n앞으로 한 칸 이동합니다."));
365 else if (Flow.Stage == EGuidedStage::SkillTarget)
366 GuidedWidget->SetNotice(NSLOCTEXT("Guided", "FixedEnemy", "표시한 독수리를 누르세요."));
367 else if (Flow.Stage == EGuidedStage::HoldSkill)
368 GuidedWidget->SetNotice(FText::Format(NSLOCTEXT("Guided", "FixedHold", "{0}\n길게 눌러 설명을 확인하세요."), ScenarioSkillName));
369 else if (Flow.Stage == EGuidedStage::SelectSkill)
370 GuidedWidget->SetNotice(FText::Format(NSLOCTEXT("Guided", "FixedSkill", "{0}\n짧게 눌러 선택하세요."), ScenarioSkillName));
371 }
372 if (Retry && Target && Target->GetFName() == TEXT("DetailCloseButton"))
373 GuidedWidget->SetNotice(NSLOCTEXT("Guided", "CloseUnexpectedDetail", "설명을 닫고 카드를 짧게 누르세요."));
374 else if (Target && Target->GetFName() == TEXT("EndTurnButton") && Flow.Stage != EGuidedStage::EndTurn)
375 GuidedWidget->SetNotice(
376 NSLOCTEXT("Guided", "Unavailable", "해당 행동을 할 수 없습니다.\n턴을 종료하세요."));
377 else if (Target && Target->GetFName() == TEXT("SkillToggleButton") &&
378 Flow.Stage != EGuidedStage::OpenSkills)
379 GuidedWidget->SetNotice(NSLOCTEXT("Guided", "Reopen", "행동 카드를 펼치세요."));
380 bool Click = Flow.Stage == EGuidedStage::OpenSkills || Flow.Stage == EGuidedStage::SelectMercenary ||
381 Flow.Stage == EGuidedStage::OpenMercenary || Flow.Stage == EGuidedStage::OpenInventory ||
382 Flow.Stage == EGuidedStage::OpenMonster || Flow.Stage == EGuidedStage::SelectMonster ||
383 Flow.Stage == EGuidedStage::CloseMercenary || Flow.Stage == EGuidedStage::CloseMonster ||
384 Flow.Stage == EGuidedStage::CloseSkill || Flow.Stage == EGuidedStage::CloseMercenarySkill ||
385 Flow.Stage == EGuidedStage::CloseArtifact || Flow.Stage == EGuidedStage::CloseMonsterSkill;
386 Click |= Flow.Stage == EGuidedStage::SelectRange || Flow.Stage == EGuidedStage::EffectRange;
387 UButton* Button = FGuidedTutorialProgress::IsReadingStep(Flow.Stage)
388 ? GuidedWidget->GetContinueButton() : (Click ? Cast<UButton>(Target) : nullptr);
389 if (BoundButton.Get() != Button || BoundStage != Flow.Stage)
390 {
391 if (BoundButton.IsValid())
392 BoundButton->OnClicked.RemoveDynamic(this, &UFirstPlayTutorialSubsystem::GuidedClicked);
393 BoundButton = Button;
394 BoundStage = Flow.Stage;
395 if (Button)
396 Button->OnClicked.AddUniqueDynamic(this, &UFirstPlayTutorialSubsystem::GuidedClicked);
397 }
