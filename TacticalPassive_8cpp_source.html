21void UTacticalPassive::ActivatePassive(
22 const FGameplayTag& TimingTag,
23 const FPassiveActivateContext& Ctx,
24 TInstancedStruct<FDynamicPassiveData>& PassiveState)
25{
26 // 상태 작업본 준비: 비어 있으면 커밋본 복사, 커밋본도 없으면 최초 생성 (무상태 패시브는 빈 상태 유지)
27 if (!PassiveState.IsValid())
28 {
29 if (mState.IsValid())
30 {
31 PassiveState = mState;
32 }
33 else
34 {
35 InitializeState(PassiveState);
36 }
37 }
38
39 // 같은 태그가 여러 시점에 걸릴 수 있으므로 else-if 없이 순서대로 전부 처리 (리셋 → 캡처 → 발동 → 해제)
40
41 // 카운터 리셋 시점: 내부 카운터 초기화
42 if (TimingTag == mCounterResetTimingTag)
43 {
44 OnCounterReset(PassiveState);
45 }
46
47 // 캡처 시점: 비교용 값 저장
48 if (TimingTag == mCaptureTimingTag)
49 {
50 OnCapture(Ctx, PassiveState);
51 }
52
53 // 발동 시점: 발동 경로 처리 (게이트 → 계산 → 적용)
54 if (TimingTag == mActivateTimingTag)
55 {
56 OnActivate(Ctx, PassiveState);
57 }
58
59 // 해제 시점: 적용 중인 이펙트를 제거 (핸들 없으면 내부에서 no-op)
60 if (TimingTag == mDeactivateTimingTag)
61 {
62 DeactivatePassive();
63 }
64}

88void UTacticalPassive::NotifyPassive(
89 const FPassiveActivateContext& Ctx,
90 TSubclassOf<UTacticalEffect> EffectClass,
91 const float Magnitude,
92 const EPassiveEffectTarget EffectTarget,
93 const bool bRefreshPrevious)
94{
95 // 적용할 이펙트가 없으면(계산 전용/무상태 등) 적용하지 않음
96 if (EffectClass == nullptr)
97 {
98 return;
99 }
100
101 // 기여가 없으면(변화량 0) 굳이 이펙트 적용할 필요가 없으므로 바로 리턴
102 // @note 이펙트 적용을 안하니까 이펙트 핸들을 갖지 않으므로 딱히 부작용은 없음
103 if (FMath::IsNearlyZero(Magnitude))
104 {
105 return;
106 }
107
108 // 소유자(시전 주체)의 속성 컴포넌트 획득. 대상과 무관하게 1회만 조회.
109 IBoardCombatTarget* OwnerTarget = Cast<IBoardCombatTarget>(Ctx.mOwner.Get());
110 if (OwnerTarget == nullptr)
111 {
112 return;
113 }
114 UAttributeSetComponentModel* OwnerComp = OwnerTarget->GetAttributeComponentModel();
115 if (OwnerComp == nullptr)
116 {
117 return;
118 }
119
120 // 적용 대상 목록: Self면 소유자 1명, Targets면 Ctx.mTargets 전부
121 TArray<TWeakObjectPtr<UBoardActorModel>> SelfOnly;
122 if (EffectTarget == EPassiveEffectTarget::Self)
123 {
124 SelfOnly.Add(Ctx.mOwner);
125 }
126 const TArray<TWeakObjectPtr<UBoardActorModel>>& ApplyTargets =
127 (EffectTarget == EPassiveEffectTarget::Self) ? SelfOnly : Ctx.mTargets;
128
129 // 적용할 대상이 없으면 종료
130 if (ApplyTargets.Num() == 0)
131 {
132 return;
133 }
134
135 // 상태이상 클래스면 수치를 스택 수로 넣음 (스태킹 검사 때문에 mDynamicMagnitude는 기본값 유지)
136 const bool bIsStatus = EffectClass->IsChildOf(UTacticalEffect_Status::StaticClass());
137 const int32 Stacks = bIsStatus ? FMath::FloorToInt(Magnitude) : -1;
138 if (bIsStatus && Stacks <= 0)
139 {
140 return;
141 }
142
143 // 이전 핸들이 남아있으면 먼저 제거 (효과 배열 적용 중에는 호출자가 false로 끔)
144 if (bRefreshPrevious && mAppliedEffects.Num() > 0)
145 {
146 DeactivatePassive();
147 }
148
149 // 대상마다 이펙트 적용하면서 핸들을 배열에 저장
150 int32 AppliedNum = 0;
151 for (const TWeakObjectPtr<UBoardActorModel>& TargetPtr : ApplyTargets)
152 {
153 // 대상의 속성 컴포넌트 획득.
154 // 유효하지 않은 대상은 건너뜀.
155 IBoardCombatTarget* HitTarget = Cast<IBoardCombatTarget>(TargetPtr.Get());
156 if (HitTarget == nullptr)
157 {
158 continue;
159 }
160 UAttributeSetComponentModel* TargetComp = HitTarget->GetAttributeComponentModel();
161 if (TargetComp == nullptr)
162 {
163 continue;
164 }
165
166 // 소유자를 시전자로 spec 생성 -> 계산된 크기를 배율(상태이상은 스택 수)로 주입 -> 대상에 적용 -> 핸들 저장
167 UTacticalEffectContext* EffectContext = OwnerComp->MakeEffectContext();
168 EffectContext->SetAbility(this);
169
170 TSharedPtr<FTacticalEffectSpec> Spec = OwnerComp->MakeOutgoingSpec(EffectClass, EffectContext);
171 if (bIsStatus)
172 {
173 Spec->SetStackCount(Stacks);
174 }
175 else
176 {
177 Spec->mDynamicMagnitude = Magnitude;
178 }
179
180 mAppliedEffects.Add({ OwnerComp->ApplyTacticalEffectSpecToTarget(*Spec, TargetComp), Stacks });
181 ++AppliedNum;
182 }
183
184 UE_LOG(LogPassive, Log, TEXT("이펙트 적용: %s → %s (수치 %.1f, 대상 %d)"), *GetNameSafe(mStaticData), *GetNameSafe(EffectClass), Magnitude, AppliedNum);
185}

187void UTacticalPassive::DeactivatePassive()
188{
189 if (mAppliedEffects.Num() == 0)
190 {
191 return;
192 }
193
194 // 핸들마다 이펙트를 활성 집합에서 제거 -> TAS가 base에서 재계산(기여분을 산술로 되돌리지 않음)
195 // 상태이상은 저장한 스택 수만 제거, 일반 이펙트는 -1이라 통째로 제거
196 for (const FPassiveAppliedEffect& Applied : mAppliedEffects)
197 {
198 if (Applied.mHandle.IsValid() == false)
199 {
200 continue;
201 }
202 UAttributeSetComponentModel* OwningComp = Applied.mHandle.GetOwningAttributeSetComponentModel();
203 if (OwningComp != nullptr)
204 {
205 OwningComp->RemoveActiveTacticalEffect(Applied.mHandle, Applied.mStacks);
206 }
207 }
208
209 UE_LOG(LogPassive, Verbose, TEXT("이펙트 해제: %s (핸들 %d)"), *GetNameSafe(mStaticData), mAppliedEffects.Num());
210
211 // 핸들 목록 전체 비움
212 mAppliedEffects.Reset();
213}

215bool UTacticalPassive::PassesTargetQuantifier(
216 const FPassiveActivateContext& Ctx,
217 const TInstancedStruct<FDynamicPassiveData>& State) const
218{
219 // 판정 대상이 없으면(빈 타일 공격 등) 대상 인덱스 없이 조건을 한 번 평가.
220 // 대상을 참조하는 조건은 정보가 없어 탈락하고, 대상을 안 보는 조건은 그대로 판정됨
221 const int32 TargetNum = Ctx.mTargets.Num();
222 if (TargetNum == 0)
223 {
224 return IsTargetQualified(Ctx, INDEX_NONE, State);
225 }
226
227 // 타겟별 자격 조건(predicate)으로 자격 타겟 수 집계
228 int32 QualifiedNum = 0;
229 for (int32 Index = 0; Index < TargetNum; ++Index)
230 {
231 if (IsTargetQualified(Ctx, Index, State))
232 {
233 ++QualifiedNum;
234 }
235 }
236
237 // 수량 조건 값: 기본값은 Any
238 const EPassiveTargetQuantifier Quantifier = (mStaticData != nullptr)
239 ? mStaticData->mTargetQuantifier
240 : EPassiveTargetQuantifier::Any;
241
242 switch (Quantifier)
243 {
244 case EPassiveTargetQuantifier::All:
245 return QualifiedNum == TargetNum;
246 case EPassiveTargetQuantifier::Any:
247 default:
248 return QualifiedNum >= 1;
249 }
250}
