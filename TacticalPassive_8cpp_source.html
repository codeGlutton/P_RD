18void UTacticalPassive::ActivatePassive(
19 const FGameplayTag& TimingTag,
20 const FPassiveActivateContext& Ctx,
21 TInstancedStruct<FDynamicPassiveData>& PassiveState)
22{
23 // 들어온 타이밍이 발동 시점이면: 적용 여부를 묻고 참이면 이펙트 적용
24 if (TimingTag == mActivateTimingTag)
25 {
26 // 수량 조건(quantifier) 게이트: 자격 타겟 수가 기준 미달이면 발동하지 않음
27 if (PassesTargetQuantifier(Ctx) == false)
28 {
29 return;
30 }
31
32 // 발동 시 대상에게 가할 기여값(크기)
33 float ContributionMagnitude = 0.f;
34
35 // 내부상태로 적용 여부를 판단하고 다음 상태를 PassiveState에 기록
36 // 여기서는 아직 내부상태가 변경되진 않음 (CommitPassive 해야 변경됨)
37 if (EvaluateActivate(Ctx, PassiveState, ContributionMagnitude))
38 {
39 // 계산된 수치를 이펙트로 적용
40 NotifyPassive(Ctx, ContributionMagnitude);
41 }
42 }
43 // 해제 시점이면: 적용 중인 이펙트를 제거 (핸들 없으면 내부에서 no-op)
44 else if (TimingTag == mDeactivateTimingTag)
45 {
46 DeactivatePassive();
47 }
48 // 둘 다 아니면 무시
49}

72void UTacticalPassive::NotifyPassive(
73 const FPassiveActivateContext& Ctx,
74 const float Magnitude)
75{
76 // 적용할 이펙트가 없으면(계산 전용/무상태 등) 적용하지 않음
77 if (mEffectClass == nullptr)
78 {
79 return;
80 }
81
82 // 기여가 없으면(변화량 0) 굳이 이펙트 적용할 필요가 없으므로 바로 리턴
83 // @note 이펙트 적용을 안하니까 이펙트 핸들을 갖지 않으므로 딱히 부작용은 없음
84 if (FMath::IsNearlyZero(Magnitude))
85 {
86 return;
87 }
88
89 // 소유자(시전 주체)의 속성 컴포넌트 획득. 대상과 무관하게 1회만 조회.
90 IBoardCombatTarget* OwnerTarget = Cast<IBoardCombatTarget>(Ctx.mOwner.Get());
91 if (OwnerTarget == nullptr)
92 {
93 return;
94 }
95 UAttributeSetComponentModel* OwnerComp = OwnerTarget->GetAttributeComponentModel();
96 if (OwnerComp == nullptr)
97 {
98 return;
99 }
100
101 // 적용할 대상이 없으면 종료
102 if (Ctx.mTargets.Num() == 0)
103 {
104 return;
105 }
106
107 // 이전 핸들이 남아있으면 먼저 제거
108 if (mActiveHandles.Num() > 0)
109 {
110 DeactivatePassive();
111 }
112
113 // 대상마다 이펙트 적용하면서 핸들을 배열에 저장
114 for (const TWeakObjectPtr<UBoardActorModel>& TargetPtr : Ctx.mTargets)
115 {
116 // 대상의 속성 컴포넌트 획득.
117 // 유효하지 않은 대상은 건너뜀.
118 IBoardCombatTarget* HitTarget = Cast<IBoardCombatTarget>(TargetPtr.Get());
119 if (HitTarget == nullptr)
120 {
121 continue;
122 }
123 UAttributeSetComponentModel* TargetComp = HitTarget->GetAttributeComponentModel();
124 if (TargetComp == nullptr)
125 {
126 continue;
127 }
128
129 // 소유자를 시전자로 spec 생성 -> 계산된 크기를 배율로 주입 -> 대상에 적용 -> 핸들 저장
130 UTacticalEffectContext* EffectContext = OwnerComp->MakeEffectContext();
131 EffectContext->SetInstigator(Ctx.mOwner.Get());
132 EffectContext->SetAttributeSetComponentModel(OwnerComp);
133 EffectContext->SetAbility(this);
134
135 TSharedPtr<FTacticalEffectSpec> Spec = OwnerComp->MakeOutgoingSpec(mEffectClass, EffectContext);
136 Spec->mDynamicMagnitude = Magnitude;
137
138 mActiveHandles.Add(OwnerComp->ApplyTacticalEffectSpecToTarget(*Spec, TargetComp));
139 }
140}

142void UTacticalPassive::DeactivatePassive()
143{
144 if (mActiveHandles.Num() == 0)
145 {
146 return;
147 }
148
149 // 핸들마다 이펙트를 활성 집합에서 제거 -> TAS가 base에서 재계산(기여분을 산술로 되돌리지 않음)
150 for (FActiveTacticalEffectHandle& Handle : mActiveHandles)
151 {
152 if (Handle.IsValid() == false)
153 {
154 continue;
155 }
156 UAttributeSetComponentModel* OwningComp = Handle.GetOwningAttributeSetComponentModel();
157 if (OwningComp != nullptr)
158 {
159 OwningComp->RemoveActiveTacticalEffect(Handle);
160 }
161 }
162
163 // 배치 핸들 전체 비움
164 mActiveHandles.Reset();
165}

167bool UTacticalPassive::PassesTargetQuantifier(const FPassiveActivateContext& Ctx) const
168{
169 // 판정 대상이 없으면(자기 대상 등) 게이트 통과
170 const int32 TargetNum = Ctx.mTargets.Num();
171 if (TargetNum == 0)
172 {
173 return true;
174 }
175
176 // 타겟별 자격 조건(predicate)으로 자격 타겟 수 집계
177 int32 QualifiedNum = 0;
178 for (int32 Index = 0; Index < TargetNum; ++Index)
179 {
180 // 스냅샷이 짝으로 있으면 넘기고, 없으면 스냅샷과 실제객체가 다르므로 nullptr
181 const FBoardCombatTargetSnapshotData* Snapshot =
182 Ctx.mTargetSnapshots.IsValidIndex(Index) ? Ctx.mTargetSnapshots[Index] : nullptr;
183
184 if (IsTargetQualified(Snapshot))
185 {
186 ++QualifiedNum;
187 }
188 }
189
190 // 수량 조건 값: 기본값은 Any
191 const EPassiveTargetQuantifier Quantifier = (mStaticData != nullptr)
192 ? mStaticData->mTargetQuantifier
193 : EPassiveTargetQuantifier::Any;
194
195 switch (Quantifier)
196 {
197 case EPassiveTargetQuantifier::All:
198 return QualifiedNum == TargetNum;
199 case EPassiveTargetQuantifier::Any:
200 default:
201 return QualifiedNum >= 1;
202 }
203}
