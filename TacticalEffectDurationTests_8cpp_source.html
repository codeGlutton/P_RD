24static UWorld* GetAnyGameWorld()
25{

30 for (const FWorldContext& Context : GEngine->GetWorldContexts())
31 {
32 if ((Context.WorldType == EWorldType::PIE || Context.WorldType == EWorldType::Game) && Context.World() != nullptr)

36 }

46bool FTacticalEffectDurationTests::RunTest(const FString& Parameters)
47{
48 /* 1. 월드 및 FrameworkModel 환경 준비 */
49 UWorld* World = GetAnyGameWorld();
50 if (World == nullptr)
51 {
52 World = GWorld;
53 }
54 if (TestNotNull(TEXT("UWorld 환경이 유효해야 합니다."), World) == false)
55 {
56 return false;
57 }
58
59 USimulationSubsystem* SimSubsystem = World->GetSubsystem<USimulationSubsystem>();
60 if (TestNotNull(TEXT("USimulationSubsystem이 존재해야 합니다."), SimSubsystem) == false)
61 {
62 return false;
63 }
64
65 URoomInstance* RoomInstance = FSimulationSubsystemTestAccessor::GetRoomInstance(SimSubsystem);
66 if (TestNotNull(TEXT("RoomInstance가 존재해야 합니다."), RoomInstance) == false)
67 {
68 return false;
69 }
70
71 UTacticalFrameworkModel* FrameworkModel = Cast<UTacticalFrameworkModel>(RoomInstance->mAliveSubsystemModels.FindRef(UTacticalFrameworkModel::StaticClass()));
72 if (FrameworkModel == nullptr)
73 {
74 FrameworkModel = NewObject<UTacticalFrameworkModel>(RoomInstance);
75 RoomInstance->mAliveSubsystemModels.Add(UTacticalFrameworkModel::StaticClass(), FrameworkModel);
76 }
77
78 /* 2. Mock ActorModel 및 AttributeSetComponentModel 생성 */
79 UTASActorModelMock* MockActorModel = NewObject<UTASActorModelMock>(World);
80 MockActorModel->Initialize();
81 MockActorModel->BeginPlay();
82
83 UAttributeSetComponentModel* CompModel = MockActorModel->GetAttributeComponent();
84 if (TestNotNull(TEXT("CompModel이 유효해야 합니다."), CompModel) == false)
85 {
86 return false;
87 }
88
89 UTacticalEffectContext* EffectContext = CompModel->MakeEffectContext();
90
91 /* 테스트 1: 턴/라운드 지속시간 단위 격리 검증 (EveryTurn vs EveryRound) */
92 {
93 FrameworkModel->AdvanceTurnDuration(0);
94 FrameworkModel->AdvanceRoundDuration(0);
95
96 // 2턴 지속 이펙트 적용
97
98
99 TSharedPtr<FTacticalEffectSpec> TurnEffectSpec = CompModel->MakeOutgoingSpec(UTestTurnDurationTacticalEffect::StaticClass(), EffectContext);
100 FActiveTacticalEffectHandle TurnHandle = CompModel->ApplyTacticalEffectSpecToSelf(*TurnEffectSpec);
101 TestTrue(TEXT("2턴 지속 이펙트가 적용되어야 합니다."), TurnHandle.IsValid());
102
103 // 라운드를 진행했을 때 (AdvanceRoundDuration) 턴 지속 이펙트가 삭제되지 않고 유지되는지 검증
104 FrameworkModel->AdvanceRoundDuration(1);
105 FrameworkModel->AdvanceRoundDuration(2);
106 TestTrue(TEXT("라운드 진행 시에는 턴 단위 이펙트가 만기되지 않아야 합니다."), CompModel->GetActiveTacticalEffect(TurnHandle) != nullptr);
107
108 // 1턴 진행 시 유지 검증 (Duration = 2턴)
109 FrameworkModel->AdvanceTurnDuration(1);
110 TestTrue(TEXT("1턴 진행 시(Duration=2) 이펙트가 살아있어야 합니다."), CompModel->GetActiveTacticalEffect(TurnHandle) != nullptr);
111
112 // 2턴 도달 시 만기 및 제거 검증
113 FrameworkModel->AdvanceTurnDuration(2);
114 TestTrue(TEXT("2턴 진행 시 이펙트가 만기되어 제거되어야 합니다."), CompModel->GetActiveTacticalEffect(TurnHandle) == nullptr);
115 }
116
117 /* 테스트 2: 스택 단일 차감 및 Duration 재시작 검증 (RemoveSingleStackAndRefreshDuration) */
118 {
119 FrameworkModel->AdvanceTurnDuration(0);
120
121 // 2턴 지속, RemoveSingleStackAndRefreshDuration 이펙트 적용 (2스택)
122 TSharedPtr<FTacticalEffectSpec> StackEffectSpec = CompModel->MakeOutgoingSpec(UTestStackRemoveSingleDurationTacticalEffect::StaticClass(), EffectContext);
123 FActiveTacticalEffectHandle StackHandle1 = CompModel->ApplyTacticalEffectSpecToSelf(*StackEffectSpec);
124 FActiveTacticalEffectHandle StackHandle2 = CompModel->ApplyTacticalEffectSpecToSelf(*StackEffectSpec);
125
126 const FActiveTacticalEffect* ActiveEffect = CompModel->GetActiveTacticalEffect(StackHandle1);
127 if (TestNotNull(TEXT("스택형 활성 이펙트가 존재해야 합니다."), ActiveEffect) == true)
128 {
129 TestEqual(TEXT("초기 스택 수가 2이어야 합니다."), ActiveEffect->mSpec.GetStackCount(), 2);
130 }
131
132 // 2턴 경과 -> 1스택 차감 후 1스택으로 남고, Duration이 새 턴(턴 2)으로 재시작되는지 검증
133 FrameworkModel->AdvanceTurnDuration(2);
134
135 ActiveEffect = CompModel->GetActiveTacticalEffect(StackHandle1);
136 if (TestNotNull(TEXT("1스택 차감 후에도 이펙트가 유지되어야 합니다."), ActiveEffect) == true)
137 {
138 TestEqual(TEXT("스택 수가 1로 줄어야 합니다."), ActiveEffect->mSpec.GetStackCount(), 1);
139 TestEqual(TEXT("StartTime이 턴 2로 갱신되어야 합니다."), ActiveEffect->mStartTime, 2);
140 }
141
142 // 다시 2턴 경과(총 턴 4) -> 남은 1스택마저 소멸하여 최종 제거되는지 검증
143 FrameworkModel->AdvanceTurnDuration(4);
144 TestTrue(TEXT("마지막 스택 차감 후 이펙트가 완전히 제거되어야 합니다."), CompModel->GetActiveTacticalEffect(StackHandle1) == nullptr);
145 }
146
147 return true;
148}
