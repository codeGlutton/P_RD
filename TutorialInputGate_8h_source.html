8// Slate routes real touch through the same down/up preprocessing path as the mouse.
9// Keep accepted presses paired with their release, even if the lesson advances on press.
10class FTutorialInputGate final : public IInputProcessor
11{
12 public:
13 TFunction<bool()> IsActive;
14 TFunction<bool()> IsGameFocused;

17
18 void Tick(float, FSlateApplication &, TSharedRef<ICursor>) override
19 {
20 }
21 bool HandleMouseButtonDownEvent(FSlateApplication &, const FPointerEvent &Event) override
22 {
23 if (!IsActive() || !IsInViewport(Event.GetScreenSpacePosition()))
24 return false;
25 const FString Pointer = PressKey(Event);
26 const bool Primary = Event.IsTouchEvent() || Event.GetEffectingButton() == EKeys::LeftMouseButton;
27 if (Primary && Accepted.IsEmpty() && IsAllowed(Event.GetScreenSpacePosition()))
28 {
29 Accepted.Add(Pointer);
30 return false;
31 }
32 Rejected.Add(Pointer);
33 return true;
34 }
35 bool HandleMouseButtonUpEvent(FSlateApplication &, const FPointerEvent &Event) override
36 {
37 const FString Pointer = PressKey(Event);
38 if (Accepted.Remove(Pointer) > 0)
39 return false;
40 if (Rejected.Remove(Pointer) > 0)
41 return true;
42 return IsActive() && IsInViewport(Event.GetScreenSpacePosition());
43 }
44 bool HandleMouseButtonDoubleClickEvent(FSlateApplication &App, const FPointerEvent &Event) override
45 {
46 return HandleMouseButtonDownEvent(App, Event);
47 }
48 bool HandleMouseMoveEvent(FSlateApplication &, const FPointerEvent &Event) override
49 {
50 for (const auto &Press : Accepted)
51 if (Press.StartsWith(PointerKey(Event)))
52 return false;
53 return IsActive() && IsInViewport(Event.GetScreenSpacePosition()) && !IsAllowed(Event.GetScreenSpacePosition());
54 }
55 bool HandleMouseWheelOrGestureEvent(FSlateApplication &, const FPointerEvent &Event, const FPointerEvent *) override
56 {
57 return IsActive() && IsInViewport(Event.GetScreenSpacePosition());
58 }
59 bool HandleKeyDownEvent(FSlateApplication &, const FKeyEvent &Event) override
60 {
61 // Window close remains an OS operation; game navigation/shortcuts are locked.
62 return IsActive() && IsGameFocused() && !(Event.IsAltDown() && Event.GetKey() == EKeys::F4);
63 }
64 bool HandleKeyUpEvent(FSlateApplication &App, const FKeyEvent &Event) override
65 {
66 return HandleKeyDownEvent(App, Event);
67 }
68 bool HandleAnalogInputEvent(FSlateApplication &, const FAnalogInputEvent &) override
69 {
70 return IsActive() && IsGameFocused();
71 }
72 const TCHAR *GetDebugName() const override
73 {
74 return TEXT("FirstRoomTutorial");
75 }
76
77 private:
78 static FString PointerKey(const FPointerEvent &Event)
79 {
80 return FString::Printf(TEXT("%u:%u:%d:"), Event.GetUserIndex(), Event.GetPointerIndex(), Event.IsTouchEvent());
81 }
82 static FString PressKey(const FPointerEvent &Event)
83 {
84 return PointerKey(Event) + Event.GetEffectingButton().ToString();
85 }
86 TSet<FString> Accepted;
87 TSet<FString> Rejected;
88};
