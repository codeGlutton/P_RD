9 // 보유 주사위 id로 정적 데이터를 얻는다. 이미 로드돼 있으면 그걸, 아니면 PrimaryAsset 경로로 동기 로드.
10 // 전투 진입 시 런 보유 주사위를 구성하는 좁은 경로라 동기 로드를 허용한다.
11 const UStaticDiceData* LoadStaticDiceData(const FPrimaryAssetId& DiceId)
12 {
13 if (DiceId.IsValid() == false)
14 {
15 return nullptr;
16 }
17
18 UAssetManager* AssetManager = UAssetManager::GetIfInitialized();
19 if (AssetManager == nullptr)
20 {
21 return nullptr;
22 }
23
24 if (const UStaticDiceData* Loaded = AssetManager->GetPrimaryAssetObject<UStaticDiceData>(DiceId))
25 {
26 return Loaded;
27 }
28
29 // 아직 메모리에 없으면 경로로 동기 로드(전투 진입 시 1회).
30 const FSoftObjectPath AssetPath = AssetManager->GetPrimaryAssetPath(DiceId);
31 return Cast<UStaticDiceData>(AssetPath.TryLoad());
32 }
33}

37void UDicePoolModel::BuildFromDiceIds(const TArray<FPrimaryAssetId>& DiceIds)
38{
39 mDices.Reset();
40 mDices.Reserve(DiceIds.Num());
41
42 /*
43 * 보유 주사위는 전부 데이터에서 가져온다 — 직업 mDiceDatas → 런 mDiceIds로 채워진 그대로.
44 * 개수·면 수·면값·희귀도 모두 각 UStaticDiceData가 정의한다(코드가 종류를 만들지 않는다).
45 * 정적 데이터를 못 찾으면 InitFromStatic 내부가 기본 d6로 폴백한다.
46 */
47 for (const FPrimaryAssetId& DiceId : DiceIds)
48 {
49 const UStaticDiceData* StaticDiceData = LoadStaticDiceData(DiceId);
50
51 UDiceModel* Dice = NewObject<UDiceModel>(this);
52 Dice->InitFromStatic(StaticDiceData, DiceId);
53 mDices.Add(Dice);
54 }
55}
