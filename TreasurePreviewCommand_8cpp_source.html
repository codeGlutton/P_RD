57 void Show(UWorld* World)
58 {
59 // 이미 떠 있으면 닫고 새로 연다
60 if (ShownWidget.IsValid())
61 {
62 ShownWidget->RemoveFromParent();
63 ShownWidget.Reset();
64 }
65
66 UTreasureUIModel* Model = FindLiveModel(World);
67 if (Model == nullptr)
68 {
69 return;
70 }
71
72 UClass* WidgetClass = LoadClass<UTreasureUIWidgetBase>(nullptr, TreasureWidgetPath);
73 if (WidgetClass == nullptr)
74 {
75 UE_LOG(LogRD, Warning,
76 TEXT("RD.TreasurePreview: %s 클래스를 찾지 못했습니다."), TreasureWidgetPath);
77 return;
78 }
79
80 APlayerController* Controller = World->GetFirstPlayerController();
81 UTreasureUIWidgetBase* Widget = Controller != nullptr
82 ? CreateWidget<UTreasureUIWidgetBase>(Controller, WidgetClass)
83 : CreateWidget<UTreasureUIWidgetBase>(World, WidgetClass);
84 if (Widget == nullptr)
85 {
86 UE_LOG(LogRD, Warning, TEXT("RD.TreasurePreview: 위젯 생성에 실패했습니다."));
87 return;
88 }
89
90 Widget->BindUIModel(Model);
91 Widget->OpenUI();
92 ShownWidget = Widget;
93
94 UE_LOG(LogRD, Display, TEXT("RD.TreasurePreview: 보물방 화면을 열었습니다."));
95 }
