30void UCombatPlaybackComponent::ApplySpeed()

33 auto* CurrentMixer = Camera ? Camera->GetTimeScaleComponent() : nullptr;
34 if (Mixer.Get() != CurrentMixer)
35 {
36 if (Mixer.IsValid()) Mixer->ReleaseTimeScaleImmediately(Handle);
37 Mixer = CurrentMixer;
38 Handle.Invalidate();

40 if (Mixer.IsValid())

48void UCombatPlaybackComponent::CycleSpeed()
49{

52 ApplySpeed();

54 {
