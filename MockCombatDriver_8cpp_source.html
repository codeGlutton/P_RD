141void UMockCombatDriver::HandleCommand(ECombatInputType Type, int32 IntPayload)
142{
143 switch (Type)
144 {
145 case ECombatInputType::RollDice:
146 for (FDiceSlotUI& Die : mDice)
147 {
148 const int32 FaceCount = FMath::Max(1, Die.mFaceValues.Num());
149 Die.mRolledFaceIndex = FMath::RandRange(0, FaceCount - 1);
150 Die.mResultValue = Die.mFaceValues.IsValidIndex(Die.mRolledFaceIndex)
151 ? Die.mFaceValues[Die.mRolledFaceIndex]
152 : Die.mRolledFaceIndex + 1;
153 Die.mIsRolled = true;
154 }
155 RebuildDicePush();
156 break;
157
158 case ECombatInputType::ToggleDice:
159 if (mSelectedDice.Contains(IntPayload))
160 {
161 mSelectedDice.Remove(IntPayload);
162 }
163 else if (mDice.IsValidIndex(IntPayload))
164 {
165 mSelectedDice.Add(IntPayload);
166 }
167 RebuildDicePush();
168 break;
169
170 case ECombatInputType::Cancel:
171 mSelectedDice.Reset();
172 RebuildDicePush();
173 break;
174
175 case ECombatInputType::LongPressUnit:
176 if (mUIModel != nullptr)
177 {
178 // HP/스탯은 FUnitDetailUI가 더 이상 보관하지 않는다 — UI가 mUnitId로 FUnitUI를 찾아 읽는다.
179 FUnitDetailUI Detail;
180 Detail.mUnitId = IntPayload;
181 Detail.mName = FText::Format(
182 LOCTEXT("Enemy {0}", "Enemy {0}"),
183 FText::AsNumber(IntPayload));
184 Detail.mLevel = 1;
185 Detail.mPassiveDescriptions = { LOCTEXT("(mock passive)", "(mock passive)") };
186 mUIModel->SetUnitDetail(Detail);
187 }
188 break;
189
190 default:
191 UE_LOG(LogRD, Display, TEXT("MockCombatDriver: command %d (payload %d)"), static_cast<int32>(Type), IntPayload);
192 break;
193 }
194}
