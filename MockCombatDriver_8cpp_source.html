231void UMockCombatDriver::HandleCommand(ECombatInputType Type, int32 IntPayload)
232{
233 switch (Type)
234 {
235 case ECombatInputType::Cancel:
236 break;
237
238 case ECombatInputType::LongPressUnit:
239 if (mUIModel != nullptr)
240 {
241 // HP/스탯은 FUnitDetailUI가 더 이상 보관하지 않는다 — UI가 mUnitId로 FUnitUI를 찾아 읽는다.
242 FUnitDetailUI Detail;
243 Detail.mUnitId = IntPayload;
244 Detail.mName = FText::Format(
245 LOCTEXT("Enemy {0}", "Enemy {0}"),
246 FText::AsNumber(IntPayload));
247 Detail.mLevel = 1;
248 Detail.mPassiveDescriptions = { LOCTEXT("(mock passive)", "(mock passive)") };
249 mUIModel->SetUnitDetail(Detail);
250 }
251 break;
252
253 default:
254 UE_LOG(LogRD, Display, TEXT("MockCombatDriver: command %d (payload %d)"), static_cast<int32>(Type), IntPayload);
255 break;
256 }
257}
