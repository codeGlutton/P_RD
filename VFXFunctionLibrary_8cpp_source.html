138bool UVFXFunctionLibrary::ExecuteVFXTimeline(UVFXTimelineComponent* TimelineComp, const FVFXTimelineExecutionData& TimelineExecutionData, FVFXTimelineEventTarget EventTarget)
139{
140 if (TimelineComp != nullptr && IsVFXPossible(TimelineComp) == true)
141 {
142 // 재생할 때마다 이번 실행 데이터의 배속을 다시 지정
143 TimelineComp->SetPlayRate(TimelineExecutionData.mKeyName, TimelineExecutionData.mPlayRate);
144
145 if (TimelineExecutionData.mDirection == ETimelineDirection::Forward)
146 {
147 if (TimelineExecutionData.mIsPlayFromStart == true)
148 {
149 return TimelineComp->PlayFromStart(TimelineExecutionData.mKeyName, EventTarget);
150 }
151 else
152 {
153 return TimelineComp->Play(TimelineExecutionData.mKeyName, EventTarget);
154 }
155 }
156 else
157 {
158 if (TimelineExecutionData.mIsPlayFromStart == true)
159 {
160 return TimelineComp->ReverseFromEnd(TimelineExecutionData.mKeyName, EventTarget);
161 }
162 else
163 {
164 return TimelineComp->Reverse(TimelineExecutionData.mKeyName, EventTarget);
165 }
166 }
167 }
168 return false;
169}

208TArray<TObjectPtr<UNiagaraComponent>> UVFXFunctionLibrary::SpawnAndExecuteVFX(const FVFXSpawnData& VFXSpawnData, const AActor* TargetActor, UVFXTimelineComponent* TimelineComp, FVFXTimelineEventTarget EventTarget, ETileActorDirection Direction)
209{
210 TArray<TObjectPtr<UNiagaraComponent>> SpawnedNiagaras;
211 if (TargetActor == nullptr)
212 {
213 return SpawnedNiagaras;
214 }
215
216 /* 나이아가라 이펙트 스폰 */
217 SpawnedNiagaras.Reserve(VFXSpawnData.mNiagaraSpawnDatas.Num());
218 for (const FNiagaraSpawnData& NiagaraSpawnData : VFXSpawnData.mNiagaraSpawnDatas)
219 {
220 UNiagaraComponent* NiagaraComp = SpawnNiagaraEffectWithDirection(NiagaraSpawnData, TargetActor, Direction);
221 if (NiagaraComp != nullptr)
222 {
223 SpawnedNiagaras.Add(NiagaraComp);
224 }
225 }
226
227 /* 타임라인 타겟 설정 및 나이아가라 컴포넌트 추가 옵션 반영 */
228 if (VFXSpawnData.mIncludeSpawnedNiagaraInTimeline == true)
229 {
230 for (UNiagaraComponent* SpawnedNiagara : SpawnedNiagaras)
231 {
232 if (SpawnedNiagara != nullptr)
233 {
234 EventTarget.mNiagaraComps.Add(SpawnedNiagara);
235 }
236 }
237 }
238
239 /* 타임라인 실행 */
240 if (TimelineComp != nullptr)
241 {
242 for (const FVFXTimelineExecutionData& TimelineExecutionData : VFXSpawnData.mTimelineExecutionDatas)
243 {
244 ExecuteVFXTimeline(TimelineComp, TimelineExecutionData, EventTarget);
245 }
246 }
247
248 return SpawnedNiagaras;
249}

304UNiagaraComponent* UVFXFunctionLibrary::SpawnNiagaraEffectWithDirection_Internal(const TObjectPtr<UNiagaraSystem>& NiagaraSystem, const AActor* TargetActor, UPrimitiveComponent* TargetComponent, const FName& SocketName, const FTransform& RelativeTransform, bool Attached, ETileActorDirection Direction)
305{
306 if (TargetActor == nullptr || NiagaraSystem == nullptr || IsVFXPossible(TargetActor->GetWorld()) == false)
307 {
308 return nullptr;
309 }
310
311 // 대상 확정
312 USceneComponent* FinalTargetComponent = TargetActor->GetRootComponent();
313 if (TargetComponent != nullptr)
314 {
315 FinalTargetComponent = TargetComponent;
316 }
317
318 // 대상 트랜스폼 수집
319 const FTransform FinalTargetTransform = FinalTargetComponent->GetComponentTransform();
320 const FTransform SocketLocalTransform = FinalTargetComponent->GetSocketTransform(SocketName, RTS_Component);
321
322 // 히트한 방향 Transform
323 const float LocalYaw = StaticCast<uint8>(Direction) * 90.f;
324 const FTransform LocalDirectionTransform(FRotator(0.f, LocalYaw, 0.f).Quaternion());
325
326 // 소켓 Transform을 컴포넌트 축으로 절대 회전 방향으로 돌리고, 그 다음에 로컬 트랜스폼 적용
327 const FTransform FinalLocalTransform = RelativeTransform * LocalDirectionTransform * SocketLocalTransform;
328 // 월드 트랜스폼으로 변환
329 const FTransform FinalWorldTransform = FinalLocalTransform * FinalTargetTransform;
330
331 // 나이아가라 이펙트 소환
332 UNiagaraComponent* NiagaraComponent = nullptr;
333 if (Attached == true)
334 {
335 NiagaraComponent = UNiagaraFunctionLibrary::SpawnSystemAttached(
336 NiagaraSystem,
337 FinalTargetComponent,
338 SocketName,
339 FinalWorldTransform.GetLocation(),
340 FinalWorldTransform.Rotator(),
341 FinalWorldTransform.GetScale3D(),
342 EAttachLocation::KeepWorldPosition,
343 true,
344 ENCPoolMethod::None
345 );
346 }
347 else
348 {
349 NiagaraComponent = UNiagaraFunctionLibrary::SpawnSystemAtLocation(
350 FinalTargetComponent->GetWorld(),
351 NiagaraSystem,
352 FinalWorldTransform.GetLocation(),
353 FinalWorldTransform.Rotator(),
354 FinalWorldTransform.GetScale3D()
355 );
356 }
357
358 return NiagaraComponent;
359}
