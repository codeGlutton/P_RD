229void UGuidedTutorialWidget::NativeTick(const FGeometry& G, float Delta)
230{
231 Super::NativeTick(G, Delta);
232 GestureTime += Delta;
233 const auto Size = G.GetLocalSize();
234 if (!Card || Size.X < 1)
235 return;
236 double PreferredWidth = 460.;
237 // A shop icon can sit in the middle of a short landscape screen. Fit the
238 // callout beside that icon instead of clamping a tall card over its spotlight.
239 if (!ReadingTitle.IsEmpty() && TargetWidget.IsValid())
240 {
241 const auto& TG = TargetWidget->GetCachedGeometry();
242 const auto TA = G.AbsoluteToLocal(TG.LocalToAbsolute(FVector2D::ZeroVector));
243 const auto TB = G.AbsoluteToLocal(TG.LocalToAbsolute(TG.GetLocalSize()));
244 const double SideSpace = FMath::Max(TA.X, Size.X - TB.X) - 108. - 28.;
245 PreferredWidth = FMath::Clamp(SideSpace, 280., 460.);
246 }
247 const double Width = FMath::Min(PreferredWidth, Size.X - 48.);
248 Card->SetWidthOverride(Width);
249 const double Height = FMath::Max(120., Card->GetDesiredSize().Y);
250 const double Pad = 24., Gap = 108.;
251 FVector2D Pos(Pad, FMath::Max(Pad, Size.Y - Height - 145.));
252 bFocus =
253 !bBoard && TargetWidget.IsValid() && !TargetWidget->GetCachedGeometry().GetLocalSize().IsNearlyZero();
254 if (bBoard && WorldFocus.Num() == 4)
255 {
256 FocusA = FVector2D(DBL_MAX, DBL_MAX);
257 FocusB = FVector2D(-DBL_MAX, -DBL_MAX);
258 for (const auto& Point : WorldFocus)
259 {
260 const auto Local = G.AbsoluteToLocal(Point);
261 FocusA.X = FMath::Min(FocusA.X, Local.X); FocusA.Y = FMath::Min(FocusA.Y, Local.Y);
262 FocusB.X = FMath::Max(FocusB.X, Local.X); FocusB.Y = FMath::Max(FocusB.Y, Local.Y);
263 }
264 bFocus = true;
265 }
266 if (bFocus)
267 {
268 if (!bBoard)
269 {
270 const auto& T = TargetWidget->GetCachedGeometry();
271 FocusA = G.AbsoluteToLocal(T.LocalToAbsolute(FVector2D::ZeroVector));
272 FocusB = G.AbsoluteToLocal(T.LocalToAbsolute(T.GetLocalSize()));
273 }
274 const FVector2D Center = (FocusA + FocusB) * .5;
275 // Keep the short callout next to its actual control and inside the viewport.
276 if (FocusA.X > Width + Gap + Pad)
277 Pos = {FocusA.X - Width - Gap, Center.Y - Height * .5};
278 else if (Size.X - FocusB.X > Width + Gap + Pad)
279 Pos = {FocusB.X + Gap, Center.Y - Height * .5};
280 else if (FocusA.Y > Height + Gap + Pad)
281 Pos = {Center.X - Width * .5, FocusA.Y - Height - Gap};
282 else
283 Pos = {Center.X - Width * .5, FocusB.Y + Gap};
284 }
285 // Keep the tactical diagram visible while the player compares its two range modes.
286 if (CurrentStage == EGuidedStage::SelectRange || CurrentStage == EGuidedStage::EffectRange)
287 Pos = FVector2D(Size.X - Width - Pad, Size.Y - Height - Pad);
288 Pos.X = FMath::Clamp(Pos.X, Pad, FMath::Max(Pad, Size.X - Width - Pad));
289 Pos.Y = FMath::Clamp(Pos.Y, Pad, FMath::Max(Pad, Size.Y - Height - Pad));
290 Cast<UCanvasPanelSlot>(Card->Slot)->SetPosition(Pos);
291 BubblePosition = Pos;
292 BubbleSize = {Width, Height};
293}
294int32 UGuidedTutorialWidget::NativePaint(const FPaintArgs& Args, const FGeometry& G, const FSlateRect& Cull,
295 FSlateWindowElementList& Out, int32 Layer, const FWidgetStyle& Style,
296 bool Enabled) const
297{
298 if (!bFocus)
299 return Super::NativePaint(Args, G, Cull, Out, Layer, Style, Enabled);
300
301 const FVector2D ViewSize = G.GetLocalSize();
302 const FVector2D Center = (FocusA + FocusB) * .5;
303 // A small icon still needs a readable spotlight and a clear arrow landing point.
304 const FVector2D HalfSize(FMath::Max(30., (FocusB.X - FocusA.X) * .5 + 12.),
305 FMath::Max(30., (FocusB.Y - FocusA.Y) * .5 + 12.));
306 const FVector2D A = Center - HalfSize, B = Center + HalfSize;
307 FVector2D HoleA = A - FVector2D(8), HoleB = B + FVector2D(8);
308 // Flying enemies extend above their tile. Keep their silhouette visible as well as the tile.
309 if (bBoard && CurrentStage == EGuidedStage::SkillTarget) HoleA.Y -= 110.;
310 HoleA.X = FMath::Clamp(HoleA.X, 0., ViewSize.X);
311 HoleA.Y = FMath::Clamp(HoleA.Y, 0., ViewSize.Y);
312 HoleB.X = FMath::Clamp(HoleB.X, HoleA.X, ViewSize.X);
313 HoleB.Y = FMath::Clamp(HoleB.Y, HoleA.Y, ViewSize.Y);
314
315 // NativePaint is called after Slate children. Cut out the callout as well as the target,
316 // otherwise the screen shade also darkens the instructions and acknowledgement button.
317 const auto& CardGeometry = Card->GetCachedGeometry();
318 const FVector2D CardA = G.AbsoluteToLocal(CardGeometry.LocalToAbsolute(FVector2D::ZeroVector));
319 const FVector2D CardB = G.AbsoluteToLocal(CardGeometry.LocalToAbsolute(CardGeometry.GetLocalSize()));
320 TArray<double> Xs = {0., ViewSize.X, HoleA.X, HoleB.X,
321 FMath::Clamp(CardA.X, 0., ViewSize.X), FMath::Clamp(CardB.X, 0., ViewSize.X)};
322 TArray<double> Ys = {0., ViewSize.Y, HoleA.Y, HoleB.Y,
323 FMath::Clamp(CardA.Y, 0., ViewSize.Y), FMath::Clamp(CardB.Y, 0., ViewSize.Y)};
324 Xs.Sort(); Ys.Sort();
325 auto Inside = [](FVector2D P, FVector2D Min, FVector2D Max)
326 { return P.X >= Min.X && P.X <= Max.X && P.Y >= Min.Y && P.Y <= Max.Y; };
327 static const FSlateColorBrush DimBrush(FLinearColor::White);
328 for (int32 X = 0; X + 1 < Xs.Num(); ++X)
329 for (int32 Y = 0; Y + 1 < Ys.Num(); ++Y)
330 {
331 const FVector2D Position(Xs[X], Ys[Y]), Size(Xs[X+1] - Xs[X], Ys[Y+1] - Ys[Y]);
332 const FVector2D Mid = Position + Size * .5;
333 if (Size.X <= 0 || Size.Y <= 0 || Inside(Mid, HoleA, HoleB) || Inside(Mid, CardA, CardB)) continue;
334 FSlateDrawElement::MakeBox(Out, Layer + 1,
335 G.ToPaintGeometry(FVector2f(Size), FSlateLayoutTransform(FVector2f(Position))),
336 &DimBrush, ESlateDrawEffect::None, FLinearColor(.015f, .02f, .025f, .68f));
337 }
338
339 int32 DrawLayer = Super::NativePaint(Args, G, Cull, Out, Layer + 2, Style, Enabled);
340 const FLinearColor Accent(1.f, .78f, .16f, 1.f), Ivory(1.f, .98f, .85f, 1.f);
341 const FLinearColor Edge(.015f, .012f, .009f, 1.f);
342 auto Line = [&](const TArray<FVector2D>& Points, float Width, FLinearColor Color)
343 {
344 FSlateDrawElement::MakeLines(Out, ++DrawLayer, G.ToPaintGeometry(), Points,
345 ESlateDrawEffect::None, Color, true, Width);
346 };
347 const float Pulse = .5f + .5f * FMath::Sin(GestureTime * 3.5f);
348 auto Outline = [&](const TArray<FVector2D>& Points)
349 {
350 Line(Points, 22.f + 5.f * Pulse, Accent.CopyWithNewOpacity(.18f + .10f * Pulse));
351 Line(Points, 14.f, Edge);
352 Line(Points, 8.f, Accent);
353 Line(Points, 2.f, Ivory);
354 };
355 if (WorldFocus.Num() == 4)
356 {
357 TArray<FVector2D> Diamond;
358 for (const auto& Point : WorldFocus) Diamond.Add(G.AbsoluteToLocal(Point));
359 const FVector2D FirstPoint = Diamond[0];
360 Diamond.Add(FirstPoint);
361 Outline(Diamond);
362 }
363 if (!bBoard)
364 Outline({A, {B.X, A.Y}, B, {A.X, B.Y}, A});
365
366 FVector2D Dir = Center - (BubblePosition + BubbleSize * .5);
367 if (!Dir.Normalize()) Dir = FVector2D(0, 1);
368 // Intersect the rectangle edge; using the smaller half-size lands inside wide controls.
369 const double EdgeDistance = FMath::Min(
370 FMath::Abs(Dir.X) > .001 ? HalfSize.X / FMath::Abs(Dir.X) : DBL_MAX,
