39{

57 }
58
59 // 약화
60 const bool IsOwnerWeakness = AttributeSetComponentModel->HasMatchingGameplayTag(EffectTags::GameplayEffect_StatusEffect_TurnDuration_Debuff_Weakness);
61 const float WeaknessRatio = IsOwnerWeakness == true ? GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_TurnDuration_Debuff_Weakness] : 1.f;
62
63 /* 데미지 적용 */
64 for (const IBoardCombatTarget* OtherCombatTarget : OtherCombatTargets)
65 {
66 UAttributeSetComponentModel* OtherAttributeSetComponentModel = OtherCombatTarget->GetAttributeComponentModel();
67 checkf(OtherAttributeSetComponentModel != nullptr, TEXT("속성 컴포넌트 nullptr"));
68
69 // 취약
70 const bool IsTargetVulnerability = OtherAttributeSetComponentModel->HasMatchingGameplayTag(EffectTags::GameplayEffect_StatusEffect_TurnDuration_Debuff_Vulnerability);
71 const float VulnerabilityRatio = IsTargetVulnerability == true ? GameBalanceSettings->mGlobalStatusEffectSetting.mEffectRatios[EffectTags::GameplayEffect_StatusEffect_TurnDuration_Debuff_Vulnerability] : 1.f;
72
73 // 최종 공격력과 방어력
74 const int32 TotalAttack = FMath::Floor(VulnerabilityRatio * WeaknessRatio * AttributeSetComponentModel->GetAttributeCurrentValue(UUnitAttributeSet::GetAttackFactorAttribute()));
75 const int32 TotalDefense = FMath::Floor(OtherAttributeSetComponentModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefenseAttribute()));
76
77 /* 방어력 까기 */
78 {
79 const float DefenseDiff = -FMath::Min(TotalAttack, TotalDefense);
80 if (DefenseDiff < 0)
81 {
82 TSharedPtr<FTacticalEffectSpec> EffectSpec = AttributeSetComponentModel->MakeOutgoingSpec(UTacticalEffect_Defense::StaticClass(), EffectContext);
83 EffectSpec->mDynamicMagnitude = DefenseDiff;
84 AttributeSetComponentModel->ApplyTacticalEffectSpecToTarget(*EffectSpec, OtherAttributeSetComponentModel);
85 }
86 }
87 /* 체력 까기 */
88 {
89 const float HPDiff = TotalDefense - TotalAttack;
90 if (HPDiff < 0)
91 {
92 TSharedPtr<FTacticalEffectSpec> EffectSpec = AttributeSetComponentModel->MakeOutgoingSpec(UTacticalEffect_HP::StaticClass(), EffectContext);
93 EffectSpec->mDynamicMagnitude = HPDiff;
94 AttributeSetComponentModel->ApplyTacticalEffectSpecToTarget(*EffectSpec, OtherAttributeSetComponentModel);
95 }
96 }
97 }
98
99 /* 포인트를 Factor에서 제거 */
100 {
101 AttributeSetComponentModel->RemoveActiveTacticalEffect(EffectHandle);
102 }
