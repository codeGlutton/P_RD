55void FTacticalAttribute::SetNumericValueChecked(float& NewValue, class UTacticalAttributeSet* Dest) const
56{
57 check(Dest != nullptr);
58
59 FNumericProperty* NumericProperty = CastField<FNumericProperty>(mAttribute.Get());
60 float OldValue = 0.f;
61 if (NumericProperty != nullptr)
62 {
63 void* ValuePtr = NumericProperty->ContainerPtrToValuePtr<void>(Dest);
64 OldValue = *static_cast<float*>(ValuePtr);
65 Dest->PreAttributeChange(*this, NewValue);
66 NumericProperty->SetFloatingPointPropertyValue(ValuePtr, NewValue);
67 Dest->PostAttributeChange(*this, OldValue, NewValue);
68 }
69 else if (IsTacticalAttributeDataProperty(mAttribute.Get()) == true)
70 {
71 FStructProperty* StructProperty = CastField<FStructProperty>(mAttribute.Get());
72 check(StructProperty != nullptr);
73 FTacticalAttributeData* DataPtr = StructProperty->ContainerPtrToValuePtr<FTacticalAttributeData>(Dest);
74 check(DataPtr != nullptr);
75 OldValue = DataPtr->GetCurrentValue();
76 Dest->PreAttributeChange(*this, NewValue);
77 DataPtr->SetCurrentValue(NewValue);
78 Dest->PostAttributeChange(*this, OldValue, DataPtr->GetCurrentValue());
79 }
80 else
81 {
82 check(false);
83 }
84}

424TSubclassOf<UTacticalAttributeSet> FindBestAttributeClass(TArray<TSubclassOf<UTacticalAttributeSet> >& ClassList, FString PartialName)
425{
426 // 1순위: 클래스명 정확 일치.
427 // 부분 포함만으로 고르면 "UnitAttributeSet"가 "PlayerUnitAttributeSet"에도 포함되어
428 // 상속 속성(MaxHP 등)이 파생 클래스로 잘못 매핑된다(반복 순서에 따라 비결정적). 정확 일치를 먼저 본다.
429 for (auto Class : ClassList)
430 {
431 if (Class->GetName() == PartialName)
432 {
433 return Class;
434 }
435 }
436
437 // 2순위: 부분 포함(기존 동작 폴백).
438 for (auto Class : ClassList)
439 {
440 if (Class->GetName().Contains(PartialName) == true)
441 {
442 return Class;
443 }
444 }
445 return nullptr;
446}

448void FTacticalAttributeSetInitterDiscreteLevels::PreloadAttributeSetData(const TArray<UCurveTable*>& CurveData)
449{
450 if (ensure(CurveData.Num() > 0) == false)
451 {
452 return;
453 }
454
455 TArray<TSubclassOf<UTacticalAttributeSet> > ClassList;
456 for (TObjectIterator<UClass> ClassIt; ClassIt; ++ClassIt)
457 {
458 UClass* TestClass = *ClassIt;
459 if (TestClass->IsChildOf(UTacticalAttributeSet::StaticClass()) == true)
460 {
461 ClassList.Add(TestClass);
462 }
463 }
464
465 for (const UCurveTable* CurTable : CurveData)
466 {
467 for (const TPair<FName, FRealCurve*>& CurveRow : CurTable->GetRowMap())
468 {
469 FString RowName = CurveRow.Key.ToString();
470 FString ClassName;
471 FString SetName;
472 FString AttributeName;
473 FString Temp;
474
475 RowName.Split(TEXT("."), &ClassName, &Temp);
476 Temp.Split(TEXT("."), &SetName, &AttributeName);
477
478 if (ensure(!ClassName.IsEmpty() && !SetName.IsEmpty() && !AttributeName.IsEmpty()) == false)
479 {
480 continue;
481 }
482
483 TSubclassOf<UTacticalAttributeSet> Set = FindBestAttributeClass(ClassList, SetName);
484 if (Set == nullptr)
485 {
486 continue;
487 }
488
489 FProperty* Property = FindFProperty<FProperty>(*Set, *AttributeName);
490 if (Property == nullptr)
491 {
492 continue;
493 }
494 else if (!FTacticalAttribute::IsSupportedProperty(Property) == true)
495 {
496 continue;
497 }
498
499 FRealCurve* Curve = CurveRow.Value;
500 FName ClassFName = FName(*ClassName);
501 FAttributeSetDefaultsCollection& DefaultCollection = mDefaults.FindOrAdd(ClassFName);
502
503 float FirstLevelFloat = 0.f;
504 float LastLevelFloat = 0.f;
505 Curve->GetTimeRange(FirstLevelFloat, LastLevelFloat);
506
507 int32 FirstLevel = FMath::RoundToInt32(FirstLevelFloat);
508 int32 LastLevel = FMath::RoundToInt32(LastLevelFloat);
509
510 if (FirstLevel != 1)
511 {
512 UE_LOG(LogTacticalFramework, Warning, TEXT("커브의 초기 레벨은 항상 1이여야 함"));
513 continue;
514 }
515
516 DefaultCollection.mLevelData.SetNum(FMath::Max(LastLevel, DefaultCollection.mLevelData.Num()));
517
518 for (int32 Level = 1; Level <= LastLevel; ++Level)
519 {
520 float Value = Curve->Eval(float(Level));
521
522 FAttributeSetDefaults& SetDefaults = DefaultCollection.mLevelData[Level-1];
523
524 FAttributeDefaultValueList* DefaultDataList = SetDefaults.mDataMap.Find(Set);
525 if (DefaultDataList == nullptr)
526 {
527 DefaultDataList = &SetDefaults.mDataMap.Add(Set);
528 }
529
530 check(DefaultDataList != nullptr);
531 DefaultDataList->AddPair(Property, Value);
532 }
533 }
534 }
535}

537void FTacticalAttributeSetInitterDiscreteLevels::InitAttributeSetDefaults(UAttributeSetComponentModel* AttributeSetComponentModel, FName GroupName, int32 Level, bool InitialInit) const
538{
539 check(AttributeSetComponentModel != nullptr);
540
541 const FAttributeSetDefaultsCollection* Collection = mDefaults.Find(GroupName);
542 if (Collection == nullptr)
543 {
544 UE_LOG(LogTacticalFramework, Warning, TEXT("'%s' 그룹의 속성 기본값을 찾을 수 없음. Default 그룹으로 폴백."), *GroupName.ToString());
545 Collection = mDefaults.Find(FName(TEXT("Default")));
546 }
547 if (Collection == nullptr)
548 {
549 // 그룹/Default 둘 다 없으면 하드 크래시 대신 초기화를 건너뛴다(원래 GAS 동작 복원).
550 UE_LOG(LogTacticalFramework, Error, TEXT("'%s'(및 Default) 그룹의 속성 기본값이 없어 초기화를 건너뜁니다."), *GroupName.ToString());
551 return;
552 }
553
554 if (Collection->mLevelData.IsValidIndex(Level - 1) == false)
555 {
556 UE_LOG(LogTacticalFramework, Warning, TEXT("%d 레벨의 속성 기본 값을 찾을 수 없음"), Level);
557 return;
558 }
559
560 const FAttributeSetDefaults& SetDefaults = Collection->mLevelData[Level - 1];
561 for (const UTacticalAttributeSet* Set : AttributeSetComponentModel->GetSpawnedAttributes())
562 {
563 if (Set == nullptr)
564 {
565 continue;
566 }
567 // 커브의 기본값 목록은 속성을 "선언한" 클래스 단위로 키가 잡힌다.
568 // 파생 AttributeSet(예: UPlayerUnitAttributeSet)은 베이스(UUnitAttributeSet)에서 상속한 속성도 가지므로,
569 // 스폰된 셋의 클래스부터 UTacticalAttributeSet까지 거슬러 올라가며 각 단계의 기본값을 모두 적용한다.
570 for (UClass* SetClass = Set->GetClass();
571 SetClass != nullptr && SetClass->IsChildOf(UTacticalAttributeSet::StaticClass());
572 SetClass = SetClass->GetSuperClass())
573 {
574 const FAttributeDefaultValueList* DefaultDataList = SetDefaults.mDataMap.Find(SetClass);
575 if (!DefaultDataList)
576 {
577 continue;
578 }
579
580 UE_LOG(LogTacticalFramework, Log, TEXT("%s 초기화 중 (%s의 기본값 참조)"), *Set->GetName(), *SetClass->GetName());
581
582 for (auto& DataPair : DefaultDataList->mList)
583 {
584 check(DataPair.mProperty);
585
586 if (Set->ShouldInitProperty(InitialInit, DataPair.mProperty))
587 {
588 FTacticalAttribute AttributeToModify(DataPair.mProperty);
589 AttributeSetComponentModel->SetAttributeBaseValue(AttributeToModify, DataPair.mValue);
590 }
591 }
592 }
593 }

597{
598 const FAttributeSetDefaultsCollection* Collection = mDefaults.Find(GroupName);
599 if (Collection == nullptr)
600 {
601 UE_LOG(LogTacticalFramework, Warning, TEXT("'%s' 그룹의 속성 기본값을 찾을 수 없음. Default 그룹으로 폴백."), *GroupName.ToString());
602 Collection = mDefaults.Find(FName(TEXT("Default")));
603 }
604 if (Collection == nullptr)
605 {
606 // 그룹/Default 둘 다 없으면 하드 크래시 대신 초기화를 건너뛴다(원래 GAS 동작 복원).
607 UE_LOG(LogTacticalFramework, Error, TEXT("'%s'(및 Default) 그룹의 속성 기본값이 없어 초기화를 건너뜁니다."), *GroupName.ToString());
608 return;
609 }
610
611 if (Collection->mLevelData.IsValidIndex(Level - 1) == false)
612 {
613 UE_LOG(LogTacticalFramework, Warning, TEXT("%d 레벨의 속성 기본 값을 찾을 수 없음"), Level);
614 return;
615 }
616
617 const FAttributeSetDefaults& SetDefaults = Collection->mLevelData[Level - 1];
618 for (const UTacticalAttributeSet* Set : AttributeSetComponentModel->GetSpawnedAttributes())
619 {
620 if (Set == nullptr)
621 {
622 continue;
623 }

626 if (DefaultDataList != nullptr)
627 {
628 UE_LOG(LogTacticalFramework, Log, TEXT("%s 초기화 중"), *Set->GetName());
629
630 for (auto& DataPair : DefaultDataList->mList)
631 {
632 check(DataPair.mProperty);
633
634 if (DataPair.mProperty == Attribute.GetUProperty())

636 FTacticalAttribute AttributeToModify(DataPair.mProperty);
637 AttributeSetComponentModel->SetAttributeBaseValue(AttributeToModify, DataPair.mValue);
638 }
639 }

644TArray<float> FTacticalAttributeSetInitterDiscreteLevels::GetAttributeSetValues(UClass* AttributeSetClass, FProperty* AttributeProperty, FName GroupName) const
645{
646 TArray<float> AttributeSetValues;
647 const FAttributeSetDefaultsCollection* Collection = mDefaults.Find(GroupName);
648 checkf(Collection != nullptr, TEXT("해당 속성 기본 값을 찾을 수 없음"));
649
650 for (const FAttributeSetDefaults& SetDefaults : Collection->mLevelData)
651 {
652 const FAttributeDefaultValueList* DefaultDataList = SetDefaults.mDataMap.Find(AttributeSetClass);
653 if (DefaultDataList != nullptr)
654 {
655 for (auto& DataPair : DefaultDataList->mList)
656 {
657 check(DataPair.mProperty);
