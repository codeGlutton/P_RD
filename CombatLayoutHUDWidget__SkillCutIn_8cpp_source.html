79bool UCombatLayoutHUDWidget::EnsureSkillCutInWidget()
80{
81 if (IsValid(mSkillCutInWidget) == false)
82 {
83 if (APlayerController* OwningPlayer = GetOwningPlayer())
84 {
85 mSkillCutInWidget = CreateWidget<USkillCutInWidget>(OwningPlayer);
86 }
87 else if (UWorld* World = GetWorld())
88 {
89 mSkillCutInWidget = CreateWidget<USkillCutInWidget>(World);
90 }
91 }
92
93 if (IsValid(mSkillCutInWidget) == false)
94 {
95 return false;
96 }
97 mSkillCutInWidget->SetCombatPlaybackSpeed(mUIModel && mUIModel->IsPlaybackSpeedAvailable() ? mUIModel->GetPlaybackSpeed() : 1);
98
99 if (mSkillCutInWidget->IsInViewport() == false)
100 {
101 mSkillCutInWidget->AddToViewport(SkillCutInZOrder);
102 mSkillCutInWidget->SetVisibility(ESlateVisibility::Collapsed);
103 }
104 return true;

107void UCombatLayoutHUDWidget::HandlePrePlaySkillCutIn(
108 const FCombatSkillCutInRequest& Request,
109 TSharedPtr<FPresentationBarrier> Barrier)
110{
111 // 앞 컷인이 배리어를 붙잡은 동안에는 정상 흐름에서 다음 요청이
112 // 오지 않지만, 별도 캐스터나 중복 브로드캐스트가 겹쳐도 두 번째
113 // 스킬이 컷인 뒤에서 먼저 실행되면 안 된다. 새 배리어도 현재 컷인의
114 // 종료 시점까지 함께 보관해 모든 본 실행을 같은 경계 뒤로 직렬화한다.
115 if (mSkillCutInPlaying || mSkillCutInBarrier.IsValid())
116 {
117 UE_LOG(LogRD, Warning,
118 TEXT("Skill cut-in request coalesced while another cut-in is playing (UnitId=%d, SkillIndex=%d)."),
119 Request.UnitId,
120 Request.SkillIndex);
121 if (Barrier.IsValid())
122 {
123 mOverlappingSkillCutInBarriers.Add(MoveTemp(Barrier));
124 }
125 return;
126 }
127
128 mSkillCutInBarrier = MoveTemp(Barrier);
129 if (EnsureSkillCutInWidget() == false)
130 {
131 UE_LOG(LogRD, Warning, TEXT("Skill cut-in widget could not be created; continuing skill immediately."));
132 FinishSkillCutIn();
133 return;
134 }
135
136 const FSkillCutInPresentationData Presentation = MakeSkillCutInPresentation(Request);
137 mSkillCutInPlaying = true;
138 const bool bStarted = mSkillCutInWidget->PlayCutIn(
139 Presentation,
140 FOnSkillCutInFinished::CreateUObject(this, &UCombatLayoutHUDWidget::FinishSkillCutIn));
141 if (bStarted == false)
142 {
143 UE_LOG(LogRD, Warning, TEXT("Skill cut-in could not start; continuing skill immediately."));
144 FinishSkillCutIn();
145 return;
146 }
147
148 // PlayCutIn 이 동기적으로 완료를 통지하는 커스텀 위젯이어도 종료된
149 // 상태에 다시 타이머를 걸지 않는다.
150 if (mSkillCutInPlaying == false)
151 {
152 return;
153 }
154
155 if (UWorld* World = GetWorld())
156 {
157 const float SafetyDelay = FMath::Max(
158 Presentation.FailSafeSeconds + HudSafetyMarginSeconds,
159 Presentation.DurationSeconds + HudSafetyMarginSeconds * 2.0f);
160 World->GetTimerManager().SetTimer(
161 mSkillCutInSafetyTimerHandle,
162 FTimerDelegate::CreateUObject(this, &UCombatLayoutHUDWidget::FinishSkillCutIn),
163 SafetyDelay,
164 false);
165 return;
166 }
167
168 // 타이머를 소유할 월드가 없으면 완료 보장을 할 수 없으므로 fail-open.
169 FinishSkillCutIn();
170}

172void UCombatLayoutHUDWidget::FinishSkillCutIn()
173{
174 if (mSkillCutInPlaying == false && mSkillCutInBarrier.IsValid() == false)
175 {
176 return;
177 }
178
179 if (UWorld* World = GetWorld())
180 {
181 World->GetTimerManager().ClearTimer(mSkillCutInSafetyTimerHandle);
182 }
183
184 mSkillCutInPlaying = false;
185 if (IsValid(mSkillCutInWidget))
186 {
187 mSkillCutInWidget->StopCutIn(/*bNotifyCompletion=*/false);
188 mSkillCutInWidget->SetVisibility(ESlateVisibility::Collapsed);
189 }
190
191 // Reset 순간 스킬 본 실행이 동기적으로 재개될 수 있다. HUD 상태와
192 // 위젯을 먼저 정리하고, 배리어 해제를 반드시 마지막 작업으로 둔다.
193 TSharedPtr<FPresentationBarrier> PrimaryBarrier = MoveTemp(mSkillCutInBarrier);
194 TArray<TSharedPtr<FPresentationBarrier>> OverlappingBarriers = MoveTemp(mOverlappingSkillCutInBarriers);
195 mOverlappingSkillCutInBarriers.Reset();
196 PrimaryBarrier.Reset();
197 for (TSharedPtr<FPresentationBarrier>& OverlappingBarrier : OverlappingBarriers)
198 {
199 OverlappingBarrier.Reset();
200 }
