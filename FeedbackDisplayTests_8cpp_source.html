37void Capture(UUserWidget *Widget, const TCHAR *Name)
38{
39 if (!Widget || GUsingNullRHI)
40 return;
41 const auto Slate = Widget->TakeWidget();
42 Slate->SetCanTick(false);
43 FAssetCompilingManager::Get().FinishAllCompilation();
44 TArray<UWidget *> Children;
45 Widget->WidgetTree->GetAllWidgets(Children);
46 for (UWidget *Child : Children)
47 if (auto *Image = Cast<UImage>(Child))
48 if (auto *Texture = Cast<UTexture2D>(Image->GetBrush().GetResourceObject()))
49 {
50 Texture->SetForceMipLevelsToBeResident(30.f);
51 Texture->WaitForStreaming();
52 }
53 FWidgetRenderer Renderer(true, false);
54 auto *Target = Renderer.DrawWidget(Slate, FVector2D(1920, 1080));
55 for (int32 Frame = 0; Frame < 3; ++Frame)
56 Renderer.DrawWidget(Target, Slate, FVector2D(1920, 1080), 0.016f);
57 // Offscreen rendering paints widgets without the application's regular tick.
58 for(auto* Child:Children) if(auto* Scroll=Cast<UScrollBox>(Child))
59 if(auto SlateScroll=Scroll->GetCachedWidget()) SlateScroll->Tick(SlateScroll->GetPaintSpaceGeometry(),0,0.016f);
60 Renderer.DrawWidget(Target,Slate,FVector2D(1920,1080),0.016f);
61 FBufferArchive Bytes;
62 FImageUtils::ExportRenderTarget2DAsPNG(Target, Bytes);
63 const FString Dir = FPaths::ProjectSavedDir() / TEXT("UI/Feedback");
64 IFileManager::Get().MakeDirectory(*Dir, true);
65 FFileHelper::SaveArrayToFile(Bytes, *(Dir / FString(Name)));
66}
