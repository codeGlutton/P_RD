235 /* Case1: 밟으면 발동, 기절로 그 칸에서 정지 */
236 AddInfo(TEXT("=== Case1: 트랩 밟음 -> 기절 부여, 수명 차감, 트랩 칸에서 정지 ==="));
237
238 // (2,2)의 기절 트랩(수명 1)을 (1,2)->(3,2) 경로 도중에 밟음

249 TestEqual(TEXT("[Case1] 수명 1 -> 0"), Fixture.Gimmick->GetRemainingTriggerCount(), 0);
250 TestTrue(TEXT("[Case1] 기절로 트랩 칸에서 정지"), Fixture.Unit->GetTileTransform().mIndex == FTileIndex(2, 2));
251 TestFalse(TEXT("[Case1] 이동 종료 상태"), Fixture.Movement->IsMoving());
252 TestTrue(TEXT("[Case1] 수명 소진으로 사망 태그"), Fixture.Gimmick->IsDead());
253
254 /* Case2: 소진된 트랩은 미발동 */
255 AddInfo(TEXT("=== Case2: 소진 후 두 번째 유닛 통과 -> 미발동 ==="));
256
257 // 한 번 밟은 트랩이 발동하지 않는 걸 봐야하는데, 첫 유닛이 기절해서 자리를 차지하고 있으니까 치움
258 // 두 번째 유닛이 진입해도 죽은 트랩은 발동하지 않음
259 Fixture.TileMap->RemoveActor(Fixture.Unit);
260
261 UMockUnitMovementComponentModel* SecondMovement = nullptr;
262 UMockGimmickVictimUnitModel* SecondUnit = MakeVictimUnit(World, Fixture.TileMap, FTileTransform(FTileIndex(1, 2), ETileActorDirection::Forward), SecondMovement);
263
264 SecondMovement->MoveAlongPath({ FTileIndex(1, 2), FTileIndex(2, 2) });
265
266 TestFalse(TEXT("[Case2] 두 번째 유닛은 기절 없음"), HasStunTag(SecondUnit));
267 TestEqual(TEXT("[Case2] 수명 0 유지"), Fixture.Gimmick->GetRemainingTriggerCount(), 0);
268
269 return true;
270}

450 // @brief 시전자 유닛을 배치하고 대상에게 당기기 이펙트 적용
451 auto ApplyPull = [World](UTileMapModel* TileMap, const FTileIndex& CasterTile, UMockGimmickVictimUnitModel* Victim, int32 PullDistance)
452 {
453 UMockEnemyUnitModel* Caster = NewObject<UMockEnemyUnitModel>(World);

455 Caster->BeginPlay();
456 TileMap->PlaceActor(FTileTransform(CasterTile, ETileActorDirection::Forward), Caster);

459 UAttributeSetComponentModel* CasterAttr = Caster->GetAttributeComponentModel();
460 TSharedPtr<FTacticalEffectSpec> Spec = CasterAttr->MakeOutgoingSpec(UTacticalEffect_GetPull::StaticClass(), CasterAttr->MakeEffectContext());
461 Spec->mDynamicMagnitude = static_cast<float>(PullDistance);
462 CasterAttr->ApplyTacticalEffectSpecToTarget(*Spec, Victim->GetAttributeComponentModel());
463 };

465 /* Case1: 발판 없는 직선 당기기 */
466 AddInfo(TEXT("=== Case1: (2,1) 대상을 (2,5) 시전자가 당김 -> (2,4) 도착, 방향 유지 ==="));

473 MakePushSkillData(World, 2),
474 1);
