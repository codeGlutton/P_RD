37bool FMediaCacheContentTest::RunTest(const FString& Parameters)
38{
39 FMediaFixture Fixture;
40 const FString Source = Fixture.Path(TEXT("source.mp4"));
41 const FString Cache = Fixture.Path(TEXT("cache/video.mp4"));
42 const TArray<uint8> First = {1, 2, 3, 4};
43 const TArray<uint8> Update = {5, 6, 7, 8};
44 TestTrue(TEXT("Create source"), FFileHelper::SaveArrayToFile(First, *Source));
45 TestTrue(TEXT("First install creates cache"), RDUITexture::EnsureCachedMediaFile(Source, Cache));
46 TestTrue(TEXT("Unchanged content reuses cache"), RDUITexture::EnsureCachedMediaFile(Source, Cache));
47 TestTrue(TEXT("Replace with equal-length source"), FFileHelper::SaveArrayToFile(Update, *Source));
48 TestTrue(TEXT("Equal-length update refreshes cache"), RDUITexture::EnsureCachedMediaFile(Source, Cache));
49 TArray<uint8> Actual;
50 TestTrue(TEXT("Read updated cache"), FFileHelper::LoadFileToArray(Actual, *Cache));
51 TestTrue(TEXT("Updated bytes are used"), Actual == Update);
52 TestTrue(TEXT("Corrupt cache without changing length"), FFileHelper::SaveArrayToFile(First, *Cache));
53 TestTrue(TEXT("Equal-length corruption is repaired"), RDUITexture::EnsureCachedMediaFile(Source, Cache));
54 FFileHelper::LoadFileToArray(Actual, *Cache);
55 TestTrue(TEXT("Repair restores source bytes"), Actual == Update);
56 IFileManager::Get().Delete(*Source);
57 TestFalse(TEXT("Missing source does not validate stale cache"), RDUITexture::EnsureCachedMediaFile(Source, Cache));
58 FFileHelper::LoadFileToArray(Actual, *Cache);
59 TestTrue(TEXT("Failed refresh retains last complete cache"), Actual == Update);
60 return !HasAnyErrors();
61}
