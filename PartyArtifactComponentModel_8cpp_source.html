56bool UPartyArtifactComponentModel::RemoveArtifact(UStaticArtifactData* Data)
57{

62 return false;

65
66 // 파티 구성원 전원에서 해제
67 TArray<TObjectPtr<UPlayerUnitModel>>& PartyMembers = GetPartyMembers();
68 for (UPlayerUnitModel* PartyMember : PartyMembers)

71 {
72 continue;

76 ArtifactComp->Unequip(Data);

78 }

80 // 목록 변경 통지 (해제 배포까지 끝난 뒤 최신 목록 전달)
