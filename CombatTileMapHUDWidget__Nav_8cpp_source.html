39void UCombatTileMapHUDWidget::BindVictoryFlowEvents()
40{
41 mVictoryWorldMapLocked = false;
42 mCombatResultFlowActive = false;
43
44 ACombatGameMode* CombatGameMode = GetWorld() != nullptr ? GetWorld()->GetAuthGameMode<ACombatGameMode>() : nullptr;
45 if (CombatGameMode == nullptr)
46 {
47 return;
48 }
49
50 CombatGameMode->OnEndCombatUI.RemoveAll(this);
51 CombatGameMode->OnEndCombatUI.AddWeakLambda(this, [this](TSharedPtr<FPresentationBarrier> Barrier, ESRPGCombatResult Result)
52 {
53 HandleEndCombatUI(MoveTemp(Barrier), Result);
54 });
55
56 // 라운드 시작 배너: 게임모드가 데이터(mRound) 갱신 후 재방송하는 OnBeginAnyRoundUI를 구독한다.
57 // 배리어를 붙잡고 배너를 재생 → 배너 종료 시(FinishTurnChangeIntro) 배리어를 놓아 그 라운드 첫 턴이 진행된다.
58 // (프레임워크가 아직 OnBeginAnyRoundUI를 방송하지 않으면 이 핸들러는 호출되지 않음 = 기존 동작 유지)
59 if (CombatGameMode != nullptr)
60 {
61 CombatGameMode->OnBeginAnyRoundUI.RemoveAll(this);
62 CombatGameMode->OnBeginAnyRoundUI.AddWeakLambda(this, [this](TSharedPtr<FPresentationBarrier> Barrier, int32 /*RoundCount*/)
63 {
64 mRoundChangeBarrier.Reset();
65 mRoundChangeBarrier = MoveTemp(Barrier);
66 // 배너를 못 틀면(프레임 에셋 없음 등) 배리어를 즉시 놓아 라운드/턴이 멈추지 않게 한다.
67 if (PlayTurnChangeIntro(false) == false)
68 {
69 mRoundChangeBarrier.Reset();
70 }
71 });
72 }
73}

140void UCombatTileMapHUDWidget::ToggleWorldMap()
141{
142 if (mCombatResultFlowActive)
143 {
144 return;
145 }
146
147 UFrontendMapWidget* WorldMapWidget = Cast<UFrontendMapWidget>(GetToggleableWorldWidget(EWorldWidgetType::WorldMap));

172
173 CloseFloatingPanels(EWorldWidgetType::WorldMap);
174 WorldMapWidget->OnCloseRequested.AddUniqueDynamic(this, &UCombatTileMapHUDWidget::HandleWorldMapCloseRequested);
175 WorldMapWidget->SetRoomSelectionEnabled(false);
176 WorldMapWidget->ClearMapStatusOverride();
177 WorldMapWidget->OpenUI();
178 WorldMapWidget->RefreshMap();
179 SetCombatPlayControlsVisible(false); // 지도 열림 → 탑바만 남기고 전투 컨트롤 숨김(풀스크린 지도 뷰).

386void UCombatTileMapHUDWidget::SetCombatPlayControlsVisible(bool bVisible)
387{
388 mCombatControlsHidden = !bVisible; // 지도 열린 동안 RebuildTurnOrderBar 등이 컨트롤을 다시 만들지 않게 막는다.
389
390 // 지도는 HUD(z=-10)보다 아래(z=-20)에 깔린다. HUD 루트는 평소 Visible(빈 타일맵 탭까지 받으려고)이라
391 // 화면 전체 입력을 먼저 삼켜서, 지도의 드래그 스크롤이 안 먹는다. 지도 열림 동안엔 루트를 SelfHitTestInvisible로
392 // 바꿔 탑바 버튼(자식)만 남기고 나머지 입력을 아래 지도로 흘려보낸다. 닫으면 다시 Visible로 되돌린다.
393 SetVisibility(bVisible ? ESlateVisibility::Visible : ESlateVisibility::SelfHitTestInvisible);
394
395 const ESlateVisibility DisplayVis = bVisible ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed;
396 const ESlateVisibility InputVis = bVisible ? ESlateVisibility::Visible : ESlateVisibility::Collapsed;
397
398 auto ToggleWidgets = [](const auto& Widgets, ESlateVisibility Vis)
399 {
400 for (const auto& Widget : Widgets)
401 {
402 if (Widget != nullptr) { Widget->SetVisibility(Vis); }
403 }
404 };
405
406 // 런타임 컨트롤: 스킬 레일 / 보유 주사위 / 턴 순서 칩 / 장비 칩.
407 ToggleWidgets(mSkillRailPanels, DisplayVis);
408 ToggleWidgets(mSkillRailTexts, DisplayVis);
409 ToggleWidgets(mSkillRailIcons, DisplayVis);
410 ToggleWidgets(mSkillInputButtons, InputVis);
411 ToggleWidgets(mOwnedDiceImages, DisplayVis);
412 ToggleWidgets(mOwnedDiceCardWidgets, InputVis);
413 ToggleWidgets(mOwnedDiceTypeTexts, DisplayVis);
414 ToggleWidgets(mTurnOrderChips, DisplayVis);
415 ToggleWidgets(mTurnOrderChipTexts, DisplayVis);
416 ToggleWidgets(mEquipmentChips, DisplayVis);
417 ToggleWidgets(mEquipmentChipTexts, DisplayVis);
418 ToggleWidgets(mEquipSlotButtons, InputVis);
419
420 // mCombatStatusBarText는 항상 Collapsed(Lv/HP/Gold는 WBP HUD_M_* 라벨이 표시)이므로 여기서 손대지 않는다.
421 // 복원 시 Visible로 켜면 Lv/Gold/HP 필과 같은 좌상단 위치라 빈 텍스트가 필 위에 겹친다.
422 if (mMoveButton != nullptr) { mMoveButton->SetVisibility(InputVis); }
423 if (EndTurnButton != nullptr) { EndTurnButton->SetVisibility(InputVis); }
424
425 // mDiceRollInputButton은 "탭해서 굴리기" 입력영역이라 입장 주사위 오버레이가 실제로 떠 있을 때만 존재해야 한다.
426 // 복원(bVisible)에서 무조건 Visible로 켜면 오버레이가 없는데도 유령 버튼이 살아나, 지도 닫은 직후 탭이 이 버튼에
427 // 꽂혀 StartIntroDiceRoll이 불린다(=주사위 재굴림 버그). 오버레이 상태로 게이트해 그 경로를 끊는다.
428 if (mDiceRollInputButton != nullptr)
429 {
430 const bool bDiceOverlayActive = mIntroDiceRollReady || mIntroDiceRollActive || mIntroDiceResultWaitingForDismiss;
431 mDiceRollInputButton->SetVisibility(
432 bVisible && bDiceOverlayActive ? ESlateVisibility::Visible : ESlateVisibility::Collapsed);
433 }
434
435 // WBP 스킨 마커(프레임 아트) — 지도 위에서는 내비/룸 배너 + 상단 상태바(Lv/Gold/HP)를 남기고 전투 컨트롤만 숨긴다.
436 // (상태바 프리픽스 R_top_status_bar/HUD_M_lv_value/HUD_M_hp_value/HUD_M_gold_value는 제외 → 지도에서도 표시)
437 if (DesignCanvas != nullptr)
438 {
439 static const TArray<FString> ControlPrefixes = {
440 TEXT("R_skill_rail"), TEXT("R_btn_move"), TEXT("R_btn_end_turn"), TEXT("R_turn_order"),
441 TEXT("HUD_SkillRail"), TEXT("HUD_DiceTray"), TEXT("HUD_Move"), TEXT("HUD_EndTurn"), TEXT("HUD_TurnOrder"),
442 TEXT("HUD_M_dice"), TEXT("HUD_M_turn"), TEXT("HUD_M_equip"), TEXT("HUD_M_btn_move"), TEXT("HUD_M_btn_end")
443 };
444 const int32 ChildCount = DesignCanvas->GetChildrenCount();
445 for (int32 ChildIndex = 0; ChildIndex < ChildCount; ++ChildIndex)
446 {
447 UWidget* Child = DesignCanvas->GetChildAt(ChildIndex);
448 if (Child == nullptr) { continue; }
449 const FString ChildName = Child->GetName();
450 for (const FString& Prefix : ControlPrefixes)
451 {
452 if (ChildName.StartsWith(Prefix))
453 {
454 Child->SetVisibility(DisplayVis);
455 break;
456 }
457 }
458 }
459 }
460
461 if (bVisible)
462 {
463 RebuildTurnOrderBar(); // 지도 닫힘 → 그동안 스킵된 턴 순서를 현재 상태로 다시 그린다.
464 }
