195void UCombatUIModel::NotifyCombatFloatingLogs(const TArray<FCombatFloatingLogRequest>& Requests)
196{
197 struct FIndexedFloatingLogRequest
198 {
199 const FCombatFloatingLogRequest* Request = nullptr;
200 int32 InputIndex = 0;
201 };
202
203 TArray<FIndexedFloatingLogRequest> SortedRequests;
204 SortedRequests.Reserve(Requests.Num());
205 for (int32 RequestIndex = 0; RequestIndex < Requests.Num(); ++RequestIndex)
206 {
207 SortedRequests.Add(FIndexedFloatingLogRequest{ &Requests[RequestIndex], RequestIndex });
208 }
209 SortedRequests.Sort([](const FIndexedFloatingLogRequest& Lhs, const FIndexedFloatingLogRequest& Rhs) {
210 if (Lhs.Request->mSequence != Rhs.Request->mSequence)
211 {
212 return Lhs.Request->mSequence < Rhs.Request->mSequence;
213 }
214 return Lhs.InputIndex < Rhs.InputIndex;
215 });
216
217 for (const FIndexedFloatingLogRequest& SortedRequest : SortedRequests)
218 {
219 if (SortedRequest.Request != nullptr)
220 {
221 NotifyCombatFloatingLog(*SortedRequest.Request);
222 }
223 }
224}
