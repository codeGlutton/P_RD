270void UCombatUIModel::NotifyCombatFloatingLogs(const TArray<FCombatFloatingLogRequest>& Requests)
271{
272 struct FIndexedFloatingLogRequest
273 {
274 const FCombatFloatingLogRequest* Request = nullptr;
275 int32 InputIndex = 0;
276 };
277
278 TArray<FIndexedFloatingLogRequest> SortedRequests;
279 SortedRequests.Reserve(Requests.Num());
280 for (int32 RequestIndex = 0; RequestIndex < Requests.Num(); ++RequestIndex)
281 {
282 SortedRequests.Add(FIndexedFloatingLogRequest{ &Requests[RequestIndex], RequestIndex });
283 }
284 SortedRequests.Sort([](const FIndexedFloatingLogRequest& Lhs, const FIndexedFloatingLogRequest& Rhs) {
285 if (Lhs.Request->mSequence != Rhs.Request->mSequence)
286 {
287 return Lhs.Request->mSequence < Rhs.Request->mSequence;
288 }
289 return Lhs.InputIndex < Rhs.InputIndex;
290 });
291
292 for (const FIndexedFloatingLogRequest& SortedRequest : SortedRequests)
293 {
294 if (SortedRequest.Request != nullptr)
295 {
296 NotifyCombatFloatingLog(*SortedRequest.Request);
297 }
298 }
299}
