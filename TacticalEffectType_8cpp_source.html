11float TacticalEffectUtilities::ComputeStackedModifierMagnitude(float BaseComputedMagnitude, int32 StackCount, ETacticalModOp::Type ModOp)
12{
13 // 이 연산의 초기값 계산
14 const float OperationBias = GetModifierBiasByModifierOp(ModOp);
15
16 StackCount = FMath::Clamp<int32>(StackCount, 0, StackCount);
17 float StackMag = BaseComputedMagnitude;
18
19 /*
20 * Override(덮어쓰기)는 스택과 무관하게 항상 같은 값을 강제하므로 보정하지 않는다.
21 * 나머지는 "항등값을 빼서 순수 증분만 남긴 뒤" 스택을 적용하고 다시 항등값을 더한다.
22 */
23 if (ModOp != ETacticalModOp::Override)
24 {
25 // 순수 증가 수치만 추출 (곱셈계열은 -1, 합산계열은 -0)
26 StackMag -= OperationBias;
27 if (ModOp == ETacticalModOp::MultiplyCompound)
28 {
29 // 복리로 스택 누적
30 StackMag = FMath::Pow(StackMag, StackCount);
31 }
32 else
33 {
34 // 선형으로 스택 누적
35 StackMag *= StackCount;
36 }
37 // 다시 초기값 더해 합산
38 StackMag += OperationBias;
39 }
40 return StackMag;
41}
