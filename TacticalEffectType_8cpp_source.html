46float TacticalEffectUtilities::GetModifierBiasByModifierOp(ETacticalModOp::Type ModOp)
47{
48 // 인덱스 = ETacticalModOp 정수값(구 EGameplayModOp와 동일). op 값을 그대로 배열
49 // 첨자로 쓰므로 값이 연속·고정이어야 한다. 배열 길이는 op 개수인 Max(6)와 일치.
50 // 각 슬롯의 항등값(아래 리터럴 순서와 일대일 대응):
51 // [0] AddBase = 0.f (합산 → 0을 더하면 무변화)
52 // [1] MultiplyAdditive = 1.f (배율 가산 → 1배면 무변화)
53 // [2] DivideAdditive = 1.f (나눗셈 가산 → 1로 나누면 무변화)
54 // [3] Override = 0.f (덮어쓰기 → 스택 계산에서 별도 제외, 값 무의미)
55 // [4] MultiplyCompound = 0.f (거듭제곱 곱의 기준값으로 사용)
56 // [5] AddFinal = 0.f (최종 합산 → 0을 더하면 무변화)
57 static const float ModifierOpBiases[ETacticalModOp::Max] = { 0.f, 1.f, 1.f, 0.f, 0.f, 0.f };
58 // 인덱스 범위 보호: op는 [0, Max) 안이어야 배열 오버런이 없다.
59 check(ModOp >= 0 && ModOp < ETacticalModOp::Max);
60
61 return ModifierOpBiases[ModOp];
62}

80float TacticalEffectUtilities::ComputeStackedModifierMagnitude(float BaseComputedMagnitude, int32 StackCount, ETacticalModOp::Type ModOp)
81{
82 // 이 연산의 항등값(곱셈계열=1, 합산계열=0). 증분 분리/복원의 기준이 된다.
83 const float OperationBias = GetModifierBiasByModifierOp(ModOp);
84
85 StackCount = FMath::Clamp<int32>(StackCount, 0, StackCount);
86
87 float StackMag = BaseComputedMagnitude;
88
89 // Override(덮어쓰기)는 스택과 무관하게 항상 같은 값을 강제하므로 보정하지 않는다.
90 // 나머지는 "항등값을 빼서 순수 증분만 남긴 뒤" 스택을 적용하고 다시 항등값을 더한다.
91 if (ModOp != ETacticalModOp::Override)
92 {
93 StackMag -= OperationBias; // 증분만 분리 (곱셈계열은 -1, 합산계열은 -0)
94 if (ModOp == ETacticalModOp::MultiplyCompound)
95 {
96 // 거듭제곱 곱: 스택마다 곱이 누적(복리)되므로 증분을 StackCount 제곱한다.
97 StackMag = FMath::Pow(StackMag, StackCount);
98 }
99 else
100 {
101 // 그 외 모든 연산: 증분을 스택 수에 선형 비례시킨다.
102 StackMag *= StackCount;
103 }
104 StackMag += OperationBias; // 항등값 복원 → 다시 "배율/합산값" 형태로
105 }
106
107 return StackMag;
108}
