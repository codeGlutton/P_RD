40bool FArtifactComponentModelTests::RunTest(const FString& Parameters)
41{
42 FTestObjectScope Scope;
43
44 const FGameplayTag StartTiming = PassiveTiming(TEXT("GameplayAbility.Passive.OnStartTurn"));
45
46 // 필요한 컴포넌트만 유닛 없이 생성
47 UArtifactComponentModel* ArtifactComp = Scope.New<UArtifactComponentModel>();
48 UPassiveComponentModel* PassiveComp = Scope.New<UPassiveComponentModel>();
49 if (!TestNotNull(TEXT("아티팩트 컴포넌트 생성"), ArtifactComp) || !TestNotNull(TEXT("패시브 컴포넌트 생성"), PassiveComp))
50 {
51 return false;
52 }
53
54 // 패시브 DA (코드 구성): 제네릭 패시브 + AttackFactor 이펙트 + 수치 5 + OnStartTurn 시점
55 UStaticPassiveData* PassiveData = Scope.New<UStaticPassiveData>();
56 PassiveData->mPassiveClass = UTacticalPassive_Generic::StaticClass();
57 FPassiveEffectEntry& Effect = PassiveData->mEffects.AddDefaulted_GetRef();
58 Effect.mEffectClass = UTacticalEffect_AttackFactor_AddBase::StaticClass();
59 Effect.mMagnitude.mConst = 5.f;
60 PassiveData->mActivateTimingTag = StartTiming;
61
62 // 아티펙트 DA (코드 구성): 위 패시브를 연관
63 UStaticArtifactData* Artifact = Scope.New<UStaticArtifactData>();
64 Artifact->mStaticPassiveData.Add(TSoftObjectPtr<UStaticPassiveData>(PassiveData));
65
66 // 장착: 패시브가 설치돼야 함
67 const bool bEquipped = ArtifactComp->EquipInternal(Artifact, PassiveComp, nullptr);
68 TestTrue(TEXT("장착 성공"), bEquipped);
69 TestEqual(TEXT("설치된 패시브 1개"), PassiveComp->GetPassives().Num(), 1);
70
71 // 발동 시점 태그로 조회됨 + 정적 데이터가 주입됨
72 TArray<UTacticalPassive*> StartList = PassiveComp->GetPassivesByTiming(StartTiming);
73 TestEqual(TEXT("OnStartTurn 패시브 1개"), StartList.Num(), 1);
74 if (StartList.Num() == 1)
75 {
76 TestEqual(TEXT("주입된 정적 데이터 일치"), StartList[0]->GetStaticData(), PassiveData);
77 }
78
79 // 장착 조회 확인
80 TestNotNull(TEXT("장착 엔트리 조회"), ArtifactComp->GetEquipped(Artifact));
81 TestEqual(TEXT("장착 목록 1개"), ArtifactComp->GetArtifacts().Num(), 1);
82
83 // 중복 장착: 같은 아티펙트를 한 번 더 장착하면 엔트리와 패시브가 2개
84 TestTrue(TEXT("중복 장착 성공"), ArtifactComp->EquipInternal(Artifact, PassiveComp, nullptr));
85 TestEqual(TEXT("장착 목록 2개"), ArtifactComp->GetArtifacts().Num(), 2);
86 TestEqual(TEXT("설치된 패시브 2개"), PassiveComp->GetPassives().Num(), 2);
87
88 // 해제: 첫 엔트리만 해제돼야 함 (중복 장착분 하나 남음)
89 TestTrue(TEXT("해제 성공"), ArtifactComp->UnequipInternal(Artifact, PassiveComp, nullptr));
90 TestEqual(TEXT("해제 후 장착 목록 1개"), ArtifactComp->GetArtifacts().Num(), 1);
91 TestEqual(TEXT("해제 후 패시브 1개"), PassiveComp->GetPassives().Num(), 1);
92
93 // 남은 엔트리 해제 후 빈 상태 재해제는 실패
94 TestTrue(TEXT("남은 엔트리 해제 성공"), ArtifactComp->UnequipInternal(Artifact, PassiveComp, nullptr));
95 TestEqual(TEXT("전부 해제 후 패시브 0개"), PassiveComp->GetPassives().Num(), 0);
96 TestNull(TEXT("전부 해제 후 조회 없음"), ArtifactComp->GetEquipped(Artifact));
97 TestFalse(TEXT("빈 상태 재해제 실패"), ArtifactComp->UnequipInternal(Artifact, PassiveComp, nullptr));
98
99 return true;
100}
