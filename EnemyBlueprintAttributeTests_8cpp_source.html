24bool FEnemyBlueprintAttributeInitializationTest::RunTest(const FString& Parameters)
25{
26 UClass* EnemyClass = LoadClass<UEnemyUnitModel>(nullptr, *Parameters);
27 if (!TestNotNull(TEXT("Cook-warning enemy Blueprint loads"), EnemyClass)) return false;
28 TStrongObjectPtr<UEnemyUnitModel> Enemy(NewObject<UEnemyUnitModel>(GetTransientPackage(), EnemyClass));
29 if (!TestNotNull(TEXT("Enemy instance created"), Enemy.Get())) return false;
30
31 // Follow the same initialization order as FinishCreatingModel, without creating a view or touching assets.
32 Enemy->Initialize();
33 UAttributeSetComponentModel* Attributes = Enemy->GetAttributeComponentModel();
34 if (!TestNotNull(TEXT("Attribute component initialized"), Attributes))
35 {
36 Enemy->Uninitialize();
37 return false;
38 }
39
40 int32 UnitSetCount = 0;
41 bool bOwnsAllSets = true;
42 for (const UTacticalAttributeSet* Set : Attributes->GetSpawnedAttributes())
43 {
44 if (!TestNotNull(TEXT("Registered attribute set is valid"), Set)) continue;
45 AddInfo(FString::Printf(TEXT("Registered set: %s, class=%s, outer=%s"),
46 *Set->GetName(), *Set->GetClass()->GetName(), *GetNameSafe(Set->GetOuter())));
47 bOwnsAllSets &= TestTrue(TEXT("Registered set belongs to the instance"), Set->GetOuter() == Enemy.Get());
48 bOwnsAllSets &= TestFalse(TEXT("Registered set is not a CDO/archetype"), Set->IsTemplate());
49 if (Set->IsA<UUnitAttributeSet>()) ++UnitSetCount;
50 }
51 TestEqual(TEXT("Exactly one Unit attribute family is registered"), UnitSetCount, 1);
52 const FObjectProperty* UnitProperty = FindFProperty<FObjectProperty>(UEnemyUnitModel::StaticClass(), TEXT("mUnitAttributeSet"));
53 if (UnitProperty)
54 {
55 AddInfo(FString::Printf(TEXT("Native member reference: %s"),
56 *GetNameSafe(UnitProperty->GetObjectPropertyValue_InContainer(Enemy.Get()))));
57 }
58
59 if (bOwnsAllSets && UnitSetCount > 0)
60 {
61 constexpr float ExpectedHP = 137.25f;
62 constexpr float ExpectedSpeed = 17.5f;
63 Attributes->ApplyModToAttribute(UUnitAttributeSet::GetMaxHPAttribute(), ETacticalModOp::Override, 300.f);
64 Attributes->ApplyModToAttribute(UUnitAttributeSet::GetHPAttribute(), ETacticalModOp::Override, ExpectedHP);
65 Attributes->ApplyModToAttribute(UUnitAttributeSet::GetRechargeSpeedPointAttribute(), ETacticalModOp::Override, ExpectedSpeed);
66 TestEqual(TEXT("Live HP can be changed after initialization"), Attributes->GetAttributeCurrentValue(UUnitAttributeSet::GetHPAttribute()), ExpectedHP);
67 TStrongObjectPtr<UBoardCombatTargetSnapshotData> Snapshot(Enemy->MakeSnapshotData());
68 if (TestNotNull(TEXT("Combat snapshot created"), Snapshot.Get()))
69 {
70 const float* HP = Snapshot->mAttributes.Find(UUnitAttributeSet::GetHPAttribute());
71 const float* Speed = Snapshot->mAttributes.Find(UUnitAttributeSet::GetRechargeSpeedPointAttribute());
72 if (TestNotNull(TEXT("Snapshot contains HP"), HP)) TestEqual(TEXT("Snapshot HP matches live HP"), *HP, ExpectedHP);
73 if (TestNotNull(TEXT("Snapshot contains recharge speed"), Speed)) TestEqual(TEXT("Snapshot speed matches live speed"), *Speed, ExpectedSpeed);
74 }
75 }
76 Enemy->Uninitialize();
77 return true;
78}
