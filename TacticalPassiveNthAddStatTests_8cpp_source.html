54 // 작업본 하나를 이어서 굴리며 사이클 검증 (threshold=3 → 3·6회차 발동)
55 TInstancedStruct<FDynamicPassiveData> Running;
56 Running.InitializeAs<FDynamicPassiveData_NthCounter>(); // 완료 0에서 시작
57
58 // 한 회차 계산 → 이번 회차 발동 여부 반환 (작업본은 NextState로 갱신되어 다음 회차로 이어짐)
59 // EvaluateActivate는 protected라 friend(테스트 클래스)로 직접 호출
60 auto Step = [&]() -> bool
61 {
62 FPassiveActivateContext Ctx;
63 float Magnitude = 0.f;
64
65 // friend는 서브클래스 override에 적용 안 되므로 베이스 타입으로 호출
66 UTacticalPassive* Base = Passive;
67 const bool bApplied = Base->EvaluateActivate(Ctx, Running, Magnitude);
68
69 // 임계 도달 회차만 적용(true, 보너스 적용), 그 외 false
70 return bApplied;
71 };
72
73 // 7회차 순차 실행: 발동된 회차(1-base) 수집
74 TArray<int32> FiredAt;
75 for (int32 Turn = 1; Turn <= 7; ++Turn)
76 {
77 if (Step())
78 {
79 FiredAt.Add(Turn);
80 }
81 }
82
83 // 7회 중 정확히 2번(3·6회차) 발동
84 TestEqual(TEXT("7회 중 발동 횟수 2"), FiredAt.Num(), 2);
85 if (FiredAt.Num() == 2)
86 {
87 TestEqual(TEXT("첫 발동 = 3회차"), FiredAt[0], 3);
88 TestEqual(TEXT("둘째 발동 = 6회차"), FiredAt[1], 6);
89 }
90
91 // 7회차 후 완료횟수 1 (6회차 리셋→0, 7회차 전진→1)
92 const int32 FinalCount = Running.Get<FDynamicPassiveData_NthCounter>().mCount;
93 TestEqual(TEXT("7회차 후 완료횟수 1"), FinalCount, 1);
94
95 return true;
96}

105bool FTacticalPassiveNthAddStatCommitTests::RunTest(const FString& Parameters)
106{
107 UTacticalPassive_NthAddStat* Passive = MakeNthAddStat();
108 if (!TestNotNull(TEXT("패시브 생성"), Passive))
109 {
110 return false;
111 }

113 // 작업 카운터를 RunningCount로 만들어 커밋 → 커밋된 mState 카운터 반환
114 // mState는 protected라 friend(테스트 클래스)로 직접 접근
115 auto Commit = [&](int32 RunningCount) -> int32
116 {
117 TInstancedStruct<FDynamicPassiveData> Running;
118 Running.InitializeAs<FDynamicPassiveData_NthCounter>();
119 Running.GetMutable<FDynamicPassiveData_NthCounter>().mCount = RunningCount;
120
121 Passive->CommitPassive(Running);
122
123 // mState는 protected라 베이스 타입으로 접근 (friend 적용)
124 const UTacticalPassive* Base = Passive;
125 const TInstancedStruct<FDynamicPassiveData>& Committed = Base->mState;
126 return Committed.IsValid() ? Committed.Get<FDynamicPassiveData_NthCounter>().mCount : -1;
127 };
128
129 // 커밋은 작업본을 그대로 mState에 확정 (리셋은 Evaluate가 NextState에 이미 반영하므로 여기선 안 함)
130 TestEqual(TEXT("커밋: 작업본 1 -> mState 1"), Commit(1), 1);
131 TestEqual(TEXT("커밋: 작업본 2 -> mState 2 (리셋 없음)"), Commit(2), 2);
132
133 return true;
134}
