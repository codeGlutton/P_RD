39 UTexture2D* LoadStatusArt(const FStatusArt& Art)
40 {
41 // Static UObject pointers alone are not GC roots. Keep cached UI art alive.
42 static TMap<FString, TStrongObjectPtr<UTexture2D>> Cache;
43 const FString Path = FString::Printf(TEXT("%sT_Status_%s.T_Status_%s"),
44 TEXT("/Game/SVN/OutSideAsset/AICreation/UI/CombatHUD/StatusIcons/"),
45 Art.Asset, Art.Asset);
46 if (const TStrongObjectPtr<UTexture2D>* Existing = Cache.Find(Path))
47 {
48 return Existing->Get();
49 }
50 UTexture2D* Texture = LoadObject<UTexture2D>(nullptr, *Path);
51 if (Texture)
52 {
53 Cache.Add(Path, TStrongObjectPtr<UTexture2D>(Texture));
54 }
55 return Texture;
56 }

127CombatStatusUI::FPresentation CombatStatusUI::Resolve(
128 const FGameplayTag& StatusTag)
129{
130 FPresentation Result;
131 const FString FullName = StatusTag.GetTagName().ToString();
132 const FString Leaf = StatusLeafName(StatusTag);
133 Result.mDisplayName = LocalizedStatusName(Leaf);
134 Result.mIsBuff = FullName.Contains(TEXT(".Buff."));
135 Result.mIsDebuff = FullName.Contains(TEXT(".Debuff."));
136 Result.mIsRoundDuration = FullName.Contains(TEXT(".RoundDuration."));
137 Result.mIsInfinite = FullName.Contains(TEXT(".Infinite."));
138 Result.mColor = Result.mIsDebuff
139 ? EFloatingLogColorType::Debuff
140 : (Result.mIsBuff ? EFloatingLogColorType::Buff
141 : EFloatingLogColorType::Neutral);
142 Result.mSortPriority = StatusPriority(Leaf, Result.mIsDebuff, Result.mIsBuff);
143
144 if (StatusTag.MatchesTag(EffectTags::GameplayEffect_StatusEffect))
145 {
146 Result.mFloatingIcon = EFloatingLogIconType::Status;
147 for (const FStatusArt& Art : StatusArt)
148 {
149 if (Leaf == Art.Leaf)
150 {
151 Result.mFloatingIcon = Art.Icon;
152 break;
153 }
154 }
155 }
156 return Result;
157}

159FText CombatStatusUI::FormatDelta(const FGameplayTag& StatusTag, const int32 Delta)
160{
161 const FPresentation Presentation = Resolve(StatusTag);
162 if (Delta > 0)
163 {
164 return FText::Format(NSLOCTEXT("CombatStatusUI", "StatusAdded", "{0} +{1}"),
165 Presentation.mDisplayName, FText::AsNumber(Delta));
166 }
167 if (Delta < 0)
168 {
169 return FText::Format(NSLOCTEXT("CombatStatusUI", "StatusRemoved", "{0} -{1}"),
170 Presentation.mDisplayName, FText::AsNumber(FMath::Abs(Delta)));
171 }
172 return Presentation.mDisplayName;
173}

175void CombatStatusUI::SortForDisplay(TArray<FStatusEffectUI>& Statuses)
176{
177 Statuses.StableSort([](const FStatusEffectUI& A, const FStatusEffectUI& B)
178 {
179 const FPresentation APresentation = Resolve(A.mTag);
180 const FPresentation BPresentation = Resolve(B.mTag);
181 if (APresentation.mSortPriority != BPresentation.mSortPriority)
182 {
183 return APresentation.mSortPriority < BPresentation.mSortPriority;
184 }
185 return A.mTag.GetTagName().LexicalLess(B.mTag.GetTagName());
186 });
187}

189UTexture2D* CombatStatusUI::ResolveIcon(const FGameplayTag& StatusTag)
190{
191 if (!StatusTag.MatchesTag(EffectTags::GameplayEffect_StatusEffect))
192 {
193 return nullptr;
194 }
195 const FString Leaf = StatusLeafName(StatusTag);
196 for (const FStatusArt& Art : StatusArt)
197 {
198 if (Leaf == Art.Leaf)
199 {
200 if (UTexture2D* Texture = LoadStatusArt(Art))
201 {
202 return Texture;
203 }
204 }
205 }
206 return nullptr;
207}

209UTexture2D* CombatStatusUI::ResolveIcon(const EFloatingLogIconType IconType)
210{
211 if (IconType == EFloatingLogIconType::Status || IconType == EFloatingLogIconType::None)
212 {
213 return nullptr;
214 }
215 for (const FStatusArt& Art : StatusArt)
216 {
217 if (Art.Icon == IconType)
218 {
219 return LoadStatusArt(Art);
220 }
221 }
222 return nullptr;
