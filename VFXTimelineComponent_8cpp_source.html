25void UVFXTimelineComponent::TickComponent(float DeltaTime, ELevelTick TickType, FActorComponentTickFunction* ThisTickFunction)
26{
27 Super::TickComponent(DeltaTime, TickType, ThisTickFunction);
28
29 if (mIgnoreTimeDilation == true)

458void UCombatTargetVFXTimelineComponent::BindOwnerModel(UObjectModel* Model)
459{
460 UBoardActorModel* BoardActorModel = Cast<UBoardActorModel>(Model);
461 if (BoardActorModel == nullptr)
462 {
463 return;
464 }
465
466 mOwnerModel = BoardActorModel;
467
468 /* 연출 커브 추가 */
469
470 const UGamePlaySettings* GamePlaySettings = GetDefault<UGamePlaySettings>();
471
472 mTimelineSetttings.Append(GamePlaySettings->mCombatTargetVFXTimelineSettings);
473 for (const FCombatTargetVFXTimelineSetting& CombatTargetVFXTimelineSetting : mTimelineSetttings)
474 {
475 UCurveBase* Curve = CombatTargetVFXTimelineSetting.mTimelineCurve.LoadSynchronous();
476 if (UCurveFloat* FloatCurve = Cast<UCurveFloat>(Curve))
477 {
478 AddInterpFloat(CombatTargetVFXTimelineSetting.mTimelineKeyName, FloatCurve, CombatTargetVFXTimelineSetting.mTimelineEvent);
479 }
480 else if (UCurveVector* VectorCurve = Cast<UCurveVector>(Curve))
481 {
482 AddInterpVector(CombatTargetVFXTimelineSetting.mTimelineKeyName, VectorCurve, CombatTargetVFXTimelineSetting.mTimelineEvent);
483 }
484 else if (UCurveLinearColor* ColorCurve = Cast<UCurveLinearColor>(Curve))
485 {
486 AddInterpLinearColor(CombatTargetVFXTimelineSetting.mTimelineKeyName, ColorCurve, CombatTargetVFXTimelineSetting.mTimelineEvent);
487 }
488 }
489
490 /* Remove VFX 연결 */
491
492 if (mOwnerModel.IsValid() == true)
493 {
494 mOwnerModel->OnRemoveTileTransform.AddUObject(this, &UCombatTargetVFXTimelineComponent::PlayRemoveVFX);
495 }
496}
