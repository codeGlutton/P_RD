35 return;

37
38 const int32 PlayerIndex = FCString::Atoi(*Args[0]);
39 const TArray<TObjectPtr<UUnitModel>>& PlayerUnits = CombatModel->GetPlayerUnits();

41 {
42 UE_LOG(LogRD, Warning, TEXT("유닛인덱스 범위 밖: %d (플레이어 %d명, RD.ListUnits로 확인)"), PlayerIndex, PlayerUnits.Num());
43 return;
44 }
45

57 UE_LOG(LogRD, Log, TEXT("텔레포트 %s: [%d] %s -> (%d,%d)"),
58 Started ? TEXT("시작") : TEXT("거부 (이동 중이거나 막힌 타일)"),
59 PlayerIndex, *Unit->GetBoardActorDisplayName().ToString(), TargetIndex.mX, TargetIndex.mY);
60 }
