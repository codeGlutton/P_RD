46void UDiceModel::InitWithFaces(ERarityType Rarity, const FPrimaryAssetId& SourceDiceId, int32 FaceCount, const TArray<int32>& FaceValues, const TArray<TObjectPtr<UTexture>>& FaceTextures)
47{
48 mRarityType = Rarity;
49 mSourceDiceId = SourceDiceId;
50 mFaceCount = SanitizeDiceFaceCount(FaceCount);
51
52 // 면 값: 들어온 배열을 면 수만큼 채우되, 모자란 면은 기본값(index+1)으로 메운다.
53 mFaceValues.Reset(mFaceCount);
54 for (int32 FaceIndex = 0; FaceIndex < mFaceCount; ++FaceIndex)
55 {
56 const int32 FaceValue = FaceValues.IsValidIndex(FaceIndex) ? FaceValues[FaceIndex] : FaceIndex + 1;
57 mFaceValues.Add(FaceValue);
58 }
59
60 // 면 텍스처: 면 값과 같은 길이로 맞춘다. 없는 면은 nullptr(=기본 면 텍스처 사용).
61 mFaceTextures.Reset(mFaceCount);
62 for (int32 FaceIndex = 0; FaceIndex < mFaceCount; ++FaceIndex)
63 {
64 mFaceTextures.Add(FaceTextures.IsValidIndex(FaceIndex) ? FaceTextures[FaceIndex] : nullptr);
65 }
66
67 // 굴림 상태 리셋.
68 mRolledFaceIndex = INDEX_NONE;
69 mCurrentValue = 0;
70 mIsRolled = false;
71 mIsUsed = false;
72}

79void UDiceModel::InitFromStatic(const UStaticDiceData* StaticData, const FPrimaryAssetId& SourceDiceId)
80{
81 const ERarityType Rarity = StaticData != nullptr ? StaticData->mRarityType : ERarityType::Common;
82
83 // 템플릿이 없으면 희귀도만 들고 기본 주사위로 초기화한다(방어적 처리).
84 if (StaticData == nullptr)
85 {
86 Init(Rarity, SourceDiceId);
87 return;
88 }
89
90 const int32 FaceCount = SanitizeDiceFaceCount(StaticData->mFaceCount);
91 TArray<int32> FaceValues;
92 TArray<TObjectPtr<UTexture>> FaceTextures;
93 FaceValues.Reserve(FaceCount);
94 FaceTextures.Reserve(FaceCount);
95 for (int32 FaceIndex = 0; FaceIndex < FaceCount; ++FaceIndex)
96 {
97 if (StaticData->mFaces.IsValidIndex(FaceIndex))
98 {
99 FaceValues.Add(StaticData->mFaces[FaceIndex].mValue);
100 FaceTextures.Add(StaticData->mFaces[FaceIndex].mTexture.LoadSynchronous());
101 }
102 else
103 {
104 // 템플릿이 면 수보다 적게 정의했으면 모자란 면은 기본값으로 메운다.
105 FaceValues.Add(FaceIndex + 1);
106 FaceTextures.Add(nullptr);
107 }
108 }
109
110 InitWithFaces(Rarity, SourceDiceId, FaceCount, FaceValues, FaceTextures);
111}

118int32 UDiceModel::Roll(const FRandomStream& Stream)
119{
120 // 면 값이 비어 있는 비정상 상태면 기본값으로 복구해 크래시를 막는다.
121 if (mFaceValues.IsEmpty())
122 {
123 BuildDefaultFaceValues(mFaceCount, mFaceValues);
124 }
125
126 mRolledFaceIndex = Stream.RandRange(0, mFaceValues.Num() - 1);
127 mCurrentValue = mFaceValues[mRolledFaceIndex];
128 mIsRolled = true;
129 mIsUsed = false; // 새 턴 굴림에서 이전 턴 사용 잠금을 해제.
130 mIsSelected = false; // 새 굴림이면 값이 바뀌므로 이전 스킬 빌드 선택도 무효화한다.
131 return mCurrentValue;
132}
