293{
294 const bool IsFirstTurn = mTurnCount == 0;
295 if (IsFirstTurn == false)

297 /* 필요 시 전투 강제 중단 */

300 if (mShouldTerminateAfterAllPlayersTurnStarted == true && IsPlayerTurn == true)

346 GetCurrentTurnContext()->ForcedSkipAIActions();

349 }));

595void USRPGCombatModel::EvaluateCombatEndState()
596{
597 /* 이미 중단 */
598
599 if (mCombatPhase == ESRPGCombatRoomPhase::CombatAbort)
600 {
601 return;
602 }
603
604 /* 플레이어가 죽어서 전투가 종료되는가? */
605
606 bool AnyPlayerAlive = false;
607 for (const TObjectPtr<UUnitModel>& PlayerUnit : mPlayerUnitModels)
608 {
609 if (PlayerUnit != nullptr && PlayerUnit->IsDead() == false)
610 {
611 AnyPlayerAlive = true;
612 break;
613 }
614 }
615
616 if (AnyPlayerAlive == false)
617 {
618 mCombatResult = ESRPGCombatResult::PlayerLose;
619 mCombatPhase = ESRPGCombatRoomPhase::CombatAbort;
620 return;
621 }
622
623 /* 적군이 모두 죽어 전투가 종료되는가? */
624
625 bool AnyEnemyAlive = false;
626 for (const TObjectPtr<UUnitModel>& Unit : mUnitModels)
627 {
628 if (FGenericTeamId::GetAttitude(EGameTeamType::Adventurer, Unit->GetGenericTeamId()) == ETeamAttitude::Hostile)
629 {
630 if (Unit->IsDead() == false)
631 {
632 AnyEnemyAlive = true;
633 }
634 }
635 }
636
637 if (AnyEnemyAlive == false)
638 {
639 mCombatResult = ESRPGCombatResult::PlayerWin;
640 mCombatPhase = ESRPGCombatRoomPhase::CombatAbort;
641 return;
642 }
643}

683 /* 턴 탐색 */
684
685 if (HasAnyTurnContext() == false)
686 {
687 return false;
688 }
689
690 auto* FoundNode = mTurnContextOrders.GetHead();
691 if (IgnoreCurTurn == true)
692 {
693 FoundNode = FoundNode->GetNextNode();
694 }
695
696 while (FoundNode != nullptr)
697 {
698 USRPGTurnContext* FoundTurnContext = mTurnContextMap[FoundNode->GetValue()];
699 if (FoundTurnContext->GetOwner() == Owner)
700 {
701 FoundTurnContext->OnBeginTurnUI.RemoveAll(this);
702 FoundTurnContext->OnEndTurnUI.RemoveAll(this);
703 FoundTurnContext->OnBeginAnyActionUI.RemoveAll(this);
704 FoundTurnContext->OnEndAnyActionUI.RemoveAll(this);
705 break;
706 }
707 FoundNode = FoundNode->GetNextNode();
708 }
709
710 if (FoundNode == nullptr)
711 {
712 return false;
713 }
714
715 /* 턴 제거 */
716
717 mTurnContextMap.Remove(FoundNode->GetValue());
718 mTurnContextOrders.RemoveNode(FoundNode);
719
720 return true;
721}

733bool USRPGCombatModel::EvaluateRound()
734{
735 // A corrupted or stalled speed configuration must not block the game thread in Shipping.
736 constexpr int32 MaxRechargePasses = 1024;
737 for (int32 RechargePass = 0; RechargePass < MaxRechargePasses; ++RechargePass)
738 {
739 /* 스피드 포인트 충전 */
740
741 for (const TObjectPtr<UUnitModel>& Unit : mUnitModels)
742 {
743 Unit->OnPreEvaluateRound();
744 }
745 for (const TObjectPtr<UBoardActorModel>& Obstacle : mObstacleModels)
746 {
747 Obstacle->OnPreEvaluateRound();
748 }
749
750 /* 현 상태 평가 */
751 if (mFixedPlayerOpening && mRoundCount == 0 && !mPlayerUnitModels.IsEmpty())
752 {
753 // The scripted opening always belongs to party slot 0, regardless of speed rolls.
754 auto* Attributes = mPlayerUnitModels[0]->GetAttributeComponentModel();
755 const float Required = GetDefault<UGameBalanceSettings>()->mRequiredSpeedPointForTurn;
756 const float Missing = Required - Attributes->GetAttributeCurrentValue(UUnitAttributeSet::GetSpeedPointAttribute());
757 if (Missing > 0)
758 {
759 auto Spec = Attributes->MakeOutgoingSpec(UTacticalEffect_SpeedPoint::StaticClass(), Attributes->MakeEffectContext());
760 Spec->mDynamicMagnitude = Missing;
761 Attributes->ApplyTacticalEffectSpecToSelf(*Spec);
762 }
763 }
764
765 TArray<FSRPGTurnCandidate> Candidates;
766 int32 NextRoundRandomSeed = INDEX_NONE;
767
768 const bool IsValid = CheckOrderedTurnCandidates(OUT Candidates, OUT NextRoundRandomSeed);
769 if (!IsValid) break;
770
771 /* 유효 라운드 발견 */
772
773 if (Candidates.IsEmpty() == false)
774 {
775 ApplyOrderedTurnCandidates(Candidates, NextRoundRandomSeed);
776 return true;
777 }
778 }
779 UE_LOG(LogSRPGCombat, Warning, TEXT("Combat speed cannot produce a turn; retaining room entry checkpoint."));
780 mCombatPhase = ESRPGCombatRoomPhase::CombatEnd;
781 OnCombatProgressBlocked.Broadcast();
782 return false;
783}

785bool USRPGCombatModel::CheckOrderedTurnCandidates(OUT TArray<FSRPGTurnCandidate>& Candidates, OUT int32& NextRoundRandomSeed) const
786{
787 const UGameBalanceSettings* GameBalanceSettings = GetDefault<UGameBalanceSettings>();
788 checkf(GameBalanceSettings != nullptr, TEXT("게임 밸런스 세팅 nullptr"));
789
790 Candidates.Empty();
791 NextRoundRandomSeed = INDEX_NONE;
792
793 /* 현재 스피드 수집 */
794
795 TArray<FSRPGTurnCandidate> CurTurnCandidates;
796 bool bCanGenerateTurn = false;
797 for (const TObjectPtr<UUnitModel>& UnitModel : mUnitModels)
798 {
799 const float Remaining = UnitModel->GetAttributeComponentModel()->GetAttributeCurrentValue(UUnitAttributeSet::GetSpeedPointAttribute());
800 const float Recharge = UnitModel->GetAttributeComponentModel()->GetAttributeCurrentValue(UUnitAttributeSet::GetLastRechargedSpeedPointAttribute());
801 if (!FMath::IsFinite(Remaining) || !FMath::IsFinite(Recharge)
802 || Remaining >= static_cast<float>(MAX_int32) || Remaining <= static_cast<float>(MIN_int32)
803 || Recharge >= static_cast<float>(MAX_int32) || Recharge <= static_cast<float>(MIN_int32)) return false;
804 const int32 SpeedPoint = FMath::FloorToInt(Remaining);
805
806 FSRPGTurnCandidate CurTurnCandidate;
807 CurTurnCandidate.mOwner = UnitModel;
808 CurTurnCandidate.mRemainSpeedPoint = SpeedPoint;
809 CurTurnCandidate.mRechargedSpeedPoint = FMath::FloorToInt(Recharge);
810
811 bCanGenerateTurn |= CanAccumulateTurn(
812 Remaining, Recharge,
813 GameBalanceSettings->mRequiredSpeedPointForTurn);
814
815 CurTurnCandidates.Add(CurTurnCandidate);
816 }
817
818 if (!bCanGenerateTurn || GameBalanceSettings->mRequiredSpeedPointForTurn <= 0) return false;
819 /* 후보 추가 */
820
821 const FRandomStream RandomStream(mNextRoundRandomSeed);
822 for (FSRPGTurnCandidate& CurTurnCandidate : CurTurnCandidates)
823 {
824 if (CurTurnCandidate.mRemainSpeedPoint >= GameBalanceSettings->mRequiredSpeedPointForTurn)
825 {
826 FSRPGTurnCandidate FixedTurnCandidate = CurTurnCandidate;
827 FixedTurnCandidate.mRemainSpeedPoint -= GameBalanceSettings->mRequiredSpeedPointForTurn;
828 FixedTurnCandidate.mRandomTieBreaker = RandomStream.FRand();
829
830 Candidates.Add(FixedTurnCandidate);
831 }
832 }
833
834 Candidates.Sort(TGreater<FSRPGTurnCandidate>());
835 if (mFixedPlayerOpening && mRoundCount == 0 && !mPlayerUnitModels.IsEmpty())
836 {
837 const int32 Opening = Candidates.IndexOfByPredicate([this](const FSRPGTurnCandidate& Candidate)
838 { return Candidate.mOwner == mPlayerUnitModels[0]; });
839 if (Opening > 0)
840 {
841 const auto First = Candidates[Opening];
842 Candidates.RemoveAt(Opening);
843 Candidates.Insert(First, 0);
844 }
845 }
846 NextRoundRandomSeed = RandomStream.GetCurrentSeed();
847
848 return true;
849}

851bool USRPGCombatModel::CanAccumulateTurn(float Remaining, float Recharge, float Required)
852{
853 return FMath::IsFinite(Remaining) && FMath::IsFinite(Recharge) && FMath::IsFinite(Required)
854 && Required > 0.f && (Recharge > 0.f || Remaining >= Required);

857TArray<FSRPGPredictedRound> USRPGCombatModel::PredictTurnRounds(
858 const TArray<FSRPGTurnCandidate>& CurrentCandidates,
859 int32 RequiredSpeedPointForTurn,
860 int32 InitialRandomSeed,
861 uint32 RoundCount)
862{
863 TArray<FSRPGPredictedRound> PredictedRounds;
864 if (RoundCount == 0 || RequiredSpeedPointForTurn <= 0
865 || CurrentCandidates.IsEmpty())
866 {
867 return PredictedRounds;
868 }
869
870 TArray<FSRPGTurnCandidate> SimulatedCandidates = CurrentCandidates;
871 int32 SimulatedRandomSeed = InitialRandomSeed;
872 PredictedRounds.Reserve(FMath::Min<uint32>(
873 RoundCount, StaticCast<uint32>(MAX_int32)));
874
875 for (uint32 RoundIndex = 0; RoundIndex < RoundCount; ++RoundIndex)
876 {
877 TArray<FSRPGTurnCandidate> RoundCandidates;
878 while (RoundCandidates.IsEmpty())
879 {
880 bool CanAccumulateSpeed = false;
881 for (FSRPGTurnCandidate& Candidate : SimulatedCandidates)
882 {
883 const int32 Recharge = FMath::Max(Candidate.mRechargedSpeedPoint, 0);
884 CanAccumulateSpeed |= Recharge > 0;
885 Candidate.mRemainSpeedPoint = StaticCast<int32>(FMath::Min<int64>(
886 StaticCast<int64>(Candidate.mRemainSpeedPoint) + Recharge,
887 StaticCast<int64>(MAX_int32)));
888 }
889
890 FRandomStream RandomStream(SimulatedRandomSeed);
891 for (FSRPGTurnCandidate& Candidate : SimulatedCandidates)
892 {
893 if (Candidate.mRemainSpeedPoint < RequiredSpeedPointForTurn)
894 {
895 continue;
896 }
897
898 Candidate.mRemainSpeedPoint -= RequiredSpeedPointForTurn;
899 FSRPGTurnCandidate FixedCandidate = Candidate;
900 FixedCandidate.mRandomTieBreaker = RandomStream.FRand();
901 RoundCandidates.Add(FixedCandidate);
902 }
903
904 if (RoundCandidates.IsEmpty())
905 {
906 if (CanAccumulateSpeed == false)
907 {
908 return PredictedRounds;
909 }
910 continue;
911 }
912
913 RoundCandidates.Sort(TGreater<FSRPGTurnCandidate>());
914 SimulatedRandomSeed = RandomStream.GetCurrentSeed();
915 }
916
917 FSRPGPredictedRound& PredictedRound = PredictedRounds.AddDefaulted_GetRef();
918 PredictedRound.mRoundOffset = StaticCast<int32>(RoundIndex + 1);
919 PredictedRound.mCandidates = MoveTemp(RoundCandidates);
920 }
921
922 return PredictedRounds;
923}
