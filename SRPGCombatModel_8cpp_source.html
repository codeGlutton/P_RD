25void USRPGCombatModel::Serialize(FArchive& Ar)
26{
27 Super::Serialize(Ar);
28
29 if (Ar.IsSaving() == true)
30 {
31 auto* Node = mTurnContextOrder.GetHead();
32 int32 TotalContextNum = mTurnContextOrder.Num();
33 Ar << TotalContextNum;
34 for (int32 i = 0; i < TotalContextNum; ++i)
35 {
36 Ar << Node->GetValue();
37 Node = Node->GetNextNode();
38 }
39 // 시뮬레이션 복제본에서도 현재 턴을 알 수 있게 현재 턴 id를 저장한다.
40 int32 CurrentTurnId = (mCurTurnContextOrder != nullptr) ? mCurTurnContextOrder->GetValue() : INDEX_NONE;
41 Ar << CurrentTurnId;
42 }
43 if (Ar.IsLoading() == true)
44 {
45 int32 TotalContextNum = INDEX_NONE;
46 Ar << OUT TotalContextNum;
47 for (int32 i = 0; i < TotalContextNum; ++i)
48 {
49 int32 TurnIndex = INDEX_NONE;
50 Ar << OUT TurnIndex;
51 mTurnContextOrder.AddTail(TurnIndex);
52 }
53 // 저장한 id로 현재 턴 노드를 다시 찾는다.
54 int32 CurrentTurnId = INDEX_NONE;
55 Ar << OUT CurrentTurnId;
56 mCurTurnContextOrder = (CurrentTurnId != INDEX_NONE) ? mTurnContextOrder.FindNode(CurrentTurnId) : nullptr;
57 }
58}

217void USRPGCombatModel::EvaluateCombatEndState()
218{
219 checkf(mPlayerUnit != nullptr, TEXT("플레이어 미동록 오류"));
220
221 /* 이미 중단 */
222
223 if (mCombatPhase == ESRPGCombatRoomPhase::CombatAbort)
224 {
225 return;
226 }
227
228 /* 플레이어가 죽어서 전투가 종료되는가? */
229
230 if (mPlayerUnit->IsDead() == true)
231 {
232 mCombatResult = ESRPGCombatResult::PlayerLose;
233 mCombatPhase = ESRPGCombatRoomPhase::CombatAbort;
234 return;
235 }
236
237 /* 적군이 모두 죽어 전투가 종료되는가? */
238
239 bool AnyEnemyAlive = false;
240 for (const TObjectPtr<UUnitModel>& Unit : mUnits)
241 {
242 if (Unit->GetTeamAttitudeTowards(*mPlayerUnit) == ETeamAttitude::Hostile)
243 {
244 if (Unit->IsDead() == false)
245 {
246 AnyEnemyAlive = true;
247 }
248 }
249 }
250
251 if (AnyEnemyAlive == false)
252 {
253 mCombatResult = ESRPGCombatResult::PlayerWin;
254 mCombatPhase = ESRPGCombatRoomPhase::CombatAbort;
255 return;
256 }

323bool USRPGCombatModel::UnregisterTurnImmediately(USRPGTurnContext* TurnContext)
324{
325 auto* CurNode = mTurnContextOrder.FindNode(TurnContext->GetTurnId());
326 if (CurNode != nullptr)
327 {
328 // 현재 진행 중인 턴을 삭제하게 되어 노드 전환이 필요한지 여부
329 const bool NeedToChangeTurnContext = mTurnContextMap[mCurTurnContextOrder->GetValue()] == TurnContext;
330
331 auto* NextNode = mTurnContextOrder.RemoveNode(CurNode);
332 mTurnContextMap.Remove(CurNode->GetValue());
333
334 if (NeedToChangeTurnContext == true)
335 {
336 mCurTurnContextOrder = NextNode;
337 }
338
339 TArray<TObjectPtr<USRPGTurnContext>> ActiveTurns = GetTurnContexts(TurnContext->GetOwner());
340 if (ActiveTurns.Num() == 0)
341 {
342 UnregisterUnit(TurnContext->GetOwner());
343 }
344 return true;
345 }
346 return false;
347}

349int32 USRPGCombatModel::UnregisterTurnsImmediately(UUnitModel* Owner)
350{
351 auto* CurNode = mTurnContextOrder.GetHead();
352
353 int32 UnregisterCount = 0;
354 const int32 TotalContextNum = mTurnContextOrder.Num();
355 for (int32 i = 0; i < TotalContextNum; ++i)
356 {
357 auto* NextNode = CurNode->GetNextNode();
358
359 USRPGTurnContext* CurTurnContext = mTurnContextMap[CurNode->GetValue()];
360 if (CurTurnContext->GetOwner() == Owner)
361 {
362 // 현재 진행 중인 턴을 삭제하게 되어 노드 전환이 필요한지 여부
363 const bool NeedToChangeTurnContext = mTurnContextMap[mCurTurnContextOrder->GetValue()] == CurTurnContext;
364
365 mTurnContextMap.Remove(CurNode->GetValue());
366 NextNode = mTurnContextOrder.RemoveNode(CurNode);
367 if (NeedToChangeTurnContext == true)
368 {
369 mCurTurnContextOrder = NextNode;
370 }
371 ++UnregisterCount;
372 }
373
374 CurNode = NextNode;
375 }
376
377 if (UnregisterCount > 0)
378 {
379 UnregisterUnit(Owner);
380 }
381
382 return UnregisterCount;
383}

385void USRPGCombatModel::FlushPendingTurnRequests()
386{
387 while (mPendingTurnRequests.Num() > mHeadRequestIndex)
388 {
389 FSRPGTurnUnregisterRequest& Request = mPendingTurnRequests[mHeadRequestIndex];
390
391 USRPGTurnContext* TargetTurnContext = Request.mTargetTurnContext;
392 UUnitModel* TargetOwner = Request.mTargetOwner;
393 if (TargetTurnContext != nullptr)
394 {
395 UnregisterTurnImmediately(TargetTurnContext);
396 }
397 else if (TargetOwner != nullptr)
398 {
399 UnregisterTurnsImmediately(TargetOwner);
400 }
401
402 ++mHeadRequestIndex;
403 if (mHeadRequestIndex > 5)
404 {
405 mPendingTurnRequests.RemoveAt(0, mHeadRequestIndex);
406 mHeadRequestIndex = 0;
407 }
408 }
409}

544 TSortedMap<int32, TArray<FEnemyUnitPlacementData*>> RegisterMap;

552 for (auto& RegisterPair : RegisterMap)
553 {
554 Algo::RandomShuffle(RegisterPair.Value);

556 {
557 RegisterEnemyUnit(*Data);

596 FlushPendingTurnRequests();
597

599 NotifyRoundEndIfNeeded();
600
601 // 턴 변경
602 mCurTurnContextOrder = mCurTurnContextOrder->GetNextNode();
603 }
604
605 // 강제 중단
606 if (mShouldTerminateBeforePlayerTurnStart == true && mTurnContextMap[mCurTurnContextOrder->GetValue()]->GetOwner()->IsPlayerUnitModel() == true)

608 mShouldTerminateBeforePlayerTurnStart = false;
609 return;
610 }
611
612 auto PresentationBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
613 // 다음 턴 시작
614 mTurnContextMap[mCurTurnContextOrder->GetValue()]->BeginTurn();
615 }));
616
617 // 라운드 시작 시 이벤트 처리
618 NotifyRoundStartIfNeeded(PresentationBarrier);
619}

622{
