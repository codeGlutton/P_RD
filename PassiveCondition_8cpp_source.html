52bool FPassiveCondition::Compare(float Lhs, EPassiveCompareOp Op, float Rhs)
53{
54 switch (Op)
55 {
56 case EPassiveCompareOp::Less: return Lhs < Rhs;
57 case EPassiveCompareOp::LessEqual: return Lhs <= Rhs;
58 case EPassiveCompareOp::Equal: return FMath::IsNearlyEqual(Lhs, Rhs);
59 case EPassiveCompareOp::NotEqual: return FMath::IsNearlyEqual(Lhs, Rhs) == false;
60 case EPassiveCompareOp::GreaterEqual: return Lhs >= Rhs;
61 case EPassiveCompareOp::Greater: return Lhs > Rhs;
62 case EPassiveCompareOp::ModuloZero:
63 {
64 // 피연산자는 float지만 %는 정수 전용이므로 정수로 바꿔서 계산
65 const int32 Divisor = FMath::RoundToInt(Rhs);
66 if (Divisor <= 0)
67 {
68 return false;
69 }
70 return FMath::RoundToInt(Lhs) % Divisor == 0;
71 }
72 default:
73 return false;
74 }
75}

96bool FPassiveCondition::Evaluate(const FPassiveActivateContext& Ctx, int32 TargetIndex, const FDynamicPassiveData_Generic& State) const
97{
98 // 좌변 계산
99 float LhsValue = 0.f;
100 if (mLhs.Resolve(Ctx, TargetIndex, State, LhsValue) == false)
101 {
102 UE_LOG(LogPassive, VeryVerbose, TEXT("조건식 좌변 계산 실패"));
103 return false;
104 }
105
106 // 우변 계산
107 float RhsValue = 0.f;
108 if (mRhs.Resolve(Ctx, TargetIndex, State, RhsValue) == false)
109 {
110 UE_LOG(LogPassive, VeryVerbose, TEXT("조건식 우변 계산 실패"));
111 return false;
112 }
113
114 const bool bPassed = Compare(LhsValue, mOp, RhsValue);
115 UE_LOG(LogPassive, VeryVerbose, TEXT("조건식 %.1f %s %.1f → %s"), LhsValue, ToOpString(mOp), RhsValue, bPassed ? TEXT("통과") : TEXT("탈락"));
116 return bPassed;
117}

119bool FPassiveOperand::Resolve(const FPassiveActivateContext& Ctx, int32 TargetIndex, const FDynamicPassiveData_Generic& State, float& OutValue) const
120{
121 float Value = 0.f;
122
123 // Kind별 계산값
124 switch (mKind)
125 {
126 case EPassiveOperandKind::Const:
127 // 고정값은 배수 없이 그대로 반환
128 OutValue = mConst;
129 return true;
130
131 case EPassiveOperandKind::Counter:
132 Value = static_cast<float>(State.mCounter);
133 break;
134
135 case EPassiveOperandKind::TargetCount:
136 Value = static_cast<float>(Ctx.mTargets.Num());
137 break;
138
139 case EPassiveOperandKind::Attribute:
140 {
141 // 출처 스냅샷에서 속성 현재값. 스냅샷이나 속성이 없으면 실패
142 const UBoardCombatTargetSnapshotData* Snapshot = PassiveConditionUtils::GetSnapshot(Ctx, mSource, TargetIndex);
143 if (Snapshot == nullptr)
144 {
145 return false;
146 }
147 const float* Found = Snapshot->mAttributes.Find(mAttribute);
148 if (Found == nullptr)
149 {
150 return false;
151 }
152 Value = *Found;
153 }
154 break;
155
156 case EPassiveOperandKind::TagCount:
157 {
158 // 출처 스냅샷의 상태이상 스택 중 mTag와 일치(하위 태그 포함)하는 것의 합
159 const UBoardCombatTargetSnapshotData* Snapshot = PassiveConditionUtils::GetSnapshot(Ctx, mSource, TargetIndex);
160 if (Snapshot == nullptr)
161 {
162 return false;
163 }
164 int32 Count = 0;
165 for (const TPair<FGameplayTag, int32>& Pair : Snapshot->mEffectCounts)
166 {
167 if (Pair.Key.MatchesTag(mTag))
168 {
169 Count += Pair.Value;
170 }
171 }
172 Value = static_cast<float>(Count);
173 }
174 break;
175
176 case EPassiveOperandKind::Distance:
177 {
178 // 소유자 타일과 타겟 타일 사이 거리. 둘 중 하나라도 스냅샷이 없으면 실패
179 const UBoardCombatTargetSnapshotData* SelfSnapshot = PassiveConditionUtils::GetSnapshot(Ctx, EPassiveOperandSource::Self, TargetIndex);
180 const UBoardCombatTargetSnapshotData* TargetSnapshot = PassiveConditionUtils::GetSnapshot(Ctx, EPassiveOperandSource::Target, TargetIndex);
181 if (SelfSnapshot == nullptr || TargetSnapshot == nullptr)
182 {
183 return false;
184 }
185 Value = static_cast<float>(PassiveConditionUtils::TileDistance(SelfSnapshot->mTileTransform.mIndex, TargetSnapshot->mTileTransform.mIndex));
186 }
187 break;
188
189 case EPassiveOperandKind::Captured:
190 {
191 // 캡처 키로 저장값 조회. 키가 없거나 타겟 인덱스가 범위 밖이면 실패
192 const FPassiveCaptureSlot* Slot = State.mCaptures.Find(mCaptureKey);
193 if (Slot == nullptr)
194 {
195 return false;
196 }
197 if (mSource == EPassiveOperandSource::Self)
198 {
199 Value = Slot->mSelf;
200 }
201 else
202 {
203 if (Slot->mTargets.IsValidIndex(TargetIndex) == false)
204 {
205 return false;
206 }
207 Value = Slot->mTargets[TargetIndex];
208 }
209 }
210 break;
211
212 case EPassiveOperandKind::MovedDistance:
213 {
214 // 캡처 시점 타일과 현재 타일 사이 거리. 캡처가 없으면(Invalid) 실패
215 const UBoardCombatTargetSnapshotData* Snapshot = PassiveConditionUtils::GetSnapshot(Ctx, mSource, TargetIndex);
216 if (Snapshot == nullptr)
217 {
218 return false;
219 }
220 FTileIndex CapturedTile = FTileIndex::Invalid;
221 if (mSource == EPassiveOperandSource::Self)
222 {
223 CapturedTile = State.mCapturedSelfTile;
224 }
225 else if (State.mCapturedTargetTiles.IsValidIndex(TargetIndex))
226 {
227 CapturedTile = State.mCapturedTargetTiles[TargetIndex];
228 }
229 if (CapturedTile == FTileIndex::Invalid)
230 {
231 return false;
232 }
233 Value = static_cast<float>(PassiveConditionUtils::TileDistance(CapturedTile, Snapshot->mTileTransform.mIndex));
234 }
235 break;
236
237 case EPassiveOperandKind::Team:
238 {
239 // 소유자 기준 타겟의 팀 관계. 모델 포인터가 없거나 전투 대상이 아니면 실패
240 const IBoardCombatTarget* Owner = Cast<IBoardCombatTarget>(Ctx.mOwner.Get());
241 const UBoardActorModel* Target = Ctx.mTargets.IsValidIndex(TargetIndex) ? Ctx.mTargets[TargetIndex].Get() : nullptr;
242 if (Owner == nullptr || Target == nullptr)
243 {
244 return false;
245 }
246 switch (Owner->GetTeamAttitudeTowards(*Target))
247 {
248 case ETeamAttitude::Friendly: OutValue = 1.f; break;
249 case ETeamAttitude::Hostile: OutValue = 2.f; break;
250 default: OutValue = 0.f; break;
251 }
252 // 범주값이라 배수 없이 반환
253 return true;
254 }
255
256 case EPassiveOperandKind::Custom:
257 {
258 // 커스텀 클래스의 Resolve 호출 (인스턴스 생성 없이 CDO 사용)
259 const UPassiveCustomOperand* Custom = mCustomClass != nullptr ? mCustomClass.GetDefaultObject() : nullptr;
260 if (Custom == nullptr || Custom->Resolve(Ctx, TargetIndex, State, Value) == false)
261 {
262 return false;
263 }
264 }
265 break;
266
267 default:
268 return false;
269 }
270
271 // 배수 적용
272 OutValue = Value * mMultiplier;
273 return true;
274}
