49int32 UImportArtifactDataCommandlet::Main(const FString& Params)
50{
51 using namespace ImportArtifactDataPrivate;
52 // -csv 는 필수, -ids 는 선택
53 FString CsvPath;
54 if (FParse::Value(*Params, TEXT("csv="), CsvPath) == false)
55 {
56 UE_LOG(LogDataAssetImport, Error, TEXT("사용법: -run=ImportArtifactData -csv=<경로> [-ids=A001,A002]"));
57 return 1;
58 }
59 FString IdsText;
60 TArray<FString> OnlyIds;
61 if (FParse::Value(*Params, TEXT("ids="), IdsText))
62 {
63 IdsText.ParseIntoArray(OnlyIds, TEXT(","));
64 }
65
66 // CSV 읽기
67 FImportCsvTable Table;
68 if (DataAssetImportUtils::LoadCsv(CsvPath, Table) == false)
69 {
70 return 1;
71 }
72
73 // 필수 컬럼이 없으면 이름을 찍고 중단. 희귀도/가격은 없으면 경고만, 생성 제외 체크박스는 없어도 됨
74 const TCHAR* RequiredColumns[] = {
75 Column::Name, Column::EnglishName, Column::Id, Column::Description, Column::PassiveId,
76 Column::Attribute, Column::ModifierOp, Column::ModifierMagnitude,
77 };
78 bool bColumnsOk = true;
79 for (const TCHAR* Name : RequiredColumns)
80 {
81 if (Table.HasColumn(Name) == false)
82 {
83 UE_LOG(LogDataAssetImport, Error, TEXT("CSV에 컬럼이 없음: %s"), Name);
84 bColumnsOk = false;
85 }
86 }
87 if (bColumnsOk == false)
88 {
89 return 1;
90 }
91 if (Table.HasColumn(Column::RarityType) == false || Table.HasColumn(Column::Price) == false)
92 {
93 UE_LOG(LogDataAssetImport, Warning, TEXT("RarityType/Price 컬럼이 없어 Common/0 으로 채움"));
94 }
95
96 // 행마다 DA 생성 또는 덮어쓰기
97 int32 NumCreated = 0;
98 int32 NumUpdated = 0;
99 int32 NumSkipped = 0;
100 int32 NumFailed = 0;
101 TSet<FString> SeenIds;
102 for (int32 Row = 0; Row < Table.mRows.Num(); ++Row)
103 {
104 const FString Id = Table.Get(Row, Column::Id).TrimStartAndEnd();
105 if (Id.IsEmpty() || (OnlyIds.Num() > 0 && OnlyIds.Contains(Id) == false))
106 {
107 continue;
108 }
109 SeenIds.Add(Id);
110
111 // 생성 제외 체크박스가 켜져 있으면 건너뜀
112 if (DataAssetImportUtils::IsChecked(Table.Get(Row, Column::Exclude)))
113 {
114 UE_LOG(LogDataAssetImport, Display, TEXT("%s: 생성 제외"), *Id);
115 ++NumSkipped;
116 continue;
117 }
118
119 // 에셋 이름: DA_Artifact_<ID>_<영명 파스칼 표기>. 영명이 비어 있으면 ID만 쓰고 경고
120 // 예) DA_Artifact_A001_ChaliceOfLife
121 const FString EnglishName = DataAssetImportUtils::ToPascalCase(Table.Get(Row, Column::EnglishName));
122 if (EnglishName.IsEmpty())
123 {
124 UE_LOG(LogDataAssetImport, Warning, TEXT("%s: 영명이 비어 있어 ID만으로 이름을 만듦. 나중에 채우면 파일이 하나 더 생기니 옛 파일은 삭제 필요"), *Id);
125 }
126 const FString AssetName = EnglishName.IsEmpty() ? AssetPrefix + Id : FString::Printf(TEXT("%s%s_%s"), AssetPrefix, *Id, *EnglishName);
127
128 // DA를 만들거나 로드해서 채우고, 검증한 뒤 저장
129 bool bCreated = false;
130 UStaticArtifactData* Data = DataAssetImportUtils::FindOrCreateAsset<UStaticArtifactData>(PackagePath, AssetName, bCreated);
131 if (Data == nullptr || FillArtifactData(Table, Row, Data) == false)
132 {
133 UE_LOG(LogDataAssetImport, Error, TEXT("%s: 생성 실패"), *Id);
134 ++NumFailed;
135 continue;
136 }
137 DataAssetImportUtils::ValidateAsset(Data);
138 if (DataAssetImportUtils::SaveAsset(Data) == false)
139 {
140 ++NumFailed;
141 continue;
142 }
143 bCreated ? ++NumCreated : ++NumUpdated;
144 }
145
146 // -ids 에 있는데 CSV에 없는 ID 경고
147 for (const FString& Id : OnlyIds)
148 {
149 if (SeenIds.Contains(Id) == false)
150 {
151 UE_LOG(LogDataAssetImport, Warning, TEXT("CSV에 없는 ID: %s"), *Id);
152 }
153 }
154
155 UE_LOG(LogDataAssetImport, Display, TEXT("아티팩트 DA 임포트 끝: 생성 %d, 갱신 %d, 제외 %d, 실패 %d"), NumCreated, NumUpdated, NumSkipped, NumFailed);
156 return (NumFailed == 0) ? 0 : 1;
157}

159bool UImportArtifactDataCommandlet::FillArtifactData(const FImportCsvTable& Table, int32 RowIndex, UStaticArtifactData* Data) const
160{
161 using namespace ImportArtifactDataPrivate;
162 const FString Id = Table.Get(RowIndex, Column::Id).TrimStartAndEnd();
163 bool bOk = true;
164
165 // 이름과 설명
166 Data->mName = FText::FromString(Table.Get(RowIndex, Column::Name).TrimStartAndEnd());
167 Data->mDescription = FText::FromString(Table.Get(RowIndex, Column::Description).TrimStartAndEnd());
168
169 // 희귀도: Common/Rare/Epic. 비었거나 틀리면 Common 으로 채우고 경고
170 const FString RarityText = Table.Get(RowIndex, Column::RarityType).TrimStartAndEnd();
171 const int64 RarityValue = StaticEnum<ERarityType>()->GetValueByNameString(RarityText);
172 if (RarityValue == INDEX_NONE)
173 {
174 UE_LOG(LogDataAssetImport, Warning, TEXT("%s: 희귀도가 비었거나 잘못돼 Common 으로 채움: '%s'"), *Id, *RarityText);
175 }
176 Data->mRarityType = (RarityValue == INDEX_NONE) ? ERarityType::Common : static_cast<ERarityType>(RarityValue);
177
178 // 가격: 숫자가 아니면 0 으로 채우고 경고
179 const FString PriceText = Table.Get(RowIndex, Column::Price).TrimStartAndEnd();
180 if (PriceText.IsNumeric() == false)
181 {
182 UE_LOG(LogDataAssetImport, Warning, TEXT("%s: 가격이 비었거나 잘못돼 0 으로 채움: '%s'"), *Id, *PriceText);
183 }
184 Data->mPrice = PriceText.IsNumeric() ? FCString::Atoi(*PriceText) : 0;
185
186 // 패시브 참조: ID가 비어 있으면 스탯 전용 아티팩트. 있으면 DA_Passive_<ID> 를 소프트 참조
187 // 패시브 DA 파일이 아직 없으면 에러
188 Data->mStaticPassiveData.Reset();
189 const FString PassiveId = Table.Get(RowIndex, Column::PassiveId).TrimStartAndEnd();
190 if (PassiveId.IsEmpty() == false)
191 {
192 const FString PassiveAssetName = PassiveAssetPrefix + PassiveId;
193 const FString PassivePackage = FString(PassivePackagePath) / PassiveAssetName;
194 if (FPackageName::DoesPackageExist(PassivePackage) == false)
195 {
196 UE_LOG(LogDataAssetImport, Error, TEXT("%s: 패시브 DA가 없음: %s"), *Id, *PassivePackage);
197 bOk = false;
198 }
199 Data->mStaticPassiveData.Add(TSoftObjectPtr<UStaticPassiveData>(FSoftObjectPath(PassivePackage + TEXT(".") + PassiveAssetName)));
200 }
201
202 // 스탯 수정자: Attribute 가 비어 있으면 없음. 있으면 속성/연산/크기 하나
203 Data->mStatModifiers.Reset();
204 const FString AttributeText = Table.Get(RowIndex, Column::Attribute).TrimStartAndEnd();
205 if (AttributeText.IsEmpty() == false)
206 {
207 FTacticalModifierInfo Modifier;
208 bOk &= DataAssetImportUtils::ParseAttribute(AttributeText, Modifier.mAttribute);
209
210 const FString OpText = Table.Get(RowIndex, Column::ModifierOp).TrimStartAndEnd();
211 const int64 OpValue = StaticEnum<ETacticalModOp::Type>()->GetValueByNameString(OpText);
212 if (OpValue == INDEX_NONE)
213 {
214 UE_LOG(LogDataAssetImport, Error, TEXT("%s: 알 수 없는 연산: '%s'"), *Id, *OpText);
215 bOk = false;
216 }
217 Modifier.mModifierOp = (OpValue == INDEX_NONE) ? ETacticalModOp::AddBase : static_cast<ETacticalModOp::Type>(OpValue);
218 Modifier.mModifierMagnitude = FCString::Atof(*Table.Get(RowIndex, Column::ModifierMagnitude));
219 Data->mStatModifiers.Add(Modifier);
220 }
221
222 return bOk;
223}
