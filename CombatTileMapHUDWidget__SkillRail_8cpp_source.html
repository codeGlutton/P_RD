92{

94 if (mSelectedSkillIndex != SkillIndex)

104 const int32 SkillDataIndex = GetSkillDataIndexForRailSlot(RailSlotIndex);

106 const bool bOwned = Skill != nullptr && Skill->mName.IsEmpty() == false;
107 const bool bUsable = bOwned && Skill->mIsUsable;

165int32 UCombatTileMapHUDWidget::GetSkillDataIndexForRailSlot(int32 RailSlotIndex) const
166{
167 int32 SkillDataIndex = INDEX_NONE;
168 if (RailSlotIndex == 0)
169 {
170 SkillDataIndex = 0; // 맨 위 고정: 기본 공격(평타)
171 }
172 else if (RailSlotIndex == CombatSkillSlotCount - 1)
173 {
174 SkillDataIndex = 1; // 맨 아래 고정: 기본 이동(STEP)
175 }
176 else if (RailSlotIndex >= 1 && RailSlotIndex < CombatSkillSlotCount - 1)
177 {
178 SkillDataIndex = RailSlotIndex + 1; // 중간 4칸: 추가 스킬(데이터 2..5)을 위에서부터
179 }
180
181 if (mCombatUIModel == nullptr)
182 {
183 return INDEX_NONE;
184 }
185 const TArray<FSkillUI>& Skills = mCombatUIModel->GetSkillUIs();
186 if (Skills.IsValidIndex(SkillDataIndex) == false || Skills[SkillDataIndex].mName.IsEmpty() == true)
187 {
188 return INDEX_NONE;
189 }
190 return SkillDataIndex;
191}
