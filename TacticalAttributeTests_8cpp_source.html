28static UWorld* GetAnyGameWorld()
29{
30 if (GEngine == nullptr)
31 {
32 return nullptr;
33 }
34 for (const FWorldContext& Context : GEngine->GetWorldContexts())
35 {
36 if ((Context.WorldType == EWorldType::PIE || Context.WorldType == EWorldType::Game) && Context.World() != nullptr)
37 {
38 return Context.World();
39 }
40 }
41 return nullptr;
42}

50bool FTacticalAttributeTests::RunTest(const FString& Parameters)
51{
52 // ----------------------------------------------------
53 // 1. 에디터에 이미 존재하는 활성화된 GWorld 환경 사용
54 // ----------------------------------------------------
55 UWorld* World = GetAnyGameWorld();
56 if (World == nullptr)
57 {
58 World = GWorld;
59 }
60 if (World == nullptr)
61 {
62 AddError(TEXT("유효한 UWorld를 찾는 데 실패했습니다."));
63 return false;
64 }
65
66 USimulationSubsystem* SimSubsystem = World->GetSubsystem<USimulationSubsystem>();
67 if (TestNotNull(TEXT("USimulationSubsystem이 초기화되어야 합니다."), SimSubsystem) == false)
68 {
69 return false;
70 }
71
72 URoomInstance* RoomInstance = FSimulationSubsystemTestAccessor::GetRoomInstance(SimSubsystem);
73 if (TestNotNull(TEXT("RoomInstance가 초기화되어야 합니다."), RoomInstance) == false)
74 {
75 return false;
76 }
77
78 // UTacticalFrameworkModel 모크 등록 (기존 등록 상태가 없을 때만 등록)
79 UTacticalFrameworkModel* FrameworkModel = Cast<UTacticalFrameworkModel>(RoomInstance->mAliveSubsystemModels.FindRef(UTacticalFrameworkModel::StaticClass()));
80 if (FrameworkModel == nullptr)
81 {
82 FrameworkModel = NewObject<UTacticalFrameworkModel>(RoomInstance);
83 RoomInstance->mAliveSubsystemModels.Add(UTacticalFrameworkModel::StaticClass(), FrameworkModel);
84 }
85
86 // ----------------------------------------------------
87 // 2. 모델 및 속성 컴포넌트 모델 생성 & 초기화
88 // ----------------------------------------------------
89 UTASActorModelMock* MockActorModel = NewObject<UTASActorModelMock>(World);
90 MockActorModel->Initialize();
91 MockActorModel->BeginPlay();
92
93 UAttributeSetComponentModel* CompModel = MockActorModel->GetAttributeComponent();
94 if (TestNotNull(TEXT("AttributeSetComponentModel이 유효해야 합니다."), CompModel) == false)
95 {
96 return false;
97 }
98
99 // AttributeSet이 정상 부착되었는지 확인
100 TestTrue(TEXT("UUnitAttributeSet이 등록되어야 합니다."), CompModel->HasAttributeSetForAttribute(UUnitAttributeSet::GetHPAttribute()));
101
102 // ----------------------------------------------------
103 // 3. 테스트 케이스 1: 기본 속성 변경 검증 (HP 및 MaxHP 초기화)
104 // ----------------------------------------------------
105 AddInfo(TEXT("=== 테스트 케이스 1: 기본값 및 현재값 초기화 검증 ==="));
106 CompModel->SetAttributeBaseValue(UUnitAttributeSet::GetMaxHPAttribute(), 100.f);
107 CompModel->SetAttributeBaseValue(UUnitAttributeSet::GetHPAttribute(), 100.f);
108
109 TestEqual(TEXT("기본 MaxHP는 100.f이어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetMaxHPAttribute()), 100.f);
110 TestEqual(TEXT("현재 MaxHP는 100.f이어야 합니다."), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetMaxHPAttribute()), 100.f);
111 TestEqual(TEXT("기본 HP는 100.f이어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetHPAttribute()), 100.f);
112 TestEqual(TEXT("현재 HP는 100.f이어야 합니다."), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetHPAttribute()), 100.f);
113
114 // ----------------------------------------------------
115 // 4. 테스트 케이스 2: Instant Effect 적용 검증 (MaxHP 데미지 차감 -20)
116 // ----------------------------------------------------
117 AddInfo(TEXT("=== 테스트 케이스 2: Instant 이펙트 적용 검증 ==="));
118
119 UTacticalEffectContext* EffectContext1 = CompModel->MakeEffectContext();
120 EffectContext1->SetInstigator(MockActorModel);
121 EffectContext1->SetAttributeSetComponentModel(CompModel);
122
123 TSharedPtr<FTacticalEffectSpec> InstantSpec = CompModel->MakeOutgoingSpec(UTestInstantTacticalEffect::StaticClass(), EffectContext1);
124 InstantSpec->mDynamicMagnitude = -20.f;
125 CompModel->ApplyTacticalEffectSpecToSelf(*InstantSpec);
126
127 // Instant Effect는 베이스 값과 현재 값을 모두 영구적으로 변화시킴
128 TestEqual(TEXT("Instant HP 변경 후, 기본 MaxHP는 80.f이어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetMaxHPAttribute()), 80.f);
129 TestEqual(TEXT("Instant HP 변경 후, 현재 MaxHP는 80.f이어야 합니다."), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetMaxHPAttribute()), 80.f);
130
131 // ----------------------------------------------------
132 // 5. 테스트 케이스 3: Infinite Effect 적용 및 해제 검증 (DefensePoint +30)
133 // ----------------------------------------------------
134 AddInfo(TEXT("=== 테스트 케이스 3: Infinite 이펙트 적용 및 제거 검증 ==="));
135 CompModel->SetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute(), 10.f);
136 TestEqual(TEXT("기본 방어력은 10.f이어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
137
138 UTacticalEffectContext* EffectContext2 = CompModel->MakeEffectContext();
139 EffectContext2->SetInstigator(MockActorModel);
140 EffectContext2->SetAttributeSetComponentModel(CompModel);
141
142 TSharedPtr<FTacticalEffectSpec> InfiniteEffect = CompModel->MakeOutgoingSpec(UTestInfiniteTacticalEffect::StaticClass(), EffectContext2);
143 InfiniteEffect->mDynamicMagnitude = 30.f;
144 FActiveTacticalEffectHandle ActiveHandle = CompModel->ApplyTacticalEffectSpecToSelf(*InfiniteEffect);
145
146 TestTrue(TEXT("ActiveHandle이 유효해야 합니다."), ActiveHandle.IsValid());
147 // Infinite Effect는 베이스 값은 그대로 두고 현재 값만 변화시킴 (Aggregator에 가산)
148 TestEqual(TEXT("기본 방어력은 10.f로 유지되어야 합니다."), CompModel->GetAttributeBaseValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
149 TestEqual(TEXT("현재 방어력은 40.f이어야 합니다. (기본 10 + 모디파이어 30)"), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 40.f);
150
151 // 이펙트 제거 후 원래대로 복원되는지 확인
152 CompModel->RemoveActiveTacticalEffect(ActiveHandle);
153 TestEqual(TEXT("Infinite 이펙트 제거 후, 현재 방어력은 10.f로 복구되어야 합니다."), CompModel->GetAttributeCurrentValue(UUnitAttributeSet::GetDefensePointAttribute()), 10.f);
154
155 // ----------------------------------------------------
156 // 6. 테스트 케이스 4: Loose Tag 부여 및 해제 검증
157 // ----------------------------------------------------
158 AddInfo(TEXT("=== 테스트 케이스 4: Loose 태그 추가 및 제거 검증 ==="));
159 // 테스트에 사용할 태그 등록
160 FGameplayTag TestTagA = AbilityTags::GameplayAbility_LevelUp;
161
162 TestFalse(TEXT("초기에는 TestTagA를 가지고 있지 않아야 합니다."), CompModel->HasMatchingGameplayTag(TestTagA));
163
164 CompModel->AddLooseGameplayTag(TestTagA);
165 TestTrue(TEXT("태그 추가 후에는 TestTagA를 가지고 있어야 합니다."), CompModel->HasMatchingGameplayTag(TestTagA));
166 TestEqual(TEXT("TestTagA의 개수는 1이어야 합니다."), CompModel->GetTagCount(TestTagA), 1);
167
168 CompModel->RemoveLooseGameplayTag(TestTagA);
169 TestFalse(TEXT("태그 제거 후에는 TestTagA를 가지고 있지 않아야 합니다."), CompModel->HasMatchingGameplayTag(TestTagA));
170
171 // ----------------------------------------------------
172 // 7. 테스트 케이스 5: Effect를 통한 Granted Tag 부여 및 해제 검증
173 // ----------------------------------------------------
174 AddInfo(TEXT("=== 테스트 케이스 5: Infinite 이펙트에 의한 Granted 태그 부여 검증 ==="));
175 FGameplayTag GrantedTag = AbilityTags::GameplayAbility_Passive_OnEndApplyingEffect;
176
177 UTacticalEffectContext* EffectContext3 = CompModel->MakeEffectContext();
178 EffectContext3->SetInstigator(MockActorModel);
179 EffectContext3->SetAttributeSetComponentModel(CompModel);
180
181 TSharedPtr<FTacticalEffectSpec> TagSpec = CompModel->MakeOutgoingSpec(UTestInfiniteTagEffect::StaticClass(), EffectContext3);
182 FActiveTacticalEffectHandle TagActiveHandle = CompModel->ApplyTacticalEffectSpecToSelf(*TagSpec);
183
184 TestTrue(TEXT("CompModel에 GrantedTag가 활성화되어야 합니다."), CompModel->HasMatchingGameplayTag(GrantedTag));
185 TestEqual(TEXT("GrantedTag의 개수는 1이어야 합니다."), CompModel->GetTagCount(GrantedTag), 1);
186
187 CompModel->RemoveActiveTacticalEffect(TagActiveHandle);
188 TestFalse(TEXT("CompModel에서 GrantedTag가 제거되어야 합니다."), CompModel->HasMatchingGameplayTag(GrantedTag));
189
190 // ----------------------------------------------------
191 // 8. 환경 정리 (Clean up)
192 // ----------------------------------------------------
193 MockActorModel->EndPlay();
194 MockActorModel->Uninitialize();
195
196 CompModel->MarkAsGarbage();
197 MockActorModel->MarkAsGarbage();
198
199 if (RoomInstance != nullptr)
200 {
201 RoomInstance->mAliveSubsystemModels.Remove(UTacticalFrameworkModel::StaticClass());
202 FrameworkModel->MarkAsGarbage();
203 }
204
205 return true;
206}
