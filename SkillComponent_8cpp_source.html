81bool USkillComponent::TestActivateSkill(int32 SkillIndex, TArray<AUnit*> UnitArray)
82{
83 if (!GetOwner())
84 return false;
85
86 // 어빌리티를 발동시킨다.
87 FGameplayEventData EventData;
88
89 EventData.Instigator = GetOwner(); // 사용자
90
91 EventData.EventTag = FGameplayTag::RequestGameplayTag(TEXT("Test.GameplayAbility.Skill"));
92
93 // 2. 빈 타겟 데이터 구조체를 동적 할당합니다.
94 FGameplayAbilityTargetData_ActorArray* ArrayTargetData = new FGameplayAbilityTargetData_ActorArray();
95
96
97 // 3. 소프트 포인터 배열을 순회하며 실제 액터 포인터를 추출해 채워 넣습니다.
98 for (AUnit* UnitPtr : UnitArray)
99 {
100 // 소프트 포인터가 가리키는 대상이 현재 메모리에 로드되어 있고 유효한지 확인
101 if (IsValid(UnitPtr))
102 {
103 // TargetActorArray는 TWeakObjectPtr<AActor> 배열이지만,
104 // 일반 포인터(AActor*)를 넣으면 자동으로 변환되어 들어갑니다.
105 ArrayTargetData->TargetActorArray.Add(UnitPtr);
106 }
107 }
108
109 // 4. 포장된 데이터를 EventData에 추가
110 EventData.TargetData.Add(ArrayTargetData);
111
112 UAbilitySystemBlueprintLibrary::SendGameplayEventToActor(GetOwner(), EventData.EventTag, EventData);
113
114 return true;
115}
