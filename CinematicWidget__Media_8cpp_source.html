40void UCinematicWidget::EnsureCinematicMediaObjects()
41{
42 if (mCinematicMediaPlayer == nullptr)
43 {
44 mCinematicMediaPlayer = NewObject<UMediaPlayer>(this, TEXT("IntroCinematicMediaPlayer"));
45#if PLATFORM_WINDOWS || PLATFORM_MAC
46 mCinematicMediaPlayer->SetDesiredPlayerName(FName(TEXT("ElectraProtron")));
47#elif PLATFORM_ANDROID
48 // Electra가 자동 선택되면 본 프로젝트의 로컬 H.264 mp4를 "No playable streams"로 거부한다(5.7 실측).
49 // OS MediaPlayer 경로(AndroidMedia)가 로컬 파일 재생에 검증돼 있어 명시 지정한다.
50 mCinematicMediaPlayer->SetDesiredPlayerName(FName(TEXT("AndroidMedia")));
51#endif
52 mCinematicMediaPlayer->SetLooping(false); // 인트로는 1회 재생 후 종료(루프 X)
53 // 열림/실패/종료 이벤트를 바인딩해 재생 시작·폴백·다음 단계 전환을 제어한다(AddUnique로 중복 바인딩 방지).
54 mCinematicMediaPlayer->OnMediaOpened.AddUniqueDynamic(this, &UCinematicWidget::HandleCinematicMediaOpened);
55 mCinematicMediaPlayer->OnMediaOpenFailed.AddUniqueDynamic(this, &UCinematicWidget::HandleCinematicMediaOpenFailed);
56 mCinematicMediaPlayer->OnEndReached.AddUniqueDynamic(this, &UCinematicWidget::HandleCinematicMediaEndReached);
57 }
58
59 if (mCinematicMediaTexture == nullptr)
60 {
61 mCinematicMediaTexture = NewObject<UMediaTexture>(this, TEXT("IntroCinematicMediaTexture"));
62 mCinematicMediaTexture->ClearColor = FLinearColor::Black; // 첫 프레임 도착 전·클리어 시 검은 화면으로 채움
63 mCinematicMediaTexture->SetMediaPlayer(mCinematicMediaPlayer); // 텍스처가 이 플레이어의 출력 프레임을 받게 연결
64 mCinematicMediaTexture->UpdateResource(); // 렌더 리소스 즉시 갱신(연결 직후 유효 텍스처 보장)
65 }
66 if (mCinematicMediaTexture != nullptr)
67 {
68 mCinematicMediaTexture->AutoClear = !mHoldLastFrameOnFinish;
69 }
70
71 if (mCinematicMediaSource == nullptr)
72 {
73 mCinematicMediaSource = NewObject<UFileMediaSource>(this, TEXT("IntroCinematicMediaSource"));
74 mCinematicMediaSource->PrecacheFile = true; // 디스크 파일을 미리 캐시해 재생 시작 끊김(stutter)을 줄임
75 }
76
77 mCinematicVideoBrush.SetResourceObject(mCinematicMediaTexture); // 브러시의 그리기 리소스를 미디어 텍스처로 지정
78}

86bool UCinematicWidget::PlayCinematicVideo()
87{
88 EnsureCinematicMediaObjects();
89
90 if (mCinematicMediaPlayer == nullptr || mCinematicMediaSource == nullptr)
91 {
92 return false; // 미디어 오브젝트 생성 실패 시 재생 불가
93 }
94
95 const FString VideoPath = ResolveCinematicVideoPath(); // 상대 경로면 Content 기준 절대 경로로 변환
96 if (FPaths::FileExists(VideoPath) == false)
97 {
98 UE_LOG(LogRD, Warning, TEXT("Intro cinematic file missing: %s"), *VideoPath);
99 return false; // mp4가 디스크에 없으면 호출측이 폴백 타이머로 진행하도록 false 반환
100 }
101
102 // cover 비율 기준이 되는 영상 실제 해상도를 파일에서 미리 확정한다(미디어 텍스처/플레이어 해상도는 첫 프레임 전까지 0이라 폴백 시 정사각이 16:9로 눌림).
103 mCinematicVideoNativeSize = ReadCinematicVideoFileDimensions(VideoPath);
104 if (mCinematicVideoNativeSize.X > 0.0f && mCinematicVideoNativeSize.Y > 0.0f)
105 {
106 mCinematicVideoBrush.ImageSize = mCinematicVideoNativeSize; // 브러시 이미지 크기를 원본 해상도로 맞춰 비율 보존
107 }
108
109 mCinematicMediaPlayer->Rewind(); // 이전 재생 위치를 처음으로 되돌림
110 mCinematicMediaPlayer->Close(); // 기존 열린 소스를 닫아 깨끗한 상태에서 다시 open
111 mCinematicMediaSource->SetFilePath(VideoPath); // 열 대상 파일 경로 지정
112 return mCinematicMediaPlayer->OpenSource(mCinematicMediaSource); // 비동기 open 시작(성공 발행 여부 반환)
113}

122FVector2D UCinematicWidget::ReadCinematicVideoFileDimensions(const FString& VideoPath) const
123{
124 TArray<uint8> Bytes;
125 if (FFileHelper::LoadFileToArray(Bytes, *VideoPath) == false)
126 {
127 return FVector2D::ZeroVector; // 파일 로드 실패 → 해상도 미상(폴백 경로에서 처리)
128 }
129
130 const int32 NumBytes = Bytes.Num();
131 // MP4 박스 헤더의 길이/해상도 필드는 모두 빅엔디안 32비트라 4바이트를 묶어 uint32로 읽는 헬퍼.
132 auto ReadBigEndian32 = [&Bytes](int32 Offset) -> uint32
133 {
134 return (static_cast<uint32>(Bytes[Offset]) << 24)
135 | (static_cast<uint32>(Bytes[Offset + 1]) << 16)
136 | (static_cast<uint32>(Bytes[Offset + 2]) << 8)
137 | static_cast<uint32>(Bytes[Offset + 3]);
138 };
139
140 // 바이트 스트림 전체를 훑어 'tkhd' 4바이트 시그니처를 탐색(인덱스 4부터: 앞 4바이트는 박스 크기 자리).
141 for (int32 Index = 4; Index + 4 <= NumBytes; ++Index)
142 {
143 if (Bytes[Index] != 't' || Bytes[Index + 1] != 'k' || Bytes[Index + 2] != 'h' || Bytes[Index + 3] != 'd')
144 {
145 continue; // 'tkhd' 가 아니면 다음 바이트로
146 }
147
148 // 박스 크기(4바이트)는 타입('tkhd') 바로 앞에 온다. 박스 끝 = (타입앞) + 크기.
149 const int32 BoxStart = Index - 4;
150 const uint32 BoxSize = ReadBigEndian32(BoxStart);
151 const int64 BoxEnd = static_cast<int64>(BoxStart) + static_cast<int64>(BoxSize);
152 if (BoxSize < 32 || BoxEnd > NumBytes)
153 {
154 continue; // 박스 길이가 비정상이거나 버퍼를 넘어가면 잘못된 매치로 보고 스킵
155 }
156
157 // 표시 해상도는 박스 끝 직전 8바이트(width 4 + height 4)에 16.16 고정소수로 저장 → 65536으로 나눠 실수화.
158 const float Width = static_cast<float>(ReadBigEndian32(static_cast<int32>(BoxEnd) - 8)) / 65536.0f;
159 const float Height = static_cast<float>(ReadBigEndian32(static_cast<int32>(BoxEnd) - 4)) / 65536.0f;
160 if (Width > 0.0f && Height > 0.0f)
161 {
162 return FVector2D(Width, Height); // 첫 양수 해상도(=비디오 트랙) 발견 시 즉시 반환
163 }
164 }
165
166 return FVector2D::ZeroVector; // tkhd 미발견(또는 모두 0) → 해상도 미상
167}

201void UCinematicWidget::HandleCinematicMediaOpened(FString OpenedUrl)
202{
203 if (mCinematicMediaPlayer != nullptr)
204 {
205 mCinematicMediaPlayer->Play(); // open 완료 시점에 비로소 재생 시작
206
207 // 영상 실제 해상도를 한 번 확보해 cover 비율 기준으로 쓴다.
208 // (미디어 텍스처 GetWidth/Height는 첫 프레임 디코드 전까지 0/부정확할 수 있어, 폴백 시 정사각(640x640) 영상이 16:9로 눌릴 수 있다.)
209 const int32 VideoTrack = mCinematicMediaPlayer->GetSelectedTrack(EMediaPlayerTrack::Video); // 선택된 비디오 트랙 인덱스
210 const int32 VideoFormat = mCinematicMediaPlayer->GetTrackFormat(EMediaPlayerTrack::Video, VideoTrack); // 해당 트랙의 포맷 인덱스
211 const FIntPoint VideoDimensions = mCinematicMediaPlayer->GetVideoTrackDimensions(VideoTrack, VideoFormat); // 트랙 포맷의 실제 픽셀 해상도
212 if (VideoDimensions.X > 0 && VideoDimensions.Y > 0)
213 {
214 mCinematicVideoNativeSize = FVector2D(VideoDimensions.X, VideoDimensions.Y); // 파일 파싱 값을 트랙 실측값으로 갱신
215 mCinematicVideoBrush.ImageSize = mCinematicVideoNativeSize; // 브러시 이미지 크기를 실해상도에 맞춰 비율 보존
216 }
217
218 const float MediaDurationSeconds = StaticCast<float>(mCinematicMediaPlayer->GetDuration().GetTotalSeconds());
219 if (MediaDurationSeconds > 0.0f)
220 {
221 StartDefaultCinematicTimer(MediaDurationSeconds + 1.0f); // 길이를 알면 (영상길이+1초) 후 강제 종료 보장(OnEndReached 누락 대비)
222 }
223 else
224 {
225 StartDefaultCinematicTimer(mDefaultCinematicDuration); // 길이 미상이면 기본 재생 시간으로 폴백 타이머 설정
226 }
227 }
228}

257void UCinematicWidget::HoldCinematicLastFrame()
258{
259 if (mCinematicMediaPlayer == nullptr)
260 {
261 return;
262 }
263
264 const FTimespan Duration = mCinematicMediaPlayer->GetDuration();
265 const FTimespan FrameSafetyOffset = FTimespan::FromMilliseconds(33);
266 if (Duration > FrameSafetyOffset && mCinematicMediaPlayer->SupportsSeeking())
267 {
268 mCinematicMediaPlayer->Seek(Duration - FrameSafetyOffset);
269 }
270 mCinematicMediaPlayer->Pause();
271 mCinematicMediaPlayer->SetRate(0.0f);
272}
