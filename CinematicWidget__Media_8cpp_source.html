53void UCinematicWidget::EnsureCinematicMediaObjects()
54{
55 if (mCinematicMediaPlayer == nullptr)
56 {
57 mCinematicMediaPlayer = NewObject<UMediaPlayer>(this, TEXT("IntroCinematicMediaPlayer"));
58#if PLATFORM_WINDOWS || PLATFORM_MAC
59 mCinematicMediaPlayer->SetDesiredPlayerName(FName(TEXT("ElectraProtron")));
60#elif PLATFORM_ANDROID
61 // Electra가 자동 선택되면 본 프로젝트의 로컬 H.264 mp4를 "No playable streams"로 거부한다(5.7 실측).
62 // OS MediaPlayer 경로(AndroidMedia)가 로컬 파일 재생에 검증돼 있어 명시 지정한다.
63 mCinematicMediaPlayer->SetDesiredPlayerName(FName(TEXT("AndroidMedia")));
64#endif
65 mCinematicMediaPlayer->SetLooping(false); // 인트로는 1회 재생 후 종료(루프 X)
66 // 열림/실패/종료 이벤트를 바인딩해 재생 시작·폴백·다음 단계 전환을 제어한다(AddUnique로 중복 바인딩 방지).
67 mCinematicMediaPlayer->OnMediaOpened.AddUniqueDynamic(this, &UCinematicWidget::HandleCinematicMediaOpened);
68 mCinematicMediaPlayer->OnMediaOpenFailed.AddUniqueDynamic(this, &UCinematicWidget::HandleCinematicMediaOpenFailed);
69 mCinematicMediaPlayer->OnEndReached.AddUniqueDynamic(this, &UCinematicWidget::HandleCinematicMediaEndReached);
70 }
71
72 if (mCinematicAudioEnabled && !mCinematicSound && GetWorld())
73 {
74 mCinematicSound = NewObject<UMediaSoundComponent>(this);
75 mCinematicSound->SoundClass = GetDefault<UGamePlaySettings>()->mSoundClasses[
76 static_cast<int32>(EGameVolumeType::SFX)].LoadSynchronous();
77 mCinematicSound->bIsUISound = true;
78 mCinematicSound->SetMediaPlayer(mCinematicMediaPlayer);
79 mCinematicSound->RegisterComponentWithWorld(GetWorld());
80 mCinematicSound->Start();
81 }
82
83 if (mCinematicMediaTexture == nullptr)
84 {
85 mCinematicMediaTexture = NewObject<UMediaTexture>(this, TEXT("IntroCinematicMediaTexture"));
86 mCinematicMediaTexture->ClearColor = FLinearColor::Black; // 첫 프레임 도착 전·클리어 시 검은 화면으로 채움
87 mCinematicMediaTexture->SetMediaPlayer(mCinematicMediaPlayer); // 텍스처가 이 플레이어의 출력 프레임을 받게 연결
88 mCinematicMediaTexture->UpdateResource(); // 렌더 리소스 즉시 갱신(연결 직후 유효 텍스처 보장)
89 }
90 if (mCinematicMediaTexture != nullptr)
91 {
92 mCinematicMediaTexture->AutoClear = !mHoldLastFrameOnFinish;
93 }
94
95 if (mCinematicMediaSource == nullptr)
96 {
97 mCinematicMediaSource = NewObject<UFileMediaSource>(this, TEXT("IntroCinematicMediaSource"));
98 mCinematicMediaSource->PrecacheFile = true; // 디스크 파일을 미리 캐시해 재생 시작 끊김(stutter)을 줄임
99 }
100
101 mCinematicVideoBrush.SetResourceObject(mCinematicMediaTexture); // 브러시의 그리기 리소스를 미디어 텍스처로 지정
102}

110bool UCinematicWidget::PlayCinematicVideo()
111{
112 EnsureCinematicMediaObjects();
113
114 if (mCinematicMediaPlayer == nullptr || mCinematicMediaSource == nullptr)
115 {
116 return false; // 미디어 오브젝트 생성 실패 시 재생 불가
117 }
118
119 const FString VideoPath = ResolveCinematicVideoPath(); // 상대 경로면 Content 기준 절대 경로로 변환
120 if (FPaths::FileExists(VideoPath) == false)
121 {
122 UE_LOG(LogRD, Warning, TEXT("Intro cinematic file missing: %s"), *VideoPath);
123 return false; // mp4가 디스크에 없으면 호출측이 폴백 타이머로 진행하도록 false 반환
124 }
125
126 // cover 비율 기준이 되는 영상 실제 해상도를 파일에서 미리 확정한다(미디어 텍스처/플레이어 해상도는 첫 프레임 전까지 0이라 폴백 시 정사각이 16:9로 눌림).
127 mCinematicVideoNativeSize = ReadCinematicVideoFileDimensions(VideoPath);
128 if (mCinematicVideoNativeSize.X > 0.0f && mCinematicVideoNativeSize.Y > 0.0f)
129 {
130 mCinematicVideoBrush.ImageSize = mCinematicVideoNativeSize; // 브러시 이미지 크기를 원본 해상도로 맞춰 비율 보존
131 }
132
133 mCinematicMediaPlayer->Rewind(); // 이전 재생 위치를 처음으로 되돌림
134 mCinematicMediaPlayer->Close(); // 기존 열린 소스를 닫아 깨끗한 상태에서 다시 open
135 mCinematicMediaSource->SetFilePath(VideoPath); // 열 대상 파일 경로 지정
136 StartDefaultCinematicTimer(mDefaultCinematicDuration + 10.f);
137 return mCinematicMediaPlayer->OpenSource(mCinematicMediaSource); // 비동기 open 시작(성공 발행 여부 반환)
138}

196void UCinematicWidget::HandleCinematicMediaOpened(FString OpenedUrl)
197{
198 if (mCinematicMediaPlayer != nullptr)
199 {
200 if (mCinematicAudioEnabled)
201 UE_LOG(LogRD, Display, TEXT("RD_BOSS_ENTRANCE media opened duration=%.2f audioTracks=%d audioOutput=%d"),
202 mCinematicMediaPlayer->GetDuration().GetTotalSeconds(),
203 mCinematicMediaPlayer->GetNumTracks(EMediaPlayerTrack::Audio), mCinematicSound && mCinematicSound->IsActive());
204 mCinematicMediaPlayer->Play(); // open 완료 시점에 비로소 재생 시작
205
206 // 영상 실제 해상도를 한 번 확보해 cover 비율 기준으로 쓴다.
207 // (미디어 텍스처 GetWidth/Height는 첫 프레임 디코드 전까지 0/부정확할 수 있어, 폴백 시 정사각(640x640) 영상이 16:9로 눌릴 수 있다.)
208 const int32 VideoTrack = mCinematicMediaPlayer->GetSelectedTrack(EMediaPlayerTrack::Video); // 선택된 비디오 트랙 인덱스
209 const int32 VideoFormat = mCinematicMediaPlayer->GetTrackFormat(EMediaPlayerTrack::Video, VideoTrack); // 해당 트랙의 포맷 인덱스
210 const FIntPoint VideoDimensions = mCinematicMediaPlayer->GetVideoTrackDimensions(VideoTrack, VideoFormat); // 트랙 포맷의 실제 픽셀 해상도
211 if (VideoDimensions.X > 0 && VideoDimensions.Y > 0)
212 {
213 mCinematicVideoNativeSize = FVector2D(VideoDimensions.X, VideoDimensions.Y); // 파일 파싱 값을 트랙 실측값으로 갱신
214 mCinematicVideoBrush.ImageSize = mCinematicVideoNativeSize; // 브러시 이미지 크기를 실해상도에 맞춰 비율 보존
215 }
216
217 if (mUsingAcceleratedCinematicSource)
218 {
219 if (mAcceleratedCinematicStartTime > FTimespan::Zero() && mCinematicMediaPlayer->SupportsSeeking())
220 {
221 mCinematicMediaPlayer->Seek(mAcceleratedCinematicStartTime);
222 }
223 const float RemainingSeconds = StaticCast<float>(FMath::Max(
224 0.0,
225 (mCinematicMediaPlayer->GetDuration() - mAcceleratedCinematicStartTime).GetTotalSeconds()));
226 StartDefaultCinematicTimer(RemainingSeconds + 1.0f);
227 UE_LOG(LogRD, Display, TEXT("Intro cinematic switched to pre-encoded 3.00x source at %.2fs"), mAcceleratedCinematicStartTime.GetTotalSeconds());
228 return;
229 }
230
231 if (mRequestedPlaybackRate > 1.0f && ApplyRequestedCinematicPlaybackRate())
232 {
233 return;
234 }
235
236 const float MediaDurationSeconds = StaticCast<float>(mCinematicMediaPlayer->GetDuration().GetTotalSeconds());
237 if (MediaDurationSeconds > 0.0f)
238 {
239 StartDefaultCinematicTimer(MediaDurationSeconds + 1.0f); // 길이를 알면 (영상길이+1초) 후 강제 종료 보장(OnEndReached 누락 대비)
240 }
241 else
242 {
243 StartDefaultCinematicTimer(mDefaultCinematicDuration); // 길이 미상이면 기본 재생 시간으로 폴백 타이머 설정
244 }
245 }

248bool UCinematicWidget::ApplyRequestedCinematicPlaybackRate()
249{
250 if (mCinematicMediaPlayer == nullptr || mCinematicMediaPlayer->IsReady() == false)
251 {
252 return false;
253 }
254
255 const float PlaybackRate = FMath::Max(mRequestedPlaybackRate, KINDA_SMALL_NUMBER);
256 if (mCinematicMediaPlayer->SetRate(PlaybackRate) == false)
257 {
258 const FString AcceleratedVideoPath = ResolveAcceleratedCinematicVideoPath();
259 if (FMath::IsNearlyEqual(PlaybackRate, 3.0f) == false
260 || FPaths::FileExists(AcceleratedVideoPath) == false
261 || mCinematicMediaSource == nullptr)
262 {
263 UE_LOG(LogRD, Warning, TEXT("Intro cinematic playback rate %.2fx is not supported by the active media player"), PlaybackRate);
264 return false;
265 }
266
267 mAcceleratedCinematicStartTime = FTimespan::FromSeconds(mCinematicMediaPlayer->GetTime().GetTotalSeconds() / 3.0);
268 mUsingAcceleratedCinematicSource = true;
269 if (mCinematicMediaTexture != nullptr)
270 {
271 mCinematicMediaTexture->AutoClear = false;
272 }
273 mCinematicMediaPlayer->Close();
274 mCinematicMediaSource->SetFilePath(AcceleratedVideoPath);
275 if (mCinematicMediaPlayer->OpenSource(mCinematicMediaSource) == false)
276 {
277 mUsingAcceleratedCinematicSource = false;
278 UE_LOG(LogRD, Warning, TEXT("Intro cinematic failed to open pre-encoded 3.00x source: %s"), *AcceleratedVideoPath);
279 return false;
280 }
281 UE_LOG(LogRD, Display, TEXT("Intro cinematic opening pre-encoded 3.00x source: %s"), *AcceleratedVideoPath);
282 return true;
283 }
284
285 const double RemainingSeconds = FMath::Max(
286 0.0,
287 (mCinematicMediaPlayer->GetDuration() - mCinematicMediaPlayer->GetTime()).GetTotalSeconds());
288 if (RemainingSeconds > 0.0)
289 {
290 StartDefaultCinematicTimer(StaticCast<float>(RemainingSeconds / PlaybackRate) + 1.0f);
291 }
292 UE_LOG(LogRD, Display, TEXT("Intro cinematic playback accelerated to %.2fx; remaining media time %.2fs"), PlaybackRate, RemainingSeconds);
293 return true;
294}

323void UCinematicWidget::HoldCinematicLastFrame()
324{
325 if (mCinematicMediaPlayer == nullptr)
326 {
327 return;
328 }
329
330 const FTimespan Duration = mCinematicMediaPlayer->GetDuration();
331 const FTimespan FrameSafetyOffset = FTimespan::FromMilliseconds(33);
332 if (Duration > FrameSafetyOffset && mCinematicMediaPlayer->SupportsSeeking())
333 {
334 mCinematicMediaPlayer->Seek(Duration - FrameSafetyOffset);
335 }
336 mCinematicMediaPlayer->Pause();
337 mCinematicMediaPlayer->SetRate(0.0f);
