58 bool TickCaptureAndQuit(const float DeltaSeconds)
59 {
60 CaptureElapsed += DeltaSeconds;
61 if (CaptureIndex < UE_ARRAY_COUNT(CaptureTimes)
62 && CaptureElapsed >= CaptureTimes[CaptureIndex])
63 {
64 const FString OutputPath = FPaths::Combine(
65 CaptureDirectory, CaptureNames[CaptureIndex]);
66 FScreenshotRequest::RequestScreenshot(
67 OutputPath, true, false, false);
68 UE_LOG(LogRD, Display,
69 TEXT("RD_REWARD_CONCEPT03_INGAME_CAPTURE requested index=%d path=%s"),
70 CaptureIndex, *OutputPath);
71 ++CaptureIndex;
72 }
73
74 if (CaptureIndex >= UE_ARRAY_COUNT(CaptureTimes)
75 && CaptureElapsed >= CaptureTimes[UE_ARRAY_COUNT(CaptureTimes) - 1] + 1.f)
76 {
77 const FString FinalPath = FPaths::Combine(
78 CaptureDirectory, CaptureNames[UE_ARRAY_COUNT(CaptureNames) - 1]);
79 UE_LOG(LogRD, Display,
80 TEXT("RD_REWARD_CONCEPT03_INGAME_CAPTURE complete final=%s exists=%d"),
81 *FinalPath, IFileManager::Get().FileExists(*FinalPath) ? 1 : 0);
82 FPlatformMisc::RequestExit(false);
83 CaptureTickerHandle.Reset();
84 return false;
85 }
86 return true;
87 }

89 void StartCaptureAndQuit()
90 {
91 if (CaptureTickerHandle.IsValid())
92 {
93 FTSTicker::GetCoreTicker().RemoveTicker(CaptureTickerHandle);

95 CaptureDirectory = FPaths::Combine(
96 FPaths::ProjectSavedDir(), TEXT("UI"), TEXT("RewardConcept03New"),
97 TEXT("InGame"));
98 IFileManager::Get().MakeDirectory(*CaptureDirectory, true);
99 CaptureElapsed = 0.f;
100 CaptureIndex = 0;
101 CaptureTickerHandle = FTSTicker::GetCoreTicker().AddTicker(
102 TEXT("RewardConcept03InGameCapture"), 0.f,
103 [](const float DeltaSeconds)
104 {
105 return TickCaptureAndQuit(DeltaSeconds);
106 });
107 }

109 bool TickVideoCaptureAndQuit(const float DeltaSeconds)
110 {
111 if (!bVideoPresentationStarted)
112 {
113 --VideoWarmupFramesRemaining;
114 if (VideoWarmupFramesRemaining <= 0 && ShownWidget.IsValid())
115 {
116 ShownWidget->ResetRewardFlow();
117 ShownWidget->SetRewardPresentationManualTick(true);
118 ShownWidget->AdvanceRewardFlow();
119 ShownWidget->OpenRewardChest();
120 VideoElapsed = 0.f;
121 bVideoPresentationStarted = true;
122 }
123 return true;
124 }
125
126 if (VideoFrameIndex < VideoFrameCount
127 && !FScreenshotRequest::IsScreenshotRequested())
128 {
129 ShownWidget->AdvanceRewardPresentation(
130 1.f / static_cast<float>(VideoFrameRate));
131 const FString OutputPath = FPaths::Combine(VideoFrameDirectory,
132 FString::Printf(TEXT("Frame_%04d.png"), VideoFrameIndex));
133 FScreenshotRequest::RequestScreenshot(
134 OutputPath, true, false, false);
135 ++VideoFrameIndex;
136 }
137
138 if (VideoFrameIndex >= VideoFrameCount && !FScreenshotRequest::IsScreenshotRequested())
139 {
140 const FString FinalPath = FPaths::Combine(VideoFrameDirectory,
141 FString::Printf(TEXT("Frame_%04d.png"), VideoFrameCount - 1));
142 UE_LOG(LogRD, Display,
143 TEXT("RD_REWARD_CONCEPT03_INGAME_VIDEO complete frames=%d dir=%s final_exists=%d"),
144 VideoFrameIndex, *VideoFrameDirectory,
145 IFileManager::Get().FileExists(*FinalPath) ? 1 : 0);
146 FPlatformMisc::RequestExit(false);
147 VideoTickerHandle.Reset();
148 return false;
149 }
150 return true;
151 }

153 void StartVideoCaptureAndQuit()
154 {
155 if (VideoTickerHandle.IsValid())
156 {
157 FTSTicker::GetCoreTicker().RemoveTicker(VideoTickerHandle);
158 }
159 VideoFrameDirectory = FPaths::Combine(
160 FPaths::ProjectSavedDir(), TEXT("UI"), TEXT("RewardConcept03New"),
161 TEXT("InGameVideoFrames"),
162 FDateTime::Now().ToString(TEXT("%Y%m%d_%H%M%S")));
163 IFileManager::Get().MakeDirectory(*VideoFrameDirectory, true);
164 VideoElapsed = 0.f;
165 VideoFrameIndex = 0;
166 VideoWarmupFramesRemaining = 8;
167 bVideoPresentationStarted = false;
168 // PNG 쓰기가 한 프레임을 지연시켜도 WBP 연출은 정확히 30fps로 진행한다.
169 FApp::SetUseFixedTimeStep(true);
170 FApp::SetFixedDeltaTime(1.0 / static_cast<double>(VideoFrameRate));
171 VideoTickerHandle = FTSTicker::GetCoreTicker().AddTicker(
172 TEXT("RewardConcept03InGameVideoCapture"), 0.f,
173 [](const float DeltaSeconds)
174 {
175 return TickVideoCaptureAndQuit(DeltaSeconds);
176 });
177 UE_LOG(LogRD, Display,
178 TEXT("RD_REWARD_CONCEPT03_INGAME_VIDEO begin fps=%d frames=%d dir=%s"),
179 VideoFrameRate, VideoFrameCount, *VideoFrameDirectory);
180 }

182 void Show(UWorld* World, const TCHAR* WidgetPath, const bool bStartOpening)
183 {
184 RemoveShownWidget();
185 if (World == nullptr || !World->IsGameWorld())
186 {
187 UE_LOG(LogRD, Warning,
188 TEXT("RD.RewardConcept03Preview: 게임 월드에서만 실행할 수 있습니다."));
189 return;
190 }
191
192 UClass* WidgetClass = LoadClass<URewardConcept03Widget>(nullptr, WidgetPath);
193 if (WidgetClass == nullptr)
194 {
195 UE_LOG(LogRD, Warning,
196 TEXT("RD.RewardConcept03Preview: WBP 클래스를 찾지 못했습니다: %s"),
197 WidgetPath);
198 return;
199 }
200
201 APlayerController* Controller = World->GetFirstPlayerController();
202 URewardConcept03Widget* Widget = Controller != nullptr
203 ? CreateWidget<URewardConcept03Widget>(Controller, WidgetClass)
204 : CreateWidget<URewardConcept03Widget>(World, WidgetClass);
205 if (Widget == nullptr)
206 {
207 UE_LOG(LogRD, Warning,
208 TEXT("RD.RewardConcept03Preview: WBP 생성에 실패했습니다."));
209 return;
210 }
211
212 Widget->AddToViewport(10000);
213 Widget->ResetRewardFlow();
214 ShownWidget = Widget;
215
216 if (Controller != nullptr)
217 {
218 FInputModeUIOnly InputMode;
219 InputMode.SetWidgetToFocus(Widget->TakeWidget());
220 Controller->SetInputMode(InputMode);
221 Controller->SetShowMouseCursor(true);
222 }
223
224 if (bStartOpening)
225 {
226 Widget->AdvanceRewardFlow();
227 Widget->OpenRewardChest();
228 }
229
230 UE_LOG(LogRD, Display,
231 TEXT("RD_REWARD_CONCEPT03_PREVIEW shown path=%s auto=%d"),
232 WidgetPath, bStartOpening ? 1 : 0);
233 }
