36void ACombatDicePreviewActor::RebuildFaceTexts(int32 FaceCount)
37{
38 // 기존 TextRender 제거.
39 for (UTextRenderComponent* FaceText : mFaceTexts)
40 {
41 if (FaceText != nullptr)
42 {
43 FaceText->DestroyComponent();
44 }
45 }
46 mFaceTexts.Reset();
47 for (UTextRenderComponent* FaceTextStroke : mFaceTextStrokes)
48 {
49 if (FaceTextStroke != nullptr)
50 {
51 FaceTextStroke->DestroyComponent();
52 }
53 }
54 mFaceTextStrokes.Reset();
55 const RDDicePolyhedron::FDicePolyhedron& Poly = RDDicePolyhedron::Get(FaceCount);
56
57 // 주사위 종류당 숫자 크기를 하나로 통일(면마다 들쭉날쭉하지 않게). 가장 작은 면에도 들어가게 최소 내접반지름 기준.
58 float MinInradius = TNumericLimits<float>::Max();
59 for (int32 ValueFaceIndex = 0; ValueFaceIndex < Poly.GetValueFaceCount(); ++ValueFaceIndex)
60 {
61 const RDDicePolyhedron::FDiceFace& Face = Poly.GetValueFace(ValueFaceIndex);
62 const int32 EdgeCount = Face.mVertexIndices.Num();
63 for (int32 Edge = 0; Edge < EdgeCount; ++Edge)
64 {
65 const FVector& V0 = Poly.mVertices[Face.mVertexIndices[Edge]];
66 const FVector& V1 = Poly.mVertices[Face.mVertexIndices[(Edge + 1) % EdgeCount]];
67 MinInradius = FMath::Min(MinInradius, static_cast<float>(((V0 + V1) * 0.5f - Face.mCenter).Size()));
68 }
69 }
70 // 주사위 간에도 너무 차이나지 않게 클램프.
71 mBaseFaceTextWorldSize = FMath::Clamp(MinInradius * DiceRadius * 1.48f, 24.0f, 48.0f);
72 if (FaceCount == 2)
73 {
74 mBaseFaceTextWorldSize = 44.0f;
75 }
76 const float UniformTextSize = mBaseFaceTextWorldSize * mFaceTextScale;
77
78 for (int32 FaceIndex = 0; FaceIndex < Poly.GetValueFaceCount(); ++FaceIndex)
79 {
80 const RDDicePolyhedron::FDiceFace& Face = Poly.GetValueFace(FaceIndex);
81 const float TextWorldSize = UniformTextSize;
82 const float TextLift = FaceCount == 2 ? CoinFaceTextLift : FaceTextLift;
83 const float StrokeLift = FaceCount == 2 ? CoinFaceTextStrokeLift : FaceTextStrokeLift;
84 const float CurrentStrokeOffset = FaceCount == 2 ? CoinFaceTextStrokeOffset : FaceTextStrokeOffset;
85
86 const FVector BaseTextLocation = Face.mCenter * DiceRadius + Face.mNormal * (DiceRadius * TextLift);
87 const FVector StrokeBaseLocation = Face.mCenter * DiceRadius + Face.mNormal * (DiceRadius * StrokeLift);
88 const FVector Right = MakeFaceRightVector(Face);
89 const float StrokeOffset = DiceRadius * CurrentStrokeOffset;
90 const FRotator TextRotation = RDCombatDicePreview::MakeFaceTextRotation(Face.mNormal, Face.mUp);
91 const FText DefaultText = FText::AsNumber(FaceIndex + 1);
92 const FVector StrokeDirections[FaceTextStrokeCount] =
93 {
94 Right,
95 -Right,
96 Face.mUp,
97 -Face.mUp
98 };
99
100 for (int32 StrokeIndex = 0; StrokeIndex < FaceTextStrokeCount; ++StrokeIndex)
101 {
102 UTextRenderComponent* FaceTextStroke = NewObject<UTextRenderComponent>(this);
103 if (FaceTextStroke == nullptr)
104 {
105 mFaceTextStrokes.Add(nullptr);
106 continue;
107 }
108
109 FaceTextStroke->SetupAttachment(mDiceRoot);
110 FaceTextStroke->RegisterComponent();
111 FaceTextStroke->SetHorizontalAlignment(EHorizTextAligment::EHTA_Center);
112 FaceTextStroke->SetVerticalAlignment(EVerticalTextAligment::EVRTA_TextCenter);
113 FaceTextStroke->SetTextRenderColor(FaceTextDefaultStrokeColor);
114 FaceTextStroke->SetWorldSize(TextWorldSize);
115 FaceTextStroke->SetCastShadow(false);
116 FaceTextStroke->SetTranslucentSortPriority(FaceCount == 2 ? 64 : 8);
117 if (mDiceNumberFont != nullptr)
118 {
119 FaceTextStroke->SetFont(mDiceNumberFont);
120 }
121 FaceTextStroke->SetRelativeLocation(StrokeBaseLocation + StrokeDirections[StrokeIndex] * StrokeOffset);
122 FaceTextStroke->SetRelativeRotation(TextRotation);
123 FaceTextStroke->SetText(DefaultText);
124 mFaceTextStrokes.Add(FaceTextStroke);
125 }
126
127 UTextRenderComponent* FaceText = NewObject<UTextRenderComponent>(this);
128 if (FaceText == nullptr)
129 {
130 mFaceTexts.Add(nullptr);
131 continue;
132 }
133 FaceText->SetupAttachment(mDiceRoot);
134 FaceText->RegisterComponent();
135 FaceText->SetHorizontalAlignment(EHorizTextAligment::EHTA_Center);
136 FaceText->SetVerticalAlignment(EVerticalTextAligment::EVRTA_TextCenter);
137 FaceText->SetTextRenderColor(FaceTextDefaultColor); // 밝은 주사위에 대비되는 진한 숫자
138 FaceText->SetWorldSize(TextWorldSize);
139 FaceText->SetCastShadow(false);
140 FaceText->SetTranslucentSortPriority(FaceCount == 2 ? 65 : 9);
141 if (mDiceNumberFont != nullptr)
142 {
143 FaceText->SetFont(mDiceNumberFont);
144 }
145
146 // 면 중심에서 바깥으로 띄워, 숫자가 주사위 몸체에 가리지 않게.
147 FaceText->SetRelativeLocation(BaseTextLocation);
148 FaceText->SetRelativeRotation(TextRotation);
149 FaceText->SetText(DefaultText);
150
151 mFaceTexts.Add(FaceText);
152 }
153
154 ApplyFaceTextStyles();
155}

219void ACombatDicePreviewActor::ApplyFaceTextStyles()
220{
221 const bool bHasHighlight = mHighlightedFaceOrdinal != INDEX_NONE;
222 for (int32 FaceIndex = 0; FaceIndex < mFaceTexts.Num(); ++FaceIndex)
223 {
224 const bool bHighlighted = bHasHighlight && mHighlightedFaceOrdinal == FaceIndex + 1;
225 const bool bHideDimText = mHideNonHighlightedFaceTexts && (bHasHighlight == false || bHighlighted == false);
226 const FColor MainColor = bHasHighlight
227 ? (bHighlighted ? FaceTextHighlightColor : FaceTextDimColor)
228 : FaceTextDefaultColor;
229 const FColor StrokeColor = bHasHighlight
230 ? (bHighlighted ? FaceTextHighlightStrokeColor : FaceTextDimStrokeColor)
231 : FaceTextDefaultStrokeColor;
232
233 if (mFaceTexts[FaceIndex] != nullptr)
234 {
235 mFaceTexts[FaceIndex]->SetVisibility(bHideDimText == false, false);
236 mFaceTexts[FaceIndex]->SetTextRenderColor(MainColor);
237 }
238
239 const bool bShowStroke = bHideDimText == false && (bHasHighlight == false || bHighlighted == true);
240 const int32 StrokeStartIndex = FaceIndex * FaceTextStrokeCount;
241 for (int32 StrokeOffsetIndex = 0; StrokeOffsetIndex < FaceTextStrokeCount; ++StrokeOffsetIndex)
242 {
243 const int32 StrokeIndex = StrokeStartIndex + StrokeOffsetIndex;
244 if (mFaceTextStrokes.IsValidIndex(StrokeIndex) && mFaceTextStrokes[StrokeIndex] != nullptr)
245 {
246 mFaceTextStrokes[StrokeIndex]->SetVisibility(bShowStroke, false);
247 mFaceTextStrokes[StrokeIndex]->SetTextRenderColor(StrokeColor);
248 }
249 }
250 }
