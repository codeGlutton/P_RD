76 void ClampToVisibleDiceBoard(float DiceRadius, FVector& LocalLocation, FVector& LocalVelocity, bool& bClamped)
77 {
78 const float SafeX = FMath::Max(DiceRadius, TableHalfDepth - DiceRadius - DiceTableVisibleMarginX);
79 const float SafeY = FMath::Max(DiceRadius, TableHalfWidth - DiceRadius - DiceTableVisibleMarginY);
80
81 // 벽 반사계수↑: 판 안에서 리코셰를 크게 만들어 "따다다다당" 손맛을 낸다.
82 if (LocalLocation.X < -SafeX)
83 {
84 LocalLocation.X = -SafeX;
85 LocalVelocity.X = FMath::Abs(LocalVelocity.X) * 0.66f;
86 bClamped = true;
87 }
88 else if (LocalLocation.X > SafeX)
89 {
90 LocalLocation.X = SafeX;
91 LocalVelocity.X = -FMath::Abs(LocalVelocity.X) * 0.66f;
92 bClamped = true;
93 }
94
95 if (LocalLocation.Y < -SafeY)
96 {
97 LocalLocation.Y = -SafeY;
98 LocalVelocity.Y = FMath::Abs(LocalVelocity.Y) * 0.66f;
99 bClamped = true;
100 }
101 else if (LocalLocation.Y > SafeY)
102 {
103 LocalLocation.Y = SafeY;
104 LocalVelocity.Y = -FMath::Abs(LocalVelocity.Y) * 0.66f;
105 bClamped = true;
106 }
107
108 const float CornerStartX = SafeX * 0.56f;
109 const float CornerStartY = SafeY * 0.56f;
110 const float AbsX = FMath::Abs(LocalLocation.X);
111 const float AbsY = FMath::Abs(LocalLocation.Y);
112 if (AbsX > CornerStartX && AbsY > CornerStartY)
113 {
114 const float ExcessX = (AbsX - CornerStartX) / FMath::Max(KINDA_SMALL_NUMBER, SafeX - CornerStartX);
115 const float ExcessY = (AbsY - CornerStartY) / FMath::Max(KINDA_SMALL_NUMBER, SafeY - CornerStartY);
116 const float ExcessSum = ExcessX + ExcessY;
117 if (ExcessSum > 1.0f)
118 {
119 const float Scale = 1.0f / ExcessSum;
120 const float NewAbsX = CornerStartX + (AbsX - CornerStartX) * Scale;
121 const float NewAbsY = CornerStartY + (AbsY - CornerStartY) * Scale;
122 LocalLocation.X = FMath::Sign(LocalLocation.X) * NewAbsX;
123 LocalLocation.Y = FMath::Sign(LocalLocation.Y) * NewAbsY;
124 LocalVelocity.X *= 0.58f;
125 LocalVelocity.Y *= 0.58f;
126 bClamped = true;
127 }
128 }
129 }

189 void BuildImpactEffectMesh(UProceduralMeshComponent* ImpactComponent)
190 {
191 if (ImpactComponent == nullptr)
192 {
193 return;
194 }
195
196 TArray<FVector> Vertices;
197 TArray<int32> Triangles;
198 TArray<FVector> Normals;
199 TArray<FVector2D> UVs;
200 TArray<FLinearColor> Colors;
201 TArray<FProcMeshTangent> Tangents;
202
203 constexpr int32 SegmentCount = 12;
204 constexpr float InnerRadius = 0.58f;
205 constexpr float OuterRadius = 1.00f;
206 constexpr float SegmentFill = 0.62f;
207 for (int32 SegmentIndex = 0; SegmentIndex < SegmentCount; SegmentIndex += 2)
208 {
209 const float AngleA = (UE_TWO_PI / SegmentCount) * StaticCast<float>(SegmentIndex);
210 const float AngleB = AngleA + (UE_TWO_PI / SegmentCount) * SegmentFill;
211 const FVector InnerA(FMath::Cos(AngleA) * InnerRadius, FMath::Sin(AngleA) * InnerRadius, 0.0f);
212 const FVector OuterA(FMath::Cos(AngleA) * OuterRadius, FMath::Sin(AngleA) * OuterRadius, 0.0f);
213 const FVector OuterB(FMath::Cos(AngleB) * OuterRadius, FMath::Sin(AngleB) * OuterRadius, 0.0f);
214 const FVector InnerB(FMath::Cos(AngleB) * InnerRadius, FMath::Sin(AngleB) * InnerRadius, 0.0f);
215 const int32 BaseIndex = Vertices.Num();
216 AddImpactVertex(InnerA, Vertices, Normals, UVs, Colors, Tangents);
217 AddImpactVertex(OuterA, Vertices, Normals, UVs, Colors, Tangents);
218 AddImpactVertex(OuterB, Vertices, Normals, UVs, Colors, Tangents);
219 AddImpactVertex(InnerB, Vertices, Normals, UVs, Colors, Tangents);
220 Triangles.Add(BaseIndex + 0);
221 Triangles.Add(BaseIndex + 1);
222 Triangles.Add(BaseIndex + 2);
223 Triangles.Add(BaseIndex + 0);
224 Triangles.Add(BaseIndex + 2);
225 Triangles.Add(BaseIndex + 3);
226 }
227
228 constexpr int32 SparkCount = 4;
229 for (int32 SparkIndex = 0; SparkIndex < SparkCount; ++SparkIndex)
230 {
231 const float Angle = (UE_TWO_PI / SparkCount) * StaticCast<float>(SparkIndex) + UE_PI * 0.25f;
232 const FVector Direction(FMath::Cos(Angle), FMath::Sin(Angle), 0.0f);
233 const FVector Perpendicular(-Direction.Y, Direction.X, 0.0f);
234 const FVector Tip = Direction * 1.24f;
235 const FVector BaseCenter = Direction * 0.22f;
236 const FVector Left = BaseCenter + Perpendicular * 0.055f;
237 const FVector Right = BaseCenter - Perpendicular * 0.055f;
238 const int32 BaseIndex = Vertices.Num();
239 AddImpactVertex(Left, Vertices, Normals, UVs, Colors, Tangents);
240 AddImpactVertex(Right, Vertices, Normals, UVs, Colors, Tangents);
241 AddImpactVertex(Tip, Vertices, Normals, UVs, Colors, Tangents);
242 Triangles.Add(BaseIndex + 0);
243 Triangles.Add(BaseIndex + 1);
244 Triangles.Add(BaseIndex + 2);
245 }
246
247 ImpactComponent->CreateMeshSection_LinearColor(0, Vertices, Triangles, Normals, UVs, Colors, Tangents, false);
248 }

380 // 벽 높이(Z 스케일)를 TableWallHeight에 맞춰 바닥 아래부터 충분히 높게 덮는다(큐브 기본 100 단위).
381 const float WallScaleZ = (TableWallHeight + 20.0f) / 100.0f;
382 const float WallThicknessScale = TableWallThickness / 100.0f;
383 const float SafeX = FMath::Max(PhysicsDiceRadius, TableHalfDepth - PhysicsDiceRadius - DiceTableVisibleMarginX);
384 const float SafeY = FMath::Max(PhysicsDiceRadius, TableHalfWidth - PhysicsDiceRadius - DiceTableVisibleMarginY);

386 const float CornerStartY = SafeY * 0.56f;
387 const float TopBottomWallScale = (CornerStartX * 2.0f + TableWallThickness) / 100.0f;
388 const float SideWallScale = (CornerStartY * 2.0f + TableWallThickness) / 100.0f;
389 const float DiagonalDx = SafeX - CornerStartX;
390 const float DiagonalDy = SafeY - CornerStartY;
391 const float DiagonalWallScale = (FMath::Sqrt(DiagonalDx * DiagonalDx + DiagonalDy * DiagonalDy) + TableWallThickness) / 100.0f;
392 const float WallZ = TableWallHeight * 0.5f;
393
394 TArray<FWallSpec> WallSpecs;
395 WallSpecs.Reserve(8);
396 WallSpecs.Add({ FVector(0.0f, -SafeY - TableWallThickness * 0.5f, WallZ), FRotator::ZeroRotator, FVector(TopBottomWallScale, WallThicknessScale, WallScaleZ) });
397 WallSpecs.Add({ FVector(0.0f, SafeY + TableWallThickness * 0.5f, WallZ), FRotator::ZeroRotator, FVector(TopBottomWallScale, WallThicknessScale, WallScaleZ) });
398 WallSpecs.Add({ FVector(-SafeX - TableWallThickness * 0.5f, 0.0f, WallZ), FRotator::ZeroRotator, FVector(WallThicknessScale, SideWallScale, WallScaleZ) });
399 WallSpecs.Add({ FVector( SafeX + TableWallThickness * 0.5f, 0.0f, WallZ), FRotator::ZeroRotator, FVector(WallThicknessScale, SideWallScale, WallScaleZ) });
400
401 for (const float SignX : { -1.0f, 1.0f })
402 {
403 for (const float SignY : { -1.0f, 1.0f })
404 {
405 const FVector2D P0(SignX * CornerStartX, SignY * SafeY);
406 const FVector2D P1(SignX * SafeX, SignY * CornerStartY);
407 const FVector2D Direction = P1 - P0;
408 const FVector2D OutwardNormal = FVector2D(SignX * DiagonalDy, SignY * DiagonalDx).GetSafeNormal();
409 const FVector2D Center2D = (P0 + P1) * 0.5f + OutwardNormal * (TableWallThickness * 0.5f);
410 const float YawDegrees = FMath::RadiansToDegrees(FMath::Atan2(Direction.Y, Direction.X));
411
412 WallSpecs.Add({
413 FVector(Center2D.X, Center2D.Y, WallZ),
414 FRotator(0.0f, YawDegrees, 0.0f),
415 FVector(DiagonalWallScale, WallThicknessScale, WallScaleZ)
416 });
417 }
418 }

866 if (Now - mLastImpactSoundTime >= ImpactSoundCooldownSeconds)
867 {
868 UGameplayStatics::PlaySound2D(this, mImpactSound, FMath::Lerp(0.68f, 1.24f, ClampedStrength));

870 }
871 }

908void ACombatDiceRollCaptureActor::HandlePhysicsBodyHit(
909 UPrimitiveComponent* HitComponent,
910 AActor* OtherActor,
911 UPrimitiveComponent* OtherComp,
912 FVector NormalImpulse,
913 const FHitResult& Hit)
914{
915 if (mRollActive == false || OtherActor != this || HitComponent == nullptr || OtherComp == nullptr)
916 {
917 return;
918 }
919
920 UProceduralMeshComponent* HitBody = Cast<UProceduralMeshComponent>(HitComponent);
921 UProceduralMeshComponent* OtherBody = Cast<UProceduralMeshComponent>(OtherComp);
922 if (HitBody == nullptr || HitBody == OtherBody)
923 {
924 return;
925 }
926
927 const int32 HitDiceIndex = mPhysicsDiceBodies.IndexOfByPredicate(
928 [HitBody](const TObjectPtr<UProceduralMeshComponent>& CandidateBody)
929 {
930 return CandidateBody.Get() == HitBody;
931 });
932 if (HitDiceIndex == INDEX_NONE)
933 {
934 return;
935 }
936
937 const int32 OtherDiceIndex = OtherBody != nullptr
938 ? mPhysicsDiceBodies.IndexOfByPredicate(
939 [OtherBody](const TObjectPtr<UProceduralMeshComponent>& CandidateBody)
940 {
941 return CandidateBody.Get() == OtherBody;
942 })
943 : INDEX_NONE;
944
945 FVector SeparationDirection = OtherBody != nullptr
946 ? HitBody->GetComponentLocation() - OtherBody->GetComponentLocation()
947 : Hit.Normal.GetSafeNormal();
948 if (SeparationDirection.Normalize() == false)
949 {
950 SeparationDirection = Hit.Normal.GetSafeNormal();
951 }
952 if (SeparationDirection.IsNearlyZero())
953 {
954 return;
955 }
956
957 const FVector HitVelocity = HitBody->GetPhysicsLinearVelocity();
958 const FVector RelativeVelocity = OtherBody != nullptr
959 ? HitVelocity - OtherBody->GetPhysicsLinearVelocity()
960 : HitVelocity;
961 const float ClosingSpeed = FMath::Abs(FVector::DotProduct(RelativeVelocity, SeparationDirection));
962 const float ImpactSpeed = FMath::Max(ClosingSpeed, RelativeVelocity.Size());
963 if (ImpactSpeed >= ImpactEffectMinSpeed || NormalImpulse.Size() >= ImpactEffectMinNormalImpulse)
964 {
965 FVector EffectLocation = HitBody->GetComponentLocation();
966 if (Hit.ImpactPoint.IsNearlyZero() == false)
967 {
968 EffectLocation = FVector(Hit.ImpactPoint);
969 }
970 const float EffectStrength = FMath::Clamp(ImpactSpeed / 360.0f, 0.25f, 1.0f);
971 TriggerImpactEffect(EffectLocation, EffectStrength);
972 }
973
974 if (OtherDiceIndex == INDEX_NONE)
975 {
976 return;
977 }
978 if (mCollisionBoostsUsed >= DiceCollisionBoostMaxCount)
979 {
980 return;
981 }
982 if (mRollElapsed - mLastCollisionBoostTime < DiceCollisionBoostCooldownSeconds)
983 {
984 return;
985 }
986 if (ClosingSpeed < DiceCollisionBoostMinClosingSpeed && NormalImpulse.Size() <= KINDA_SMALL_NUMBER)
987 {
988 return;
989 }
990
991 const FVector WorldUp = GetActorTransform().TransformVectorNoScale(FVector::UpVector).GetSafeNormal();
992 FVector TangentDirection = FVector::CrossProduct(WorldUp, SeparationDirection).GetSafeNormal();
993 if (TangentDirection.IsNearlyZero())
994 {
995 TangentDirection = FVector::RightVector;
996 }
997
998 const float BoostScale = (mCollisionBoostsUsed == 0) ? 1.0f : 0.72f;
999 const FVector HitImpulse = SeparationDirection * (DiceCollisionBoostLinearImpulse * BoostScale)
1000 + WorldUp * (DiceCollisionBoostLiftImpulse * BoostScale);
1001 const FVector OtherImpulse = -SeparationDirection * (DiceCollisionBoostLinearImpulse * BoostScale)
1002 + WorldUp * (DiceCollisionBoostLiftImpulse * BoostScale);
1003 const float SpinSign = ((HitDiceIndex + OtherDiceIndex + mCollisionBoostsUsed) % 2 == 0) ? 1.0f : -1.0f;
1004 const FVector HitAngularImpulse = (TangentDirection + SeparationDirection * 0.45f).GetSafeNormal()
1005 * (DiceCollisionBoostAngularImpulse * BoostScale * SpinSign);
1006 const FVector OtherAngularImpulse = (-TangentDirection - SeparationDirection * 0.45f).GetSafeNormal()
1007 * (DiceCollisionBoostAngularImpulse * BoostScale * SpinSign);
1008
1009 HitBody->AddImpulse(HitImpulse, NAME_None, true);
1010 OtherBody->AddImpulse(OtherImpulse, NAME_None, true);
1011 HitBody->AddAngularImpulseInDegrees(HitAngularImpulse, NAME_None, true);
1012 OtherBody->AddAngularImpulseInDegrees(OtherAngularImpulse, NAME_None, true);
1013 HitBody->WakeAllRigidBodies();
1014 OtherBody->WakeAllRigidBodies();
1015
1016 mLastCollisionBoostTime = mRollElapsed;
1017 ++mCollisionBoostsUsed;
1018 mRollStillElapsed = 0.0f;
1019}

1233 const float CurrentRollSpeed = FVector::DotProduct(AngularVelocity, RollAxis);

1239 if (mRollElapsed >= CoinSettleAssistStartSeconds
1240 && PlanarSpeed <= CoinSettleLinearSpeedMax
1241 && AngularVelocity.Size() <= CoinSettleAngularSpeedMax

1245 const FVector TipAxis = FVector::CrossProduct(CoinAxis, TargetAxis).GetSafeNormal();
1246 if (TipAxis.IsNearlyZero() == false)

1248 const float EdgeAlpha = FMath::Clamp((CoinSettleFlatDotMin - Flatness) / CoinSettleFlatDotMin, 0.0f, 1.0f);
1249 const float YawSpin = FVector::DotProduct(AngularVelocity, WorldUp);

1252 AngularVelocity -= WorldUp * YawSpin * YawDampingAlpha;

1254 }
