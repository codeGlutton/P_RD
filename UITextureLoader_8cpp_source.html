10 bool HashMediaFile(const FString& Path, FSHAHash& OutHash)
11 {
12 TUniquePtr<FArchive> Reader(IFileManager::Get().CreateFileReader(*Path, FILEREAD_Silent));
13 if (!Reader || Reader->TotalSize() <= 0) return false;
14 FSHA1 Hash;
15 uint8 Buffer[64 * 1024];
16 while (!Reader->AtEnd() && !Reader->IsError())
17 {
18 const int64 Count = FMath::Min<int64>(sizeof(Buffer), Reader->TotalSize() - Reader->Tell());
19 Reader->Serialize(Buffer, Count);
20 if (Reader->IsError()) return false;
21 Hash.Update(Buffer, Count);
22 }
23 if (!Reader->Close()) return false;
24 OutHash = Hash.Finalize();
25 return true;
26 }

34 FVector2D ReadTrackDimensions(FArchive& Reader, int64 Begin, int64 End, int32 Depth, int32& RemainingBoxes)
35 {
36 if (Depth > 8) return FVector2D::ZeroVector;
37 for (int64 Offset = Begin; End - Offset >= 8 && RemainingBoxes-- > 0;)
38 {
39 uint8 Header[16] = {};
40 Reader.Seek(Offset);
41 Reader.Serialize(Header, 8);
42 if (Reader.IsError()) return FVector2D::ZeroVector;
43 uint64 Size = ReadBE32(Header);
44 int64 HeaderSize = 8;
45 if (Size == 1)
46 {
47 if (End - Offset < 16) return FVector2D::ZeroVector;
48 Reader.Serialize(Header + 8, 8);
49 if (Reader.IsError()) return FVector2D::ZeroVector;
50 Size = (uint64(ReadBE32(Header + 8)) << 32) | ReadBE32(Header + 12);
51 HeaderSize = 16;
52 }
53 else if (Size == 0) Size = End - Offset;
54 if (Size < uint64(HeaderSize) || Size > uint64(End - Offset)) return FVector2D::ZeroVector;
55 const int64 BoxEnd = Offset + int64(Size);
56 const uint32 Type = ReadBE32(Header + 4);
57 if (Type == 0x746b6864 /* tkhd */ && Size >= uint64(HeaderSize + 84))
58 {
59 uint8 Dimensions[8];
60 Reader.Seek(BoxEnd - 8);
61 Reader.Serialize(Dimensions, sizeof(Dimensions));
62 if (Reader.IsError()) return FVector2D::ZeroVector;
63 const FVector2D Result(double(ReadBE32(Dimensions)) / 65536.0,
64 double(ReadBE32(Dimensions + 4)) / 65536.0);
65 if (Result.X > 0 && Result.Y > 0) return Result;
66 }
67 else if (Type == 0x6d6f6f76 /* moov */ || Type == 0x7472616b /* trak */)
68 {
69 const FVector2D Result = ReadTrackDimensions(Reader, Offset + HeaderSize, BoxEnd, Depth + 1, RemainingBoxes);
70 if (Result.X > 0 && Result.Y > 0) return Result;
71 }
72 Offset = BoxEnd;
73 }
74 return FVector2D::ZeroVector;
75 }

78bool RDUITexture::EnsureCachedMediaFile(const FString& SourcePath, const FString& CachePath)
79{
80 TRACE_CPUPROFILER_EVENT_SCOPE(RDMediaCacheResolve);
81 FSHAHash SourceHash;
82 if (!HashMediaFile(SourcePath, SourceHash)) return false;
83 IFileManager& Files = IFileManager::Get();
84 FSHAHash CacheHash;
85 if (Files.FileSize(*SourcePath) == Files.FileSize(*CachePath)
86 && HashMediaFile(CachePath, CacheHash) && SourceHash == CacheHash) return true;
87
88 if (!Files.MakeDirectory(*FPaths::GetPath(CachePath), true)) return false;
89 const FString TemporaryPath = CachePath + TEXT(".tmp-") + FGuid::NewGuid().ToString(EGuidFormats::Digits);
90 ON_SCOPE_EXIT { Files.Delete(*TemporaryPath, false, true, true); };
91 if (Files.Copy(*TemporaryPath, *SourcePath, true, false) != COPY_OK) return false;
92 FSHAHash CopiedHash;
93 if (!HashMediaFile(TemporaryPath, CopiedHash) || CopiedHash != SourceHash) return false;
94 return Files.Move(*CachePath, *TemporaryPath, true, false, false, true);
95}
