121void USimulationEventLogger::BeginMotionLog()
122{
123 Super::BeginMotionLog();
124
125 if (mCurrentTurnEventLog == nullptr || mCurrentActionEventLog == nullptr)
126 {
127 UE_LOG(LogEventLogger, Log, TEXT("턴 외의 모션 시작"));
128 return;
129 }
130
131 checkf(mCurrentTurnEventLog != nullptr, TEXT("턴 로그 시작 없이 모션 로그 시작 오류"));
132 checkf(mCurrentActionEventLog != nullptr, TEXT("액션 로그 시작 없이 모션 로그 시작 오류"));
133
134 FSRPGMotionEventLog MotionLog;
135
136 checkf(MotionLog.IsValid() == true, TEXT("모션 로그 불량"));
137
138 if (mCurrentMotionEventCount == 0)
139 {
140 mCurrentActionEventLog->mMotionEventLogs.Add(MoveTemp(MotionLog));
141 mCurrentMotionEventLog = &mCurrentActionEventLog->mMotionEventLogs.Last();
142
143 UE_LOG(LogEventLogger, Log, TEXT("모션 이벤트 로그 시작"));
144 }
145 ++mCurrentMotionEventCount;
146}

169void USimulationEventLogger::LogTagEffect(int32 TargetActorID, UClass* BoardActorModelClass, const FSRPGTagEffectEventLog& Log)
170{
171 if (mCurrentMotionEventLog == nullptr)
172 {
173 UE_LOG(LogEventLogger, Log, TEXT("모션 외 범위에서의 잘못된 로그 요청으로 무시"));
174 return;
175 }
176 checkf(Log.IsValid() == true, TEXT("태그 변경 로그 불량"));
177
178 Super::LogTagEffect(TargetActorID, BoardActorModelClass, Log);
179
180 FSRPGBoardActorEventLog& BoardActorLog = mCurrentMotionEventLog->mBoardActorEventLogs.FindOrAdd(TargetActorID);
181 BoardActorLog.mTargetActorID = TargetActorID;
182 BoardActorLog.mBoardActorModelClass = BoardActorModelClass;
183
184 checkf(BoardActorLog.IsValid() == true, TEXT("보드 액터 로그 불량"));
185
186 bool IsAlreadyExisted = false;
187 FSRPGTagEffectEventLog& TagEffectEventLog = BoardActorLog.mTagEffectEventLogs.FindOrAdd(Log, &IsAlreadyExisted);
188 if (IsAlreadyExisted == true)
189 {
190 TagEffectEventLog.mEffectTag = Log.mEffectTag;
191 TagEffectEventLog.mCount += Log.mCount;
192 }
193
194 UE_LOG(LogEventLogger, Log, TEXT("[%d][%s : %d] 태그 변경"), TargetActorID, *Log.mEffectTag.ToString(), Log.mCount);
195}
