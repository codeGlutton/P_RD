79bool UBoardActorSequencePlayer::PlaySequenceUsingTag_Internal(FBoardActorAnimationContext Context)
80{
81 if (mTagSequenceSets.Contains(Context.mMontageTag) == false)
82 {
83 // 존재하지 않는 레벨 시퀀스 태그
84 return false;
85 }
86
87 TObjectPtr<ULevelSequence>& TargetLevelSequence = mTagSequenceSets[Context.mMontageTag].mLevelSequences[StaticCast<int32>(Context.mMontageDir)];
88 if (TargetLevelSequence == nullptr)
89 {
90 // nullptr 레벨 시퀀스
91 return false;
92 }
93
94 if (IsPlayingSequenceUsingTag() == true && GetPlayingSequenceSetUsingTag()->mPlayPriority > mTagSequenceSets[Context.mMontageTag].mPlayPriority)
95 {
96 // 우선순위가 낮은 시퀀스
97 return false;
98 }
99
100 StopSequenceUsingTag();
101
102 UDefaultLevelSequenceInstanceData* InstanceData = Cast<UDefaultLevelSequenceInstanceData>(mDefaultInstanceData);
103 if (InstanceData != nullptr)
104 {
105 InstanceData->TransformOriginActor = GetOwningActor();
106 }
107 mLevelSequencePlayer->Initialize(TargetLevelSequence, GetOwningActor()->GetLevel(), mCameraSettings);
108 ApplyDynamicBindings(TargetLevelSequence);
109
110 mActiveAnimationContext = MoveTemp(Context);
111
112 mLevelSequencePlayer->OnNativeFinished.BindUObject(this, &UBoardActorSequencePlayer::OnEndSequenceUsingTag, false);
113 mLevelSequencePlayer->Play();
114
115 return true;
116}

126bool UBoardActorSequencePlayer::TriggerMontageTagEvent(const FGameplayTag& EventTag, const FEventTriggerPayloadBase* Payload)
127{
128 if (IsPlayingSequenceUsingTag() == false)
129 {
130 // 태그 기반 시퀀스 실행 중이 아님
131 return false;
132 }
133
134 const bool HasMontageEvent = mActiveAnimationContext.mMontageEvents.Contains(EventTag);
135 const bool HasAllMontageEvent = mAllMontageEvents.Contains(EventTag);
136 if (HasMontageEvent == false && HasAllMontageEvent == false)
137 {
138 // 존재하지 않는 이벤트
139 return false;
140 }
141
142 if (HasMontageEvent == true)
143 {
144 /* 현 시퀀스 대상 이벤트 실행 */
145 FBoardActorAnimationEvent& TargetEvent = mActiveAnimationContext.mMontageEvents[EventTag];
146 if (TargetEvent.OnTriggerAnimationEvent.IsBound() == true)
147 {
148 TargetEvent.OnTriggerAnimationEvent.Broadcast(mActiveAnimationContext, GetPlayingSequenceUsingTag(), Payload);
149 }
150
151 if (TargetEvent.mIsOneTimeEvent == true)
152 {
153 UnregisterTagEventOnMontage(EventTag);
154 }
155 }
156 if (HasAllMontageEvent == true)
157 {
158 /* 모든 시퀀스 대상 이벤트 실행 */
159 if (mAllMontageEvents[EventTag].OnTriggerAnimationEvent.IsBound() == true)
160 {
161 mAllMontageEvents[EventTag].OnTriggerAnimationEvent.Broadcast(mActiveAnimationContext, GetPlayingSequenceUsingTag(), Payload);
162 }
163 }
164
165 return true;
166}

302void UBoardActorSequencePlayer::ApplyDynamicBindings(const ULevelSequence* Sequence)
303{
304 if (Sequence == nullptr)
305 {
306 return;
307 }
308
309 if (mAutoBindOwnerActor == true)
310 {
311 AActor* OwnerActor = GetOwningActor();
312 if (OwnerActor != nullptr)
313 {
314 TArray<AActor*> OwnerActorArray;
315 OwnerActorArray.Add(OwnerActor);
316 SetBindingOnSequenceByTag(Sequence, mDefaultOwnerActorBindingTag, OwnerActorArray);
317 }
318 }
319
320 for (const auto& Pair : mDynamicActorBindings)
321 {
322 TArray<AActor*> ValidActors;
323 for (const TWeakObjectPtr<AActor>& WeakActor : Pair.Value)
324 {
325 if (WeakActor.IsValid() == true)
326 {
327 ValidActors.Add(WeakActor.Get());
328 }
329 }
330
331 if (ValidActors.IsEmpty() == true)
332 {
333 continue;
334 }
335
336 SetBindingOnSequenceByTag(Sequence, Pair.Key, ValidActors);
