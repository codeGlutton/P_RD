68void ACombatDicePreviewActor::ApplyDiceMesh(int32 FaceCount)
69{
70 if (mDiceMesh == nullptr)
71 {
72 return;
73 }
74
75 TObjectPtr<UStaticMesh>* Found = mDiceMeshAssets.Find(FaceCount);
76 if (Found == nullptr)
77 {
78 Found = mDiceMeshAssets.Find(6); // 폴백: 큐브
79 }
80 UStaticMesh* Mesh = (Found != nullptr) ? Found->Get() : nullptr;
81 mDiceMesh->SetStaticMesh(Mesh);
82
83 if (Mesh != nullptr)
84 {
85 // 바운딩 반지름을 DiceRadius로 맞춰 주사위 종류별 크기 통일(FBX 임포트 스케일과 무관하게).
86 const float MeshRadius = static_cast<float>(Mesh->GetBounds().SphereRadius);
87 const float Scale = (MeshRadius > KINDA_SMALL_NUMBER) ? (DiceRadius / MeshRadius) : 1.0f;
88
89 // 외접구(코너까지) 기준 정규화는 뾰족한 주사위(d4/d6)의 몸통을 작게 만든다.
90 // 종류별 보정 배율로 체감 크기를 맞춘다(눈으로 튜닝하는 값).
91 float SizeCorrection = 1.0f;
92 switch (FaceCount)
93 {
94 case 4: SizeCorrection = 1.45f; break;
95 case 6: SizeCorrection = 1.15f; break;
96 case 8: SizeCorrection = 1.18f; break;
97 case 10: SizeCorrection = 1.10f; break;
98 case 12: SizeCorrection = 0.92f; break;
99 case 20: SizeCorrection = 0.85f; break;
100 default: break;
101 }
102
103 // 보정은 부모(mDiceRoot)에 걸어 몸통·면 텍스처판·숫자가 함께 스케일되게 한다.
104 // (몸통 메시에만 걸면 면판/숫자가 원래 반경에 남아 커진 몸통 속에 파묻힌다.)
105 if (mDiceRoot != nullptr)
106 {
107 mDiceRoot->SetRelativeScale3D(FaceCount == 2 ? FVector(0.96f) : FVector(SizeCorrection));
108 }
109
110 const FVector MeshScale = FaceCount == 2
111 ? FVector(Scale * 1.08f, Scale * 1.08f, Scale * 0.105f)
112 : FVector(Scale);
113 mDiceMesh->SetRelativeScale3D(MeshScale);
114
115 if (mCoinRimMesh != nullptr)
116 {
117 const bool bShowCoinRim = false;
118 mCoinRimMesh->SetStaticMesh(bShowCoinRim ? Mesh : nullptr);
119 mCoinRimMesh->SetVisibility(bShowCoinRim);
120 if (bShowCoinRim)
121 {
122 mCoinRimMesh->SetRelativeLocation(FVector(0.0f, 0.0f, -Scale * 3.2f));
123 mCoinRimMesh->SetRelativeScale3D(FVector(Scale * 1.12f, Scale * 1.12f, Scale * 0.08f));
124 }
125 }
126 }
127 else if (mCoinRimMesh != nullptr)
128 {
129 mCoinRimMesh->SetStaticMesh(nullptr);
130 mCoinRimMesh->SetVisibility(false);
131 }
132
133 if (mDiceMaterial != nullptr)
134 {
135 ApplyMaterialToEverySlot(mDiceMesh, mDiceMaterial);
136 }
137 if (mCoinRimMesh != nullptr && mCoinRimMaterial != nullptr)
138 {
139 ApplyMaterialToEverySlot(mCoinRimMesh, mCoinRimMaterial);
140 }
141}
