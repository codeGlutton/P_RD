61bool FSkillComponentStatusEffectTests::RunTest(const FString& Parameters)
62{
63 FTestObjectScope Scope;
64
65 UWorld* World = GetAnyGameWorldForSkillTests();
66 if (World == nullptr)
67 {
68 World = GWorld;
69 }
70 if (TestNotNull(TEXT("유효한 UWorld"), World) == false)
71 {
72 return false;
73 }
74
75 // 유닛 생성 후 슬롯 0에 더미 스킬 장착
76 UMockEnemyUnitModel* Unit = Scope.New<UMockEnemyUnitModel>(World);
77 Unit->Initialize();
78 Unit->BeginPlay();
79
80 USkillComponentModel* SkillComp = Unit->GetSkillComponentModel();
81 if (TestNotNull(TEXT("스킬 컴포넌트 모델"), SkillComp) == false)
82 {
83 return false;
84 }
85 UAttributeSetComponentModel* AttrComp = Unit->GetAttributeComponentModel();
86 if (TestNotNull(TEXT("속성 컴포넌트 모델"), AttrComp) == false)
87 {
88 return false;
89 }
90
91 SkillComp->SetSkillFrom(TArray<TSoftObjectPtr<UStaticSkillData>>());
92 SkillComp->SetSkill(0, MakeDummySkill(World, Scope));
93
94 /* [1] 기본 상태: 사용 가능 */
95 TestTrue(TEXT("기본: 스킬 사용 가능"), SkillComp->CanActiveSkill(0));
96
97 /* [2] 속박: 사용 가능 */
98 AttrComp->AddLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Root);
99 TestTrue(TEXT("속박: 스킬 사용 가능"), SkillComp->CanActiveSkill(0));
100
101 /* [3] 기절 추가: 사용 불가 */
102 AttrComp->AddLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Stun);
103 TestFalse(TEXT("기절: 스킬 사용 불가"), SkillComp->CanActiveSkill(0));
104
105 /* [4] 기절 제거, 속박만 남음: 사용 가능 */
106 AttrComp->RemoveLooseGameplayTag(EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Stun);
107 TestTrue(TEXT("기절 해제 후 속박만: 스킬 사용 가능"), SkillComp->CanActiveSkill(0));
108
109 return true;
110}
