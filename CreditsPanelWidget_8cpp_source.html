255 const TCHAR* DocumentFile = Owner->IsKorean() && Page.KoreanFile ? Page.KoreanFile : Page.File;

257 if (!FFileHelper::LoadFileToString(Content, *Path))

259 UE_LOG(LogTemp, Error, TEXT("Credits: unable to read %s"), *Path);

261 }
262 Content.ReplaceInline(TEXT("\r"), TEXT(""));

295 Entry->AddSlot().AutoHeight()[ Text(FText::FromString(Author), 26, RDCredits::Ink, ETextJustify::Left, true) ];
