178bool USkillTacticalDiagramWidget::SelectCovers(const int32 Row,
179 const int32 Column, const int32 OriginRow, const int32 OriginColumn,
180 const int32 Range, const ECombatSkillSelectShapeUI Shape) const
181{
182 const int32 DeltaRow = FMath::Abs(Row - OriginRow);
183 const int32 DeltaColumn = FMath::Abs(Column - OriginColumn);
184 const int32 Manhattan = DeltaRow + DeltaColumn;
185 const int32 Chebyshev = FMath::Max(DeltaRow, DeltaColumn);
186 switch (Shape)
187 {
188 case ECombatSkillSelectShapeUI::Cross:
189 case ECombatSkillSelectShapeUI::Line:
190 return (DeltaRow == 0 || DeltaColumn == 0) && Manhattan <= Range;
191 case ECombatSkillSelectShapeUI::Square:
192 return Chebyshev <= Range;
193 case ECombatSkillSelectShapeUI::Diagonal:
194 return Chebyshev <= Range && (DeltaRow == 0 || DeltaColumn == 0
195 || DeltaRow == DeltaColumn);
196 case ECombatSkillSelectShapeUI::Single:
197 case ECombatSkillSelectShapeUI::None:
198 default:
199 return DeltaRow == 0 && DeltaColumn == 0;
200 }
201}

203bool USkillTacticalDiagramWidget::HitCovers(const int32 Row, const int32 Column,
204 const int32 TargetRow, const int32 TargetColumn, const int32 Range,
205 const ECombatSkillHitShapeUI Shape) const
206{
207 const int32 DeltaRow = FMath::Abs(Row - TargetRow);
208 const int32 DeltaColumn = FMath::Abs(Column - TargetColumn);
209 switch (Shape)
210 {
211 case ECombatSkillHitShapeUI::Cross:
212 return (DeltaRow == 0 || DeltaColumn == 0)
213 && DeltaRow + DeltaColumn <= Range;
214 // 8방향: 직교 축 또는 정확한 대각선(체비셰프 거리 이내).
215 // 게임플레이 진실(TileMapModel::GetEffectTiles의 Star)과 같은 모양이다.
216 case ECombatSkillHitShapeUI::Star:
217 return (DeltaRow == 0 || DeltaColumn == 0 || DeltaRow == DeltaColumn)
218 && FMath::Max(DeltaRow, DeltaColumn) <= Range;
219 case ECombatSkillHitShapeUI::Circle:
220 return FMath::Max(DeltaRow, DeltaColumn) <= Range;
221 case ECombatSkillHitShapeUI::Single:
222 case ECombatSkillHitShapeUI::None:
223 default:
224 return DeltaRow == 0 && DeltaColumn == 0;
225 }

227FVector2D USkillTacticalDiagramWidget::CellCenter(const int32 Row,
228 const int32 Column) const
229{
230 const int32 Index = Row * ColumnCount + Column;
231 const UImage* Cell = mCells.IsValidIndex(Index) ? mCells[Index].Get() : nullptr;
232 const UCanvasPanelSlot* CanvasSlot = Cell != nullptr
233 ? Cast<UCanvasPanelSlot>(Cell->Slot) : nullptr;
234 if (CanvasSlot == nullptr)
235 {
236 return FVector2D::ZeroVector;
237 }
238
239 // TacticalBoardRotate(45도) -> TacticalBoardTilt(Y 55%) 순서의
240 // 렌더 변환을 그대로 계산해, 판 밖에 있는 직립 마커를 셀 중심에 맞춘다.
241 const FVector2D LocalCenter = CanvasSlot->GetPosition()
242 + CanvasSlot->GetSize() * .5f;
243 const FVector2D FromPivot = LocalCenter - FVector2D(BoardCenter);
244 const FVector2D Rotated(
245 (FromPivot.X - FromPivot.Y) * InvSqrtTwo + BoardCenter,
246 (FromPivot.X + FromPivot.Y) * InvSqrtTwo + BoardCenter);
247 const FVector2D Tilted(Rotated.X,
248 (Rotated.Y - BoardCenter) * BoardTiltY + BoardCenter);
249 return BoardOrigin + Tilted;
250}

263void USkillTacticalDiagramWidget::RefreshDiagram()
264{
265 if (mCells.Num() != RowCount * ColumnCount
266 || mNeutralTileTexture == nullptr || mSelectTileTexture == nullptr
267 || mEffectTileTexture == nullptr)
268 {
269 return;
270 }
271
272 const int32 ActualSelectRange = FMath::Max(0,
273 FMath::RoundToInt(mDetail.mTargeting.mSelectRange));
274 const int32 ActualHitRange = FMath::Max(0,
275 FMath::RoundToInt(mDetail.mTargeting.mHitRange));
276 const bool bSelfTarget = mDetail.mTargeting.mSelectShape
277 == ECombatSkillSelectShapeUI::Single
278 || mDetail.mTargeting.mSelectShape == ECombatSkillSelectShapeUI::None;
279 // 범위 모식도는 실제 전장 위치를 흉내 내는 미니맵이 아니라 모양과 거리를
280 // 비교하는 설명판이다. 사정 범위는 시전자, 영향 범위는 선택 지점을 각각
281 // 9x9 정중앙에 놓아 비대칭 여백 때문에 범위가 왜곡돼 보이지 않게 한다.
282 const int32 TargetRow = RowCount / 2;
283 const int32 TargetColumn = ColumnCount / 2;
284 const int32 PreviewHitRange = FMath::Clamp(ActualHitRange, 0, 2);
285 const bool bBoardVisible = mPreviewMode != EPreviewMode::Hidden;
286 const bool bSelectMode = mPreviewMode == EPreviewMode::SelectRange;
287 const bool bEffectMode = mPreviewMode == EPreviewMode::EffectRange;
288 if (mBackdrop != nullptr)
289 {
290 mBackdrop->SetVisibility(bBoardVisible
291 ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
292 }
293 if (mBoardTilt != nullptr)
294 {
295 mBoardTilt->SetVisibility(bBoardVisible
296 ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
297 }
298
299 for (int32 Row = 0; Row < RowCount; ++Row)
300 {
301 for (int32 Column = 0; Column < ColumnCount; ++Column)
302 {
303 const int32 Index = Row * ColumnCount + Column;
304 UImage* Cell = mCells[Index];
305 if (Cell == nullptr)
306 {
307 continue;
308 }
309 const bool bEffect = bEffectMode && HitCovers(Row, Column, TargetRow,
310 TargetColumn, PreviewHitRange, mDetail.mTargeting.mHitShape);
311 const bool bSelect = bSelectMode && SelectCovers(Row, Column,
312 CasterRow, CasterColumn, ActualSelectRange,
313 mDetail.mTargeting.mSelectShape);
314 UTexture2D* Texture = bEffect ? mEffectTileTexture.Get()
315 : (bSelect ? mSelectTileTexture.Get() : mNeutralTileTexture.Get());
316 Cell->SetBrushFromTexture(Texture, false);
317 FSlateBrush Brush = Cell->GetBrush();
318 Brush.DrawAs = ESlateBrushDrawType::Image;
319 Cell->SetBrush(Brush);
320 const FLinearColor CellTint = bEffect
321 ? FLinearColor(1.f, .72f, .52f, 1.f)
322 : (bSelect ? FLinearColor(.12f, .82f, 1.f, 1.f)
323 : FLinearColor(.58f, .55f, .50f, .58f));
324 Cell->SetColorAndOpacity(CellTint);
325 }
326 }
327
328 const FVector2D CasterCenter = CellCenter(CasterRow, CasterColumn)
329 + FVector2D(0.f, -12.f);
330 const FVector2D TargetCenter = CellCenter(TargetRow, TargetColumn)
331 + FVector2D(0.f, -12.f);
332 // 모바일에서 마커가 1칸보다 크게 읽히지 않도록 링을 포함한 플레이어/적
333 // 표시 묶음을 기존 크기의 정확히 50%로 축소한다.
334 PlaceCentered(mCasterRing, CasterCenter, FVector2D(38.f, 38.f));
335 PlaceCentered(mCasterMarker, CasterCenter, FVector2D(25.f, 29.f));
336 PlaceCentered(mTargetRing, TargetCenter, FVector2D(38.f, 38.f));
337 PlaceCentered(mTargetMarker, TargetCenter, FVector2D(25.f, 29.f));
338 if (mCasterRing != nullptr)
339 {
340 mCasterRing->SetVisibility(bSelectMode
341 ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
342 }
343 if (mCasterMarker != nullptr)
344 {
345 mCasterMarker->SetVisibility(bSelectMode
346 ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
347 }
348 if (mTargetRing != nullptr)
349 {
350 mTargetRing->SetVisibility(bEffectMode
351 ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
352 }
353 if (mTargetMarker != nullptr)
354 {
355 mTargetMarker->SetVisibility(bEffectMode
356 ? ESlateVisibility::HitTestInvisible : ESlateVisibility::Collapsed);
357 }
358
359 if (mEffectRing != nullptr)
360 {
361 mEffectRing->SetVisibility(ESlateVisibility::Collapsed);
362 }
363
364 // LineToTarget(PR #466): 영향 시점이 시전자→조준 경로 전체다. 사정 범위
365 // 프리뷰에서 시전자 중심에서 조준 방향으로 사거리만큼 경로 점선을 그려
366 // "지나가는 칸 전부가 맞는다" 는 것을 알린다. 조준 칸 하나짜리(Default)는
367 // 점선이 오해만 부르므로 계속 숨긴다.
368 const bool bLineToTarget = mDetail.mTargeting.mTargetPattern
369 == ECombatSkillTargetPatternUI::LineToTarget;
370 const int32 RayLength = FMath::Clamp(
371 ActualSelectRange, 1, ColumnCount - 1 - CasterColumn);
372 const FVector2D RayEnd =
373 CellCenter(CasterRow, CasterColumn + RayLength) + FVector2D(0.f, -12.f);
374 for (int32 Index = 0; Index < mRouteDots.Num(); ++Index)
375 {
376 UImage* Dot = mRouteDots[Index];
377 if (Dot == nullptr)
378 {
379 continue;
380 }
381 const bool bShow = bSelectMode && bLineToTarget;
382 Dot->SetVisibility(bShow ? ESlateVisibility::HitTestInvisible
383 : ESlateVisibility::Collapsed);
384 if (bShow)
385 {
386 const float Alpha = float(Index + 1) / float(mRouteDots.Num() + 1);
387 PlaceCentered(Dot, FMath::Lerp(CasterCenter, RayEnd, Alpha),
388 FVector2D(6.f, 6.f));
389 }
390 }
391
392 if (mSelectLegend != nullptr)
393 {
394 mSelectLegend->SetText(bSelfTarget
395 ? LOCTEXT("SelectLegendSelf", "자신을 중심으로 발동")
396 : FText::Format(LOCTEXT("SelectLegendRange", "사정 범위 {0}칸"),
397 FText::AsNumber(ActualSelectRange)));
398 }
399 if (mEffectLegend != nullptr)
400 {
401 mEffectLegend->SetText(ActualHitRange <= 0
402 ? LOCTEXT("EffectLegendSingle", "영향 범위 대상 1칸")
403 : FText::Format(LOCTEXT("EffectLegendRange", "영향 범위 {0}칸"),
404 FText::AsNumber(ActualHitRange)));
405 }
406 if (mSelectButton != nullptr)
407 {
408 mSelectButton->SetBackgroundColor(bSelectMode
409 ? FLinearColor(.08f, .48f, .92f, 1.f)
410 : FLinearColor(.30f, .25f, .18f, .92f));
411 }
412 if (mEffectButton != nullptr)
413 {
414 mEffectButton->SetBackgroundColor(bEffectMode
415 ? FLinearColor(.78f, .18f, .06f, 1.f)
416 : FLinearColor(.30f, .25f, .18f, .92f));
417 }
418}
