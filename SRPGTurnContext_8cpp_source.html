156void FSRPGTurnContext::OnEndCurrentAction(TSharedRef<FSRPGAction> Action, ESRPGActionResult ActionResult)
157{
158 mActions.Pop();
159
160 // 턴 소모 액션 처리
161 bool IsBlockTurnSuccessfully = Action->IsTurnEndingAction() == true && ActionResult == ESRPGActionResult::Succeeded;
162 if (IsBlockTurnSuccessfully == true)
163 {
164 mPhase = ESRPGTurnPhase::TurnAbort;
165 mResult = ESRPGTurnResult::Succeeded;
166 }
167
168 // 턴 종료 여부 체크
169 if (mPhase == ESRPGTurnPhase::TurnAbort)
170 {
171 EndTurn();
172 return;
173 }
174
175 checkf(mPhase == ESRPGTurnPhase::ActionPlay, TEXT("액션 종료 절차 오류"));
176 mPhase = ESRPGTurnPhase::ActionSelect;
177
178 StartNextAction();
179}
