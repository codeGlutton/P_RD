121ESRPGActionCommandResult FSRPGTurnContext::RouteCommand(TSharedPtr<const FSRPGActionCommand> Command)
122{
123 ESRPGActionCommandResult Result = ESRPGActionCommandResult::Ignored;
124
125 if (mTurnPhase != ESRPGTurnPhase::TurnPlay)
126 {
127 // 턴 실행 외 상황에서 진입한 명령은 무시 처리
128 return Result;
129 }
130
131 TSharedPtr<FSRPGAction> CurAction = nullptr;
132 mActions.Peek(CurAction);
133 if (CurAction != nullptr)
134 {
135 /* 실행 중인 액션 객체에 커맨드 처리 요청 */
136
137 Result = CombineSRPGActionCommandResult(CurAction->HandleCommand(Command), Result);
138 if (Result == ESRPGActionCommandResult::Handled)
139 {
140 return Result;
141 }
142 }
143
144 /* 액션 대기 중인 경우 직접 처리 */
145
146 Result = CombineSRPGActionCommandResult(HandleActionCreationCommand(Command), Result);
147 if (Result == ESRPGActionCommandResult::Handled)
148 {
149 return Result;
150 }
151
152 /* 커맨드 처리 실패 시 Fallback 처리 */
153
154 return CombineSRPGActionCommandResult(HandleFallbackCommand(Command), Result);
155}

157ESRPGActionCommandResult FSRPGTurnContext::HandleActionCreationCommand(TSharedPtr<const FSRPGActionCommand> Command)
158{
159 TSharedPtr<FSRPGAction> NewAction = Command->CreateAction();
160 if (NewAction == nullptr)
161 {
162 /* 생성 명령 외는 처리 않함 */
163
164 return ESRPGActionCommandResult::Ignored;
165 }
166
167 /* 새로운 Action 생성 후, 예약 걸어두기 */
168
169 NewAction->ReserveCommand(Command);
170 EnqueueAction(NewAction);
171
172 return ESRPGActionCommandResult::Handled;
173}

231void FSRPGTurnContext::OnEndCurrentAction(TSharedRef<FSRPGAction> Action, ESRPGActionResult ActionResult)
232{
233 mActions.Pop();
234
235 // 턴 소모 액션 처리
236 bool IsBlockTurnSuccessfully = Action->ConsumesTurn() == true && ActionResult == ESRPGActionResult::Succeeded;
237 if (IsBlockTurnSuccessfully == true)
238 {
239 mTurnPhase = ESRPGTurnPhase::TurnAbort;
240 mTurnResult = ESRPGTurnResult::Succeeded;
241 }
242
243 USRPGCombatSubsystem* CombatSubsystem = GetWorld()->GetSubsystem<USRPGCombatSubsystem>();
244 checkf(CombatSubsystem != nullptr, TEXT("전투 서브시스템 nullptr"));
245
246 // 전투 상태 평가
247 CombatSubsystem->EvaluateCombatStates();
248
249 // 턴 종료 여부 체크
250 if (mTurnPhase == ESRPGTurnPhase::TurnAbort)
251 {
252 EndTurn();
253 return;
254 }
255
256 DequeueAction();

259void FSRPGTurnContext::EvaluateTurnEndState(bool ForceAbort)
260{
261 checkf(mOwner != nullptr, TEXT("턴 주인 미존재"));
262
263 if (mTurnPhase != ESRPGTurnPhase::TurnPlay)
264 {
265 return;
266 }
267
268 /* 외부에서 강제 종료되는가? */

270
271 if (ForceAbort == true || mOwner->IsDead() == true)
272 {
