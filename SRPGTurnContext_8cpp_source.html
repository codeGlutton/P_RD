25ESRPGCommandResult USRPGActionCreationCommandHandler::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
26{
27 if (Command.Get().mRequestedAction == nullptr)
28 {
29 /* 생성 명령 외는 처리 않함 */
30
31 return ESRPGCommandResult::Ignored;
32 }
33
34 USRPGAction* NewAction = NewObject<USRPGAction>(this, Command.Get().mRequestedAction);
35 if (NewAction == nullptr)
36 {
37 /* 생성 명령 외는 처리 않함 */
38
39 return ESRPGCommandResult::Ignored;
40 }
41
42 /* 새로운 액션에 명령 예약 걸고 넣기 */
43
44 NewAction->ReserveInitializeCommand(Command);
45
46 USRPGTurnContext* TurnContext = mParent.Get();
47 checkf(TurnContext != nullptr, TEXT("턴 컨텍스트 nullptr"));
48 USRPGCombatModel* CombatModel = TurnContext->GetParent();
49 checkf(CombatModel != nullptr, TEXT("전투 모델 nullptr"));
50
51 if (CombatModel->PushAction(NewAction) == false)
52 {
53 return ESRPGCommandResult::Ignored;
54 }
55 return ESRPGCommandResult::Handled;
56}

120void USRPGTurnContext::BeginTurn()
121{
122 checkf(mTurnPhase == ESRPGTurnPhase::TurnInit, TEXT("이미 턴 진행 중에 재실행 오류"));
123 mTurnPhase = ESRPGTurnPhase::TurnStart;
124
125 UE_LOG(LogSRPGCombat, Log, TEXT("턴 시작"));
126
127 // 턴 시작 연출
128 TSharedPtr<FPresentationBarrier> PresentationBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
129
130 // 턴 실행
131 checkf(mTurnPhase == ESRPGTurnPhase::TurnStart, TEXT("턴 진입 절차 오류"));
132 mTurnPhase = ESRPGTurnPhase::TurnPlay;
133
134 USRPGCommandRouterModel* CommandRouterModel = GetWorldSubsystemModel<USRPGCommandRouterModel>(this);
135 checkf(CommandRouterModel != nullptr, TEXT("명령 라우터 모델 nullptr"));
136 UPassiveComponentModel* PassiveComponentModel = mOwner->GetPassiveComponentModel();
137 checkf(PassiveComponentModel != nullptr, TEXT("패시브 컴포넌트 nullptr"));
138 USRPGCombatModel* CombatModel = mParent.Get();
139 checkf(CombatModel != nullptr, TEXT("전투 모델 nullptr"));
140
141 // 핸들러 등록
142 {
143 CommandRouterModel->OnHandleCommand.AddDynamic(this, &USRPGTurnContext::OnHandleCommand);
144 for (TScriptInterface<ISRPGCommandHandler>& TurnDefaultCommandHandler : mTurnDefaultCommandHandlers)
145 {
146 CommandRouterModel->RegisterCommandHandler(TurnDefaultCommandHandler);
147 }
148 }
149
150 // 로그 작성
151 GetWorldEventLogger(this)->BeginTurnLog(mOwner->GetModelId(), mOwner->GetClass());
152
153 // 유닛 턴 시작 단계
154 mOwner->OnBeginTurn();
155
156 // 전투 상태 평가
157 CombatModel->EvaluateCombatStates();
158
159 // 턴 종료 여부 체크
160 if (mTurnPhase == ESRPGTurnPhase::TurnAbort)
161 {
162 EndTurn();
163 return;
164 }
165
166 if (mOwner->IsPlayerUnitModel() == true)
167 {
168 /* 플레이어의 경우 주사위 굴리기 액션 추가 */
169
170 TInstancedStruct<FSRPGCommand> DicePrepareCommand;
171 DicePrepareCommand.InitializeAs<FSRPGDicePrepareCommand>();
172 DicePrepareCommand.GetMutable<FSRPGDicePrepareCommand>().OnShowDicePanelUI.AddWeakLambda(this, [this]() {
173 OnShowDicePanelAtTurnStartUI.Broadcast(this);
174 });
175
176 CommandRouterModel->SummitCommand(DicePrepareCommand);
177 }
178 else
179 {
180 /* AI의 경우 움직임 판단 로직 시작 */
181
182 UEnemyUnitModel* Enemy = Cast<UEnemyUnitModel>(mOwner.Get());
183 UUnitModel* Player = CombatModel->GetPlayerUnit();
184 UTileMapModel* TileMap = CombatModel->GetTileMap();
185
186 // PlanTurn에서 Command 리스트를 리턴하면 순서대로 라우터에 전달
187 // @note 스킬 랜덤 선택은 시뮬/라이브 동일 결과를 위해 룸의 이벤트 스트림 사용
188 for (TInstancedStruct<FSRPGCommand>& Command : USRPGEnemyTurnPlanner::PlanTurn(Enemy, Player, TileMap, URandomStreamFunctionLibrary::GetEventStream(this)))
189 {
190 CommandRouterModel->SummitCommand(Command);
191 }
192 }
193 }));
194 OnBeginTurnUI.Broadcast(PresentationBarrier, this);
195}

207void USRPGTurnContext::EndTurn()
208{
209 checkf(mTurnPhase == ESRPGTurnPhase::TurnAbort, TEXT("턴 종료 절차 오류"));
210 mTurnPhase = ESRPGTurnPhase::TurnEnd;
211
212 UPassiveComponentModel* PassiveComponentModel = mOwner->GetPassiveComponentModel();
213 checkf(PassiveComponentModel != nullptr, TEXT("패시브 컴포넌트 nullptr"));
214
215 UE_LOG(LogSRPGCombat, Log, TEXT("턴 종료"));
216
217 // 유닛 턴 종료 단계
218 mOwner->OnEndTurn();
219
220 // 로그 작성
221 GetWorldEventLogger(this)->EndTurnLog();
222
223 // 턴 종료 연출
224 TSharedPtr<FPresentationBarrier> PresentationBarrier = FPresentationBarrier::Make(FOnFinishPresentation::CreateWeakLambda(this, [this]() {
225 if (IsPermanent() == false)
226 {
227 --mLifeCount;
228 }
229 mReservedActions.Empty();
230 // 큐를 비우면 소비 인덱스도 함께 되돌려야 한다. 이 컨텍스트는 다음 턴에 재사용되므로, 인덱스가 남아 있으면
231 // 새로 등록된 액션이 mReservedActions.Num() > mHeadActionIndex 조건을 통과하지 못해 영원히 시작되지 않는다.
232 mHeadActionIndex = 0;
233
234 // 핸들러 등록 해제
235 USRPGCommandRouterModel* CommandRouterModel = GetWorldSubsystemModel<USRPGCommandRouterModel>(this);
236 checkf(CommandRouterModel != nullptr, TEXT("명령 라우터 모델 nullptr"));
237 for (TScriptInterface<ISRPGCommandHandler>& TurnDefaultCommandHandler : mTurnDefaultCommandHandlers)
238 {
239 CommandRouterModel->UnregisterCommandHandler(TurnDefaultCommandHandler);
240 }
241 CommandRouterModel->OnHandleCommand.RemoveDynamic(this, &USRPGTurnContext::OnHandleCommand);
242
243 checkf(mTurnPhase == ESRPGTurnPhase::TurnEnd, TEXT("턴 종료 절차 오류"));
244 mTurnPhase = ESRPGTurnPhase::TurnInit;
245
246 USRPGCombatModel* CombatModel = mParent.Get();
247 checkf(CombatModel != nullptr, TEXT("전투 모델 nullptr"));
248 CombatModel->OnEndCurrentTurn(this, mTurnResult);
249 }));
250 OnEndTurnUI.Broadcast(PresentationBarrier, this, mTurnResult);
251}

264void USRPGTurnContext::OnEndCurrentAction(USRPGAction* Action, ESRPGActionResult ActionResult)
265{
266 // 턴 소모 액션 처리
267 bool IsBlockTurnSuccessfully = Action->ConsumesTurn() == true && ActionResult == ESRPGActionResult::Succeeded;
268 if (IsBlockTurnSuccessfully == true)
269 {
270 mTurnPhase = ESRPGTurnPhase::TurnAbort;
271 mTurnResult = ESRPGTurnResult::Succeeded;
272 }
273
274 USRPGCombatModel* CombatModel = mParent.Get();
275 checkf(CombatModel != nullptr, TEXT("전투 모델 nullptr"));
276
277 // 전투 상태 평가
278 CombatModel->EvaluateCombatStates();
279
280 // 강제 중단
281 if (mShouldTerminateAfterAction == true)
282 {
283 mShouldTerminateAfterAction = false;
284 return;
285 }
286
287 // 턴 종료 여부 체크
288 if (mTurnPhase == ESRPGTurnPhase::TurnAbort)
289 {
290 EndTurn();
291 return;
292 }
293
294 // 예약된 액션 꺼내기
295 DequeueAction();
296
297 // 다음 액션 실행
298 if (mTurnPhase == ESRPGTurnPhase::TurnPlay && mReservedActions.Num() > mHeadActionIndex)
299 {
300 mReservedActions[mHeadActionIndex]->TryBeginAction();
301 }
302}

324void USRPGTurnContext::EnqueueAction(USRPGAction* NewAction)
325{
326 const bool IsPlayingInPlayAction = mReservedActions.Num() > mHeadActionIndex && mReservedActions[mHeadActionIndex]->GetActionType() == ESRPGActionType::InPlayAction;
327 if (IsPlayingInPlayAction == true && NewAction->GetActionType() == ESRPGActionType::BuildAction)
328 {
329 return;
330 }
331
332 NewAction->OnBeginActionUI.AddWeakLambda(this, [this](TSharedPtr<FPresentationBarrier> Barrier, const USRPGAction* Action) {
333 OnBeginAnyActionUI.Broadcast(Barrier, this, Action);
334 });
335 NewAction->OnEndActionUI.AddWeakLambda(this, [this](TSharedPtr<FPresentationBarrier> Barrier, const USRPGAction* Action, ESRPGActionResult Result) {
336 OnEndAnyActionUI.Broadcast(Barrier, this, Action, Result);
337 });
338
339 // 액션 등록
340 mReservedActions.Add(NewAction);
341 // 액션 초기화
342 NewAction->InitAction(this, mOwner.Get());
343}

397{
398 if (mTurnPhase == ESRPGTurnPhase::TurnPlay && mReservedActions.Num() > mHeadActionIndex)
399 {
400 const int32 NewHeadActionIndex = mHeadActionIndex + 1;
401 while (mReservedActions.Num() != NewHeadActionIndex)

412 mReservedActions[mHeadActionIndex]->MarkActionCompleted(ESRPGActionResult::Cancelled);
413 mReservedActions[mHeadActionIndex]->TryEndAction();
