79int32 UDicePanelWidget::FindDicePanelIndexAtPosition(const FVector2D& ScreenPosition) const
80{
81 int32 BestIndex = INDEX_NONE;
82 int32 BestZOrder = TNumericLimits<int32>::Lowest();
83 const int32 DiceCount = FMath::Min(mDicePanelViews.Num(), DicePanelSlotCount);
84
85 for (int32 DiceIndex = 0; DiceIndex < DiceCount; ++DiceIndex)
86 {
87 const UImage* DiceImage = mDicePanelImages.IsValidIndex(DiceIndex) ? mDicePanelImages[DiceIndex].Get() : nullptr;
88 if (DiceImage == nullptr || DiceImage->GetVisibility() == ESlateVisibility::Collapsed)
89 {
90 continue;
91 }
92
93 const bool bIsUnderDiceImage = DiceImage->GetCachedGeometry().IsUnderLocation(ScreenPosition);
94 const bool bIsUnderCard = mDicePanelCardBorders.IsValidIndex(DiceIndex)
95 && mDicePanelCardBorders[DiceIndex] != nullptr
96 && mDicePanelCardBorders[DiceIndex]->GetCachedGeometry().IsUnderLocation(ScreenPosition);
97 if (bIsUnderDiceImage == false && bIsUnderCard == false)
98 {
99 continue;
100 }
101
102 int32 ZOrder = 0;
103 if (const UCanvasPanelSlot* CanvasSlot = Cast<UCanvasPanelSlot>(DiceImage->Slot))
104 {
105 ZOrder = CanvasSlot->GetZOrder();
106 }
107 // Carousel에서는 카드가 겹칠 수 있으므로 같은 좌표에 여러 카드가 걸리면 가장 높은 ZOrder를 선택한다.
108 if (ZOrder >= BestZOrder)
109 {
110 BestZOrder = ZOrder;
111 BestIndex = DiceIndex;
112 }
113 }
114
115 return BestIndex;
116}

133bool UDicePanelWidget::FinishDicePanelPress(const FVector2D& ScreenPosition)
134{

137 && ReleasedIndex == mPressedDicePanelIndex
