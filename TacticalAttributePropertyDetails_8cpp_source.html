53void FTacticalAttributePropertyDetails::BuildOptions()
54{
55 mOptions.Reset();
56 mPropertyByOption.Reset();
57
58 // 해제용 선택지
59 mOptions.Add(MakeShared<FString>(TEXT("None")));
60
61 TArray<FProperty*> Properties;
62 FTacticalAttribute::GetAllAttributeProperties(Properties);
63 for (FProperty* Property : Properties)
64 {
65 const UStruct* Owner = Property->GetOwnerStruct();
66 FString Option = FString::Printf(TEXT("%s.%s"), *GetNameSafe(Owner), *Property->GetName());
67 mPropertyByOption.Add(Option, Property);
68 mOptions.Add(MakeShared<FString>(MoveTemp(Option)));
69 }
70}
