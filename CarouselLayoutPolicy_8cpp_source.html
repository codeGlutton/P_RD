1// 이 파일: 카드 한 장이 화면 어디에·얼마 크기·기울기·투명도로 놓일지를 "계산만" 한다.
2// 위젯을 직접 건드리지 않는 순수 계산(상태 없음)이라 테스트·재사용이 쉽다.
3// 두 모드: ① 비활성 = 가로 리스트 ② 활성 = 선택 카드 중심의 원형(타원) 캐러셀.
4#include "UI/CarouselLayoutPolicy.h"

6namespace
7{
8 const FVector2D CarouselOrigin(0.0f, 36.0f); // 원형 캐러셀 중심
9 const FVector2D CarouselBaseSize(180.0f, 244.0f); // 카드 기본 크기
10 const FVector2D LinearListOrigin(0.0f, 36.0f); // 가로 리스트 중심
11 constexpr float CarouselRadiusX = 360.0f; // 타원 가로 반지름(좌우로 벌어지는 정도)
12 constexpr float CarouselRadiusY = 58.0f; // 타원 세로 반지름(앞/뒤 깊이로 살짝 위아래)
13 constexpr float LinearListMaxWidth = 920.0f; // 리스트 전체 최대 폭(넘으면 간격 좁힘)
14 constexpr float LinearListPreferredSpacing = 168.0f; // 리스트 카드 간 기본 간격
15 constexpr float LinearListScale = 0.74f; // 리스트 모드 카드 축소율
16 constexpr float MinCarouselScale = 0.44f; // 가장 뒤 카드 크기
17 constexpr float MaxCarouselScale = 1.18f; // 가장 앞(선택) 카드 크기
18 constexpr float MinCarouselOpacity = 0.30f; // 가장 뒤 카드 투명도
19 constexpr float MaxCarouselOpacity = 1.00f; // 가장 앞 카드 투명도

21 // 비활성 모드: 카드들을 가로로 균등 배치. 개수가 많아 폭을 넘기면 간격을 줄여 한 줄에 맞춘다.
22 void BuildLinearListLayouts(int32 VisibleItemCount, TArray<FCarouselItemLayout>& OutLayouts)
23 {
24 const float Spacing = VisibleItemCount > 1
25 ? FMath::Min(LinearListPreferredSpacing, LinearListMaxWidth / StaticCast<float>(VisibleItemCount - 1))
26 : 0.0f;
27 const float StartX = -Spacing * StaticCast<float>(VisibleItemCount - 1) * 0.5f;
28
29 for (int32 Index = 0; Index < VisibleItemCount; ++Index)
30 {
31 FCarouselItemLayout& Layout = OutLayouts[Index];
32 Layout.mVisible = true;
33 Layout.mPosition = LinearListOrigin + FVector2D(StartX + Spacing * Index, 0.0f);
34 Layout.mSize = CarouselBaseSize;
35 Layout.mScale = FVector2D(LinearListScale, LinearListScale);
36 Layout.mAngle = 0.0f;
37 Layout.mOpacity = 1.0f;
38 Layout.mZOrder = 100 + Index;
39 }
40 }
41
42 // 활성 모드: 카드들을 타원 위에 빙 둘러 놓고, 선택 카드를 맨 앞 중앙에 둔다.
43 // 핵심 아이디어 — 선택 카드 기준 상대 위치를 각도(Angle)로 환산하고,
44 // 그 각도의 cos를 "깊이(Depth)"로 본다(앞=+1, 옆=0, 뒤=-1).
45 // 깊이로 크기·투명도·ZOrder를 동시에 정해 "앞 카드는 크고 또렷, 뒤 카드는 작고 흐릿"하게 만든다.
46 void BuildActivatedCarouselLayouts(
47 int32 VisibleItemCount,
48 int32 SelectedItemIndex,
49 const TArray<float>& ItemAngles,
50 TArray<FCarouselItemLayout>& OutLayouts)
51 {
52 const int32 SelectedIndex = (SelectedItemIndex < 0 || SelectedItemIndex >= VisibleItemCount) ? 0 : SelectedItemIndex;
53 for (int32 Index = 0; Index < VisibleItemCount; ++Index)
54 {
55 // 선택 카드를 0번(앞)으로 두는 상대 순번 → 0~2π 각도로 균등 분배.
56 const int32 RelativeIndex = (Index - SelectedIndex + VisibleItemCount) % VisibleItemCount;
57 const float Angle = 2.0f * PI * StaticCast<float>(RelativeIndex) / StaticCast<float>(VisibleItemCount);
58 const float Depth = FMath::Cos(Angle); // 앞(+1)~뒤(-1)
59 const float NormalizedDepth = (Depth + 1.0f) * 0.5f; // 0(뒤)~1(앞)로 정규화
60 const float X = FMath::Sin(Angle) * CarouselRadiusX; // 좌우 위치(sin)
61 const float Y = Depth * CarouselRadiusY; // 앞일수록 살짝 아래로(타원 깊이)
62 const float Scale = FMath::Lerp(MinCarouselScale, MaxCarouselScale, NormalizedDepth); // 앞일수록 큼
63 const float Opacity = FMath::Lerp(MinCarouselOpacity, MaxCarouselOpacity, NormalizedDepth); // 앞일수록 또렷
64 const int32 ZOrder = FMath::RoundToInt(100.0f + Depth * 100.0f); // 앞일수록 위로 겹침
65
66 FCarouselItemLayout& Layout = OutLayouts[Index];
67 Layout.mVisible = true;
68 Layout.mPosition = CarouselOrigin + FVector2D(X, Y);
69 Layout.mSize = CarouselBaseSize;
70 Layout.mScale = FVector2D(Scale, Scale);
71 Layout.mAngle = ItemAngles.IsValidIndex(Index) ? ItemAngles[Index] : 0.0f;
72 Layout.mOpacity = Opacity;
73 Layout.mZOrder = ZOrder;
74 }
75 }
76}

78void FCarouselLayoutPolicy::BuildLayouts(
79 int32 TotalItemCount,
80 int32 VisibleItemCount,
81 bool bActivated,
82 int32 SelectedItemIndex,
83 const TArray<float>& ItemAngles,
84 TArray<FCarouselItemLayout>& OutLayouts)
85{
86 OutLayouts.Reset();
87 OutLayouts.SetNum(FMath::Max(0, TotalItemCount));
88
89 const int32 ClampedVisibleItemCount = FMath::Clamp(VisibleItemCount, 0, OutLayouts.Num());
90 if (ClampedVisibleItemCount <= 0)
91 {
92 return;
93 }
94
95 if (bActivated == false)
96 {
97 BuildLinearListLayouts(ClampedVisibleItemCount, OutLayouts);
98 return;
99 }
100
101 BuildActivatedCarouselLayouts(ClampedVisibleItemCount, SelectedItemIndex, ItemAngles, OutLayouts);
102}

104float FCarouselLayoutPolicy::EaseOutCubic(float RawAlpha)
105{
106 const float ClampedAlpha = FMath::Clamp(RawAlpha, 0.0f, 1.0f);
107 const float InverseAlpha = 1.0f - ClampedAlpha;
108 return 1.0f - InverseAlpha * InverseAlpha * InverseAlpha;
109}

111FCarouselItemLayout FCarouselLayoutPolicy::InterpolateLayout(
112 const FCarouselItemLayout& StartLayout,
113 const FCarouselItemLayout& TargetLayout,
114 float Alpha,
115 float RawAlpha)
116{
117 FCarouselItemLayout ResolvedStartLayout = StartLayout;
118 FCarouselItemLayout ResolvedTargetLayout = TargetLayout;
119
120 if (ResolvedStartLayout.mVisible == false && ResolvedTargetLayout.mVisible == true)
121 {
122 ResolvedStartLayout = ResolvedTargetLayout;
123 ResolvedStartLayout.mOpacity = 0.0f;
124 }
125 else if (ResolvedStartLayout.mVisible == true && ResolvedTargetLayout.mVisible == false)
126 {
127 ResolvedTargetLayout = ResolvedStartLayout;
128 ResolvedTargetLayout.mOpacity = 0.0f;
129 }
130
131 FCarouselItemLayout CurrentLayout;
132 CurrentLayout.mVisible = ResolvedStartLayout.mVisible || ResolvedTargetLayout.mVisible;
133 CurrentLayout.mPosition = ResolvedStartLayout.mPosition + (ResolvedTargetLayout.mPosition - ResolvedStartLayout.mPosition) * Alpha;
134 CurrentLayout.mSize = ResolvedStartLayout.mSize + (ResolvedTargetLayout.mSize - ResolvedStartLayout.mSize) * Alpha;
135 CurrentLayout.mScale = ResolvedStartLayout.mScale + (ResolvedTargetLayout.mScale - ResolvedStartLayout.mScale) * Alpha;
136 CurrentLayout.mAngle = FMath::Lerp(ResolvedStartLayout.mAngle, ResolvedTargetLayout.mAngle, Alpha);
137 CurrentLayout.mOpacity = FMath::Lerp(ResolvedStartLayout.mOpacity, ResolvedTargetLayout.mOpacity, Alpha);
138 CurrentLayout.mZOrder = RawAlpha < 0.5f ? ResolvedStartLayout.mZOrder : ResolvedTargetLayout.mZOrder;
139 return CurrentLayout;
140}
