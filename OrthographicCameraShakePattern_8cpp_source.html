6float FOrthographicNoiseShaker::Update(float DeltaTime, float AmplitudeMultiplier, float FrequencyMultiplier, float& InOutCurrentOffset) const
7{
8 const float TotalAmplitude = Amplitude * AmplitudeMultiplier;
9 if (TotalAmplitude != 0.f)
10 {
11 InOutCurrentOffset += DeltaTime * Frequency * FrequencyMultiplier;
12 return TotalAmplitude * FMath::PerlinNoise1D(InOutCurrentOffset);
13 }
14 return 0.f;
15}

25void UOrthographicCameraShakePattern::StartShakePatternImpl(const FCameraShakePatternStartParams& Params)
26{
27 Super::StartShakePatternImpl(Params);
28
29 if (!Params.bIsRestarting)
30 {
31 // All offsets are random. This is because the core perlin noise implementation
32 // uses permutation tables, so if two shakers have the same initial offset and the same
33 // frequency, they will have the same exact values.
34 InitialLocationOffset = FVector3f((float)FMath::RandHelper(255), (float)FMath::RandHelper(255), (float)FMath::RandHelper(255));
35 InitialRotationOffset = FVector3f((float)FMath::RandHelper(255), (float)FMath::RandHelper(255), (float)FMath::RandHelper(255));
36 InitialOrthoWidthOffset = (float)FMath::RandHelper(255);
37
38 CurrentLocationOffset = InitialLocationOffset;
39 CurrentRotationOffset = InitialRotationOffset;
40 CurrentOrthoWidthOffset = InitialOrthoWidthOffset;
41 }
42}

52void UOrthographicCameraShakePattern::ScrubShakePatternImpl(const FCameraShakePatternScrubParams& Params, FCameraShakePatternUpdateResult& OutResult)
53{
54 // Scrubbing is like going back to our initial state and updating directly to the scrub time.
55 CurrentLocationOffset = InitialLocationOffset;
56 CurrentRotationOffset = InitialRotationOffset;
57 CurrentOrthoWidthOffset = InitialOrthoWidthOffset;
58
59 UpdateOrthographicNoise(Params.AbsoluteTime, OutResult);
60
61 const float BlendWeight = State.Scrub(Params.AbsoluteTime);
62 OutResult.ApplyScale(BlendWeight);
63}

65void UOrthographicCameraShakePattern::UpdateOrthographicNoise(float DeltaTime, FCameraShakePatternUpdateResult& OutResult)
66{
67 OutResult.Location.Y = Y.Update(DeltaTime, LocationAmplitudeMultiplier, LocationFrequencyMultiplier, CurrentLocationOffset.Y);
68 OutResult.Location.Z = Z.Update(DeltaTime, LocationAmplitudeMultiplier, LocationFrequencyMultiplier, CurrentLocationOffset.Z);
69
70 OutResult.Rotation.Roll = Roll.Update(DeltaTime, RotationAmplitudeMultiplier, RotationFrequencyMultiplier, CurrentRotationOffset.Z);
71
72 CurrentOrhoWidthValue = OrthoWidth.Update(DeltaTime, 1.f, 1.f, CurrentOrthoWidthOffset);
73}
