77bool FStatusEffectRoundDurationTests::RunTest(const FString& Parameters)
78{
79 FTestObjectScope Scope;
80
81 /* 1. 월드 및 FrameworkModel 환경 준비 */
82
83 UWorld* World = FindAnyGameWorld();
84 if (World == nullptr)
85 {
86 World = GWorld;
87 }
88 if (TestNotNull(TEXT("UWorld 환경이 유효해야 합니다."), World) == false)
89 {
90 return false;
91 }
92
93 USimulationSubsystem* SimSubsystem = World->GetSubsystem<USimulationSubsystem>();
94 if (TestNotNull(TEXT("USimulationSubsystem이 존재해야 합니다."), SimSubsystem) == false)
95 {
96 return false;
97 }
98
99 URoomInstance* RoomInstance = FSimulationSubsystemTestAccessor::GetRoomInstance(SimSubsystem);
100 if (TestNotNull(TEXT("RoomInstance가 존재해야 합니다."), RoomInstance) == false)
101 {
102 return false;
103 }
104
105 UTacticalFrameworkModel* FrameworkModel = Cast<UTacticalFrameworkModel>(RoomInstance->mAliveSubsystemModels.FindRef(UTacticalFrameworkModel::StaticClass()));
106 if (FrameworkModel == nullptr)
107 {
108 FrameworkModel = Scope.New<UTacticalFrameworkModel>(RoomInstance);
109 RoomInstance->mAliveSubsystemModels.Add(UTacticalFrameworkModel::StaticClass(), FrameworkModel);
110 // 테스트가 등록한 것이므로 파괴 전에 등록 해제
111 Scope.Defer([RoomInstance]() { RoomInstance->mAliveSubsystemModels.Remove(UTacticalFrameworkModel::StaticClass()); });
112 }
113
114 const FGameplayTag StunTag = EffectTags::GameplayEffect_StatusEffect_RoundDuration_Debuff_Stun;
115
116 /* 테스트 1: 다른 유닛만 턴을 받은 라운드에서도 기절 스택이 라운드당 1씩 감소 */
117 {
118 UTASActorModelMock* MockActorModel = Scope.New<UTASActorModelMock>(World);
119 MockActorModel->Initialize();
120 MockActorModel->BeginPlay();
121
122 UAttributeSetComponentModel* CompModel = MockActorModel->GetAttributeComponent();
123 if (TestNotNull(TEXT("CompModel이 유효해야 합니다."), CompModel) == false)
124 {
125 return false;
126 }
127 UTacticalEffectContext* EffectContext = CompModel->MakeEffectContext();
128
129 int32 TurnCount = 0;
130 FrameworkModel->AdvanceTurnDuration(TurnCount);
131 FrameworkModel->AdvanceRoundDuration(0);
132
133 // 기절 3스택 부여 (스킬의 TagGain 3에 해당)
134 const FActiveTacticalEffectHandle StunHandle = ApplyStunStacks(CompModel, EffectContext, 3);
135 if (TestTrue(TEXT("기절 이펙트가 적용되어야 합니다."), StunHandle.IsValid()) == false)
136 {
137 return false;
138 }
139 TestEqual(TEXT("부여 직후 스택 수는 3이어야 합니다."), CompModel->GetCurrentStackCount(StunHandle), 3);
140 TestTrue(TEXT("부여 직후 기절 태그를 보유해야 합니다."), CompModel->HasMatchingGameplayTag(StunTag));
141
142 // 라운드마다 소모되는 턴 수를 다르게 둠. 기절한 유닛은 느려서 턴을 못 받고,
143 // 아래 턴들은 전부 다른 유닛이 소모한 것으로 간주함.
144 // 감소량이 턴 수가 아니라 라운드 진행에만 좌우되는지 확인하기 위한 구성임.
145 const int32 OtherUnitTurnCounts[] = { 3, 5, 2 };
146 const int32 ExpectedStackCounts[] = { 2, 1, 0 };
147
148 for (int32 RoundIndex = 0; RoundIndex < UE_ARRAY_COUNT(OtherUnitTurnCounts); ++RoundIndex)
149 {
150 for (int32 TurnIndex = 0; TurnIndex < OtherUnitTurnCounts[RoundIndex]; ++TurnIndex)
151 {
152 FrameworkModel->AdvanceTurnDuration(++TurnCount);
153 }
154
155 // 한 명이라도 턴을 받은 라운드이므로 라운드가 진행됨
156 FrameworkModel->AdvanceRoundDuration(RoundIndex + 1);
157
158 const int32 ExpectedStackCount = ExpectedStackCounts[RoundIndex];
159 if (ExpectedStackCount > 0)
160 {
161 TestEqual(
162 *FString::Printf(TEXT("라운드 %d 진행 후 스택 수가 %d이어야 합니다."), RoundIndex + 1, ExpectedStackCount),
163 CompModel->GetCurrentStackCount(StunHandle), ExpectedStackCount);
164 TestTrue(
165 *FString::Printf(TEXT("라운드 %d 진행 후에도 기절 태그가 유지되어야 합니다."), RoundIndex + 1),
166 CompModel->HasMatchingGameplayTag(StunTag));
167 }
168 else
169 {
170 TestTrue(
171 *FString::Printf(TEXT("라운드 %d 진행 후 이펙트가 제거되어야 합니다."), RoundIndex + 1),
172 CompModel->GetActiveTacticalEffect(StunHandle) == nullptr);
173 TestFalse(
174 TEXT("이펙트 제거 후 기절 태그가 해제되어야 합니다."),
175 CompModel->HasMatchingGameplayTag(StunTag));
176 }
177 }
178 }
179
180 /* 테스트 2: 라운드가 진행되지 않으면 기절 스택이 줄지 않음 */
181 {
182 UTASActorModelMock* MockActorModel = Scope.New<UTASActorModelMock>(World);
183 MockActorModel->Initialize();
184 MockActorModel->BeginPlay();
185
186 UAttributeSetComponentModel* CompModel = MockActorModel->GetAttributeComponent();
187 if (TestNotNull(TEXT("CompModel이 유효해야 합니다."), CompModel) == false)
188 {
189 return false;
190 }
191 UTacticalEffectContext* EffectContext = CompModel->MakeEffectContext();
192
193 int32 TurnCount = 0;
194 FrameworkModel->AdvanceTurnDuration(TurnCount);
195 FrameworkModel->AdvanceRoundDuration(0);
196
197 const FActiveTacticalEffectHandle StunHandle = ApplyStunStacks(CompModel, EffectContext, 2);
198 if (TestTrue(TEXT("기절 이펙트가 적용되어야 합니다."), StunHandle.IsValid()) == false)
199 {
200 return false;
201 }
202
203 // 아무도 턴을 받지 못해 생략된 사이클은 AdvanceRoundDuration을 호출하지 않음.
204 // 턴만 흘러가는 경우까지 포함해, 라운드가 진행되지 않는 한 감소가 없어야 함.
205 for (int32 TurnIndex = 0; TurnIndex < 5; ++TurnIndex)
206 {
207 FrameworkModel->AdvanceTurnDuration(++TurnCount);
208 }
209
210 TestEqual(TEXT("라운드가 진행되지 않으면 스택 수가 그대로여야 합니다."), CompModel->GetCurrentStackCount(StunHandle), 2);
211 TestTrue(TEXT("라운드가 진행되지 않으면 기절 태그가 유지되어야 합니다."), CompModel->HasMatchingGameplayTag(StunTag));
212
213 // 유효한 라운드가 오면 그때 1 감소
214 FrameworkModel->AdvanceRoundDuration(1);
215 TestEqual(TEXT("유효한 라운드가 진행된 시점에만 스택이 1로 줄어야 합니다."), CompModel->GetCurrentStackCount(StunHandle), 1);
216 }
217
218 return true;
219}
