99 // PlayerController가 존재하지 않으면 함수를 종료합니다.

114 && PlayerController->IsInputKeyDown(EKeys::LeftMouseButton);
115 UpdatePointerGestures(Touches[0], Touches[1], bMousePressed, MousePosition);

118void ACombatCameraPawn::UpdatePointerGestures(const FTouchState& FirstTouch,
119 const FTouchState& SecondTouch, bool bMousePressed, const FVector2D& MousePosition)
120{
121 mTouchStates.SetNum(2);
122 if (!mTouchGestureInputEnabled)
123 {
124 for (FTouchState& State : mTouchStates) State = FTouchState();
125 return;
126 }
127 // Real/emulated touch wins, so mouse-for-touch never produces two gestures.
128 const bool bUseMouse = !FirstTouch.bIsCurrentlyPressed && !SecondTouch.bIsCurrentlyPressed
129 && bMousePressed;
130 if (bUseMouse != mUsingMouseGesture)
131 {
132 for (FTouchState& State : mTouchStates) State = FTouchState();
133 mPinchActive = false;
134 mHadTwoTouches = false;
135 mPanActive = false;
136 }
137 mUsingMouseGesture = bUseMouse;
138 FTouchState Samples[2] = { FirstTouch, SecondTouch };
139 const bool HadTwoTouches = mTouchStates[0].bIsCurrentlyPressed && mTouchStates[1].bIsCurrentlyPressed;
140 if (bUseMouse)
141 {
142 Samples[0].bIsCurrentlyPressed = true;
143 Samples[0].CurTouchPos = MousePosition;
144 }
145 // Mouse and touch share the same camera projection, bounds and modal gate.
146 for (int i = 0; i < 2; ++i)
147 {
148 mTouchStates[i].PreTouchPos = mTouchStates[i].CurTouchPos;
149 bool bPreTickTouch = mTouchStates[i].bIsCurrentlyPressed;
150 mTouchStates[i].CurTouchPos = Samples[i].CurTouchPos;
151 mTouchStates[i].bIsCurrentlyPressed = Samples[i].bIsCurrentlyPressed;
152
153 if (bPreTickTouch == 0 && mTouchStates[i].bIsCurrentlyPressed)
154 {
155 if (i == 0) mPanActive = false;
156 mTouchStates[i].StartTouchPos = mTouchStates[i].CurTouchPos;
157 mTouchStates[i].PreTouchPos = mTouchStates[i].CurTouchPos;
158 }
159 }
160
161
162 const bool HasTwoTouches = mTouchStates[0].bIsCurrentlyPressed && mTouchStates[1].bIsCurrentlyPressed;
163 if (HasTwoTouches && !HadTwoTouches)
164 {
165 mPinchStartDistance = FVector2D::Distance(mTouchStates[0].CurTouchPos, mTouchStates[1].CurTouchPos);
166 mPinchActive = false;
167 mHadTwoTouches = true;
168 }
169 if (!mTouchStates[0].bIsCurrentlyPressed && !mTouchStates[1].bIsCurrentlyPressed)
170 {
171 mPinchActive = false;
172 mHadTwoTouches = false;
173 mPanActive = false;
174 }
175
176 // Pinch 중
177 if (IsPinch())
178 {
179 if (OnPinching.IsBound())
180 {
181 OnPinching.Broadcast(mTouchStates);
182 }
183 }
184 // 드래그 중
185 else if (IsDrag())
186 {
187 if (OnDragging.IsBound())
188 {
189 OnDragging.Broadcast(mTouchStates);
190 }
191 }
192}

237bool ACombatCameraPawn::IsDrag()
238{
239 if (!mTouchStates[0].bIsCurrentlyPressed || mTouchStates[1].bIsCurrentlyPressed || mHadTwoTouches) return false;
240 mPanActive |= RDPointerGesture::HasMoved(mTouchStates[0].StartTouchPos, mTouchStates[0].CurTouchPos, !mUsingMouseGesture);
241 return mTouchStates[0].bIsCurrentlyPressed &&
242 !mTouchStates[1].bIsCurrentlyPressed &&
243 !mHadTwoTouches &&
244 mPanActive &&
245 !mTouchStates[0].PreTouchPos.Equals(mTouchStates[0].CurTouchPos, 0.01f);
246}

248bool ACombatCameraPawn::IsPinch()
249{
250 float PrePinchDis = FVector2D::Distance(mTouchStates[0].PreTouchPos, mTouchStates[1].PreTouchPos);
251 float CurPinchDis = FVector2D::Distance(mTouchStates[0].CurTouchPos, mTouchStates[1].CurTouchPos);
252
253 if (!mTouchStates[0].bIsCurrentlyPressed || !mTouchStates[1].bIsCurrentlyPressed)
254 return false;
255 mPinchActive |= FMath::Abs(CurPinchDis - mPinchStartDistance) > RDPointerGesture::PinchSlop;
256 return mPinchActive && !FMath::IsNearlyEqual(PrePinchDis, CurPinchDis);
257}

269void ACombatCameraPawn::Pinching(const TArray<FTouchState>& TouchState)
270{
271 if (!ensureMsgf(IsValid(mCameraMovementComponent), TEXT("CameraMovementComponent가 없습니다")))
272 {
273 return;
274 }
275
276 float PrePinchDis = FVector2D::Distance(mTouchStates[0].PreTouchPos, mTouchStates[1].PreTouchPos);
277 float CurPinchDis = FVector2D::Distance(mTouchStates[0].CurTouchPos, mTouchStates[1].CurTouchPos);
278
279 //mCameraMovementComponent.Get()->ZoomCamera_Instant(PrePinchDis - CurPinchDis);
280 mCameraMovementComponent.Get()->PinchZoomCamera_InstantAndMoveToViewportPosition_Instant(PrePinchDis - CurPinchDis, (mTouchStates[0].CurTouchPos + mTouchStates[1].CurTouchPos)/2);
281}
