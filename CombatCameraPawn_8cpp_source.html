59// Called every frame
60void ACombatCameraPawn::Tick(float DeltaTime)
61{
62 Super::Tick(DeltaTime);
63
64 AController* DefaultController = GetController();
65 // DefaultController가 존재하지 않으면 함수를 종료합니다.
66 if (!ensureMsgf(IsValid(DefaultController), TEXT("컨트롤러가 없습니다")))
67 {
68 return;
69 }
70
71 APlayerController* PlayerController = Cast<APlayerController>(GetController());
72 // PlayerController가 존재하지 않으면 함수를 종료합니다.
73 if (!ensureMsgf(IsValid(PlayerController), TEXT("Player 컨트롤러가 없습니다")))
74 {
75 return;
76 }
77
78 // 터치 상태를 보고 제스처를 판단한다.
79 for (int i = 0; i < 2; ++i)
80 {
81 mTouchStates[i].PreTouchPos = mTouchStates[i].CurTouchPos;
82 bool bPreTickTouch = mTouchStates[i].bIsCurrentlyPressed;
83 PlayerController->GetInputTouchState((ETouchIndex::Type)i, mTouchStates[i].CurTouchPos.X, mTouchStates[i].CurTouchPos.Y, mTouchStates[i].bIsCurrentlyPressed);
84
85 if (bPreTickTouch == 0 && mTouchStates[i].bIsCurrentlyPressed)
86 {
87 mTouchStates[i].StartTouchPos = mTouchStates[i].CurTouchPos;
88 mTouchStates[i].PreTouchPos = mTouchStates[i].CurTouchPos;
89 }
90 }
91
92
93 // Pinch 중
94 if (IsPinch())
95 {
96 if (OnPinching.IsBound())
97 {
98 OnPinching.Broadcast(mTouchStates);
99 }
100 }
101 // 드래그 중
102 else if (IsDrag())
103 {
104 if (OnDragging.IsBound())
105 {
106 OnDragging.Broadcast(mTouchStates);
107 }
108 }
109}

133bool ACombatCameraPawn::IsDrag()
134{
135 return mTouchStates[0].bIsCurrentlyPressed &&
136 mImageStabilization < FVector2D::Distance(mTouchStates[0].PreTouchPos, mTouchStates[0].CurTouchPos);
137}
138
139bool ACombatCameraPawn::IsPinch()
140{
141 float PrePinchDis = FVector2D::Distance(mTouchStates[0].PreTouchPos, mTouchStates[1].PreTouchPos);
142 float CurPinchDis = FVector2D::Distance(mTouchStates[0].CurTouchPos, mTouchStates[1].CurTouchPos);
143
144 return mTouchStates[0].bIsCurrentlyPressed &&
145 mTouchStates[1].bIsCurrentlyPressed &&
146 mImageStabilization < FMath::Abs(PrePinchDis - CurPinchDis);
147}

173void ACombatCameraPawn::RDMoveTo(int32 X, int32 Y)
174{
175 // 치트는 출시 빌드에서 본문만 비움 (UFUNCTION 선언은 전처리 제외 불가)
176#if !UE_BUILD_SHIPPING
177 // 전투 모델·타일맵·플레이어 유닛 확보 (전투 중이 아니면 무시)
178 USRPGCombatModel* CombatModel = GetWorldSubsystemModel<USRPGCombatModel>(this);
179 UTileMapModel* TileMap = CombatModel != nullptr ? CombatModel->GetTileMap() : nullptr;
180 UUnitModel* PlayerUnit = CombatModel != nullptr ? CombatModel->GetPlayerUnit() : nullptr;
181 if (TileMap == nullptr || PlayerUnit == nullptr)
182 {
183 UE_LOG(LogSRPGCombat, Warning, TEXT("RDMoveTo: 전투 진행 중이 아니라 무시"));
184 return;
185 }
186
187 // 현재 칸 → 목표 칸 최단 경로 계산 (막히거나 맵 밖이면 빈 경로)
188 const TArray<FTileIndex> Path = TileMap->FindPath(PlayerUnit->GetTileTransform().mIndex, FTileIndex(X, Y));
189 if (Path.Num() < 2)
190 {
191 UE_LOG(LogSRPGCombat, Warning, TEXT("RDMoveTo: (%d,%d)까지 경로 없음"), X, Y);
192 return;
193 }
194
195 // 확정 경로를 실은 이동 커맨드 발행 — 빌드 액션 없이 MoveAction 직행 (BuildMove와 동일 형식)
196 USRPGCommandRouterModel* CommandRouter = GetWorldSubsystemModel<USRPGCommandRouterModel>(this);
197 checkf(CommandRouter != nullptr, TEXT("명령 라우터 모델 nullptr"));
198
199 TInstancedStruct<FSRPGCommand> MoveCommand;
200 MoveCommand.InitializeAs<FSRPGMoveCommand>();
201 MoveCommand.GetMutable<FSRPGMoveCommand>().mPathTileIndexes = Path;
202 const bool Submitted = CommandRouter->SummitCommand(MoveCommand);
203
204 UE_LOG(LogSRPGCombat, Log, TEXT("RDMoveTo: (%d,%d) 경로 %d칸, 커맨드 %s"), X, Y, Path.Num(), Submitted ? TEXT("제출됨") : TEXT("거부됨"));
205#endif
206}

208void ACombatCameraPawn::RDRotate(int32 Direction)
209{
210 // 치트는 출시 빌드에서 본문만 비움 (UFUNCTION 선언은 전처리 제외 불가)
211#if !UE_BUILD_SHIPPING
212 // 전투 모델·타일맵·플레이어 유닛 확보 (전투 중이 아니면 무시)
213 USRPGCombatModel* CombatModel = GetWorldSubsystemModel<USRPGCombatModel>(this);
214 UTileMapModel* TileMap = CombatModel != nullptr ? CombatModel->GetTileMap() : nullptr;
215 UUnitModel* PlayerUnit = CombatModel != nullptr ? CombatModel->GetPlayerUnit() : nullptr;
216 if (TileMap == nullptr || PlayerUnit == nullptr)
217 {
218 UE_LOG(LogSRPGCombat, Warning, TEXT("RDRotate: 전투 진행 중이 아니라 무시"));
219 return;
220 }
221
222 // 방향 인덱스 검증 (0=Forward 1=Right 2=Backward 3=Left)
223 if (Direction < 0 || Direction >= static_cast<int32>(ETileActorDirection::Count))
224 {
225 UE_LOG(LogSRPGCombat, Warning, TEXT("RDRotate: 잘못된 방향 %d (0~3)"), Direction);
226 return;
227 }
228
229 // 배리어 없이 즉발 요청 — 뷰는 내부 무통지 배리어로 회전을 완주
230 TileMap->RotateActor(static_cast<ETileActorDirection>(Direction), PlayerUnit);
231 UE_LOG(LogSRPGCombat, Log, TEXT("RDRotate: 방향 %d로 전환 요청"), Direction);
232#endif
233}
