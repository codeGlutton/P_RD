29 bool RepairDungeonSystem(UNiagaraSystem* System)
30 {
31 if (!System || System->GetPathName() != DungeonSystemPath) return false;
32 UNiagaraNodeFunctionCall* Node = FindLegacyNode(System);
33 if (!Node || Node->NodeGuid != LegacyNodeGuid || Node->FunctionScript) return false;
34
35 // This is the converter's parameter-map module, not the similarly named dynamic input.
36 UNiagaraScript* Module = nullptr;
37 {
38 // Only the compiled system ships; this standalone module supplies editor graph source.
39 FCookLoadScope EditorOnlyLoad(ECookLoadType::EditorOnly);
40 Module = LoadObject<UNiagaraScript>(nullptr, ConverterModulePath);
41 }
42 if (!Module)
43 {
44 UE_LOG(LogRDEditor, Error, TEXT("Dungeon Niagara requires the editor-only CascadeToNiagaraConverter module: %s"), ConverterModulePath);
45 return false;
46 }
47
48 const bool bPackageWasDirty = System->GetOutermost()->IsDirty();
49 Node->FunctionScript = Module;
50 FPropertyChangedEvent ChangedProperty(FindFProperty<FObjectPropertyBase>(
51 UNiagaraNodeFunctionCall::StaticClass(), GET_MEMBER_NAME_CHECKED(UNiagaraNodeFunctionCall, FunctionScript)));
52 Node->PostEditChangeProperty(ChangedProperty);
53 System->RequestCompile(true);
54 System->WaitForCompilationComplete(false, false);
55 // The correction is regenerated on each editor/cook load. Do not prompt to save SVN content.
56 if (!bPackageWasDirty) System->GetOutermost()->SetDirtyFlag(false);
57 UE_LOG(LogRDEditor, Display, TEXT("Restored legacy dungeon Niagara module in memory: %s (valid=%d)"),
58 *System->GetPathName(), System->IsValid());
59 return true;
60 }

62 void OnAssetLoaded(UObject* Asset)

65 }

68void RegisterDungeonNiagaraCompatibility()
