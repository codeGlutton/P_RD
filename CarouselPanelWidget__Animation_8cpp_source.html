11void UCarouselPanelWidget::NativeTick(const FGeometry& MyGeometry, float InDeltaTime)
12{
13 Super::NativeTick(MyGeometry, InDeltaTime);
14
15 // 전환 중이 아니면 매 프레임 일 없음(정적 배치는 Set/Clear 때만 갱신).
16 if (mCarouselTransitionActive == false)
17 {
18 return;
19 }
20
21 // 경과/전체 길이 = 선형 진행도(RawAlpha). 여기에 EaseOutCubic을 먹여
22 // 처음 빠르게 → 끝에 천천히 멈추는 감속 곡선(Alpha)을 만든다.
23 mCarouselTransitionElapsed += InDeltaTime;
24 const float RawAlpha = mCarouselTransitionDuration > 0.0f
25 ? FMath::Clamp(mCarouselTransitionElapsed / mCarouselTransitionDuration, 0.0f, 1.0f)
26 : 1.0f;
27 const float Alpha = FCarouselLayoutPolicy::EaseOutCubic(RawAlpha);
28
29 TArray<FCarouselItemLayout> CurrentLayouts;
30 CurrentLayouts.SetNum(mCarouselTargetLayouts.Num());
31 for (int32 Index = 0; Index < mCarouselTargetLayouts.Num(); ++Index)
32 {
33 const FCarouselItemLayout StartLayout = mCarouselStartLayouts.IsValidIndex(Index)
34 ? mCarouselStartLayouts[Index]
35 : mCarouselTargetLayouts[Index];
36 CurrentLayouts[Index] = FCarouselLayoutPolicy::InterpolateLayout(
37 StartLayout,
38 mCarouselTargetLayouts[Index],
39 Alpha,
40 RawAlpha);
41 }
42 ApplyCarouselLayouts(CurrentLayouts);
43
44 // 진행도 100%면 전환 종료 + 부동소수 누적 오차 없이 목표 배치를 정확히 한 번 더 적용.
45 if (RawAlpha >= 1.0f)
46 {
47 mCarouselTransitionActive = false;
48 ApplyCarouselLayouts(mCarouselTargetLayouts);
49 }
50}

78void UCarouselPanelWidget::CaptureCurrentCarouselLayouts(TArray<FCarouselItemLayout>& OutLayouts) const
79{
80 OutLayouts.Reset();
81 OutLayouts.SetNum(mCarouselItems.Num());
82
83 for (int32 Index = 0; Index < mCarouselItems.Num(); ++Index)
84 {
85 const UWidget* Item = mCarouselItems[Index];
86 if (Item == nullptr)
87 {
88 continue;
89 }
90
91 FCarouselItemLayout& Layout = OutLayouts[Index];
92 Layout.mVisible = Item->GetVisibility() != ESlateVisibility::Collapsed;
93 Layout.mScale = Item->GetRenderTransform().Scale;
94 Layout.mAngle = Item->GetRenderTransform().Angle;
95 Layout.mOpacity = Item->GetRenderOpacity();
96 if (const UCanvasPanelSlot* CanvasSlot = Cast<UCanvasPanelSlot>(Item->Slot))
97 {
98 Layout.mPosition = CanvasSlot->GetPosition();
99 Layout.mSize = CanvasSlot->GetSize();
100 Layout.mZOrder = CanvasSlot->GetZOrder();
101 }
102 }
103}

105// 계산된 배치 값들을 실제 카드 위젯에 기록한다.
106// - 보이는 카드: 렌더 transform(중심 피벗 기준 스케일/회전/투명도) + CanvasSlot(위치/크기/ZOrder) 설정
107// - 안 보이는 카드: Collapsed로 숨김(자리도 차지 안 함)
108void UCarouselPanelWidget::ApplyCarouselLayouts(const TArray<FCarouselItemLayout>& Layouts)
109{
110 for (int32 Index = 0; Index < mCarouselItems.Num(); ++Index)
111 {
112 UWidget* Item = mCarouselItems[Index];
113 if (Item == nullptr)
114 {
115 continue;
116 }
117
118 const FCarouselItemLayout Layout = Layouts.IsValidIndex(Index) ? Layouts[Index] : FCarouselItemLayout();
119 if (Layout.mVisible == false)
120 {
121 Item->SetVisibility(ESlateVisibility::Collapsed);
122 continue;
123 }
124
125 Item->SetRenderTransformPivot(FVector2D(0.5f, 0.5f));
126 Item->SetRenderScale(Layout.mScale);
127 Item->SetRenderTransformAngle(Layout.mAngle);
128 Item->SetRenderOpacity(Layout.mOpacity);
129 Item->SetVisibility(ESlateVisibility::Visible);
130
131 if (UCanvasPanelSlot* CanvasSlot = Cast<UCanvasPanelSlot>(Item->Slot))
132 {
133 CanvasSlot->SetAnchors(FAnchors(0.5f, 0.5f));
134 CanvasSlot->SetAlignment(FVector2D(0.5f, 0.5f));
135 CanvasSlot->SetPosition(Layout.mPosition);
136 CanvasSlot->SetSize(Layout.mSize);
137 CanvasSlot->SetZOrder(Layout.mZOrder);
138 }
139 }
140}
