80bool UArtifactComponentModel::EquipInternal(UStaticArtifactData* Data, UPassiveComponentModel* PassiveComp, UAttributeSetComponentModel* AttrComp)
81{
82 if (Data == nullptr)
83 {
84 return false;
85 }
86
87 FArtifactEntry Entry;
88 Entry.mData = Data;
89
90 // 패시브 설치: 아티펙트가 참조하는 패시브 DA를 로드해 PassiveComponent에 설치
91 if (PassiveComp != nullptr)
92 {
93 for (const TSoftObjectPtr<UStaticPassiveData>& PassivePtr : Data->mStaticPassiveData)
94 {
95 UStaticPassiveData* PassiveData = PassivePtr.LoadSynchronous();
96 if (PassiveData == nullptr)
97 {
98 UE_LOG(LogPassive, Warning, TEXT("아티팩트 패시브 로드 실패: %s -> %s"), *Data->GetName(), *PassivePtr.ToString());
99 continue;
100 }
101
102 UTacticalPassive* Installed = PassiveComp->AddPassiveFromData(PassiveData);
103 if (Installed == nullptr)
104 {
105 UE_LOG(LogPassive, Warning, TEXT("아티팩트 패시브 설치 실패: %s -> %s"), *Data->GetName(), *PassiveData->GetName());
106 continue;
107 }
108 Entry.mInstalledPassives.Add(Installed);
109 }
110 }
111 else if (Data->mStaticPassiveData.Num() > 0)
112 {
113 UE_LOG(LogPassive, Warning, TEXT("패시브 컴포넌트 없음: %s 패시브 설치 생략"), *Data->GetName());
114 }
115
116 // 아티펙트 고유 스탯이 있으면 무한 이펙트로 자신에게 적용 (해제 시 핸들로 제거)
117 if (AttrComp != nullptr && Data->mStatModifiers.Num() > 0)
118 {
119 // 데이터의 수정자들을 런타임 이펙트 인스턴스에 담아 무한 정책으로 구성
120 UTacticalEffect* StatEffect = NewObject<UTacticalEffect>(this);
121 StatEffect->mModifiers = Data->mStatModifiers;
122 StatEffect->mDurationPolicy = ETacticalEffectDurationType::Infinite;
123
124 // 소유자를 시전자로 컨텍스트 구성 후 self 적용, 인스턴스·핸들 보관
125 UTacticalEffectContext* EffectContext = AttrComp->MakeEffectContext();
126 EffectContext->SetAttributeSetComponentModel(AttrComp);
127
128 FTacticalEffectSpec Spec(StatEffect, EffectContext);
129 Entry.mStatEffect = StatEffect;
130 Entry.mStatEffectHandle = AttrComp->ApplyTacticalEffectSpecToSelf(Spec);
131 }
132
133 mArtifacts.Add(MoveTemp(Entry));
134
135 // 장착 확인용 로그 (소유 유닛, 아티팩트, 설치된 패시브 수)
136 UE_LOG(LogRD, Log, TEXT("아티팩트 장착: %s -> %s (패시브 %d개)"), *GetNameSafe(GetOwnerModel()), *Data->GetName(), mArtifacts.Last().mInstalledPassives.Num());
137 return true;
138}

151bool UArtifactComponentModel::UnequipAt(int32 Index, UPassiveComponentModel* PassiveComp, UAttributeSetComponentModel* AttrComp)
152{
153 if (mArtifacts.IsValidIndex(Index) == false)
154 {
155 return false;
156 }
157 FArtifactEntry& Entry = mArtifacts[Index];
158
159 // 설치한 패시브 해제 (적용 중 이펙트가 있으면 내리고 보유 목록에서 제거)
160 if (PassiveComp != nullptr)
161 {
162 for (const TObjectPtr<UTacticalPassive>& Installed : Entry.mInstalledPassives)
163 {
164 if (Installed != nullptr)
165 {
166 Installed->DeactivatePassive();
167 PassiveComp->RemovePassive(Installed);
168 }
169 }
170 }
171
172 // 아티펙트 고유 스탯 이펙트 제거 (적용돼 있던 경우)
173 if (Entry.mStatEffectHandle.IsValid() && AttrComp != nullptr)
174 {
175 AttrComp->RemoveActiveTacticalEffect(Entry.mStatEffectHandle);
176 }
177 Entry.mStatEffectHandle.Reset();
178
179 mArtifacts.RemoveAt(Index);
180 return true;
181}
