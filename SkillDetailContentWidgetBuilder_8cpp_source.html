85 FSlateBrush ImageBrush(UTexture2D* Source,
86 const FLinearColor& Tint = FLinearColor::White)
87 {
88 FSlateBrush Brush;
89 Brush.SetResourceObject(Source);
90 Brush.DrawAs = Source != nullptr
91 ? ESlateBrushDrawType::Image : ESlateBrushDrawType::NoDrawType;
92 Brush.Margin = FMargin(0.f);
93 Brush.TintColor = FSlateColor(Tint);
94 if (Source != nullptr)
95 {
96 const FIntPoint Imported = Source->GetImportedSize();
97 Brush.ImageSize = FVector2D(Imported.X, Imported.Y);
98 }
99 return Brush;

102 FSlateBrush RangeButtonBrush(UTexture2D* Source)
103 {
104 FSlateBrush Brush = ImageBrush(Source);
105 if (Source == nullptr)
106 {
107 return Brush;
108 }
109 // 원본 소스 알파 실측값. Normal 1024x128=(32,0)-(994,118),
110 // Selected=(33,0)-(992,121). 투명 캔버스를 Slate 배치에 남기지 않는다.
111 const bool bSelected = Source->GetName().Contains(TEXT("Selected"));
112 const FVector2f Min = bSelected
113 ? FVector2f(33.f / 1024.f, 0.f)
114 : FVector2f(32.f / 1024.f, 0.f);
115 const FVector2f Max = bSelected
116 ? FVector2f(992.f / 1024.f, 121.f / 128.f)
117 : FVector2f(994.f / 1024.f, 118.f / 128.f);
118 Brush.SetUVRegion(FBox2f(Min, Max));
119 return Brush;
120 }

224 AutoFit->SetClipping(EWidgetClipping::ClipToBoundsAlways);
225 AutoFit->SetVisibility(ESlateVisibility::SelfHitTestInvisible);
226 Button->SetContent(AutoFit);
