41ESRPGCommandResult USRPGMoveBuildAction::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
42{
43 ESRPGCommandResult Result = Super::HandleCommand(Command);
44 if (Result == ESRPGCommandResult::Handled)
45 {
46 return Result;
47 }
48
49 switch (Command.Get().GetCommandType())
50 {
51 case ESRPGCommandType::MoveSelect:
52 {
53 /* 이동 빌드 진입 시 외부에서 넘어온 이동 포인트를 받아 도달 범위 계산 */
54
55 if (mMoveBuildPhase != ESRPGMoveBuildPhase::None)
56 {
57 MarkActionCompleted(ESRPGActionResult::Cancelled);
58 SetBuildPhase(ESRPGMoveBuildPhase::None);
59 Result = ESRPGCommandResult::Handled;
60 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
61 }
62
63 UPlayerUnitModel* PlayerUnitModel = Cast<UPlayerUnitModel>(mInstigator);
64 checkf(PlayerUnitModel != nullptr, TEXT("플레이어 유닛 모델 nullptr"));
65
66 // 속박 등으로 이동 불가면 빌드에 진입하지 않고 취소
67 UUnitMovementComponentModel* MovementCompModel = Cast<UUnitMovementComponentModel>(PlayerUnitModel->GetBoardMovementComponentModel());
68 if (MovementCompModel != nullptr && MovementCompModel->CanSelfMove() == false)
69 {
70 MarkActionCompleted(ESRPGActionResult::Cancelled);
71 SetBuildPhase(ESRPGMoveBuildPhase::None);
72 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
73 }
74
75 UAttributeSetComponentModel* AttributeSetComponentModel = PlayerUnitModel->GetAttributeComponentModel();
76 checkf(AttributeSetComponentModel != nullptr, TEXT("속성 컴포넌트 모델 nullptr"));
77 mMovePoint = AttributeSetComponentModel->GetAttributeCurrentValue(UPlayerUnitAttributeSet::GetActionPointAttribute());
78
79 const FSRPGMoveSelectCommand& MoveSelectCommand = Command.Get<FSRPGMoveSelectCommand>();
80 // 커맨드 델리깃을 복사해서 페이즈가 바뀔 때 통지
81 OnChangeMoveBuildPhase = MoveSelectCommand.OnChangeMoveBuildPhase;
82
83 EnterMoveBuild();
84 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
85 }
86 case ESRPGCommandType::SkillSelect:
87 {
88 /* 이동 대상 선택 중 스킬을 누르면 이동 빌드를 취소한다(기획: 스킬↔이동 전환 시 이전 것 취소).
89 * Ignored를 반환해, 라우터(USRPGActionCreationCommandHandler)가 이 SkillSelect로 스킬 빌드를 새로 생성하게 둔다. */
90
91 MarkActionCompleted(ESRPGActionResult::Cancelled);
92 SetBuildPhase(ESRPGMoveBuildPhase::None);
93 return ESRPGCommandResult::Ignored;
94 }
95 case ESRPGCommandType::TurnEnd:
96 {
97 /* 이동 조준 중 엔드턴을 누르면 이동 빌드를 먼저 취소한다(스킬 빌드와 동일한 꼬임 방지).
98 * Ignored를 반환해, 라우터가 이 TurnEnd로 TurnEndAction을 정상 생성하게 둔다. */
99
100 MarkActionCompleted(ESRPGActionResult::Cancelled);
101 SetBuildPhase(ESRPGMoveBuildPhase::None);
102 return ESRPGCommandResult::Ignored;
103 }
104 case ESRPGCommandType::WorldTrace:
105 {
106 /* 월드 공간 터치 시 선택 위치에 따라서 결정 */
107
108 return CombineSRPGCommandResult(HandleWorldTraceCommand(Command), Result);
109 }
110 }
111
112 return ESRPGCommandResult::Ignored;
113}

115ESRPGCommandResult USRPGMoveBuildAction::HandleWorldTraceCommand(const TInstancedStruct<FSRPGCommand>& Command)
116{
117 ESRPGCommandResult Result = ESRPGCommandResult::Ignored;
118
119 const FSRPGWorldTraceCommand& WorldTraceCommand = Command.Get<FSRPGWorldTraceCommand>();
120 if (WorldTraceCommand.mIsLongPress == true)
121 {
122 return Result;
123 }
124
125 UTileMapModel* TileMap = GetTileMap();
126
127 AActor* TargetActor = nullptr;
128 FTileIndex TargetTileIndex = FTileIndex::Invalid;
129 GetTileActorForCommand(
130 GetWorld(),
131 RDTraceChannels::TileOnlyTrace,
132 WorldTraceCommand,
133 OUT TargetActor,
134 OUT TargetTileIndex
135 );
136
137 IActorView* ActorView = Cast<IActorView>(TargetActor);
138 const bool IsContactedTileMap = ActorView != nullptr && ActorView->GetModel() == TileMap;
139 // 타일맵을 아예 벗어난 탭(IsContactedTileMap=false)도 무효 타일과 똑같이 "타일 밖" 취소로 취급한다
140 // (기획: 이동 대상 선택 중 타일 밖을 누르면 취소). TargetTileIndex가 유효하면 정상 처리로 간다.
141 if (IsContactedTileMap == true || TargetTileIndex == FTileIndex::Invalid)
142 {
143 if (TargetTileIndex == FTileIndex::Invalid)
144 {
145 /* 한단계 취소작업 (무효 타일 또는 타일 밖 탭) */
146
147 switch (mMoveBuildPhase)
148 {
149 case ESRPGMoveBuildPhase::Preview:
150 {
151 /* 프리뷰 단계에서 한단계 취소 시, 마지막 경유지 취소 */
152
153 RemoveLastWaypoint();
154 Result = ESRPGCommandResult::Handled;
155 break;
156 }
157 case ESRPGMoveBuildPhase::DestSelection:
158 {
159 /* 목적지 선택 단계에서 한단계 취소 시, 빌드 자체 종료 */
160
161 MarkActionCompleted(ESRPGActionResult::Cancelled);
162 SetBuildPhase(ESRPGMoveBuildPhase::None);
163 Result = ESRPGCommandResult::Handled;
164 break;
165 }
166 }
167 }
168 else
169 {
170 /* 클릭 타일에 따라 확정 / 경유지 추가 / 취소 분기 */
171
172 switch (mMoveBuildPhase)
173 {
174 case ESRPGMoveBuildPhase::Preview:
175 {
176 /* 마지막 경유지(도착 후보) 재클릭 시, 이동 발행 */
177
178 if (GetLastWaypoint() == TargetTileIndex)
179 {
180 BuildMove();
181 MarkActionCompleted(ESRPGActionResult::Succeeded);
182 SetBuildPhase(ESRPGMoveBuildPhase::Build);
183 Result = ESRPGCommandResult::Handled;
184 break;
185 }
186
187 // 중간 경유지 재클릭도 잘라내지 않고 아래에서 새 경유지로 추가 (왔다갔다 이동 허용)
188 [[fallthrough]];
189 }
190 case ESRPGMoveBuildPhase::DestSelection:
191 {
192 /* 도달 가능한 칸 클릭 시, 경유지 추가 (마지막 경유지 위치 재클릭은 위에서 처리, 출발 타일 복귀 허용) */
193
194 if (TargetTileIndex != GetLastWaypoint() && mReachableTileIndexes.Contains(TargetTileIndex) == true)
195 {
196 AddWaypoint(TargetTileIndex);
197 Result = ESRPGCommandResult::Handled;
198 break;
199 }
200
201 /* 범위 밖 유효 타일 클릭 시, 마지막 경유지 취소 (경유지가 있는 프리뷰 단계만) */
202
203 if (mMoveBuildPhase == ESRPGMoveBuildPhase::Preview)
204 {
205 RemoveLastWaypoint();
206 Result = ESRPGCommandResult::Handled;
207 }
208 break;
209 }
210 }
211 }
212 }
213
214 return Result;
215}

254 /* 부분경로 계산 및 프리뷰 표시 — 빌드 시점에 확정한 경로를 실행까지 그대로 사용(심=라이브 보장) */

264
265 {
266 // 스킬의 CalculateSkillResult(가하는 데미지)의 역방향.
267 // 받는 효과의 실제 적용은 실행 시 도착·경유 타일의 OnBeginTileOverlap이 담당.
268 // 심 복제본에서 이 경로 이동을 실행해 출발/도착 스냅샷(MakeSnapshotData) 비교 또는
269 // EventLog(mAttributeEffectEventLogs)로 스탯 델타를 산출해 프리뷰로 표시.
270 // TODO : 이동 예측 계산 연결
271 }
272
273 /* 상태 변경되면서 외부에서 바인딩된 UI 변경 */

292void USRPGMoveBuildAction::BuildMove()
293{
294 checkf(mMoveBuildPhase == ESRPGMoveBuildPhase::Preview, TEXT("이동 빌드 순서 오류"));
295
296 USRPGCommandRouterModel* CommandRouterModel = GetWorldSubsystemModel<USRPGCommandRouterModel>(this);
297 checkf(CommandRouterModel != nullptr, TEXT("명령 라우터 서브시스템 모델 nullptr"));
298
299 /* 부분경로들을 하나의 경로로 이어붙여 이동 액션 생성 명령 발행 */
300
301 TArray<FTileIndex> PathTileIndexes;
302 for (int32 SegmentIndex = 0; SegmentIndex < mPathSegments.Num(); ++SegmentIndex)
303 {
304 const TArray<FTileIndex>& Segment = mPathSegments[SegmentIndex];
305
306 // 두 번째 부분경로부터는 첫 타일이 앞 부분경로의 끝 타일과 같으므로 제외
307 for (int32 Index = (SegmentIndex == 0 ? 0 : 1); Index < Segment.Num(); ++Index)
308 {
309 PathTileIndexes.Add(Segment[Index]);
310 }
311 }
312
313 TInstancedStruct<FSRPGCommand> MoveCastCommand;
314 MoveCastCommand.InitializeAs<FSRPGMoveCommand>();
315 MoveCastCommand.GetMutable<FSRPGMoveCommand>().mPathTileIndexes = PathTileIndexes;
316
317 CommandRouterModel->SummitCommand(MoveCastCommand);
318}

375void USRPGMoveBuildAction::RefreshPathPreview()
376{
377 UTileMapModel* TileMap = GetTileMap();
378
379 UE_LOG(LogTemp, Warning, TEXT("[화살표] RefreshPathPreview 조각=%d"), mPathSegments.Num());
380
381 // 부분경로가 없으면 프리뷰 해제
382 if (mPathSegments.IsEmpty())
383 {
384 TileMap->ClearMovePath();
385 return;
386 }
387
388 /* 부분경로들을 하나의 표시용 경로로 이어붙임 */
389
390 TArray<FMovePathTile> PathTiles;
391 for (int32 SegmentIndex = 0; SegmentIndex < mPathSegments.Num(); ++SegmentIndex)
392 {
393 const TArray<FTileIndex>& Segment = mPathSegments[SegmentIndex];
394
395 // 두 번째 부분경로부터는 첫 타일이 앞 부분경로의 끝 타일과 같으므로 제외
396 for (int32 Index = (SegmentIndex == 0 ? 0 : 1); Index < Segment.Num(); ++Index)
397 {
398 // 부분경로의 끝 타일 = 경유지 (단, 마지막 부분경로의 끝 = 도착 후보라 제외)
399 const bool IsWaypoint = (Index == Segment.Num() - 1) && (SegmentIndex < mPathSegments.Num() - 1);
400 PathTiles.Emplace(Segment[Index], IsWaypoint);
401 }
402 }
403
404 TileMap->SetMovePath(PathTiles);
405}
