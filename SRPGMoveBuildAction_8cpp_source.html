40ESRPGCommandResult USRPGMoveBuildAction::HandleCommand(const TInstancedStruct<FSRPGCommand>& Command)
41{
42 ESRPGCommandResult Result = Super::HandleCommand(Command);
43 if (Result == ESRPGCommandResult::Handled)
44 {
45 return Result;
46 }
47
48 switch (Command.Get().GetCommandType())
49 {
50 case ESRPGCommandType::MoveSelect:
51 {
52 /* 이동 빌드 진입 시 외부에서 넘어온 이동 포인트를 받아 도달 범위 계산 */
53
54 if (mMoveBuildPhase != ESRPGMoveBuildPhase::None)
55 {
56 MarkActionCompleted(ESRPGActionResult::Cancelled);
57 SetBuildPhase(ESRPGMoveBuildPhase::None);
58 Result = ESRPGCommandResult::Handled;
59 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
60 }
61
62 UPlayerUnitModel* PlayerUnitModel = Cast<UPlayerUnitModel>(mInstigator);
63 checkf(PlayerUnitModel != nullptr, TEXT("플레이어 유닛 모델 nullptr"));
64 UAttributeSetComponentModel* AttributeSetComponentModel = PlayerUnitModel->GetAttributeComponentModel();
65 checkf(AttributeSetComponentModel != nullptr, TEXT("속성 컴포넌트 모델 nullptr"));
66 mMovePoint = AttributeSetComponentModel->GetAttributeCurrentValue(UPlayerUnitAttributeSet::GetMovementAttribute());
67
68 const FSRPGMoveSelectCommand& MoveSelectCommand = Command.Get<FSRPGMoveSelectCommand>();
69 // 커맨드 델리깃을 복사해서 페이즈가 바뀔 때 통지
70 OnChangeMoveBuildPhase = MoveSelectCommand.OnChangeMoveBuildPhase;
71
72 EnterMoveBuild();
73 return CombineSRPGCommandResult(ESRPGCommandResult::Handled, Result);
74 }
75 case ESRPGCommandType::SkillSelect:
76 {
77 /* 이동 대상 선택 중 스킬을 누르면 이동 빌드를 취소한다(기획: 스킬↔이동 전환 시 이전 것 취소).
78 * Ignored를 반환해, 라우터(USRPGActionCreationCommandHandler)가 이 SkillSelect로 스킬 빌드를 새로 생성하게 둔다. */
79
80 MarkActionCompleted(ESRPGActionResult::Cancelled);
81 SetBuildPhase(ESRPGMoveBuildPhase::None);
82 return ESRPGCommandResult::Ignored;
83 }
84 case ESRPGCommandType::TurnEnd:
85 {
86 /* 이동 조준 중 엔드턴을 누르면 이동 빌드를 먼저 취소한다(스킬 빌드와 동일한 꼬임 방지).
87 * Ignored를 반환해, 라우터가 이 TurnEnd로 TurnEndAction을 정상 생성하게 둔다. */
88
89 MarkActionCompleted(ESRPGActionResult::Cancelled);
90 SetBuildPhase(ESRPGMoveBuildPhase::None);
91 return ESRPGCommandResult::Ignored;
92 }
93 case ESRPGCommandType::WorldTrace:
94 {
95 /* 월드 공간 터치 시 선택 위치에 따라서 결정 */
96
97 return CombineSRPGCommandResult(HandleWorldTraceCommand(Command), Result);
98 }
99 }
100
101 return ESRPGCommandResult::Ignored;
102}

104ESRPGCommandResult USRPGMoveBuildAction::HandleWorldTraceCommand(const TInstancedStruct<FSRPGCommand>& Command)
105{
106 ESRPGCommandResult Result = ESRPGCommandResult::Ignored;
107
108 const FSRPGWorldTraceCommand& WorldTraceCommand = Command.Get<FSRPGWorldTraceCommand>();
109 if (WorldTraceCommand.mIsLongPress == true)
110 {
111 return Result;
112 }
113
114 UTileMapModel* TileMap = GetTileMap();
115
116 AActor* TargetActor = nullptr;
117 FTileIndex TargetTileIndex = FTileIndex::Invalid;
118 GetTileActorUnderCursor(GetWorld(), RDTraceChannels::TileOnlyTrace, WorldTraceCommand.mScreenPosition, OUT TargetActor, OUT TargetTileIndex);
119
120 IActorView* ActorView = Cast<IActorView>(TargetActor);
121 const bool IsContactedTileMap = ActorView != nullptr && ActorView->GetModel() == TileMap;
122 // 타일맵을 아예 벗어난 탭(IsContactedTileMap=false)도 무효 타일과 똑같이 "타일 밖" 취소로 취급한다
123 // (기획: 이동 대상 선택 중 타일 밖을 누르면 취소). TargetTileIndex가 유효하면 정상 처리로 간다.
124 if (IsContactedTileMap == true || TargetTileIndex == FTileIndex::Invalid)
125 {
126 if (TargetTileIndex == FTileIndex::Invalid)
127 {
128 /* 한단계 취소작업 (무효 타일 또는 타일 밖 탭) */
129
130 switch (mMoveBuildPhase)
131 {
132 case ESRPGMoveBuildPhase::Preview:
133 {
134 /* 프리뷰 단계에서 한단계 취소 시, 목적지 취소 처리 */
135
136 ResetTargetTile();
137 Result = ESRPGCommandResult::Handled;
138 break;
139 }
140 case ESRPGMoveBuildPhase::DestSelection:
141 {
142 /* 목적지 선택 단계에서 한단계 취소 시, 빌드 자체 종료 */
143
144 MarkActionCompleted(ESRPGActionResult::Cancelled);
145 SetBuildPhase(ESRPGMoveBuildPhase::None);
146 Result = ESRPGCommandResult::Handled;
147 break;
148 }
149 }
150 }
151 else
152 {
153 /* 한단계 처리작업 */
154
155 switch (mMoveBuildPhase)
156 {
157 case ESRPGMoveBuildPhase::Preview:
158 {
159 /* 확정 칸 클릭 시, 이동 발행 */
160
161 if (mTargetIndex == TargetTileIndex)
162 {
163 BuildMove();
164 MarkActionCompleted(ESRPGActionResult::Succeeded);
165 SetBuildPhase(ESRPGMoveBuildPhase::Build);
166 Result = ESRPGCommandResult::Handled;
167 break;
168 }
169 [[fallthrough]];
170 }
171 case ESRPGMoveBuildPhase::DestSelection:
172 {
173 /* 도달 가능한 칸 클릭 시, 프리뷰 단계까지 보여주기 */
174
175 if (mReachableTileIndexes.Contains(TargetTileIndex) == true)
176 {
177 ResetTargetTile();
178 SetTargetTile(TargetTileIndex);
179 Result = ESRPGCommandResult::Handled;
180 break;
181 }
182 break;
183 }
184 }
185 }
186 }
187
188 return Result;
189}

222void USRPGMoveBuildAction::SetTargetTile(const FTileIndex& TileIndex)
223{
224 checkf(mMoveBuildPhase == ESRPGMoveBuildPhase::DestSelection, TEXT("이동 빌드 순서 오류"));
225
226 UTileMapModel* TileMap = GetTileMap();
227
228 /* 경로 계산 및 프리뷰 표시 — 빌드 시점에 확정한 경로를 실행까지 그대로 사용(심=라이브 보장) */
229
230 {
231 const FTileIndex Origin = mInstigator->GetTileTransform().mIndex;
232 mTargetIndex = TileIndex;
233 mPathTileIndexes = TileMap->FindPath(Origin, TileIndex);
234 TileMap->SetMovePath(Origin, TileIndex);
235 }
236
237 /* 예측 시스템 — 이동 시 '받는' 영향(경로/도착 타일의 함정·장판 데미지·상태이상) 예측 */
238
239 {
240 // 스킬의 CalculateSkillResult(가하는 데미지)의 역방향.
241 // 받는 효과의 실제 적용은 실행 시 도착·경유 타일의 OnBeginTileOverlap이 담당.
242 // 심 복제본에서 이 경로 이동을 실행해 출발/도착 스냅샷(MakeSnapshotData) 비교 또는
243 // EventLog(mAttributeEffectEventLogs)로 스탯 델타를 산출해 프리뷰로 표시.
244 // TODO : 이동 예측 계산 연결
245 }
246
247 /* 상태 변경되면서 외부에서 바인딩된 UI 변경 */
248
249 SetBuildPhase(ESRPGMoveBuildPhase::Preview);
250}
