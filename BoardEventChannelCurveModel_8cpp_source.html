56{
57 FBoardEventTriggerChannel* Channel = mChannelHandle.Get();
58 UMovieSceneSection* Section = mWeakSection.Get();
59
60 if (Channel && Section)
61 {
62 FFrameRate TickResolution = Section->GetTypedOuter<UMovieScene>()->GetTickResolution();
63
64 TMovieSceneChannelData<FBoardEventTriggerData> ChannelData = Channel->GetData();
65 TArrayView<const FFrameNumber> Times = ChannelData.GetTimes();
66
67 const FFrameNumber StartFrame = MinTime <= MIN_int32 ? MIN_int32 : (MinTime * TickResolution).CeilToFrame();
68 const FFrameNumber EndFrame = MaxTime >= MAX_int32 ? MAX_int32 : (MaxTime * TickResolution).FloorToFrame();
69
70 const int32 StartingIndex = Algo::LowerBound(Times, StartFrame);
71 const int32 EndingIndex = Algo::UpperBound(Times, EndFrame);
72
73 for (int32 KeyIndex = StartingIndex; KeyIndex < EndingIndex; ++KeyIndex)
74 {
75 OutKeyHandles.Add(ChannelData.GetHandle(KeyIndex));
76 }
77 }

156void FBoardEventChannelCurveModel::GetTimeRange(double& MinTime, double& MaxTime) const

159 UMovieSceneSection* Section = mWeakSection.Get();
